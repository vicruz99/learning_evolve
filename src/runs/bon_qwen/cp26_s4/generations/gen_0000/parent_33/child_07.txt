import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    def objective(vars_):
        return -np.sum(vars_[2::3])

    def boundary_constraints(vars_):
        cons = []
        for i in range(n):
            x, y, r = vars_[3*i], vars_[3*i+1], vars_[3*i+2]
            # x - r >= 0
            cons.append(x - r)
            # x + r <= 1  => 1 - x - r >= 0
            cons.append(1 - x - r)
            # y - r >= 0
            cons.append(y - r)
            # y + r <= 1  => 1 - y - r >= 0
            cons.append(1 - y - r)
        return np.array(cons)

    def overlap_constraints(vars_):
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1, r1 = vars_[3*i], vars_[3*i+1], vars_[3*i+2]
                x2, y2, r2 = vars_[3*j], vars_[3*j+1], vars_[3*j+2]
                dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                min_dist = r1 + r2
                # dist >= min_dist  => dist^2 - min_dist^2 >= 0
                cons.append(dist_sq - min_dist**2)
        return np.array(cons)

    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None

    np.random.seed(42)
    # Run multiple random starts to escape local minima
    for _ in range(20):
        # Initialize on a dense grid slightly larger than the square
        # This helps the solver find a high-density packing
        cols = 6
        rows = 5
        x_coords = np.linspace(0, 1, cols + 1)
        y_coords = np.linspace(0, 1, rows + 1)
        xs, ys = np.meshgrid(x_coords, y_coords)
        xs = xs.flatten()
        ys = ys.flatten()
        
        # Select first n points
        xs_init = xs[:n]
        ys_init = ys[:n]
        
        # Initial radii: small enough to be valid
        rs_init = np.full(n, 0.05)
        
        # Add some randomness
        xs_init += np.random.uniform(-0.05, 0.05, n)
        ys_init += np.random.uniform(-0.05, 0.05, n)
        
        # Clip to valid range [r, 1-r] approximately
        rs_init = np.clip(rs_init, 0.01, 0.1)
        xs_init = np.clip(xs_init, rs_init, 1 - rs_init)
        ys_init = np.clip(ys_init, rs_init, 1 - rs_init)

        initial_vars = np.zeros(3 * n)
        initial_vars[0::3] = xs_init
        initial_vars[1::3] = ys_init
        initial_vars[2::3] = rs_init

        # Bounds for variables: x, y in [0, 1], r in [0, 1]
        bnds = [(0, 1)] * (2 * n) + [(0, 1)] * n

        # Optimization
        res = minimize(
            objective,
            initial_vars,
            method='SLSQP',
            bounds=bnds,
            constraints=[
                {'type': 'ineq', 'fun': boundary_constraints},
                {'type': 'ineq', 'fun': overlap_constraints}
            ],
            options={'maxiter': 1000, 'ftol': 1e-12}
        )

        if res.success:
            sum_r = -res.fun
            if sum_r > best_sum_radii:
                best_sum_radii = sum_r
                best_centers = res.x[0::3].reshape(n, 2)
                best_radii = res.x[2::3]

    # Validation and safety check
    if best_centers is None or best_radii is None:
        # Fallback to a simple valid configuration if optimization fails
        best_centers = np.tile(np.array([0.5, 0.5]), (n, 1))
        best_radii = np.zeros(n)
        best_sum_radii = 0.0

    return best_centers, best_radii, best_sum_radii

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    print(f"Sum of radii: {sum_r}")
