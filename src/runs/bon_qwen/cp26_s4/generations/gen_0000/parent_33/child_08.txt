import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26

    def create_equal_constraints(centers, r_var, n):
        constraints = []

        # These are linear in terms of centers and r_var if we view r_var as scalar
        # But scipy constraints need to return values >= 0
        
        # Since r_var is a single variable in this phase, we can define constraints directly
        # However, SLSQP handles inequality constraints g(x) >= 0.
        
        # We can add linear constraints for boundaries
        # But mixing linear and non-linear in a list is fine.
        
        # Boundary constraints
        # x_i - r >= 0
        # 1 - x_i - r >= 0
        # y_i - r >= 0
        # 1 - y_i - r >= 0
        
        # Non-overlap constraints: dist(i,j)^2 - (2r)^2 >= 0
        
        # It's better to define a function that computes all constraint violations
        def constraint_func(vars_flat):
            # vars_flat = [x0, y0, ..., x25, y25, r]
            x = vars_flat[0:n]
            y = vars_flat[n:2*n]
            r = vars_flat[2*n]
            
            vals = []
            
            # Boundary
            vals.extend(x - r)
            vals.extend(1 - x - r)
            vals.extend(y - r)
            vals.extend(1 - y - r)
            
            # Overlaps
            # Vectorized computation
            # dist_sq = (x_i - x_j)^2 + (y_i - y_j)^2
            # We need dist_sq >= 4*r^2
            
            # Compute pairwise distances squared
            # Using broadcasting
            x_col = x.reshape(-1, 1)
            y_col = y.reshape(-1, 1)
            
            dx = x_col - x_col.T
            dy = y_col - y_col.T
            dist_sq = dx**2 + dy**2
            
            # We only care about upper triangle (i < j)
            # Mask for upper triangle
            mask = np.triu(np.ones((n, n), dtype=bool), k=1)
            upper_dist_sq = dist_sq[mask]
            
            # Constraint: dist_sq - 4*r^2 >= 0
            vals.extend(upper_dist_sq - 4*r**2)
            
            return np.array(vals)

        return constraint_func

            y = vars_flat[1::3]
            r = vars_flat[2::3]

            vals.extend(1 - x - r)

            upper_dist_sq = dist_sq[mask]
            
            # Constraint: dist_sq >= (r_i + r_j)^2

            upper_r_sum = r_sum[mask]
            
            vals.extend(upper_dist_sq - upper_r_sum**2)
            
            return np.array(vals)
        return constraint_func
