import numpy as np
from scipy.optimize import minimize

def compute_loss(centers, radii, n):
    """Computes negative sum of radii plus penalties for constraint violations."""
    loss = -np.sum(radii)
    pen = 0.0
    
    # Boundary penalties
    for i in range(n):
        if centers[i, 0] < radii[i]:
            pen += (radii[i] - centers[i, 0])**2
        if centers[i, 0] > 1 - radii[i]:
            pen += (centers[i, 0] + radii[i] - 1)**2
        if centers[i, 1] < radii[i]:
            pen += (radii[i] - centers[i, 1])**2
        if centers[i, 1] > 1 - radii[i]:
            pen += (centers[i, 1] + radii[i] - 1)**2
            
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            md = radii[i] + radii[j]
            if d < md:
                pen += (md - d)**2
                
    return loss + 2000.0 * pen

def objective(vars, n):
    """Wrapper for scipy optimizer. Transforms log-radii back to linear."""
    centers = vars[:2*n].reshape(n, 2)
    radii = np.exp(vars[2*n:])
    return compute_loss(centers, radii, n)

def run_packing():
    n = 26
    best_score = -np.inf
    best_res = None
    
    for seed in range(20):
        np.random.seed(seed)
        
        # Hexagonal-like staggered grid initialization
        centers = np.zeros((n, 2))
        idx = 0
        counts = [5, 5, 6, 5, 5] # Sums to 26
        y = 0.08
        for cnt in counts:
            x_start = 0.1
            for k in range(cnt):
                if idx < n:
                    centers[idx] = [x_start + k * 0.18, y]
                    idx += 1
            y += 0.15
            x_start += 0.09 # Offset for hexagonal staggering
            
        # Break symmetry and ensure initial points are strictly inside bounds
        centers += np.random.randn(n, 2) * 0.002
        centers = np.clip(centers, 0.01, 0.99)
        
        radii = np.full(n, 0.09)
        x0 = np.hstack([centers.flatten(), np.log(radii)])
        
        # Bounds: x,y in [0,1], log(r) in [-4, 1] -> r in [0.018, 2.718]
        bounds = [(0.0, 1.0)] * (2*n) + [(-4.0, 1.0)] * n
        
        try:
            res = minimize(objective, x0, args=(n,), bounds=bounds, method='L-BFGS-B',
                           options={'maxiter': 4000, 'ftol': 1e-12, 'gtol': 1e-10})
            c = res.x[:2*n].reshape(n, 2)
            r = np.exp(res.x[2*n:])
            
            # Strict validation against problem constraints
            valid = True
            for i in range(n):
                if c[i, 0] < r[i] - 1e-9 or c[i, 0] > 1 - r[i] + 1e-9: valid = False; break
                if c[i, 1] < r[i] - 1e-9 or c[i, 1] > 1 - r[i] + 1e-9: valid = False; break
            for i in range(n):
                for j in range(i + 1, n):
                    if np.linalg.norm(c[i] - c[j]) < r[i] + r[j] - 1e-9: valid = False; break
                if not valid: break
                
            if valid and np.sum(r) > best_score:
                best_score = np.sum(r)
                best_res = (c.copy(), r.copy())
        except Exception:
            pass
            
    # Fallback in case optimization fails entirely
    if best_res is None:
        c = np.random.rand(n, 2) * 0.6 + 0.2
        r = np.full(n, 0.01)
        return c, r, np.sum(r)
        
    return best_res[0], best_res[1], best_score
