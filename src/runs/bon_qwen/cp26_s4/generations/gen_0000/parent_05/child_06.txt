import math
import random

N = 26
CURRENT_PENALTY = 1000.0

def penalty_objective(params):
    global CURRENT_PENALTY
    n = N
    obj = -np.sum(params[2::3])
    pw = CURRENT_PENALTY
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = params[3 * i] - params[3 * j]
            dy = params[3 * i + 1] - params[3 * j + 1]
            dist = math.sqrt(dx * dx + dy * dy)
            min_dist = params[3 * i + 2] + params[3 * j + 2]
            violation = min_dist - dist
            if violation > 0:
                obj += pw * violation * violation
    
    # Boundary penalties
    for i in range(n):
        x = params[3 * i]
        y = params[3 * i + 1]
        r = params[3 * i + 2]
        
        v = r - x
        if v > 0:
            obj += pw * v * v
        v = r - (1.0 - x)
        if v > 0:
            obj += pw * v * v
        v = r - y
        if v > 0:
            obj += pw * v * v
        v = r - (1.0 - y)
        if v > 0:
            obj += pw * v * v
        v = -r
        if v > 0:
            obj += pw * v * v
    
    return obj

def gradient_penalty_objective(params):

    n = N
    grad = np.zeros(3 * n)
    pw = CURRENT_PENALTY

                dist = 1e-15
            min_dist = params[3 * i + 2] + params[3 * j + 2]
            violation = min_dist - dist
            if violation > 0:

                # d/dx_i: -2*pw*violation * (-dx/dist) = 2*pw*violation*dx/dist
                # d/dy_i: 2*pw*violation*dy/dist
                # d/dr_i: 2*pw*violation

                grad[3 * i] += factor * dx
                grad[3 * i + 1] += factor * dy
                grad[3 * i + 2] += 2.0 * pw * violation
                
                grad[3 * j] -= factor * dx
                grad[3 * j + 1] -= factor * dy
                grad[3 * j + 2] += 2.0 * pw * violation

        x = params[3 * i]
        y = params[3 * i + 1]
        r = params[3 * i + 2]
        
        # x >= r: violation = r - x
        v = r - x
        if v > 0:
            grad[3 * i] -= 2.0 * pw * v
            grad[3 * i + 2] += 2.0 * pw * v
        
        # x <= 1 - r: violation = r - (1 - x) = r + x - 1
        v = r + x - 1.0
        if v > 0:
            grad[3 * i] += 2.0 * pw * v
            grad[3 * i + 2] += 2.0 * pw * v
        
        # y >= r: violation = r - y
        v = r - y
        if v > 0:
            grad[3 * i + 1] -= 2.0 * pw * v
            grad[3 * i + 2] += 2.0 * pw * v
        
        # y <= 1 - r: violation = r - (1 - y) = r + y - 1
        v = r + y - 1.0
        if v > 0:
            grad[3 * i + 1] += 2.0 * pw * v
            grad[3 * i + 2] += 2.0 * pw * v
    
    return grad

def init_hexagonal_config(n, row_counts, r_init=0.08):
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    
    idx = 0
    for row in range(len(row_counts)):
        count = row_counts[row]
        y = r_init + row * r_init * math.sqrt(3)
        x_offset = r_init if row % 2 == 1 else 0.0
        for col in range(count):
            if idx < n:
                x = r_init + col * 2.0 * r_init + x_offset
                centers[idx] = [x, y]
                idx += 1
    
    return centers, radii

def init_square_config(n, cols, r_init=0.09):
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    
    rows = math.ceil(n / cols)
    idx = 0
    x_spacing = (1.0 - 2.0 * r_init) / (cols - 1) if cols > 1 else 0.0
    y_spacing = (1.0 - 2.0 * r_init) / (rows - 1) if rows > 1 else 0.0
    
    for row in range(rows):
        for col in range(cols):
            if idx < n:
                x = r_init + col * x_spacing
                y = r_init + row * y_spacing
                centers[idx] = [x, y]
                idx += 1
    
    return centers, radii

def init_random_config(n, r_init=0.05):
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    
    for i in range(n):
        centers[i, 0] = r_init + (1.0 - 2.0 * r_init) * random.random()
        centers[i, 1] = r_init + (1.0 - 2.0 * r_init) * random.random()
    
    return centers, radii

def flatten_params(centers, radii, n):
    params = np.zeros(3 * n)
    for i in range(n):
        params[3 * i] = centers[i, 0]
        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = radii[i]
    return params

def unflatten_params(params, n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = params[3 * i]
        centers[i, 1] = params[3 * i + 1]
        radii[i] = params[3 * i + 2]
    return centers, radii

def get_bounds(n):

    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((1e-8, 0.5))
    return bounds
