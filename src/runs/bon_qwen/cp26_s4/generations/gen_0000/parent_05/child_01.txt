import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def compute_constraints(x):
    """Compute all constraints as an array (must be >= 0).
    
    x: flattened vector [x1, y1, x2, y2, ..., r1, r2, ...]
    Returns array where each element must be non-negative.
    """
    n = N_CIRCLES
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    constraints_list = []
    
    # Boundary constraints: 4 per circle
    for i in range(n):
        # x_i >= r_i (left boundary)
        constraints_list.append(centers[i, 0] - radii[i])
        # 1 - x_i >= r_i (right boundary)
        constraints_list.append(1.0 - centers[i, 0] - radii[i])
        # y_i >= r_i (bottom boundary)
        constraints_list.append(centers[i, 1] - radii[i])
        # 1 - y_i >= r_i (top boundary)
        constraints_list.append(1.0 - centers[i, 1] - radii[i])
    
    # Distance constraints: pairwise
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            # dist >= r_i + r_j
            constraints_list.append(dist - radii[i] - radii[j])
    
    return np.array(constraints_list)


def compute_jacobian(x):
    """Compute analytical Jacobian of constraints.
    
    Returns matrix of shape (n_constraints, n_variables).
    """
    n = N_CIRCLES
    nc = 3 * n  # total number of variables
    n_constraints = 4 * n + n * (n - 1) // 2
    
    jac = np.zeros((n_constraints, nc))
    
    centers = x[:2*n].reshape(n, 2)
    
    row = 0
    
    # Boundary constraint Jacobians
    for i in range(n):
        # x_i - r_i >= 0
        jac[row, 2*i] = 1.0
        jac[row, 2*n + i] = -1.0
        row += 1
        
        # 1 - x_i - r_i >= 0
        jac[row, 2*i] = -1.0
        jac[row, 2*n + i] = -1.0
        row += 1
        
        # y_i - r_i >= 0
        jac[row, 2*i + 1] = 1.0
        jac[row, 2*n + i] = -1.0
        row += 1
        
        # 1 - y_i - r_i >= 0
        jac[row, 2*i + 1] = -1.0
        jac[row, 2*n + i] = -1.0
        row += 1
    
    # Distance constraint Jacobians
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist > 1e-10:
                # d(dist - r_i - r_j)/d(x_i) = -dx/dist
                jac[row, 2*i] = -dx / dist
                jac[row, 2*i + 1] = -dy / dist
                jac[row, 2*j] = dx / dist
                jac[row, 2*j + 1] = dy / dist
            # d(dist - r_i - r_j)/d(r_i) = -1
            jac[row, 2*n + i] = -1.0
            jac[row, 2*n + j] = -1.0
            row += 1
    
    return jac


def objective_function(x):
    """Negative sum of radii (we minimize negative to maximize sum)."""
    n = N_CIRCLES
    return -np.sum(x[2*n:])


def generate_hexagonal_packing(n):
    """Generate initial configuration using hexagonal lattice."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.035
    
    idx = 0
    spacing = 0.32
    
    row = 0
    while idx < n:
        col = 0
        while idx < n:
            x = 0.12 + spacing * col + (spacing * 0.5 if row % 2 == 1 else 0)
            y = 0.12 + spacing * np.sqrt(3) / 2.0 * row
            if x <= 0.88 and y <= 0.88:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1
    
    return centers, radii


def generate_corner_emphasis_packing(n):
    """Generate initial config with larger circles in corners."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04
    
    # Place 4 circles in corners with larger initial radius
    corner_positions = [[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]]
    for i, (x, y) in enumerate(corner_positions):
        centers[i] = [x, y]
        radii[i] = 0.09
    
    # Fill remaining with hexagonal pattern in center
    idx = 4
    spacing = 0.28
    row = 0
    while idx < n:
        col = 0
        while idx < n:
            x = 0.35 + spacing * col + (spacing * 0.5 if row % 2 == 1 else 0)
            y = 0.35 + spacing * np.sqrt(3) / 2.0 * row
            if x <= 0.65 and y <= 0.65:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1
    
    # Fill any remaining
    idx2 = idx
    if idx2 < n:
        centers[idx2] = [0.5, 0.5]
    
    return centers, radii


def generate_dense_grid_packing(n):
    """Generate initial config using a dense grid."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04
    
    idx = 0
    # Try different grid spacings
    for spacing in [0.18, 0.19, 0.20]:
        for row in range(6):
            for col in range(6):
                if idx < n:
                    x = 0.08 + spacing * col
                    y = 0.08 + spacing * row
                    if x <= 0.92 and y <= 0.92:
                        centers[idx] = [x, y]
                        idx += 1
    
    return centers, radii


def run_packing():
    """Run circle packing optimization to maximize sum of radii."""
    n = N_CIRCLES
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    np.random.seed(123)
    
    # Define optimization bounds and constraints once
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    cons = {
        'type': 'ineq',
        'fun': compute_constraints,
        'jac': compute_jacobian
    }
    
    # Try multiple initial configurations
    initial_configs = [
        generate_hexagonal_packing(n),
        generate_corner_emphasis_packing(n),
        generate_dense_grid_packing(n),
    ]
    
    for config_idx, (centers0, radii0) in enumerate(initial_configs):
        for trial in range(2):
            # Create a working copy with perturbation
            centers_trial = centers0.copy()
            radii_trial = radii0.copy()
            
            if trial > 0:
                centers_trial += np.random.randn(n, 2) * 0.015
                radii_trial += np.random.randn(n) * 0.003
                radii_trial = np.maximum(radii_trial, 0.01)
            
            # Clip centers to valid range
            centers_trial[:, 0] = np.clip(centers_trial[:, 0], 0.01, 0.99)
            centers_trial[:, 1] = np.clip(centers_trial[:, 1], 0.01, 0.99)
            
            x0 = np.concatenate([centers_trial.flatten(), radii_trial])
            
            result = minimize(
                objective_function,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={
                    'maxiter': 20000,
                    'ftol': 1e-14,
                    'disp': False
                }
            )
            
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = result.x[:2*n].reshape(n, 2).copy()
                best_radii = result.x[2*n:].copy()
    
    # Final refinement: run one more optimization from the best solution found
    x0_final = np.concatenate([best_centers.flatten(), best_radii])
    result_final = minimize(
        objective_function,
        x0_final,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={
            'maxiter': 30000,
            'ftol': 1e-15,
            'disp': False
        }
    )
    
    best_centers = result_final.x[:2*n].reshape(n, 2)
    best_radii = result_final.x[2*n:]
    
    # Post-process: ensure strict validity (fix numerical edge cases)
    for i in range(n):
        best_radii[i] = max(best_radii[i], 1e-10)
        best_centers[i, 0] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 0]))
        best_centers[i, 1] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 1]))
    
    # Final sum
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, final_sum
