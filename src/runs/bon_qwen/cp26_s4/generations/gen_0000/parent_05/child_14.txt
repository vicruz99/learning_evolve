import numpy as np

def get_initial_layout(n):
    """Initialize centers in a grid and set small radii."""
    centers = np.zeros((n, 2))
    cols = 6
    rows = 5
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            centers[idx, 0] = (c + 0.5) / cols
            centers[idx, 1] = (r + 0.5) / rows
            idx += 1
        if idx >= n:
            break
    radii = np.ones(n) * 0.02
    return centers, radii

def optimize_packing(centers, radii, n):
    """Run force-directed relaxation with expanding radii."""
    n_iter = 5000
    dt = 0.02
    k_repulse = 20.0
    k_boundary = 50.0
    growth_rate = 0.0003
    
    idx_i, idx_j = np.triu_indices(n, k=1)
    
    for _ in range(n_iter):
        radii += growth_rate
        
        diff = centers[idx_i] - centers[idx_j]
        dist = np.linalg.norm(diff, axis=1)
        safe_dist = np.maximum(dist, 1e-9)
        dir_vec = diff / safe_dist[:, None]
        
        min_dist_req = radii[idx_i] + radii[idx_j]
        overlap = min_dist_req - dist
        f_mag = np.maximum(overlap, 0.0) * k_repulse
        
        forces = np.zeros_like(centers)
        np.add.at(forces, idx_i, dir_vec * f_mag[:, None])
        np.add.at(forces, idx_j, -dir_vec * f_mag[:, None])
        
        forces[:, 0] += np.maximum(radii - centers[:, 0], 0.0) * k_boundary
        forces[:, 0] -= np.maximum(centers[:, 0] + radii - 1.0, 0.0) * k_boundary
        forces[:, 1] += np.maximum(radii - centers[:, 1], 0.0) * k_boundary
        forces[:, 1] -= np.maximum(centers[:, 1] + radii - 1.0, 0.0) * k_boundary
        
        centers += forces * dt
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    min_gap = 1e-4
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            gap = d - radii[i] - radii[j]
            if gap < min_gap:
                min_gap = gap
        d2b = min(centers[i,0]-radii[i], 1-centers[i,0]-radii[i], centers[i,1]-radii[i], 1-centers[i,1]-radii[i])
        if d2b < min_gap:
            min_gap = d2b
            
    if min_gap < 0:
        reduction = -min_gap + 1e-6
        radii -= reduction
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii

def run_packing():
    n = 26
    centers, radii = get_initial_layout(n)
    centers, radii = optimize_packing(centers, radii, n)
    return centers, radii, float(np.sum(radii))
