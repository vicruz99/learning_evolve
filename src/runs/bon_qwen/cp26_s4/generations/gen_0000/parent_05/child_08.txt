def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def objective_with_penalties(params, n):

    Objective function to minimize.
    params: flat array of [x1, y1, r1, x2, y2, r2, ...]
    We want to maximize sum(r), so we minimize -sum(r).
    We add penalties for constraint violations.

    score = -np.sum(radii)

    if np.any(radii < 0):

    # Circle i is inside if r <= x <= 1-r and r <= y <= 1-r
    # Violation amount: max(0, r - x, x - (1-r), r - y, y - (1-r))
    # Or simpler: distance from center to boundary must be >= r
    # dist to left: x - r. if < 0, penalty.
    
    boundary_violations = np.maximum(0, radii - centers[:, 0]) + \
                          np.maximum(0, radii - centers[:, 1]) + \
                          np.maximum(0, centers[:, 0] - (1 - radii)) + \
                          np.maximum(0, centers[:, 1] - (1 - radii))
    score += 1000.0 * np.sum(boundary_violations ** 2)

    # diff = centers[i] - centers[j]

    # Using broadcasting
    # centers_diff: (n, n, 2)
    # dists: (n, n)

    # Vectorized:
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    dists = np.sqrt(np.sum(diff**2, axis=2)) # (n, n)
    
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :] # (n, n)
    overlaps = rad_sum - dists
    
    # Only consider upper triangle i < j
    # Create mask
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap_amounts = np.maximum(0, overlaps[mask])
    
    # Penalty is large for overlaps
    score += 10000.0 * np.sum(overlap_amounts ** 2)
    
    return score
