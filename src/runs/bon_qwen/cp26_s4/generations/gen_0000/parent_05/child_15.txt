def constraints_factory(n):
    """
    Create list of constraint dictionaries for scipy.
    Constraints:
    1. x_i - r_i >= 0
    2. 1 - x_i - r_i >= 0
    3. y_i - r_i >= 0
    4. 1 - y_i - r_i >= 0
    5. r_i >= 0
    6. dist(i,j)^2 - (r_i + r_j)^2 >= 0
    """

        # r_i >= 0 (handled by bounds usually, but explicit is safe)
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+2]})
