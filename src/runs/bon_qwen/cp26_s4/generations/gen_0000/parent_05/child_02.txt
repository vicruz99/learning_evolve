import numpy as np
import scipy.optimize
import scipy.spatial

def calculate_max_equal_radius(centers):
    """
    Calculates the maximum radius r such that all circles with these centers
    and radius r are non-overlapping and inside the unit square.
    """
    n = centers.shape[0]
    r_boundary = np.min([np.min(centers), np.min(1.0 - centers)], axis=1)
    
    # Pairwise distances
    dists = scipy.spatial.distance.pdist(centers, metric='euclidean')
    min_dist = np.min(dists)
    
    # r is limited by half the minimum distance between any pair
    r_pairwise = min_dist / 2.0
    
    return min(np.min(r_boundary), r_pairwise)

def get_initial_configs(n):
    """Generates a list of initial center configurations."""
    configs = []
    
    # 1. Random configuration
    rng = np.random.RandomState(42)
    for _ in range(2):
        centers = rng.rand(n, 2) * 0.6 + 0.2 # Start in middle-ish
        configs.append(centers)

    # 2. 5x5 Grid + 1 (shifted)
    grid_centers = []
    for i in range(5):
        for j in range(5):
            grid_centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    grid_centers.append([0.5, 0.5]) # Center point (will cause overlap, optimizer fixes)
    configs.append(np.array(grid_centers))

    # 3. Hexagonal Lattice approximation
    hex_centers = []
    r_approx = 0.11
    dy = np.sqrt(3) * r_approx
    for row in range(7): # 7 rows to fit 26
        for col in range(6):
            x = col * 2 * r_approx + (row % 2) * r_approx + 0.05
            y = row * dy + 0.05
            if x < 0.95 and y < 0.95:
                hex_centers.append([x, y])
                if len(hex_centers) == 26:
                    break
        if len(hex_centers) == 26:
            break
    if len(hex_centers) == 26:
        configs.append(np.array(hex_centers))

    return configs

def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None

    # Stage 1: Optimize Centers for Equal Radii
    initial_configs = get_initial_configs(n)

    for init_centers in initial_configs:
        # Flatten centers for optimizer
        x0 = init_centers.flatten()
        
        # Objective: Maximize the minimal feasible radius
        # We minimize the negative of this radius
        def objective(x_flat):
            c = x_flat.reshape((n, 2))
            # Penalize if out of bounds significantly to keep search space valid
            # though the radius function handles it naturally
            return -calculate_max_equal_radius(c)

        # Nelder-Mead is robust for non-smooth objectives
        res = scipy.optimize.minimize(
            objective, 
            x0, 
            method='Nelder-Mead', 
            options={'maxiter': 5000, 'xatol': 1e-5, 'fatol': 1e-5}
        )
        
        if res.success or (not res.success and -res.fun > best_sum / n):
            current_r = -res.fun
            if current_r > best_sum / n:
                best_sum = current_r * n
                best_centers = res.x.reshape((n, 2))

    # Stage 2: Linear Programming to optimize radii given centers
    # Maximize sum(r_i)
    # Constraints:
    # 1. r_i + r_j <= dist_ij for all i < j
    # 2. r_i <= dist_to_boundary_i
    # 3. r_i >= 0
    
    # Calculate boundary limits for each circle
    centers = best_centers
    r_boundary_limits = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                                   np.minimum(centers[:, 1], 1 - centers[:, 1]))
    
    # Pairwise constraints
    # r_i + r_j <= d_ij  =>  r_i + r_j - d_ij <= 0
    # We need to construct A_ub * r <= b_ub
    n_constraints = 0
    for i in range(n):
        for j in range(i + 1, n):
            n_constraints += 1
            
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Add boundary constraints: r_i <= limit_i
    # r_i - limit_i <= 0
    for i in range(n):
        A_ub = np.vstack([A_ub, np.zeros(n)])
        b_ub = np.append(b_ub, 0)
        A_ub[-1, i] = 1.0
        b_ub[-1] = r_boundary_limits[i]

    # Objective: Maximize sum(r) => Minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res_lp = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res_lp.success:
            best_radii = res_lp.x
            best_sum = -res_lp.fun
            # Centers remain the same
        else:
            # Fallback to equal radii if LP fails
            best_radii = np.ones(n) * (-best_centers.shape[0] * 0.0) # Placeholder
            # Actually, just use the radius found in stage 1
            r_eq = calculate_max_equal_radius(centers)
            best_radii = np.ones(n) * r_eq
            best_sum = r_eq * n
    except Exception:
        r_eq = calculate_max_equal_radius(centers)
        best_radii = np.ones(n) * r_eq
        best_sum = r_eq * n

    # Validation check (mental or explicit)
    # Ensure no NaNs
    if np.isnan(best_radii).any():
        best_radii = np.zeros(n)
        best_sum = 0
        
    return best_centers, best_radii, best_sum
