import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None

    def make_initial_config(config_type):
        centers_list = []
        if config_type == 0:
            # 5x5 grid + 1 circle in gap
            for i in range(5):
                for j in range(5):
                    centers_list.append([0.1 + 0.2 * i, 0.1 + 0.2 * j])
            centers_list.append([0.25, 0.25])
            r_init = 0.085
        elif config_type == 1:
            # Hexagonal: 5,6,5,6,4
            r_init = 0.075
            row_counts = [5, 6, 5, 6, 4]
            for row_idx, count in enumerate(row_counts):
                y = r_init + row_idx * np.sqrt(3) * r_init
                x_start = r_init if row_idx % 2 == 0 else r_init * 2
                for col_idx in range(count):
                    x = x_start + col_idx * 2 * r_init
                    centers_list.append([x, y])
        elif config_type == 2:
            # Hexagonal: 6,5,6,5,4
            r_init = 0.075
            row_counts = [6, 5, 6, 5, 4]
            for row_idx, count in enumerate(row_counts):
                y = r_init + row_idx * np.sqrt(3) * r_init
                x_start = r_init * 2 if row_idx % 2 == 0 else r_init
                for col_idx in range(count):
                    x = x_start + col_idx * 2 * r_init
                    centers_list.append([x, y])
        elif config_type == 3:
            # 5x5 with adjusted spacing
            for i in range(5):
                for j in range(5):
                    centers_list.append([0.11 + 0.21 * i, 0.11 + 0.21 * j])
            centers_list.append([0.55, 0.11])
            r_init = 0.08
        elif config_type == 4:
            # Staggered 5x5 + 1
            for i in range(5):
                for j in range(5):
                    x = 0.12 + 0.22 * i + (0.11 if j % 2 == 1 else 0)
                    y = 0.12 + 0.19 * j
                    centers_list.append([x, y])
            centers_list.append([0.5, 0.5])
            r_init = 0.075
        else:
            # Random perturbation of 5x5
            for i in range(5):
                for j in range(5):
                    centers_list.append([0.1 + 0.2 * i, 0.1 + 0.2 * j])
            centers_list.append([0.25, 0.25])
            r_init = 0.085

        centers_arr = np.array(centers_list)
        radii_arr = np.ones(n) * r_init
        return centers_arr, radii_arr

    def force_directed_optimize(centers, radii, iterations=5000):
        centers = centers.copy()
        radii = radii.copy()
        dt = 0.02
        growth_rate = 2e-5

        for _ in range(iterations):
            forces = np.zeros_like(centers)

            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist_sq = np.sum(diff ** 2)
                    dist = np.sqrt(dist_sq)
                    min_dist = radii[i] + radii[j]
                    if dist < min_dist and dist > 1e-10:
                        overlap = min_dist - dist
                        force_mag = overlap / dist
                        forces[i] += force_mag * diff
                        forces[j] -= force_mag * diff

            for i in range(n):
                if centers[i, 0] < radii[i]:
                    forces[i, 0] += (radii[i] - centers[i, 0]) * 20
                if centers[i, 0] > 1 - radii[i]:
                    forces[i, 0] -= (centers[i, 0] - (1 - radii[i])) * 20
                if centers[i, 1] < radii[i]:
                    forces[i, 1] += (radii[i] - centers[i, 1]) * 20
                if centers[i, 1] > 1 - radii[i]:
                    forces[i, 1] -= (centers[i, 1] - (1 - radii[i])) * 20

            centers += forces * dt

            can_grow = True
            for i in range(n):
                if (centers[i, 0] - radii[i] < -1e-8 or
                        centers[i, 0] + radii[i] > 1 + 1e-8 or
                        centers[i, 1] - radii[i] < -1e-8 or
                        centers[i, 1] + radii[i] > 1 + 1e-8):
                    can_grow = False
                    break

            if can_grow:
                radii += growth_rate
            else:
                growth_rate *= 0.5
                if growth_rate < 1e-10:
                    break

        return centers, radii

    def get_cr(x):
        c = x[:52].reshape(-1, 2)
        r = x[52:]
        return c, r

    def objective(x):
        return -np.sum(x[52:])

    def boundary_con(x):
        c, r = get_cr(x)
        out = []
        for i in range(n):
            out.append(c[i, 0] - r[i])
            out.append(1 - c[i, 0] - r[i])
            out.append(c[i, 1] - r[i])
            out.append(1 - c[i, 1] - r[i])
        return np.array(out)

    def overlap_con(x):
        c, r = get_cr(x)
        out = []
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                out.append(d - r[i] - r[j])
        return np.array(out)

    constraints = [
        {'type': 'ineq', 'fun': boundary_con},
        {'type': 'ineq', 'fun': overlap_con}
    ]

    bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)] * 26

    for cfg in range(5):
        centers, radii = make_initial_config(cfg)
        centers_fd, radii_fd = force_directed_optimize(centers, radii, 3000)

        x0 = np.concatenate([centers_fd.flatten(), radii_fd])

        try:
            result = minimize(objective, x0, method='SLSQP',
                              constraints=constraints, bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})

            if result.success or (-result.fun > best_sum):
                c_opt = result.x[:52].reshape(-1, 2)
                r_opt = result.x[52:]

                # Validate and fix small violations
                valid = True
                for i in range(n):
                    for coord in range(2):
                        if c_opt[i, coord] - r_opt[i] < 0:
                            c_opt[i, coord] = r_opt[i] + 1e-10
                        if c_opt[i, coord] + r_opt[i] > 1:
                            c_opt[i, coord] = 1 - r_opt[i] - 1e-10

                s = np.sum(r_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass

        # Also try direct SLSQP without force-directed
        x0_direct = np.concatenate([centers.flatten(), radii])
        try:
            result2 = minimize(objective, x0_direct, method='SLSQP',
                               constraints=constraints, bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if -result2.fun > best_sum:
                c_opt = result2.x[:52].reshape(-1, 2)
                r_opt = result2.x[52:]
                for i in range(n):
                    for coord in range(2):
                        if c_opt[i, coord] - r_opt[i] < 0:
                            c_opt[i, coord] = r_opt[i] + 1e-10
                        if c_opt[i, coord] + r_opt[i] > 1:
                            c_opt[i, coord] = 1 - r_opt[i] - 1e-10
                s = np.sum(r_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass

    # Final refinement on best result
    if best_centers is not None:
        x0_final = np.concatenate([best_centers.flatten(), best_radii])
        try:
            result_final = minimize(objective, x0_final, method='SLSQP',
                                    constraints=constraints, bounds=bounds,
                                    options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
            if -result_final.fun > best_sum:
                c_opt = result_final.x[:52].reshape(-1, 2)
                r_opt = result_final.x[52:]
                for i in range(n):
                    for coord in range(2):
                        if c_opt[i, coord] - r_opt[i] < 0:
                            c_opt[i, coord] = r_opt[i] + 1e-10
                        if c_opt[i, coord] + r_opt[i] > 1:
                            c_opt[i, coord] = 1 - r_opt[i] - 1e-10
                best_sum = np.sum(r_opt)
                best_centers = c_opt
                best_radii = r_opt
        except Exception:
            pass

    if best_centers is None:
        centers, radii = make_initial_config(0)
        return centers, radii, np.sum(radii)

    return best_centers, best_radii, best_sum
