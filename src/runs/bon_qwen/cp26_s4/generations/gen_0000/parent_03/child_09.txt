def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        radii = vars[2::3]
        return -np.sum(radii)

        radii = vars[2::3]
        return centers, radii

        centers, radii = get_centers_radii(vars)
        cons = []
        
        # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v: v[0::3] - v[2::3]})
        # 1 - (x + r) >= 0 => 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v: 1.0 - v[0::3] - v[2::3]})
        # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v: v[1::3] - v[2::3]})
        # 1 - (y + r) >= 0 => 1 - y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v: 1.0 - v[1::3] - v[2::3]})

        # However, SLSQP constraints take full variable vector.

            c = get_centers_radii(v)[0]

        return cons
