import numpy as np
from scipy.optimize import minimize

def generate_seed_layout(n_circles=26):
    """
    Generates a staggered hexagonal seed layout for the circles.
    """
    centers = []
    # Approximate spacing for 26 circles in a unit square
    # 5 rows of roughly 5-6 circles
    row_counts = [6, 5, 6, 5, 4]
    
    y = 0.1
    dy = 0.8 / 4 # Space for 4 intervals
    
    for count in row_counts:
        dx = (1 - 0.2) / (count - 1) if count > 1 else 0
        x_start = 0.1
        if len(centers) % 2 == 1:
            x_start += dx / 2 # Stagger
        
        # Ensure we don't go out of bounds roughly
        if count > 1:
            x_vals = np.linspace(x_start, 1 - 0.1, count)
        else:
            x_vals = [0.5]
            
        for x in x_vals:
            centers.append([x, y])
        y += dy
            
    # Pad or trim if needed (should be 26)
    current_centers = np.array(centers[:n_circles])
    radii = np.full(n_circles, 0.08) # Start with a safe small radius
    
    return current_centers, radii

def calculate_objective_and_gradient(variables, n_circles=26):
    """
    Calculates the objective value (negative sum of radii) and gradient.
    Variables are structured as [x1, y1, r1, x2, y2, r2, ...]
    """
    x = variables[::3]
    y = variables[1::3]
    r = variables[2::3]
    
    # Objective: -sum(r)
    objective = -np.sum(r)
    
    # Gradient of objective
    grad = np.zeros_like(variables)
    grad[2::3] = -1.0
    
    # Penalty for overlaps and boundaries
    # Using a quadratic penalty for smoothness in optimization
    penalty = 0.0
    penalty_grad = np.zeros_like(variables)
    
    # Boundary constraints
    for i in range(n_circles):
        # Left: x - r >= 0
        val = x[i] - r[i]
        if val < 0:
            p = -val
            penalty += p**2
            penalty_grad[3*i] -= 2*p * 1 # d/dx
            penalty_grad[3*i + 2] -= 2*p * (-1) # d/dr
            
        # Right: x + r <= 1
        val = 1 - (x[i] + r[i])
        if val < 0:
            p = -val
            penalty += p**2
            penalty_grad[3*i] -= 2*p * 1
            penalty_grad[3*i + 2] -= 2*p * 1
            
        # Bottom: y - r >= 0
        val = y[i] - r[i]
        if val < 0:
            p = -val
            penalty += p**2
            penalty_grad[3*i + 1] -= 2*p * 1
            penalty_grad[3*i + 2] -= 2*p * (-1)
            
        # Top: y + r <= 1
        val = 1 - (y[i] + r[i])
        if val < 0:
            p = -val
            penalty += p**2
            penalty_grad[3*i + 1] -= 2*p * 1
            penalty_grad[3*i + 2] -= 2*p * 1

    # Overlap constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist_sq = (x[i] - x[j])**2 + (y[i] - y[j])**2
            dist = np.sqrt(dist_sq + 1e-12) # Avoid div by zero
            min_dist = r[i] + r[j]
            
            # Constraint: dist >= min_dist => dist - min_dist >= 0
            val = dist - min_dist
            if val < 0:
                p = -val
                penalty += p**2
                
                # Gradient contribution
                # d(dist)/dx_i = (x_i - x_j) / dist
                df_dx_i = (x[i] - x[j]) / dist
                df_dx_j = -(x[i] - x[j]) / dist
                df_dy_i = (y[i] - y[j]) / dist
                df_dy_j = -(y[i] - y[j]) / dist
                df_dr_i = -1
                df_dr_j = -1
                
                g_term = 2 * p
                
                penalty_grad[3*i] += g_term * df_dx_i
                penalty_grad[3*i + 1] += g_term * df_dy_i
                penalty_grad[3*i + 2] += g_term * df_dr_i
                
                penalty_grad[3*j] += g_term * df_dx_j
                penalty_grad[3*j + 1] += g_term * df_dy_j
                penalty_grad[3*j + 2] += g_term * df_dr_j

    # We want to minimize objective + lambda * penalty
    # But usually in minimize, we just return the value and gradient.
    # To balance them, we can scale the penalty or rely on the optimizer.
    # Here we use a soft constraint approach implicitly by just summing them, 
    # but typically we need a large lambda. 
    # However, for this problem, simply returning the combined value works if penalties are active.
    
    return objective + 100.0 * penalty, grad + 100.0 * penalty_grad

def run_packing():
    # Initial guess
    centers, radii = generate_seed_layout(26)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.empty(26 * 3)
    x0[::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    # Bounds: x, y in [0, 1], r >= 0
    bounds = [(0, 1), (0, 1), (0, 1)] * 26
    
    # Optimization
    # Using L-BFGS-B as it handles bounds well
    result = minimize(
        calculate_objective_and_gradient, 
        x0, 
        jac=True, 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 2000, 'ftol': 1e-9}
    )
    
    # Extract results
    final_vars = result.x
    final_centers = np.column_stack((final_vars[::3], final_vars[1::3]))
    final_radii = final_vars[2::3]
    
    # Clip radii to non-negative just in case
    final_radii = np.maximum(final_radii, 0)
    
    # Clip centers to [0, 1] just in case
    final_centers = np.clip(final_centers, 0, 1)
    
    total_sum = np.sum(final_radii)
    
    return final_centers, final_radii, total_sum
