import numpy as np

def compute_overlap_matrix(centers, radii):
    """
    Compute a matrix of overlaps between all pairs of circles and with boundaries.
    Positive values indicate overlap.
    """
    n = centers.shape[0]
    overlaps = np.zeros(n)
    
    # Pairwise overlaps
    for i in range(n):
        r_i = radii[i]
        c_i = centers[i]
        # Check boundaries
        # x < r_i => overlap r_i - x
        if centers[i, 0] < r_i:
            overlaps[i] += (r_i - centers[i, 0])
        if centers[i, 0] > 1 - r_i:
            overlaps[i] += (centers[i, 0] - (1 - r_i))
        if centers[i, 1] < r_i:
            overlaps[i] += (r_i - centers[i, 1])
        if centers[i, 1] > 1 - r_i:
            overlaps[i] += (centers[i, 1] - (1 - r_i))
            
        for j in range(i + 1, n):
            r_j = radii[j]
            dist = np.sqrt(np.sum((c_i - centers[j]) ** 2))
            overlap = (r_i + r_j) - dist
            if overlap > 0:
                overlaps[i] += overlap
                overlaps[j] += overlap
                
    return overlaps

def resolve_packing(centers, radii, steps=100, stiffness=200.0, dt=0.005):
    """
    Iteratively push circles apart to resolve overlaps.
    """
    n = centers.shape[0]
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        
        # Calculate forces
        for i in range(n):

            c_i = centers[i]
            
            # Boundary forces (strong push to keep inside)
            # Left
            if c_i[0] < r_i:
                forces[i, 0] += stiffness * (r_i - c_i[0])
            # Right
            if c_i[0] > 1 - r_i:
                forces[i, 0] -= stiffness * (c_i[0] - (1 - r_i))
            # Bottom
            if c_i[1] < r_i:
                forces[i, 1] += stiffness * (r_i - c_i[1])
            # Top
            if c_i[1] > 1 - r_i:
                forces[i, 1] -= stiffness * (c_i[1] - (1 - r_i))
            
            for j in range(i + 1, n):
                r_j = radii[j]
                diff = c_i - centers[j]
                dist_sq = np.sum(diff ** 2)
                
                # Avoid division by zero
                if dist_sq < 1e-10:
                    dist = 1e-5
                    norm_diff = np.random.rand(2) * 2 - 1 # Random push
                else:
                    dist = np.sqrt(dist_sq)
                    norm_diff = diff / dist
                
                req_dist = r_i + r_j
                
                if dist < req_dist:
                    overlap = req_dist - dist
                    force_mag = stiffness * overlap
                    forces[i] += norm_diff * force_mag
                    forces[j] -= norm_diff * force_mag
        
        # Update positions
        centers += dt * forces
        
        # Strict clamping to [0, 1] to prevent numerical explosion, 
        # though forces should keep them valid. 
        # Note: Clamping to 0 might technically violate radius constraint immediately 
        # if r > 0, but the next force step will push it out to r.
        centers = np.clip(centers, 0.0, 1.0)
        
    return centers
