To maximize the sum of radii for 26 circles in a unit square, I will employ a hybrid optimization strategy that separates the geometry of the centers from the sizing of the radii. 

1. **Decoupling the Problem**: For any fixed set of center coordinates, finding the maximum radii is a Linear Programming (LP) problem. The objective is to maximize $\sum r_i$ subject to non-negativity, boundary constraints ($r_i \le x_i \le 1-r_i$), and non-overlap constraints ($r_i + r_j \le \text{distance}_{ij}$).
2. **Outer Optimization**: I will treat the center coordinates as the primary variables. Since the LP objective function is non-smooth with respect to center positions, I will use a derivative-free optimizer (`Nelder-Mead`) to adjust the centers.
3. **Variable Transformation**: To handle the $[0,1]$ boundary constraints for centers, I will transform the optimization variables using a hyperbolic tangent function, allowing the optimizer to search in an unconstrained space.
4. **Multi-Start Search**: To escape local minima, I will run the optimization from multiple starting configurations:
    - A 5x5 grid with a small perturbation.
    - A hexagonal packing pattern.
    - Randomly scattered centers.
5. **Finalization**: The best solution found will be returned.

This approach ensures that for every candidate configuration of centers, the radii are expanded to their absolute maximum limit, efficiently navigating the high-dimensional search space.

import numpy as np
from scipy.optimize import minimize, linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # ---------------------------------------------------------
    # Helper: Solve LP for optimal radii given fixed centers
    # ---------------------------------------------------------
    def solve_radii_for_centers(centers_flat):
        centers = centers_flat.reshape((n, 2))
        xs = centers[:, 0]
        ys = centers[:, 1]
        
        # Objective: maximize sum(r) -> minimize -sum(r)
        c_obj = -np.ones(n)
        
        # Constraints: A_ub * r <= b_ub
        constraints_count = 4 * n + n * (n - 1) // 2
        A_ub = np.zeros((constraints_count, n))
        b_ub = np.zeros(constraints_count)
        
        row_idx = 0
        
        # Boundary constraints
        for i in range(n):
            # r_i <= x_i
            A_ub[row_idx, i] = 1.0
            b_ub[row_idx] = xs[i]
            row_idx += 1
            
            # r_i <= 1 - x_i
            A_ub[row_idx, i] = 1.0
            b_ub[row_idx] = 1.0 - xs[i]
            row_idx += 1
            
            # r_i <= y_i
            A_ub[row_idx, i] = 1.0
            b_ub[row_idx] = ys[i]
            row_idx += 1
            
            # r_i <= 1 - y_i
            A_ub[row_idx, i] = 1.0
            b_ub[row_idx] = 1.0 - ys[i]
            row_idx += 1
            
        # Inter-circle constraints: r_i + r_j <= dist_ij
        # To ensure stability, we filter extremely close points
        min_dist_threshold = 1e-6
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                
                # If points are extremely close, set dist to 0 to force radii to 0
                # This avoids numerical issues in linprog
                if dist < min_dist_threshold:
                    dist = 0.0
                    
                A_ub[row_idx, i] = 1.0
                A_ub[row_idx, j] = 1.0
                b_ub[row_idx] = dist
                row_idx += 1

        # Solve LP
        # Bounds for r: [0, 0.5] (max radius in unit square is 0.5)
        r_bounds = [(0, 0.5) for _ in range(n)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
        
        if res.success:
            return -res.fun, res.x
        else:
            # If LP fails, return a penalty (very low sum) and zero radii
            return 0.0, np.zeros(n)

    # ---------------------------------------------------------
    # Objective function for the center optimizer
    # ---------------------------------------------------------
    def objective(z):
        # Transform unconstrained z to [0, 1] using tanh
        # x = 0.5 * (1 + tanh(z)) maps R -> (0, 1)
        centers_flat = 0.5 * (1.0 + np.tanh(z))
        
        sum_radii, _ = solve_radii_for_centers(centers_flat)
        return -sum_radii # We minimize negative sum

    # ---------------------------------------------------------
    # Generate Initial Guesses
    # ---------------------------------------------------------
    best_sum_radii = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    def run_optimization_from(initial_centers_flat):
        nonlocal best_sum_radii, best_centers, best_radii
        
        # Transform initial centers to z-space for Nelder-Mead
        # x = 0.5 * (1 + tanh(z)) => tanh(z) = 2x - 1 => z = arctanh(2x - 1)
        # Clip to avoid log(0)
        val = np.clip(2.0 * initial_centers_flat - 1.0, -0.99999, 0.99999)
        z0 = np.arctanh(val)
        
        res = minimize(objective, z0, method='Nelder-Mead', 
                       options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-8})
        
        current_sum = -res.fun
        if current_sum > best_sum_radii:
            # Transform back to get actual centers
            z_opt = res.x
            centers_opt = 0.5 * (1.0 + np.tanh(z_opt))
            _, radii_opt = solve_radii_for_centers(centers_opt)
            
            best_sum_radii = current_sum
            best_centers = centers_opt
            best_radii = radii_opt

    # 1. Grid-based initialization (5x5 + 1)
    grid_centers = []
    # 5x5 grid centered slightly to avoid exact boundaries
    step = 1.0 / 6.0
    for i in range(5):
        for j in range(5):
            x = step + i * step
            y = step + j * step
            grid_centers.append([x, y])
    # Add one extra circle at center
    grid_centers.append([0.5, 0.5])
    run_optimization_from(np.array(grid_centers).flatten())

    # 2. Hexagonal Packing Initialization
    hex_centers = []
    row_height = np.sqrt(3) / 2 * 0.2 # Approx spacing
    # Try to fit 26 circles
    y = 0.1
    row_count = 0
    while len(hex_centers) < 26:
        x = 0.1
        if row_count % 2 == 1:
            x = 0.15 # Offset
        while x <= 0.9 and len(hex_centers) < 26:
            hex_centers.append([x, y])
            x += 0.2
        y += row_height
        row_count += 1
    
    if len(hex_centers) > 0:
        run_optimization_from(np.array(hex_centers[:26]).flatten())

    # 3. Random Initializations (5 restarts)
    rng = np.random.default_rng(42)
    for _ in range(5):
        rand_centers = rng.random((n, 2))
        run_optimization_from(rand_centers.flatten())

    return best_centers, best_radii, best_sum_radii

# Execution
if __name__ == "__main__":
    centers, radii, total_sum = run_packing()
    print(f"Sum of radii: {total_sum}")
