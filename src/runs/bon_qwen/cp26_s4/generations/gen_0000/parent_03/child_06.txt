from scipy.optimize import differential_evolution, linprog

def run_packing():

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses Differential Evolution for position search and Linear Programming for radius determination.
    """

    dim = 2 * n_circles
    
    def calculate_max_radii_sum(centers_flat):
        # Reshape flat array to (26, 2) centers

        # 1 per pair for non-overlap (325)

        b_ub = []

        # These are upper bounds on r_i, but linprog uses A_ub @ x <= b_ub.
        # For r_i <= bound, we can just set bounds on variables or add constraints.
        # Let's use explicit bounds for simplicity, but linprog bounds are per variable.
        # However, the bounds depend on the centers (which are fixed here).
        
        r_bounds = []
        for i in range(n_circles):
            x, y = centers[i]
            min_dist = min(x, 1 - x, y, 1 - y)
            # Clip to 0 to avoid negative bounds
            r_bounds.append((0, max(0, min_dist)))

        # We need to construct rows for A_ub
        # Each row has 1 at index i, 1 at index j, and 0 elsewhere.
        # RHS is distance between centers i and j.
        
        dists = np.zeros((n_circles, n_circles))
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                dists[i, j] = dist
                dists[j, i] = dist

        A_ub = np.zeros((n_circles * (n_circles - 1) // 2, n_circles))
        b_ub = np.zeros(n_circles * (n_circles - 1) // 2)

        for i in range(n_circles):

                return 0.0
