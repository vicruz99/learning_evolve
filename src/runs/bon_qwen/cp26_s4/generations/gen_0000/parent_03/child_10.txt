

<strategy>
I'll use an alternating optimization strategy:
1. Initialize with multiple configurations (hexagonal, grid patterns)
2. Alternate between:
   a. Given fixed centers, solve an LP to find optimal radii (maximize sum of radii subject to non-overlap and boundary constraints)
   b. Given fixed radii, use force-based relaxation and Nelder-Mead optimization to push circles apart and reduce overlaps
3. Try multiple restarts with different initial configurations
4. The LP for radii is exact and efficient - it transforms the radius optimization into a linear program
5. For position optimization, I'll use both force-based repulsion and scipy's Nelder-Mead for refinement
6. The key insight is that unequal circles allow much higher sum of radii than equal circles - we want a few large circles and many smaller ones filling gaps
</strategy>

```python
import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26


def initialize_hexagonal(n):
    centers = []
    rows = 6
    for i in range(rows):
        cols = 5 if i % 2 == 0 else 4
        for j in range(cols):
            if len(centers) >= n:
                break
            if i % 2 == 0:
                x = (j + 0.5) / 5.0
            else:
                x = (j + 0.5) / 4.0 + 0.05
            y = (i + 0.5) / 6.0
            centers.append([x, y])
        if len(centers) >= n:
            break
    return np.array(centers[:n])


def initialize_grid(n, rows=None):
    if rows is None:
        rows = int(np.ceil(np.sqrt(n)))
    centers = []
    cols = int(np.ceil(n / rows))
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = (j + 0.5) / cols
            y = (i + 0.5) / rows
            centers.append([x, y])
        if len(centers) >= n:
            break
    return np.array(centers[:n])


def initialize_corners_plus_grid(n):
    centers = []
    # Place some circles near corners
    corner_positions = [
        [0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]
    ]
    centers.extend(corner_positions)
    # Fill rest with grid
    remaining = n - len(centers)
    rows = int(np.ceil(np.sqrt(remaining)))
    cols = int(np.ceil(remaining / rows))
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = 0.3 + (j + 0.5) * 0.4 / cols
            y = 0.3 + (i + 0.5) * 0.4 / rows
            centers.append([x, y])
        if len(centers) >= n:
            break
    return np.array(centers[:n])


def initialize_pentagonal_layers(n):
    centers = []
    # Center circle
    centers.append([0.5, 0.5])
    # First ring
    for i in range(6):
        angle = i * np.pi / 3
        x = 0.5 + 0.2 * np.cos(angle)
        y = 0.5 + 0.2 * np.sin(angle)
        centers.append([x, y])
        if len(centers) >= n:
            break
    # Second ring
    for i in range(12):
        if len(centers) >= n:
            break
        angle = i * np.pi / 6
        r = 0.35
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers.append([x, y])
    # Fill remaining randomly in gaps
    while len(centers) < n:
        x = np.random.uniform(0.05, 0.95)
        y = np.random.uniform(0.05, 0.95)
        ok = True
        for cx, cy in centers:
            if np.sqrt((x - cx)**2 + (y - cy)**2) < 0.12:
                ok = False
                break
        if ok:
            centers.append([x, y])
    return np.array(centers[:n])


def compute_optimal_radii(centers, n):
    c_obj = -np.ones(n)
    
    r_max = np.ones(n)
    for i in range(n):
        x, y = centers[i]
        r_max[i] = min(x, 1 - x, y, 1 - y)
        r_max[i] = max(r_max[i], 0.0001)
    
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
    
    bounds = [(0.0, r_max[i]) for i in range(n)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and np.all(result.x >= -1e-10):
        return np.maximum(result.x, 0.0)
    return np.ones(n) * 0.01


def position_objective(pos, radii, n):
    c = pos.reshape((n, 2))
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_d = radii[i] + radii[j]
            violation = min_d - dist
            if violation > 0:
                penalty += 5000.0 * violation ** 2
    for i in range(n):
        x, y = c[i]
        r = radii[i]
        if x < r:
            penalty += 5000.0 * (r - x) ** 2
        if x > 1 - r:
            penalty += 5000.0 * (x - (1 - r)) ** 2
        if y < r:
            penalty += 5000.0 * (r - y) ** 2
        if y > 1 - r:
            penalty += 5000.0 * (y - (1 - r)) ** 2
    return penalty


def force_relaxation(centers, radii, n, max_iter=5000, dt=0.005):
    c = centers.copy()
    for it in range(max_iter):
        forces = np.zeros((n, 2))
        max_force = 0.0
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[j, 0] - c[i, 0]
                dy = c[j, 1] - c[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    repulsion = overlap / dist * 5.0
                    fx = dx * repulsion
                    fy = dy * repulsion
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
                    max_force = max(max_force, overlap)
        
        for i in range(n):
            x, y = c[i]
            r = radii[i]
            if x < r:
                forces[i, 0] += (r - x) * 50.0
            if x > 1 - r:
                forces[i, 0] -= (x - (1 - r)) * 50.0
            if y < r:
                forces[i, 1] += (r - y) * 50.0
            if y > 1 - r:
                forces[i, 1] -= (y - (1 - r)) * 50.0
        
        c = c + forces * dt
        c = np.clip(c, 0.001, 0.999)
        
        if max_force < 1e-8:
            break
    
    return c


def optimize_positions_nm(centers, radii, n):
    result = minimize(
        position_objective,
        centers.flatten(),
        args=(radii, n),
        method='Nelder-Mead',
        options={'maxiter': 20000, 'xatol': 1e-12, 'fatol': 1e-14}
    )
    return result.x.reshape((n, 2))


def run_packing():
    n = N_CIRCLES
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    inits = [
        initialize_hexagonal(n),
        initialize_grid(n, 5),
        initialize_grid(n, 6),
        initialize_corners_plus_grid(n),
        initialize_pentagonal_layers(n),
    ]
    
    np.random.seed(42)
    
    for init_idx, centers_base in enumerate(inits):
        for trial in range(5):
            centers = centers_base.copy()
            if trial > 0:
                scale = 0.01 * trial + 0.005
                centers = centers + np.random.randn(n, 2) * scale
            centers = np.clip(centers, 0.03, 0.97)
            
            radii = np.ones(n) * 0.02
            
            # Phase 1: Force relaxation with small radii
            centers = force_relaxation(centers, radii, n, max_iter=2000, dt=0.01)
            
            # Phase 2: Alternating optimization
            for iteration in range(80):
                radii = compute_optimal_radii(centers, n)
                centers = force_relaxation(centers, radii, n, max_iter=3000, dt=0.003)
                
                if iteration % 10 == 5:
                    centers = optimize_positions_nm(centers, radii, n)
                    radii = compute_optimal_radii(centers, n)
            
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()
    
    # Phase 3: Fine-tuning with more iterations
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for iteration in range(150):
        radii_new = compute_optimal_radii(centers, n)
        centers = force_relaxation(centers, radii_new, n, max_iter=5000, dt=0.002)
        radii_new2 = compute_optimal_radii(centers, n)
        
        if np.sum(radii_new2) > np.sum(radii) - 1e-10:
            centers = optimize_positions_nm(centers, radii_new2, n)
            radii = compute_optimal_radii(centers, n)
        else:
            radii = radii_new2
    
    # Final LP to ensure radii are optimal for these centers
    radii = compute_optimal_radii(centers, n)
    
    # Verify no negative radii
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
```
