This ensures a high-quality starting configuration with minimal overlaps.

2.  **Numerical Optimization:** I will then use `scipy.optimize.minimize` with the `SLSQP` method. The objective will be to maximize the sum of radii (minimize the negative sum). To ensure a valid packing, I will enforce hard constraints for boundaries ($r_i \le x_i, y_i \le 1-r_i$) and non-overlap ($(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$). import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.01) # Initial small radius

    # 1. Initialize centers using a hexagonal grid pattern
    # This provides a dense, structured starting point
    r_init = 0.1
    points = []
    # Generate a dense hexagonal grid
    for row in range(20):
        y = r_init + row * r_init * np.sqrt(3)
        if y + r_init > 1.0:
            break
        for col in range(20):
            x = r_init + col * 2 * r_init
            if row % 2 == 1:
                x += r_init
            if x + r_init <= 1.0:
                points.append([x, y])

    # Select the first 26 points if we have enough, otherwise fallback to random
    if len(points) >= n:
        # To ensure diversity, we might want to pick points that are somewhat spread, 
        # but a dense grid is usually fine for the optimizer to expand from.
        # However, to help the optimizer, let's pick points that fit well.
        # Sorting by distance from center or just taking first n is okay.
        centers[:n, :] = np.array(points[:n])
    else:
        # Fallback: Random uniform distribution if grid fails
        centers = np.random.uniform(0.1, 0.9, size=(n, 2))

    # Initial radii estimation: assume equal radii for a dense packing
    # Area argument: 26 * pi * r^2 ~ 0.9 (high density) -> r ~ 0.105
    # Start slightly smaller to ensure feasibility
    radii[:] = 0.09

    def objective(x):
        # x contains [x0, y0, x1, y1, ..., r0, r1, ...]
        # We want to maximize sum(r), so minimize -sum(r)
        return -np.sum(x[2*n:])

    def boundary_constraints(x):
        # Returns array of values that must be >= 0
        # r_i <= x_i, r_i <= y_i, x_i + r_i <= 1, y_i + r_i <= 1
        # -> x_i - r_i >= 0, y_i - r_i >= 0, 1 - x_i - r_i >= 0, 1 - y_i - r_i >= 0
        cons = []
        for i in range(n):
            xi = x[i*2]
            yi = x[i*2 + 1]
            ri = x[2*n + i]
            cons.append(xi - ri)
            cons.append(yi - ri)
            cons.append(1 - xi - ri)
            cons.append(1 - yi - ri)
        return np.array(cons)

    def overlap_constraints(x):
        # Returns array of values that must be >= 0
        # dist^2 >= (ri + rj)^2  ->  dist^2 - (ri + rj)^2 >= 0
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi = x[i*2], x[i*2+1]
                xj, yj = x[j*2], x[j*2+1]
                ri, rj = x[2*n + i], x[2*n + j]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                
                # Constraint: dist_sq - sum_r^2 >= 0
                cons.append(dist_sq - sum_r**2)
        return np.array(cons)

    # Combine constraints
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]

    # Bounds: radii must be non-negative
    # x, y can be technically outside if not for boundary constraints, 
    # but we can set loose bounds or rely on constraints.
    # Let's set loose bounds for x, y and [0, 0.5] for r.
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1)]) # x, y
    for _ in range(n):
        bounds.extend([(0, 0.5)]) # r

    # Flatten initial state
    x0 = np.hstack([centers.flatten(), radii])

    # Optimize
    # Using SLSQP for constrained optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})

    # Extract results
    final_centers = res.x[:2*n].reshape((n, 2))
    final_radii = res.x[2*n:]

    # Validate and fix numerical issues if any (clip radii to be non-negative)
    final_radii = np.maximum(final_radii, 0)

    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii
