import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # Helper to create initial hexagonal packing
    def create_hex_packing(n):
        # Heuristic row distribution for 26 circles to fit in square
        # 5, 4, 5, 4, 5, 3 -> 26 circles
        row_counts = [5, 4, 5, 4, 5, 3]
        
        centers = []
        # Estimate radius to fit roughly in square
        # Width constraint: 2*r*max_row <= 1 => r <= 1/10 = 0.1
        # Height constraint: 2*r + (rows-1)*r*sqrt(3) <= 1
        # 2 + 5*1.732 = 10.66 => r <= 0.093
        # Let's start with r=0.08
        r_init = 0.08
        
        y = r_init
        for count in row_counts:
            # Determine x start
            # If row is even index (1-based, i.e., 2nd, 4th), shift by r
            if len(centers) % 2 == 1:
                x_start = r_init + r_init # Shifted
            else:
                x_start = r_init
            
            # Calculate width of this row to center it if needed, 
            # but simple placement from left is safer for boundaries
            # Actually, to maximize packing, centering might be better.
            # Let's just place them.
            
            # Adjust x_start to fit width if count is small? 
            # No, standard lattice is fine.
            
            # If shifted, x_start is 2r. If not, x_start is r.
            # Wait, in hex packing:
            # Row 0: centers at r, 3r, 5r... (starts at r)
            # Row 1: centers at 2r, 4r, 6r... (starts at 2r)
            
            if len(centers) % 2 == 0:
                x_start = r_init
            else:
                x_start = 2 * r_init
                
            for _ in range(count):
                centers.append([x_start, y])
                x_start += 2 * r_init
            
            y += r_init * np.sqrt(3)
            
        # Trim or pad if needed (though row_counts sums to 26)
        centers = centers[:n]
        while len(centers) < n:
            # Fallback if needed (shouldn't be)
            centers.append([0.5, 0.5])
            
        return np.array(centers), np.full(n, r_init)

    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Objective: minimize negative sum of radii
    def objective(vars):
        r = vars[2::3] # radii are at indices 2, 5, 8...
        return -np.sum(r)

    # Constraints
    # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    #    x - r >= 0  =>  x - r >= 0
    #    1 - x - r >= 0
    #    y - r >= 0
    #    1 - y - r >= 0
    
    # 2. Non-overlap: dist >= r_i + r_j
    #    (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    
    # We will use SLSQP.
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    
    # Bounds
    # x, y in [0, 1]
    # r in [0, 0.5] (actually max r is 0.5)
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])

    def get_constraints():
        cons = []
        # Boundary constraints
        for i in range(n_circles):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]})
            # 1 - x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[idx_x] - v[idx_r]})
            # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - v[idx_r]})
            # 1 - y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[idx_y] - v[idx_r]})
            
        # Overlap constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
                idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
                
                def dist_sq_constraint(v, i=i, j=j):
                    xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                    xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    radii_sum_sq = (ri + rj)**2
                    return dist_sq - radii_sum_sq
                
                cons.append({'type': 'ineq', 'fun': dist_sq_constraint})
        return cons

    constraints = get_constraints()
    
    # Run optimization from multiple starts
    # Strategy: Use the hex packing as base, add small noise
    base_centers, base_radii = create_hex_packing(n_circles)
    
    # Combine into vector
    base_vars = np.zeros(3 * n_circles)
    for i in range(n_circles):
        base_vars[3*i] = base_centers[i, 0]
        base_vars[3*i+1] = base_centers[i, 1]
        base_vars[3*i+2] = base_radii[i]

    for seed in range(10): # 10 restarts
        np.random.seed(seed)
        # Perturb initial guess
        # Slightly scale radii down to ensure feasibility, then optimizer will expand
        # Or just random noise
        x0 = base_vars.copy()
        x0[::3] += np.random.normal(0, 0.02, n_circles) # x noise
        x0[1::3] += np.random.normal(0, 0.02, n_circles) # y noise
        # Clamp coordinates to [0, 1]
        x0[::3] = np.clip(x0[::3], 0.01, 0.99)
        x0[1::3] = np.clip(x0[1::3], 0.01, 0.99)
        
        # Maybe slightly reduce radius to start feasible
        x0[2::3] = 0.08 

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success or res.fun < -best_sum + 1e-6: # Check if better
                # Check validity manually just in case
                # (The validation function is separate, but good to be safe)
                # However, we trust SLSQP mostly.
                # Extract radii
                current_radii = res.x[2::3]
                current_sum = np.sum(current_radii)
                
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = np.column_stack((res.x[::3], res.x[1::3]))
                    best_radii = current_radii
        except Exception:
            continue
            
    # If optimization failed or result is bad, return the initial hex packing
    if best_centers is None:
        best_centers = base_centers
        best_radii = base_radii
        best_sum = np.sum(base_radii)

    # Ensure non-negative radii and clip coordinates just in case
    best_radii = np.maximum(best_radii, 0)
    best_centers[:, 0] = np.clip(best_centers[:, 0], 0, 1)
    best_centers[:, 1] = np.clip(best_centers[:, 1], 0, 1)
    
    return best_centers, best_radii, best_sum
