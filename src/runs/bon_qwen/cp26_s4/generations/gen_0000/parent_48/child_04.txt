import numpy as np

def adjust_for_validity(centers, radii):
    """Resolve overlaps and boundary violations iteratively."""
    n = centers.shape[0]
    for _ in range(50):
        # Boundary constraints
        for i in range(n):
            for d in range(2):
                if centers[i, d] < radii[i]:
                    centers[i, d] = radii[i] + 1e-8
                elif centers[i, d] > 1.0 - radii[i]:
                    centers[i, d] = 1.0 - radii[i] - 1e-8
                    
        # Inter-circle repulsion to resolve overlaps
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-9:
                    dist = 1e-9
                    diff = np.array([1.0, 0.0])
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    shift = (diff / dist) * overlap / 2.0
                    centers[i] += shift
                    centers[j] -= shift
                    
    return centers
