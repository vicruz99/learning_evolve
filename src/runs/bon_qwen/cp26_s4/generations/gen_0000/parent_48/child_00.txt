import numpy as np
from scipy.optimize import minimize

def get_initial_guess(seed=42):
    """
    Initializes 26 circles using a hexagonal lattice pattern.
    """
    np.random.seed(seed)
    n = 26
    r_init = 0.09
    
    centers = []
    y_val = r_init
    row_idx = 0
    
    # Generate rows until we have enough points or exceed the square height
    while y_val <= 1 - r_init + 1e-9:
        if row_idx % 2 == 0:
            # Even rows aligned with left wall
            xs = [r_init + k * 2 * r_init for k in range(6)]
        else:
            # Odd rows shifted
            xs = [2 * r_init + k * 2 * r_init for k in range(6)]
        
        for x in xs:
            if x >= r_init - 1e-9 and x <= 1 - r_init + 1e-9:
                centers.append([x, y_val])
        
        y_val += r_init * np.sqrt(3)
        row_idx += 1
        
        if len(centers) >= 26:
            break
            
    # Select exactly 26 points
    if len(centers) > 26:
        centers = centers[:26]
    elif len(centers) < 26:
        while len(centers) < 26:
            cx = np.random.uniform(r_init, 1 - r_init)
            cy = np.random.uniform(r_init, 1 - r_init)
            centers.append([cx, cy])
            
    centers = np.array(centers[:n])
    radii = np.array([r_init] * n)
    return centers, radii

def constraints_func(vars, n=26):
    """
    Computes the vector of constraint values.
    Constraints are satisfied if returned values >= 0.
    """
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    # 1. Boundary constraints
    # x - r >= 0
    c1 = x - r
    # 1 - x - r >= 0
    c2 = 1 - x - r
    # y - r >= 0
    c3 = y - r
    # 1 - y - r >= 0
    c4 = 1 - y - r
    
    # 2. Overlap constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    dist_sq = dx**2 + dy**2
    min_dist_sq = dr**2
    violation = dist_sq - min_dist_sq
    
    # Upper triangle indices
    rows, cols = np.triu_indices(n, k=1)
    overlap_cons = violation[rows, cols]
    
    return np.concatenate([c1, c2, c3, c4, overlap_cons])

def run_packing():
    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None
    
    # Try multiple initial seeds to find a better local optimum
    seeds = [42, 123, 456, 789, 101]
    
    for seed in seeds:
        centers, radii = get_initial_guess(seed)
        x0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
        
        # Bounds for variables
        bounds = []
        for i in range(n):
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((1e-6, 0.5)) # r > 0
            
        cons = {'type': 'ineq', 'fun': lambda v: constraints_func(v, n)}
        
        res = minimize(lambda v: -np.sum(v[2*n:]), x0, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 2000, 'ftol': 1e-10})
        
        if res.success:
            x_opt = res.x[:n]
            y_opt = res.x[n:2*n]
            r_opt = res.x[2*n:]
            current_sum = np.sum(r_opt)
            
            if current_sum > best_sum_r:
                best_sum_r = current_sum
                best_centers = np.column_stack((x_opt, y_opt))
                best_radii = r_opt

    # Post-processing to strictly enforce constraints
    x_opt = best_centers[:, 0]
    y_opt = best_centers[:, 1]
    r_opt = best_radii

    for i in range(n):
        # Ensure radius is positive
        r_opt[i] = max(1e-9, r_opt[i])
        # Clip centers to be inside the circle's valid region
        x_opt[i] = np.clip(x_opt[i], r_opt[i], 1 - r_opt[i])
        y_opt[i] = np.clip(y_opt[i], r_opt[i], 1 - r_opt[i])

    # Resolve any minor overlaps by shrinking radii
    # A simple iterative fix for numerical precision
    for _ in range(10):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((x_opt[i] - x_opt[j])**2 + (y_opt[i] - y_opt[j])**2)
                sum_r = r_opt[i] + r_opt[j]
                if dist < sum_r:
                    reduction = (sum_r - dist) * 0.5
                    r_opt[i] -= reduction
                    r_opt[j] -= reduction
                    # Re-clip after reduction
                    r_opt[i] = max(1e-9, r_opt[i])
                    r_opt[j] = max(1e-9, r_opt[j])
                    x_opt[i] = np.clip(x_opt[i], r_opt[i], 1 - r_opt[i])
                    y_opt[i] = np.clip(y_opt[i], r_opt[i], 1 - r_opt[i])
                    x_opt[j] = np.clip(x_opt[j], r_opt[j], 1 - r_opt[j])
                    y_opt[j] = np.clip(y_opt[j], r_opt[j], 1 - r_opt[j])

    final_centers = np.column_stack((x_opt, y_opt))
    final_sum = np.sum(r_opt)
    
    return final_centers, r_opt, final_sum
