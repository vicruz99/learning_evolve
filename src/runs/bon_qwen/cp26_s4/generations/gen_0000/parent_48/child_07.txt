import numpy as np
from scipy.optimize import minimize
import math

def compute_validity_and_objective(params, n=26):
    """
    Helper to compute the objective (sum of radii) and validate constraints internally for optimization.
    Returns (objective_value, is_valid).
    objective_value is -sum(radii).
    """
    centers = params[:n*2].reshape(n, 2)
    radii = params[n*2:]
    
    # Penalty initialization
    penalty = 0.0
    
    # Boundary constraints
    # x - r >= 0 => r - x <= 0
    # 1 - x - r >= 0 => x + r - 1 <= 0
    # Same for y
    
    # We use a soft penalty for the optimizer, but strict check for final validation
    # However, for SLSQP, we pass constraints as functions returning >= 0
    
    # Check for NaN
    if np.isnan(centers).any() or np.isnan(radii).any():
        return np.inf, False

    # Basic radius non-negativity (handled by bounds, but check here)
    if np.any(radii < 0):
        return np.inf, False

    sum_radii = np.sum(radii)
    obj_val = -sum_radii

    # Validation checks for "is_valid" return (used for filtering results)
    is_valid = True
    
    # Boundary
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            is_valid = False
            # Add penalty
            penalty += max(0, x - r) + max(0, -(x - r) * 0) # logic check
            # Actually just rely on constraints in scipy
            break

    # Overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                is_valid = False
                break
    
    return obj_val, is_valid

def objective_function(params, n=26):
    """Objective to minimize: -sum(radii)"""
    radii = params[n*2:]
    return -np.sum(radii)

def get_constraints(n=26):
    """
    Define constraints for scipy.optimize.minimize.
    """
    constraints = []
    
    # 1. Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    # In scipy constraints, fun(x) >= 0
    # x - r >= 0
    # 1 - x - r >= 0
    
    for i in range(n):
        # Indices in params: x_i = 2*i, y_i = 2*i + 1, r_i = 2*n + i
        idx_x = 2 * i
        idx_y = 2 * i + 1
        idx_r = 2 * n + i
        
        # x - r >= 0
        def make_boundary_x(i):
            def f(params):
                return params[2*i] - params[2*n + i]
            return f
        constraints.append({'type': 'ineq', 'fun': make_boundary_x(i)})
        
        # 1 - x - r >= 0
        def make_boundary_1mx(i):
            def f(params):
                return 1.0 - params[2*i] - params[2*n + i]
            return f
        constraints.append({'type': 'ineq', 'fun': make_boundary_1mx(i)})
        
        # y - r >= 0
        def make_boundary_y(i):
            def f(params):
                return params[2*i+1] - params[2*n + i]
            return f
        constraints.append({'type': 'ineq', 'fun': make_boundary_y(i)})
        
        # 1 - y - r >= 0
        def make_boundary_1my(i):
            def f(params):
                return 1.0 - params[2*i+1] - params[2*n + i]
            return f
        constraints.append({'type': 'ineq', 'fun': make_boundary_1my(i)})

    # 2. Non-overlap constraints: dist(c_i, c_j) >= r_i + r_j
    # sqrt(dx^2 + dy^2) - r_i - r_j >= 0
    # To avoid sqrt issues with gradients, we could use squared, but inequality direction changes if negative.
    # Standard way: dist - r_i - r_j >= 0.
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_x1, idx_y1 = 2*i, 2*i + 1
            idx_x2, idx_y2 = 2*j, 2*j + 1
            idx_r1, idx_r2 = 2*n + i, 2*n + j
            
            def make_non_overlap(i, j):
                def f(params):
                    dx = params[idx_x1] - params[idx_x2]
                    dy = params[idx_y1] - params[idx_y2]
                    dist = math.sqrt(dx*dx + dy*dy)
                    return dist - (params[idx_r1] + params[idx_r2])
                return f
            constraints.append({'type': 'ineq', 'fun': make_non_overlap(i, j)})
            
    return constraints

def generate_initial_guess(n=26, seed=0):
    """
    Generate an initial guess for centers and radii.
    Strategy: 5x5 grid with one extra circle in the center gap, perturbed.
    """
    rng = np.random.RandomState(seed)
    
    # Base grid for 25 circles
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09 # Start slightly smaller than 0.1 to allow movement
    
    # 5x5 grid positions
    k = 0
    for r in range(5):
        for c in range(5):
            if k < 25:
                centers[k, 0] = 0.1 + c * 0.2 + rng.uniform(-0.02, 0.02)
                centers[k, 1] = 0.1 + r * 0.2 + rng.uniform(-0.02, 0.02)
                k += 1
                
    # 26th circle in the center
    if k < 26:
        centers[k, 0] = 0.5 + rng.uniform(-0.05, 0.05)
        centers[k, 1] = 0.5 + rng.uniform(-0.05, 0.05)
        k += 1

    # Ensure centers are within [0,1]
    centers = np.clip(centers, 0.01, 0.99)
    
    return np.concatenate([centers.flatten(), radii])

def run_packing():
    n = 26
    constraints = get_constraints(n)
    
    best_params = None
    best_obj = np.inf
    best_sum = 0.0

    # 26 circles is small enough for a few runs, but many constraints might slow it down.

    seeds = list(range(10))

        bounds.append((0.0, 0.5))

            # Optimization
            # SLSQP handles constraints well
            res = minimize(
                objective_function, 
                x0, 
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-9}

                                valid = False; break

                    best_params = res.x

    # Final validation pass to ensure no numerical drift made it invalid

    # However, to be safe against the "1e-12" tolerance in validation:
    # The optimizer constraints used 0 tolerance.
    
    return final_centers, final_radii, np.sum(final_radii)
