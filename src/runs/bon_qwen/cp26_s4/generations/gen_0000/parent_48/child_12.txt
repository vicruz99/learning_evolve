def objective_and_constraints(x_vec, n):
    """
    Prepares the objective function and constraints for scipy optimizer.
    x_vec contains [x1, y1, x2, y2, ..., x26, y26, r]
    """
    # Unpack
    centers = np.array(x_vec[:2*n]).reshape(n, 2)
    r = x_vec[2*n]
    
    # Objective: minimize -r (maximize r)
    # We return a scalar
    obj = -r
    
    # Constraints list
    cons = []
    
    # 1. Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    # Equivalent to: x - r >= 0, etc. -> r - x <= 0
    # Inequality constraint type 'ineq' means g(x) >= 0
    # So we need x - r >= 0.
    
    for i in range(n):
        xi, yi = centers[i]
        # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[2*idx] - v[2*n]})
        # 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[2*idx] - v[2*n]})
        # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[2*idx + 1] - v[2*n]})
        # 1 - y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[2*idx + 1] - v[2*n]})
        
    # 2. Non-overlap constraints: dist(i, j) >= 2r
    # dist^2 >= 4r^2
    # (xi-xj)^2 + (yi-yj)^2 - 4r^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            def make_ineq(idx_i, idx_j):
                def ineq(v):
                    xi, yi = v[2*idx_i], v[2*idx_i+1]
                    xj, yj = v[2*idx_j], v[2*idx_j+1]
                    r_val = v[2*n]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    return dist_sq - 4 * r_val**2
                return ineq
            
            cons.append({'type': 'ineq', 'fun': make_ineq(i, j)})

    return obj, cons
