def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a physics-based simulation (repulsion + inflation) to find a dense packing.
    """
    n = 26
    
    # 1. Initialization: Grid-like placement with jitter
    # We try to fit 26 circles in 6 rows. 
    # Pattern: 5, 5, 5, 5, 4, 2 = 26 circles
    rows = [5, 5, 5, 5, 4, 2]
    centers = []
    r_init = 0.05  # Start small
    
    y_spacing = 1.0 / 7.0
    for r_idx, count in enumerate(rows):
        y = (r_idx + 1) * y_spacing
        # Distribute circles in x
        for i in range(count):
            # Center circles within [r_init, 1-r_init]
            x_step = (1.0 - 2 * r_init) / (count + 1) if count > 0 else 1
            x = r_init + (i + 1) * x_step
            # Add jitter to break symmetry
            x += np.random.uniform(-0.02, 0.02)
            y += np.random.uniform(-0.01, 0.01)
            centers.append([x, y])
            
    centers = np.array(centers)
    radii = np.full(n, r_init)
    
    # 2. Simulation Parameters
    dt = 0.01
    k_repulse = 1.0  # Stiffness of repulsion
    k_inflate = 0.0005 # Rate of radius growth
    
    # 3. Optimization Loop
    for step in range(5000):
        # Gradually reduce step size for finer tuning
        current_dt = dt * max(0.01, 1.0 - step / 5000)
        
        forces = np.zeros_like(centers)
        
        # Compute pairwise interactions
        for i in range(n):
            for j in range(i + 1, n):
                dist_vec = centers[i] - centers[j]
                dist = np.linalg.norm(dist_vec)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    # Overlap detected, apply repulsion
                    force_mag = (min_dist - dist) * k_repulse
                    force_dir = dist_vec / dist
                    forces[i] += force_dir * force_mag
                    forces[j] -= force_dir * force_mag
        
        # Apply boundary constraints and repulsion
        for i in range(n):
            # Check walls
            # Left
            if centers[i, 0] - radii[i] < 0:
                forces[i, 0] += (radii[i] - centers[i, 0]) * k_repulse
            # Right
            if centers[i, 0] + radii[i] > 1:
                forces[i, 0] -= (centers[i, 0] + radii[i] - 1) * k_repulse
            # Bottom
            if centers[i, 1] - radii[i] < 0:
                forces[i, 1] += (radii[i] - centers[i, 1]) * k_repulse
            # Top
            if centers[i, 1] + radii[i] > 1:
                forces[i, 1] -= (centers[i, 1] + radii[i] - 1) * k_repulse

        # Update centers
        centers += forces * current_dt
        
        # Clamp centers to valid range [0, 1]
        centers = np.clip(centers, 0, 1)

        # The max radius is limited by walls and neighbors.
        # We increase radii by a small amount, but clamp them immediately.
        # This acts as a "try to grow" strategy.

        # This ensures validity even if the physics simulation drifts
        for i in range(n):
            # Wall constraints
            max_r_walls = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])

            max_r_neighbors = np.inf
            for j in range(n):
                if i == j: continue
                dist = np.linalg.norm(centers[i] - centers[j])
                # dist >= r_i + r_j  =>  r_i <= dist - r_j
                max_r_neighbors = min(max_r_neighbors, dist - radii[j])
            
            # Take the tightest constraint
            allowed_r = min(max_r_walls, max_r_neighbors)

            # During simulation, we can be slightly loose to help escape local minima,
            # but here we clamp strictly to ensure validity.
            radii[i] = min(radii[i], allowed_r + 1e-5)
            radii[i] = max(radii[i], 0.0)

    # Final cleanup:

    # We perform a final strict clamp to ensure 100% validity.
