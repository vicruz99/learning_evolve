

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii, targeting 2.636.

<strategy>
1. **Initial Configuration**: Use a hexagonal grid pattern (densest packing for equal circles) to get a strong starting point for 26 circles.

2. **Penalty-based Optimization**: Use scipy's L-BFGS-B optimizer with a penalty function that:
   - Maximizes sum of radii (the objective)
   - Penalizes circle-circle overlaps
   - Penalizes boundary violations

3. **Multiple Random Restarts**: Run the optimizer from many different perturbed starting points to escape local minima.

4. **Radius Expansion Phase**: After optimization, iteratively expand each circle's radius while maintaining feasibility.

5. **Second Round of Optimization**: After radius expansion, run another optimization to find better positions.

6. **Key Insight**: The hexagonal arrangement is optimal for equal circles. For unequal circles, we want to position circles well first, then expand them optimally. The penalty approach allows continuous optimization while handling all constraints simultaneously.

7. **High Penalty Weight**: Use very large penalty weights (1e6) to ensure solutions remain feasible.
</strategy>

import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26
    
    # Phase 1: Create initial hexagonal configuration
    r_start = 0.09
    x_space = r_start * np.sqrt(3)
    y_space = r_start * 1.5
    
    centers = []
    idx = 0
    for row in range(7):
        for col in range(8):
            if idx >= n:
                break
            x = col * x_space + (x_space / 2 if row % 2 == 1 else 0)
            y = row * y_space
            if x + r_start <= 1 and y + r_start <= 1:
                centers.append((x, y))
                idx += 1
    
    while len(centers) < n:
        centers.append((0.5, 0.5))
    
    centers = np.array(centers)
    radii_init = np.full(n, r_start)
    
    def compute_overlap_penalty(centers_arr, radii_arr, n):
        penalty = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_arr[i, 0] - centers_arr[j, 0]
                dy = centers_arr[i, 1] - centers_arr[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                sum_r = radii_arr[i] + radii_arr[j]
                if dist < sum_r:
                    penalty += (sum_r - dist) ** 2
        return penalty
    
    def compute_boundary_penalty(centers_arr, radii_arr, n):
        penalty = 0.0
        for i in range(n):
            x_i, y_i = centers_arr[i]
            r_i = radii_arr[i]
            if x_i - r_i < 0:
                penalty += (x_i - r_i) ** 2
            if x_i + r_i > 1:
                penalty += (x_i + r_i - 1) ** 2
            if y_i - r_i < 0:
                penalty += (y_i - r_i) ** 2
            if y_i + r_i > 1:
                penalty += (y_i + r_i - 1) ** 2
        return penalty
    
    def objective(x, n_circles):
        centers_opt = x[:2 * n_circles].reshape(n_circles, 2)
        radii_opt = x[2 * n_circles:]
        
        value = -np.sum(radii_opt)
        
        penalty = 0.0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                sum_r = radii_opt[i] + radii_opt[j]
                if dist < sum_r:
                    penalty += (sum_r - dist) ** 2
        
        for i in range(n_circles):
            x_i, y_i = centers_opt[i]
            r_i = radii_opt[i]
            if x_i - r_i < 0:
                penalty += (x_i - r_i) ** 2
            if x_i + r_i > 1:
                penalty += (x_i + r_i - 1) ** 2
            if y_i - r_i < 0:
                penalty += (y_i - r_i) ** 2
            if y_i + r_i > 1:
                penalty += (y_i + r_i - 1) ** 2
        
        return value + 1e6 * penalty
    
    best_sum_radii = -1
    best_centers = centers.copy()
    best_radii = radii_init.copy()
    
    # Multiple random restarts
    num_restarts = 30
    perturbation_scale = 0.03
    
    for restart in range(num_restarts):
        x0 = np.concatenate([centers.ravel(), radii_init])
        
        bounds = []
        for i in range(2 * n):
            bounds.append((0, 1))
        for i in range(n):
            bounds.append((0.01, 0.5))
        
        rng = np.random.RandomState(restart * 42 + 7)
        x_perturbed = x0 + rng.uniform(-perturbation_scale, perturbation_scale, size=3 * n)
        x_perturbed[:2 * n] = np.clip(x_perturbed[:2 * n], 0, 1)
        x_perturbed[2 * n:] = np.clip(x_perturbed[2 * n:], 0.01, 0.5)
        
        result = minimize(objective, x_perturbed, method='L-BFGS-B', 
                         bounds=bounds, args=(n,), 
                         options={'maxiter': 500, 'ftol': 1e-15})
        
        current_centers = result.x[:2 * n].reshape(n, 2)
        current_radii = result.x[2 * n:]
        
        overlap_pen = compute_overlap_penalty(current_centers, current_radii, n)
        boundary_pen = compute_boundary_penalty(current_centers, current_radii, n)
        total_penalty = overlap_pen + boundary_pen
        
        if total_penalty < 1e-6:
            current_sum = np.sum(current_radii)
            if current_sum > best_sum_radii:
                best_sum_radii = current_sum
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
    
    # Phase 2: Radius expansion - expand each circle as much as possible
    centers_final = best_centers.copy()
    radii_final = best_radii.copy()
    
    def expand_radii(centers_exp, radii_exp, num_circles, max_iterations=50):
        for iteration in range(max_iterations):
            improved = False
            for i in range(num_circles):
                x_i, y_i = centers_exp[i]
                
                max_r = min(x_i, 1 - x_i, y_i, 1 - y_i)
                for j in range(num_circles):
                    if i != j:
                        dx = centers_exp[i, 0] - centers_exp[j, 0]
                        dy = centers_exp[i, 1] - centers_exp[j, 1]
                        dist = np.sqrt(dx * dx + dy * dy)
                        max_r = min(max_r, dist - radii_exp[j])
                
                max_r = max(0.001, max_r)
                if max_r > radii_exp[i] + 1e-12:
                    radii_exp[i] = max_r
                    improved = True
            
            if not improved:
                break
        return radii_exp
    
    radii_final = expand_radii(centers_final, radii_final, n)
    
    # Phase 3: Second round of optimization with expanded radii
    best_sum_radii_2 = np.sum(radii_final)
    best_centers_2 = centers_final.copy()
    best_radii_2 = radii_final.copy()
    
    for restart in range(20):
        x0 = np.concatenate([centers_final.ravel(), radii_final])
        
        bounds = []
        for i in range(2 * n):
            bounds.append((0, 1))
        for i in range(n):
            bounds.append((0.01, 0.5))
        
        rng = np.random.RandomState(restart * 99 + 13)
        x_perturbed = x0 + rng.uniform(-0.01, 0.01, size=3 * n)
        x_perturbed[:2 * n] = np.clip(x_perturbed[:2 * n], 0, 1)
        x_perturbed[2 * n:] = np.clip(x_perturbed[2 * n:], 0.01, 0.5)
        
        result = minimize(objective, x_perturbed, method='L-BFGS-B',
                         bounds=bounds, args=(n,),
                         options={'maxiter': 500, 'ftol': 1e-15})
        
        current_centers = result.x[:2 * n].reshape(n, 2)
        current_radii = result.x[2 * n:]
        
        overlap_pen = compute_overlap_penalty(current_centers, current_radii, n)
        boundary_pen = compute_boundary_penalty(current_centers, current_radii, n)
        total_penalty = overlap_pen + boundary_pen
        
        if total_penalty < 1e-6:
            current_sum = np.sum(current_radii)
            if current_sum > best_sum_radii_2:
                best_sum_radii_2 = current_sum
                best_centers_2 = current_centers.copy()
                best_radii_2 = current_radii.copy()
    
    # Final radius expansion
    best_radii_2 = expand_radii(best_centers_2, best_radii_2, n)
    
    # Phase 4: Try hexagonal grid with different densities
    for row_density in [0.08, 0.085, 0.09, 0.095, 0.10]:
        for col_density in [0.14, 0.15, 0.16, 0.17]:
            test_centers = []
            test_radii = []
            idx = 0
            for row in range(8):
                for col in range(8):
                    if idx >= n:
                        break
                    x = col * col_density + (col_density / 2 if row % 2 == 1 else 0)
                    y = row * row_density
                    if x + row_density <= 1 and y + row_density <= 1:
                        test_centers.append([x, y])
                        test_radii.append(row_density)
                        idx += 1
            
            while len(test_centers) < n:
                test_centers.append([0.5, 0.5])
                test_radii.append(0.05)
            
            test_centers = np.array(test_centers)
            test_radii = np.array(test_radii)
            
            # Expand radii for this configuration
            test_radii = expand_radii(test_centers, test_radii, n)
            test_sum = np.sum(test_radii)
            
            if test_sum > best_sum_radii_2:
                best_sum_radii_2 = test_sum
                best_centers_2 = test_centers.copy()
                best_radii_2 = test_radii.copy()
                
                # Optimize this promising configuration
                x0 = np.concatenate([test_centers.ravel(), test_radii])
                bounds = []
                for i in range(2 * n):
                    bounds.append((0, 1))
                for i in range(n):
                    bounds.append((0.01, 0.5))
                
                result = minimize(objective, x0, method='L-BFGS-B',
                                 bounds=bounds, args=(n,),
                                 options={'maxiter': 500, 'ftol': 1e-15})
                
                current_centers = result.x[:2 * n].reshape(n, 2)
                current_radii = result.x[2 * n:]
                
                overlap_pen = compute_overlap_penalty(current_centers, current_radii, n)
                boundary_pen = compute_boundary_penalty(current_centers, current_radii, n)
                if overlap_pen + boundary_pen < 1e-6:
                    current_radii = expand_radii(current_centers, current_radii, n)
                    current_sum = np.sum(current_radii)
                    if current_sum > best_sum_radii_2:
                        best_sum_radii_2 = current_sum
                        best_centers_2 = current_centers.copy()
                        best_radii_2 = current_radii.copy()
    
    centers_final = best_centers_2
    radii_final = best_radii_2
    sum_radii = np.sum(radii_final)
    
    return centers_final, radii_final, sum_radii
