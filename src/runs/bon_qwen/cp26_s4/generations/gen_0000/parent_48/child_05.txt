import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    for seed in range(80):
        np.random.seed(seed)
        centers0, radii0 = generate_start(n, seed)
        x0 = np.concatenate([centers0.ravel(), radii0])

        bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.4)] * n

        result = minimize(
            obj, x0, args=(n,),
            bounds=bounds,
            method="L-BFGS-B",
            options={"maxiter": 30000, "ftol": 1e-20}
        )

        co = result.x[:2 * n].reshape((n, 2))
        ro = result.x[2 * n:]

        if check_valid(co, ro, n):
            s = float(np.sum(ro))
            if s > best_sum:
                best_sum = s
                best_centers = co.copy()
                best_radii = ro.copy()

    if best_centers is None:
        best_centers, best_radii = make_hex(n)
        best_sum = float(np.sum(best_radii))

    return best_centers, best_radii, best_sum


def obj(x, n):
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]

    value = -np.sum(radii)
    pen = 0.0
    C = 2000.0

    for i in range(n):
        xi = centers[i, 0]
        yi = centers[i, 1]
        ri = radii[i]

        if xi - ri < 0:
            pen += C * (xi - ri) ** 2
        if xi + ri > 1:
            pen += C * (xi + ri - 1) ** 2
        if yi - ri < 0:
            pen += C * (yi - ri) ** 2
        if yi + ri > 1:
            pen += C * (yi + ri - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy + 1e-18)
            md = radii[i] + radii[j]
            if d < md:
                pen += C * (md - d) ** 2

    return value + pen


def check_valid(centers, radii, n):
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        if r < 0 or np.isnan(r):
            return False
        if x - r < -1e-9 or x + r > 1 + 1e-9:
            return False
        if y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            if d < radii[i] + radii[j] - 1e-9:
                return False
    return True


def generate_start(n, seed):
    style = seed % 5
    if style == 0:
        return make_hex(n)
    elif style == 1:
        return make_hex_perturbed(n, seed)
    elif style == 2:
        return make_hex_wide(n)
    elif style == 3:
        return make_corners(n)
    else:
        return make_grid_pert(n)


def make_hex(n):
    centers = []
    idx = 0
    row = 0
    while idx < n:
        offset = (row % 2) * 0.5
        cols = 8
        for col in range(cols):
            if idx >= n:
                break
            x = offset + col
            y = row * np.sqrt(3.0) / 2.0
            centers.append([x, y])
            idx += 1
        row += 1

    centers = np.array(centers[:n])
    lo = centers.min(axis=0)
    hi = centers.max(axis=0)
    sz = hi - lo
    centers = (centers - lo) / (sz + 1e-12) * 0.85 + 0.075

    radii = np.full(n, 0.095)
    return centers, radii


def make_hex_perturbed(n, seed):
    c, r = make_hex(n)
    np.random.seed(seed)
    c = c + np.random.randn(n, 2) * 0.015
    c = np.clip(c, 0.08, 0.92)
    return c, r

def make_corners(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    centers[0] = [0.15, 0.15]
    centers[1] = [0.85, 0.15]
    centers[2] = [0.15, 0.85]
    centers[3] = [0.85, 0.85]
    radii[0:4] = 0.15

    idx = 4
    remaining = n - 4
    gs = int(np.ceil(np.sqrt(remaining)))
    for i in range(gs):
        for j in range(gs):
            if idx >= n:
                break
            centers[idx, 0] = 0.25 + j * 0.15
            centers[idx, 1] = 0.25 + i * 0.15
            radii[idx] = 0.07
            idx += 1

    return centers, radii


def make_grid_pert(n):
    gs = int(np.ceil(np.sqrt(n)))
    centers = np.zeros((n, 2))
    for i in range(n):
        r = i // gs
        c = i % gs
        centers[i, 0] = (c + 0.5) / gs * 0.85 + 0.075
        centers[i, 1] = (r + 0.5) / gs * 0.85 + 0.075

    np.random.seed(100)
    centers += np.random.randn(n, 2) * 0.02
    centers = np.clip(centers, 0.05, 0.95)
    radii = np.full(n, 0.095)
    return centers, radii
