        min_dist = 1.0

        for i in range(n):
            x, y = centers[i]

                dy = centers[i, 1] - centers[j, 1]
                d = math.sqrt(dx*dx + dy*dy)
                if d < min_dist:

        # We want to maximize min_dist, so we minimize -min_dist
        return -min_dist
