from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    def get_initial_guess(strategy, seed=42):

        centers = np.zeros((n, 2))
        radii = np.full(n, 0.05) # Start with small radii
        
        if strategy == 'grid':
            # 5x5 grid plus one extra
            # Grid spacing 0.2, centers at 0.1, 0.3, 0.5, 0.7, 0.9
            xs = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
            ys = np.array([0.1, 0.3, 0.5, 0.7, 0.9])

            for x in xs:
                for y in ys:
                    if idx < n:
                        centers[idx] = [x, y]
                        idx += 1
            if idx < n:
                # Add extra circle in a gap, e.g., (0.2, 0.2)
                centers[idx] = [0.2, 0.2]
        elif strategy == 'hex':
            # Hexagonal packing approximation
            # Try to fit rows
            # Estimate radius for hex packing of 26 circles ~ 0.095
            r_est = 0.09
            dy = np.sqrt(3) * r_est
            row_idx = 0
            y_curr = r_est
            while True:
                if row_idx % 2 == 0:
                    # Even row: starts at r_est
                    # Max x is 1 - r_est
                    # centers at r_est + k*2*r_est
                    xs = np.arange(r_est, 1.0 + 1e-9, 2 * r_est)
                else:
                    # Odd row: shifted by r_est

                    break

                        row_idx += 1

                        break
                if row_idx >= n:
                    break
                y_curr += dy
                if y_curr + r_est > 1.0:
                    # Try to squeeze in if needed, or just stop
                    # For initialization, valid placement is preferred.
                    # If we can't place all, fall back to random?
                    # But hex logic above might fail if r_est is too big.

                    break

                row_idx += 1

        # vars is flattened [x1, y1, r1, x2, y2, r2, ...]

        return -np.sum(rs)

            })
            # 1 - x - r >= 0 => 1 - x >= r
            constraints.append({
                'type': 'ineq',
                'fun': lambda vars, idx=i: 1.0 - vars[3*idx] - vars[3*idx+2]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda vars, idx=i: vars[3*idx+1] - vars[3*idx+2]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda vars, idx=i: 1.0 - vars[3*idx+1] - vars[3*idx+2]
            })

        for i in range(n):

                constraints.append({

                    'fun': lambda vars, i=i, j=j: \

                })
        return constraints
