import numpy as np

def compute_forces(centers, radii, n, k_pair=300.0, k_bound=300.0):
    """Compute repulsive forces for overlaps and boundary violations."""
    forces = np.zeros_like(centers)
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            if dist < 1e-8:
                diff = np.random.uniform(-0.01, 0.01, 2)
                dist = np.linalg.norm(diff)
            
            overlap = (radii[i] + radii[j]) - dist
            if overlap > 0:
                f_mag = k_pair * overlap
                dir_vec = diff / dist
                forces[i] += dir_vec * f_mag
                forces[j] -= dir_vec * f_mag
                
        x, y = centers[i]
        r = radii[i]
        if x < r:
            forces[i][0] += k_bound * (r - x)
        elif x > 1 - r:
            forces[i][0] -= k_bound * (x - (1 - r))
        if y < r:
            forces[i][1] += k_bound * (r - y)
        elif y > 1 - r:
            forces[i][1] -= k_bound * (y - (1 - r))
    return forces
