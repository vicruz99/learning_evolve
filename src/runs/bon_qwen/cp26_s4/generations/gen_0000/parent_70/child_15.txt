import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # 1. Initialize parameters for a staggered layout (5, 6, 5, 6, 4 circles per row)
    n_circles = 26
    row_counts = [5, 6, 5, 6, 4]
    base_radius = 0.098
    
    # 2. Construct the initial centers and radii arrays
    centers = np.zeros((n_circles, 2))
    radii = np.full(n_circles, base_radius)
    
    # 3. Place circles in a staggered grid pattern
    idx = 0
    # y-coordinates for the 5 rows, placed near boundaries
    y_positions = [base_radius, 0.298, 0.5, 0.702, 1.0 - base_radius]
    
    for i, count in enumerate(row_counts):
        # x-coordinates: centered or aligned to edges based on row size
        if count == 6:
            # 6 circles: spans most of the width
            x_start = base_radius + 0.001 
            x_positions = np.linspace(x_start, 1.0 - x_start, count)
        else:
            # 5 or 4 circles: centered horizontally
            if count == 5:
                x_positions = np.linspace(base_radius + 0.002, 1.0 - base_radius - 0.002, count)
            else: # count == 4
                x_positions = np.linspace(base_radius + 0.02, 1.0 - base_radius - 0.02, count)
        
        for j in range(count):
            centers[idx] = [x_positions[j], y_positions[i]]
            idx += 1

    # 4. Numerical Optimization to maximize sum of radii
    # Iteratively scale up radii based on the tightest constraints
    max_iterations = 2000
    convergence_threshold = 1e-10
    
    for _ in range(max_iterations):
        min_slack = float('inf')
        n = len(radii)
        
        # Check all pairwise distances and boundary distances
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            
            # Boundary constraints: distance to edges
            slack_x1 = xi - ri
            slack_x2 = (1.0 - xi) - ri
            slack_y1 = yi - ri
            slack_y2 = (1.0 - yi) - ri
            
            for slack in [slack_x1, slack_x2, slack_y1, slack_y2]:
                if slack < min_slack:
                    min_slack = slack

            for j in range(i + 1, n):
                xj, yj = centers[j]
                rj = radii[j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                slack = dist - (ri + rj)
                if slack < min_slack:
                    min_slack = slack

        # If minimum slack is very small, we've reached the limit
        if min_slack < convergence_threshold:
            break
            
        # Update radii: increase them slightly proportional to the slack
        # A dampening factor of 0.5 prevents oscillation
        growth_factor = 1.0 + (min_slack * 0.5)
        radii *= growth_factor

    # 5. Calculate final sum of radii and return
    total_sum = np.sum(radii)
    
    return centers, radii, total_sum
