

<strategy>
To maximize the sum of radii for 26 circles in a unit square, the solution must balance high packing density with efficient use of boundary space. A standard square grid (5x5 + 1) yields a sum of approximately 2.5, which is suboptimal. A hexagonal (honeycomb) lattice is denser and should be used as the primary layout. 

My strategy involves:
1.  **Initialization:** Generating a dense initial configuration using a perturbed hexagonal grid. This starts with larger radii, providing the optimizer with a much better "head start" than random initialization.
2.  **Global Exploration:** Running the local optimization from multiple different starting seeds (perturbations of the grid) to escape local minima.
3.  **Local Optimization:** Using `scipy.optimize.minimize` with the 'SLSQP' method. This solver handles non-linear inequality constraints (non-overlap) effectively.
4.  **Vectorization:** The constraints and objective function are vectorized using NumPy to ensure fast evaluation of the 325 pairwise distance checks.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    def objective(x):
        # x layout: x_coords, y_coords, radii
        radii = x[2 * n:]
        # Maximize sum of radii (minimize negative sum)
        return -np.sum(radii)

    def constraint(x):
        # Returns array of values that must be >= 0
        # Constraint: dist^2 >= (r_i + r_j)^2
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]

        # Vectorized pair calculation
        # c_diff: (n, n, 2)
        c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(c_diff ** 2, axis=2)

        # r_sum: (n, n)
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        # Constraint matrix
        cons = dist_sq - r_sum ** 2
        
        # Extract upper triangular part to avoid duplicates and self-check
        # mask where j > i
        triu_idx = np.triu_indices(n, k=1)
        return cons[triu_idx]

    # Bounds: [0, 1] for x, y; [0, 0.5] for radii
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n

    # Initial seed generation: Hexagonal Grid
    def generate_hex_grid(n_circles):
        centers = np.zeros((n_circles, 2))
        idx = 0
        # Estimate row length
        row_len = int(np.ceil(np.sqrt(n_circles)))
        row_idx = 0
        
        while idx < n_circles:
            y = 0.05 + row_idx * 0.17  # Vertical spacing approx sqrt(3)/2 * diameter
            x_start = 0.05
            if row_idx % 2 == 1:
                x_start = 0.15  # Shift for hex pattern
                
            for col in range(row_len):
                if idx >= n_circles:
                    break
                x = x_start + col * 0.2
                if x > 0.95:
                    break
                centers[idx] = [x, y]
                idx += 1
            row_idx += 1
        return centers

    # Run multiple optimizations with different initializations
    for _ in range(20): # 20 restarts for robustness
        # 1. Start with a dense grid layout
        init_centers = generate_hex_grid(n)
        
        # 2. Add small random perturbations to break symmetry
        init_centers += np.random.uniform(-0.005, 0.005, size=(n, 2))
        
        # 3. Clip to valid square
        init_centers = np.clip(init_centers, 0, 1)
        
        # 4. Set initial radii (small to ensure feasibility)
        # Calculate distance to nearest neighbor to set a safe starting radius
        dists = np.sqrt(((init_centers[:, None, :] - init_centers[None, :, :]) ** 2).sum(axis=2))
        # Mask diagonal (distance to self)
        dists_flat = dists.flatten()
        dists_flat[np.arange(n) * n + np.arange(n)] = np.inf
        min_dists = dists_flat.reshape(n, n).min(axis=1)
        init_radii = min_dists * 0.3 # Start at 30% of min distance to allow room for growth
        
        x0 = np.concatenate([init_centers.flatten(), init_radii])
        
        # Optimization
        result = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'ineq', 'fun': constraint}, 
            options={'maxiter': 2000, 'ftol': 1e-9}
        )

        if not result.success:
            continue

        current_sum = -result.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = result.x[:2 * n].reshape(n, 2)
            best_radii = result.x[2 * n:]

    return best_centers, best_radii, best_sum
```
