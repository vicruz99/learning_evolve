from scipy.optimize import minimize, differential_evolution

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # This provides a denser packing than a square grid.
    n = 26
    centers_init = []
    row, col = 0, 0
    while len(centers_init) < n:
        x = 0.1 + col * 0.2 + (0.1 if row % 2 == 1 else 0)
        y = 0.1 + row * 0.173205 # sqrt(3)/2 * 0.2
        if x <= 0.9:
            centers_init.append([x, y])
            col += 1
        else:
            row += 1
            col = 0

    radii_init = np.full(n, 0.05)

    def objective(state):
        n_circles = state.shape[0] // 3
        centers = state[:2 * n_circles].reshape(n_circles, 2)
        radii = state[2 * n_circles:]
        
        # 1. Maximize sum of radii (minimize negative sum)
        score = -np.sum(radii)
        
        # 2. Penalty for negative radii
        score += 1000.0 * np.sum(np.maximum(0, -radii))
        
        # 3. Penalty for boundary violations (center - r < 0 or center + r > 1)
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            # Bottom/Left boundary
            score += 1000.0 * np.maximum(0, -(x - r))
            score += 1000.0 * np.maximum(0, -(y - r))
            # Top/Right boundary
            score += 1000.0 * np.maximum(0, x + r - 1.0)
            score += 1000.0 * np.maximum(0, y + r - 1.0)
            
        # 4. Penalty for overlaps (dist < r1 + r2)
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:

        return score

    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    de_result = differential_evolution(
        objective, 
        bounds, 
        seed=42, 
        maxiter=500, 
        popsize=30, 
        mutation=(0.5, 1.5), 
        recombination=0.9,
        tol=1e-6
    )

        objective, 
        de_result.x,

    # 6. Clamp radii to be non-negative and ensure no NaN
    radii_final = np.clip(radii_final, 0, None)
    
    # 7. Calculate sum of radii
    sum_radii = float(np.sum(radii_final))
    
    return centers_final, radii_final, sum_radii
