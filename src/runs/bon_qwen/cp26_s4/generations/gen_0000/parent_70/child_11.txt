import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None

    # Define the objective: minimize negative sum of radii
    def objective(x):
        r = x[2::3]
        return -np.sum(r)

    # Define constraints
    constraints = []
    
    # Boundary constraints: x_i - r_i >= 0, x_i + r_i <= 1, same for y
    for i in range(n_circles):
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+2]})
        # 1 - (x + r) >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - (x[3*i] + x[3*i+2])})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]})
        # 1 - (y + r) >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - (x[3*i+1] + x[3*i+2])})

    # Separation constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: 
                    (x[3*i] - x[3*j])**2 + (x[3*i+1] - x[3*j+1])**2 - (x[3*i+2] + x[3*j+2])**2
            })

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r

    # Function to generate initial hexagonal-like configurations
    def get_initial_guess(seed=0):
        np.random.seed(seed)
        centers = []
        radii = []

        y_curr = r_init
        for r_count in rows:
            row_centers_x = []

            span = (r_count - 1) * 2 * r_init
            margin = (1.0 - span) / 2.0

                cx = margin + k * 2 * r_init

                cy += np.random.uniform(-0.005, 0.005)

                row_centers_x.append(cx)
            
            centers.extend([(cx, y_curr) for cx in row_centers_x])
            radii.extend([r_init] * r_count)
            y_curr += r_init * np.sqrt(3) # Hexagonal row spacing
        
        # Flatten to optimization vector: x1, y1, r1, x2, y2, r2...
        x0 = []
        for c, r in zip(centers, radii):
            x0.extend([c[0], c[1], r])
            
        return np.array(x0)

            res = minimize(
                objective, 
                x0, 
                method='SLSQP',

            )

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    import numpy as np # Ensure numpy is available inside
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
