import numpy as np
from scipy.optimize import minimize
import math


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = -1.0

    for restart in range(40):
        np.random.seed(restart * 137 + 42)

        if restart < 20:
            x0 = init_hexagonal_flat(n)
        else:
            x0 = init_grid_flat(n)

        for i in range(n):
            x0[3*i] += np.random.uniform(-0.03, 0.03)
            x0[3*i + 1] += np.random.uniform(-0.03, 0.03)
            x0[3*i + 2] += np.random.uniform(-0.015, 0.015)

            x0[3*i] = np.clip(x0[3*i], 0.03, 0.97)
            x0[3*i + 1] = np.clip(x0[3*i + 1], 0.03, 0.97)
            x0[3*i + 2] = np.clip(x0[3*i + 2], 0.01, 0.25)
        
        # Bounds for optimization
        bounds = []
        for i in range(n):
            bounds.append((0.001, 0.999))
            bounds.append((0.001, 0.999))

        result = minimize(

            args=(n, 1e9),

def init_hexagonal_flat(n):
    """Initialize with hexagonal packing pattern."""
    x = np.zeros(3 * n)
    
    r = 0.088
    h = r * math.sqrt(3)
    
    row = 0
    col = 0
    idx = 0
    for i in range(n):
        x_val = r + col * 2 * r
        y_val = r + row * h
        if row % 2 == 1:
            x_val += r
        
        x[idx] = x_val
        x[idx + 1] = y_val
        x[idx + 2] = r
        idx += 3
        
        col += 1
        # Alternate row lengths for hexagonal packing
        max_cols = 5 if row % 2 == 0 else 4
        if col >= max_cols:
            col = 0
            row += 1
    
    return x


def init_grid_flat(n):
    """Initialize with grid pattern."""
    x = np.zeros(3 * n)
    
    r = 0.085
    rows = 5
    cols = 6
    
    count = 0
    for i in range(rows):
        for j in range(cols):
            if count < n:
                x[3*count] = r + j * 2 * r * 0.95
                x[3*count + 1] = r + i * 2 * r * 0.95
                x[3*count + 2] = r
                count += 1
    
    return x


def penalized_objective(x, n, penalty_weight=1e9):
    """Objective: maximize sum of radii with penalties for constraint violations."""
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    obj = -np.sum(radii)  # Negative because we minimize
    penalty = 0.0
    
    # Boundary penalties (vectorized)
    x_coords = centers[:, 0]
    y_coords = centers[:, 1]
    
    # x - r >= 0
    penalty += penalty_weight * np.sum(np.maximum(0.0, -(x_coords - radii)) ** 2)
    # x + r <= 1
    penalty += penalty_weight * np.sum(np.maximum(0.0, x_coords + radii - 1.0) ** 2)
    # y - r >= 0
    penalty += penalty_weight * np.sum(np.maximum(0.0, -(y_coords - radii)) ** 2)
    # y + r <= 1
    penalty += penalty_weight * np.sum(np.maximum(0.0, y_coords + radii - 1.0) ** 2)
    # r >= 0
    penalty += penalty_weight * np.sum(np.maximum(0.0, -radii) ** 2)
    
    # Overlap penalties (partially vectorized)
    for i in range(n):
        dx = centers[i, 0] - centers[:, 0]
        dy = centers[i, 1] - centers[:, 1]
        dist = np.sqrt(dx * dx + dy * dy)
        overlap = radii[i] + radii - dist
        overlap = np.maximum(0.0, overlap)
        # Only j > i to avoid double counting
        penalty += penalty_weight * np.sum(overlap[i + 1:] ** 2)
    
    return obj + penalty


def check_validity(centers, radii, n):
    """Check if the packing is valid (no overlaps, all within bounds)."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False
        if r < 0:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True
