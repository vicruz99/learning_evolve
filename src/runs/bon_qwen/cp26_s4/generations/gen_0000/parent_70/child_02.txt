import numpy as np
from scipy.optimize import differential_evolution

def compute_radii(centers):
    """
    Computes the maximum possible radius for each circle given a configuration of centers.
    Radii are constrained by the unit square boundaries and non-overlap with other circles.
    """
    n = centers.shape[0]
    
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    x, y = centers[:, 0], centers[:, 1]
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Distance to nearest neighbor / 2
    # Efficient pairwise distance computation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_pairs = np.min(dists, axis=1) / 2.0
    
    # Final radius is the minimum of boundary constraint and neighbor constraint
    return np.minimum(r_bound, r_pairs)

def objective(centers_flat):
    """
    Objective function to minimize: negative sum of radii.
    Centers are clipped to [0,1] to maintain feasibility during optimization steps.
    """
    centers = centers_flat.reshape(26, 2)
    centers = np.clip(centers, 0.0, 1.0)
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing():
    """
    Optimizes circle positions to maximize the sum of radii using Differential Evolution.
    Returns centers, radii, and the sum of radii.
    """
    # 26 circles * 2 coordinates = 52 dimensions
    bounds = [(0.0, 1.0)] * 52
    
    # Global optimization with differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        seed=42, 
        maxiter=800, 
        popsize=20, 
        tol=1e-8,
        atol=1e-8,
        polish=True
    )
    
    centers = result.x.reshape(26, 2)
    centers = np.clip(centers, 0.0, 1.0) # Final boundary enforcement
    
    radii = compute_radii(centers)
    total_sum = float(np.sum(radii))
    
    return centers, radii, total_sum
