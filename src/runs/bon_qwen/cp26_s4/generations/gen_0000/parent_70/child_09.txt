def hexagonal_init(n, r=0.09):
    """Initialize circles in various hexagonal patterns"""
    centers_list = []
    
    # Various row patterns for 26 circles
    row_patterns = [
        [5, 5, 5, 5, 6],
        [6, 5, 5, 5, 5],
        [5, 6, 5, 5, 5],

        [6, 4, 5, 5, 6],
        [5, 6, 5, 4, 6],
    ]
    
    for row_lengths in row_patterns:
        total = sum(row_lengths)

            continue
        centers = np.zeros((n, 2))
        idx = 0
        for row, row_len in enumerate(row_lengths):

        max_ext = max(max_x, max_y)
        scale = 1.0 / max_ext
        centers *= scale
        centers_list.append(centers)

def grid_init(n):

def random_init(n, seed=42):
    """Initialize circles randomly with jitter"""
    rng = np.random.RandomState(seed)
    centers = rng.rand(n, 2) * 0.8 + 0.1
    # Add jitter
    centers += rng.randn(n, 2) * 0.05
    centers = np.clip(centers, 0.05, 0.95)
    return [centers]


def force_directed_relax(centers, radii, n, steps=500, step_size=0.01):
    """Force-directed relaxation to resolve overlaps"""
    centers = centers.copy()
    radii = radii.copy()
    
    for step in range(steps):
        forces = np.zeros((n, 2))
        current_step = step_size * (1.0 - step / steps)
        
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            
            # Boundary repulsion
            if xi - ri < 0:
                forces[i, 0] += (ri - xi) * 50
            if xi + ri > 1:
                forces[i, 0] -= (xi + ri - 1) * 50
            if yi - ri < 0:
                forces[i, 1] += (ri - yi) * 50
            if yi + ri > 1:
                forces[i, 1] -= (yi + ri - 1) * 50
            
            # Circle repulsion
            for j in range(i + 1, n):
                dx = centers[j, 0] - xi
                dy = centers[j, 1] - yi
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = ri + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force = overlap / dist * 20
                    fx = force * dx
                    fy = force * dy
                    forces[i] -= np.array([fx, fy])
                    forces[j] += np.array([fx, fy])
        
        # Apply forces with damping
        max_force = np.max(np.abs(forces))
        if max_force > 1e-10:
            centers += current_step * forces / max(max_force, 1)
        
        # Clamp to valid range
        centers = np.clip(centers, radii[:, np.newaxis], 1 - radii[:, np.newaxis])
    
    return centers


def objective_function(params, n, penalty_weight=1000.0):
    """Objective: maximize sum of radii with penalty constraints"""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    # Ensure non-negative radii
    radii = np.maximum(radii, 1e-10)
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary constraint penalties (smooth)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left
        v = r - x
        if v > 0:
            penalty += penalty_weight * v * v
        # Bottom
        v = r - y
        if v > 0:
            penalty += penalty_weight * v * v
        # Right
        v = x + r - 1
        if v > 0:
            penalty += penalty_weight * v * v
        # Top
        v = y + r - 1
        if v > 0:
            penalty += penalty_weight * v * v
    
    # Pairwise non-overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq)
            
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            
            if overlap > 0:
                penalty += penalty_weight * overlap * overlap
    
    return obj + penalty


def optimize_from_init(centers_init, radii_init, n, penalty_weight=1000.0):

def optimize_equal_radii(centers_init, n, target_r=0.1):

def is_valid_packing(centers, radii, tol=1e-10):
    """Check if packing is valid"""
    n = centers.shape[0]
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -tol:
            return False
        if x - r < -tol or x + r > 1 + tol or y - r < -tol or y + r > 1 + tol:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + 
                          (centers[i, 1] - centers[j, 1]) ** 2)
            if dist < radii[i] + radii[j] - tol:
                return False
    
    return True
