from scipy.optimize import minimize

    radii = []

            centers.append([x, y])
            radii.append(r_init)
            count += 1
            if count == 26:
                break
        if count == 26:
            break
            
    centers = np.array(centers)
    radii = np.array(radii)

    def objective(vars):
        # vars contains 26 radii followed by 26*2 coordinates (x, y)
        # Total 26 + 52 = 78 variables
        radii_opt = vars[:n]
        centers_opt = vars[n:].reshape((n, 2))

        penalty = np.sum(np.maximum(0, -radii_opt)**2) * 1000

        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]
            # x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
            if x - r < 0: penalty += (x - r)**2 * 1000

    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
