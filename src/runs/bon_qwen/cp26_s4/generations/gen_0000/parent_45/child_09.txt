import numpy as np
from scipy.optimize import minimize
import math

    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Try multiple initial configurations
    row_patterns = [
        [6, 5, 5, 5, 5],
        [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5],
        [5, 5, 5, 5, 6],
        [7, 6, 6, 7],
        [6, 6, 7, 7],
        [7, 7, 6, 6],
    ]

                    best_centers = centers.copy()

    if best_centers is None:
        # Fallback
        centers, radii = make_initial(n, [6, 5, 5, 5, 5], 0)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, float(best_sum)


def make_initial(n, rows_counts, seed):
    rng = np.random.RandomState(seed)
    r = 0.075

    centers = np.zeros((n, 2))
    idx = 0

    for row, count in enumerate(rows_counts):
        y = r + row * r * math.sqrt(3)
        offset = r if row % 2 == 1 else 0
        denom = max(count - 1, 1)
        x_space = (1 - 2 * r - offset) / denom

        for col in range(count):
            x = r + offset + col * x_space
            x += rng.uniform(-0.008, 0.008)
            y += rng.uniform(-0.008, 0.008)
            x = np.clip(x, r + 0.002, 1 - r - 0.002)
            y = np.clip(y, r + 0.002, 1 - r - 0.002)
            centers[idx] = [x, y]
            idx += 1

    radii = np.full(n, r)
    return centers, radii


def full_optimize(centers, radii, n):
    # Phase 1: Force-based relaxation to resolve overlaps
    centers = repulsion_optimize(centers, radii, n, steps=2000)

    # Phase 2: Gradually expand radii while relaxing positions
    for _ in range(80):
        radii *= 1.008
        centers = repulsion_optimize(centers, radii, n, steps=150)

    # Phase 3: Fine joint optimization with scipy
    params = np.concatenate([centers.flatten(), radii])
    bounds = [(0.002, 0.998)] * (2 * n) + [(0.001, 0.5)] * n

    result = minimize(
        joint_objective,
        params,
        args=(n,),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 30000, 'ftol': 1e-16, 'gtol': 1e-12}
    )

    opt_centers = result.x[:2 * n].reshape(n, 2)
    opt_radii = result.x[2 * n:]
    opt_radii = np.maximum(opt_radii, 1e-8)

    return opt_centers, opt_radii


def repulsion_optimize(centers, radii, n, steps=1000):
    dt = 0.003
    for step in range(steps):
        forces = np.zeros_like(centers)

        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.dot(diff, diff)
                dist = math.sqrt(dist_sq) + 1e-10
                min_dist = radii[i] + radii[j]

                if dist < min_dist * 1.4:
                    force_mag = max(0.0, min_dist * 1.15 - dist) / (dist + 1e-10)
                    forces[i] += diff * force_mag
                    forces[j] -= diff * force_mag

            for d in range(2):
                lower = radii[i] + 0.0005
                upper = 1.0 - radii[i] - 0.0005
                if centers[i][d] < lower:
                    forces[i][d] += (lower - centers[i][d]) * 800
                if centers[i][d] > upper:
                    forces[i][d] -= (centers[i][d] - upper) * 800

        cooling = 1.0 - step / steps * 0.85
        actual_dt = dt * cooling
        centers += actual_dt * forces
        centers = np.clip(centers, 0.001, 0.999)

    return centers


def joint_objective(params, n):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    obj = -np.sum(radii)
    penalty = 0.0

    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = math.sqrt(np.dot(diff, diff))
            min_dist = radii[i] + radii[j]
            gap = dist - min_dist
            if gap < 0.01:
                penalty += max(0.0, -gap) ** 2 * 200000

        for d in range(2):
            if centers[i][d] < radii[i] + 1e-6:
                penalty += (radii[i] + 1e-6 - centers[i][d]) ** 2 * 200000
            if centers[i][d] > 1.0 - radii[i] - 1e-6:
                penalty += (centers[i][d] - (1.0 - radii[i] - 1e-6)) ** 2 * 200000

    return obj + penalty


def is_valid(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0 or x < 0 or x > 1 or y < 0 or y > 1:
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt(
                (centers[i][0] - centers[j][0]) ** 2 +
                (centers[i][1] - centers[j][1]) ** 2
            )
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True
