        radii = np.zeros(n)

        idx = 0

                    break

                rep_str = 80
            elif stage == 1:
                num_iters = 30000
                lr_base = 0.008

                rep_str = 150

                num_iters = 40000
                lr_base = 0.004
                r_growth = 0.000008
                rep_str = 300
            else:

                lr_base = 0.002
                r_growth = 0.000003
                rep_str = 500

                        diff = centers[i] - centers[j]

                        dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10

                            forces[i, 0] += fx
                            forces[i, 1] += fy
                            forces[j, 0] -= fx
                            forces[j, 1] -= fy

                radii += r_growth
