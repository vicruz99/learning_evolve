import numpy as np
from scipy.optimize import minimize
import math

def calculate_overlap(centers, radii):
    """
    Calculate the total penalty for overlaps and boundary violations.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Distance to boundaries
        left = x - r
        right = 1 - (x + r)
        bottom = y - r
        top = 1 - (y + r)
        
        # Penalty is squared max(0, violation)
        if left < 0:
            penalty += left**2
        if right < 0:
            penalty += right**2
        if bottom < 0:
            penalty += bottom**2
        if top < 0:
            penalty += top**2
            
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j]) ** 2)
            dist = math.sqrt(dist_sq)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += overlap**2
                
    return penalty

def objective_function(vars, radii):
    """
    Objective for scipy.optimize.minimize to minimize overlap.
    vars contains flattened centers.
    """
    centers = vars.reshape(-1, 2)
    return calculate_overlap(centers, radii)
