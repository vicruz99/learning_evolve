- **Objective**: Maximize $\sum r_i$ (minimize $-\sum r_i$)
- **Variables**: 78-dimensional vector containing $(x_i, y_i, r_i)$ for each circle.
- **Constraints**:
  1. Boundary constraints: $r_i \le x_i \le 1-r_i$, $r_i \le y_i \le 1-r_i$, $r_i \ge 0$
  2. Non-overlap constraints: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ for all $i < j$

2. **SLSQP optimizer**: Robust for handling inequality constraints. We pass all ~430 constraints as a single vectorized function.
3. **Post-processing projection**: Ensures strict feasibility by clamping centers to boundaries and iteratively shrinking overlapping circles to handle numerical tolerances.

import numpy as np
from scipy.optimize import minimize

N = 26

def objective_func(x):
    return -np.sum(x[2::3])

def constraint_func(x):
    n = N
    con = np.empty(5*n + n*(n-1)//2)
    idx = 0
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        con[idx]   = xi - ri
        con[idx+1] = 1.0 - xi - ri
        con[idx+2] = yi - ri
        con[idx+3] = 1.0 - yi - ri
        con[idx+4] = ri - 1e-6
        idx += 5
    for i in range(n):
        for j in range(i+1, n):
            xi, yi = x[3*i], x[3*i+1]
            xj, yj = x[3*j], x[3*j+1]
            ri, rj = x[3*i+2], x[3*j+2]
            d2 = (xi-xj)**2 + **(yi-yj)2
            con[idx] = d2 - (ri+rj)**2
            idx += 1
    return con

def generate_initial(trial):
    n = N
    c = np.zeros((n, 2))
    r = np.ones(n) * 0.095
    idx = 0
    for i in range(5):
        for j in range(5):
            c[idx] = [i/4.0 + 0.05, j/4.0 + 0.05]
            idx += 1
    c[25] = [0.5, 0.5]
    r[25] = 0.05
    
    scale = 0.015 * (1 + trial)
    c += np.random.randn(n, 2) * scale
    r += np.random.randn(n) * 0.005
    r = np.clip(r, 1e-5, 0.3)
    
    for i in range(n):
        c[i, 0] = np.clip(c[i, 0], r[i], 1-r[i])
        c[i, 1] = np.clip(c[i, 1], r[i], 1-r[i])
        
    return np.hstack([c.flatten(), r])

def project_and_fix(centers, radii):
    n = N
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r + 1e-9, 1 - r - 1e-9)
        centers[i, 1] = np.clip(centers[i, 1], r + 1e-9, 1 - r - 1e-9)
        
    for _ in range(100):
        overlap = False
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt((centers[i,0]-centers[j,0])**2 + **(centers[i,1]-centers[j,1])2)
                req = radii[i] + radii[j]
                if dist < req - 1e-10:
                    shrink = (req - dist)/2 * 1.02
                    radii[i] = max(radii[i] - shrink/2, 1e-6)
                    radii[j] = max(radii[j] - shrink/2, 1e-6)
                    overlap = True
        if not overlap:
            break
    return centers, radii

def run_packing():
    np.random.seed(42)
    best_sum = 0.0
    best_c = None
    best_r = None

    bounds = [(0, 1)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    for trial in range(15):
        x0 = generate_initial(trial)
        
        res = minimize(objective_func, x0, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 2500, 'ftol': 1e-10, 'disp': False})
        
        if res.success:
            s = np.sum(res.x[2*N:])
            if s > best_sum:
                best_sum = s
                best_c = res.x[:2*N].reshape(N, 2)
                best_r = res.x[2*N:]

    if best_c is None:
        x0 = generate_initial(0)
        best_c = x0[:2*N].reshape(N, 2)
        best_r = x0[2*N:]
        best_sum = np.sum(best_r)

    best_c, best_r = project_and_fix(best_c, best_r)
    return best_c, best_r, np.sum(best_r)
