

To maximize the sum of radii for 26 circles in a unit square, I will use an iterative expansion algorithm. This strategy starts with a valid grid configuration and gradually increases the radii, using `scipy.optimize` to adjust the centers and resolve overlaps at each step. By keeping the radii equal, we leverage the high packing density of a lattice structure, which is efficient for this number of circles.

<strategy>
1.  **Initialization**: Place 26 circles in a $6 \times 5$ grid pattern (skipping 4 spots to reach 26) with a small initial radius. This provides a non-overlapping, well-distributed starting point.
2.  **Iterative Expansion**: Gradually increase the common radius of all circles.
3.  **Constraint Resolution**: At each radius step, minimize an energy function that penalizes overlaps between circles and violations of the square's boundaries.
4.  **Optimization**: Use `scipy.optimize.minimize` with the 'L-BFGS-B' method to find the optimal center positions for the current radius.
5.  **Termination**: The process stops when the optimizer can no longer resolve overlaps for a larger radius, indicating the maximum packing density has been reached.
</strategy>

```python
import numpy as np
import scipy.optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    # Start with a small radius
    r = 0.05
    # Step size for radius increase
    dr = 0.0005
    # Max iterations to prevent infinite loops
    max_iter = 1000
    
    # Initialize centers in a 6x5 grid pattern, skipping 4 to get 26 circles
    # 6 cols, 5 rows = 30 spots. We'll pick 26.
    # Use a slightly offset grid to keep away from boundaries initially
    centers = np.zeros((n, 2))
    idx = 0
    
    # Create a 6x5 grid of potential spots
    # x in [0, 1], y in [0, 1]
    # Spacing roughly 1/6
    for row in range(5):
        for col in range(6):
            if idx < n:
                # Offset to center within the grid cell
                x = (col + 0.5) / 6.0
                y = (row + 0.5) / 5.0
                centers[idx] = [x, y]
                idx += 1
            else:
                break
                
    # Define the energy function to minimize (penalties for overlaps and boundaries)
    def energy_function(pos, radius):
        """
        Calculate the penalty for a given configuration of centers and a fixed radius.
        pos: flattened array of [x1, y1, x2, y2, ...]
        """
        centers = pos.reshape(-1, 2)
        penalty = 0.0
        
        # Overlap penalty: sum of squared overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                min_dist = 2 * radius
                if dist < min_dist:
                    # Squared distance of overlap
                    penalty += (min_dist - dist)**2
        
        # Boundary penalty: sum of squared distances outside the unit square
        for i in range(n):
            x, y = centers[i]
            # Left wall
            if x < radius:
                penalty += (radius - x)**2
            # Right wall
            if x > 1 - radius:
                penalty += (x - (1 - radius))**2
            # Bottom wall
            if y < radius:
                penalty += (radius - y)**2
            # Top wall
            if y > 1 - radius:
                penalty += (y - (1 - radius))**2
                
        return penalty

    # Optimization parameters
    # We flatten the centers array for scipy
    pos = centers.flatten()
    bounds = [(0, 1) for _ in range(2 * n)] # x, y in [0, 1]
    
    # Iterate to increase radius
    for _ in range(max_iter):
        # Increase radius
        r += dr
        
        # Optimize positions to minimize energy for the new radius
        # Using L-BFGS-B which handles bounds efficiently
        result = scipy.optimize.minimize(
            energy_function,
            pos,
            args=(r,),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        
        # Update positions
        pos = result.x
        centers = pos.reshape(-1, 2)
        
        # Check if the optimization was successful (energy close to 0)
        # If energy is too high, it means we can't fit circles of this radius
        if result.fun > 1e-6:
            # Radius is too large, revert the last increase
            r -= dr
            break
            
    # Finalize the radii array
    radii = np.full(n, r)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

# For local testing or if called as a script
if __name__ == "__main__":
    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
    print(f"Average radius: {s/26}")
    # Basic self-validation
    valid = True
    for i in range(len(c)):
        if r[i] < 0 or c[i][0] < r[i] or c[i][0] > 1-r[i] or c[i][1] < r[i] or c[i][1] > 1-r[i]:
            valid = False
            break
        for j in range(i+1, len(c)):
            dist = np.sqrt(np.sum((c[i]-c[j])**2))
            if dist < r[i] + r[j] - 1e-12:
                valid = False
                break
        if not valid: break
    print(f"Valid packing: {valid}")
```
