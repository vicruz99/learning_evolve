import numpy as np
import scipy.optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Random seed for reproducibility of "random" initializations in this session
    rng = np.random.default_rng(42)

    def overlap_constraint(centers, radii, i, j):
        """Constraint: distance >= r_i + r_j"""
        d = np.sqrt(np.sum((centers[i] - centers[j])**2))
        return d - (radii[i] + radii[j])

    def boundary_constraint_x(centers, radii, i):
        """Constraint: x_i - r_i >= 0 and x_i + r_i <= 1"""
        return [centers[i, 0] - radii[i], 1.0 - (centers[i, 0] + radii[i])]

    def boundary_constraint_y(centers, radii, i):
        """Constraint: y_i - r_i >= 0 and y_i + r_i <= 1"""
        return [centers[i, 1] - radii[i], 1.0 - (centers[i, 1] + radii[i])]

    def get_constraints(n, centers, radii):
        constraints = []
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda vars_, i=i, j=j: overlap_constraint(
                        vars_[:2*n].reshape(n, 2), 
                        vars_[2*n:], i, j
                    )
                })
        
        # Boundary constraints
        for i in range(n):
            constraints.extend([
                {'type': 'ineq', 'fun': lambda vars_, i=i: boundary_constraint_x(vars_[:2*n].reshape(n, 2), vars_[2*n:], i)[0]},
                {'type': 'ineq', 'fun': lambda vars_, i=i: boundary_constraint_x(vars_[:2*n].reshape(n, 2), vars_[2*n:], i)[1]},
                {'type': 'ineq', 'fun': lambda vars_, i=i: boundary_constraint_y(vars_[:2*n].reshape(n, 2), vars_[2*n:], i)[0]},
                {'type': 'ineq', 'fun': lambda vars_, i=i: boundary_constraint_y(vars_[:2*n].reshape(n, 2), vars_[2*n:], i)[1]},
            ])
        return constraints

    def objective(vars_):
        return -np.sum(vars_[2*n:]) # Maximize sum of radii

    def generate_hex_grid(n):
        """Generate a hexagonal grid initialization."""
        # Approximate radius for 26 circles
        r_est = 0.09
        centers = np.zeros((n, 2))
        idx = 0
        
        # Estimate grid dimensions
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        
        spacing_x = 2.0 * r_est
        spacing_y = np.sqrt(3) * r_est
        
        y = r_est
        while idx < n:
            x = r_est
            row_offset = (int((y - r_est) / spacing_y) % 2) * r_est
            while x + r_est <= 1.0 and idx < n:
                centers[idx] = [x + row_offset, y]
                x += spacing_x
                idx += 1
            y += spacing_y
            if y + r_est > 1.0:
                break
                
        return centers

    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    # Multiple restarts to find global optimum
    for restart in range(10):
        # Initialize
        if restart == 0:
            # Hexagonal grid start
            centers = generate_hex_grid(n)
        else:
            # Random start within square, small radii
            centers = rng.random((n, 2))
        
        # Scale centers to be within bounds and radii to a reasonable start
        radii = np.full(n, 0.04)

        # Prepare initial vector
        x0 = np.hstack([centers.flatten(), radii])
        
        # Bounds for variables: 
        # x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for _ in range(2 * n):
            bounds.append((0, 1))
        for _ in range(n):
            bounds.append((0, 0.5))

        cons = get_constraints(n, centers, radii)

        try:
            res = scipy.optimize.minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
            )
            
            if res.success:
                current_centers = res.x[:2*n].reshape(n, 2)
                current_radii = res.x[2*n:]
                current_sum = np.sum(current_radii)
                
                # Final validation of the result
                if validate_packing(current_centers, current_radii):
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = current_centers.copy()
                        best_radii = current_radii.copy()
        except Exception:
            continue

    return best_centers, best_radii, best_sum

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
