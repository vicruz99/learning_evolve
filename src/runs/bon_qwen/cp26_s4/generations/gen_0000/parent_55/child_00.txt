from scipy.optimize import minimize

def objective(vars):

    return -np.sum(vars[52:])

def boundary_constraints(vars):
    """Constraints ensuring circles stay within the unit square."""
    vals = []
    for i in range(26):
        x = vars[i]
        y = vars[i + 26]
        r = vars[i + 52]
        # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        vals.extend([x - r, 1 - x - r, y - r, 1 - y - r])
    return np.array(vals)

def overlap_constraints(vars):
    """Constraints ensuring circles do not overlap."""
    centers = vars[:52].reshape(26, 2)

    return (dist - rad_sum)[mask]
