import numpy as np
from scipy.optimize import minimize


def compute_penalty(centers, radii, n):
    """Compute constraint violation penalty"""
    penalty = 0.0

    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            penalty += 500 * (r - x) ** 2
        if x > 1 - r:
            penalty += 500 * (x - 1 + r) ** 2
        if y < r:
            penalty += 500 * (r - y) ** 2
        if y > 1 - r:
            penalty += 500 * (y - 1 + r) ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += 500 * overlap ** 2

    return penalty


def objective(params, n):
    """Objective: minimize negative sum of radii + penalty"""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-10)

    penalty = compute_penalty(centers, radii, n)
    return -np.sum(radii) + penalty


def initialize_hexagonal(n):
    """Initialize circles in a hexagonal pattern"""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09

    idx = 0
    # 5 rows, with varying columns for hexagonal pattern
    for row in range(5):
        cols_in_row = 5 if row % 2 == 0 else 6
        for col in range(cols_in_row):
            if idx >= n:
                break
            x = (col + 0.5) / float(cols_in_row + 1)
            y = (row + 0.5) / 5.0
            if row % 2 == 1:
                x += 1.0 / (2 * (cols_in_row + 1))
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break

    return centers, radii


def initialize_grid(n):
    """Initialize circles in a square grid pattern"""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.085

    cols = 6
    rows = 5
    idx = 0

    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = (col + 0.5) / float(cols)
            y = (row + 0.5) / float(rows)
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break

    return centers, radii


def initialize_random(n, seed=42):
    """Initialize circles randomly but spread out"""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.15, 0.85, (n, 2))
    radii = np.ones(n) * 0.08
    return centers, radii


def enforce_validity(centers, radii, n):
    """Post-process to ensure strict validity by reducing radii of overlapping circles"""
    centers = centers.copy()
    radii = radii.copy()

    # Ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        max_r = min(x, 1 - x, y, 1 - y)
        radii[i] = min(radii[i], max_r)

    # Iteratively fix overlaps
    for iteration in range(1000):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    violation = min_dist - dist
                    max_violation = max(max_violation, violation)
                    # Reduce the smaller radius
                    if radii[i] < radii[j]:
                        radii[i] = max(0, radii[i] - violation * 0.5)
                    else:
                        radii[j] = max(0, radii[j] - violation * 0.5)

        if max_violation < 1e-10:
            break

    # Re-enforce boundary
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        radii[i] = min(radii[i], max_r)

    return centers, radii


def try_optimization(centers_init, radii_init, n, bounds, seed=0):
    """Run optimization from a given initial point"""
    rng = np.random.RandomState(seed)
    params0 = np.concatenate([centers_init.flatten(), radii_init])

    # Add small random perturbation
    params0 += rng.normal(0, 0.01, params0.shape)
    params0 = np.clip(params0, 0.01, 0.99)
    params0[2 * n:] = np.clip(params0[2 * n:], 0.01, 0.45)

    result = minimize(
        objective,
        params0,
        args=(n,),
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 100000, "ftol": 1e-15, "gtol": 1e-10},
    )

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]

    return centers_opt, radii_opt


def run_packing():

    bounds = [(0, 1)] * (2 * n) + [(0.001, 0.45)] * n

    best_centers = None
    best_radii = None
    best_sum = 0

        initialize_hexagonal,
        initialize_grid,
        initialize_random,
    ]

    for strategy_idx, init_func in enumerate(init_strategies):
        for restart in range(20):
            seed = strategy_idx * 100 + restart
            centers_init, radii_init = init_func(n, seed=seed)

            # For random init, use different seeds

                centers_init, radii_init = initialize_random(n, seed=seed + 1000)

                centers_init, radii_init, n, bounds, seed=seed + 2000
            )

            # Enforce validity
            centers_valid, radii_valid = enforce_validity(centers_opt, radii_opt, n)

            current_sum = np.sum(radii_valid)

            if current_sum > best_sum:
                # Verify validity
                valid = True

                # Check boundaries
                for i in range(n):
                    x, y = centers_valid[i]
                    r = radii_valid[i]
                    if x - r < -1e-12 or x + r > 1 + 1e-12:
                        valid = False
                        break
                    if y - r < -1e-12 or y + r > 1 + 1e-12:
                        valid = False
                        break

                if valid:
                    # Check overlaps

                            dist = np.sqrt(
                                np.sum((centers_valid[i] - centers_valid[j]) ** 2)
                            )

                                valid = False
                                break
                        if not valid:
                            break

                if valid:

                    best_radii = radii_valid.copy()

        best_centers = centers_fallback
        best_radii = radii_fallback
