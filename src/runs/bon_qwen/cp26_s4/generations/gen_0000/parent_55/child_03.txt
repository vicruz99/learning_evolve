import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a penalty-based optimization approach with multiple restarts.
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Penalty weight. Higher is stricter, but might cause numerical instability.
    # 10000 is usually sufficient for this scale.
    LAMBDA = 10000.0 
    
    # Number of restarts to find a good global optimum
    num_restarts = 15

    for seed in range(num_restarts):
        np.random.seed(seed)
        
        # Initial configuration
        # Random centers inside the square
        centers = np.random.rand(n, 2)
        # Initial small radii to ensure no overlaps initially
        radii = np.ones(n) * 0.01 
        
        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.hstack([centers.ravel(), radii])
        
        # Bounds: x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for i in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((1e-7, 0.5)) # r (strictly positive to avoid degeneracy)

        def objective(vars_):
            # Unpack
            c = vars_[:2*n].reshape(n, 2)
            r = vars_[2*n:]
            
            # Term 1: Minimize negative sum of radii (i.e., Maximize sum of radii)
            obj = -np.sum(r)
            
            # Term 2: Boundary penalties
            # Left: x >= r  => r - x <= 0. Violation: max(0, r - x)
            # Right: x + r <= 1 => x + r - 1 <= 0. Violation: max(0, x + r - 1)
            # Same for y
            
            # Vectorized calculations
            r_vec = r
            x = c[:, 0]
            y = c[:, 1]
            
            v_left = np.maximum(0.0, r_vec - x)
            v_right = np.maximum(0.0, x + r_vec - 1.0)
            v_bottom = np.maximum(0.0, r_vec - y)
            v_top = np.maximum(0.0, y + r_vec - 1.0)
            
            pen_boundary = np.sum(v_left**2 + v_right**2 + v_bottom**2 + v_top**2)
            
            # Term 3: Overlap penalties
            # Distance matrix
            # Shape (n, n, 2) differences
            diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            
            # Sum of radii matrix
            r_sum_mat = r_vec[:, np.newaxis] + r_vec[np.newaxis, :]
            
            # Overlap amount: positive means overlap
            overlaps = r_sum_mat - dists
            
            # We only care about unique pairs (upper triangle)
            # Mask for upper triangle excluding diagonal
            mask = np.triu(np.ones((n, n), dtype=bool), k=1)
            relevant_overlaps = overlaps[mask]
            
            # Squared penalty for overlaps
            pen_overlap = np.sum(np.maximum(0.0, relevant_overlaps)**2)
            
            # Total objective
            return obj + LAMBDA * (pen_boundary + pen_overlap)

        # Run optimization
        # L-BFGS-B is good for bounded problems
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
            
            if res.success or res.nit > 0:
                final_c = res.x[:2*n].reshape(n, 2)
                final_r = res.x[2*n:]
                current_sum = np.sum(final_r)
                
                # Verify validity with a stricter tolerance than the validator's 1e-12
                # to ensure we don't return a solution that is barely invalid.
                # However, the validator uses 1e-12, so we should be safe if we are close.
                # Let's do a quick check.
                
                # Check boundaries
                valid = True
                # x - r >= 0, x + r <= 1
                if np.any(final_r - final_c[:, 0] > 1e-10) or \
                   np.any(final_c[:, 0] + final_r - 1.0 > 1e-10) or \
                   np.any(final_r - final_c[:, 1] > 1e-10) or \
                   np.any(final_c[:, 1] + final_r - 1.0 > 1e-10):
                    valid = False
                
                # Check overlaps
                if valid:
                    diff = final_c[:, np.newaxis, :] - final_c[np.newaxis, :, :]
                    dists = np.sqrt(np.sum(diff**2, axis=2))
                    r_sum = final_r[:, np.newaxis] + final_r[np.newaxis, :]
                    overlaps = r_sum - dists
                    overlaps = overlaps[np.triu_indices(n, k=1)]
                    if np.any(overlaps > 1e-10):
                        valid = False
                
                if valid and current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = final_c.copy()
                    best_radii = final_r.copy()
                    
        except Exception:
            continue

    # If no valid packing found (unlikely), return a fallback grid
    if best_centers is None:
        # Fallback: simple grid
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        # Place in a 5x5 grid (25 circles) and one extra small one
        # This is just a fallback, optimizer should find better.
        k = 0
        for r_idx in range(5):
            for c_idx in range(5):
                if k < 25:
                    best_centers[k] = [0.1 + c_idx * 0.2, 0.1 + r_idx * 0.2]
                    best_radii[k] = 0.1
                    k += 1
        if k < 26:
            best_centers[k] = [0.5, 0.5] # Center
            best_radii[k] = 0.01 # Small
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
