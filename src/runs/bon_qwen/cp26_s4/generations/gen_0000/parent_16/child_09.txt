import numpy as np
from scipy.optimize import minimize

def constraint_fun(v, n):
    """
    Computes all boundary and overlap constraints for the optimization.
    Returns an array of constraint values that must be >= 0.
    """
    nc = 4 * n + n * (n - 1) // 2
    c = np.zeros(nc)
    idx = 0
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(n):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        c[idx] = xi - ri
        c[idx+1] = 1 - xi - ri
        c[idx+2] = yi - ri
        c[idx+3] = 1 - yi - ri
        idx += 4
        
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        for j in range(i+1, n):
            xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
            c[idx] = (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            idx += 1
    return c

def objective(v):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(v[2::3])

def run_packing():
    n = 26
    r_init = 0.09
    
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    
    # 1. Generate a valid initial configuration using a hexagonal grid pattern
    idx = 0
    y = r_init
    row = 0
    dy = r_init * np.sqrt(3)
    while idx < n and y + r_init <= 1.0:
        x_start = r_init if row % 2 == 0 else 2 * r_init
        x = x_start
        while x + r_init <= 1.0 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r_init
        row += 1
        y += dy
        
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Add small perturbation to break symmetry and aid optimization
    np.random.seed(42)
    x0 += np.random.uniform(-1e-5, 1e-5, size=x0.shape)
    
    # Define bounds for variables
    bounds = [(0, 1) if i % 3 != 2 else (0, 0.5) for i in range(3 * n)]
    
    # Setup constraints
    cons = {'type': 'ineq', 'fun': lambda v: constraint_fun(v, n)}
        
    # 2. Run optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-12})
                   
    final_centers = res.x[:3*n].reshape(n, 3)[:, :2]
    final_radii = res.x[2::3]
    
    # Ensure radii are non-negative
    final_radii = np.maximum(final_radii, 0)
    
    return final_centers, final_radii, np.sum(final_radii)
