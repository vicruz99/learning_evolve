import numpy as np
import scipy.optimize as opt
import random

def solve_lp_radii(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Constraints:
    0 <= r_i <= x_i
    0 <= r_i <= 1 - x_i
    0 <= r_i <= y_i
    0 <= r_i <= 1 - y_i
    r_i + r_j <= dist(centers[i], centers[j])
    """
    n = centers.shape[0]
    c = np.ones(n) # Coefficients for max sum r_i

    # Constraints matrix A_ub @ r <= b_ub
    # We need to collect all inequality constraints.
    
    n_constraints = 4 * n + n * (n - 1) // 2
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    
    # Wall constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[idx, i] = 1
        b_ub[idx] = x
        idx += 1
        
        # r_i <= 1 - x
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x
        idx += 1
        
        # r_i <= y
        A_ub[idx, i] = 1
        b_ub[idx] = y
        idx += 1
        
        # r_i <= 1 - y
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y
        idx += 1

    # Overlap constraints: r_i + r_j <= dist
    # dist_ij is constant for fixed centers
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf) # Ignore self
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Bounds for r_i: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # method='highs' is efficient
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback to small radii if LP fails
        return np.full(n, 0.01)

def compute_repulsion_forces(centers, radii):
    """
    Compute forces to push circles apart if they are close (touching).
    Also push away from walls if touching.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Interaction between circles
    # Only apply force if they are effectively touching or overlapping
    # Threshold to define "touching": dist < r_i + r_j + epsilon
    epsilon = 1e-4
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    radii_sum = radii[:, None] + radii[None, :]
    
    diff = dists - radii_sum
    
    # We only care about pairs that are very close (active constraints)
    # If dist > sum_radii + margin, no force needed
    # If dist < sum_radii, they overlap (shouldn't happen if valid, but just in case)
    # We want to increase sum_radii, so we want to increase dist.
    # Force direction: away from neighbor.
    
    # Vector from j to i
    vecs = centers[:, None, :] - centers[None, :, :] # Shape (n, n, 2)
    
    # Normalize vectors where dist > 0
    safe_dists = np.maximum(dists, 1e-9)
    unit_vecs = vecs / safe_dists[:, :, None]
    
    # Force magnitude: proportional to how "tight" the constraint is?
    # Actually, any separation helps. 
    # Let's apply a constant repulsion if dist is small.
    # Or better: force is inverse square of distance?
    # Let's use a soft repulsion that is active only when close.
    
    # Mask for close pairs
    close_mask = diff < 0.1 # Consider pairs within 0.1 distance margin
    
    for i in range(n):
        for j in range(i + 1, n):
            if close_mask[i, j]:
                # Strength of repulsion
                # If they are touching (diff approx 0), we push hard?
                # If they are far, push weak.
                # Let's use 1/dist^2
                strength = 0.5 / (dists[i, j] ** 2 + 1e-6)
                
                f = strength * unit_vecs[i, j]
                forces[i] += f
                forces[j] -= f

    # Wall forces
    # If circle is touching wall, push towards center
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left wall
        if x - r < 0.01:
            forces[i, 0] += 0.1
        # Right wall
        if 1 - (x + r) < 0.01:
            forces[i, 0] -= 0.1
        # Bottom wall
        if y - r < 0.01:
            forces[i, 1] += 0.1
        # Top wall
        if 1 - (y + r) < 0.01:
            forces[i, 1] -= 0.1
            
    return forces

def run_packing():
    best_sum = 0
    best_centers = None
    best_radii = None
    
    n_circles = 26
    iterations = 2000 # Steps per run
    n_restarts = 10
    
    for restart in range(n_restarts):
        # Initialize centers randomly
        centers = np.random.rand(n_circles, 2)
        radii = np.full(n_circles, 0.01)
        
        # Pre-solve LP to get valid radii for random positions
        radii = solve_lp_radii(centers)
        
        current_sum = np.sum(radii)
        
        # Optimization loop
        step_size = 0.02
        for step in range(iterations):
            # 1. Reposition centers based on forces
            forces = compute_repulsion_forces(centers, radii)
            
            # Update centers
            centers += step_size * forces
            
            # Clip centers to valid range [r, 1-r] is hard because r changes
            # Just clip to [0.01, 0.99] to stay inside
            centers = np.clip(centers, 0.01, 0.99)
            
            # 2. Re-solve LP for radii
            new_radii = solve_lp_radii(centers)
            
            new_sum = np.sum(new_radii)
            
            # Accept if improved (or just always accept to explore?)
            # Since LP maximizes radii for FIXED centers, 
            # moving centers might increase or decrease the max possible sum.
            # We want to maximize sum.
            
            radii = new_radii
            current_sum = new_sum
            
            # Decay step size
            if step % 100 == 0 and step > 0:
                step_size *= 0.8
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Final polish: one last LP solve
    best_radii = solve_lp_radii(best_centers)
    
    # Ensure strict validity (clip radii slightly if needed due to numerical errors)
    # The LP guarantees constraints, but float precision might be an issue.
    # However, LP is exact enough.
    
    return best_centers, best_radii, np.sum(best_radii)
