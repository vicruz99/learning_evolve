import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_fn(vars):
    """
    Computes all inequality constraints.
    Returns an array of constraint values that must be >= 0.
    """
    n = N_CIRCLES
    num_boundary = 4 * n
    num_overlaps = n * (n - 1) // 2
    c = np.zeros(num_boundary + num_overlaps)
    k = 0
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        c[k] = xi - ri
        c[k+1] = 1.0 - xi - ri
        c[k+2] = yi - ri
        c[k+3] = 1.0 - yi - ri
        k += 4
        
    # Overlap constraints: dist >= ri + rj
    for i in range(n):
        for j in range(i + 1, n):
            dx = vars[3*i] - vars[3*j]
            dy = vars[3*i+1] - vars[3*j+1]
            dist = np.sqrt(dx*dx + dy*dy)
            ri_rj = vars[3*i+2] + vars[3*j+2]
            c[k] = dist - ri_rj
            k += 1
            
    return c

def run_packing():
    n = N_CIRCLES
    
    # Initialize with a hexagonal lattice pattern
    centers = np.zeros((n, 2))
    idx = 0
    row_counts = [5, 4, 5, 4, 5, 3]
    y = 0.1
    
    for count in row_counts:
        xs = np.linspace(0.1, 0.9, count)
        for x in xs:
            centers[idx] = [x, y]
            idx += 1
            if idx >= n:
                break
        if idx >= n:
            break
        y += np.sqrt(3) / 10.0
        
    # Start with a feasible radius slightly less than theoretical 0.1 limit
    radii = np.full(n, 0.092)
    
    # Flatten to optimization vector [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Variable bounds
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.25)] * n
    
    # Package constraints
    cons = NonlinearConstraint(constraint_fn, 0.0, np.inf)
    
    # Run optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=[cons], 
                   options={'maxiter': 10000, 'ftol': 1e-12})
                   
    # Extract results
    centers_opt = res.x[:3*n].reshape(n, 3)[:, :2]
    radii_opt = res.x[2::3]
    radii_opt = np.maximum(radii_opt, 0.0)
    
    sum_radii = float(np.sum(radii_opt))
    
    return centers_opt, radii_opt, sum_radii
