import numpy as np
from scipy.optimize import minimize


def hexagonal_initialization(n):
    """Create hexagonal grid initialization for n circles."""
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        y = 0.12 + row * 0.19
        x_offset = (row % 2) * 0.095
        x_start = 0.12 + x_offset
        col = 0
        while idx < n:
            x = x_start + col * 0.19
            if x <= 0.88 and y <= 0.88:
                centers[idx] = [x, y]
                idx += 1
            col += 1
            if x > 0.88:
                break
        row += 1
    return centers


def grid_initialization(n):
    """Create grid initialization for n circles."""
    centers = np.zeros((n, 2))
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    spacing_x = 0.8 / cols
    spacing_y = 0.8 / rows
    offset_x = (1 - cols * spacing_x) / 2
    offset_y = (1 - rows * spacing_y) / 2
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx < n:
                centers[idx] = [offset_x + c * spacing_x, offset_y + r * spacing_y]
                idx += 1
    return centers


def random_initialization(n):
    """Create random initialization for n circles."""
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    return centers


def compute_loss(params, n, pw):
    """Compute loss: negative sum of radii plus penalty for constraint violations."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    radii = np.maximum(radii, 0)

    loss = -np.sum(radii)

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq + 1e-16)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                loss += pw * overlap * overlap

    for i in range(n):
        r = radii[i]
        x = centers[i, 0]
        y = centers[i, 1]
        if x < r:
            loss += pw * (x - r) * (x - r)
        if x > 1 - r:
            loss += pw * (x + r - 1) * (x + r - 1)
        if y < r:
            loss += pw * (y - r) * (y - r)
        if y > 1 - r:
            loss += pw * (y + r - 1) * (y + r - 1)

    return loss


def force_directed_relax(centers, radii, n, steps=500, growth_per_step=0.00008):
    """Force-directed method to relax overlapping circles while growing them."""
    centers = centers.copy()
    radii = radii.copy()
    dt = 0.05

    for step in range(steps):
        radii += growth_per_step
        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq + 1e-16)
                min_dist = radii[i] + radii[j]

                if dist < min_dist:
                    overlap = min_dist - dist
                    strength = overlap * 0.5
                    if dist > 1e-8:
                        fx = dx / dist * strength
                        fy = dy / dist * strength
                    else:
                        angle = np.random.uniform(0, 2 * np.pi)
                        fx = np.cos(angle) * strength
                        fy = np.sin(angle) * strength
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy

        centers += dt * forces

        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1 - r, centers[i, 1]))

    return centers, radii


def clip_and_validate(centers, radii, n):
    """Ensure all circles are within bounds and radii are non-negative."""
    radii = np.maximum(radii, 0)
    for i in range(n):
        r = radii[i]
        centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1 - r, centers[i, 1]))
    return centers, radii


def optimize_from(centers, radii, n, penalty_weight, max_iter=10000):
    """Run scipy optimization from given initial state."""
    params = np.concatenate([centers.flatten(), radii])

    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.45))

    result = minimize(
        compute_loss,
        params,
        args=(n, penalty_weight),
        method="L-BFGS-B",
        bounds=bounds,
        options={
            "maxiter": max_iter,
            "ftol": 1e-16,
            "gtol": 1e-10,
            "disp": False,
        },
    )

    opt_params = result.x
    opt_centers = opt_params[:2 * n].reshape(n, 2)
    opt_radii = opt_params[2 * n:]

    return opt_centers, opt_radii
