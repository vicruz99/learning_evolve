import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars_arr):
    """Objective: minimize -r to maximize the common radius."""
    return -vars_arr[-1]

def constraint_func(vars_arr):
    """
    Computes all inequality constraints >= 0:
    1. Boundary constraints for each circle
    2. Non-overlap constraints for each pair
    """
    r = vars_arr[-1]
    n = N_CIRCLES
    num_constraints = 4 * n + n * (n - 1) // 2
    c = np.empty(num_constraints)
    idx = 0
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(n):
        c[idx] = vars_arr[2*i] - r
        idx += 1
        c[idx] = 1.0 - vars_arr[2*i] - r
        idx += 1
        c[idx] = vars_arr[2*i+1] - r
        idx += 1
        c[idx] = 1.0 - vars_arr[2*i+1] - r
        idx += 1
        
    # Pairwise non-overlap: dist^2 >= (2r)^2
    for i in range(n):
        xi, yi = vars_arr[2*i], vars_arr[2*i+1]
        for j in range(i + 1, n):
            dx = xi - vars_arr[2*j]
            dy = yi - vars_arr[2*j+1]
            c[idx] = dx*dx + dy*dy - 4.0 * r * r
            idx += 1
            
    return c

def run_packing():
    # 1. Initialize centers in a hexagonal lattice
    r_init = 0.1
    pts = []
    for row in range(8):
        y = r_init + row * r_init * np.sqrt(3)
        if y > 1 - r_init:
            break
        for col in range(8):
            x = r_init + col * 2 * r_init + (row % 2) * r_init
            if x > 1 - r_init:
                break
            pts.append([x, y])
            
    pts = np.array(pts[:N_CIRCLES])
    x0 = np.concatenate([pts.flatten(), [r_init]])
    
    # 2. Setup bounds and constraints
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.05, 0.15)]
    cons = [{'type': 'ineq', 'fun': constraint_func}]
    
    # 3. Optimize
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-8, 'disp': False})
        
        if res.success or res.fun < -0.101:
            centers = res.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
            r_opt = res.x[-1]
            radii = np.full(N_CIRCLES, r_opt)
            return centers, radii, N_CIRCLES * r_opt
    except Exception:
        pass
        
    # 4. Fallback to optimized initial guess if needed
    radii = np.full(N_CIRCLES, r_init)
    return pts, radii, N_CIRCLES * r_init
