The objective is to maximize $\sum r_i$, subject to non-overlap constraints $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ and boundary constraints $r_i \le x_i \le 1-r_i$, $r_i \le y_i \le 1-r_i$.

2. **Continuation Method**: We start with a small penalty coefficient $\mu$ and gradually increase it. This helps the optimizer escape poor local minima and converge to a tighter packing.
3. **Multiple Restarts**: We initialize from several diverse configurations (hexagonal grid, randomized layouts) to improve the chance of finding a global optimum.
4. **Gradient Computation**: We provide an exact analytical gradient for fast convergence. The gradient handles pairwise repulsion and boundary reflections explicitly.

import numpy as np
from scipy.optimize import minimize

def compute_loss_and_grad(vars, n, mu):
    """Computes the penalized objective and its gradient."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    loss = -np.sum(r)
    grad = np.zeros_like(vars)
    grad[2::3] = -1.0  # Gradient of -sum(r)
    
    # Pairwise repulsion constraints
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        base_i = 3 * i
        for j in range(i + 1, n):
            dx = xi - x[j]
            dy = yi - y[j]
            d = np.hypot(dx, dy)
            if d < 1e-10:
                d = 1e-10
            vr = ri + r[j] - d
            if vr > 0:
                loss += mu * vr**2
                g = 2 * mu * vr
                dxd = dx / d
                dyd = dy / d
                
                grad[base_i] -= g * dxd
                grad[base_i + 1] -= g * dyd
                grad[base_i + 2] += g
                
                base_j = 3 * j
                grad[base_j] += g * dxd
                grad[base_j + 1] += g * dyd
                grad[base_j + 2] += g
                
    # Boundary constraints
    for i in range(n):
        base_i = 3 * i
        ri = r[i]
        xi = x[i]
        yi = y[i]
        
        # x - r >= 0
        vr = ri - xi
        if vr > 0:
            loss += mu * vr**2
            g = 2 * mu * vr
            grad[base_i] -= g
            grad[base_i + 2] += g
        # x + r <= 1
        vr = xi + ri - 1.0
        if vr > 0:
            loss += mu * vr**2
            g = 2 * mu * vr
            grad[base_i] += g
            grad[base_i + 2] += g
        # y - r >= 0
        vr = ri - yi
        if vr > 0:
            loss += mu * vr**2
            g = 2 * mu * vr
            grad[base_i + 1] -= g
            grad[base_i + 2] += g
        # y + r <= 1
        vr = yi + ri - 1.0
        if vr > 0:
            loss += mu * vr**2
            g = 2 * mu * vr
            grad[base_i + 1] += g
            grad[base_i + 2] += g
            
    return loss, grad

def obj(v, n, mu):
    return compute_loss_and_grad(v, n, mu)[0]

def jac(v, n, mu):
    return compute_loss_and_grad(v, n, mu)[1]

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Variable bounds: x,y allowed slight overflow for penalty handling, r strictly non-negative
    bounds = []
    for _ in range(n):
        bounds.append((-0.1, 1.1))
        bounds.append((-0.1, 1.1))
        bounds.append((0.0, 0.6))
        
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal-like grid layout
    rows = [5, 6, 5, 6, 4]
    pts1 = []
    y_val = 0.15
    for cnt in rows:
        dx = (1.0 - 0.3) / (cnt + 1)
        for k in range(cnt):
            pts1.extend([0.15 + k * dx, y_val, 0.04])
        y_val += 0.18
    configs.append(np.array(pts1))
    
    # 2. Randomized layouts
    for seed in [42, 123, 999, 2024]:
        rng = np.random.RandomState(seed)
        pts = np.zeros(3 * n)
        pts[0::3] = rng.rand(n) * 0.8 + 0.1
        pts[1::3] = rng.rand(n) * 0.8 + 0.1
        pts[2::3] = 0.03
        configs.append(pts)
        
    # Continuation schedule for penalty coefficient
    mus = [100, 500, 2000, 5000, 10000]
    
    for init_arr in configs:
        current_vars = init_arr.copy()
        for mu in mus:
            res = minimize(
                fun=obj,
                jac=jac,
                x0=current_vars,
                args=(n, mu),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-10}
            )
            current_vars = res.x
            
        r_vals = current_vars[2::3]
        s = np.sum(r_vals)
        if s > best_sum:
            best_sum = s
            best_centers = current_vars[0::3].reshape((n, 2))
            best_radii = r_vals.copy()
            
    # Final projection to strictly satisfy validator constraints
    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    return best_centers, best_radii, np.sum(best_radii)
