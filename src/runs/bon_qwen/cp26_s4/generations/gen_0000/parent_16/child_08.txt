import numpy as np
from scipy.optimize import minimize

def compute_cost(params, n, mu):
    """
    Computes the penalized objective function.
    Minimizes negative sum of radii + penalties for boundary and overlap violations.
    """
    r = params[0::3]
    x = params[1::3]
    y = params[2::3]
    
    cost = -np.sum(r)
    
    # Boundary violation penalties: r <= x <= 1-r and r <= y <= 1-r
    cost += mu * np.sum(np.maximum(0, r - x)**2)
    cost += mu * np.sum(np.maximum(0, r + x - 1)**2)
    cost += mu * np.sum(np.maximum(0, r - y)**2)
    cost += mu * np.sum(np.maximum(0, r + y - 1)**2)
    
    # Overlap violation penalties: distance >= r_i + r_j
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)  # Ignore self-distance
    
    rad_sum = r[:, np.newaxis] + r[np.newaxis, :]
    cost += mu * np.sum(np.maximum(0, rad_sum - dist)**2)
    
    return cost

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds for r, x, y
    bounds = [(0.001, 0.5) for _ in range(n)] + \
             [(0.0, 1.0) for _ in range(n)] + \
             [(0.0, 1.0) for _ in range(n)]
             
    # Penalty continuation schedule
    mu_schedule = [50, 300, 1500, 8000, 25000]
    
    # Run multiple trials with different perturbations to escape local minima
    for trial in range(4):
        np.random.seed(123 + trial)
        
        # Initialize with a 5x5 grid + center circle
        r0 = np.full(n, 0.08)
        r0[-1] = 0.04
        x0 = np.tile(np.linspace(0.1, 0.9, 5), 5)
        y0 = np.repeat(np.linspace(0.1, 0.9, 5), 5)
        x0[-1] = 0.5
        y0[-1] = 0.5
        
        # Add controlled random noise
        r0 += np.random.uniform(-0.008, 0.008, n)
        x0 += np.random.uniform(-0.02, 0.02, n)
        y0 += np.random.uniform(-0.02, 0.02, n)
        
        # Clip to safe initial ranges
        r0 = np.clip(r0, 0.01, 0.4)
        x0 = np.clip(x0, 0.01, 0.99)
        y0 = np.clip(y0, 0.01, 0.99)
        
        params0 = np.empty(n * 3)
        params0[0::3] = r0
        params0[1::3] = x0
        params0[2::3] = y0
        
        current_params = params0.copy()
        
        # Sequential penalty optimization
        for mu in mu_schedule:
            res = minimize(compute_cost, current_params, args=(n, mu), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 1200, 'ftol': 1e-11, 'gtol': 1e-8})
            current_params = res.x
            
        # Extract optimized parameters
        r_opt = current_params[0::3]
        x_opt = current_params[1::3]
        y_opt = current_params[2::3]
        
        # Hard projection onto feasible set to eliminate numerical drift
        for i in range(n):
            r_opt[i] = min(r_opt[i], x_opt[i], 1.0 - x_opt[i], y_opt[i], 1.0 - y_opt[i])
            r_opt[i] = max(r_opt[i], 0.0)
            
        current_sum = np.sum(r_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_radii = r_opt.copy()
            best_centers = np.column_stack((x_opt, y_opt))
            
    return best_centers, best_radii, float(best_sum)
