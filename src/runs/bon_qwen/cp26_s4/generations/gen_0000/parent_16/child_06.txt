import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
OVERLAP_INDICES = np.triu_indices(N_CIRCLES, k=1)

def objective(x):
    return -np.sum(x[2::3])

def grad_obj(x):
    g = np.zeros_like(x)
    g[2::3] = -1.0
    return g

def bound_constraints(x):
    pts = x.reshape(N_CIRCLES, 3)
    xs, ys, rs = pts[:, 0], pts[:, 1], pts[:, 2]
    return np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])

def overlap_constraints(x):
    pts = x.reshape(N_CIRCLES, 3)
    xs, ys, rs = pts[:, 0], pts[:, 1], pts[:, 2]
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    return (dx**2 + dy**2 - dr**2)[OVERLAP_INDICES]

def run_packing():
    bounds = []
    for i in range(N_CIRCLES * 3):
        if i % 3 == 2:
            bounds.append((1e-12, 0.5))
        else:
            bounds.append((0.0, 1.0))
            
    cons = [
        {'type': 'ineq', 'fun': bound_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]
    
    best_sum = 0.0
    best_x = None
    
    for restart in range(150):
        if restart < 30:
            x0 = np.zeros(N_CIRCLES * 3)
            idx = 0
            for r in range(5):
                for c in range(6):
                    if idx >= N_CIRCLES: break
                    x0[idx] = 0.1 + c * 0.15
                    x0[idx+1] = 0.1 + r * 0.18
                    x0[idx+2] = 0.04
                    idx += 1
        else:
            x0 = np.random.uniform(0.15, 0.85, N_CIRCLES * 3)
            x0[2::3] = 0.04
            
        try:
            res = minimize(objective, x0, method='SLSQP', jac=grad_obj,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        best_x = np.random.uniform(0.2, 0.8, N_CIRCLES * 3)
        best_x[2::3] = 0.01
        
    centers = np.array([best_x[i*3:i*3+2] for i in range(N_CIRCLES)])
    radii = np.array([best_x[i*3+2] for i in range(N_CIRCLES)])
    
    # Shrink slightly to strictly satisfy checker tolerance
    radii -= 1e-9
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, np.sum(radii)
