import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    best_centers = None
    best_radii = None
    best_sum = -np.inf

    # Penalty weight for constraint violations
    penalty_weight = 100.0

    def calculate_objective(x):
        """
        Calculates the objective value (negative sum of radii + penalties).
        x contains [x0, y0, r0, x1, y1, r1, ...]
        """
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)

        for i in range(n_circles):
            centers[i, 0] = x[3 * i]
            centers[i, 1] = x[3 * i + 1]
            radii[i] = x[3 * i + 2]

        # Ensure radii are positive for penalty calculation
        radii = np.maximum(radii, 1e-9)

        obj = -np.sum(radii)

        # Boundary constraints: circle must be inside [0,1]x[0,1]
        # x - r >= 0  =>  r - x <= 0
        # x + r <= 1  =>  x + r - 1 <= 0
        # y - r >= 0  =>  r - y <= 0
        # y + r <= 1  =>  y + r - 1 <= 0
        
        for i in range(n_circles):
            r = radii[i]
            cx, cy = centers[i]
            
            # Penalties for being outside the square
            if cx - r < 0: obj += penalty_weight * (r - cx)**2
            if cx + r > 1: obj += penalty_weight * (cx + r - 1)**2
            if cy - r < 0: obj += penalty_weight * (r - cy)**2
            if cy + r > 1: obj += penalty_weight * (cy + r - 1)**2

        # Overlap constraints: dist >= r_i + r_j
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                               (centers[i, 1] - centers[j, 1])**2)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    obj += penalty_weight * (min_dist - dist)**2

        return obj

    def hexagonal_lattice_init(perturbation_scale=0.01):
        """Generates a hexagonal lattice arrangement as an initial guess."""
        centers = np.zeros((n_circles, 2))
        # Rough radius estimate
        r_guess = 0.10
        
        # Try to fit in a grid-like hexagonal pattern
        # 5 rows, approx 5-6 circles per row
        count = 0
        y = r_guess
        row = 0
        while count < n_circles and y <= 1 - r_guess:
            x_start = r_guess if row % 2 == 0 else r_guess + r_guess * np.sqrt(3)/2 * 0.1 # slight shift
            # Standard hex shift is r, but we approximate
            x = x_start
            if row % 2 == 1:
                x = x_start + r_guess # Shift for hex
            
            while count < n_circles and x <= 1 - r_guess:
                if count < n_circles:
                    centers[count, 0] = x
                    centers[count, 1] = y
                    count += 1
                x += 2 * r_guess
            
            y += r_guess * np.sqrt(3)
            row += 1
            
        # Add some random perturbation
        noise = np.random.normal(0, perturbation_scale, centers.shape)
        centers = centers + noise
        
        # Clip to valid range roughly
        radii = np.full(n_circles, r_guess)
        
        return centers, radii

    def run_optimization():
        centers, radii = hexagonal_lattice_init(perturbation_scale=0.005)
        
        # Flatten to vector for scipy
        x0 = np.zeros(3 * n_circles)
        for i in range(n_circles):
            x0[3 * i] = centers[i, 0]
            x0[3 * i + 1] = centers[i, 1]
            x0[3 * i + 2] = radii[i]
            
        # Bounds: x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for _ in range(n_circles):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
            
        res = minimize(calculate_objective, x0, method='Nelder-Mead', 
                       options={'maxiter': 20000, 'xatol': 1e-7, 'fatol': 1e-7})
        
        # Extract results
        opt_centers = np.zeros((n_circles, 2))
        opt_radii = np.zeros(n_circles)
        
        valid = True
        for i in range(n_circles):
            opt_centers[i, 0] = max(0.0, min(1.0, res.x[3 * i]))
            opt_centers[i, 1] = max(0.0, min(1.0, res.x[3 * i + 1]))
            opt_radii[i] = max(0.0, res.x[3 * i + 2])
            
            # Basic validity check
            if opt_centers[i, 0] < opt_radii[i] or opt_centers[i, 0] > 1 - opt_radii[i]: valid = False
            if opt_centers[i, 1] < opt_radii[i] or opt_centers[i, 1] > 1 - opt_radii[i]: valid = False

        # Check overlaps
        if valid:
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dist = np.sqrt(np.sum((opt_centers[i] - opt_centers[j])**2))
                    if dist < opt_radii[i] + opt_radii[j] - 1e-12:
                        valid = False
                        break
                if not valid: break
        
        if valid:
            return opt_centers, opt_radii, np.sum(opt_radii)
        else:
            return None, None, -np.inf

    # Run multiple times with different seeds to find global optimum
    np.random.seed(42) # Fixed seed for reproducibility of best attempt
    
    for _ in range(20): # 20 restarts
        centers, radii, total = run_optimization()
        if total > best_sum:
            best_sum = total
            best_centers = centers
            best_radii = radii

    # If optimization failed to find a valid packing, fallback to simple grid
    if best_centers is None:
        # 5x5 grid (25 circles) + 1 in corner
        centers = np.zeros((26, 2))
        radii = np.zeros(26)
        count = 0
        for r in range(5):
            for c in range(5):
                centers[count, 0] = 0.1 + c * 0.2
                centers[count, 1] = 0.1 + r * 0.2
                radii[count] = 0.1
                count += 1
        # 26th circle

        radii[25] = 0.05 # Small circle in corner
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
