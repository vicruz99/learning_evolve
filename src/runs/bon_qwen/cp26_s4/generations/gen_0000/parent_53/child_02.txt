import numpy as np
from scipy.optimize import minimize

def _compute_objective(vars, n, C):
    """
    Computes the penalized objective function.
    vars: flattened array [x1, y1, r1, ..., xn, yn, rn]
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # 1. Boundary penalties: max(0, violation)^2
    # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    v_left = np.maximum(0, radii - centers[:, 0])
    v_right = np.maximum(0, centers[:, 0] + radii - 1.0)
    v_bottom = np.maximum(0, radii - centers[:, 1])
    v_top = np.maximum(0, centers[:, 1] + radii - 1.0)
    bound_pen = np.sum(v_left**2 + v_right**2 + v_bottom**2 + v_top**2)
    
    # 2. Overlap penalties: max(0, r_i + r_j - dist)^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Only consider upper triangle to avoid double counting and self-comparison
    mask = np.triu(np.ones((n, n)), k=1)
    overlaps = np.maximum(0, rad_sum - dists)
    overlap_pen = np.sum(overlaps**2 * mask)
    
    # Objective: minimize negative sum of radii + penalty terms
    return -np.sum(radii) + C * (bound_pen + overlap_pen)

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    init_configs = []
    
    # Generate diverse initial configurations to escape local minima
    # 1. Perturbed square grid layouts
    for seed in [42, 123, 456, 789, 2024, 3000, 5000]:
        rng = np.random.RandomState(seed)
        gx = np.tile(np.linspace(0.12, 0.88, 5), 5)
        gy = np.repeat(np.linspace(0.12, 0.88, 5), 5)
        centers = np.column_stack([gx, gy])
        centers = centers[:n]
        centers += rng.uniform(-0.02, 0.02, centers.shape)
        radii = np.full(n, 0.10 + rng.uniform(-0.005, 0.005, n))
        init_configs.append(np.concatenate([centers.ravel(), radii]))
        
    # 2. Hexagonal lattice layout (denser packing structure)
    pts = []
    y = 0.10
    for i in range(6):
        x_start = 0.10 if i % 2 == 0 else 0.18
        n_in_row = 5 if i % 2 == 0 else 6
        for j in range(n_in_row):
            if len(pts) < n:
                pts.append([x_start + j * 0.15, y])
        y += 0.13
    if len(pts) >= n:
        init_configs.append(np.array(pts[:n]).ravel())

    # Progressive penalty coefficients for annealing
    Cs = [500, 5000, 50000, 200000]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    for config in init_configs:
        x = config.copy()
        x[2*n:] = np.maximum(x[2*n:], 0.001) # Ensure strictly positive radii initially
        
        # Multi-stage optimization with increasing penalty weight
        for C in Cs:
            res = minimize(_compute_objective, x, args=(n, C), method='L-BFGS-B',
                          bounds=bounds, options={'ftol': 1e-14, 'maxiter': 5000, 'disp': False})
            x = res.x
            
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        
        # Numerical cleanup to strictly satisfy validation constraints
        radii = np.maximum(radii, 0.0)
        centers = np.clip(centers, radii[:, None], 1.0 - radii[:, None])
        
        curr_sum = np.sum(radii)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, best_sum
