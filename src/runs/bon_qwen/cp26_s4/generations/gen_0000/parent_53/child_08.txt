from scipy.optimize import minimize

def compute_penalty(centers_flat, radii):
    """
    Computes a smooth penalty for overlaps and boundary violations.
    Lower penalty indicates a more valid packing.
    """
    n = len(radii)
    centers = centers_flat.reshape((n, 2))
    penalty = 0.0
    
    # Pairwise overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            min_dist_sq = (radii[i] + radii[j])**2
            
            if dist_sq < min_dist_sq:
                dist = np.sqrt(dist_sq)
                diff = (radii[i] + radii[j]) - dist
                penalty += diff * diff
                
    # Boundary violation penalty
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            penalty += (r - x) ** 2
        if x > 1.0 - r:
            penalty += (x - (1.0 - r)) ** 2
        if y < r:
            penalty += (r - y) ** 2
        if y > 1.0 - r:
            penalty += (y - (1.0 - r)) ** 2
            
    return penalty
