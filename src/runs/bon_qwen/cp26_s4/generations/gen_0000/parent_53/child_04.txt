import numpy as np
from scipy.optimize import minimize

def _objective(vars_):
    # vars_ shape (78,) -> x(26), y(26), r(26)
    # We maximize sum of radii, so minimize negative sum
    return -np.sum(vars_[52:])

def _constraints(vars_):
    x = vars_[:26]
    y = vars_[26:52]
    r = vars_[52:]
    cons = []
    # Boundary constraints: circle inside [0,1]x[0,1]
    for i in range(26):
        cons.append(x[i] - r[i])
        cons.append(1.0 - x[i] - r[i])
        cons.append(y[i] - r[i])
        cons.append(1.0 - y[i] - r[i])
    # Overlap constraints: distance >= sum of radii
    for i in range(26):
        for j in range(i + 1, 26):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx * dx + dy * dy)
            cons.append(dist - (r[i] + r[j]))
    return np.array(cons)

def run_packing():
    n = 26
    bounds = [(0, 1)] * 26 + [(0, 1)] * 26 + [(0, 0.5)] * 26
    
    best_x = None
    best_sum_radii = -1.0
    
    # Generate multiple initial configurations to avoid local minima
    configs = []
    
    # 1. Regular Grid
    c1 = []
    for i in range(5):
        for j in range(6):
            if len(c1) >= 26: break
            c1.append([0.2 + i * 0.2, 0.15 + j * 0.15])
    configs.append(np.array(c1[:26]))
    
    # 2. Hexagonal-like packing
    c2 = []
    for i in range(5):
        for j in range(6):

            x = 0.19 + i * 0.20

                y += 0.09
            c2.append([x, y])
    configs.append(np.array(c2[:26]))
    
    # 3-6. Random initializations (confined to center to ensure feasibility)
    np.random.seed(42)
    for _ in range(4):
        c_rand = np.random.rand(26, 2) * 0.6 + 0.2
        configs.append(c_rand)
        
    for centers_init in configs:
        radii_init = np.full(26, 0.04)
        x0 = np.concatenate([centers_init[:, 0], centers_init[:, 1], radii_init])
        
        try:
            res = minimize(_objective, x0, method='SLSQP', bounds=bounds,
                          constraints={'type': 'ineq', 'fun': _constraints},
                          options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            
            # Evaluate result regardless of success flag, as SLSQP sometimes 
            # finds good points without formally converging within tol
            r = res.x[52:]
            if np.all(r >= -1e-6):
                current_sum = np.sum(np.maximum(r, 0.0))
                # Simple validation of constraints with tolerance
                if np.all(_constraints(res.x) >= -1e-9):
                    if current_sum > best_sum_radii:
                        best_sum_radii = current_sum
                        best_x = res.x
        except Exception:
            continue
            
    if best_x is None:
        # Fallback to first config with small radii
        centers = configs[0]
        radii = np.full(26, 0.04)
        return centers, radii, np.sum(radii)
        
    centers = best_x[:52].reshape((26, 2))
    radii = best_x[52:]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, np.sum(radii)
