import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    
    # --- 1. Initialization: Hexagonal Grid ---
    # We want to fit 26 circles. A hexagonal packing is optimal.
    # Let's try to arrange them in rows.
    # Width 1, Height 1.
    # Approximate radius for 25 circles (5x5) is 0.1.
    # For 26, we need a bit smaller or a 6th row.
    
    # Let's construct a hexagonal grid.
    # We estimate an initial radius. 
    # Area heuristic: 26 * pi * r^2 approx 0.9 * 1.0 => r approx 0.105. 
    # But boundary effects reduce this. Let's start with r = 0.09.
    
    r_init = 0.09
    centers = []
    radii = []
    
    # Hexagonal packing parameters
    # Vertical spacing between rows
    dy = r_init * np.sqrt(3)
    # Horizontal spacing within a row
    dx = 2 * r_init
    
    # Generate rows
    row_idx = 0
    count = 0
    
    # We will fill rows from bottom to top
    y_pos = r_init
    
    while count < n_circles:
        # Determine x positions for current row
        # Alternate rows are shifted by dx/2 = r_init
        x_start = r_init
        if row_idx % 2 == 1:
            x_start = r_init + dx / 2.0 # Shifted row
        
        x_pos = x_start
        while x_pos + r_init <= 1.0 + 1e-9 and count < n_circles:
            centers.append([x_pos, y_pos])
            radii.append(r_init)
            x_pos += dx
            count += 1
        
        y_pos += dy
        row_idx += 1
        
        # Safety break to prevent infinite loop if logic fails
        if y_pos > 2.0:
            break

    # Convert to numpy arrays
    centers = np.array(centers[:n_circles])
    radii = np.array(radii[:n_circles])
    
    # If we didn't generate enough (should not happen with r=0.09), fallback
    if len(centers) < n_circles:
        centers = np.random.rand(n_circles, 2) * 0.5 + 0.25
        radii = np.full(n_circles, 0.05)

    # --- 2. Optimization Setup ---
    
    # Flatten variables: x1, y1, r1, x2, y2, r2, ...
    x0 = []
    for i in range(n_circles):
        x0.extend([centers[i, 0], centers[i, 1], radii[i]])
    x0 = np.array(x0)

    # Objective: Minimize -sum(radii)
    def objective(params):
        r = params[2::3]
        return -np.sum(r)

    # Constraints
    constraints = []

    # 1. Boundary constraints: r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
    # Same for y
    for i in range(n_circles):
        idx_x = i * 3
        idx_y = i * 3 + 1
        idx_r = i * 3 + 2
        
        # x >= r  => x - r >= 0
        def bound_x_low(p, i=i):
            return p[i*3] - p[i*3+2]
        constraints.append({'type': 'ineq', 'fun': bound_x_low})
        
        # x <= 1-r => 1 - x - r >= 0
        def bound_x_high(p, i=i):
            return 1.0 - p[i*3] - p[i*3+2]
        constraints.append({'type': 'ineq', 'fun': bound_x_high})
        
        # y >= r
        def bound_y_low(p, i=i):
            return p[i*3+1] - p[i*3+2]
        constraints.append({'type': 'ineq', 'fun': bound_y_low})
        
        # y <= 1-r
        def bound_y_high(p, i=i):
            return 1.0 - p[i*3+1] - p[i*3+2]
        constraints.append({'type': 'ineq', 'fun': bound_y_high})

        # r >= 0
        def bound_r(p, i=i):
            return p[i*3+2]
        constraints.append({'type': 'ineq', 'fun': bound_r})

    # 2. Non-overlap constraints: dist^2 >= (r1 + r2)^2
    # (x1-x2)^2 + (y1-y2)^2 - (r1+r2)^2 >= 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_x1, idx_y1, idx_r1 = i*3, i*3+1, i*3+2
            idx_x2, idx_y2, idx_r2 = j*3, j*3+1, j*3+2
            
            def dist_constraint(p, i=i, j=j):
                dx = p[idx_x1] - p[idx_x2]
                dy = p[idx_y1] - p[idx_y2]
                dr = p[idx_r1] + p[idx_r2]
                return dx**2 + dy**2 - dr**2
            constraints.append({'type': 'ineq', 'fun': dist_constraint})

    # Bounds for variables (loose bounds, constraints handle tight ones)
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r
        ])

    # --- 3. Run Optimization ---
    # SLSQP is a local solver, so initialization matters. 
    # We used a hexagonal grid which is a very good start.
    
    try:
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-12}
        )
        
        if res.success:
            params = res.x
        else:
            # If optimization fails, keep the best feasible point found or fallback
            # Usually SLSQP returns a result even if not perfect success
            params = res.x
    except Exception as e:
        print(f"Optimization failed: {e}")
        # Fallback to initial configuration
        params = x0

    # --- 4. Extract Results ---
    final_centers = np.zeros((n_circles, 2))
    final_radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        final_centers[i, 0] = params[i*3]
        final_centers[i, 1] = params[i*3+1]
        final_radii[i] = params[i*3+2]
        
    # Clamp small negative radii due to numerical error
    final_radii = np.maximum(final_radii, 0.0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
