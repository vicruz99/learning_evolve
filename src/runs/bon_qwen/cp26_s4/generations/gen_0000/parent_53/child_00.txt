import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    """
    Given fixed centers, compute the maximum possible radii using Linear Programming.
    Maximizes sum(r_i) s.t. r_i + r_j <= dist(i,j) and r_i <= distance to boundary.
    """
    n = centers.shape[0]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 0)

    # Distance to square boundaries
    bd = np.minimum(
        np.minimum(centers[:, 0], 1 - centers[:, 0]),
        np.minimum(centers[:, 1], 1 - centers[:, 1])
    )

    # LP Setup: minimize -sum(r_i)
    c = -np.ones(n)
    
    # Pairwise constraints: r_i + r_j <= dists[i,j]
    i_idx, j_idx = np.triu_indices(n, k=1)
    m_pairs = len(i_idx)
    A_pairs = np.zeros((m_pairs, n))
    A_pairs[np.arange(m_pairs), i_idx] = 1
    A_pairs[np.arange(m_pairs), j_idx] = 1
    b_pairs = dists[i_idx, j_idx]

    # Boundary constraints: r_i <= bd[i]
    A_bd = np.eye(n)
    b_bd = bd

    # Combine constraints
    A_ub = np.vstack([A_pairs, A_bd])
    b_ub = np.concatenate([b_pairs, b_bd])

    bounds = [(0, None)] * n
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = np.maximum(res.x, 0)
        return radii, np.sum(radii)
    
    # Fallback for numerical edge cases
    return np.full(n, 1e-7), 1e-7 * n

def objective(centers_flat):
    """Objective function for center optimization: negative sum of max radii."""
    centers = centers_flat.reshape(26, 2)
    centers = np.clip(centers, 0.0, 1.0)
    _, val = solve_radii(centers)
    return -val

def run_packing():
    n = 26
    best_val = -np.inf
    best_centers = None
    
    rng = np.random.default_rng(42)
    
    # Multi-start optimization to escape local optima
    for _ in range(10):
        # Initialize on a hexagonal lattice pattern
        centers = np.zeros((n, 2))
        idx = 0
        y = 0.1
        row = 0
        while idx < n:
            x = 0.1 + (row % 2) * 0.05
            while idx < n and x <= 0.9:
                centers[idx] = [x, y]
                idx += 1
                x += 0.1732  # approx sqrt(3)/10
            y += 0.15
            row += 1
            
        # Add stochastic perturbation
        centers += rng.uniform(-0.04, 0.04, size=centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        
        # Local refinement using gradient-free friendly L-BFGS-B
        res = minimize(objective, centers.flatten(), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * (2 * n),
                       options={'maxiter': 800, 'ftol': 1e-12})
                       
        if -res.fun > best_val:
            best_val = -res.fun
            best_centers = res.x.reshape(n, 2)
            
    # Final radii computation on best centers
    radii, sum_radii = solve_radii(best_centers)
    return best_centers, radii, sum_radii
