import numpy as np

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses an iterative inflation and repulsion strategy.
    """
    n = 26
    
    # --- 1. Initialization ---
    # Place circles in a grid pattern that approximates a hexagonal packing.
    # We use 5 columns and 6 rows (30 spots), taking the first 26.
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    cols = 5
    rows = 6
    
    # Spacing to fit circles inside [0,1] with some margin initially
    # We start with a small radius, so spacing doesn't need to be huge
    x_spacing = 1.0 / (cols + 1)
    y_spacing = 1.0 / (rows + 1)
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx < n:
                centers[idx, 0] = (c + 1) * x_spacing
                centers[idx, 1] = (r + 1) * y_spacing
                radii[idx] = 0.02  # Start with small radius
                idx += 1
            else:
                break
        if idx >= n:
            break
            
    # --- 2. Optimization Loop ---
    iterations = 8000
    growth_rate = 0.0004  # Initial growth step
    decay = 0.9995        # Decay factor for growth rate
    
    # Precompute indices for pairs to avoid checking all pairs redundantly if needed,
    # but vectorized full matrix is faster for N=26.
    
    for it in range(iterations):
        # A. Grow radii uniformly
        radii += growth_rate
        
        # B. Resolve Constraints (Repulsion and Walls)
        # We perform multiple sub-steps of repulsion to stabilize positions
        for _ in range(3):
            # Vectorized distance calculation
            # diff[i, j] = centers[i] - centers[j]
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # Shape (n, n, 2)
            dist_sq = np.sum(diff**2, axis=2)
            dist = np.sqrt(dist_sq + 1e-12) # Avoid division by zero
            
            # Required distance (sum of radii)
            r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
            
            # Overlap amount: positive if dist < r_sum
            overlap = r_sum - dist
            
            # We only care about overlaps (overlap > 0)
            # Also, to prevent double counting forces in a symmetric way that might cancel out 
            # or cause issues, we can just apply forces based on overlap.
            # However, applying force F_ij on i and -F_ij on j is standard.
            # Since the matrix is symmetric, summing over all j for each i works.
            
            # Force direction: unit vector from j to i (which is diff / dist)
            # Note: diff is (c_i - c_j). So diff points from j to i.
            # If overlap > 0, i is too close to j. i should move away from j (direction of diff).
            # j should move away from i (direction of -diff).
            
            # Normalize diff
            # Add epsilon to dist to avoid div by zero
            safe_dist = np.where(dist < 1e-9, 1e-9, dist)
            norm_diff = diff / safe_dist[:, :, np.newaxis]
            
            # Force magnitude proportional to overlap
            # Use a stiffness factor. If overlap is large, force is large.
            force_mag = np.maximum(0, overlap)
            
            # Force vector on i from j
            # F_ij = force_mag * norm_diff
            # This contributes to the net force on i.
            forces_ij = force_mag * norm_diff # Shape (n, n, 2)
            
            # Net force on each circle i is sum over all j
            net_force = np.sum(forces_ij, axis=1) # Shape (n, 2)
            
            # Move centers
            # Step size scaling. 
            # If overlap is exactly resolved, shift should be overlap/2.
            # Here we use a fraction to allow dynamic adjustment.
            move_step = 0.5 
            centers += move_step * net_force
            
            # C. Wall Constraints
            # Ensure circles stay within [0, 1] considering their radius
            # x - r >= 0  => x >= r
            # x + r <= 1  => x <= 1 - r
            
            # Lower bound
            min_pos = radii[:, np.newaxis]
            centers = np.maximum(centers, min_pos)
            
            # Upper bound
            max_pos = 1.0 - radii[:, np.newaxis]
            centers = np.minimum(centers, max_pos)
            
        # Decay growth rate to settle into local optimum
        growth_rate *= decay
        
    # --- 3. Final Validation and Cleanup ---
    # Ensure no negative radii (should not happen)
    radii = np.maximum(radii, 0)
    
    # One final check to ensure valid coordinates (just in case)
    # Clamp centers again
    centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
    
    total_sum = np.sum(radii)
    
    return centers, radii, total_sum
