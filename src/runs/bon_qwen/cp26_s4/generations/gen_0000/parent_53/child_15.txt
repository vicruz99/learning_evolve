import numpy as np
from scipy.optimize import minimize

def run_packing():

    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n_circles = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Helper for objective function
    def objective(x):
        centers = x[:n_circles*2].reshape(n_circles, 2)
        radii = x[n_circles*2:]
        return -np.sum(radii)

    # Helper for non-linear constraints (circle overlaps)
    def overlap_constraints(x):
        centers = x[:n_circles*2].reshape(n_circles, 2)
        radii = x[n_circles*2:]
        constraints_vals = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints_vals.append(dist - radii[i] - radii[j])
        return constraints_vals

    # Helper for linear constraints (boundary)
    def boundary_constraints(x):
        centers = x[:n_circles*2].reshape(n_circles, 2)
        radii = x[n_circles*2:]
        constraints_vals = []
        for i in range(n_circles):
            x_c, y_c = centers[i]
            r = radii[i]
            # x - r >= 0
            constraints_vals.append(x_c - r)
            # 1 - x - r >= 0
            constraints_vals.append(1.0 - x_c - r)
            # y - r >= 0
            constraints_vals.append(y_c - r)
            # 1 - y - r >= 0
            constraints_vals.append(1.0 - y_c - r)
        return constraints_vals

    # Setup bounds for all variables
    # Centers in [0.01, 0.99] to give solver some breathing room initially
    # Radii in [0, 0.5]
    lb = [0.01] * (n_circles * 2) + [0.0] * n_circles
    ub = [0.99] * (n_circles * 2) + [0.5] * n_circles
    bounds = list(map(tuple, zip(lb, ub)))

    # Define constraints
    cons = [
        {'type': 'ineq', 'fun': overlap_constraints},
        {'type': 'ineq', 'fun': boundary_constraints}
    ]

    random_seed = 42
    for _ in range(15):
        np.random.seed(random_seed)

        radii = np.full(n_circles, 0.09)

            col = i % 6
            centers[i, 0] = 0.1 + col * 0.15 + np.random.normal(0, 0.01)
            centers[i, 1] = 0.1 + row * 0.15 + np.random.normal(0, 0.01)

        # Initial guess vector
        x0 = np.concatenate([centers.flatten(), radii])

        # Optimize
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,

        if res.success:
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x[:n_circles*2].reshape(n_circles, 2)

    # If optimization was completely unsuccessful or low, fallback to grid

        radii = np.full(n_circles, 0.095)

            centers[i, 0] = 0.1 + (i % 6) * 0.15
            centers[i, 1] = 0.1 + (i // 6) * 0.15
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, float(best_sum)
