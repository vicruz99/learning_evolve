import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # --- 1. Initialization ---
    # Create a 5x5 grid of centers (25 points) and add one extra point
    # Grid points at 0.1, 0.3, 0.5, 0.7, 0.9
    grid_points = []
    coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    for y in coords:
        for x in coords:
            grid_points.append([x, y])
    
    # Add 26th point. Place it in a "hole" or edge to allow optimization to work.
    # (0.2, 0.2) is in the center of the square formed by (0.1,0.1), (0.3,0.1), etc.
    grid_points.append([0.2, 0.2])
    
    centers_init = np.array(grid_points)
    
    # Initial radii small to ensure no overlaps initially
    radii_init = np.full(n, 0.02)
    
    # Flatten into optimization vector: [x0, y0, r0, x1, y1, r1, ...]
    # Actually, grouping by variable type might be easier for slicing, 
    # but interleaved is fine. Let's use interleaved: [x0, y0, r0, x1, y1, r1, ...]
    x0 = centers_init[:, 0]
    y0 = centers_init[:, 1]
    r0 = radii_init
    
    # Interleaved vector
    v0 = np.zeros(3 * n)
    v0[0::3] = x0
    v0[1::3] = y0
    v0[2::3] = r0
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # --- 2. Objective and Constraints ---
    
    def objective(v):
        # Maximize sum of radii => Minimize negative sum
        radii = v[2::3]
        return -np.sum(radii)

    def constraints_fun(v):
        # Returns a vector of constraint values that must be >= 0
        # We will collect them in a list and concatenate
        cons_list = []
        
        centers = np.zeros((n, 2))
        centers[:, 0] = v[0::3]
        centers[:, 1] = v[1::3]
        radii = v[2::3]
        
        # 1. Boundary constraints:
        # x - r >= 0
        # y - r >= 0
        # 1 - x - r >= 0
        # 1 - y - r >= 0
        # Flattened: 4 * n constraints
        
        # x >= r  => x - r >= 0
        cons_list.append(v[0::3] - radii)
        # y >= r  => y - r >= 0
        cons_list.append(v[1::3] - radii)
        # 1 - x >= r => 1 - x - r >= 0
        cons_list.append(1.0 - v[0::3] - radii)
        # 1 - y >= r => 1 - y - r >= 0
        cons_list.append(1.0 - v[1::3] - radii)
        
        # 2. Non-overlap constraints:
        # dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
        # We only need to check pairs i < j
        
        # Vectorized computation for pairs
        # Create arrays for all pairs
        # This might be memory heavy for large N, but for 26 it's small (325 pairs)
        
        # Using broadcasting or loops? Loop is simpler and fast enough for 325 iterations
        for i in range(n):
            for j in range(i + 1, n):
                # Centers
                xi, yi = v[3*i], v[3*i+1]
                xj, yj = v[3*j], v[3*j+1]
                ri, rj = v[3*i+2], v[3*j+2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                cons_val = dist_sq - sum_r**2
                cons_list.append(cons_val)
        
        return np.concatenate(cons_list)

    # Constraint definition for scipy
    cons = {'type': 'ineq', 'fun': constraints_fun}

    # --- 3. Optimization ---
    
    # Run optimizer
    # SLSQP is good for bounded constrained optimization
    res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})
    
    # --- 4. Post-processing ---
    
    # Extract best result
    best_v = res.x
    
    best_centers = np.zeros((n, 2))
    best_centers[:, 0] = best_v[0::3]
    best_centers[:, 1] = best_v[1::3]
    best_radii = best_v[2::3]
    
    # Clip radii to be non-negative (should be enforced by bounds/constraints but safety check)
    best_radii = np.maximum(best_radii, 0.0)
    
    # Calculate sum
    total_sum = np.sum(best_radii)
    
    # Validation (internal check)
    # The validation function provided in prompt is strict, we assume solver respects it.
    # However, numerical noise might occur.
    # If the result is invalid, we might want to try a restart or fallback, 
    # but usually SLSQP handles it well.
    
    return best_centers, best_radii, total_sum
