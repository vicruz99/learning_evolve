from scipy.optimize import minimize

def _objective(x):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(x[2::3])

def _constraint_func(v):

    res = []

    for i in range(n):
        xi, yi, ri = 3*i, 3*i+1, 3*i+2

        res.append(v[xi] - v[ri])
        # 1 - x - r >= 0
        res.append(1.0 - v[xi] - v[ri])
        # y - r >= 0
        res.append(v[yi] - v[ri])
        # 1 - y - r >= 0
        res.append(1.0 - v[yi] - v[ri])

            dy = v[yi] - v[yj]

            res.append(dx*dx + dy*dy - (v[ri] + v[rj])**2)
