import numpy as np
from scipy.optimize import minimize

def get_overlap_penalty(centers, radii):
    """
    Calculates a penalty score based on overlaps and boundary violations.
    Returns 0 if valid.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary penalties
    # Constraint: r <= x <= 1-r  =>  x >= r, x <= 1-r
    # Constraint: r <= y <= 1-r  =>  y >= r, y <= 1-r
    # Violation amount: max(0, r - x), max(0, x + r - 1), etc.
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary
        if x < r:
            penalty += (r - x)**2
        # Right boundary
        if x + r > 1:
            penalty += (x + r - 1)**2
        # Bottom boundary
        if y < r:
            penalty += (r - y)**2
        # Top boundary
        if y + r > 1:
            penalty += (y + r - 1)**2
            
    # Pairwise overlap penalties
    # Constraint: dist >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist)**2
                
    return penalty
