import numpy as np
from scipy.optimize import minimize
import math

def hex_init_guess(n):
    """Generate an initial hexagonal packing guess for n circles."""
    s = math.sqrt(2.0 / (n * math.sqrt(3.0)))
    points = []
    range_val = int(1.5 / s) + 3
    for j in range(range_val):
        for i in range(range_val):
            x = i * s + (j % 2) * s / 2.0
            y = j * s * math.sqrt(3.0) / 2.0
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
    
    # Fallback to grid if hex grid is too sparse for n
    if len(points) < n:
        grid_size = int(math.ceil(math.sqrt(n)))
        dx = 1.0 / grid_size
        dy = 1.0 / grid_size
        for i in range(grid_size + 1):
            for j in range(grid_size + 1):
                points.append((i * dx, j * dy))
                
    # Select n points closest to the center
    dists = [math.hypot(p[0] - 0.5, p[1] - 0.5) for p in points]
    idx = np.argsort(dists)[:n]
    selected = [points[k] for k in idx]
    
    r_init = s / 2.0 * 0.8
    centers = np.array(selected)
    radii = np.full(n, r_init)
    return centers, radii

def objective(vars, n):
    """Objective function: minimize negative sum of radii."""
    r = vars[2::3]
    return -np.sum(r)

def constraints(vars, n):
    """Constraint function: returns array of values that must be >= 0."""
    cons = []
    # Boundary constraints
    for i in range(n):
        x = vars[3 * i]
        y = vars[3 * i + 1]
        r = vars[3 * i + 2]
        cons.append(x - r)
        cons.append(1.0 - x - r)
        cons.append(y - r)
        cons.append(1.0 - y - r)
        
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
            xj, yj, rj = vars[3 * j], vars[3 * j + 1], vars[3 * j + 2]
            dist_sq = (xi - xj) ** 2 + (yi - yj) ** 2
            r_sum = ri + rj
            cons.append(dist_sq - r_sum ** 2)
            
    return np.array(cons)

def run_packing():
    n = 26
    best_obj = np.inf
    best_x = None
    
    # Run multiple trials with different seeds
    for seed in range(5):
        np.random.seed(seed)
        centers, radii = hex_init_guess(n)
        
        # Add small random perturbation to break symmetry
        centers += np.random.uniform(-0.02, 0.02, centers.shape)
        centers = np.clip(centers, 0.01, 0.99)
        radii += np.random.uniform(-0.01, 0.01, radii.shape)
        radii = np.clip(radii, 0.01, 0.4)
        
        # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3 * i] = centers[i, 0]
            x0[3 * i + 1] = centers[i, 1]
            x0[3 * i + 2] = radii[i]
            
        bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
        
        # Define constraints without using lambdas
        cons = {'type': 'ineq', 'fun': constraints, 'args': (n,)}
        
        try:
            res = minimize(
                objective, x0, args=(n,), method='SLSQP', 
                bounds=bounds, constraints=cons, 
                options={'ftol': 1e-9, 'maxiter': 2000}
            )
            
            if np.isfinite(res.fun) and res.fun < best_obj:
                best_obj = res.fun
                best_x = res.x
        except Exception:
            continue
            
    # Fallback if optimization fails completely
    if best_x is None:
        centers, radii = hex_init_guess(n)
        return centers, radii, np.sum(radii)
        
    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [best_x[3 * i], best_x[3 * i + 1]]
        radii[i] = best_x[3 * i + 2]
        
    # Ensure strict validity
    radii = np.maximum(radii, 0.0)
    centers[:, 0] = np.clip(centers[:, 0], 0, 1)
    centers[:, 1] = np.clip(centers[:, 1], 0, 1)
    
    return centers, radii, np.sum(radii)
