import numpy as np
from scipy.optimize import minimize

def compute_overlap(centers, r):
    """
    Computes the overlap penalty energy and its gradient with respect to centers.
    """
    n = centers.shape[0]
    energy = 0.0
    grad = np.zeros_like(centers)
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff**2))
            
            if dist < 2 * r:
                val = 2 * r - dist
                energy += val**2
                
                if dist > 1e-9:
                    factor = -2.0 * val / dist
                    grad[i] += factor * diff
                    grad[j] -= factor * diff
                    
    return energy, grad

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Binary search bounds for radius
    r_low, r_high = 0.080, 0.115
    best_r = r_low
    best_centers = None
    
    # Binary search iterations for high precision
    for _ in range(20):
        r_mid = (r_low + r_high) / 2.0
        bounds = [(r_mid, 1.0 - r_mid)] * (n * 2)
        
        best_energy = np.inf
        best_cfg = None
        
        # Generate diverse initial configurations to escape local minima
        inits = []
        
        # 1. Structured Grid Initialization
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        c_grid = np.zeros((n, 2))
        idx = 0
        for i in range(rows):
            for j in range(cols):
                if idx < n:
                    c_grid[idx, 0] = r_mid + (j + 0.5) * (1.0 - 2.0 * r_mid) / cols
                    c_grid[idx, 1] = r_mid + (i + 0.5) * (1.0 - 2.0 * r_mid) / rows
                    idx += 1
        inits.append(c_grid)
        
        # 2 & 3. Random Initializations
        inits.append(np.random.uniform(r_mid + 0.001, 1.0 - r_mid - 0.001, (n, 2)))
        inits.append(np.random.uniform(r_mid + 0.001, 1.0 - r_mid - 0.001, (n, 2)))
        
        # Optimize over each initialization
        for init in inits:
            # Perturb slightly
            x0 = init + np.random.uniform(-0.002, 0.002, init.shape)
            x0 = np.clip(x0, r_mid + 1e-5, 1.0 - r_mid - 1e-5)
            
            res = minimize(compute_overlap, x0.flatten(), args=(r_mid,), 
                           method='L-BFGS-B', jac=True, bounds=bounds,
                           options={'maxiter': 1500, 'ftol': 1e-11, 'gtol': 1e-8})
            
            if res.fun < best_energy:
                best_energy = res.fun
                best_cfg = res.x.reshape(n, 2)

        if best_energy < 1e-5:
            best_r = r_mid
            best_centers = best_cfg
            r_low = r_mid  # Try larger radius
        else:
            r_high = r_mid  # Must reduce radius
            
    # Fallback if search fails (unlikely)
    if best_centers is None:
        best_r = 0.05
        best_centers = np.random.uniform(best_r, 1.0 - best_r, (n, 2))
        
    radii = np.full(n, best_r)
    sum_radii = np.sum(radii)
    
    return best_centers, radii, sum_radii
