import numpy as np
from scipy.optimize import linprog, differential_evolution
import warnings

# Suppress warnings from scipy
warnings.filterwarnings('ignore')

def solve_lp_for_radii(centers):
    """
    Solves the Linear Programming problem to find the maximum sum of radii 
    for a fixed set of centers.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        sum_radii: float
        radii: np.array of shape (n)
        success: bool
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0, np.array([]), True

    # Variables: r_1, r_2, ..., r_n
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = -np.ones(n)

    # Constraints matrix A_ub @ r <= b_ub
    # We will build this list of constraints
    A_ub_list = []
    b_ub_list = []

    # 1. Boundary Constraints:
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    # This can be represented as 1*r_i <= val
    
    # Pre-allocate for speed if possible, but list append is fast enough for n=26
    # Constraints:
    # r_i <= x_i  => r_i - 0 <= x_i
    # r_i <= 1-x_i
    # r_i <= y_i
    # r_i <= 1-y_i
    
    # We can add these row by row
    for i in range(n):
        # Row for r_i <= x_i
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(centers[i, 0])
        
        # Row for r_i <= 1 - x_i
        A_ub_list.append(row)
        b_ub_list.append(1.0 - centers[i, 0])
        
        # Row for r_i <= y_i
        A_ub_list.append(row)
        b_ub_list.append(centers[i, 1])
        
        # Row for r_i <= 1 - y_i
        A_ub_list.append(row)
        b_ub_list.append(1.0 - centers[i, 1])

    # 2. Pairwise Constraints:
    # r_i + r_j <= dist(i, j)
    # 1*r_i + 1*r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)

    # Convert to numpy arrays
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    # Bounds for variables r_i >= 0
    bounds = [(0, None) for _ in range(n)]

    try:
        # Use 'highs' method for speed if available, else 'simplex'
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            return -res.fun, radii, True
        else:
            # Fallback or error handling
            # Return a small valid radius if LP fails (e.g. infeasible due to numerical issues)
            # Though with r>=0 it should be feasible.
            return 0.0, np.zeros(n), False
    except Exception:
        return 0.0, np.zeros(n), False

def negative_sum_radii(centers_flat):
    """
    Objective function for the center optimizer.
    Minimizes the negative sum of radii.
    """
    n = 26
    centers = centers_flat.reshape((n, 2))
    sum_radii, radii, success = solve_lp_for_radii(centers)
    
    # Penalize if centers are out of bounds (though DE handles bounds)
    # Just in case
    if np.any(centers < -1e-9) or np.any(centers > 1.0 + 1e-9):
        return 1e6 

    return -sum_radii
