import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP optimization starting from a valid grid packing.
    """
    n = 26
    
    # --- 1. Initialization ---
    # Start with a valid grid packing of equal circles.
    # A radius of 0.08 allows 6 columns (width 0.96) and 5 rows (height 0.80).
    # This gives 30 positions, we take the first 26.
    r_init = 0.08
    
    # Generate grid coordinates
    # x centers: r, r+2r, r+4r, ... -> r(1 + 2k)
    # We need 6 columns: k=0..5. Last x = 0.08 * (1 + 10) = 0.88. 
    # Circle extent 0.88 + 0.08 = 0.96 <= 1. OK.
    xs = r_init * (1 + 2 * np.arange(6))
    # y centers: r, r+2r, ...
    # We need 5 rows: k=0..4. Last y = 0.08 * (1 + 8) = 0.72.
    # Circle extent 0.72 + 0.08 = 0.80 <= 1. OK.
    ys = r_init * (1 + 2 * np.arange(5))
    
    centers = []
    radii = []
    count = 0
    
    # Fill grid row by row
    for y in ys:
        for x in xs:
            if count < n:
                centers.append([x, y])
                radii.append(r_init)
                count += 1
                
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Flatten into optimization variables vector
    # Structure: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
        
    # --- 2. Bounds ---
    # x, y in [0, 1], r in [0, 0.5] (0.5 is max possible radius in unit square)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # --- 3. Constraints ---
    constraints = []
    
    # Boundary constraints: Circle i must be inside [0,1]x[0,1]
    # r <= x <= 1-r  =>  x - r >= 0  and  1 - x - r >= 0
    # r <= y <= 1-r  =>  y - r >= 0  and  1 - y - r >= 0
    for i in range(n):
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: v[3 * idx] - v[3 * idx + 2]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: 1.0 - v[3 * idx] - v[3 * idx + 2]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: v[3 * idx + 1] - v[3 * idx + 2]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: 1.0 - v[3 * idx + 1] - v[3 * idx + 2]
        })
        
    # Overlap constraints: dist(i, j) >= r_i + r_j
    # Using squared distance to keep it differentiable and avoid sqrt:
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # => (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: 
                    (v[3 * i] - v[3 * j])**2 + 
                    (v[3 * i + 1] - v[3 * j + 1])**2 - 
                    (v[3 * i + 2] + v[3 * j + 2])**2
            })

    # --- 4. Objective Function ---
    # Maximize sum of radii <=> Minimize -sum(radii)
    def objective(v):
        return -np.sum(v[2::3])

    # --- 5. Optimization ---
    # Using SLSQP method
    try:
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={
                'maxiter': 2000,
                'ftol': 1e-12,
                'disp': False
            }
        )
        
        # Extract results
        final_centers = np.zeros((n, 2))
        final_radii = np.zeros(n)
        
        for i in range(n):
            final_centers[i, 0] = res.x[3 * i]
            final_centers[i, 1] = res.x[3 * i + 1]
            final_radii[i] = res.x[3 * i + 2]
            
    except Exception as e:
        # Fallback to initial grid if optimization fails
        final_centers = centers
        final_radii = radii

    # --- 6. Post-processing / Validation ---
    # Ensure non-negative radii (though bounds should handle it)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Ensure centers are within [0,1] (clamping just in case of numerical issues)
    # Note: Constraints should enforce x in [r, 1-r], so x in [0,1] is implied if r>=0
    # But strict clamping might be safer if constraints were violated slightly.
    # However, validate_packing checks strict constraints.
    # Let's trust the optimizer but maybe clean up tiny violations.
    
    # Simple iterative repair for any tiny constraint violations due to numerical precision
    # This helps pass the validation function which has a tolerance of 1e-12.
    
    # Repair overlap
    for _ in range(10): # Iterate a few times
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                r1 = final_radii[i]
                r2 = final_radii[j]
                c1 = final_centers[i]
                c2 = final_centers[j]
                
                dist = np.sqrt(np.sum((c1 - c2)**2))
                min_dist = r1 + r2
                
                if dist < min_dist - 1e-14:
                    # Overlap detected, push apart
                    overlap = min_dist - dist
                    # Move each circle away by half the overlap
                    if dist > 1e-12:
                        direction = (c1 - c2) / dist
                        move = 0.5 * overlap + 1e-8
                        final_centers[i] += direction * move
                        final_centers[j] -= direction * move
                        changed = True
                    else:
                        # Centers coincident, push randomly? Or just one?
                        # With r > 0, this shouldn't happen often if valid.
                        final_centers[i] += [1e-6, 0]
        
        if not changed:
            break

    # Repair boundary
    for i in range(n):
        r = final_radii[i]
        x, y = final_centers[i]
        
        # x constraint
        if x - r < 0:
            final_centers[i, 0] = r
        if x + r > 1:
            final_centers[i, 0] = 1 - r
            
        # y constraint
        if y - r < 0:
            final_centers[i, 1] = r
        if y + r > 1:
            final_centers[i, 1] = 1 - r

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
