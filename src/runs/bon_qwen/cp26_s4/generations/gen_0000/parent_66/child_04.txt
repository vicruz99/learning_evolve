import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- Helper Functions ---

    def get_initial_guesses():
        """Generates diverse initial configurations for the optimizer."""
        guesses = []
        
        # 1. Hexagonal Lattice Initialization
        # High density packing structure. Estimated r ~ 0.105
        r_est = 0.105
        rows = 6
        cols = 6
        centers = []
        for r in range(rows):
            for c in range(cols):
                x = 0.1 + c * 2 * r_est + (r % 2) * r_est
                y = 0.1 + r * np.sqrt(3) * r_est
                if x <= 0.9 and y <= 0.9:
                    centers.append([x, y])
        
        # Pad or trim to exactly 26
        if len(centers) < n:
            # Add random points if lattice doesn't yield 26
            while len(centers) < n:
                centers.append([np.random.rand(), np.random.rand()])
        else:
            centers = centers[:n]
        
        guesses.append(np.array(centers))

        # 2. Random Uniform Initialization
        guesses.append(np.random.rand(n, 2))

        # 3. Grid Initialization
        x_grid = np.linspace(0.1, 0.9, 6)
        y_grid = np.linspace(0.1, 0.9, 6)
        centers = []
        for y in y_grid:
            for x in x_grid:
                if len(centers) < n:
                    centers.append([x, y])
        guesses.append(np.array(centers))
        
        return guesses

    def objective_and_constraints(v):
        """
        Computes objective and constraints for the optimizer.
        v: flattened array of [x1, y1, r1, x2, y2, r2, ...]
        """
        centers = v[:2*n].reshape(n, 2)
        radii = v[2*n:]
        
        # Objective: Maximize sum of radii => Minimize -sum
        obj = -np.sum(radii)
        
        # Constraints list
        cons = []
        
        # 1. Boundary Constraints
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        for i in range(n):
            # x >= r
            cons.append({
                'fun': lambda v, i=i: v[3*i] - v[3*i+2],
                'jac': lambda v, i=i: np.zeros_like(v) if False else (lambda v, i=i: np.zeros_like(v)) # Placeholder for jac logic below
            })
            # ... (SLSQP with dictionary constraints is verbose, we will use a vectorized approach below)
            pass

        return obj, centers, radii

    # --- Optimizer Logic ---

    best_sum_radii = -1.0
    best_centers = None
    best_radii = None

    # We will use a custom wrapper to handle SLSQP constraints efficiently
    # SLSQP handles bounds and constraints.
    # Variables: [x_0, y_0, r_0, ..., x_25, y_25, r_25]
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Constraint functions for SLSQP
    # We define a function that returns the vector of constraint values
    # And another for the Jacobian
    
    def constraint_fun(v):
        centers = v[:2*n].reshape(n, 2)
        radii = v[2*n:]
        c_list = []
        
        # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
        for i in range(n):
            c_list.append(centers[i, 0] - radii[i])       # x - r >= 0
            c_list.append(1.0 - centers[i, 0] - radii[i]) # 1 - x - r >= 0
            c_list.append(centers[i, 1] - radii[i])       # y - r >= 0
            c_list.append(1.0 - centers[i, 1] - radii[i]) # 1 - y - r >= 0
            
        # Overlap constraints: dist >= r_i + r_j
        # Vectorized pairwise distance calculation is tricky in a callback, 
        # but for n=26, loop is acceptable.
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                c_list.append(dist - radii[i] - radii[j])
                
        return np.array(c_list)

    def constraint_jac(v):
        centers = v[:2*n].reshape(n, 2)
        radii = v[2*n:]
        jac = np.zeros((4*n + n*(n-1)//2, 3*n))
        
        row = 0
        # Boundary constraints Jacobian
        for i in range(n):
            idx_x = 3*i
            idx_y = 3*i + 1
            idx_r = 3*i + 2
            
            # x - r >= 0
            jac[row, idx_x] = 1.0
            jac[row, idx_r] = -1.0
            row += 1
            
            # 1 - x - r >= 0
            jac[row, idx_x] = -1.0
            jac[row, idx_r] = -1.0
            row += 1
            
            # y - r >= 0
            jac[row, idx_y] = 1.0
            jac[row, idx_r] = -1.0
            row += 1
            
            # 1 - y - r >= 0
            jac[row, idx_y] = -1.0
            jac[row, idx_r] = -1.0
            row += 1
            
        # Overlap constraints Jacobian
        # dist - r_i - r_j >= 0
        # dist = sqrt(dx^2 + dy^2)
        # d(dist)/dx_i = dx/dist, d(dist)/dx_j = -dx/dist
        # d(dist)/dr_i = -1, d(dist)/dr_j = -1
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff**2))
                
                # Avoid division by zero
                if dist < 1e-9:
                    grad_dist = np.array([0.0, 0.0])
                else:
                    grad_dist = diff / dist
                
                idx_xi = 3*i
                idx_yi = 3*i + 1
                idx_ri = 3*i + 2
                idx_xj = 3*j
                idx_yj = 3*j + 1
                idx_rj = 3*j + 2
                
                # Current row in jac matrix
                r_idx = row
                
                # x_i component
                jac[r_idx, idx_xi] = grad_dist[0]
                # y_i component
                jac[r_idx, idx_yi] = grad_dist[1]
                # r_i component
                jac[r_idx, idx_ri] = -1.0
                
                # x_j component
                jac[r_idx, idx_xj] = -grad_dist[0]
                # y_j component
                jac[r_idx, idx_yj] = -grad_dist[1]
                # r_j component
                jac[r_idx, idx_rj] = -1.0
                
                row += 1
                
        return jac

    cons = ({'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac})

    # Objective function and gradient
    def objective(v):
        radii = v[2*n:]
        return -np.sum(radii)

    def objective_jac(v):
        grad = np.zeros_like(v)
        grad[2*n:] = -1.0
        return grad

    # --- Execution ---
    
    guesses = get_initial_guesses()
    
    # Add a few more random restarts to ensure diversity
    for _ in range(3):
        guesses.append(np.random.rand(n, 2))

    for idx, centers_init in enumerate(guesses):
        # Initialize radii small to ensure feasibility
        radii_init = np.full(n, 0.01)
        v0 = np.concatenate([centers_init.flatten(), radii_init])
        
        try:
            res = minimize(
                objective, 
                v0, 
                method='SLSQP', 
                jac=objective_jac,
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
            )
            
            if res.success or (np.isnan(res.fun) == False):
                centers_opt = res.x[:2*n].reshape(n, 2)
                radii_opt = res.x[2*n:]
                
                # Validate strictly
                # Check for negative radii due to numerical noise
                radii_opt = np.maximum(radii_opt, 0.0)
                
                # Check basic validity manually before returning
                # (The validation function will be run externally, but good to be safe)
                if validate_packing_local(centers_opt, radii_opt):
                    current_sum = np.sum(radii_opt)
                    if current_sum > best_sum_radii:
                        best_sum_radii = current_sum
                        best_centers = centers_opt
                        best_radii = radii_opt
                        
        except Exception:
            continue

    if best_centers is None:
        # Fallback to a simple grid if optimization fails
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.full(n, 0.01)
        best_sum_radii = 0.26

    return best_centers, best_radii, best_sum_radii

def validate_packing_local(centers, radii):
    """Local validation to check validity before updating best solution."""
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < -1e-9: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True
