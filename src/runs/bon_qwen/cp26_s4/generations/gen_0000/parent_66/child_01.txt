import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def generate_hexagonal_initialization(n, r_init):
    """Generates centers for n circles in a hexagonal pattern with radius r_init."""
    centers = []
    y = r_init
    row_idx = 0
    
    # Safety break to prevent infinite loops if logic is flawed
    max_rows = 20
    
    while y + r_init <= 1.0 + 1e-9 and len(centers) < n and row_idx < max_rows:
        # Determine x start based on row parity for staggering
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init # Shift by r (half of 2r spacing)
        
        x = x_start
        while x + r_init <= 1.0 + 1e-9:
            centers.append([x, y])
            if len(centers) == n:
                break
            x += 2 * r_init
        
        y += np.sqrt(3) * r_init
        row_idx += 1
        
    return np.array(centers[:n])

def optimize_packing(n, centers_init, radii_init):
    """
    Optimizes the packing to maximize sum of radii.
    """
    # Initialize variables: x0, y0, r0, x1, y1, r1, ...
    vars_init = np.zeros(3 * n)
    for i in range(n):
        vars_init[3*i] = centers_init[i, 0]
        vars_init[3*i+1] = centers_init[i, 1]
        vars_init[3*i+2] = radii_init[i]

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(1e-6, 0.5)] * n # Small lower bound for r to avoid degeneracy

    # Objective: Maximize sum(r) -> Minimize -sum(r)
    def objective(vars):
        return -np.sum(vars[2::3])

    # Constraints
    constraints = []

    # Boundary constraints: x - r >= 0, 1 - (x + r) >= 0, etc.
    def get_constraints():
        cons = []
        
        # Helper for linear constraints (though SLSQP handles generic fun)
        # x_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v: v[0::3] - v[2::3]})
        # 1 - (x_i + r_i) >= 0
        cons.append({'type': 'ineq', 'fun': lambda v: 1.0 - (v[0::3] + v[2::3])})
        # y_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v: v[1::3] - v[2::3]})
        # 1 - (y_i + r_i) >= 0
        cons.append({'type': 'ineq', 'fun': lambda v: 1.0 - (v[1::3] + v[2::3])})

        # Distance constraints: dist^2 - (r_i + r_j)^2 >= 0
        def dist_constraints(vars):
            xs = vars[0::3]
            ys = vars[1::3]
            rs = vars[2::3]
            vals = []
            for i in range(n):
                for j in range(i + 1, n):
                    dx = xs[i] - xs[j]
                    dy = ys[i] - ys[j]
                    dist_sq = dx*dx + dy*dy
                    sum_r = rs[i] + rs[j]
                    vals.append(dist_sq - sum_r*sum_r)
            return np.array(vals)

        cons.append({'type': 'ineq', 'fun': dist_constraints})
        return cons

    constraints = get_constraints()

    # Run SLSQP
    try:
        res = minimize(objective, vars_init, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 500, 'ftol': 1e-12})

                best_radii[i] = best_vars[3*i+2]
            return best_centers, best_radii
    except Exception:
        pass
        
    return centers_init, radii_init
