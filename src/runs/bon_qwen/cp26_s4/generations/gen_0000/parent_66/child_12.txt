import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Strategy:
    1. Generate a good initial configuration using a hexagonal lattice pattern.
    2. Use SLSQP optimization to maximize sum of radii subject to non-overlap and boundary constraints.
    3. Run multiple optimizations with slight perturbations to escape local minima.
    """
    n = 26
    
    # Helper to generate hexagonal initial guess
    def get_hexagonal_initial_guess(n):
        # Try to fit n circles in a hexagonal grid
        # Estimate radius for equal circles packing
        # Area density ~ 0.9, 26 * pi * r^2 ~ 0.9 => r ~ 0.105
        # Let's try r = 0.1 as a safe start for grid generation
        r_init = 0.1
        dy = np.sqrt(3) * r_init
        
        centers = []
        row = 0
        while len(centers) < n:
            row_len = 5 if row % 2 == 0 else 4 # Alternating 5 and 4
            y = r_init + row * dy
            
            # Check if y fits in square
            if y + r_init > 1.0 + 1e-5:
                # Try to squeeze or stop? 
                # If we can't fit more rows, maybe we stop. 
                # But we need 26 circles.
                # Let's force them in if possible, or just proceed and let optimizer fix.
                pass 
            
            x_start = r_init
            if row % 2 == 1:
                x_start += r_init # Shift for hexagonal
            
            for col in range(row_len):
                x = x_start + col * (2 * r_init)
                if x + r_init <= 1.0 + 1e-5:
                    centers.append([x, y])
                else:
                    # If circle goes out, maybe place it somewhere else?
                    # For initialization, just appending close to boundary is okay
                    centers.append([1.0 - r_init, y])
            
            if len(centers) >= n:
                break
            row += 1
            
        # If we still don't have enough, fill with random
        while len(centers) < n:
            centers.append([np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)])
        
        return np.array(centers[:n])

    best_sum = 0
    best_centers = None
    best_radii = None

    # Run multiple attempts
    for seed in range(10):
        np.random.seed(seed)
        
        # Generate initial centers
        centers = get_hexagonal_initial_guess(n)
        
        # Initial radii: start small to satisfy constraints, then grow
        # Or start with estimated r
        # Let's start with r = 0.05 to be safe, optimizer will increase
        radii = np.ones(n) * 0.05
        
        # Construct initial variable vector
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
        
        # Define objective: Maximize sum of radii -> Minimize negative sum
        def objective(v):
            return -np.sum(v[2::3])
        
        # Define constraints
        # 1. Boundary constraints
        # x_i >= r_i  => x_i - r_i >= 0
        # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
        # Same for y
        
        cons = []
        
        # Boundary constraints
        for i in range(n):
            idx = 3*i
            # x >= r
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx] - v[idx+2]})
            # x <= 1 - r
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx] - v[idx+2]})
            # y >= r
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx+1] - v[idx+2]})
            # y <= 1 - r
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx+1] - v[idx+2]})
            
        # Non-overlap constraints
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_i = 3 * i
                idx_j = 3 * j
                cons.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: 
                        (v[idx_i] - v[idx_j])**2 + (v[idx_i+1] - v[idx_j+1])**2 - (v[idx_i+2] + v[idx_j+2])**2
                })

        # Bounds
        # x, y in [0, 1], r >= 0
        bounds = []
        for i in range(n):
            bounds.extend([
                (0.0, 1.0), # x
                (0.0, 1.0), # y
                (0.0, 1.0)  # r (upper bound 1 is loose but safe)
            ])
        
        # Optimization
        # SLSQP is good for constrained problems
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
        
        if res.success or res.fun < -best_sum:
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = np.zeros((n, 2))
                best_radii = np.zeros(n)
                for i in range(n):
                    best_centers[i, 0] = res.x[3*i]
                    best_centers[i, 1] = res.x[3*i+1]
                    best_radii[i] = res.x[3*i+2]

    # Final validation and return
    # The problem requires non-negative radii and valid centers.
    # The optimizer should have respected bounds.
    
    return best_centers, best_radii, best_sum
