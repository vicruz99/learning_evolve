import numpy as np
from scipy.optimize import minimize


def penalty_objective(params, n, penalty):
    """Penalized objective: minimize negative sum of radii with constraint penalties."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    obj = -np.sum(radii)

    # Boundary penalties: circle must be inside [0,1]x[0,1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left boundary: x >= r
        viol = r - x
        if viol > 0:
            obj += penalty * viol * viol
        # Right boundary: x + r <= 1, i.e., r <= 1 - x
        viol = r - (1.0 - x)
        if viol > 0:
            obj += penalty * viol * viol
        # Bottom boundary: y >= r
        viol = r - y
        if viol > 0:
            obj += penalty * viol * viol
        # Top boundary: y + r <= 1, i.e., r <= 1 - y
        viol = r - (1.0 - y)
        if viol > 0:
            obj += penalty * viol * viol

    # Overlap penalties: distance >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = diff[0] * diff[0] + diff[1] * diff[1]
            if dist_sq > 1e-20:
                dist = np.sqrt(dist_sq)
                viol = radii[i] + radii[j] - dist
                if viol > 0:
                    obj += penalty * viol * viol
            else:
                # Circles are on top of each other
                viol = radii[i] + radii[j]
                obj += penalty * viol * viol

    return obj


def initialize_grid(n, seed):
    """Initialize with 5x5 grid + 1 extra circle, with perturbation."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)

    for i in range(5):
        for j in range(5):
            idx = i * 5 + j
            centers[idx, 0] = 0.1 + j * 0.2 + np.random.uniform(-0.015, 0.015)
            centers[idx, 1] = 0.1 + i * 0.2 + np.random.uniform(-0.015, 0.015)

    centers[25, 0] = 0.5 + np.random.uniform(-0.1, 0.1)
    centers[25, 1] = 0.5 + np.random.uniform(-0.1, 0.1)

    return centers, radii


def initialize_hexagonal(n, seed):
    """Initialize with hexagonal arrangement."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)

    r = 0.08
    row_height = r * np.sqrt(3.0)

    idx = 0
    rows = [5, 6, 5, 6, 4]  # Total: 26 circles

    for row_i, num_circles in enumerate(rows):
        y = r + row_i * row_height
        if row_i % 2 == 0:
            x_start = r
        else:
            x_start = r + r

        for col in range(num_circles):
            if idx < n:
                x = x_start + col * 2 * r
                centers[idx, 0] = x + np.random.uniform(-0.01, 0.01)
                centers[idx, 1] = y + np.random.uniform(-0.01, 0.01)
                idx += 1

    while idx < n:
        centers[idx, 0] = np.random.uniform(0.1, 0.9)
        centers[idx, 1] = np.random.uniform(0.1, 0.9)
        idx += 1

    return centers, radii


def initialize_random(n, seed):
    """Initialize with random placement."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.04)
    return centers, radii


def optimize_packing(centers, radii, n):
    """Optimize packing using penalty method with L-BFGS-B."""
    params = np.concatenate([centers.flatten(), radii])

    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))

    current_params = params.copy()

    # Phase 1: Medium penalty, many iterations
    for iteration in range(250):
        penalty = 200 * (1.08 ** iteration)

        result = minimize(
            penalty_objective,
            current_params,
            args=(n, penalty),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 20, "ftol": 1e-15, "gtol": 1e-12},
        )

        current_params = result.x

    return current_params[: 2 * n].reshape(n, 2), current_params[2 * n:]


def refine_packing(centers, radii, n):
    """Fine-tune packing with high penalty for strict constraint satisfaction."""
    params = np.concatenate([centers.flatten(), radii])

    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))

    current_params = params.copy()

    for iteration in range(100):
        penalty = 5000 * (1.1 ** iteration)

        result = minimize(
            penalty_objective,
            current_params,
            args=(n, penalty),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 10, "ftol": 1e-16, "gtol": 1e-13},
        )

        current_params = result.x

    return current_params[: 2 * n].reshape(n, 2), current_params[2 * n:]


def fix_violations(centers, radii, n):
    """Fix any remaining constraint violations by reducing radii."""
    centers = centers.copy()
    radii = radii.copy()

    # Fix boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if max_r < 0:
            max_r = 0.0
        radii[i] = min(radii[i], max_r)

    # Fix overlap violations
    for _ in range(10):
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(diff[0] * diff[0] + diff[1] * diff[1])
                if dist < radii[i] + radii[j] - 1e-14:
                    # Reduce the larger radius
                    total = radii[i] + radii[j]
                    if total > dist + 1e-14:
                        scale = (dist + 1e-12) / total
                        radii[i] *= np.sqrt(scale)
                        radii[j] *= np.sqrt(scale)

    # Re-fix boundaries after radius changes
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if max_r < 0:
            max_r = 0.0
        radii[i] = min(radii[i], max_r)

    return centers, radii


def run_packing():
    n = 26

    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Try multiple initializations
    init_methods = [initialize_grid, initialize_hexagonal, initialize_random]

    for init_method in init_methods:
        for seed in range(30):
            centers, radii = init_method(n, seed)
            centers_opt, radii_opt = optimize_packing(centers, radii, n)

            # Refine
            centers_opt, radii_opt = refine_packing(centers_opt, radii_opt, n)

            # Fix any violations
            centers_opt, radii_opt = fix_violations(centers_opt, radii_opt, n)

            current_sum = np.sum(radii_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    # Final verification fix
    best_centers, best_radii = fix_violations(best_centers, best_radii, n)
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
