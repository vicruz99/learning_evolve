import numpy as np
from scipy.optimize import minimize


def objective(params, n):
    """Objective: minimize negative sum of radii (i.e., maximize sum of radii)"""
    return -np.sum(params[2 * n :])


class NonOverlapConstraint:
    """Constraint: distance between circle centers >= sum of radii"""

    def __init__(self, n, i, j):
        self.n = n
        self.i = i
        self.j = j

    def __call__(self, params):
        x_i = params[2 * self.i]
        y_i = params[2 * self.i + 1]
        x_j = params[2 * self.j]
        y_j = params[2 * self.j + 1]
        r_i = params[2 * self.n + self.i]
        r_j = params[2 * self.n + self.j]
        dist = np.sqrt((x_i - x_j) ** 2 + (y_i - y_j) ** 2)
        return dist - r_i - r_j


class BoundaryConstraint:
    """Constraint: circle stays inside unit square"""

    def __init__(self, n, i, ctype):
        self.n = n
        self.i = i
        self.ctype = ctype

    def __call__(self, params):
        x = params[2 * self.i]
        y = params[2 * self.i + 1]
        r = params[2 * self.n + self.i]
        if self.ctype == 0:  # x >= r
            return x - r
        elif self.ctype == 1:  # x <= 1 - r
            return 1 - x - r
        elif self.ctype == 2:  # y >= r
            return y - r
        elif self.ctype == 3:  # y <= 1 - r
            return 1 - y - r


def make_constraints(n):
    """Build all constraint dictionaries for n circles"""
    constraints = []
    # Non-overlap constraints: C(n,2) = 325 for n=26
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({"type": "ineq", "fun": NonOverlapConstraint(n, i, j)})
    # Boundary constraints: 4*n = 104 for n=26
    for i in range(n):
        for ct in range(4):
            constraints.append({"type": "ineq", "fun": BoundaryConstraint(n, i, ct)})
    return constraints


def make_bounds(n):
    """Build bounds for all variables"""
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x in [0,1]
        bounds.append((0, 1))  # y in [0,1]
    for i in range(n):
        bounds.append((1e-8, 0.5))  # r in [epsilon, 0.5]
    return bounds


def initial_hex(n):
    """Initial hexagonal grid placement"""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09
    row_counts = [5, 5, 5, 5, 6]
    idx = 0
    for row_i, count in enumerate(row_counts):
        y = 0.12 + row_i * 0.19
        for col_i in range(count):
            x = 0.12 + col_i * 0.17
            if row_i % 2 == 1:
                x += 0.085
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def initial_corner_heavy(n):
    """Initial placement with larger circles in corners"""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07

    # Corner circles (4)
    corner_pos = [(0.15, 0.15), (0.85, 0.15), (0.15, 0.85), (0.85, 0.85)]
    for i, (x, y) in enumerate(corner_pos):
        centers[i] = [x, y]
        radii[i] = 0.14

    # Edge circles (8)
    edge_pos = [

        (0.42, 0.92), (0.58, 0.92),  # top

        (0.92, 0.42), (0.92, 0.58),  # right
    ]
    for i, (x, y) in enumerate(edge_pos):
        centers[4 + i] = [x, y]
        radii[4 + i] = 0.11

    # Interior circles (14)
    interior_pos = [
        (0.3, 0.3), (0.5, 0.3), (0.7, 0.3),

    ]
    for i, (x, y) in enumerate(interior_pos):
        centers[12 + i] = [x, y]
        radii[12 + i] = 0.07

def initial_random(n, seed):
    """Random initial placement"""
    rng = np.random.RandomState(seed)
    centers = rng.rand(n, 2)
    radii = rng.rand(n) * 0.06 + 0.04
    return centers, radii


def penalty_objective(params, n, pw):
    """Penalty-based objective for refinement"""
    centers = params[: 2 * n].reshape((n, 2))
    radii = params[2 * n :]

    obj = -np.sum(radii)
    penalty = 0.0

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += (x - r) ** 2
        if x + r > 1:
            penalty += (x + r - 1) ** 2
        if y - r < 0:
            penalty += (y - r) ** 2
        if y + r > 1:
            penalty += (y + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2

    return obj + pw * penalty


def optimize_from(centers, radii, n):
    """Run optimization from given initial state"""
    params = np.concatenate([centers.flatten(), radii])
    bounds = make_bounds(n)
    constraints = make_constraints(n)

    result = minimize(
        objective,
        params,
        args=(n,),
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": 2000, "ftol": 1e-14, "disp": False},
    )

    return result.x[: 2 * n].reshape((n, 2)), result.x[2 * n :]


def refine_penalty(centers, radii, n):
    """Refine using penalty method with increasing weights"""
    params = np.concatenate([centers.flatten(), radii])
    bounds = make_bounds(n)

    for pw in [500, 2000, 8000, 30000, 100000]:
        result = minimize(
            penalty_objective,
            params,
            args=(n, pw),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 500, "ftol": 1e-16},
        )
        params = result.x

    return params[: 2 * n].reshape((n, 2)), params[2 * n :]


def jitter(centers, radii, n, scale):
    """Add small random perturbation"""
    rng = np.random.RandomState(42)
    new_centers = centers.copy()
    new_radii = radii.copy()
    for i in range(n):
        new_centers[i, 0] += rng.normal(0, scale)
        new_centers[i, 1] += rng.normal(0, scale)
        new_radii[i] *= 1 + rng.normal(0, 0.1)
        # Clip to valid range
        new_centers[i, 0] = np.clip(new_centers[i, 0], 0.01, 0.99)
        new_centers[i, 1] = np.clip(new_centers[i, 1], 0.01, 0.99)
        new_radii[i] = np.clip(new_radii[i], 1e-6, 0.45)
    return new_centers, new_radii
