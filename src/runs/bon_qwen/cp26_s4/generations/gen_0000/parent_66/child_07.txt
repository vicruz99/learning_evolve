import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Try multiple initial configurations with different strategies
    for trial in range(15):
        centers, radii = generate_initial(n, trial)
        centers_opt, radii_opt, sum_r = optimize_packing(centers, radii, n)
        if sum_r > best_sum:
            best_sum = sum_r
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Final refinement on best solution
    centers_refined, radii_refined, sum_refined = optimize_packing(
        best_centers, best_radii, n
    )
    if sum_refined > best_sum:
        best_sum = sum_refined
        best_centers = centers_refined
        best_radii = radii_refined

    return best_centers, best_radii, best_sum


def generate_initial(n, trial):
    """Generate initial configuration based on trial number."""
    if trial < 5:
        return init_hexagonal_v1(n, trial)
    elif trial < 10:
        return init_hexagonal_v2(n, trial)
    else:
        return init_random(n, trial)


def init_hexagonal_v1(n, trial):
    """Hexagonal packing with varying initial radii."""
    r = 0.07 + trial * 0.005
    centers = []
    row_counts = [5, 5, 5, 5, 6]

    for row_idx, count in enumerate(row_counts):
        y = r + row_idx * r * np.sqrt(3)
        for col_idx in range(count):
            x = r + col_idx * 2 * r
            if row_idx % 2 == 1:
                x += r
            if len(centers) < n:
                centers.append([x, y])

    centers = np.array(centers[:n])
    radii = np.array([r] * n)
    return centers, radii


def init_hexagonal_v2(n, trial):
    """Hexagonal packing with different row arrangements."""
    r = 0.08 + trial * 0.003
    centers = []
    # Try different row count patterns
    row_counts = [6, 5, 5, 5, 5]

    for row_idx, count in enumerate(row_counts):
        y = r + row_idx * r * np.sqrt(3)

    centers = np.array(centers[:n])

def init_random(n, trial):
    """Random initialization with some structure."""
    np.random.seed(trial * 100 + 42)
    # Place circles in a grid with some randomness
    rows = 5
    cols = 6
    centers = []
    radii = []

    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = 0.1 + j * 0.85 / (cols - 1) + np.random.randn() * 0.02
            y = 0.1 + i * 0.85 / (rows - 1) + np.random.randn() * 0.02
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            centers.append([x, y])
            radii.append(0.06)

    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    return centers, radii


def optimize_packing(centers, radii, n):
    """Optimize circle packing using SLSQP."""
    x0 = np.concatenate([centers.ravel(), radii])

    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((1e-8, 0.5))  # r

    constraints = build_constraints(n)

    result = minimize(
        objective_func,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": 5000, "ftol": 1e-15, "disp": False},
    )

    x_opt = result.x if result.success else x0

    centers_opt = x_opt[: 3 * n].reshape(n, 2)
    radii_opt = x_opt[3 * n :]

    # Ensure non-negative radii
    radii_opt = np.maximum(radii_opt, 1e-10)

    # Ensure circles are within bounds
    for i in range(n):
        r = radii_opt[i]
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], r, 1.0 - r)
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], r, 1.0 - r)

    return centers_opt, radii_opt, np.sum(radii_opt)


def objective_func(x):
    """Negative sum of radii (minimize this to maximize sum)."""
    return -np.sum(x[2::3])


def constraint_x_ge_r(x, i):
    return x[3 * i] - x[3 * i + 2]


def constraint_x_le_1_minus_r(x, i):
    return 1.0 - x[3 * i] - x[3 * i + 2]


def constraint_y_ge_r(x, i):
    return x[3 * i + 1] - x[3 * i + 2]


def constraint_y_le_1_minus_r(x, i):
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def constraint_no_overlap(x, i, j):
    ii, jj = 3 * i, 3 * j
    dx = x[ii] - x[jj]
    dy = x[ii + 1] - x[jj + 1]
    dist_sq = dx * dx + dy * dy
    r_sum = x[ii + 2] + x[jj + 2]
    return dist_sq - r_sum * r_sum


def build_constraints(n):
    """Build all constraints for the optimization."""
    constraints = []

    # Boundary constraints for each circle
    for i in range(n):
        constraints.append({"type": "ineq", "fun": constraint_x_ge_r, "args": (i,)})
        constraints.append(
            {"type": "ineq", "fun": constraint_x_le_1_minus_r, "args": (i,)}
        )
        constraints.append({"type": "ineq", "fun": constraint_y_ge_r, "args": (i,)})
        constraints.append(
            {"type": "ineq", "fun": constraint_y_le_1_minus_r, "args": (i,)}
        )

    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(
                {"type": "ineq", "fun": constraint_no_overlap, "args": (i, j)}
            )

    return constraints
