import numpy as np

def compute_objective(params, n, penalty_weight):
    """
    Computes the objective function: -sum(radii) + penalty for violations.
    
    Args:
        params: array of shape (3*n,) containing [x1, y1, r1, x2, y2, r2, ...]
        n: number of circles
        penalty_weight: weight for the penalty term
        
    Returns:
        objective value
    """
    # Reshape parameters
    centers = params[:2*n].reshape((n, 2))
    radii = params[2*n:]
    
    # Clip radii to be non-negative (though bounds should handle this, good for safety)
    radii = np.maximum(radii, 0.0)
    
    # Objective: minimize negative sum of radii
    obj = -np.sum(radii)
    
    # Penalty for boundary violations
    # Circles must be inside [0,1]x[0,1]
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => r - (1-x) <= 0
    # y - r >= 0  => r - y <= 0
    # y + r <= 1  => r - (1-y) <= 0
    
    # Vectorized boundary checks
    # r - x
    p1 = np.maximum(radii - centers[:, 0], 0.0)
    # r - (1 - x)
    p2 = np.maximum(radii - (1.0 - centers[:, 0]), 0.0)
    # r - y
    p3 = np.maximum(radii - centers[:, 1], 0.0)
    # r - (1 - y)
    p4 = np.maximum(radii - (1.0 - centers[:, 1]), 0.0)
    
    boundary_penalty = np.sum(p1**2 + p2**2 + p3**2 + p4**2)
    
    # Penalty for overlaps
    # Distance squared between all pairs
    # dist^2 >= (r_i + r_j)^2
    # violation = max(0, (r_i + r_j)^2 - dist^2)
    
    # We can compute pairwise distances efficiently
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # radii sum shape (n, n)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Violation is where r_sum_sq > dist_sq
    # We only care about i < j, but summing all i!=j and dividing by 2 is fine
    # Using a large weight for penalty
    violations = np.maximum(r_sum_sq - dist_sq, 0.0)
    overlap_penalty = np.sum(violations)
    
    total_penalty = penalty_weight * (boundary_penalty + overlap_penalty)
    
    return obj + total_penalty
