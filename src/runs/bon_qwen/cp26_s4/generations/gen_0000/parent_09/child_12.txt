import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """

    # Distribute circles in 5 rows to maximize vertical density

    row_idx = 0

        # Calculate x spacing based on row length
        # We want to center the row
        x_spacing = 1.0 / (count + 1)
        current_x = x_spacing
        
        for _ in range(count):
            centers_init.append([current_x, current_y])
            current_x += x_spacing
        
        # Shift y based on hexagonal packing spacing
        if row_idx % 2 == 1:
            current_y += 0.28 # Smaller vertical step for staggered rows
        else:
            current_y += 0.30
        row_idx += 1
        
    centers_init = np.array(centers_init)
    
    # Initial radii (uniform guess)
    radii_init = np.full(n_circles, 0.08)

    # --- 2. Optimization Objective ---
    def objective(params):
        # params: [x1, y1, r1, x2, y2, r2, ...]
        centers = params[0:n_circles*2].reshape((n_circles, 2))
        radii = params[n_circles*2:]
        
        sum_radii = np.sum(radii)
        penalty = 0.0
        
        # Boundary penalty (circles must be inside [0,1]^2)
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0: penalty += (x - r)**2 * 10000
            if x + r > 1: penalty += (x + r - 1)**2 * 10000

        return -sum_radii + penalty

        bounds.append((0.0, 0.5)) # r

    x0 = np.concatenate([centers_init.flatten(), radii_init])

    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,

    final_radii = np.maximum(final_radii, 0.0)
    
    # Final Sum
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
