import numpy as np
from scipy.optimize import minimize

def compute_cost(centers_flat, r):
    """Computes the sum of squared overlaps for packing validation."""
    centers = centers_flat.reshape(26, 2)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diffs, axis=2)
    mask = np.triu(np.ones((26, 26), dtype=bool), k=1)
    overlaps = np.maximum(0.0, 2.0 * r - dists)
    cost = np.sum(overlaps[mask]**2)
    
    x = centers[:, 0]
    y = centers[:, 1]
    cost += np.sum(np.maximum(0.0, r - x)**2)
    cost += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    cost += np.sum(np.maximum(0.0, r - y)**2)
    cost += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    return cost

def get_initial_configs():
    """Generates deterministic initial configurations for robustness."""
    configs = []
    # 1. Hexagonal lattice (high density)
    hex_c = []
    row = 0
    while len(hex_c) < 26:
        col = 0
        while len(hex_c) < 26:
            x = 0.15 + col * 0.085 * np.sqrt(3)
            y = 0.15 + row * 0.085 * 1.5
            if row % 2 == 1:
                x += 0.085 * np.sqrt(3) / 2
            hex_c.append([x, y])
            col += 1
        row += 1
    configs.append(np.array(hex_c[:26]))
    
    # 2. Regular grid
    grid = []
    for i in range(6):
        for j in range(5):
            if len(grid) < 26:
                grid.append([0.12 + i * 0.14, 0.12 + j * 0.16])
    configs.append(np.array(grid))
    
    # 3. Deterministic random spread
    np.random.seed(42)
    configs.append(np.random.uniform(0.2, 0.8, (26, 2)))
    return configs

def optimize_packing(init_centers):
    """Iteratively maximizes radius while optimizing positions."""
    N = 26
    centers = init_centers.copy()
    r = 0.06
    step = 0.004
    tol = 1e-8
    max_iters = 200
    
    for _ in range(max_iters):
        res = minimize(compute_cost, centers.flatten(), args=(r,), 
                       method='L-BFGS-B', bounds=[(0, 1)]*52,
                       options={'maxiter': 200, 'ftol': 1e-11, 'disp': False})
        centers = res.x.reshape(N, 2)
        
        if res.fun < tol:
            r += step
        else:
            # Backtrack slightly and perturb to escape local minima
            r = max(0.06, r - step * 3)
            centers += np.random.normal(0, 0.02, centers.shape)
            centers = np.clip(centers, 0.01, 0.99)
            
    return centers, r

def run_packing():
    np.random.seed(123)
    best_r = 0.0
    best_centers = None
    
    # Run optimization from multiple starts and keep the best
    for init in get_initial_configs():
        c, r = optimize_packing(init)
        if r > best_r:
            best_r = r
            best_centers = c
            
    centers = best_centers
    
    # Post-processing: compute exact minimum clearance to guarantee validity
    min_d = 1.0
    for i in range(26):
        x, y = centers[i]
        min_d = min(min_d, x, 1.0 - x, y, 1.0 - y)
        for j in range(i + 1, 26):
            d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                        (centers[i, 1] - centers[j, 1])**2)
            if d < min_d:
                min_d = d
                
    # Safety margin for numerical precision in validator
    safe_r = min_d / 2.0 - 1e-9
    radii = np.full(26, safe_r)
    
    return centers, radii, 26 * safe_r
