import numpy as np
from scipy.optimize import minimize

def get_constraints(v):
    """
    Compute constraint violations.
    Returns an array where each element should be >= 0 for a valid packing.
    """
    n = 26
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    
    # Boundary constraints: x-r >= 0, y-r >= 0, 1-x-r >= 0, 1-y-r >= 0
    b = np.concatenate([c - r[:, None], np.ones((n, 1)) - c - r[:, None]]).flatten()
    
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = c[:, 0, None] - c[None, 0, :]
    dy = c[:, 1, None] - c[None, 1, :]
    d2 = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((n, n)), k=1).astype(bool)
    p = d2[mask] - r_sum[mask]**2
    
    return np.concatenate([b, p])

def objective(v):
    """
    Objective function: minimize negative sum of radii.
    """
    n = 26
    return -np.sum(v[2*n:])

def run_packing():
    n = 26
    # Bounds: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_val = -np.inf
    best_v = None

    for seed in range(20):

        xs = np.linspace(0.1, 0.9, 5)

        v0 = np.concatenate([centers.flatten(), radii])
        
        # Run optimization
        res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,

        best_v = np.concatenate([centers.flatten(), radii])
        best_val = np.sum(radii)
