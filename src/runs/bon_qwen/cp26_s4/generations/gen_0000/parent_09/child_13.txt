        # We center this row in [0, 1]
        total_width = (count - 1) * dx + 2 * r_init
        x_start = (1.0 - total_width) / 2.0
        
        for k in range(count):
            x = x_start + k * dx
            centers.append([x, y_current])
            radii.append(r_init)
        
        y_current += dy
        
        # Check if we exceeded height, if so, stop or wrap? 
        # With 6 rows and r=0.09, height ~ 0.09 + 5*0.155 + 0.09 ~ 0.99. Fits.
        
    centers = np.array(centers)
    radii = np.array(radii)
    
    # If we somehow have more/less than 26 (logic check), adjust

    def objective(vars):

        r = vars[2::3] # radii are at indices 2, 5, 8...
        return -np.sum(r)

    def get_centers_radii(vars):
        c_x = vars[0::3]
        c_y = vars[1::3]
        c_r = vars[2::3]
        return c_x, c_y, c_r

    # 1. Boundary constraints for each circle i
    # x_i - r_i >= 0  => x_i - r_i >= 0
    # 1 - x_i - r_i >= 0 => 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0

    # To speed up, we might vectorize, but for 26 circles, explicit loops in constraint funcs are okay.
    # However, SLSQP calls constraints frequently. 
    # It's better to define a single constraint function that returns a vector of values if possible, 
    # but SLSQP expects list of dicts or a single dict with 'fun' returning array.
    # Let's use a single 'fun' that returns all constraint values.
    
    def constraints_vec(vars):
        c_x, c_y, c_r = get_centers_radii(vars)
        val = []
        
        # Boundary constraints (4 per circle)
        # x - r >= 0
        val.extend(c_x - c_r)
        # 1 - x - r >= 0
        val.extend(1.0 - c_x - c_r)
        # y - r >= 0
        val.extend(c_y - c_r)
        # 1 - y - r >= 0
        val.extend(1.0 - c_y - c_r)

        return np.array(val)

    def constraints_jac(vars):

            idx_y = 3 * i + 1

            jac[row, idx_r] = -1.0

            jac[row + 1, idx_x] = -1.0
            jac[row + 1, idx_r] = -1.0

            jac[row + 2, idx_y] = 1.0
            jac[row + 2, idx_r] = -1.0
            
            # 1 - y - r >= 0
            jac[row + 3, idx_y] = -1.0
            jac[row + 3, idx_r] = -1.0

        # Deriv wrt rj: -1

            for j in range(i + 1, n):
                row = row_offset + counter
                counter += 1

                di_y = c_y[i] - c_y[j]

                    continue
                
                unit_x = di_x / d
                unit_y = di_y / d
                
                idx_xi = 3 * i
                idx_yi = 3 * i + 1
                idx_ri = 3 * i + 2
                
                idx_xj = 3 * j
                idx_yj = 3 * j + 1
                idx_rj = 3 * j + 2
                
                # d/dxi
                jac[row, idx_xi] = unit_x
                # d/dyi
                jac[row, idx_yi] = unit_y
                # d/dri
                jac[row, idx_ri] = -1.0
                
                # d/dxj
                jac[row, idx_xj] = -unit_x
                # d/dyj
                jac[row, idx_yj] = -unit_y
                # d/drj
                jac[row, idx_rj] = -1.0
                
        return jac

        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]

    # Optimization
    # Using SLSQP
    cons = {'type': 'ineq', 'fun': constraints_vec, 'jac': constraints_jac}
    
    # Note: 'jac' in constraints dict expects a function that returns the Jacobian matrix.
    # The order of rows in Jacobian must match order of values in 'fun'.
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=[cons], 
                      options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
    
    if result.success:
        final_x = result.x
    else:
        # If optimization fails, fallback to initial or best effort

        final_x = result.x
        # Note: result.x might violate constraints slightly. 
        # We should check validity.

    final_centers = np.column_stack((final_centers_x, final_centers_y))
    
    # Post-processing: Ensure strict validity (clip radii if negative, etc.)
    final_radii = np.maximum(final_radii, 0)

    # We assume the validation function will be called.
    # If our result is invalid (overlaps), we might need to shrink radii slightly.

    # However, numerical precision might cause tiny violations.
    
    # Let's do a quick safety check and shrink if needed
    # Check pairwise overlaps
    # If dist < r1 + r2, shrink both? 
    # Or just scale down all radii by a small factor?
    
    # A robust way: check min distance vs sum radii.
    # Compute pairwise distances
    diff_x = final_centers_x[:, None] - final_centers_x[None, :]

        # Or just rely on optimizer.

        pass
