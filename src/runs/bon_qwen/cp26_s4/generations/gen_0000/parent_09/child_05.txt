import numpy as np
from scipy.optimize import minimize


def smooth_penalty(x):
    """Smooth penalty for positive violations."""
    return 0.5 * (x + np.abs(x)) ** 2


def compute_cost(params, n):
    """Objective: maximize sum of radii with penalties for constraint violations."""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-10)

    cost = -np.sum(radii)
    pw = 50000.0

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Boundary penalties
        cost += pw * smooth_penalty(r - x)
        cost += pw * smooth_penalty(r - (1.0 - x))
        cost += pw * smooth_penalty(r - y)
        cost += pw * smooth_penalty(r - (1.0 - y))

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            cost += pw * smooth_penalty(overlap)

    return cost


def init_hexagonal(n, r):
    """Initialize circles in a hexagonal lattice pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    idx = 0
    row = 0
    while idx < n:
        col = 0
        while col < 20 and idx < n:
            x = r + col * 2.0 * r + (row % 2) * r
            y = r + row * np.sqrt(3.0) * r
            if x + r <= 1.0 and y + r <= 1.0:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1
    return centers, radii


def init_grid(n, r):
    """Initialize circles in a square grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    idx = 0
    for row in range(10):
        for col in range(10):
            if idx >= n:
                break
            x = r + col * 2.0 * r
            y = r + row * 2.0 * r
            if x + r <= 1.0 and y + r <= 1.0:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break
    return centers, radii


def init_corners_grid(n):
    """Initialize with larger corner circles and grid for rest."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # 4 corner circles with larger radius
    r_corner = 0.14
    corner_positions = [
        [r_corner, r_corner],
        [1 - r_corner, r_corner],
        [r_corner, 1 - r_corner],
        [1 - r_corner, 1 - r_corner]
    ]
    for i, pos in enumerate(corner_positions):
        centers[i] = pos
        radii[i] = r_corner
    
    # Remaining circles in a grid
    r_grid = 0.08
    idx = 4
    row = 0
    while idx < n:
        col = 0
        while col < 10 and idx < n:
            x = r_grid + col * 2.0 * r_grid
            y = r_grid + row * 2.0 * r_grid
            if x + r_grid <= 1.0 and y + r_grid <= 1.0:
                centers[idx] = [x, y]
                radii[idx] = r_grid
                idx += 1
            col += 1
        row += 1
    return centers, radii


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    num_restarts = 80
    
    for seed in range(num_restarts):
        np.random.seed(seed)
        
        # Choose initialization type
        init_type = seed % 4
        
        if init_type == 0:
            r_init = 0.07 + seed * 0.0015
            centers, radii = init_hexagonal(n, r_init)
        elif init_type == 1:
            r_init = 0.07 + seed * 0.0015
            centers, radii = init_grid(n, r_init)
        elif init_type == 2:
            centers, radii = init_corners_grid(n)
        else:
            # Random initialization
            centers = np.random.uniform(0.1, 0.9, (n, 2))
            radii = np.full(n, 0.07 + seed * 0.001)
        
        # Add random perturbation
        centers += np.random.normal(0, 0.002, centers.shape)
        centers = np.clip(centers, 0.01, 0.99)
        
        # Flatten to parameter vector
        params = np.concatenate([centers.flatten(), radii])
        
        # Define bounds
        bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.35)] * n
        
        # Optimization
        result = minimize(
            compute_cost,
            params,
            args=(n,),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50000, 'ftol': 1e-14, 'gtol': 1e-12}
        )
        
        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
        current_sum = np.sum(radii_opt)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Final refinement on best solution
    params = np.concatenate([best_centers.flatten(), best_radii])
    bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.35)] * n
    result = minimize(
        compute_cost,
        params,
        args=(n,),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100000, 'ftol': 1e-15, 'gtol': 1e-13}
    )
    
    best_centers = result.x[:2 * n].reshape(n, 2)
    best_radii = result.x[2 * n:]
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
