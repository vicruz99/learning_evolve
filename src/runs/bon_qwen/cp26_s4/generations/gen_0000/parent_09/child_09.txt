        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Compute boundary limits
        bounds_vec = np.minimum(
            np.minimum(centers[:, 0], 1 - centers[:, 0]),
            np.minimum(centers[:, 1], 1 - centers[:, 1])
        )
        
        # LP variables: r_0, ..., r_25
        # Maximize sum r_i  <=>  Minimize -sum r_i
        c = -np.ones(n)
        
        # Constraints:
        # 1. r_i <= bounds_vec[i]  (r_i - bounds[i] <= 0)
        # 2. r_i + r_j <= dists[i,j] for i < j
        
        # A_ub x <= b_ub
        # We need to construct A_ub and b_ub.
        # Number of constraints: n (bounds) + n*(n-1)/2 (distances)
        # 26 + 325 = 351 constraints.
        
        num_constraints = n + n * (n - 1) // 2
        A_ub = np.zeros((num_constraints, n))
        b_ub = np.zeros(num_constraints)
        
        idx = 0
        # Bounds
        for i in range(n):
            A_ub[idx, i] = 1.0
            b_ub[idx] = bounds_vec[i]
            idx += 1
            
        # Distances
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[idx, i] = 1.0
                A_ub[idx, j] = 1.0
                b_ub[idx] = dists[i, j]
                idx += 1
                
        # Non-negativity r_i >= 0
        bounds = [(0, None) for _ in range(n)]
        
        # Solve LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            centers = centers # centers are fixed
            sum_radii = np.sum(radii)
        else:
            # Fallback to equal radii
            radii = np.full(n, final_r)
            sum_radii = np.sum(radii)
    else:
        # Fallback centers
        # 5x5 grid with 1 extra?
        # Let's make a valid 6x5 grid subset.
        centers = np.array([
            [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
            [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
            [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
            [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
            [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9],
            [0.2, 0.2] # 26th circle in a gap?
        ])
        # This might overlap.
        # Let's just use the equal radius solution if found.
        radii = np.full(26, 0.08) # Safe radius
        sum_radii = np.sum(radii)

    return centers, radii, float(sum_radii)
