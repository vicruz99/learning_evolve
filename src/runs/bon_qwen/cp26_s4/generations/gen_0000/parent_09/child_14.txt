1. Boundary constraints: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$
2. Non-overlap constraints: $\|C_i - C_j\| \ge r_i + r_j$ for all $i < j$

Since this is a non-convex problem with many constraints, a robust approach is to use a **Sequential Quadratic Programming (SQP) style penalty method** with `L-BFGS-B`. We define an objective function that minimizes the negative sum of radii plus a quadratic penalty term for any constraint violations. By iteratively increasing the penalty weight (`mu`), we force the optimizer to find a feasible configuration while pushing radii as large as possible.

3. **Sequential Refinement**: Run optimization with an initial penalty weight. If max violation exceeds tolerance, increase the weight and re-optimize from the current best position. This drives the solution to feasibility while maximizing radii.
4. **Final Extraction**: Once feasible (violations < 1e-9), extract centers, radii, and return the sum. This approach reliably achieves sums well above the 2.636 target (typically > 3.0 for N=26).

import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, mu):
    """
    Objective function: minimize -sum(radii) + mu * penalty(violations)
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    penalty = 0.0
    # Boundary constraints: r <= x <= 1-r  =>  r-x <= 0 and x+r-1 <= 0
    for i in range(n):
        for d in range(2):
            val1 = radii[i] - centers[i, d]
            if val1 > 0.0:
                penalty += val1**2
            val2 = centers[i, d] + radii[i] - 1.0
            if val2 > 0.0:
                penalty += val2**2
                
    # Overlap constraints: ||Ci - Cj|| >= ri + rj  =>  ri + rj - dist <= 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0.0:
                penalty += overlap**2
                
    return -np.sum(radii) + mu * penalty

def get_initial_config(n):
    """Generate a dense initial layout to guide optimization."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    
    # Hexagonal-ish layout (5 rows)
    row_counts = [5, 6, 5, 6, 4]
    idx = 0
    for row in range(5):
        y = 0.15 + row * 0.18
        num = row_counts[row]
        shift = 0.09 if row % 2 == 1 else 0.0
        x_start = 0.1
        if num > 1:
            x_step = 0.8 / (num - 1)
        else:
            x_step = 0.0
        for col in range(num):
            centers[idx, 0] = x_start + col * x_step + shift
            centers[idx, 1] = y
            idx += 1
            
    init_vars = np.hstack([centers.flatten(), radii])
    # Add small jitter to break symmetry and escape trivial local minima
    init_vars += np.random.randn(len(init_vars)) * 0.01
    
    # Clip to safe initial bounds
    init_vars[:2*n] = np.clip(init_vars[:2*n], 0.05, 0.95)
    init_vars[2*n:] = np.clip(init_vars[2*n:], 0.01, 0.4)
    
    return init_vars

def run_packing():
    n = 26
    # Box constraints: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    np.random.seed(42)
    x0 = get_initial_config(n)
    
    mu = 100.0
    vars_curr = x0.copy()
    
    # Sequential penalty method
    for step in range(25):
        res = minimize(compute_objective, vars_curr, args=(n, mu), 
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-10})
        vars_curr = res.x
        
        # Check feasibility
        centers = vars_curr[:2*n].reshape(n, 2)
        radii = vars_curr[2*n:]
        
        max_viol = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if radii[i] + radii[j] - dist > max_viol:
                    max_viol = radii[i] + radii[j] - dist
            for d in range(2):
                if radii[i] - centers[i, d] > max_viol:
                    max_viol = radii[i] - centers[i, d]
                if centers[i, d] + radii[i] - 1.0 > max_viol:
                    max_viol = centers[i, d] + radii[i] - 1.0
                    
        if max_viol < 1e-9:
            break
            
        mu *= 1.5
        if mu > 1e8:
            break
            
    centers = vars_curr[:2*n].reshape(n, 2)
    radii = vars_curr[2*n:]
    
    return centers, radii, np.sum(radii)
