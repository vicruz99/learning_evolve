import numpy as np
from scipy.optimize import minimize

def compute_loss(params, n, weight):
    """
    Computes the objective function: negative sum of radii plus penalty for constraint violations.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Boundary violations
    x, y = centers[:, 0], centers[:, 1]
    r = radii
    v_bound = (np.maximum(0, r - x)**2 + 
               np.maximum(0, r - y)**2 + 
               np.maximum(0, r - (1 - x))**2 + 
               np.maximum(0, r - (1 - y))**2)
    
    # Pairwise violations
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    sum_r = r[:, np.newaxis] + r[np.newaxis, :]
    v_pair = np.maximum(0, sum_r - dists)
    
    idx = np.triu_indices(n, k=1)
    pair_penalty = np.sum(v_pair[idx]**2)
    bound_penalty = np.sum(v_bound)
    
    return -np.sum(r) + weight * (bound_penalty + pair_penalty)

def run_packing():
    n = 26
    best_score = -np.inf
    best_centers = None
    best_radii = None
    
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    for seed in range(20):
        np.random.seed(seed)
        # Structured initialization: roughly hexagonal/grid pattern
        params = np.zeros(3*n)
        
        # Generate positions
        y_vals = np.linspace(0.12, 0.88, 5)
        positions = []
        for i, y in enumerate(y_vals):
            # Row pattern: 5, 6, 5, 6, 4 -> sums to 26
            if i == 0: count = 5
            elif i == 1: count = 6
            elif i == 2: count = 5
            elif i == 3: count = 6
            else: count = 4
            
            x_vals = np.linspace(0.12, 0.88, count)
            # Shift odd rows for hexagonal arrangement
            if i % 2 == 1:
                x_vals += 0.08
            
            for x in x_vals:
                if len(positions) < n:
                    positions.append([x, y])
                    if len(positions) == n: break
            if len(positions) == n: break
            
        positions = np.array(positions[:n])
        params[:2*n] = positions.flatten()
        params[2*n:] = 0.095  # Start with radius slightly above 0.1
        
        # Add small random perturbation to break symmetry
        params += np.random.normal(0, 0.01, 3*n)
        
        # Clip to bounds
        np.clip(params[:2*n], 0, 1, out=params[:2*n])
        np.clip(params[2*n:], 0.01, 0.4, out=params[2*n:])
        
        # Successive penalty optimization
        weight = 100.0
        for _ in range(4):
            res = minimize(compute_loss, params, args=(n, weight), 
                          method='L-BFGS-B', bounds=bounds, 
                          options={'maxiter': 1500, 'ftol': 1e-11})
            params = res.x
            weight *= 10
            
        current_radii = params[2*n:]
        current_sum = np.sum(current_radii)
        
        if current_sum > best_score:
            best_score = current_sum
            best_centers = params[:2*n].reshape(n, 2).copy()
            best_radii = current_radii.copy()
            
    # Safety shrink to guarantee strict validity against numerical tolerances
    best_radii *= 0.9999
    
    return best_centers, best_radii, np.sum(best_radii)
