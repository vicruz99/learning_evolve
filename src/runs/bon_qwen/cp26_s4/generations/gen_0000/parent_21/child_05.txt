import numpy as np
from scipy.optimize import minimize

def compute_objective(vars):

    # vars layout: [x0..x25, y0..y25, r0..r25]
    return -np.sum(vars[52:])

def compute_constraints(vars):
    n = 26
    xs = vars[:n]
    ys = vars[n:2*n]
    rs = vars[2*n:]
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c_bound = np.concatenate([
        xs - rs,
        1.0 - rs - xs,
        ys - rs,
        1.0 - rs - ys
    ])
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # Only consider upper triangle to avoid duplicates and self-pairs
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    dist_sq = dx**2 + dy**2
    req_sq = dr**2
    
    c_pair = (dist_sq - req_sq)[mask]
    return np.concatenate([c_bound, c_pair])

def run_packing():
    n = 26
    best_sum = 0.0
    best_vars = None
    
    cons = [{'type': 'ineq', 'fun': compute_constraints}]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    
    # Multiple restarts to escape local optima
    for seed in range(30):
        np.random.seed(seed)
        
        # Initialize with a hexagonal-like grid pattern
        pts = []
        row = 0
        while len(pts) < n:
            col = 0

            row += 1
            if len(pts) >= n: break

        pts += np.random.uniform(-0.03, 0.03, size=pts.shape)

        x0 = np.concatenate([pts[:, 0], pts[:, 1], r_init])

            best_vars = res.x.copy()

    yc = best_vars[n:2*n]

    radii = rc
    return centers, radii, np.sum(radii)
