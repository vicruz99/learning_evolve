import numpy as np
from scipy.optimize import minimize

def compute_violations(centers, radii):
    """Compute total violation of constraints (boundary and overlap)."""
    n = len(radii)
    violations = 0.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        violations += max(0, r - x)
        violations += max(0, r - y)
        violations += max(0, x + r - 1)
        violations += max(0, y + r - 1)
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j]:
                violations += radii[i] + radii[j] - dist
    
    return violations

def objective(x, n):
    """Objective function: maximize sum of radii with penalty for violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    penalty = compute_violations(centers, radii)
    return -np.sum(radii) + 1000 * penalty ** 2

def force_based_relaxation(centers, radii, n, iterations=2000):
    """Use repulsive forces to separate overlapping circles and push them away from boundaries."""
    for iteration in range(iterations):
        forces = np.zeros_like(centers)
        
        # Repulsive forces between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq)
                
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force_mag = overlap / dist
                    forces[i, 0] += force_mag * dx
                    forces[i, 1] += force_mag * dy
                    forces[j, 0] -= force_mag * dx
                    forces[j, 1] -= force_mag * dy
        
        # Boundary repulsion forces
        for i in range(n):
            if centers[i, 0] - radii[i] < 0:
                forces[i, 0] += (radii[i] - centers[i, 0]) * 200
            if centers[i, 0] + radii[i] > 1:
                forces[i, 0] -= (centers[i, 0] + radii[i] - 1) * 200
            if centers[i, 1] - radii[i] < 0:
                forces[i, 1] += (radii[i] - centers[i, 1]) * 200
            if centers[i, 1] + radii[i] > 1:
                forces[i, 1] -= (centers[i, 1] + radii[i] - 1) * 200
        
        # Adaptive step size
        step_size = 0.01 / (1 + iteration * 0.0005)
        centers += forces * step_size
        centers = np.clip(centers, 0.001, 0.999)
        
        # Gradually grow radii
        radii += 0.000012
    
    return centers, radii

def generate_hexagonal_grid(n):
    """Generate n points in a hexagonal grid pattern in [0,1]^2."""
    points = []
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    y_step = 1.0 / (rows + 1)
    x_step = 1.0 / (cols + 1)
    
    for row in range(rows):
        for col in range(cols):
            if len(points) >= n:
                break
            x = (col + 0.5 + (row % 2) * 0.5) * x_step
            y = (row + 0.5) * y_step
            points.append([x, y])
        if len(points) >= n:
            break
    
    return np.array(points[:n])

def generate_corner_heavy_grid(n):
    """Generate points with emphasis on corners and edges for larger circles."""
    points = []
    
    # Place circles in corners first (larger potential radius)
    corner_points = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    for p in corner_points:
        if len(points) < n:
            points.append(p)
    
    # Place along edges
    edge_points = []
    for i in np.linspace(0.3, 0.7, 4):
        edge_points.append([0.08, i])
        edge_points.append([0.92, i])
        edge_points.append([i, 0.08])
        edge_points.append([i, 0.92])
    
    for p in edge_points:
        if len(points) < n:
            points.append(p)
    
    # Fill remaining with hexagonal grid
    remaining = n - len(points)
    if remaining > 0:
        rows = int(np.ceil(np.sqrt(remaining)))
        cols = int(np.ceil(remaining / rows))
        y_step = 0.6 / (rows + 1)
        x_step = 0.6 / (cols + 1)
        
        for row in range(rows):
            for col in range(cols):
                if len(points) >= n:
                    break
                x = 0.2 + (col + 0.5 + (row % 2) * 0.5) * x_step
                y = 0.2 + (row + 0.5) * y_step
                points.append([x, y])
            if len(points) >= n:
                break
    
    return np.array(points[:n])

def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Try multiple initial configurations
    for trial in range(40):
        np.random.seed(trial)
        
        # Choose initial configuration strategy
        if trial < 15:
            # Hexagonal grid with perturbation
            centers = generate_hexagonal_grid(n).copy()
            centers += np.random.uniform(-0.02, 0.02, centers.shape)
        elif trial < 25:
            # Corner-heavy grid with perturbation
            centers = generate_corner_heavy_grid(n).copy()
            centers += np.random.uniform(-0.02, 0.02, centers.shape)
        else:
            # Random configuration
            centers = np.random.uniform(0.15, 0.85, (n, 2))
        
        centers = np.clip(centers, 0.05, 0.95)
        radii = np.full(n, 0.008)
        
        # Force-based relaxation to get a good feasible configuration
        centers, radii = force_based_relaxation(centers, radii, n, iterations=1500)
        
        # Gradient-based optimization to refine
        x0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0.01, 0.99)] * (2 * n) + [(0.001, 0.5)] * n
        
        result = minimize(objective, x0, args=(n,), method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 150000, 'ftol': 1e-15})
        
        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
        
        violations = compute_violations(centers_opt, radii_opt)
        s = np.sum(radii_opt)
        
        # Accept if nearly feasible and better than current best
        if violations < 1e-4 and s > best_sum:
            best_sum = s
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
        
        # Additional refinement if close to feasible
        if violations < 0.01 and best_centers is not None:
            for refine_step in range(2):
                result2 = minimize(objective, result.x, args=(n,), method='L-BFGS-B', bounds=bounds,
                                 options={'maxiter': 50000, 'ftol': 1e-15})
                centers_refined = result2.x[:2 * n].reshape(n, 2)
                radii_refined = result2.x[2 * n:]
                violations_refined = compute_violations(centers_refined, radii_refined)
                s_refined = np.sum(radii_refined)
                
                if violations_refined < 1e-4 and s_refined > best_sum:
                    best_sum = s_refined
                    best_centers = centers_refined.copy()
                    best_radii = radii_refined.copy()
    
    # Fallback to a safe configuration if no good solution found
    if best_centers is None or np.sum(best_radii) < 0.5:
        centers = generate_hexagonal_grid(n)
        radii = np.full(n, 0.02)
        return centers, radii, np.sum(radii)
    
    # Ensure final solution is valid
    best_centers = np.clip(best_centers, 0, 1)
    best_radii = np.clip(best_radii, 0, 0.5)
    
    return best_centers, best_radii, best_sum
