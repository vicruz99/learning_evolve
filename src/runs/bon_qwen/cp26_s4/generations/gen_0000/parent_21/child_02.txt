For any fixed set of centers, the maximum valid radius for each circle is determined by its distance to the boundaries and to other circles. Specifically, $r_i = \min(x_i, 1-x_i, y_i, 1-y_i, \min_{j \neq i} \frac{\|c_i - c_j\|}{2})$. This formulation automatically guarantees a valid packing (no overlaps, within bounds) for any center configuration.

4.  **Selection**: Track the best configuration found across all runs.

import numpy as np
from scipy.optimize import minimize

def compute_radii(centers):
    """
    Computes the maximum possible radius for each circle given fixed centers.
    Ensures circles stay within [0,1]^2 and do not overlap.
    """
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        cx, cy = centers[i]
        # Distance to boundaries
        r = min(cx, 1.0 - cx, cy, 1.0 - cy)
        if r < 0:
            r = 0.0
            
        # Distance to other circles
        for j in range(n):
            if i == j:
                continue
            dx = cx - centers[j, 0]
            dy = cy - centers[j, 1]
            d = np.hypot(dx, dy)
            if d / 2.0 < r:
                r = d / 2.0
        radii[i] = r
    return radii

def objective(params):
    """
    Objective function for optimization.
    Returns negative sum of radii (to be minimized).
    Adds penalty if centers go outside [0,1]^2.
    """
    n = 26
    centers = params.reshape(n, 2)
    
    # Penalty for out-of-bounds centers to guide the optimizer
    if np.any(centers < -1e-4) or np.any(centers > 1.0 + 1e-4):
        return 1e5
        
    radii = compute_radii(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_params = None
    best_val = -np.inf
    
    initial_configs = []
    
    # 1. Hexagonal Lattice Initialization
    hex_pts = []
    # Estimate spacing for ~26 circles in unit square
    # Area per circle ~ 1/26. Hex spacing s ~ sqrt(2/(sqrt(3)*26)) ~ 0.21
    s = 0.21
    for row in range(8):
        y = (row + 0.5) * s * np.sqrt(3) / 2
        if y > 0.95:
            break
        shift = (row % 2) * s / 2
        for col in range(8):
            x = (col + 0.5) * s + shift
            if x > 0.95:
                break
            hex_pts.append([x, y])
    hex_pts = np.array(hex_pts)
    if len(hex_pts) >= n:
        initial_configs.append(hex_pts[:n])
        initial_configs.append(hex_pts[np.random.choice(len(hex_pts), n, replace=False)])
    else:
        initial_configs.append(hex_pts)

    # 2. Grid Initialization
    grid_pts = []
    for i in range(6):
        for j in range(5):
            x = 0.08 + i * 0.155
            y = 0.12 + j * 0.18
            if x <= 0.95 and y <= 0.95:
                grid_pts.append([x, y])
    grid_pts = np.array(grid_pts)
    if len(grid_pts) >= n:
        initial_configs.append(grid_pts[:n])
        initial_configs.append(grid_pts[np.random.choice(len(grid_pts), n, replace=False)])

    # 3. Random Initializations
    np.random.seed(42) # Fixed seed for reproducibility within the run
    initial_configs.append(np.random.rand(n, 2))
    initial_configs.append(np.random.rand(n, 2))
    initial_configs.append(np.random.rand(n, 2))

    # Optimization loop
    for cfg in initial_configs:
        # Ensure initial config is strictly inside bounds
        cfg = np.clip(cfg, 0.02, 0.98)
        
        res = minimize(
            objective, 
            cfg.flatten(), 
            method='Nelder-Mead', 
            options={'maxiter': 4000, 'xatol': 1e-7, 'fatol': 1e-8}
        )
        
        if res.fun < best_val:
            best_val = res.fun
            best_params = res.x
            
    centers = best_params.reshape(n, 2)
    radii = compute_radii(centers)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
