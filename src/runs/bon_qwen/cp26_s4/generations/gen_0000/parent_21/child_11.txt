def resolve_forces(centers, radii, n, n_steps):

                    forces[i, 1] -= force_magnitude * dy / dist

def pack_to_vars(centers, radii):
    n = len(radii)
    vars = np.zeros(3 * n)
    for i in range(n):
        vars[3 * i] = centers[i, 0]
        vars[3 * i + 1] = centers[i, 1]
        vars[3 * i + 2] = radii[i]
    return vars

def vars_to_pack(vars, n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = vars[3 * i]
        centers[i, 1] = vars[3 * i + 1]
        radii[i] = vars[3 * i + 2]
    return centers, radii

def objective_penalty(vars, n):
    centers, radii = vars_to_pack(vars, n)
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        r = radii[i]
        x, y = centers[i, 0], centers[i, 1]
        
        if x - r < 0:
            penalty += (r - x) ** 2 * 5000
        if x + r > 1:
            penalty += (x + r - 1) ** 2 * 5000
        if y - r < 0:
            penalty += (r - y) ** 2 * 5000
        if y + r > 1:
            penalty += (y + r - 1) ** 2 * 5000
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2 * 5000
    
    return obj + penalty

def build_bounds(n):
    bounds = []
    for i in range(n):
        bounds.append((0.001, 0.999))
        bounds.append((0.001, 0.999))
        bounds.append((0.001, 0.5))
    return bounds

def validate_internal(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True
