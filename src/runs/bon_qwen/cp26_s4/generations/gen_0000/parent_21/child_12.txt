import numpy as np
import scipy.optimize as opt

def generate_initial_centers(n_circles):
    """Generates an initial staggered grid layout for the circles."""
    rows = [5, 4, 5, 4, 5, 3]
    centers = np.zeros((n_circles, 2))
    idx = 0
    y_coord = 0.12
    row_idx = 0

        x_start = 0.5 / count  # Center the row
        x_step = 1.0 / count   # Spacing between centers
        
        for i in range(count):
            centers[idx, 0] = x_start + i * x_step
            centers[idx, 1] = y_coord
            idx += 1
            
        # Alternate shift for hexagonal packing
        if row_idx % 2 == 0:
            y_coord += 0.18
        else:
            y_coord += 0.16
            row_idx += 1
            # Apply x-shift to the next row's centers
            for j in range(idx, idx + count): # This logic is slightly flawed for direct assignment
                pass 
            # Instead, we will shift the centers directly in the loop below or pre-calc
            pass
    
    # Let's rebuild centers with explicit hexagonal shifts
    centers = np.zeros((n_circles, 2))
    idx = 0
    y_coord = 0.12
    current_rows = [5, 4, 5, 4, 5, 3]
    
    for r_idx, count in enumerate(current_rows):
        # Hexagonal shift: offset x centers by half a step relative to a grid of size count+1 or similar
        # Simple shift: offset by 0.5 * (1/count) is too complex. 
        # Standard hex shift: offset by half the spacing of the row above.
        # Let's use a fixed staggered offset of 1.0 / (2 * max_count) approx?
        # Better: offset by 0.5 * (1/5) = 0.1 if row above had 5.
        
        if r_idx % 2 == 1:
            x_start = 0.5 / 5 + 0.5 * (1.0/5) # Shifted
        else:
            x_start = 0.5 / count
            
        x_step = 1.0 / count # Spacing
        
        # For the 4-count rows, let's center them and ensure they nest
        if count == 4:
             x_start = 0.5 / 5 + 0.5 * (1.0/5)
             x_step = 1.0/5 # Keep same step as 5-count rows to nest perfectly
        else:

            centers[idx, 0] = x_start + i * x_step

            idx += 1

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initial centers
    centers = np.zeros((n, 2))
    
    # Define a robust 6-row hexagonal structure
    # Rows: 5, 4, 5, 4, 5, 3
    row_counts = [5, 4, 5, 4, 5, 3]
    
    y_pos = 0.12
    idx = 0
    for r_idx, count in enumerate(row_counts):
        if r_idx % 2 == 1:
            # Staggered row
            x_start = 0.2
            x_step = 0.2
        else:
            # Aligned row
            x_start = 0.1
            x_step = 0.2
        
        for i in range(count):
            centers[idx, 0] = x_start + i * x_step
            centers[idx, 1] = y_pos
            idx += 1
        y_pos += 0.18

    # Radii initialization
    radii = np.ones(n) * 0.09
    
    # Objective: Maximize sum of radii
    # Constraints: Non-overlap, Boundary
    
    def objective(x):
        r = x[:n]
        # Negative sum because we minimize
        return -np.sum(r)

    def constraints(x):
        c = []
        r = x[:n]
        pos = x[n:]
        centers_opt = pos.reshape((n, 2))
        
        # 1. Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        for i in range(n):
            c.append(r[i] - centers_opt[i, 0]) # x - r >= 0
            c.append(r[i] - (1.0 - centers_opt[i, 0])) # 1-x - r >= 0
            c.append(r[i] - centers_opt[i, 1]) # y - r >= 0
            c.append(r[i] - (1.0 - centers_opt[i, 1])) # 1-y - r >= 0
            
        # 2. Non-overlap: dist >= r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                c.append(dist - r[i] - r[j])
                
        return c

    x0 = np.concatenate([radii, centers.flatten()])
    
    # Bounds: radii > 0, centers in [0, 1]
    bnds = [(1e-6, 0.5)] * n + [(0.0, 1.0)] * (2 * n)
    
    cons = {'type': 'ineq', 'fun': constraints}
    
    res = opt.minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
    
    if res.success:
        final_radii = res.x[:n]
        final_centers = res.x[n:].reshape((n, 2))
    else:
        # Fallback to initial if optimization fails (unlikely with this setup)
        final_radii = radii
        final_centers = centers
        
    # Final validation check and slight scaling to ensure validity
    # Sometimes optimization pushes to boundary, we scale back slightly to be safe
    # But the constraint logic above handles it.
    
    total_sum = np.sum(final_radii)
    return final_centers, final_radii, total_sum
