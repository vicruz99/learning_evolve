import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    """Computes a smooth penalty for overlaps and boundary violations."""
    n = centers.shape[0]
    c = centers.reshape(n, 2)
    r = radii
    
    # Pairwise overlaps
    diff = c[:, None, :] - c[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    min_dist = r[:, None] + r[None, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = np.maximum(0, min_dist - dist)
    pair_pen = np.sum(overlaps[mask]**2)
    
    # Boundary violations
    x, y = c[:, 0], c[:, 1]
    bound_pen = np.sum(np.maximum(0, r - x)**2 + np.maximum(0, r - (1 - x))**2 +
                       np.maximum(0, r - y)**2 + np.maximum(0, r - (1 - y))**2)
                       
    return pair_pen + bound_pen

def objective(x, radii, n):
    """Objective function for scipy optimizer."""
    return compute_penalty(x.reshape(n, 2), radii)

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Initialize on a hexagonal grid with small noise
    centers = np.zeros((n, 2))
    idx = 0

        for j in range(6):
            if idx >= n: break
            x = 0.12 + j * 0.14 + (0.07 if i % 2 == 1 else 0)
            y = 0.12 + i * 0.14
            centers[idx] = [x, y] + np.random.randn(2) * 0.005
            idx += 1
            
    radii = np.full(n, 0.02)
    
    growth = 1.0005
    max_fails = 150
    fails = 0
    
    while fails < max_fails:
        bounds = [(radii[i], 1 - radii[i]) for i in range(n)] * 2
        res = minimize(objective, centers.flatten(), args=(radii, n),
                       method='L-BFGS-B', bounds=bounds,
                       options={'ftol': 1e-12, 'maxiter': 2000, 'disp': False})
                       
        centers = res.x.reshape(n, 2)
        # Strictly enforce bounds after optimization
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)

            radii *= growth
            fails = 0
            growth *= 0.999  # Slow down growth to refine
        else:
            fails += 1
            if fails > 50:
                break

    pen = compute_penalty(centers, radii)
    if pen > 1e-8:

        for _ in range(50):
            if compute_penalty(centers, radii * scale) < 1e-9:
                break
            scale *= 0.99
        radii *= scale
        
    return centers, radii, np.sum(radii)
