

To maximize the sum of radii for 26 circles in a unit square, we employ a strategy of packing circles of approximately equal size (optimal for this objective) and refining their positions and sizes using numerical optimization.

### Strategy

1.  **Initialization**: We initialize the 26 circles in a dense hexagonal grid. We set an initial radius slightly smaller than the theoretical maximum ($r \approx 0.09$) to ensure a valid starting configuration.
2.  **Numerical Optimization**: Using `scipy.optimize.minimize`, we maximize the sum of the radii (minimize the negative sum) subject to non-overlap and boundary constraints.
3.  **Robustness**: To avoid local minima, the process is repeated multiple times with randomized perturbations to the initial positions. The configuration with the highest sum of radii is selected.
4.  **Validation**: The final solution is validated against the strict constraints of the unit square and non-overlap.

import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Define the objective function to maximize sum of radii (minimize negative sum)
    def objective(x):
        centers = x[:2 * N].reshape(N, 2)
        radii = x[2 * N:]
        return -np.sum(radii)

    # Define constraints
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    # This implies r + x - 1 <= 0, r - x <= 0, etc.
    for i in range(N):
        r_idx = 2 * N + i
        x_idx = i * 2
        y_idx = i * 2 + 1
        
        # x >= r
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i, r_idx=r_idx, x_idx=x_idx: x[x_idx] - x[r_idx]})
        # x <= 1 - r  =>  r + x - 1 <= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i, r_idx=r_idx, x_idx=x_idx: 1.0 - x[x_idx] - x[r_idx]})
        # y >= r
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i, r_idx=r_idx, y_idx=y_idx: x[y_idx] - x[r_idx]})
        # y <= 1 - r  =>  r + y - 1 <= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i, r_idx=r_idx, y_idx=y_idx: 1.0 - x[y_idx] - x[r_idx]})
        
    # Non-overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(N):
        for j in range(i + 1, N):
            r_i_idx = 2 * N + i
            r_j_idx = 2 * N + j
            x_i_idx = i * 2
            y_i_idx = i * 2 + 1
            x_j_idx = j * 2
            y_j_idx = j * 2 + 1
            
            def non_overlap(x, i=i, j=j, r_i=r_i_idx, r_j=r_j_idx, 
                            xi=x_i_idx, yi=y_i_idx, xj=x_j_idx, yj=y_j_idx):
                dx = x[xi] - x[xj]
                dy = x[yi] - x[yj]
                dist_sq = dx**2 + dy**2
                r_sum = x[r_i] + x[r_j]
                return dist_sq - r_sum**2

            constraints.append({'type': 'ineq', 'fun': non_overlap})

    # Hexagonal initialization
    side = 1.0
    init_r = 0.095
    rows = int(np.ceil(np.sqrt(N * np.sqrt(3) / 2)))
    cols = int(np.ceil(N / rows))
    
    # Try multiple random restarts to find global optimum
    for attempt in range(10):
        x0 = []
        centers_init = np.zeros((N, 2))
        
        # Generate hexagonal grid positions with some noise
        count = 0
        r = init_r
        for row in range(rows):
            for col in range(cols):
                if count >= N:
                    break

                # Add small noise
                x += np.random.uniform(-0.02, 0.02)
                y += np.random.uniform(-0.02, 0.02)
                
                centers_init[count] = [x, y]
                count += 1
        
        # Initialize radii
        radii_init = np.full(N, init_r)
        
        # Flatten for optimizer
        x0 = np.concatenate([centers_init.flatten(), radii_init])

            res = opt.minimize(objective, x0, method='SLSQP',

                current_sum = -res.fun

                    best_sum = current_sum

        best_radii = np.full(N, 0.08)
        best_sum = np.sum(best_radii)

    # Ensure non-negative radii
    best_radii = np.maximum(best_radii, 1e-9)
    # Ensure centers are within bounds (though constraints should handle this)
    best_centers = np.clip(best_centers, best_radii.reshape(-1, 1), 1 - best_radii.reshape(-1, 1))

    return best_centers, best_radii, best_sum
