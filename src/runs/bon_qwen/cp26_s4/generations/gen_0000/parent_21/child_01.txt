def calculate_overlap_energy(centers, radii):

    Calculates the total overlap energy.
    Energy = sum of squared overlaps for all pairs and boundary violations.
    Overlap for pair (i, j) is max(0, r_i + r_j - dist(i, j))^2.
    Boundary violation is max(0, r - x)^2 + ...
    """
    n = centers.shape[0]
    energy = 0.0
    
    # Pairwise overlaps
    # Vectorized calculation for performance
    # centers shape (n, 2), radii shape (n,)
    
    # Compute pairwise distance matrix
    # dist_sq[i, j] = (x_i - x_j)^2 + (y_i - y_j)^2
    # Using broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    dist = np.sqrt(dist_sq)
    
    # Radii sums matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount: r_sum - dist. If negative, no overlap.
    # We only care about positive overlaps.
    overlap_amt = r_sum - dist
    # Set lower triangle and diagonal to 0 to avoid double counting and self-overlap
    np.fill_diagonal(overlap_amt, 0.0)
    # Keep only upper triangle (i < j)
    overlap_amt = np.triu(overlap_amt, k=1)
    
    # Squared overlap energy
    # We sum max(0, overlap)^2. Since we are minimizing, we can just square the positive parts.
    # But wait, if overlap_amt is negative (no overlap), we want 0.
    # np.maximum(0, overlap_amt)**2
    overlap_energy = np.sum(np.maximum(0, overlap_amt)**2)
    
    # Boundary violations
    # Circle i is inside [0,1] if r_i <= x_i <= 1-r_i and r_i <= y_i <= 1-r_i
    # Violation terms:
    # left: max(0, r_i - x_i)
    # right: max(0, r_i - (1 - x_i)) = max(0, r_i + x_i - 1)
    # bottom: max(0, r_i - y_i)
    # top: max(0, r_i - (1 - y_i)) = max(0, r_i + y_i - 1)
    
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    violations = np.maximum(0, r - x)**2 + \

                 np.maximum(0, r - y)**2 + \
                 np.maximum(0, r + y - 1)**2
    
    boundary_energy = np.sum(violations)
    
    return overlap_energy + boundary_energy

def optimize_centers(centers, radii):
    """
    Optimizes centers to minimize overlap energy for fixed radii.
    """
    # Reshape centers for optimizer
    x0 = centers.flatten()
    
    def objective(x):
        c = x.reshape(26, 2)
        # We can use the energy function directly, but maybe simpler to just call helper
        # Helper expects centers array
        return calculate_overlap_energy(c, radii)
    
    def gradient(x):
        c = x.reshape(26, 2)
        g = get_overlap_energy_gradient(c, radii)
        return g.flatten()
    
    # Bounds for centers: [r, 1-r]
    bounds = [(radii[i], 1 - radii[i]) for _ in range(2)]
    bounds = bounds * 26
    
    # Use L-BFGS-B for bounds and gradient
    res = minimize(objective, x0, jac=gradient, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 100, 'ftol': 1e-12})
    
    return res.x.reshape(26, 2)
