import numpy as np
from scipy.optimize import minimize
import math

def compute_violation(centers, radii):
    """Compute total squared constraint violation."""
    n = len(radii)
    v = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v += max(0, r - x) ** 2
        v += max(0, r - y) ** 2
        v += max(0, x + r - 1) ** 2
        v += max(0, y + r - 1) ** 2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = math.sqrt(dx * dx + dy * dy)
            min_d = radii[i] + radii[j]
            if d < min_d:
                v += (d - min_d) ** 2
    return v

def objective_func(params, n_circles, penalty_weight):
    """Objective: negative sum of radii plus penalty for constraint violations."""
    centers = params[:2 * n_circles].reshape(n_circles, 2)
    radii = np.maximum(params[2 * n_circles:], 1e-8)
    return -np.sum(radii) + penalty_weight * compute_violation(centers, radii)

def is_valid_packing(centers, radii, n_circles):
    """Check if the packing satisfies all constraints."""
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.ones(n) * 0.01

    for trial in range(300):
        np.random.seed(trial)
        
        # Initialize with hexagonal grid pattern
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.065
        
        idx = 0
        row = 0
        while idx < n:
            col = 0
            while idx < n and col < 7:
                x = 0.08 + col * 0.155 + (row % 2) * 0.0775
                y = 0.08 + row * 0.135
                if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                    centers[idx] = [x, y]
                col += 1
                idx += 1
            row += 1
        
        # Add random perturbations
        centers += np.random.uniform(-0.03, 0.03, (n, 2))
        radii += np.random.uniform(-0.015, 0.015, n)
        radii = np.maximum(radii, 0.02)
        centers = np.clip(centers, 0.05, 0.95)
        
        # Flatten for optimization
        params = np.concatenate([centers.flatten(), radii])
        
        # Penalty continuation optimization
        for pw in [50, 200, 1000, 5000, 25000]:
            res = minimize(
                objective_func,
                params,
                args=(n, pw),
                method='BFGS',
                options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-10}
            )
            params = res.x.copy()
        
        # Extract optimized parameters
        c_opt = params[:2 * n].reshape(n, 2)
        r_opt = np.maximum(params[2 * n:], 1e-8)
        
        # Validate the solution
        valid = is_valid_packing(c_opt, r_opt, n)
        
        if valid:
            current_sum = np.sum(r_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
    
    return best_centers, best_radii, best_sum
