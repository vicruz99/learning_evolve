import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(p):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(p[2 * N_CIRCLES:])

def constraint_function(p):
    """
    Returns an array of inequality constraints >= 0.
    Constraints:
    1. Boundary constraints for x and y
    2. Non-overlap constraints for all pairs
    """
    x = p[:N_CIRCLES]
    y = p[N_CIRCLES:2 * N_CIRCLES]
    r = p[2 * N_CIRCLES:]
    
    c = []
    
    # Boundary constraints
    for i in range(N_CIRCLES):
        c.append(x[i] - r[i])
        c.append(1.0 - x[i] - r[i])
        c.append(y[i] - r[i])
        c.append(1.0 - y[i] - r[i])
        
    # Non-overlap constraints
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            r_sum = r[i] + r[j]
            c.append(dx * dx + dy * dy - r_sum * r_sum)
            
    return np.array(c)

def get_initial_guess(seed):
    """
    Generates a feasible initial configuration using a perturbed hexagonal grid.
    """
    rng = np.random.RandomState(seed)
    pts = []
    
    # Hexagonal lattice generation
    spacing = 0.22
    for i in range(8):
        for j in range(8):
            x = j * spacing + (i % 2) * spacing / 2.0
            y = i * spacing * np.sqrt(3.0) / 2.0
            if 0.15 < x < 0.85 and 0.15 < y < 0.85:
                pts.append((x, y))
                
    # Shuffle and select
    rng.shuffle(pts)
    if len(pts) < N_CIRCLES:
        # Fallback to random if not enough points
        while len(pts) < N_CIRCLES:
            pts.append((rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)))
            
    pts = pts[:N_CIRCLES]
    x = np.array([p[0] for p in pts])
    y = np.array([p[1] for p in pts])
    
    # Add small random perturbation to break symmetry
    x += rng.uniform(-0.03, 0.03, N_CIRCLES)
    y += rng.uniform(-0.03, 0.03, N_CIRCLES)
    x = np.clip(x, 0.15, 0.85)
    y = np.clip(y, 0.15, 0.85)
    
    # Initial radii small enough to guarantee feasibility
    r = rng.uniform(0.03, 0.06, N_CIRCLES)
    
    return np.concatenate([x, y, r])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_val = -np.inf
    best_p = None
    
    # Variable bounds
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(1e-6, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_function}
    
    # Multiple restarts to avoid local optima
    for seed in range(8):
        p0 = get_initial_guess(seed)
        try:
            res = minimize(
                objective,
                p0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={
                    'maxiter': 1500,
                    'ftol': 1e-10,
                    'disp': False
                }
            )
            
            if np.isnan(res.x).any():
                continue
                
            current_val = -res.fun
            if current_val > best_val:
                best_val = current_val
                best_p = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_p is None:
        p0 = get_initial_guess(0)
        res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000})
        best_p = res.x
        best_val = -res.fun
        
    centers = np.column_stack((best_p[:N_CIRCLES], best_p[N_CIRCLES:2 * N_CIRCLES]))
    radii = best_p[2 * N_CIRCLES:]
    
    return centers, radii, float(best_val)
