import numpy as np
from scipy.optimize import minimize, linprog
import math

def compute_optimal_radii(centers):
    """
    Given centers of 26 circles, compute the maximum possible radii
    such that they don't overlap and are inside the unit square.
    Solved via Linear Programming.
    
    Args:
        centers: np.array of shape (26, 2)
        
    Returns:
        radii: np.array of shape (26,) with optimal radii
        sum_radii: float, sum of radii
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0

    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c_obj = -np.ones(n)

    # Constraints A_ub @ x <= b_ub
    
    # 1. Non-overlap constraints: r_i + r_j <= dist(i, j)
    # For each pair (i, j) with i < j
    pairs = []
    dists = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            dists.append(dist)
            
    num_pairs = len(pairs)
    A_overlap = np.zeros((num_pairs, n))
    b_overlap = np.array(dists)
    
    for idx, (i, j) in enumerate(pairs):
        A_overlap[idx, i] = 1.0
        A_overlap[idx, j] = 1.0

    # 2. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # Total 4*n constraints
    A_boundary = np.zeros((4 * n, n))
    b_boundary = np.zeros(4 * n)
    
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_boundary[4*i, i] = 1.0
        b_boundary[4*i] = x
        # r_i <= 1-x
        A_boundary[4*i+1, i] = 1.0
        b_boundary[4*i+1] = 1.0 - x
        # r_i <= y
        A_boundary[4*i+2, i] = 1.0
        b_boundary[4*i+2] = y
        # r_i <= 1-y
        A_boundary[4*i+3, i] = 1.0
        b_boundary[4*i+3] = 1.0 - y

    # 3. Non-negativity: r_i >= 0
    # linprog handles bounds via the `bounds` argument
    bounds = [(0, None)] * n

    # Combine constraints
    A_ub = np.vstack([A_overlap, A_boundary])
    b_ub = np.hstack([b_overlap, b_boundary])

    # Solve LP
    # method='highs' is usually fast and robust
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        sum_radii = -res.fun # Because we minimized -sum
        return radii, sum_radii
    else:
        # If LP fails (should not happen if centers are valid), return 0s
        return np.zeros(n), 0.0

def objective_function(centers_flat):
    """
    Objective function for scipy.optimize.
    Minimizes negative sum of radii.
    """
    n = 26
    centers = centers_flat.reshape((n, 2))
    
    # Clip centers to be strictly inside to avoid numerical issues with LP bounds
    # Although LP handles r_i <= x_i, if x_i is 0, r_i must be 0.
    # Keeping centers slightly away from edges helps.
    centers = np.clip(centers, 1e-6, 1 - 1e-6)
    
    _, sum_radii = compute_optimal_radii(centers)
    return -sum_radii
