def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def compute_loss(params, n):

    Computes the loss function (negative sum of radii + penalties).
    """
    penalty_weight = 1000.0

    # We separate centers and radii for easier math

    loss = -np.sum(radii)
    
    # Penalty for boundary violations
    # x - r >= 0  => r - x <= 0
    # 1 - x - r >= 0 => x + r - 1 <= 0
    # Same for y
    
    # Boundary penalties (soft constraints)
    # max(0, violation)^2
    # Violations:
    # r - x
    # r + x - 1
    # r - y
    # r + y - 1

    # Check x boundaries
    viol_x1 = np.maximum(0, radii - centers[:, 0])
    viol_x2 = np.maximum(0, radii + centers[:, 0] - 1)
    # Check y boundaries
    viol_y1 = np.maximum(0, radii - centers[:, 1])
    viol_y2 = np.maximum(0, radii + centers[:, 1] - 1)
    
    boundary_penalty = np.sum(viol_x1**2 + viol_x2**2 + viol_y1**2 + viol_y2**2)
    
    # Overlap penalties
    # dist >= r_i + r_j  => r_i + r_j - dist <= 0
    # Violation: max(0, r_i + r_j - dist)

    dist_sq = np.sum(diff**2, axis=2)

    # though we mask it.

    dist = np.sqrt(dist_sq)
    
    # Sum of radii matrix (n, n)
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]

    violation = radii_sum - dist

    tri_indices = np.triu_indices(n, k=1)
    violations = np.maximum(0, violation[tri_indices])
    
    overlap_penalty = np.sum(violations**2)
    
    total_loss = loss + penalty_weight * (boundary_penalty + overlap_penalty)
    return total_loss
