import numpy as np
import math


def compute_violations(centers, radii):
    """Compute total constraint violation"""
    n = len(radii)
    violations = 0.0

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            violations += r * r
        if x - r < 0:
            violations += (x - r) ** 2
        if x + r > 1:
            violations += (x + r - 1) ** 2
        if y - r < 0:
            violations += (y - r) ** 2
        if y + r > 1:
            violations += (y + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violations += (min_dist - dist) ** 2

    return violations


def force_optimize(centers, radii, n, iterations=30000):
    """Force-directed optimization to separate circles and expand radii"""
    lr = 0.008
    expand_step = 5e-5
    force_scale = 80.0

    for iteration in range(iterations):
        forces = np.zeros((n, 2))
        max_overlap = 0.0
        min_gap = float('inf')

        # Compute pairwise forces
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = math.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10

                min_dist = radii[i] + radii[j]
                gap = dist - min_dist

                if gap < 0:
                    max_overlap = max(max_overlap, -gap)
                else:
                    min_gap = min(min_gap, gap)

                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force_mag = overlap / dist * force_scale
                    fx = force_mag * dx
                    fy = force_mag * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]

        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]

            if x - r < 0:
                forces[i, 0] += (r - x) * force_scale
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * force_scale
            if y - r < 0:
                forces[i, 1] += (r - y) * force_scale
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * force_scale

        # Update positions
        centers = centers + forces * lr

        # Clip to valid bounds
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1 - r, centers[i, 1]))

        # Expand radii periodically
        if iteration % 50 == 0:
            can_expand = True
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    if dist < radii[i] + radii[j] + 0.001:
                        can_expand = False
                        break
                if not can_expand:
                    break

            if can_expand and max_overlap < 0.0001:
                radii = radii + expand_step
            elif max_overlap > 0.001:
                radii = radii - expand_step * 0.05

        # Decay learning rate
        if iteration % 2000 == 0 and iteration > 0:
            lr *= 0.95

    return centers, radii


def hex_init(n):
    """Initialize circles on hexagonal grid"""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.03

    idx = 0
    spacing = 0.14
    for row in range(8):
        for col in range(8):
            if idx >= n:
                break
            x = col * spacing + spacing / 2
            y = row * spacing * math.sqrt(3) / 2 + spacing / 2
            if row % 2 == 1:
                x += spacing / 2
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def corner_init(n):

    radii = np.zeros(n)

    # 4 corner circles

        radii[i] = corner_size

    edge_size = 0.08

        edges.append([0.15 + k * 0.2, edge_size])

    for k in range(2):
        edges.append([1 - edge_size, 0.35 + k * 0.3])

    idx = 4
    for e in edges:
        if idx >= n:
            break
        centers[idx] = e
        radii[idx] = edge_size
        idx += 1

    # Fill remaining with smaller circles
    fill_size = 0.05
    remaining = n - idx
    fill_positions = []
    for row in range(3):
        for col in range(3):
            x = 0.3 + col * 0.15
            y = 0.3 + row * 0.15
            fill_positions.append([x, y])

    for k in range(remaining):
        if k < len(fill_positions):
            centers[idx] = fill_positions[k]
        else:
            centers[idx] = [0.5 + 0.05 * (k % 3), 0.5 + 0.05 * (k // 3)]
        radii[idx] = fill_size
        idx += 1

    return centers, radii


def grid_init(n):
    """Initialize on a regular grid"""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04

    cols = 6
    for i in range(n):
        row = i // cols
        col = i % cols
        x = (col + 0.5) / cols
        y = (row + 0.5) / 5
        centers[i] = [x, y]

    return centers, radii
