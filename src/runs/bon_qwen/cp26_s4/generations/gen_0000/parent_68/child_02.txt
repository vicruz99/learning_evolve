4. Apply fine-grained local optimization (try many small moves, keep best)
5. Use multiple optimization passes with decreasing perturbation magnitude

The key insight is that just growing circles in place leads to local optima. By shaking (perturbing) positions and regrowing, we can escape local optima and find better packings.

import numpy as np
import math
import random

def compute_max_radius(cx, cy, centers, radii, exclude_i):
    """Compute maximum radius for a circle at (cx, cy) given other circles."""
    r = min(cx, 1 - cx, cy, 1 - cy)
    n = len(centers)
    for j in range(n):
        if j != exclude_i:
            dx = cx - centers[j, 0]
            dy = cy - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            r = min(r, dist - radii[j])
    return max(0, r)


def grow_all_circles(centers, radii, max_iterations=500):
    """Iteratively grow all circles to their maximum possible size."""
    n = len(centers)
    for iteration in range(max_iterations):
        improved = False
        for i in range(n):
            r_new = compute_max_radius(
                centers[i, 0], centers[i, 1], centers, radii, i
            )
            if r_new > radii[i] + 1e-12:
                radii[i] = r_new
                improved = True
        if not improved:
            break
    return radii


def shake_circles(centers, radii, magnitude):
    """Randomly perturb circle positions within valid bounds."""
    n = len(centers)
    for i in range(n):
        dx = random.uniform(-magnitude, magnitude)
        dy = random.uniform(-magnitude, magnitude)
        new_x = centers[i, 0] + dx
        new_y = centers[i, 1] + dy
        # Clamp to valid range given current radius
        new_x = max(radii[i] + 1e-10, min(1 - radii[i] - 1e-10, new_x))
        new_y = max(radii[i] + 1e-10, min(1 - radii[i] - 1e-10, new_y))
        centers[i, 0] = new_x
        centers[i, 1] = new_y


def try_local_moves(centers, radii, i, step, n_attempts=32):
    """Try random moves for circle i and return the best one."""
    best_move = None
    best_gain = 0
    n = len(centers)
    
    for _ in range(n_attempts):
        angle = random.random() * 2 * math.pi
        dx = step * math.cos(angle)
        dy = step * math.sin(angle)
        
        new_x = centers[i, 0] + dx
        new_y = centers[i, 1] + dy
        
        # Check boundary constraints
        if new_x < radii[i] + 1e-10 or new_x > 1 - radii[i] - 1e-10:
            continue
        if new_y < radii[i] + 1e-10 or new_y > 1 - radii[i] - 1e-10:
            continue
        
        # Check overlap constraints
        valid = True
        for j in range(n):
            if i != j:
                dist = math.sqrt(
                    (new_x - centers[j, 0]) ** 2
                    + (new_y - centers[j, 1]) ** 2
                )
                if dist < radii[i] + radii[j] - 1e-10:
                    valid = False
                    break
        
        if valid:
            r_max = compute_max_radius(new_x, new_y, centers, radii, i)
            gain = r_max - radii[i]
            if gain > best_gain:
                best_gain = gain
                best_move = (new_x, new_y, r_max)
    
    return best_move, best_gain


def run_packing():
    n = 26
    random.seed(42)
    
    # Initial hexagonal arrangement: 6,5,6,5,4 circles per row
    centers = np.zeros((n, 2))
    row_counts = [6, 5, 6, 5, 4]
    
    initial_r = 0.065
    y_pos = initial_r
    idx = 0
    
    for row_idx, count in enumerate(row_counts):
        offset = 0 if row_idx % 2 == 0 else initial_r * 0.866
        x_start = initial_r + offset
        if count > 1:
            available_width = 1 - 2 * initial_r - offset
            spacing = available_width / (count - 1)
        else:
            spacing = 0
        
        for col in range(count):
            x = x_start + col * spacing
            centers[idx, 0] = x
            centers[idx, 1] = y_pos
            idx += 1
        
        y_pos += initial_r * math.sqrt(3) * 0.92
    
    radii = np.full(n, initial_r)
    
    # Phase 1: Initial growth
    radii = grow_all_circles(centers, radii)
    current_sum = np.sum(radii)
    
    # Phase 2: Shake and grow cycles with decreasing magnitude
    for cycle in range(40):
        magnitude = 0.04 * (0.88 ** cycle)
        shake_circles(centers, radii, magnitude)
        radii = grow_all_circles(centers, radii)
        new_sum = np.sum(radii)
        if new_sum > current_sum:
            current_sum = new_sum
    
    # Phase 3: Fine-grained local optimization
    for iteration in range(30000):
        i = random.randint(0, n - 1)
        step = 0.008 * (0.99995 ** iteration)
        
        move, gain = try_local_moves(centers, radii, i, step, n_attempts=24)
        if move is not None and gain > 1e-12:
            centers[i, 0] = move[0]
            centers[i, 1] = move[1]
            radii[i] = move[2]
    
    # Phase 4: Another round of shake and grow with tiny perturbations

        magnitude = 0.01 * (0.95 ** cycle)
        shake_circles(centers, radii, magnitude)

    # Phase 5: Final fine-tuning
    for iteration in range(20000):
        i = random.randint(0, n - 1)
        step = 0.003 * (0.99997 ** iteration)

        if move is not None and gain > 1e-14:
            centers[i, 0] = move[0]
            centers[i, 1] = move[1]
            radii[i] = move[2]
    
    # Final growth pass
    radii = grow_all_circles(centers, radii)
    
    return centers, radii, np.sum(radii)
