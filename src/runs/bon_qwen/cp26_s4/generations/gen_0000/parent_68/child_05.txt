import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(params):
    """Negative sum of radii (minimization equivalent)"""
    return -np.sum(params[2::3])

def constraints_fun(params):
    """Returns array of constraint values >= 0 for feasibility"""
    n = N_CIRCLES
    c = np.zeros(5*n + n*(n-1)//2)
    idx = 0
    
    # Boundary and non-negativity constraints
    for i in range(n):
        b = i * 3
        r = params[b+2]
        x = params[b]
        y = params[b+1]
        
        c[idx] = r                  # r >= 0
        idx += 1
        c[idx] = x - r              # x - r >= 0
        idx += 1
        c[idx] = 1.0 - x - r        # 1 - x - r >= 0
        idx += 1
        c[idx] = y - r              # y - r >= 0
        idx += 1
        c[idx] = 1.0 - y - r        # 1 - y - r >= 0
        idx += 1
        
    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            xi, yi, ri = params[i*3], params[i*3+1], params[i*3+2]
            xj, yj, rj = params[j*3], params[j*3+1], params[j*3+2]
            # dist^2 >= (ri + rj)^2
            c[idx] = (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            idx += 1
            
    return c

def run_packing():
    n = N_CIRCLES
    best_params = None
    best_val = -np.inf
    
    # Variable bounds: x,y in [0,1], r in [0,0.5]
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraints_fun}
    
    # Multiple restarts to escape local optima
    for seed in range(10):
        np.random.seed(seed)
        init_pts = np.zeros((n, 3))
        
        # Grid-based initialization with small perturbation
        for i in range(n):
            row = i // 6
            col = i % 6
            cx = (col + 0.5) / 6.0 + 0.02 * np.random.randn()
            cy = (row + 0.5) / 5.0 + 0.02 * np.random.randn()
            cx = np.clip(cx, 0.1, 0.9)
            cy = np.clip(cy, 0.1, 0.9)
            init_pts[i, 0] = cx
            init_pts[i, 1] = cy
            init_pts[i, 2] = 0.01  # Small initial radius
            
        init_params = init_pts.flatten()
        
        try:
            res = minimize(objective, init_params, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
            
            # Check if solution is feasible enough (SLSQP tolerance)
            if np.all(constraints_fun(res.x) >= -1e-9):
                if -res.fun > best_val:
                    best_val = -res.fun
                    best_params = res.x
        except Exception:
            pass
            
    if best_params is None:
        # Fallback in case of failure
        return np.zeros((n, 2)), np.zeros(n), 0.0
        
    centers = best_params.reshape(n, 3)[:, :2]
    radii = best_params.reshape(n, 3)[:, 2]
    
    return centers, radii, np.sum(radii)
