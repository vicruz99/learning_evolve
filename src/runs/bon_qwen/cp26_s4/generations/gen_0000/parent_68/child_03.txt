import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple[np.ndarray, np.ndarray, float]: (centers, radii, sum_radii)
    """
    np.random.seed(123)
    n = 26
    
    # --- 1. Initialization ---
    # Start with circles slightly inside the boundary to avoid initial constraint violations
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.ones(n) * 0.02  # Small initial radii
    
    # --- 2. Simulation Parameters ---
    steps = 3000          # Total optimization steps
    growth_step = 0.0004  # Radii expansion rate per step
    repulsion_strength = 1.5 # Stiffness of circle-circle repulsion
    boundary_strength = 10.0 # Stiffness of boundary repulsion
    damping = 0.5         # Velocity damping factor to prevent oscillation
    
    velocities = np.zeros((n, 2))
    
    # --- 3. Simulation Loop ---
    for _ in range(steps):
        # A. Expansion: Grow all radii
        radii += growth_step
        
        # B. Calculate Forces
        forces = np.zeros((n, 2))
        
        # -- Pairwise Repulsion --
        # Vectorized computation for all pairs
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # Shape (n, n, 2)
        dists = np.sqrt(np.sum(diff**2, axis=2)) # Shape (n, n)
        
        # Sum of radii for all pairs
        sum_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        # Calculate overlap (positive if overlapping)
        # Overlap = (r_i + r_j) - distance
        overlaps = sum_radii - dists
        
        # We only apply force if there is an overlap (or contact)
        # To smooth the force field and prevent singularities at dist=0,
        # we clamp distances.
        safe_dists = np.where(dists < 1e-9, 1e-9, dists)
        
        # Unit direction vectors (from j to i)
        dirs = diff / safe_dists[:, :, np.newaxis]
        
        # Force magnitude: proportional to overlap
        force_mag = np.maximum(0, overlaps) * repulsion_strength
        
        # Accumulate forces: sum of forces exerted by all other circles on i
        # forces[i] += sum_j ( force_mag[i, j] * dirs[i, j] )
        # Since diff[i, j] = c_i - c_j, pushing i away from j is in direction +diff
        forces += np.sum(force_mag[:, :, np.newaxis] * dirs, axis=1)
        
        # -- Boundary Repulsion --
        # Check distance to boundaries and apply force if inside margin
        # Left boundary (x=0)
        dist_to_left = centers[:, 0] - radii
        mask_left = dist_to_left < 0
        forces[mask_left, 0] -= boundary_strength * dist_to_left[mask_left]
        
        # Right boundary (x=1)
        dist_to_right = (1 - centers[:, 0]) - radii
        mask_right = dist_to_right < 0
        forces[mask_right, 0] += boundary_strength * dist_to_right[mask_right]
        
        # Bottom boundary (y=0)
        dist_to_bottom = centers[:, 1] - radii
        mask_bottom = dist_to_bottom < 0
        forces[mask_bottom, 1] -= boundary_strength * dist_to_bottom[mask_bottom]
        
        # Top boundary (y=1)
        dist_to_top = (1 - centers[:, 1]) - radii
        mask_top = dist_to_top < 0
        forces[mask_top, 1] += boundary_strength * dist_to_top[mask_top]
        
        # C. Update Positions
        # v = damping * v + force
        velocities = velocities * damping + forces
        centers += velocities
        
        # D. Hard Clamp to ensure strict validity (optional but good for stability)
        # Although forces should keep them in, numerical integration might overshoot.
        # We enforce strict bounds relative to current radii.
        # However, since radii are growing, strict clamping might be tricky if 
        # we clamp to old radii. The forces handle the continuous constraint.
        # We just ensure centers don't fly out of [0,1].
        centers[:, 0] = np.clip(centers[:, 0], 0, 1)
        centers[:, 1] = np.clip(centers[:, 1], 0, 1)
        
    # --- 4. Final Validation & Adjustment ---
    # Ensure strict validity for the final output
    # If any circle is slightly out, shrink it or move it.
    # Given the strong boundary forces, this is unlikely to be needed for the whole shape,
    # but let's be safe.
    
    # Check overlaps and resolve any residual numerical errors
    # This step is crucial for the strict validation function
    for _ in range(100):
        changed = False
        # Pairwise check
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    # Push apart
                    if dist == 0:
                        # If centers coincide, move randomly
                        centers[i] += np.random.randn(2) * 0.01
                    else:
                        direction = (centers[i] - centers[j]) / dist
                        push = (min_dist - dist) / 2 + 1e-7
                        centers[i] += direction * push
                        centers[j] -= direction * push
                    changed = True
        
        # Boundary check
        for i in range(n):
            r = radii[i]
            # Clip centers to valid range [r, 1-r]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
            
        if not changed:
            break

    # Re-calculate valid radii based on final positions to be absolutely safe
    # Although the simulation maximized radii, the final clip might make them slightly invalid 
    # if positions were forced. But our loop above handles this.
    # Actually, to be 100% compliant with max radii, we should just output what we have.
    # The validation function allows 1e-12 tolerance.
    
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
