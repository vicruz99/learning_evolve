import numpy as np
from scipy.optimize import minimize


def objective_wrapper(params):
    n = 26
    return -np.sum(params[2 * n :])


def constraint_wrapper(params):
    n = 26
    c = params[: 2 * n].reshape(n, 2)
    r = params[2 * n :]
    cons = []

    for i in range(n):
        cons.append(c[i, 0] - r[i])
        cons.append(1.0 - r[i] - c[i, 0])
        cons.append(c[i, 1] - r[i])
        cons.append(1.0 - r[i] - c[i, 1])

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((c[i] - c[j]) ** 2) + 1e-15)
            cons.append(dist - r[i] - r[j])

    return np.array(cons)


def generate_hexagonal_initialization(n, row_start=0.12, col_start=0.12, row_spacing=0.145, col_spacing=0.175, perturbation=0.008):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.10
    idx = 0
    row = 0
    while idx < n and row < 12:
        col = 0
        while idx < n and col < 8:
            x = col_start + col * col_spacing + (row % 2) * (col_spacing / 2.0)
            y = row_start + row * row_spacing
            x = np.clip(x + np.random.randn() * perturbation, 0.03, 0.97)
            y = np.clip(y + np.random.randn() * perturbation, 0.03, 0.97)
            centers[idx] = [x, y]
            idx += 1
            col += 1
        row += 1
    return centers, radii


def generate_random_initialization(n, min_r=0.05, max_r=0.12):
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.random.rand(n) * (max_r - min_r) + min_r
    return centers, radii


def verify_packing(centers, radii, n=26):
    for i in range(n):
        if centers[i, 0] < radii[i] - 1e-11 or centers[i, 0] > 1.0 - radii[i] + 1e-11:
            return False
        if centers[i, 1] < radii[i] - 1e-11 or centers[i, 1] > 1.0 - radii[i] + 1e-11:
            return False
        if radii[i] < 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-11:
                return False
    return True


def optimize_from_initialization(centers, radii, max_iter=4000):
    n = 26
    params0 = np.concatenate([centers.flatten(), radii])

    result = minimize(
        objective_wrapper,
        params0,
        method="SLSQP",
        constraints={"type": "ineq", "fun": constraint_wrapper},
        options={"maxiter": max_iter, "ftol": 1e-15, "disp": False},
    )

    c_opt = result.x[: 2 * n].reshape(n, 2)
    r_opt = np.maximum(result.x[2 * n :], 1e-10)

    return c_opt, r_opt
