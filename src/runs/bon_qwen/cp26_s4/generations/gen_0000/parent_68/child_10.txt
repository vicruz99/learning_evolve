import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Generate a good initial guess using a hexagonal lattice
    # Hexagonal packing allows for higher density than square packing.
    # We estimate a radius that might fit 26 circles. 
    # Area argument: 26 * pi * r^2 approx 1 * density. 
    # Hex density ~ 0.907. r ~ sqrt(0.907 / (26 * pi)) ~ 0.105.
    # However, boundary effects reduce this. Let's start with r=0.095.
    
    initial_r = 0.095
    centers = []
    
    # Create a hexagonal grid
    # Rows spaced by sqrt(3) * diameter / 2 ? No, vertical distance is sqrt(3)*r * 2 / 2?
    # Distance between centers in hex packing: horizontal 2r, vertical sqrt(3)r.
    # Actually, if we arrange rows, vertical step is sqrt(3)/2 * (2r) = sqrt(3)r.
    
    row_spacing = np.sqrt(3) * initial_r
    col_spacing = 2 * initial_r
    
    y = initial_r
    row_idx = 0
    
    # We need to collect 26 points
    while len(centers) < n:
        x = initial_r
        # Offset odd rows
        if row_idx % 2 == 1:
            x += initial_r # Shift by r (half of 2r)
        
        while x <= 1 - initial_r and len(centers) < n:
            centers.append([x, y])
            x += col_spacing
        
        y += row_spacing
        row_idx += 1
        
    centers = np.array(centers[:n])
    radii = np.full(n, initial_r)
    
    # Combine variables: x_0, y_0, r_0, x_1, y_1, r_1, ...
    # Or separate arrays? Scipy minimizes a 1D array.
    # Let's flatten: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.hstack([centers.flatten(), radii])
    
    # 2. Define Objective Function
    def objective(vars_flat):
        # vars_flat shape (78,) -> 26 * 3
        r = vars_flat[2::3] # Every 3rd element starting from index 2
        return -np.sum(r) # Maximize sum(r) -> minimize -sum(r)

    # 3. Define Constraints
    # Constraints must return values >= 0
    
    def boundary_constraints(vars_flat):
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        
        cons = []
        for i in range(n):
            idx = i * 3
            x = vars_flat[idx]
            y = vars_flat[idx+1]
            r = vars_flat[idx+2]
            
            # Boundary constraints
            cons.append(x - r)
            cons.append(1 - x - r)
            cons.append(y - r)
            cons.append(1 - y - r)
        return np.array(cons)

    def overlap_constraints(vars_flat):
        cons = []
        for i in range(n):
            idx_i = i * 3
            xi = vars_flat[idx_i]
            yi = vars_flat[idx_i+1]
            ri = vars_flat[idx_i+2]
            
            for j in range(i + 1, n):
                idx_j = j * 3
                xj = vars_flat[idx_j]
                yj = vars_flat[idx_j+1]
                rj = vars_flat[idx_j+2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                # dist >= ri + rj  => dist^2 >= (ri+rj)^2
                # But sqrt is safer for gradient? Or just use dist - (ri+rj) >= 0
                # Using dist - (ri+rj) is non-smooth at 0? 
                # Actually dist >= sum_r is better.
                dist = np.sqrt(dist_sq)
                cons.append(dist - (ri + rj))
        return np.array(cons)

    # Combine constraints into a list of dictionaries for SLSQP
    constraints = []
    
    # Boundary constraints
    constraints.append({
        'type': 'ineq',
        'fun': boundary_constraints
    })
    
    # Overlap constraints
    constraints.append({
        'type': 'ineq',
        'fun': overlap_constraints
    })

    # Bounds for variables
    # x, y in [0, 1]
    # r >= 0. To prevent solver from finding trivial solution r=0, 
    # we rely on objective to push r up, but lower bound 0 is safe.
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 1.0)) # r (upper bound 1 is loose but safe)

    # 4. Run Optimization
    # Using SLSQP. It handles bounds and constraints.
    # We might need to tune options or use a robust method.
    # N=26 -> 78 variables, ~400 constraints.
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'ftol': 1e-9, 'maxiter': 1000})

    if not result.success:
        # Fallback or print warning, but we return best found
        pass

    # Extract results
    final_vars = result.x
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        idx = i * 3
        final_centers[i] = [final_vars[idx], final_vars[idx+1]]
        final_radii[i] = final_vars[idx+2]
        
    # Clip any negative radii due to numerical noise (though bounds prevent it)
    final_radii = np.maximum(final_radii, 0.0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
