import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_initial_guess():
    """Creates a feasible hexagonal lattice initial configuration."""
    n = N_CIRCLES
    r = 0.1
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    idx = 0
    y = r
    row = 0
    while idx < n and y <= 1 - r:
        x = r
        offset = r if row % 2 == 1 else 0
        while x <= 1 - r and idx < n:
            centers[idx] = [x + offset, y]
            idx += 1
            x += 2 * r
        y += r * np.sqrt(3)
        row += 1
    return centers, radii

def boundary_constraints(vars):
    """Ensures all circles stay within [0,1]x[0,1]."""
    n = N_CIRCLES
    res = np.zeros(4 * n)
    for i in range(n):
        res[4*i] = vars[3*i] - vars[3*i+2]
        res[4*i+1] = 1 - vars[3*i] - vars[3*i+2]
        res[4*i+2] = vars[3*i+1] - vars[3*i+2]
        res[4*i+3] = 1 - vars[3*i+1] - vars[3*i+2]
    return res

def pair_constraints(vars):
    """Ensures no two circles overlap. Returns lower-triangle pairwise separations."""
    n = N_CIRCLES
    C = vars[:2*n].reshape(n, 2)
    R = vars[2*n:]
    diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    sum_r = R[:, np.newaxis] + R[np.newaxis, :]
    mask = np.tril(np.ones((n, n), dtype=bool), -1)
    return dists[mask] - sum_r[mask]

def obj_fn(vars):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars[2::3])

def run_packing():
    centers_init, radii_init = get_initial_guess()
    x0 = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]
        
    bounds = [(0, 1)] * (2*N_CIRCLES) + [(1e-4, 0.5)] * N_CIRCLES
    
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': pair_constraints}
    ]
    
    res = minimize(obj_fn, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    centers_opt = np.array([[res.x[3*i], res.x[3*i+1]] for i in range(N_CIRCLES)])
    radii_opt = np.array([res.x[3*i+2] for i in range(N_CIRCLES)])
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
