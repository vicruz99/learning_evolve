import numpy as np
from scipy.optimize import minimize
import math


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def all_constraints(x):
    """Compute all constraint values. Must be >= 0 for feasibility."""
    n = len(x) // 3
    constraints = []

    # Boundary and radius constraints
    for i in range(n):
        xi = x[3 * i]
        yi = x[3 * i + 1]
        ri = x[3 * i + 2]
        constraints.append(xi - ri)
        constraints.append(1.0 - (xi + ri))
        constraints.append(yi - ri)
        constraints.append(1.0 - (yi + ri))
        constraints.append(ri)

    # Non-overlap constraints (using squared distance)
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[3 * i] - x[3 * j]
            dy = x[3 * i + 1] - x[3 * j + 1]
            dist_sq = dx * dx + dy * dy
            sum_r = x[3 * i + 2] + x[3 * j + 2]
            constraints.append(dist_sq - sum_r * sum_r)

    return np.array(constraints)


def make_hexagonal_init(n, r_init, row_pattern):
    """Create hexagonal packing initialization."""
    sqrt3 = math.sqrt(3)
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    idx = 0

    for row_idx, count in enumerate(row_pattern):
        y = r_init + row_idx * r_init * sqrt3
        if row_idx % 2 == 0:
            for col_idx in range(count):
                x = r_init + col_idx * 2.0 * r_init
                centers[idx] = [x, y]
                idx += 1
        else:
            for col_idx in range(count):
                x = 2.0 * r_init + col_idx * 2.0 * r_init
                centers[idx] = [x, y]
                idx += 1

    return centers, radii


def flatten_state(centers, radii):
    """Flatten centers and radii into optimization vector."""
    n = centers.shape[0]
    x = np.zeros(3 * n)
    for i in range(n):
        x[3 * i] = centers[i, 0]
        x[3 * i + 1] = centers[i, 1]
        x[3 * i + 2] = radii[i]
    return x


def unflatten_state(x):
    """Unflatten optimization vector back to centers and radii."""
    n = len(x) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = x[3 * i + 2]
    return centers, radii


def clamp_state(centers, radii):
    """Ensure circles are within bounds."""
    n = centers.shape[0]
    for i in range(n):
        r = max(0.0, radii[i])
        radii[i] = r
        centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))
    return centers, radii


def random_init(n):
    """Create random initialization."""
    centers = np.random.uniform(0.2, 0.8, size=(n, 2))
    radii = np.full(n, 0.02)
    return centers, radii


def try_optimization(x0, max_iter=2000, tol=1e-14):
    """Run optimization from given initial state."""
    cons = {'type': 'ineq', 'fun': all_constraints}
    try:
        result = minimize(
            objective, x0, method='SLSQP', constraints=cons,
            options={'maxiter': max_iter, 'ftol': tol, 'disp': False}
        )
        return result.x, result.success, result.fun
    except Exception:
        return x0, False, 0.0
