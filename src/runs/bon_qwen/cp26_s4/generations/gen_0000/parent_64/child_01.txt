import numpy as np
from scipy.optimize import minimize

def objective(vars, n=26, lam=1000.0):
    centers = vars[:n*2].reshape(n, 2)
    radii = vars[n*2:]
    
    x = centers[:,0]
    y = centers[:,1]
    
    # Boundary penalties: circles must be inside [0,1]x[0,1]
    pen = np.sum(np.maximum(0, radii - x)**2)
    pen += np.sum(np.maximum(0, radii - (1 - x))**2)
    pen += np.sum(np.maximum(0, radii - y)**2)
    pen += np.sum(np.maximum(0, radii - (1 - y))**2)
    
    # Overlap penalties: distance between centers >= sum of radii
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, 1e9)
    
    sums_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = np.maximum(0, sums_radii - dists)
    pen += np.sum(overlap**2)
    
    # Objective: maximize sum of radii (minimize negative sum) + penalty
    return -np.sum(radii) + lam * pen

def run_packing():
    n = 26
    best_vars = None
    best_score = np.inf
    
    # Multiple restarts to find good global configuration
    for seed in range(80):
        np.random.seed(seed)
        centers0 = np.random.rand(n, 2)
        radii0 = np.full(n, 0.06)
        vars0 = np.concatenate([centers0.flatten(), radii0])
        
        # Bounds: x,y in [0,1], r in [small, 0.5]
        bounds = [(0, 1)] * (n*2) + [(1e-6, 0.5)] * n
        
        # Phase 1: Moderate penalty to grow circles and arrange them
        res1 = minimize(objective, vars0, args=(n, 500.0), method='L-BFGS-B',
                        bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-10})
        
        # Phase 2: Higher penalty to strictly enforce constraints
        res2 = minimize(objective, res1.x, args=(n, 1000.0), method='L-BFGS-B',
                        bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-10})
        
        if res2.fun < best_score:
            best_score = res2.fun
            best_vars = res2.x.copy()
            
    centers = best_vars[:n*2].reshape(n, 2)
    radii = best_vars[n*2:]
    
    # Repair step to guarantee strict validity within tolerance
    for i in range(n):
        r = radii[i]
        cx, cy = centers[i]
        max_r = min(cx, 1-cx, cy, 1-cy)
        if r > max_r + 1e-9:
            radii[i] = max_r
            
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            if d < radii[i] + radii[j] - 1e-9:
                excess = radii[i] + radii[j] - d
                radii[i] -= excess/2
                radii[j] -= excess/2
                
    radii = np.maximum(radii, 1e-9)
    return centers, radii, np.sum(radii)
