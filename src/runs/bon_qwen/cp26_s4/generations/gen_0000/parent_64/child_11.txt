import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_neg_sum_radii(centers_flat):
    """
    Computes the negative sum of maximal feasible radii for a given set of centers.
    Used as the objective function for minimization.
    """
    centers = centers_flat.reshape(-1, 2)
    
    # Vectorized pairwise distance computation
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_neighbor_dist = np.min(dists / 2, axis=1)
    
    # Distance to square boundaries
    x = centers[:, 0]
    y = centers[:, 1]
    min_boundary_dist = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    
    # Maximal valid radius for each circle
    radii = np.minimum(min_neighbor_dist, min_boundary_dist)
    return -np.sum(radii)

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Stage 1: Global optimization with Differential Evolution
    # popsize=20 and maxiter=2000 provide a strong exploration for 52 dimensions
    result = differential_evolution(compute_neg_sum_radii, bounds, 
                                    maxiter=2000, popsize=20, seed=42, 
                                    tol=1e-7, atol=1e-7, workers=1)
    
    # Stage 2: Local refinement with L-BFGS-B for high precision
    res_local = minimize(compute_neg_sum_radii, result.x, method='L-BFGS-B', 
                         bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-12})
    
    # Ensure centers stay strictly within bounds after optimization
    centers = np.clip(res_local.x.reshape(-1, 2), 0.0, 1.0)
    
    # Final radius computation from optimized centers
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_neighbor_dist = np.min(dists / 2, axis=1)
    
    x = centers[:, 0]
    y = centers[:, 1]
    min_boundary_dist = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    radii = np.minimum(min_neighbor_dist, min_boundary_dist)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
