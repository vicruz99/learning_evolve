def get_boundary_constraints(centers):
    """
    Compute the upper bounds for radii based on circle centers 
    and the unit square boundaries [0,1]x[0,1].
    
    r_i <= x_i
    r_i <= 1 - x_i
    r_i <= y_i
    r_i <= 1 - y_i
    
    So r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    """
    n = centers.shape[0]
    bounds = []
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1.0 - x, y, 1.0 - y)
        if r_max < 0:
            r_max = 0.0 # Should not happen if centers in [0,1]
        bounds.append(r_max)
    return np.array(bounds)

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= dist(c_i, c_j) for all i < j
      0 <= r_i <= boundary_limit_i
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Bounds for r_i: [0, boundary_limit]
    boundary_limits = get_boundary_constraints(centers)
    bounds_r = [(0, b) for b in boundary_limits]
    
    # Constraints: A_ub * r <= b_ub
