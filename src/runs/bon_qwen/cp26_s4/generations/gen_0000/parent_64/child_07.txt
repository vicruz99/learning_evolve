import numpy as np
from scipy.optimize import linprog, minimize

def compute_sum_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Returns the maximum sum of radii.
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum(r_i) <=> Minimize -sum(r_i)
    
    # Constraints: 4 per circle (boundaries) + 1 per pair (non-overlap)
    n_con = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_con, n))
    b_ub = np.zeros(n_con)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        for bound in (x, 1 - x, y, 1 - y):
            A_ub[k, i] = 1.0
            b_ub[k] = bound
            k += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dist
            k += 1
            
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun
    return 0.0

def run_packing():

    centers = best_x.reshape(n, 2)
