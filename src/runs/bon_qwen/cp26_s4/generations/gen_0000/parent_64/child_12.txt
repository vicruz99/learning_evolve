**Strategy**:
1. **Nested Optimization**: Fix centers, solve LP for optimal radii. Then optimize centers.

import numpy as np
from scipy.optimize import linprog, minimize


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii.
    
    Maximizes sum(r_i) subject to:
    - r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i
    - r_i + r_j <= ||c_i - c_j|| for all i < j
    - r_i >= 0 for all i
    """
    n = centers.shape[0]
    c = -np.ones(n)
    
    n_constraints = n + n * (n - 1) // 2
    A = np.zeros((n_constraints, n))
    b = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1.0 - x, y, 1.0 - y)
        A[idx, i] = 1.0
        b[idx] = r_max
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dist
            idx += 1
    
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return np.full(n, 0.001)


def neg_sum_radii(centers_flat):
    """Objective function: negative sum of optimal radii."""
    n = 26
    centers = centers_flat.reshape(n, 2)
    centers = np.clip(centers, 1e-10, 1.0 - 1e-10)
    radii = solve_radii_lp(centers)
    return -np.sum(radii)


def generate_hex_initial(n, seed):
    """Generate hexagonal grid initial configuration."""
    rng = np.random.RandomState(seed)
    centers = []
    
    spacing_x = 0.22
    spacing_y = spacing_x * np.sqrt(3) / 2
    
    row = 0
    while len(centers) < n:
        col = 0
        while len(centers) < n and col < 8:
            x = col * spacing_x + (spacing_x / 2.0 if row % 2 == 1 else 0)
            y = row * spacing_y
            
            x += rng.uniform(-0.03, 0.03)
            y += rng.uniform(-0.03, 0.03)
            
            x = 0.05 + (x / (5.0 * spacing_x)) * 0.9
            y = 0.05 + (y / (5.0 * spacing_y)) * 0.9
            
            x = np.clip(x, 0.02, 0.98)
            y = np.clip(y, 0.02, 0.98)
            
            centers.append([x, y])
            col += 1
        row += 1
        if row > 12:
            break
    
    while len(centers) < n:
        centers.append([rng.uniform(0.05, 0.95), rng.uniform(0.05, 0.95)])
    
    return np.array(centers[:n])


def generate_random_initial(n, seed):
    """Generate random initial configuration with good spread."""
    rng = np.random.RandomState(seed)
    centers = []
    attempts = 0
    
    while len(centers) < n and attempts < 20000:
        x = rng.uniform(0.1, 0.9)
        y = rng.uniform(0.1, 0.9)
        
        min_dist = 1.0
        for c in centers:
            d = np.sqrt((x - c[0]) ** 2 + (y - c[1]) ** 2)
            min_dist = min(min_dist, d)
        
        if min_dist > 0.10:
            centers.append([x, y])
        attempts += 1
    
    while len(centers) < n:
        centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
    
    return np.array(centers[:n])


def generate_grid_initial(n, seed):
    """Generate grid-based initial configuration."""
    rng = np.random.RandomState(seed)
    centers = []
    
    nx = 5
    ny = 6
    xs = np.linspace(0.08, 0.92, nx)
    ys = np.linspace(0.08, 0.92, ny)
    
    for y in ys:
        for x in xs:
            if len(centers) >= n:
                break
            x += rng.uniform(-0.015, 0.015)
            y += rng.uniform(-0.015, 0.015)
            centers.append([np.clip(x, 0.02, 0.98), np.clip(y, 0.02, 0.98)])
        if len(centers) >= n:
            break
    
    while len(centers) < n:
        centers.append([rng.uniform(0.05, 0.95), rng.uniform(0.05, 0.95)])
    
    return np.array(centers[:n])


def coordinate_descent(centers, max_outer=80, max_inner=200):
    """Coordinate descent: optimize one center at a time."""
    n = centers.shape[0]
    
    for _ in range(max_outer):
        for i in range(n):
            old_center = centers[i].copy()
            best_sum = -1e10
            best_pos = old_center.copy()
            
            radii_before = solve_radii_lp(centers)
            current_sum = np.sum(radii_before)
            
            for step in [0.015, 0.008, 0.004, 0.002]:
                directions = [
                    [step, 0], [-step, 0], [0, step], [0, -step],
                    [step, step], [-step, step], [step, -step], [-step, -step]
                ]
                
                improved = False
                for dx, dy in directions:
                    centers[i] = old_center + np.array([dx, dy])
                    centers[i] = np.clip(centers[i], 1e-10, 1.0 - 1e-10)
                    
                    radii = solve_radii_lp(centers)
                    s = np.sum(radii)
                    
                    if s > best_sum:
                        best_sum = s
                        best_pos = centers[i].copy()
                        improved = True
                
                if not improved:
                    break
            
            centers[i] = best_pos
    
    return centers


def force_relax(centers, n_iter=3000):
    """Force-directed relaxation to improve packing."""
    n = centers.shape[0]
    radii = solve_radii_lp(centers)
    temp = 0.5
    
    for iteration in range(n_iter):
        forces = np.zeros_like(centers)
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0.05:
                forces[i, 0] += 2.0 / max(0.02, x - r + 0.01)
            if x + r > 0.95:
                forces[i, 0] -= 2.0 / max(0.02, 1.0 - x - r + 0.01)
            if y - r < 0.05:
                forces[i, 1] += 2.0 / max(0.02, y - r + 0.01)
            if y + r > 0.95:
                forces[i, 1] -= 2.0 / max(0.02, 1.0 - y - r + 0.01)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist * 1.15 and dist > 1e-10:
                    repulse = (min_dist * 1.15 - dist) / max(1e-8, dist) * 3.0
                    direction = diff / max(1e-8, dist)
                    forces[i] += direction * repulse
                    forces[j] -= direction * repulse
        
        centers += forces * temp * 0.008
        centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
        
        radii = solve_radii_lp(centers)
        temp *= 0.9995
    
    return centers


def run_packing():
    n = 26
    
    best_centers = None

    best_sum = -1.0
    
    # Phase 1: Nelder-Mead optimization with multiple starts
    configs = []
    for seed in range(8):

        configs.append(generate_random_initial(n, seed))

    for idx, init_centers in enumerate(configs):
        centers_flat = init_centers.flatten()
        
        result = minimize(

            centers_flat,
            method='Nelder-Mead',
            options={
                'maxiter': 25000,
                'xatol': 1e-11,
                'fatol': 1e-11,
                'adaptive': True
            }
        )
        
        opt_centers = result.x.reshape(n, 2)

        # Phase 2: Coordinate descent refinement
        opt_centers = coordinate_descent(opt_centers, max_outer=60, max_inner=200)

        # Phase 4: Another round of coordinate descent
