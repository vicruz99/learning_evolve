import numpy as np

def initialize_centers(N, trial_type, seed):
    centers = np.zeros((N, 2))
    if trial_type == "hex":
        rows = [5, 6, 5, 6, 4]
        idx = 0
        for r_i, cnt in enumerate(rows):
            y = r_i * np.sqrt(3) / 2.0
            x_shift = 0.5 * (r_i % 2)
            for c_i in range(cnt):
                centers[idx] = [c_i + x_shift, y]
                idx += 1
    elif trial_type == "grid":
        idx = 0
        for i in range(5):
            for j in range(5):
                centers[idx] = [i, j]
                idx += 1
        centers[25] = [2.0, 2.5]
    else:
        np.random.seed(seed)
        centers = np.random.rand(N, 2)
        
    c_min = centers.min(axis=0)
    c_max = centers.max(axis=0)
    span = c_max - c_min
    if span[0] > 1e-9 and span[1] > 1e-9:
        scale = min((0.8)/span[0], (0.8)/span[1])
        centers = (centers - c_min) * scale + (1.0 - scale * span) / 2.0
    else:
        centers = np.random.rand(N, 2) * 0.8 + 0.1
    return centers

def optimize_packing(centers, radii, N, steps=15000):
    dt = 2e-4
    decay = 0.999
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    for step in range(steps):
        radii += dt
        
        # Vectorized pairwise difference and distance
        diff = centers[:, None, :] - centers[None, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        dist = np.sqrt(np.maximum(dist_sq, 1e-12))
        
        # Overlap calculation (0 if no overlap)
        min_dist = radii[:, None] + radii[None, :]
        overlap = np.maximum(min_dist - dist, 0.0)
        
        # Compute displacement vectors
        inv_dist = 1.0 / dist[:, :, None]
        unit_vec = diff * inv_dist
        push_mag = (overlap * 0.5)[:, :, None]
        disp = unit_vec * push_mag
        
        # Apply forces
        centers += np.sum(disp, axis=1)
        
        # Project to boundaries
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
        
        current_sum = radii.sum()
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        dt *= decay
        
    return best_centers, best_radii, best_sum

def run_packing():
    N = 26
    trials = [
        ("hex", 0),
        ("rand", 1),
        ("rand", 2),
        ("grid", 3)
    ]
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    for t_type, t_seed in trials:

        radii = np.full(N, 0.01)
        opt_c, opt_r, opt_sum = optimize_packing(centers, radii, N)
        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_c
            best_radii = opt_r
            
    # Final validation adjustment
    valid = True
    for i in range(N):
        x, y = best_centers[i]
        r = best_radii[i]
        if x < r - 1e-10 or x > 1 - r + 1e-10 or y < r - 1e-10 or y > 1 - r + 1e-10:
            valid = False
            break
    if valid:
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if d < best_radii[i] + best_radii[j] - 1e-10:
                    valid = False
                    break
            if not valid: break
            
    if not valid:
        best_radii *= 0.9995
        best_sum = best_radii.sum()
        
    return best_centers, best_radii, best_sum
