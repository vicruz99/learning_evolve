import numpy as np
from scipy.optimize import minimize

def objective_func(x):
    """Objective function: minimize negative sum of radii (equivalent to maximizing sum)"""
    return -np.sum(x[2::3])

def constraint_func(x):
    """
    Returns array of constraint values g(x) >= 0.
    Includes boundary constraints and non-overlap constraints.
    """
    n = 26
    m = 4 * n + n * (n - 1) // 2
    c = np.zeros(m)
    idx = 0
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    for i in range(n):
        xi = x[3*i]
        yi = x[3*i+1]
        ri = x[3*i+2]
        c[idx] = xi - ri
        c[idx+1] = 1 - xi - ri
        c[idx+2] = yi - ri
        c[idx+3] = 1 - yi - ri
        idx += 4
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            d2 = (xi - xj)**2 + (yi - yj)**2
            r_sum = ri + rj
            c[idx] = d2 - r_sum**2
            idx += 1
    return c

def get_initial_configs():
    """Generate multiple deterministic initial configurations to avoid poor local minima"""
    n = 26
    configs = []
    
    # 1. Standard 5x5 grid + one in center
    c1 = []
    for i in range(5):
        for j in range(5):
            c1.append([0.1 + 0.2*i, 0.1 + 0.2*j])
    c1.append([0.5, 0.5])
    c1 = np.array(c1)
    configs.append(np.concatenate([c1.flatten(), np.full(26, 0.04)]))
    
    # 2. Perturbed grid to break symmetry
    c2 = c1.copy()
    for i in range(26):
        c2[i, 0] += 0.01 * (i % 5)
        c2[i, 1] += 0.01 * (i // 5)
    configs.append(np.concatenate([c2.flatten(), np.full(26, 0.04)]))
    
    # 3. Dense corner-starting layout
    c3 = []
    y = 0.05
    while y < 0.95 and len(c3) < 26:
        x = 0.05
        while x < 0.95 and len(c3) < 26:
            c3.append([x, y])
            x += 0.2
        y += 0.2
    c3 = np.array(c3)
    configs.append(np.concatenate([c3.flatten(), np.full(26, 0.04)]))
        
    return configs

def run_packing():
    n = 26
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_x = None
    
    inits = get_initial_configs()
    
    for x0 in inits:
        x0 = np.copy(x0)
        # Ensure initial guess respects bounds strictly
        for i in range(2*n):
            if x0[i] < 0.05: x0[i] = 0.1
            elif x0[i] > 0.95: x0[i] = 0.9
        for i in range(n):
            if x0[2*n+i] < 0.01: x0[2*n+i] = 0.05
            
        res = minimize(
            objective_func, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons, 
            options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
        )
        
        s = -res.fun
        if s > best_sum:
            best_sum = s
            best_x = res.x
            
    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]
    return centers, radii, best_sum
