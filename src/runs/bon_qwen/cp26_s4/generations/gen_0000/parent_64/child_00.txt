import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, N, lambda_pen):
    centers = vars[:N*2].reshape(N, 2)
    radii = vars[N*2:]

    x = centers[:, 0]
    y = centers[:, 1]
    r = radii

    # Boundary penalties: circles must be inside [0,1]x[0,1]
    # Penalty is quadratic for any violation
    b_pen = np.sum(np.maximum(0.0, r - x)**2)
    b_pen += np.sum(np.maximum(0.0, x + r - 1.0)**2)
    b_pen += np.sum(np.maximum(0.0, r - y)**2)
    b_pen += np.sum(np.maximum(0.0, y + r - 1.0)**2)

    # Overlap penalties: distance between centers must be >= sum of radii
    dx = centers[:, 0, np.newaxis] - centers[:, 0, np.newaxis].T
    dy = centers[:, 1, np.newaxis] - centers[:, 1, np.newaxis].T
    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
    rad_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    violations = rad_sum - dist
    penalties = np.maximum(0.0, violations)
    # Only consider each pair once (upper triangle)
    o_pen = np.sum(np.triu(penalties, k=1)**2)

    # Objective: maximize sum of radii -> minimize negative sum
    return -np.sum(radii) + lambda_pen * (b_pen + o_pen)

def run_packing():
    N = 26
    best_sum = -np.inf
    best_sol = None
    
    # Try multiple initial configurations to escape poor local minima
    for seed in range(15):
        rng = np.random.default_rng(seed)
        
        # Initialize centers in a perturbed hexagonal-like grid
        init_c = []
        # Row distribution to sum to 26
        row_lengths = [5, 5, 5, 5, 4, 2]
        y_step = 1.0 / 7.0
        c = 0
        for r_idx, length in enumerate(row_lengths):
            y_base = 0.12 + r_idx * y_step
            x_step = 1.0 / (length + 1)
            for col in range(length):
                if c >= N: break
                x = (col + 1) * x_step
                # Add controlled jitter
                x += rng.uniform(-0.03, 0.03)
                y = y_base + rng.uniform(-0.03, 0.03)
                init_c.append([np.clip(x, 0.05, 0.95), np.clip(y, 0.05, 0.95)])
                c += 1
        while len(init_c) < N:
            init_c.append([0.5 + rng.uniform(-0.2, 0.2), 0.5 + rng.uniform(-0.2, 0.2)])
            
        init_c = np.array(init_c)
        init_r = np.full(N, 0.06)
        x0 = np.concatenate([init_c.flatten(), init_r])
        
        bounds = [(0.0, 1.0)] * (N*2) + [(1e-4, 0.5)] * N
        
        # Optimize using L-BFGS-B with a high penalty weight
        res = minimize(compute_loss, x0, args=(N, 10000.0), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 4000, 'ftol': 1e-10, 'gtol': 1e-8})
        
        curr_c = res.x[:N*2].reshape(N, 2)
        curr_r = res.x[N*2:]
        
        # Strict validity check to ensure we only keep valid packings
        valid = True
        for i in range(N):
            xi, yi = curr_c[i]
            ri = curr_r[i]
            if xi - ri < -1e-9 or xi + ri > 1.0 + 1e-9 or yi - ri < -1e-9 or yi + ri > 1.0 + 1e-9:
                valid = False
                break
            for j in range(i+1, N):
                d = np.hypot(curr_c[i,0]-curr_c[j,0], curr_c[i,1]-curr_c[j,1])
                if d < curr_r[i] + curr_r[j] - 1e-9:
                    valid = False
                    break
            if not valid:
                break
                
        if valid:
            s = np.sum(curr_r)
            if s > best_sum:
                best_sum = s
                best_sol = (curr_c.copy(), curr_r.copy())
                
    # Fallback to a simple valid configuration if optimization fails
    if best_sol is None:
        c_fb = []
        r_fb = []
        for i in range(5):
            for j in range(5):
                c_fb.append([0.1 + j*0.16, 0.1 + i*0.16])
                r_fb.append(0.07)
        c_fb.append([0.5, 0.5])
        r_fb.append(0.01)
        best_sol = (np.array(c_fb), np.array(r_fb))
        best_sum = np.sum(r_fb)
        
    return best_sol[0], best_sol[1], best_sum
