import numpy as np

def compute_radii(centers):
    """
    Computes the maximum possible radius for each circle given their centers.
    Radius is limited by distance to boundaries and half the distance to the nearest neighbor.
    """
    n = centers.shape[0]
    
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    dx = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    dy = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    r_bound = np.minimum(dx, dy)

    # Pairwise distances between all circles
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Radius limited by nearest neighbor
    r_pair = np.min(dists, axis=1) / 2.0

    return np.minimum(r_bound, r_pair)
