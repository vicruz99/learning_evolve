import numpy as np
from scipy.optimize import minimize

def compute_obj(v, lam, n, tri):
    """
    Objective function for penalty-based circle packing optimization.
    v: flattened array [x1, y1, r1, x2, y2, r2, ...]
    lam: penalty weight
    n: number of circles
    tri: tuple of indices for upper triangle of distance matrix
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary violations: circle must be inside [0,1]x[0,1]
    b_viol = np.maximum(0, r - x) + np.maximum(0, x + r - 1)
    b_viol += np.maximum(0, r - y) + np.maximum(0, y + r - 1)
    pen_b = np.sum(b_viol**2)
    
    # Overlap violations
    C = np.column_stack((x, y))
    diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    R = r[:, np.newaxis] + r[np.newaxis, :]
    overlap = np.maximum(0, R - dist)
    pen_o = np.sum(overlap[tri]**2)
    
    # Minimize negative sum of radii + penalty
    return -np.sum(r) + lam * (pen_b + pen_o)

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = 0
    best_v = None
    best_radii = None
    
    tri = np.triu_indices(n, k=1)
    
    inits = []
    # 1. Hexagonal lattice initialization
    v_hex = np.zeros(n*3)
    idx = 0
    y = 0.15
    while idx < n:
        x = 0.15
        offset = 0.13 if int(y*10) % 2 == 0 else 0.0
        while x < 0.9 and idx < n:
            v_hex[3*idx] = x + offset
            v_hex[3*idx+1] = y
            v_hex[3*idx+2] = 0.08
            idx += 1
            x += 0.18
        y += 0.15
    inits.append(v_hex)
    
    # 2. Random starts
    for _ in range(6):
        v = np.random.uniform(size=n*3)
        v[0::3] = np.clip(v[0::3], 0.1, 0.9)
        v[1::3] = np.clip(v[1::3], 0.1, 0.9)
        v[2::3] = 0.05 + np.random.uniform(0, 0.05, n)
        inits.append(v)
        
    # Continuation method: gradually increase penalty weight
    for lam in [100, 500, 2000, 5000]:
        for v0 in inits:
            bounds = [(0, 1)]*n + [(0, 1)]*n + [(0, 0.5)]*n
            res = minimize(compute_obj, v0, args=(lam, n, tri), method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 2000, 'ftol': 1e-12})
            
            v_opt = res.x
            x = v_opt[0::3].copy()
            y = v_opt[1::3].copy()
            r = v_opt[2::3].copy()
            
            # Strict boundary clamping
            r = np.minimum(r, x)
            r = np.minimum(r, 1-x)
            r = np.minimum(r, y)
            r = np.minimum(r, 1-y)
            
            # Strict overlap resolution
            for _ in range(100):
                changed = False
                for i in range(n):
                    for j in range(i+1, n):
                        d = np.sqrt((x[i]-x[j])**2 + (y[i]-y[j])**2)
                        if d < r[i] + r[j]:
                            s = d / (r[i] + r[j])
                            r[i] *= s
                            r[j] *= s
                            changed = True
                if not changed:
                    break
                    
            s_curr = np.sum(r)
            if s_curr > best_sum:
                best_sum = s_curr
                best_v = np.column_stack((x, y))
                best_radii = r.copy()
                
    return best_v, best_radii, best_sum
