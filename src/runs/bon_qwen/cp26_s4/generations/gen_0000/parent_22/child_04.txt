import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import pdist

def packing_objective(x):
    """
    Objective function for differential evolution.
    Computes negative minimum separation distance (between circles and boundaries).
    DE minimizes this, thus maximizing the minimum separation.
    """
    centers = x.reshape(-1, 2)
    
    # Distance from each center to the nearest boundary
    d_x = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    d_y = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    d_bound = np.min(np.minimum(d_x, d_y))
    
    # Minimum pairwise distance between centers
    d_pair = np.min(pdist(centers))
    
    # We want to maximize the minimum of these two distances
    return -min(d_bound, d_pair)

def run_packing():
    n = 26
    # Bounds for each coordinate: all within [0, 1]
    bounds = [(0.0, 1.0)] * (n * 2)
    
    # Global optimization using Differential Evolution
    result = differential_evolution(
        packing_objective,
        bounds,
        strategy='best1bin',
        maxiter=2000,
        popsize=25,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=42,
        tol=1e-7,
        polish=True
    )
    
    centers = result.x.reshape(n, 2)
    
    # Compute final optimal radius
    min_dist = -packing_objective(result.x)
    r = min_dist / 2.0
    
    radii = np.full(n, r)
    total_sum = float(26.0 * r)
    
    return centers, radii, total_sum
