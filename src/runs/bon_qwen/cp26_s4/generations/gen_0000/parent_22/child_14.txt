import numpy as np
from scipy.optimize import linprog, differential_evolution

# Global constants
N = 26

# Precompute pair indices for the 26 circles
# We will have constraints for every pair (i, j) with i < j
pair_indices = []
for i in range(N):
    for j in range(i + 1, N):
        pair_indices.append((i, j))

num_pairs = len(pair_indices)

# Precompute the structure of the A_ub matrix for the LP solver
# Each row corresponds to a pair (i, j) and has 1 at index i and 1 at index j
# This matrix is constant for the optimization process.
A_ub = np.zeros((num_pairs, N))
for idx, (i, j) in enumerate(pair_indices):
    A_ub[idx, i] = 1.0
    A_ub[idx, j] = 1.0

# Objective coefficients for linprog: we minimize -sum(r)
c_linprog = -np.ones(N)

def solve_radii_lp(centers):
    """
    Solves the Linear Programming problem to find the maximum radii for fixed centers.
    
    Args:
        centers: np.array of shape (N, 2)
        
    Returns:
        sum_radii: float
        radii: np.array of shape (N,)
    """
    # 1. Compute pairwise distances (b_ub vector)
    dists = np.zeros(num_pairs)
    # Using a loop for clarity and sufficient speed for N=26
    cx, cy = centers[:, 0], centers[:, 1]
    for idx, (i, j) in enumerate(pair_indices):
        dists[idx] = np.sqrt((cx[i] - cx[j])**2 + (cy[i] - cy[j])**2)
    
    # 2. Compute boundary limits (upper bounds for variables)
    # r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    # This is derived from x - r >= 0, x + r <= 1, etc.
    limits = np.minimum(np.minimum(cx, 1.0 - cx), np.minimum(cy, 1.0 - cy))
    
    # 3. Define bounds for LP variables (0 <= r_i <= limit_i)
    bounds = [(0.0, limits[i]) for i in range(N)]
    
    # 4. Solve LP: Minimize -sum(r) s.t. A_ub r <= dists and bounds
    try:
        res = linprog(c_linprog, A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, -res.x
        else:
            # Fallback if LP fails (should be rare with valid bounds)
            return 0.0, np.zeros(N)
    except Exception:
        return 0.0, np.zeros(N)

def objective(centers_flat):
    """
    Objective function for the center optimization.
    Minimizes the negative sum of radii.
    """
    centers = centers_flat.reshape((N, 2))
    # Ensure centers are strictly within bounds to avoid numerical issues in LP bounds
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_r, radii = solve_radii_lp(centers)
    return -sum_r

def generate_hex_start():
    """Generates a hexagonal grid arrangement for 26 circles."""
    centers = np.zeros((N, 2))
    # We want to fit 26 circles. A 5x5 grid is 25. 
    # A hexagonal arrangement allows tighter packing.
    # Try rows with 5 and 6 circles.
    # 6, 5, 5, 5, 5 = 26.
    
    # Approximate spacing for N=26 equal circles is around 0.1 radius.
    # Let's assume a radius of roughly 0.105.
    # Horizontal spacing 2r ~ 0.21, Vertical spacing sqrt(3)r ~ 0.182.
    # We will just place them on a grid and let the optimizer move them.
    
    row_sizes = [6, 5, 5, 5, 5]
    y_spacing = 0.18
    x_spacing = 0.21
    
    idx = 0
    for row_idx, count in enumerate(row_sizes):
        y = 0.12 + row_idx * y_spacing
        # Shift alternate rows for hex pattern
        x_offset = 0.105 if row_idx % 2 == 1 else 0.10
        
        # Calculate total width of this row to center it
        total_width = (count - 1) * x_spacing
        start_x = (1.0 - total_width) / 2
        
        for col in range(count):
            x = start_x + col * x_spacing
            centers[idx] = [x, y]
            idx += 1
            
    return centers

def run_packing():
    """
    Optimizes the positions of 26 circles in a unit square to maximize the sum of radii.
    """
    # 1. Generate a smart initial guess (Hexagonal Grid)
    start_centers = generate_hex_start()
    # Flatten for optimizer
    x0 = start_centers.flatten()
    
    # 2. Define bounds for differential evolution (centers must be in [0, 1])
    bounds = [(0.0, 1.0)] * (2 * N)
    
    # 3. Run Differential Evolution
    # We use a moderate population size to balance exploration and speed.
    # 52 variables is high dimension, so we rely on the good start.
    # popsize=20, maxiter=100 should find a very good local/global optimum.
    result = differential_evolution(
        objective, 
        bounds, 
        popsize=25, 
        maxiter=150, 
        tol=1e-8, 
        mutation=(0.5, 1.2), 
        recombination=0.9,
        seed=42, # Fixed seed for reproducibility
        initial_guess=None # Let DE generate population, but it's robust enough
    )
    
    # 4. Retrieve the best centers
    best_centers = result.x.reshape((N, 2))
    
    # 5. Compute final radii for the best centers
    sum_radii, radii = solve_radii_lp(best_centers)
    
    return best_centers, radii, sum_radii
