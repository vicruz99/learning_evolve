import numpy as np
from scipy.optimize import minimize

def compute_loss(centers, radii):
    """
    Computes the objective function value: -sum(radii) + penalty.
    Penalty terms enforce boundary and non-overlap constraints.
    """
    n = centers.shape[0]
    r_clamped = np.maximum(radii, 0)
    pen = 0.0
    
    # Boundary violations: circles must be inside [0,1]x[0,1]
    for i in range(n):
        x, y = centers[i]
        r = r_clamped[i]
        if x - r < 0: 
            pen += (x - r)**2
        if x + r > 1: 
            pen += (x + r - 1)**2
        if y - r < 0: 
            pen += (y - r)**2
        if y + r > 1: 
            pen += (y + r - 1)**2
            
    # Overlap violations: distance >= r1 + r2
    # Using squared distance and squared radii sum to avoid sqrt and maintain smoothness
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            dist_sq = dx*dx + dy*dy
            r_sum = r_clamped[i] + r_clamped[j]
            r_sum_sq = r_sum * r_sum
            
            # If circles overlap, dist_sq < r_sum_sq
            if r_sum_sq > dist_sq:
                pen += (r_sum_sq - dist_sq)**2
                
    # We want to maximize sum(r), so we minimize -sum(r)
    # The penalty coefficient 1e6 ensures constraints are prioritized
    return -np.sum(r_clamped) + 1e6 * pen

def get_initialization(n, init_type):
    """
    Generates initial variables for the optimizer.
    Types: 'hex' (hexagonal lattice) or 'random'.
    """
    vars = np.zeros(3*n)
    if init_type == 'hex':

        r_init = 0.095
        coords = []

        row = 0

        while len(coords) < n and y + r_init <= 1.0 + 1e-9:
            x = r_init

                x += r_init

                x += 2 * r_init

            row += 1

            vars[2*i] = x
            vars[2*i+1] = y

            vars[2*i] = 0.5
            vars[2*i+1] = 0.5
            vars[2*n+i] = 0.05

        for i in range(n):
            vars[2*i] = np.random.rand()
            vars[2*i+1] = np.random.rand()
            vars[2*n+i] = 0.05
    return vars

def objective_func(vars, n):
    """
    Wrapper for compute_loss to match scipy.optimize signature.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    return compute_loss(centers, radii)

def run_packing():
    n = 26
    # Bounds for x, y in [0, 1] and r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n

    best_loss = np.inf

    init_vars = get_initialization(n, 'hex')

        best_loss = res.fun
        best_vars = res.x

            best_vars = res.x
