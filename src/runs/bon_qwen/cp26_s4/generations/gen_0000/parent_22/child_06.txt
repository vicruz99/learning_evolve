import numpy as np
from scipy.optimize import minimize


def objective(params, n):
    radii = params[2 * n:]
    return -np.sum(radii)


def boundary_x_min(params, i, n):
    return params[2 * i] - params[2 * n + i]


def boundary_x_max(params, i, n):
    return 1.0 - params[2 * i] - params[2 * n + i]


def boundary_y_min(params, i, n):
    return params[2 * i + 1] - params[2 * n + i]


def boundary_y_max(params, i, n):
    return 1.0 - params[2 * i + 1] - params[2 * n + i]


def non_overlap(params, i, j, n):
    dx = params[2 * i] - params[2 * j]
    dy = params[2 * i + 1] - params[2 * j + 1]
    dist_sq = dx * dx + dy * dy
    if dist_sq < 1e-20:
        return 1e-10 - params[2 * n + i] - params[2 * n + j]
    dist = np.sqrt(dist_sq)
    return dist - params[2 * n + i] - params[2 * n + j]


def radius_nn(params, i, n):
    return params[2 * n + i]


def create_hexagonal_grid(n, row_configs, spacing, x_start, y_start):
    centers = np.zeros((n, 2))
    idx = 0
    for row_idx, (count, offset) in enumerate(row_configs):
        y = y_start + row_idx * spacing * np.sqrt(3.0) / 2.0
        for col_idx in range(count):
            x = x_start + col_idx * spacing
            if offset:
                x += spacing / 2.0
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    return centers


def optimize_from_initial(centers_init, n, r_init, max_iter=3000):
    params0 = np.concatenate([centers_init.flatten(), np.full(n, r_init)])

    constraints = []
    for i in range(n):
        constraints.append({"type": "ineq", "fun": boundary_x_min, "args": (i, n)})
        constraints.append({"type": "ineq", "fun": boundary_x_max, "args": (i, n)})
        constraints.append({"type": "ineq", "fun": boundary_y_min, "args": (i, n)})
        constraints.append({"type": "ineq", "fun": boundary_y_max, "args": (i, n)})
        constraints.append({"type": "ineq", "fun": radius_nn, "args": (i, n)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({"type": "ineq", "fun": non_overlap, "args": (i, j, n)})

    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    try:
        result = minimize(
            objective,
            params0,
            args=(n,),
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": max_iter, "ftol": 1e-15},
        )
        return result
    except Exception:
        return None


def refine_with_forces(centers, radii, n, num_iters=50000):
    lr = 0.0005
    for iteration in range(num_iters):
        forces = np.zeros_like(centers)

        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq)
                if dist < 1e-12:
                    dist = 1e-12
                    diff = np.random.randn(2) * 0.001
                required = radii[i] + radii[j]
                if dist < required:
                    force_mag = (required - dist) * 500
                    forces[i] += force_mag * diff / dist
                    forces[j] -= force_mag * diff / dist

        for i in range(n):
            for dim in range(2):
                if centers[i, dim] < radii[i] + 1e-12:
                    forces[i, dim] += (radii[i] - centers[i, dim]) * 1000
                if centers[i, dim] > 1.0 - radii[i] - 1e-12:
                    forces[i, dim] -= (centers[i, dim] - (1.0 - radii[i])) * 1000

        centers += forces * lr

        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))

        if iteration % 200 == 0:
            growth = 0.00002
            radii += growth

    return centers, radii


def run_packing():
    n = 26

    best_sum = 0.0
    best_centers = None
    best_radii = None

    row_config_options = [
        [(5, False), (5, True), (5, False), (5, True), (6, False)],
        [(6, False), (5, True), (5, False), (5, True), (5, False)],
        [(5, False), (6, True), (5, False), (5, True), (5, False)],
        [(5, True), (5, False), (5, True), (5, False), (6, True)],
    ]

    for config_idx, row_configs in enumerate(row_config_options):
        for trial in range(5):
            r_init = 0.06
            centers_init = create_hexagonal_grid(n, row_configs, 0.18, 0.1, 0.1)

            if trial > 0:
                noise = np.random.RandomState(config_idx * 5 + trial).randn(n, 2) * 0.015
                centers_init = centers_init + noise
                centers_init = np.clip(centers_init, r_init, 1.0 - r_init)

            result = optimize_from_initial(centers_init, n, r_init, max_iter=4000)

            if result is not None and result.fun < 0:
                current_sum = -result.fun
                current_centers = result.x[: 2 * n].reshape(n, 2)
                current_radii = result.x[2 * n:]

                current_centers, current_radii = refine_with_forces(
                    current_centers, current_radii, n, num_iters=30000
                )

                new_sum = np.sum(current_radii)
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()

    if best_centers is None:
        row_configs = [(5, False), (5, True), (5, False), (5, True), (6, False)]
        best_centers = create_hexagonal_grid(n, row_configs, 0.18, 0.1, 0.1)
        best_radii = np.full(n, 0.06)
        best_sum = np.sum(best_radii)

    for i in range(n):
        best_centers[i, 0] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 0]))
        best_centers[i, 1] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 1]))

    return best_centers, best_radii, np.sum(best_radii)
