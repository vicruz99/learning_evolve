import numpy as np
from scipy.optimize import minimize

N = 26

def objective_func(x):
    """Maximize sum of radii by minimizing negative sum."""
    return -np.sum(x[2 * N:])

def constraint_func(x):
    """Return all constraint values (must be >= 0 for feasibility)."""
    centers = x[:2 * N].reshape(N, 2)
    radii = x[2 * N:]
    
    c = []
    
    # Boundary constraints: circle inside unit square
    for i in range(N):
        c.append(centers[i, 0] - radii[i] + 1e-8)
        c.append(1.0 - centers[i, 0] - radii[i] + 1e-8)
        c.append(centers[i, 1] - radii[i] + 1e-8)
        c.append(1.0 - centers[i, 1] - radii[i] + 1e-8)
    
    # Non-overlap constraints
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
            c.append(dist - radii[i] - radii[j] + 1e-8)
    
    # Non-negative radii
    for i in range(N):
        c.append(radii[i] + 1e-8)
    
    return np.array(c)

def make_hex_init(rows_config, x_sp, y_sp, x_even, x_odd, y_start, noise=0.0):
    """Create hexagonal initial configuration."""
    centers = np.zeros((N, 2))
    idx = 0
    for row_idx, ncols in enumerate(rows_config):
        x_start = x_even if row_idx % 2 == 0 else x_odd
        for col in range(ncols):
            if idx >= N:
                break
            cx = x_start + col * x_sp
            cy = y_start + row_idx * y_sp
            if noise > 0:
                cx += (np.random.rand() - 0.5) * noise
                cy += (np.random.rand() - 0.5) * noise
            centers[idx, 0] = cx
            centers[idx, 1] = cy
            idx += 1
    return centers

def equal_circle_objective(x):
    """Objective for equal-circle optimization."""
    return -x[2 * N]

def equal_circle_constraint(x):
    """Constraints for equal-circle optimization."""
    centers = x[:2 * N].reshape(N, 2)
    r = x[2 * N]
    
    c = []
    for i in range(N):
        c.append(centers[i, 0] - r + 1e-8)
        c.append(1.0 - centers[i, 0] - r + 1e-8)
        c.append(centers[i, 1] - r + 1e-8)
        c.append(1.0 - centers[i, 1] - r + 1e-8)
    
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
            c.append(dist - 2.0 * r + 1e-8)
    
    return np.array(c)

def equal_to_unequal_init(equal_centers, equal_r):
    """Convert equal-circle solution to unequal-circle initial guess."""
    centers = equal_centers.copy()
    radii = np.full(N, equal_r * 0.95)
    return centers, radii

def penalty_objective(x, penalty_weight):
    """Objective with penalty for constraint violations."""
    centers = x[:2 * N].reshape(N, 2)
    radii = x[2 * N:]
    
    penalty = 0.0
    
    for i in range(N):
        for d in range(2):
            if centers[i, d] < radii[i]:
                penalty += penalty_weight * (radii[i] - centers[i, d]) ** 2
            if centers[i, d] > 1.0 - radii[i]:
                penalty += penalty_weight * (centers[i, d] - (1.0 - radii[i])) ** 2
    
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j]:
                penalty += penalty_weight * (radii[i] + radii[j] - dist) ** 2
    
    return -np.sum(radii) + penalty

def optimize_with_slsqp(x0, maxiter=15000):
    """Run SLSQP optimization."""
    cons = {'type': 'ineq', 'fun': constraint_func}
    bounds = [(0, 1)] * (3 * N)
    
    result = minimize(objective_func, x0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
    return result

def refine_with_penalty(x0, num_stages=15):
    """Refine solution using penalty method."""
    x = x0.copy()
    penalty_weight = 50.0
    
    for stage in range(num_stages):
        local_penalty = penalty_weight
        
        def local_obj(x):
            return penalty_objective(x, local_penalty)
        
        bounds = [(0, 1)] * (3 * N)
        result = minimize(local_obj, x, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-16})
        x = result.x
        penalty_weight *= 1.3
    
    return x

def post_process(centers, radii):
    """Ensure solution is valid."""
    centers = centers.copy()
    radii = radii.copy()
    
    for i in range(N):
        radii[i] = max(radii[i], 1e-10)
        centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-12, 1.0 - radii[i] - 1e-12)
        centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-12, 1.0 - radii[i] - 1e-12)
    
    return centers, radii

            if result.success or -result.fun > 0.08:

                centers_eq = result.x[:2 * N].reshape(N, 2)

                result2 = optimize_with_slsqp(x0_uneq, maxiter=12000)

                # Phase 3: Penalty refinement

                    best_radii = radii_opt.copy()

        centers_pert = best_centers.copy()
