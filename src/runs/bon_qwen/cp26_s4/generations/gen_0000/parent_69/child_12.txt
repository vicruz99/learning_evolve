import numpy as np
from scipy.optimize import minimize


def compute_objective(params, n, pw):
    """Objective function with smooth penalty terms.
    
    params: flattened array of [x1, y1, ..., xn, yn, r1, ..., rn]
    n: number of circles
    pw: penalty weight
    """
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 0.001)
    
    # Objective: maximize sum of radii
    obj = -np.sum(radii)
    
    # Smooth penalty function using softplus-like approximation
    # max(0, x)^2 approximated by: for small x, use quadratic; for large x, use exact
    pen = 0.0
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left: x - r >= 0
        v = r - x
        if v > 0:
            pen += v ** 2
        # Right: 1 - x - r >= 0
        v = x + r - 1
        if v > 0:
            pen += v ** 2
        # Bottom: y - r >= 0
        v = r - y
        if v > 0:
            pen += v ** 2
        # Top: 1 - y - r >= 0
        v = y + r - 1
        if v > 0:
            pen += v ** 2
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]

            v = radii[i] + radii[j] - dist
            if v > 0:
                pen += v ** 2
    
    return obj + pw * pen

def is_valid(n, centers, radii):
    """Check if packing satisfies all constraints."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            if d < radii[i] + radii[j] - 1e-8:
                return False
    
    return True
