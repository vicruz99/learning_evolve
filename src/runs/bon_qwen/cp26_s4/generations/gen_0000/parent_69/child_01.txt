def compute_penalized_objective(x, n_circles=26, penalty_weight=5000):

    x contains [r_0, ..., r_{n-1}, x_0, y_0, ..., x_{n-1}, y_{n-1}]

    total_penalty = 0.0

            required_dist = r[i] + r[j]

                total_penalty += violation * violation
