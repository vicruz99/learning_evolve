import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def obj_func(x):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2::3])

def combined_constraints(x):
    """Vectorized boundary and overlap constraints. Returns array where all values >= 0."""
    C = x.reshape(-1, 3)
    
    # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    b = np.empty(4 * N_CIRCLES)
    b[0::4] = C[:, 0] - C[:, 2]
    b[1::4] = 1.0 - C[:, 0] - C[:, 2]
    b[2::4] = C[:, 1] - C[:, 2]
    b[3::4] = 1.0 - C[:, 1] - C[:, 2]

    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = C[:, 0][:, None] - C[:, 0][None, :]
    dy = C[:, 1][:, None] - C[:, 1][None, :]
    dr = C[:, 2][:, None] + C[:, 2][None, :]
    val = dx**2 + dy**2 - dr**2
    
    # Extract upper triangle (i < j)
    mask = np.triu(np.ones((N_CIRCLES, N_CIRCLES), dtype=bool), k=1)
    o = val[mask]

    return np.concatenate([b, o])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    best_sum = -1.0
    best_c = None
    best_r = None

    for trial in range(15):
        centers = np.zeros((n, 2))
        radii = np.zeros(n)

        # Hex-like staggered grid initialization
        row, col = 0, 0
        for i in range(n):
            shift = 0.5 if row % 2 == 1 else 0.0
            cx = 0.12 + 0.19 * (col + shift)
            cy = 0.12 + 0.16 * row
            centers[i] = [cx, cy]
            col += 1
            if (row % 2 == 0 and col >= 5) or (row % 2 == 1 and col >= 4):
                col = 0
                row += 1

        # Perturb to break symmetry
        np.random.seed(trial * 17 + 5)
        centers += np.random.uniform(-0.02, 0.02, centers.shape)
        centers = np.clip(centers, 0.08, 0.92)

        # Ensure strictly feasible initial radii
        for i in range(n):
            cx, cy = centers[i]
            max_r = min(cx, 1.0 - cx, cy, 1.0 - cy)
            radii[i] = max_r * 0.80

        # Flatten to optimization variable vector [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.zeros(n * 3)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]

        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': combined_constraints}

        # Optimize
        res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 1000, 'ftol': 1e-9})

        # Extract results
        c_new = np.zeros((n, 2))
        r_new = np.zeros(n)
        for i in range(n):
            c_new[i, 0] = res.x[3*i]
            c_new[i, 1] = res.x[3*i+1]
            r_new[i] = res.x[3*i+2]

        # Strict validation
        valid = True
        for i in range(n):
            if (c_new[i,0] < r_new[i] - 1e-8 or c_new[i,0] > 1.0 - r_new[i] + 1e-8 or 
                c_new[i,1] < r_new[i] - 1e-8 or c_new[i,1] > 1.0 - r_new[i] + 1e-8):
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i+1, n):
                    d = np.hypot(c_new[i,0]-c_new[j,0], c_new[i,1]-c_new[j,1])
                    if d < r_new[i] + r_new[j] - 1e-8:
                        valid = False
                        break
                if not valid:
                    break

        if valid:
            s = np.sum(r_new)
            if s > best_sum:
                best_sum = s
                best_c = c_new.copy()
                best_r = r_new.copy()

    return best_c, best_r, best_sum
