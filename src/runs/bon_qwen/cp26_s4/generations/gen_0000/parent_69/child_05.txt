import numpy as np
from scipy.optimize import minimize

def run_packing():

    Finds an optimal packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    # Helper to check validity
    def is_valid(centers, radii):
        # Check bounds
        if np.any(radii < 0):
            return False
        if np.any(centers < radii) or np.any(centers > 1 - radii):
            return False
        # Check overlaps
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-6:
                    return False
        return True

    # Objective function: maximize sum of radii -> minimize negative sum
    def objective(vars):
        centers = vars[:2 * n_circles].reshape(-1, 2)
        radii = vars[2 * n_circles:]
        return -np.sum(radii)

    # Constraints
    # 1. Boundary constraints: x_i >= r_i, 1 - x_i >= r_i, y_i >= r_i, 1 - y_i >= r_i
    # 2. Non-overlap: dist(i, j) >= r_i + r_j

    def get_constraints():
        cons = []
        # Boundary constraints
        # x_i - r_i >= 0
        for i in range(n_circles):
            # x index: i, y index: i + n_circles, r index: 2*n_circles + i

            # Actually, standard is usually [x0, x1..., y0, y1..., r0, r1...] or interleaved.
            # Let's stick to the reshape logic in objective.
            # vars[:52] are centers. vars[0::2] are x, vars[1::2] are y.
            # vars[52:] are radii.
            
            idx_x = 2 * i
            idx_y = 2 * i + 1
            idx_r = 2 * n_circles + i
            
            # x >= r  => x - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, ix=idx_x, ir=idx_r: v[ix] - v[ir]
            })
            # 1 - x >= r => 1 - x - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, ix=idx_x, ir=idx_r: 1.0 - v[ix] - v[ir]
            })
            # y >= r
            cons.append({
                'type': 'ineq',
                'fun': lambda v, iy=idx_y, ir=idx_r: v[iy] - v[ir]
            })

            cons.append({
                'type': 'ineq',
                'fun': lambda v, iy=idx_y, ir=idx_r: 1.0 - v[iy] - v[ir]
            })

                idx_xj, idx_yj = 2 * j, 2 * j + 1
                idx_ri, idx_rj = 2 * n_circles + i, 2 * n_circles + j
                
                # dist^2 >= (r_i + r_j)^2  => (x_i-x_j)^2 + ... - (r_i+r_j)^2 >= 0

                    xi, yi = v[2*i], v[2*i+1]
                    xj, yj = v[2*j], v[2*j+1]
                    ri, rj = v[2*n_circles+i], v[2*n_circles+j]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r = ri + rj
                    return dist_sq - sum_r**2

                cons.append({
                    'type': 'ineq',
                    'fun': overlap_con
                })
        return cons

        pts = []

                x += 0.2

            row += 1
        return np.array(pts[:n])

    def generate_square_grid(n):

        pts = []
