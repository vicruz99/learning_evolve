import numpy as np
from scipy.optimize import minimize

def objective(params):
    radii = params[52:]
    return -np.sum(radii)

def objective_jac(params):
    jac = np.zeros_like(params)
    jac[52:] = -1.0
    return jac

def constraint_func(params):
    n = 26
    centers = params[:52].reshape((n, 2))
    radii = params[52:]
    
    # Boundary constraints: centers +/- radii within [0, 1]
    b1 = centers[:, 0] - radii
    b2 = 1.0 - centers[:, 0] - radii
    b3 = centers[:, 1] - radii
    b4 = 1.0 - centers[:, 1] - radii
    
    # Non-overlap constraints: dist^2 >= (r1 + r2)^2
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diffs**2, axis=2)
    rad_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    rad_sums_sq = rad_sums**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = dists_sq[mask] - rad_sums_sq[mask]
    
    return np.concatenate([b1, b2, b3, b4, overlaps])

def get_bounds():
    bounds = [(0.0, 1.0)] * 52
    bounds += [(1e-6, 0.5)] * 26
    return bounds

def generate_initial_guess(strategy, seed):
    rng = np.random.default_rng(seed)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    
    if strategy == 'grid':
        pts = np.linspace(0.15, 0.85, 5)
        idx = 0
        for x in pts:
            for y in pts:
                centers[idx] = [x, y]
                idx += 1
        centers[25] = [0.5, 0.5]
        radii[25] = 0.02
        centers += rng.uniform(-0.02, 0.02, size=centers.shape)
        centers = np.clip(centers, 0.1, 0.9)
        
    elif strategy == 'hex':
        idx = 0
        y = 0.1
        while idx < n and y <= 0.9:
            offset = 0.1 if int(y / 0.1) % 2 != 0 else 0.0
            x_start = 0.1 + offset
            x = x_start
            while idx < n and x <= 0.9:
                centers[idx] = [x, y]
                x += 0.2
                idx += 1
            y += 0.1732
        while idx < n:
            centers[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
            
    elif strategy == 'corners':
        corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        for i in range(4):
            centers[i] = corners[i]
            radii[i] = 0.1
        idx = 4
        while idx < n:
            centers[idx] = rng.uniform(0.2, 0.8, 2)
            idx += 1
            
    elif strategy == 'random':
        centers = rng.uniform(0.1, 0.9, (n, 2))
        
    params = np.concatenate([centers.flatten(), radii])
    return params

def run_packing():
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_params = None
    best_sum = -1.0
    
    strategies = ['grid', 'hex', 'corners', 'random']
    seeds = [42, 123, 456, 789, 101, 202]
    
    # Phase 1: Multiple restarts from diverse configurations
    for strat in strategies:
        for seed in seeds:
            x0 = generate_initial_guess(strat, seed)
            try:
                res = minimize(objective, x0, method='SLSQP', jac=objective_jac, 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 2000, 'ftol': 1e-10})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_params = res.x.copy()
            except Exception:
                pass
                
    # Phase 2: Local refinement via perturbation to escape shallow local minima
    if best_params is not None:
        for _ in range(10):
            pert = best_params.copy()
            rng = np.random.default_rng()
            pert[:52] += rng.uniform(-0.005, 0.005, 52)
            pert[:52] = np.clip(pert[:52], 0.05, 0.95)
            pert[52:] += rng.uniform(-0.002, 0.002, 26)
            pert[52:] = np.clip(pert[52:], 1e-6, 0.5)
            
            try:
                res = minimize(objective, pert, method='SLSQP', jac=objective_jac,
                               bounds=bounds, constraints=cons,

                pass

    radii = best_params[52:]

    radii = np.maximum(radii, 1e-7)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, float(np.sum(radii))
