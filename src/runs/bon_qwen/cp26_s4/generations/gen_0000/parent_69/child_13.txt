import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initial Configuration
    # Start with a 5x5 grid of circles with radius 0.08.
    # This is feasible since 5 * diameter (0.16) = 0.8 < 1.
    # We place the 26th circle in a gap.
    
    centers = []
    radii = []
    
    # Grid coordinates for 5x5
    grid_points = [0.1, 0.3, 0.5, 0.7, 0.9]
    r_init_grid = 0.08
    
    for x in grid_points:
        for y in grid_points:
            centers.append([x, y])
            radii.append(r_init_grid)
    
    # Place 26th circle in the gap at (0.2, 0.2)
    # Distance to nearest grid center (0.1, 0.1) is sqrt(0.02) approx 0.141.
    # Sum of radii 0.08 + 0.05 = 0.13 < 0.141, so it fits.
    centers.append([0.2, 0.2])
    radii.append(0.05)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Define Objective Function
    # We want to maximize sum of radii, so minimize negative sum.
    def objective(vars):
        r = vars[2::3]
        return -np.sum(r)
    
    # 3. Define Constraints
    constraints = []
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    
    for i in range(n):
        idx_x = i * 3
        idx_y = i * 3 + 1
        idx_r = i * 3 + 2
        
        # x >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # x <= 1 - r  =>  1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # y <= 1 - r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })

    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = i * 3
            idx_yi = i * 3 + 1
            idx_ri = i * 3 + 2
            idx_xj = j * 3
            idx_yj = j * 3 + 1
            idx_rj = j * 3 + 2
            
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: (v[idx_xi] - v[idx_xj])**2 + (v[idx_yi] - v[idx_yj])**2 - (v[idx_ri] + v[idx_rj])**2
            })

    # 4. Define Bounds
    # x, y in [0, 1], r >= 0
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (radius cannot exceed 0.5)

    # 5. Run Optimization
    # SLSQP is suitable for constrained non-linear optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-9})
        final_vars = res.x
    except Exception as e:
        # Fallback to initial if optimization fails (unlikely)
        final_vars = x0

    # 6. Extract results
    final_centers = final_vars[0::3].reshape((n, 2))
    final_radii = final_vars[2::3]
    
    # Ensure radii are non-negative (just in case)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Recalculate sum
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
