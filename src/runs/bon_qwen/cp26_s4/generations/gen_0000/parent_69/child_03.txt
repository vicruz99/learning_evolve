import numpy as np


def resolve_overlaps(centers, r, n, iterations=200):
    """Use force-based relaxation to resolve overlaps for a given radius."""
    for it in range(iterations):
        forces = np.zeros((n, 2))
        
        # Pairwise repulsion between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                
                if dist < 2 * r:
                    overlap = 2 * r - dist
                    force_mag = overlap * 50.0
                    nx = dx / dist
                    ny = dy / dist
                    forces[i, 0] -= force_mag * nx
                    forces[i, 1] -= force_mag * ny
                    forces[j, 0] += force_mag * nx
                    forces[j, 1] += force_mag * ny
        
        # Boundary repulsion
        for i in range(n):
            for axis in range(2):
                if centers[i, axis] < r:
                    forces[i, axis] += (r - centers[i, axis]) * 50.0
                if centers[i, axis] > 1.0 - r:
                    forces[i, axis] -= (centers[i, axis] - (1.0 - r)) * 50.0
        
        # Apply forces with adaptive damping
        damping = 0.05 / (1.0 + it * 0.02)
        centers = centers + forces * damping
    
    return centers


def check_valid(centers, r, n):
    """Check if configuration is valid for given radius."""
    # Check boundaries
    for i in range(n):
        for axis in range(2):
            if centers[i, axis] < r - 1e-10 or centers[i, axis] > 1.0 - r + 1e-10:
                return False
    
    # Check overlaps

            dy = centers[j, 1] - centers[i, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < 2 * r - 1e-10:
                return False
    
    return True

    num_rows = len(row_counts)

        start_x = (1.0 - row_width) / 2.0

        if row % 2 == 1:
            start_x += spacing_x / 2.0
        
        for col in range(count):
            x = start_x + col * spacing_x
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
    
    return centers
