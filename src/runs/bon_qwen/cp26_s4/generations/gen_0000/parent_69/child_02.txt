The problem asks to pack 26 circles in a unit square to maximize the sum of radii. This is a non-convex optimization problem due to the pairwise distance constraints. I will use a hybrid numerical optimization strategy:
1. **Initialization**: Start with multiple configurations (hexagonal grid and random) to escape local minima.
2. **Sequential Least Squares Programming (SLSQP)**: Use `scipy.optimize.minimize` to optimize centers and radii simultaneously. The objective is to minimize the negative sum of radii. Constraints enforce boundary conditions and non-overlap. I will provide analytical gradients for the objective to improve convergence.
3. **Linear Programming (LP) Refinement**: After SLSQP converges, the centers are fixed. I will solve a Linear Program to find the mathematically optimal radii for those fixed centers. This step ensures radii are pushed to their theoretical maximum without overlap, often yielding a better sum than the nonlinear solver alone due to tighter constraint handling.
4. **Iteration**: Repeat the process from different seeds and keep the best valid result.

import numpy as np
from scipy.optimize import minimize, linprog
import math

N_CIRCLES = 26

def objective(vars):
    radii = vars[2::3]
    return -np.sum(radii)

def gradient(vars):
    grad = np.zeros_like(vars)
    grad[2::3] = -1.0
    return grad

def constraint_fun(vars):
    n = N_CIRCLES
    vals = []
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    for i in range(n):
        idx = i * 3
        x, y, r = vars[idx], vars[idx+1], vars[idx+2]
        vals.append(x - r)
        vals.append(1.0 - x - r)
        vals.append(y - r)
        vals.append(1.0 - y - r)
        
    # Overlap constraints: dist(i,j) - (r_i + r_j) >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i, idx_j = i * 3, j * 3
            dx = vars[idx_i] - vars[idx_j]
            dy = vars[idx_i+1] - vars[idx_j+1]
            dist = math.hypot(dx, dy)
            vals.append(dist - (vars[idx_i+2] + vars[idx_j+2]))
            
    return np.array(vals)

def refine_radii_lp(centers):
    n = centers.shape[0]
    # Maximize sum(r) -> Minimize -sum(r)
    c = -np.ones(n)
    
    pairs = []
    dists = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
            dists.append(np.linalg.norm(centers[i] - centers[j]))
            
    num_con = len(pairs)
    A_ub = np.zeros((num_con, n))
    b_ub = np.array(dists)
    
    for k, (i, j) in enumerate(pairs):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        
    bounds = []
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, r_max)))
        
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.zeros(n)

def run_optimization(init_vars):
    cons = {'type': 'ineq', 'fun': constraint_fun}
    bounds = [(0.0, 1.0) if i % 3 < 2 else (0.0, None) for i in range(3 * N_CIRCLES)]
    
    try:
        res = minimize(objective, init_vars, method='SLSQP', jac=gradient, 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
        return res.x, res.success
    except Exception:
        return init_vars, False

def generate_hex_grid():
    # Hexagonal packing initialization
    init_x = np.linspace(0.05, 0.95, 6)
    init_y = np.linspace(0.05, 0.95, 5)
    pts = []
    for i, y in enumerate(init_y):
        x_offset = 0.08 * (i % 2)
        for x in init_x:
            pts.append([x + x_offset, y])
    return np.array(pts[:N_CIRCLES])

def run_packing():
    best_centers = np.zeros((N_CIRCLES, 2))
    best_radii = np.zeros(N_CIRCLES)
    best_sum = 0.0
    
    # Strategy 1: Hexagonal grid starts
    grid_pts = generate_hex_grid()
    for seed in range(5):
        np.random.seed(seed)
        init_vars = np.zeros(3 * N_CIRCLES)
        # Add slight random perturbation to grid
        init_pts = grid_pts + np.random.randn(N_CIRCLES, 2) * 0.01
        init_pts = np.clip(init_pts, 0.05, 0.95)
        
        init_vars[0::3] = init_pts[:, 0]
        init_vars[1::3] = init_pts[:, 1]
        init_vars[2::3] = 0.01
        
        res_vars, success = run_optimization(init_vars)
        
        centers = res_vars.reshape(-1, 3)[:, :2]
        radii = refine_radii_lp(centers)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    # Strategy 2: Random starts
    for seed in range(5):
        np.random.seed(seed + 100)
        init_vars = np.zeros(3 * N_CIRCLES)
        init_vars[0::3] = np.random.uniform(0.1, 0.9, N_CIRCLES)
        init_vars[1::3] = np.random.uniform(0.1, 0.9, N_CIRCLES)
        init_vars[2::3] = 0.01
        
        res_vars, success = run_optimization(init_vars)
        
        centers = res_vars.reshape(-1, 3)[:, :2]
        radii = refine_radii_lp(centers)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
