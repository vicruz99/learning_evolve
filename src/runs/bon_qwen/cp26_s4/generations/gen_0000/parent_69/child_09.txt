import numpy as np
from scipy.optimize import minimize

def get_initial_guess(n_circles):
    """
    Generates a hexagonal lattice initial guess for circle centers and radii.
    """
    # Start with a guess radius that fits roughly. 
    # For 26 circles, r ~ 0.1 is a good starting point for equal circles.
    # Hexagonal packing allows slightly denser packing.
    r_guess = 0.09 
    
    # Hexagonal packing parameters
    # Row spacing
    dy = np.sqrt(3)/2 * (2 * r_guess)
    # Col spacing
    dx = 2 * r_guess
    
    centers = []
    radii = []
    
    y = r_guess
    row = 0
    count = 0
    while count < n_circles:
        # Offset for alternating rows
        x_start = r_guess if row % 2 == 0 else r_guess + dx/2
        
        x = x_start
        while x + r_guess <= 1.0 + 1e-9:
            if count < n_circles:
                centers.append([x, y])
                radii.append(r_guess)
                count += 1
                x += dx
            else:
                break
        y += dy
        row += 1
        
    # If we didn't fit enough (unlikely with r=0.09), or fit too many, adjust
    # But with r=0.09, we should fit plenty.
    # Trim or pad if necessary (though logic above handles count)
    centers = centers[:n_circles]
    radii = radii[:n_circles]
    
    # If we have fewer than n_circles (shouldn't happen), fill with random
    while len(centers) < n_circles:
        centers.append([0.5, 0.5])
        radii.append(0.01)
        
    return np.array(centers), np.array(radii)

def objective(params, n):
    """
    Objective function: negative sum of radii (to minimize).
    params: [x0, y0, r0, x1, y1, r1, ...]
    """
    radii = params[2::3]
    return -np.sum(radii)

def constraints_func(params, n):
    """
    Returns an array of constraint values. All must be >= 0.
    Includes:
    1. Boundary constraints for each circle (4 per circle)
    2. Non-overlap constraints for each pair (1 per pair)
    """
    # Reshape for easier access
    # params shape (3*n,)
    c = params.reshape(n, 3)
    x = c[:, 0]
    y = c[:, 1]
    r = c[:, 2]
    
    cons = []
    
    # Boundary constraints: x-r >= 0, y-r >= 0, 1-x-r >= 0, 1-y-r >= 0
    # Vectorized
    cons.append(x - r)
    cons.append(y - r)
    cons.append(1.0 - x - r)
    cons.append(1.0 - y - r)
    
    # Pairwise distance constraints
    # dist_ij^2 - (ri + rj)^2 >= 0
    # Compute all pairwise squared distances efficiently
    # Centers shape (n, 2)
    centers = c[:, :2]
    
    # Compute pairwise squared Euclidean distances
    # Using broadcasting: (n, 1, 2) - (1, n, 2) -> (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Compute (ri + rj)^2
    r_sum_sq = (r[:, np.newaxis] + r[np.newaxis, :])**2
    
    # Constraint value
    pair_cons = dist_sq - r_sum_sq
    
    # We only need upper triangle (i < j)
    # Mask for upper triangle
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pair_cons_vals = pair_cons[mask]
    
    cons.append(pair_cons_vals)
    
    # Radius non-negativity (r >= 0)
    cons.append(r)
    
    # Flatten and concatenate
    return np.concatenate([c.flatten() for c in cons])

def run_packing():
    n = 26
    
    # We will run the optimizer multiple times with different seeds/initializations
    best_result = None
    best_sum = -np.inf
    
    # Try a few different initial configurations
    # 1. Hexagonal grid
    # 2. Perturbed hexagonal grid
    
    for seed in range(5):
        np.random.seed(seed)
        centers, radii = get_initial_guess(n)
        
        # Add random perturbation to centers to avoid symmetry locking
        if seed > 0:
            perturbation = (np.random.rand(n, 2) - 0.5) * 0.02
            centers += perturbation
        
        # Initial parameters vector
        params0 = np.zeros(3 * n)
        for i in range(n):
            params0[3*i] = centers[i, 0]
            params0[3*i+1] = centers[i, 1]
            params0[3*i+2] = radii[i]
            
        # Bounds: x, y in [0, 1], r in [0, 1] (loosely)
        # Actually r max is 0.5
        bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
        bounds = tuple(bounds)
        
        # Define constraints dict
        # SLSQP expects a list of constraint dicts or a single dict with fun returning array
        cons = {'type': 'ineq', 'fun': lambda p: constraints_func(p, n)}
        
        try:
            # Use SLSQP
            res = minimize(objective, params0, args=(n,), method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success or res.fun < -np.inf: # Check if valid
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_result = res.x
                    
        except Exception as e:
            print(f"Optimization failed with seed {seed}: {e}")
            continue
            
    if best_result is None:
        # Fallback to initial guess if optimization failed completely
        centers, radii = get_initial_guess(n)
        best_sum = np.sum(radii)
        best_result = np.zeros(3*n)
        for i in range(n):
            best_result[3*i] = centers[i, 0]
            best_result[3*i+1] = centers[i, 1]
            best_result[3*i+2] = radii[i]

    # Extract final solution
    centers_final = best_result[:2*n].reshape(n, 2)
    radii_final = best_result[2*n:]
    
    # Sanity check: ensure radii are valid (non-negative)
    radii_final = np.maximum(radii_final, 0.0)
    
    # Ensure centers are within [0,1]
    centers_final = np.clip(centers_final, 0.0, 1.0)
    
    return centers_final, radii_final, np.sum(radii_final)
