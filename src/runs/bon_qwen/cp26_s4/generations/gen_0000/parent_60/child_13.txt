def get_bottleneck_distance(centers):
    """
    Computes the minimum distance between any pair of centers
    and the minimum distance from any center to the boundary.
    The feasible radius for equal circles is half of this value.
    """
    n = centers.shape[0]
    # Calculate pairwise distances
    # Centers shape: (n, 2)
    # Use broadcasting for distance matrix
    # dist_matrix[i, j] = ||c_i - c_j||
    # To save memory, we can compute half the matrix or just loop, 
    # but for N=26, full matrix is fine (26x26 = 676 elements)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    # x coords are centers[:, 0], y are centers[:, 1]
    x = centers[:, 0]
    y = centers[:, 1]
    
    dist_to_left = x
    dist_to_right = 1.0 - x
    dist_to_bottom = y
    dist_to_top = 1.0 - y
    
    # Create a vector of boundary distances for each center
    # It's the minimum of the 4 distances for that center
    boundary_dists = np.minimum(np.minimum(dist_to_left, dist_to_right), 
                                np.minimum(dist_to_bottom, dist_to_top))
    
    # We want the global minimum distance that constrains the radius.
    # For equal circles, radius r must satisfy:
    # r + r <= dist_matrix[i, j]  => 2r <= dist_matrix[i, j]
    # r <= boundary_dists[i]
    # So 2r <= min(min(dist_matrix), min(boundary_dists))
    # Actually, strictly speaking, r <= boundary_dists[i] is correct.
    # And 2r <= dist_matrix[i,j].
    # So the constraint is 2r <= min( min_{i!=j} dist_matrix[i,j], min_i boundary_dists[i] ).
    
    # Extract lower triangle of distance matrix to get unique pairs
    # np.fill_diagonal with infinity to ignore self-distance
    np.fill_diagonal(dist_matrix, np.inf)
    min_pair_dist = np.min(dist_matrix)
    
    min_boundary_dist = np.min(boundary_dists)
    
    bottleneck = min(min_pair_dist, min_boundary_dist)
    return bottleneck

    return -bottleneck
