import numpy as np
from scipy.optimize import minimize

def generate_hex_layout(r0, n=26):
    """Generate an initial hexagonal packing layout for n circles."""
    centers = []
    row_counts = [6, 5, 6, 5, 4]
    for row, count in enumerate(row_counts):
        y = r0 * (1.0 + row * np.sqrt(3.0))
        x_start = r0 if row % 2 == 0 else 2.0 * r0
        for col in range(count):
            x = x_start + col * 2.0 * r0
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, r0)
    return np.concatenate([centers.flatten(), radii])

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    constraints = []
    # Boundary constraints
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]})
        
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: 
                             np.sqrt((v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2) - (v[3*i+2] + v[3*j+2])})

    def objective(v):
        return -np.sum(v[2::3])

    best_sum = -1.0
    best_x = None
    
    # Multi-start optimization to escape local minima
    for r_start in [0.075, 0.082, 0.088]:
        x0 = generate_hex_layout(r_start, n)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'ftol': 1e-10, 'maxiter': 3000, 'disp': False})
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_x = res.x
        except Exception:
            continue
            
    # Fallback if optimization fails completely
    if best_x is None:
        best_x = generate_hex_layout(0.08, n)
        
    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]
    
    # Clip to physical bounds and ensure non-negative radii
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, np.sum(radii)
