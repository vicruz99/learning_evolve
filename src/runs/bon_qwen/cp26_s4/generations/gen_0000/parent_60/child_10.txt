import numpy as np

# Constants
N_CIRCLES = 26
ITERATIONS = 2000
INITIAL_LEARNING_RATE = 0.05
PENALTY_WEIGHT = 10000.0
BOUNDARY_PENALTY_WEIGHT = 100000.0

def compute_loss_and_gradient(vars_flat):
    """
    Computes the loss (negative sum of radii + penalties) and its gradient.
    vars_flat is a 1D numpy array of shape (3 * N_CIRCLES).
    Structure: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    """
    n = N_CIRCLES
    xs = vars_flat[0::3]
    ys = vars_flat[1::3]
    rs = vars_flat[2::3]
    
    # Objective: minimize -sum(r)
    # Gradient w.r.t r is -1, w.r.t x,y is 0
    grad = np.zeros_like(vars_flat)
    grad[2::3] = -1.0
    
    loss = -np.sum(rs)
    
    # 1. Boundary Penalties
    # Circle i must be inside [0, 1] x [0, 1]
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    # Violations: r - x, r + x - 1, r - y, r + y - 1
    
    # x < r  => r - x > 0
    diff_x_left = rs - xs
    pos_diff_x_left = np.maximum(0.0, diff_x_left)
    penalty_x_left = np.sum(pos_diff_x_left**2)
    loss += BOUNDARY_PENALTY_WEIGHT * penalty_x_left
    # Gradients: d/dr (r-x)^2 = 2(r-x), d/dx (r-x)^2 = -2(r-x)
    grad[0::3] += BOUNDARY_PENALTY_WEIGHT * (-2.0 * pos_diff_x_left)
    grad[2::3] += BOUNDARY_PENALTY_WEIGHT * (2.0 * pos_diff_x_left)
    
    # x > 1-r => r + x - 1 > 0
    diff_x_right = rs + xs - 1.0
    pos_diff_x_right = np.maximum(0.0, diff_x_right)
    penalty_x_right = np.sum(pos_diff_x_right**2)
    loss += BOUNDARY_PENALTY_WEIGHT * penalty_x_right
    # Gradients: d/dr (r+x-1)^2 = 2(r+x-1), d/dx = 2(r+x-1)
    grad[0::3] += BOUNDARY_PENALTY_WEIGHT * (2.0 * pos_diff_x_right)
    grad[2::3] += BOUNDARY_PENALTY_WEIGHT * (2.0 * pos_diff_x_right)
    
    # y < r => r - y > 0
    diff_y_bot = rs - ys
    pos_diff_y_bot = np.maximum(0.0, diff_y_bot)
    penalty_y_bot = np.sum(pos_diff_y_bot**2)
    loss += BOUNDARY_PENALTY_WEIGHT * penalty_y_bot
    grad[1::3] += BOUNDARY_PENALTY_WEIGHT * (-2.0 * pos_diff_y_bot)
    grad[2::3] += BOUNDARY_PENALTY_WEIGHT * (2.0 * pos_diff_y_bot)
    
    # y > 1-r => r + y - 1 > 0
    diff_y_top = rs + ys - 1.0
    pos_diff_y_top = np.maximum(0.0, diff_y_top)
    penalty_y_top = np.sum(pos_diff_y_top**2)
    loss += BOUNDARY_PENALTY_WEIGHT * penalty_y_top
    grad[1::3] += BOUNDARY_PENALTY_WEIGHT * (2.0 * pos_diff_y_top)
    grad[2::3] += BOUNDARY_PENALTY_WEIGHT * (2.0 * pos_diff_y_top)
    
    # 2. Overlap Penalties
    # dist(i, j) >= r_i + r_j
    # Violation: r_i + r_j - dist(i, j) > 0
    # We iterate over pairs
    
    # To speed up, we can use vectorization, but for N=26, double loop is fine.
    # However, vectorization is cleaner.
    
    # Coordinates matrix (N, 2)
    coords = np.column_stack((xs, ys))
    
    # Compute pairwise distances
    # diff shape (N, N, 2)
    diff = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-9) # add epsilon to avoid div by zero
    
    # Lower triangular part of distance matrix (excluding diagonal)
    # r_i + r_j shape (N, N)
    sum_radii = rs[:, np.newaxis] + rs[np.newaxis, :]
    
    # Violation matrix
    # violation = sum_radii - dists
    # Only care where violation > 0
    violations = sum_radii - dists
    
    # Create a mask for upper triangle to count each pair once
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    # Extract violations for pairs
    pair_violations = violations[mask]
    
    # Only penalize positive violations
    pos_violations = np.maximum(0.0, pair_violations)
    
    penalty_overlap = np.sum(pos_violations**2)
    loss += PENALTY_WEIGHT * penalty_overlap
    
    # Compute gradients for overlaps
    # The term is (r_i + r_j - dist_ij)^2
    # Let v_ij = r_i + r_j - dist_ij
    # d/dr_i (v_ij^2) = 2 * v_ij * (1 - (-1/dist_ij) * (x_i - x_j)? No)
    # dist_ij = sqrt((x_i-x_j)^2 + ...)
    # d(dist)/dr = 0
    # d(v)/dr_i = 1
    # So d(v^2)/dr_i = 2 * v_ij
    
    # d(v)/dx_i = - d(dist)/dx_i = - (x_i - x_j) / dist_ij
    # d(v^2)/dx_i = 2 * v_ij * (-(x_i - x_j)/dist_ij)
    
    # We need to distribute gradients back to the variables.
    # Since we used a mask, it's easier to loop or use broadcasting carefully.
    # Given N=26, explicit loop for gradient accumulation is safe and clear.
    
    for i in range(n):
        for j in range(i + 1, n):
            v = sum_radii[i, j] - dists[i, j]
            if v > 0:
                # Gradient contribution
                # Factor 2 * PENALTY_WEIGHT * v
                factor = 2.0 * PENALTY_WEIGHT * v
                
                # d/dr_i and d/dr_j
                grad[i*3 + 2] += factor
                grad[j*3 + 2] += factor
                
                # d/dx_i, d/dy_i, d/dx_j, d/dy_j
                # dx_i term: -factor * (x_i - x_j) / dist
                # dx_j term: -factor * (x_j - x_i) / dist = factor * (x_i - x_j) / dist
                
                inv_dist = 1.0 / dists[i, j]
                
                dx = (xs[i] - xs[j]) * inv_dist
                dy = (ys[i] - ys[j]) * inv_dist
                
                grad[i*3] -= factor * dx
                grad[i*3 + 1] -= factor * dy
                grad[j*3] += factor * dx
                grad[j*3 + 1] += factor * dy
                
    return loss, grad

def run_packing():
    n = N_CIRCLES
    
    # Initialization: Hexagonal-like grid
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.06 # Start small to avoid initial huge penalties
    
    # Place circles in a staggered pattern
    # Try to fit 26 circles. 
    # 5 rows of 5 is 25. We need 1 more.
    # Let's try 6 rows: 4, 5, 4, 5, 4, 4 -> sum 26.
    row_counts = [4, 5, 4, 5, 4, 4]
    idx = 0
    
    # Spacing
    dy = 0.17 # vertical spacing
    y = 0.1
    
    for count in row_counts:
        if count == 5:
            xs = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
        else:
            xs = np.array([0.2, 0.4, 0.6, 0.8])
            
        for x in xs:
            centers[idx] = [x, y]
            idx += 1
        y += dy
        
    # Ensure we have 26 circles (safety check)
    # The logic above produces 26.
    
    # Flatten to vars: x1, y1, r1, ...
    vars_flat = np.zeros(3 * n)
    vars_flat[0::3] = centers[:, 0]
    vars_flat[1::3] = centers[:, 1]
    vars_flat[2::3] = radii
    
    # Optimization loop (Gradient Descent with adaptive step size)
    lr = INITIAL_LEARNING_RATE
    
    # We can try to decrease learning rate over time
    for it in range(ITERATIONS):
        loss, grad = compute_loss_and_gradient(vars_flat)
        
        # Update
        # vars_flat -= lr * grad
        # But we need to project to bounds? 
        # The penalties handle bounds, but explicit clamping helps.
        
        vars_flat = vars_flat - lr * grad
        
        # Clamp bounds
        vars_flat[0::3] = np.clip(vars_flat[0::3], 0.0, 1.0)
        vars_flat[1::3] = np.clip(vars_flat[1::3], 0.0, 1.0)
        vars_flat[2::3] = np.clip(vars_flat[2::3], 0.0, 0.5)
        
        # Adaptive learning rate decay
        if it > 500:
            lr *= 0.995
            
    # Extract final results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = vars_flat[i*3]
        final_centers[i, 1] = vars_flat[i*3 + 1]
        final_radii[i] = vars_flat[i*3 + 2]
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
