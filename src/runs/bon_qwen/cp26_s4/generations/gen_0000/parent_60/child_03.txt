1. Boundary constraints: $0 \le x_i - r_i$ and $x_i + r_i \le 1$ (similarly for $y$)
2. Non-overlap constraints: $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$ for all $i < j$

This is a non-convex problem, so we use `scipy.optimize.minimize` with the SLSQP algorithm, which handles non-linear constraints efficiently. We vectorize the constraint evaluations for performance and run multiple random restarts to escape local minima. The constraints are formulated as `fun(v) >= 0` for SLSQP compatibility.

- Flattened variable vector: `[x_0, y_0, r_0, x_1, y_1, r_1, ...]`

import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N_CIRCLES = 26

def objective(v):
    """Negative sum of radii (to be minimized)"""
    return -np.sum(v[2::3])

def constraints_func(v):
    """Returns array of constraint values, all must be >= 0"""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    n = len(r)
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    idx = np.triu_indices(n, k=1)
    c.append(dx[idx]**2 + dy[idx]**2 - dr[idx]**2)
    
    return np.concatenate(c)

def run_packing():
    n = N_CIRCLES
    best_sum = -1.0
    best_v = None
    
    nl_con = NonlinearConstraint(constraints_func, 0, np.inf)
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    restarts = 30
    for i in range(restarts):
        np.random.seed(i + 100)
        # Initialize centers away from boundaries to give optimizer room
        pts = np.random.uniform(0.1, 0.9, size=(n, 2))
        r_init = np.full(n, 0.04)
        
        v0 = np.zeros(3 * n)
        v0[0::3] = pts[:, 0]
        v0[1::3] = pts[:, 1]
        v0[2::3] = r_init
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=nl_con,
                           options={'maxiter': 1500, 'ftol': 1e-9, 'disp': False})
            if res.fun < -best_sum:
                best_sum = -res.fun
                best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        # Fallback (should not be reached)
        centers = np.column_stack((np.random.rand(n, 2) * 0.8 + 0.1,))
        radii = np.full(n, 0.01)
        return centers, radii, 0.26
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    return centers, radii, float(np.sum(radii))
