import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal initial guess and optimizes the radius.
    """
    n = 26
    
    # 1. Generate Initial Centers using a Hexagonal Grid pattern
    # This is a dense packing pattern and provides a good starting point.
    centers = np.zeros((n, 2))
    
    # Estimate a starting radius for grid generation. 
    # 26 circles in a square. Hexagonal packing density ~0.9069.
    # Area = n * pi * r^2 approx 0.9 * 1. 
    # r approx sqrt(0.9 / (26 * pi)) approx 0.105.
    # We use a slightly smaller r to ensure they fit in the grid generation.
    r_init = 0.09
    
    # Generate points
    idx = 0
    # Hexagonal rows
    # Vertical spacing: sqrt(3)/2 * 2*r = sqrt(3)*r
    # Horizontal spacing: 2*r
    # Rows alternate shift
    
    row_idx = 0
    while idx < n:
        y = r_init + row_idx * (np.sqrt(3) * r_init)
        
        # Determine x starting position based on row parity
        # Even rows start at r_init, odd rows shifted by r_init (so start at 2*r_init)
        if row_idx % 2 == 0:
            x_start = r_init
            max_x = 1.0 - r_init
        else:
            x_start = 2 * r_init
            max_x = 1.0 - r_init
            
        x = x_start
        while x <= max_x + 1e-9 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r_init
        row_idx += 1
        
    # Trim or adjust if we generated too many (though loop condition handles n)
    # The loop above might stop exactly or we might need to ensure we have exactly n.
    # If the grid logic produces fewer than n (unlikely with r=0.09), we fill with random.
    if idx < n:
        # Fallback: random placement for remaining
        for k in range(idx, n):
            centers[k] = np.random.uniform(0.1, 0.9, 2)

    # 2. Optimization Strategy:
    # We want to maximize sum of radii. 
    # We will assume equal radii for simplicity and robustness first.
    # We perform a binary search for the maximum radius R such that 
    # 26 circles of radius R fit in the square.
    
    # Function to check feasibility of radius R
    def check_feasibility(R):
        # Objective: Minimize total overlap penalty
        # Penalty = sum over pairs of max(0, 2R - dist)^2 
        #         + sum over boundaries of max(0, R - margin)^2
        
        # We minimize this penalty. If min penalty ~ 0, R is feasible.
        
        def objective(vars):
            c = vars.reshape((n, 2))
            penalty = 0.0
            
            # Boundary penalties
            # For each circle, must be within [R, 1-R]
            for i in range(n):
                x, y = c[i]
                # Left
                if x < R:
                    penalty += (R - x)**2
                # Right
                if x > 1.0 - R:
                    penalty += (x - (1.0 - R))**2
                # Bottom
                if y < R:
                    penalty += (R - y)**2
                # Top
                if y > 1.0 - R:
                    penalty += (y - (1.0 - R))**2
            
            # Overlap penalties
            # We only need to check pairs, but O(N^2) is fine for N=26
            for i in range(n):
                for j in range(i + 1, n):
                    dx = c[i, 0] - c[j, 0]
                    dy = c[i, 1] - c[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    min_dist = 2.0 * R
                    if dist < min_dist:
                        penalty += (min_dist - dist)**2
                        
            return penalty

        # Initial guess for optimization is the current 'centers'
        # But 'centers' might be scaled for R_init. 
        # For a new R, we should scale centers or just use the shape.
        # Actually, the shape (relative positions) is what matters.
        # We can scale the initial centers to fit the new R roughly?
        # Or just use the hex grid logic again with new R?

        # This helps the optimizer converge faster.

        idx_g = 0
        row_idx_g = 0
        r_grid = R

            y = r_grid + row_idx_g * (np.sqrt(3) * r_grid)
            if y > 1.0 - r_grid + 1e-6:
                # If row is out of bounds vertically, break (shouldn't happen if R is feasible)
                # But we still need to fill guess_centers
                break
                
            if row_idx_g % 2 == 0:
                x_start = r_grid
            else:
                x_start = 2 * r_grid
                
            x = x_start

                guess_centers[idx_g] = [x, y]
                idx_g += 1
                x += 2 * r_grid
            row_idx_g += 1
            
        # Fill any remaining if grid didn't produce enough (rare for feasible R)
        if idx_g < n:
            for k in range(idx_g, n):
                guess_centers[k] = centers[k] # Fallback to previous

        # Run optimization
        # Use multiple restarts to avoid local minima
        best_penalty = np.inf
        
        # We try a few perturbations of the grid
        for _ in range(3):
            # Small random jitter
            perturbed_centers = guess_centers + np.random.uniform(-0.005, 0.005, (n, 2))
            # Clamp to [0,1]

            res = minimize(objective, perturbed_centers.flatten(), method='BFGS', options={'maxiter': 200})
            if res.fun < best_penalty:
                best_penalty = res.fun
                best_centers = res.x.reshape((n, 2))
                
        # Threshold for feasibility
        if best_penalty < 1e-5:
            return True, best_centers
        else:
            return False, best_centers

    # Binary Search
    low = 0.05
    high = 0.12
    best_r = low
    best_centers_result = centers
    
    # 10 iterations of binary search should be enough for precision

        mid = (low + high) / 2
        feasible, centers_res = check_feasibility(mid)
        
        if feasible:
            best_r = mid
            best_centers_result = centers_res
            low = mid
        else:
            high = mid

    # Final radii
    radii = np.full(n, best_r)
    sum_radii = np.sum(radii)
    
    # Validation check just in case
    # (We assume check_feasibility worked)
    
    return best_centers_result, radii, sum_radii
