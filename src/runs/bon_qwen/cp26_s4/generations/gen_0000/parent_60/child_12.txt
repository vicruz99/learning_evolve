from scipy.optimize import minimize

def compute_constraints(vars):

    Computes all boundary and pairwise non-overlap constraints.
    Returns an array of constraint values. Values >= 0 indicate satisfaction.
    """
    n = 26
    num_constraints = 4 * n + n * (n - 1) // 2
    vals = np.empty(num_constraints)
    idx = 0

    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        vals[idx] = xi - ri
        vals[idx+1] = 1.0 - xi - ri
        vals[idx+2] = yi - ri
        vals[idx+3] = 1.0 - yi - ri
        idx += 4
        
    # Pairwise non-overlap constraints
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            radii_sum_sq = (ri + rj)**2
            vals[idx] = dist_sq - radii_sum_sq
            idx += 1
            
    return vals

def objective(vars):
    """
    Objective function: minimize negative sum of radii.
    """
    return -np.sum(vars[2::3])

    dy = r_init * np.sqrt(3)
