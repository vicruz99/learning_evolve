import numpy as np
from scipy.optimize import minimize

N = 26

def eval_constraints(v):
    """Evaluate all inequality constraints: g(v) >= 0"""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    boundary = np.hstack([c1, c2, c3, c4])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum_sq = (r[:, np.newaxis] + r[np.newaxis, :])**2
    diff = dist_sq - r_sum_sq
    
    # Upper triangular indices (excluding diagonal)
    upper = np.triu_indices(N, k=1)
    overlap = diff[upper]
    
    return np.hstack([boundary, overlap])

def objective(v):
    """Objective: minimize negative sum of radii (i.e., maximize sum of radii)"""
    return -np.sum(v[2::3])

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Prepare deterministic initial configurations
    inits = []
    
    # 1. Centered Grid
    v0 = np.zeros(N * 3)
    for i in range(N):
        col = i % 5
        row = i // 5
        v0[3*i] = 0.2 + col * 0.15
        v0[3*i+1] = 0.2 + row * 0.15
        v0[3*i+2] = 0.04
    inits.append(v0)
    
    # 2. Hexagonal-like arrangement
    v1 = np.zeros(N * 3)
    idx = 0
    for row in range(5):
        n_in_row = 5 if row % 2 == 0 else 4
        for col in range(n_in_row):
            if idx >= N: break
            x = 0.2 + col * 0.16 + (row % 2) * 0.08
            y = 0.2 + row * 0.14
            v1[3*idx] = np.clip(x, 0.1, 0.9)
            v1[3*idx+1] = np.clip(y, 0.1, 0.9)
            v1[3*idx+2] = 0.04
            idx += 1
    inits.append(v1)
    
    # 3. Perturbed Grid (deterministic shift)
    v2 = v0.copy()
    shift = np.array([0.03, -0.02, 0.0])
    for i in range(N):
        v2[3*i] += shift[0] * ((i % 3) - 1)
        v2[3*i+1] += shift[1] * ((i % 2) - 0.5)
    # Ensure bounds
    v2[0::3] = np.clip(v2[0::3], 0.1, 0.9)
    v2[1::3] = np.clip(v2[1::3], 0.1, 0.9)
    inits.append(v2)
    
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': eval_constraints}
    
    for v_init in inits:
        try:
            res = minimize(
                objective, v_init, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 300, 'ftol': 1e-10}
            )
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                best_radii = res.x[2::3].copy()
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_centers is None:
        best_centers = inits[0][0::3].reshape(-1, 2)
        best_radii = inits[0][2::3]
        best_sum = np.sum(best_radii)
        
    # Minor projection to guarantee strict feasibility within tolerance
    # (SLSQP might return results extremely close to boundaries)
    eps = 1e-14
    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, best_radii[:, np.newaxis], 1.0 - best_radii[:, np.newaxis])
    
    return best_centers, best_radii, best_sum
