from scipy.optimize import minimize, linprog
import math

def calc_dist_matrix(centers):
    """Calculates the distance matrix between all circle centers."""
    return np.sqrt(((centers[:, np.newaxis, :] - centers[:, :, :]) ** 2).sum(axis=2))

def calc_wall_distances(centers):
    """Calculates the distance from each center to the nearest wall."""
    x, y = centers[:, 0], centers[:, 1]
    return np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
