import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    def objective(vars):
        # We want to maximize sum of radii, so minimize negative sum
        return -np.sum(vars[2::3])

    def get_constraints():
        cons = []
        # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
        for i in range(n):
            idx = 3 * i
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx] - v[idx+2]})
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx] - v[idx+2]})
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx+1] - v[idx+2]})
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx+1] - v[idx+2]})
            
        # Overlap constraints: dist^2 >= (r1 + r2)^2
        for i in range(n):
            for j in range(i + 1, n):
                def make_con(ia, ib):
                    return {'type': 'ineq', 'fun': lambda v, i=ia, j=ib: (v[3*i]-v[3*j])**2 + (v[3*i+1]-v[3*j+1])**2 - (v[3*i+2]+v[3*j+2])**2}
                cons.append(make_con(i, j))
        return cons

    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    constraints = get_constraints()
    
    best_val = np.inf
    best_x = None
    
    # Multiple restarts to avoid local minima
    for seed in range(10):
        np.random.seed(seed)
        if seed < 5:
            # Grid-like initialization with perturbation
            pts = []
            for r in range(5):
                for c in range(5):
                    pts.append([c/5.0 + 0.1, r/5.0 + 0.1])
            pts.append([0.5, 0.5])
            pts = np.array(pts)
            pts += np.random.normal(0, 0.015, size=pts.shape)
        else:
            # Random initialization
            pts = np.random.rand(n, 2) * 0.8 + 0.1
            
        pts = np.clip(pts, 0.01, 0.99)
        
        # Compute feasible initial radii
        radii = np.full(n, 0.01)
        for i in range(n):
            min_d = min(pts[i,0], 1-pts[i,0], pts[i,1], 1-pts[i,1])
            for j in range(n):
                if i != j:
                    d = np.linalg.norm(pts[i] - pts[j])
                    if d < min_d:
                        min_d = d
            radii[i] = min_d / 2.0
            
        x0 = np.zeros(3*n)
        for i in range(n):
            x0[3*i] = pts[i,0]
            x0[3*i+1] = pts[i,1]
            x0[3*i+2] = radii[i]
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-10})
            if res.fun < best_val:
                best_val = res.fun
                best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        # Fallback to simple grid
        x0 = np.zeros(3*n)
        for i in range(n):
            r = i // 5
            c = i % 5
            x0[3*i] = c/5.0 + 0.1
            x0[3*i+1] = r/5.0 + 0.1
            x0[3*i+2] = 0.08
        best_x = x0
        
    centers = best_x[::3].reshape(n, 2)
    radii = best_x[2::3]
    
    # Ensure strict feasibility and clean up numerical errors
    for i in range(n):
        radii[i] = max(radii[i], 1e-5)
        centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-7, 1.0 - radii[i] - 1e-7)
        centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-7, 1.0 - radii[i] - 1e-7)
        
    # Final overlap check and minor adjustment if needed
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            req = radii[i] + radii[j]
            if dist < req - 1e-9:
                shrink = (req - dist + 1e-6) / 2.0
                radii[i] = max(radii[i] - shrink, 1e-5)
                radii[j] = max(radii[j] - shrink, 1e-5)
                
    return centers, radii, float(np.sum(radii))
