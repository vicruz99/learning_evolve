import numpy as np
import scipy.optimize as opt

def compute_loss_and_constraints(vars, n):
    """
    Computes the negative sum of radii (loss) and all inequality constraints.
    Constraints are returned as an array where each element should be >= 0.
    """
    c = vars[:2 * n].reshape(n, 2)
    r = vars[2 * n:]
    
    # Objective: maximize sum(r) <=> minimize -sum(r)
    loss = -np.sum(r)
    
    # Preallocate constraint array
    # 4*n boundary constraints + n*(n-1)/2 distance constraints
    num_dist = n * (n - 1) // 2
    cons = np.zeros(4 * n + num_dist)
    
    idx = 0
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    cons[idx:idx + n] = c[:, 0] - r
    idx += n
    cons[idx:idx + n] = 1.0 - c[:, 0] - r
    idx += n
    cons[idx:idx + n] = c[:, 1] - r
    idx += n
    cons[idx:idx + n] = 1.0 - c[:, 1] - r
    idx += n
    
    # Distance constraints: ||c_i - c_j|| - r_i - r_j >= 0
    # Vectorized computation using broadcasting
    c_diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]  # Shape (n, n, 2)
    dists = np.linalg.norm(c_diff, axis=2)              # Shape (n, n)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]         # Shape (n, n)
    
    # Extract upper triangular part (excluding diagonal)
    triu_indices = np.triu_indices(n, k=1)
    cons[idx:] = dists[triu_indices] - r_sum[triu_indices]
    
    return loss, cons

    best_vars = None

    cons_dict = {

        return compute_loss_and_constraints(v, n)[0]

    def run_optimization(initial_vars):

            res = opt.minimize(
                objective,
                initial_vars,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_dict,
                options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
            )
            return res.fun, res.x
        except Exception:
            return np.inf, initial_vars

    pts = []
