import numpy as np

from scipy.spatial.distance import pdist, squareform

def solve_lp_for_centers(centers):
    """
    Solves the LP to find optimal radii for a fixed set of centers.
    Maximizes sum(r_i) subject to r_i + r_j <= dist(i,j) and boundary constraints.
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Compute pairwise distances
    dists = squareform(pdist(centers))
    
    # Compute boundary distances for each center
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    bounds_x = np.minimum(centers[:, 0], 1 - centers[:, 0])
    bounds_y = np.minimum(centers[:, 1], 1 - centers[:, 1])
    bound_dists = np.minimum(bounds_x, bounds_y)
    
    # Construct constraints A_ub * r <= b_ub
    # 1. Pairwise constraints: r_i + r_j <= dists[i, j] for i < j
    # 2. Boundary constraints: r_i <= bound_dists[i]
    
    num_pairwise = 0
    for i in range(n):
        for j in range(i + 1, n):
            num_pairwise += 1
            
    num_boundary = n
    
    A_ub = np.zeros((num_pairwise + num_boundary, n))
    b_ub = np.zeros(num_pairwise + num_boundary)
    
    row_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dists[i, j]
            row_idx += 1
            
    for i in range(n):
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = bound_dists[i]
        row_idx += 1
        
    # Bounds for r_i >= 0
    bounds_r = [(0, None)] * n
    
    # Solve LP
    # method='highs' is efficient
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if res.success:
        radii = res.x
        sum_radii = np.sum(radii)
        return sum_radii, radii
    else:
        # Fallback if LP fails (should not happen with feasible region r=0)
        # Return 0 radii
        return 0.0, np.zeros(n)

    n = 26
    centers = center_params.reshape(n, 2)
    
    # Quick check to avoid invalid inputs if any (though bounds handle it)
    # If centers are outside [0,1], boundary dists might be negative?
    # But bounds in DE keep them in [0,1].
    # However, if center is exactly on boundary, bound_dist is 0, r=0.
    # That's valid.
    
    sum_radii, _ = solve_lp_for_centers(centers)
    return -sum_radii
