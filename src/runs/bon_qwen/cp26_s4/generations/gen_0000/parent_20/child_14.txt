from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26)
        sum_radii: float
    """

    best_radii = None

    def calculate_cost_and_penalties(params):
        # Reshape parameters to (N, 3) -> [x, y, r]
        points = params.reshape(N, 3)
        centers = points[:, :2]
        radii = points[:, 2]
        
        # Ensure radii are non-negative for penalty calculation
        radii_clamped = np.maximum(radii, 0)
        
        # 1. Boundary Penalties
        # x - r >= 0  => r - x <= 0
        # x + r <= 1  => x + r - 1 <= 0
        # Same for y
        pen_boundary = 0.0
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii_clamped[i]
            
            # Left/Right walls
            if x - r < 0:
                pen_boundary += (x - r)**2
            if x + r > 1:
                pen_boundary += (x + r - 1)**2
            # Top/Bottom walls
            if y - r < 0:
                pen_boundary += (y - r)**2
            if y + r > 1:
                pen_boundary += (y + r - 1)**2
                
        # 2. Overlap Penalties
        # dist >= r_i + r_j  => (r_i + r_j)^2 - dist^2 <= 0
        pen_overlap = 0.0
        # Vectorized overlap check for efficiency
        # centers shape (N, 2), radii_clamped shape (N,)
        # Compute pairwise distances squared
        # diff = centers[i] - centers[j]
        # Using broadcasting
        c1 = centers[:, np.newaxis, :] # (N, 1, 2)
        c2 = centers[np.newaxis, :, :] # (1, N, 2)
        diff = c1 - c2                 # (N, N, 2)
        dist_sq = np.sum(diff**2, axis=2) # (N, N)
        
        r_sum = radii_clamped[:, np.newaxis] + radii_clamped[np.newaxis, :] # (N, N)
        overlap_constraint = r_sum**2 - dist_sq
        
        # We only care about upper triangle to avoid double counting and self
        # mask for i < j
        mask = np.triu(np.ones((N, N), dtype=bool), k=1)
        violations = overlap_constraint[mask]
        
        # Only penalize positive violations (overlaps)
        pos_violations = np.maximum(violations, 0)
        pen_overlap = np.sum(pos_violations**2)
        
        # Objective: Maximize sum of radii -> Minimize negative sum
        obj = -np.sum(radii_clamped)
        
        # Total cost
        penalty_weight = 1000.0
        total_cost = obj + penalty_weight * (pen_boundary + pen_overlap)
        
        return total_cost, centers, radii_clamped
