import numpy as np
from scipy.optimize import minimize
import math

def compute_loss_and_grad(params):
    """
    Computes the objective function and its gradient for the circle packing problem.
    Uses a penalty method to handle boundary and non-overlap constraints.
    """
    n = 26
    mu = 20000.0  # Penalty weight
    
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    loss = -np.sum(radii)
    grad = np.zeros_like(params)
    grad[2*n:] = -1.0  # Gradient of -sum(radii)
    
    # Boundary penalties: x, y must be in [r, 1-r]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # x - r >= 0
        v = x - r
        if v < 0:
            loss += mu * v**2
            grad[3*i] += mu * 2 * v
            grad[3*i+2] -= mu * 2 * v
            
        # 1 - x - r >= 0
        v = 1.0 - x - r
        if v < 0:
            loss += mu * v**2
            grad[3*i] -= mu * 2 * v
            grad[3*i+2] -= mu * 2 * v
            
        # y - r >= 0
        v = y - r
        if v < 0:
            loss += mu * v**2
            grad[3*i+1] += mu * 2 * v
            grad[3*i+2] -= mu * 2 * v
            
        # 1 - y - r >= 0
        v = 1.0 - y - r
        if v < 0:
            loss += mu * v**2
            grad[3*i+1] -= mu * 2 * v
            grad[3*i+2] -= mu * 2 * v
            
    # Overlap penalties: dist(i, j) >= r_i + r_j
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            dist = math.sqrt(dist_sq)
            
            if dist < 1e-12:
                dist = 1e-12
                dx = 1e-12
                
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                term = mu * 2.0 * overlap
                ddx = -dx / dist
                ddy = -dy / dist
                
                loss += mu * overlap**2
                
                grad[3*i] += term * ddx
                grad[3*i+1] += term * ddy
                grad[3*i+2] += term
                
                grad[3*j] -= term * ddx
                grad[3*j+1] -= term * ddy
                grad[3*j+2] += term
                
    return loss, grad

def run_packing():
    n = 26
    # Bounds: x,y in [0,1], r in [0.01, 0.25]
    bounds = [(0.0, 1.0)] * 2*n + [(0.01, 0.25)] * n
    
    best_params = None
    best_loss = np.inf
    
    # Multi-start optimization
    for seed in range(30):
        rng = np.random.default_rng(seed + 42)
        params = np.zeros(3*n)
        
        # Initialize with a perturbed hexagonal grid
        for i in range(n):
            row = i // 6
            col = i % 6
            base_x = 0.1 + col * 0.14 + (0.07 if row % 2 == 1 else 0.0)
            base_y = 0.1 + row * 0.13
            
            params[3*i] = np.clip(base_x + rng.uniform(-0.02, 0.02), 0.05, 0.95)
            params[3*i+1] = np.clip(base_y + rng.uniform(-0.02, 0.02), 0.05, 0.95)
            params[3*i+2] = 0.085 + rng.uniform(-0.005, 0.005)
            
        res = minimize(compute_loss_and_grad, params, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-9})
                       
        if res.fun < best_loss:
            best_loss = res.fun
            best_params = res.x
            
    centers = best_params[:2*n].reshape(n, 2)
    radii = best_params[2*n:]
    
    # Post-processing: ensure strict validity within numerical tolerance
    for _ in range(50):

        for i in range(n):
