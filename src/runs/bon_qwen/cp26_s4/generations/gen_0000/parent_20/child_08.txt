import numpy as np
from scipy.optimize import minimize, dual_annealing


def compute_overlap_penalty(centers, radii):
    """Compute penalty for overlapping circles."""
    n = centers.shape[0]
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq + 1e-16)
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            if overlap > 0:
                penalty += overlap ** 2
    return penalty


def compute_boundary_penalty(centers, radii):
    """Compute penalty for circles outside the unit square."""
    penalty = 0.0
    for i in range(centers.shape[0]):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += (r - x) ** 2
        if x + r > 1:
            penalty += (x + r - 1) ** 2
        if y - r < 0:
            penalty += (r - y) ** 2
        if y + r > 1:
            penalty += (y + r - 1) ** 2
    return penalty


def objective_function(params, n, penalty_weight):
    """Objective: minimize negative sum of radii plus penalties."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    obj = -np.sum(radii)
    overlap_pen = compute_overlap_penalty(centers, radii)
    boundary_pen = compute_boundary_penalty(centers, radii)
    
    return obj + penalty_weight * (overlap_pen + boundary_pen)

def force_directed_optimize(centers, radii, n, iterations=500, lr=0.01):
    """Apply force-directed optimization to reduce overlaps."""
    centers = centers.copy()
    radii = radii.copy()
    
    for it in range(iterations):
        forces = np.zeros_like(centers)
        
        # Repulsive forces between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq + 1e-16)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    scale = overlap / (dist + 1e-16)
                    forces[i, 0] += dx * scale
                    forces[i, 1] += dy * scale
                    forces[j, 0] -= dx * scale
                    forces[j, 1] -= dy * scale
        
        # Boundary repulsion forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                forces[i, 0] += (r - x) * 5
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 5
            if y - r < 0:
                forces[i, 1] += (r - y) * 5
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 5
        
        centers += lr * forces
        centers = np.clip(centers, 0.001, 0.999)
    
    return centers, radii


def optimize_positions_for_radii(centers, radii, n, penalty_weight=500.0):
    """Optimize circle positions for fixed radii."""
    def pos_objective(params):
        c = params.reshape(n, 2)
        return compute_overlap_penalty(c, radii) + compute_boundary_penalty(c, radii)
    
    params0 = centers.flatten()
    bounds = [(0, 1)] * (2 * n)
    
    result = minimize(pos_objective, params0, bounds=bounds, method='L-BFGS-B', 
                      options={'maxiter': 5000})
    
    return result.x.reshape(n, 2)
