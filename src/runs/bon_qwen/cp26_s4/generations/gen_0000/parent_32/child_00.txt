import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars, n):
    """Constraint function: boundaries and non-overlap conditions."""
    c = vars.reshape(n, 3)
    x, y, r = c[:, 0], c[:, 1], c[:, 2]
    
    cons = []
    # Boundary constraints: circle must be inside [0,1]^2
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    pairwise = dx**2 + dy**2 - dr**2
    
    cons.append(pairwise)
    return np.concatenate(cons)

def run_packing():
    n = 26
    
    # 1. Generate initial hexagonal packing configuration
    pts = []
    a = 0.16  # Base spacing for hexagonal lattice
    for i in range(10):
        for j in range(10):
            x = (i + 0.5) * a
            y = (j + 0.5 * (i % 2)) * a * np.sqrt(3) / 2
            if 0 < x < 1 and 0 < y < 1:
                pts.append([x, y])
    pts = np.array(pts)
    
    # Select exactly n points, preferring a regular grid-like distribution
    if len(pts) >= n:
        order = np.lexsort((pts[:, 0], pts[:, 1]))
        centers = pts[order[:n]]
    else:
        centers = pts
        while len(centers) < n:
            centers = np.vstack([centers, [0.5, 0.5]])
            
    # Ensure initial centers are safely inside the square
    centers = np.clip(centers, 0.05, 0.95)
    
    # 2. Compute feasible initial radii
    d_min = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            if d < d_min:
                d_min = d
                
    # Radii must be small enough to satisfy both non-overlap and boundary constraints
    max_r = min(d_min / 2.0, 0.04)
    radii = np.ones(n) * max_r
    
    # Flatten variables for optimizer: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
        
    # 3. Setup bounds and constraints
    bnds = [(0, 1) for _ in range(2 * n)] + [(0, 0.5) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    # 4. Run optimization
    res = minimize(objective, x0, args=(n,), method='SLSQP', bounds=bnds, 
                   constraints=cons, options={'maxiter': 2500, 'ftol': 1e-10, 'disp': False})
                   
    # 5. Extract and validate results
    if res.success:
        final_centers = res.x[:2 * n].reshape(n, 2)
        final_radii = res.x[2 * n:]
    else:
        final_centers = centers
        final_radii = radii
        
    # Ensure non-negative radii (safety clamp)
    final_radii = np.maximum(final_radii, 0.0)
    
    return final_centers, final_radii, np.sum(final_radii)
