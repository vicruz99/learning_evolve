import numpy as np

def penalty_func(x, r, n):
    centers = x.reshape(n, 2)
    x_coords = centers[:, 0]
    y_coords = centers[:, 1]
    
    # Wall penalties
    # x >= r => r - x <= 0. Violation if r - x > 0.
    # Penalty (r - x)^2
    p = np.maximum(0, r - x_coords)**2
    p += np.maximum(0, x_coords - (1 - r))**2
    p += np.maximum(0, r - y_coords)**2
    p += np.maximum(0, y_coords - (1 - r))**2
    wall_pen = np.sum(p)
    
    # Overlap penalties
    # Distance matrix
    # diff[i, j, k] = centers[i, k] - centers[j, k]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Indices for upper triangle
    i_idx, j_idx = np.triu_indices(n, k=1)

    # Violation if 2r - dist > 0.
    overlap_vals = np.maximum(0, 2 * r - dists[i_idx, j_idx])
    overlap_pen = np.sum(overlap_vals**2)
