def run_packing():

        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float

    def solve_from_initial(centers_init, radii_init):

        for i in range(n):
            x0[2*i] = centers_init[i, 0]
            x0[2*i + 1] = centers_init[i, 1]
            x0[3*n + i] = radii_init[i]

            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 1.0)) # r (loose upper bound)

            r = vars[3*n:]
            return -np.sum(r)

        def constraints(vars):
            centers = np.column_stack((vars[:n], vars[n:2*n]))
            r = vars[3*n:]
            c_list = []

                y = centers[i, 1]

                    c_list.append(dist_sq - sum_r**2)
            
            return c_list
