import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.spatial.distance import cdist
import time

def solve_radii(centers):
    """
    Given a set of centers, solves the LP to find radii that maximize sum of radii.
    Constraints:
    1. r_i >= 0
    2. Circle inside square: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    3. Non-overlap: r_i + r_j <= distance(centers[i], centers[j])
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r) <=> Minimize -sum(r)
    c = -np.ones(n)
    
    # Constraints A_ub @ r <= b_ub
    # We will construct lists of rows for the matrix
    A_rows = []
    b_rows = []
    
    # Wall constraints: 4 per circle
    # r_i <= x_i  =>  1*r_i <= x_i
    # r_i <= 1-x_i => 1*r_i <= 1-x_i
    # r_i <= y_i  =>  1*r_i <= y_i
    # r_i <= 1-y_i => 1*r_i <= 1-y_i
    
    # Pre-allocate for speed if needed, but list append is fine for n=26
    for i in range(n):
        x, y = centers[i]
        
        # Row for r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(x)
        
        # Row for r_i <= 1-x
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(1.0 - x)
        
        # Row for r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(y)
        
        # Row for r_i <= 1-y
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(1.0 - y)
        
    # Pair constraints: r_i + r_j <= d_ij
    # Number of pairs = n*(n-1)/2
    # For n=26, 325 constraints.
    # Constructing dense matrix might be memory heavy? 
    # 430 rows x 26 cols is tiny. No problem.
    
    # Calculate all pairwise distances efficiently
    dist_matrix = cdist(centers, centers)
    # Set diagonal to infinity or large number to ignore, though we won't access it
    np.fill_diagonal(dist_matrix, np.inf)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = dist_matrix[i, j]
            # If circles are on top of each other, distance is 0, radii must be 0.
            # But optimizer should prevent this.
            
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_rows.append(d)
            
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bounds for r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # method='highs' is the default and efficient solver in recent scipy
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # If LP fails, return 0 radii
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective(centers_flat):
    """
    Objective function for optimization.
    We want to MAXIMIZE sum of radii, so we MINIMIZE negative sum.
    """
    centers = centers_flat.reshape(-1, 2)
    # Clip centers to [0, 1] just in case, though bounds handle it
    # But linprog handles radii constraints based on centers.
    # If centers are outside, constraints might allow negative radii? 
    # No, r <= x. If x < 0, r <= negative. But r >= 0 bound enforces r=0.
    # So it's safe.
    
    total_radius, radii = solve_radii(centers)
    return -total_radius
