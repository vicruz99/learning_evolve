import numpy as np
from scipy.optimize import linprog, differential_evolution

def solve_lp_for_radii(centers):
    """
    Solves the LP problem to find the maximum sum of radii for fixed centers.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        sum_radii: float
        radii: np.array of shape (n)
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r_i) is equivalent to maximizing sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints setup
    # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # 2. Pairwise constraints: r_i + r_j <= dist(i, j)
    
    A_ub_rows = []
    b_ub_rows = []
    
    # Add boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_rows.append(x)
        
        # r_i <= 1 - x_i
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_rows.append(1 - x)
        
        # r_i <= y_i
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_rows.append(y)
        
        # r_i <= 1 - y_i
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_rows.append(1 - y)
        
    # Add pairwise distance constraints
    # Compute pairwise distances
    dist_matrix = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = dist_matrix[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_rows.append(dist)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        return -res.fun, radii
    else:
        # If LP fails (e.g. infeasible), return 0 and zeros
        return 0.0, np.zeros(n)

def objective(centers_flat):
    """
    Objective function for optimization.
    Minimize the negative sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    sum_radii, radii = solve_lp_for_radii(centers)
    return -sum_radii

def generate_hexagonal_grid(n, jitter=0.01):
    """
    Generates an initial hexagonal grid of n points in the unit square.
    """
    points = []
    # Approximate spacing for n points in a square
    s = 1.0 / np.sqrt(n)
    
    y = 0.5 * s
    while True:
        if len(points) >= n:
            break
        # Row offset
        x_start = 0.5 * s if int((y - 0.5 * s) / (s * np.sqrt(3)/2)) % 2 == 1 else 0.0
        
        x = x_start + 0.5 * s
        while True:
            if x > 1.0:
                break
            # Check if point is valid and add with some jitter
            if len(points) < n:
                points.append([x, y])
            x += s
        y += s * np.sqrt(3) / 2
        
    points = np.array(points[:n])
    # Add jitter to avoid symmetry issues
    points += np.random.uniform(-jitter, jitter, points.shape)
    
    # Clip to valid range to ensure radii can be positive
    points = np.clip(points, 0.001, 0.999)
    
    return points.flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Runs the packing optimization.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    np.random.seed(42) # For reproducibility
    n = 26
    
    # Generate initial guess
    x0 = generate_hexagonal_grid(n, jitter=0.02)
    
    # Define bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    
    # Run Differential Evolution to optimize centers
    # popsize and maxiter are tuned for balance between speed and quality
    result = differential_evolution(
        objective, 
        bounds, 
        popsize=20, 
        maxiter=100, 
        tol=1e-5, 
        seed=42, 
        workers=1
    )
    
    best_centers = result.x.reshape(n, 2)
    
    # Solve final LP for the best centers found
    sum_radii, final_radii = solve_lp_for_radii(best_centers)
    
    return best_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
