import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(123)
    n = 26

    # Helper to calculate penalty
    def calc_loss(params):
        c = params[:2*n].reshape((n, 2))
        r = params[2*n:]
        
        loss = 0.0
        
        # Boundary penalty
        # We want r <= x <= 1-r and r <= y <= 1-r
        # Violation: max(0, r-x) + max(0, x-(1-r)) etc.
        # Using squared penalty
        for i in range(n):
            x, y = c[i]
            rad = max(r[i], 0.0) # Ensure non-negative for calculation
            
            # Left
            if x < rad:
                loss += 1000 * (rad - x)**2
            # Right
            if x > 1 - rad:
                loss += 1000 * (x - (1 - rad))**2
            # Bottom
            if y < rad:
                loss += 1000 * (rad - y)**2
            # Top
            if y > 1 - rad:
                loss += 1000 * (y - (1 - rad))**2

        # Overlap penalty
        for i in range(n):
            rad_i = max(r[i], 0.0)
            for j in range(i + 1, n):
                rad_j = max(r[j], 0.0)
                dist = np.sqrt((c[i,0]-c[j,0])**2 + (c[i,1]-c[j,1])**2)
                req_dist = rad_i + rad_j
                if dist < req_dist:
                    loss += 5000 * (req_dist - dist)**2
        
        # We maximize sum(r), so minimize -sum(r)
        return -np.sum(r) + loss

    # Initialization: Hexagonal-like grid perturbed
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.1
    
    # Place 25 circles in a 5x5 grid with some spacing
    # Then add 1 in the middle? Or distribute.
    # Let's try a dense random start to avoid grid lock-in
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.ones(n) * 0.08
    
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    # Optimization
    # Using SLSQP as it handles bounds and is generally effective
    # We might need to run multiple times or use a robust method
    res = opt.minimize(calc_loss, x0, method='SLSQP', bounds=bounds, 
                       options={'maxiter': 2000, 'ftol': 1e-12})
    
    # Extract results
    c_final = res.x[:2*n].reshape((n, 2))
    r_final = res.x[2*n:]
    
    # Ensure non-negative radii
    r_final = np.maximum(r_final, 1e-9)
    
    # Clip centers to valid range [r, 1-r]
    for i in range(n):
        r_i = r_final[i]
        c_final[i, 0] = np.clip(c_final[i, 0], r_i, 1 - r_i)
        c_final[i, 1] = np.clip(c_final[i, 1], r_i, 1 - r_i)
        
    return c_final, r_final, np.sum(r_final)
