def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    # 1. Initialize in a hexagonal pattern
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.15
    dx = 0.13
    dy = dx * np.sqrt(3) / 2
    row = 0
    while idx < n:
        # Alternate row lengths to approximate hexagonal packing
        num = 6 if row % 2 == 0 else 5
        x_start = 0.5 - (num - 1) * dx / 2
        for i in range(num):
            if idx >= n: break
            centers[idx, 0] = x_start + i * dx
            centers[idx, 1] = y
            idx += 1
        y += dy
        row += 1
        
    # Add small random jitter to break symmetry and avoid local minima
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    
    radii = np.full(n, 0.05)
    
    # 2. Iterative expansion & relaxation simulation
    steps = 120000
    growth_rate = 1e-4
    repulsion_k = 1.5
    
    for step in range(steps):
        radii += growth_rate
        
        # Resolve constraints iteratively until stable
        for _ in range(40):
            moved = False
            # Handle boundaries: clamp centers to keep circles inside [0,1]
            for i in range(n):
                if centers[i, 0] < radii[i]: centers[i, 0] = radii[i]; moved = True
                if centers[i, 0] > 1.0 - radii[i]: centers[i, 0] = 1.0 - radii[i]; moved = True
                if centers[i, 1] < radii[i]: centers[i, 1] = radii[i]; moved = True
                if centers[i, 1] > 1.0 - radii[i]: centers[i, 1] = 1.0 - radii[i]; moved = True

            for i in range(n):
                for j in range(i + 1, n):
                    dx_c = centers[i, 0] - centers[j, 0]
                    dy_c = centers[i, 1] - centers[j, 1]

                    req = radii[i] + radii[j]

                        overlap = req - dist

                        mx = overlap * dx_c / dist * repulsion_k
                        my = overlap * dy_c / dist * repulsion_k

                        centers[i, 1] += my * 0.5
                        centers[j, 0] -= mx * 0.5
                        centers[j, 1] -= my * 0.5
                        moved = True
            if not moved:
                break
                
        # Slow down expansion as we approach maximum packing density
        if step % 10000 == 0:
            growth_rate *= 0.85
            
    # Final safety check: slightly shrink if any tiny numerical violations persist

    max_viol = 0.0
    for i in range(n):
        if centers[i,0] - radii[i] < max_viol: max_viol = centers[i,0] - radii[i]

                max_viol = d - radii[i] - radii[j]
                
    if max_viol < -1e-9:
        scale = 1.0 + max_viol / np.max(radii)
        radii *= scale
        
    return centers, radii, float(np.sum(radii))
