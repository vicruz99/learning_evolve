import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # --- Initialization ---
    # Start with a grid structure to ensure a valid starting point, then perturb.
    # A 5x5 grid is 25 circles. We place the 26th in a gap.
    centers = []
    for i in range(5):
        for j in range(5):
            # Add slight random jitter to break symmetry
            x = (i + 0.5) / 5.0 + np.random.uniform(-0.05, 0.05)
            y = (j + 0.5) / 5.0 + np.random.uniform(-0.05, 0.05)
            centers.append([x, y])
    
    # Add 26th circle in a random position
    centers.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
    centers = np.array(centers)
    
    # Initialize radii slightly smaller than 0.1 to ensure initial validity
    radii = np.full(n_circles, 0.09)

    # --- Optimization Setup ---
    
    # Flatten centers and radii into a single vector for the optimizer
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(n_circles * 3)
    for i in range(n_circles):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]

    def objective(x):
        """
        Objective function: Maximize sum of radii.
        Scipy minimize minimizes, so we return the negative sum.
        """
        return -np.sum(x[2::3])  # Every 3rd variable is a radius

    def constraints_gen(n):
        """
        Generator for constraints.
        Returns a list of constraint dictionaries compatible with SLSQP.
        """
        cons = []
        
        # 1. Boundary Constraints:
        # x >= r  => x - r >= 0  => -(x - r) <= 0
        # x <= 1-r => x + r - 1 <= 0
        # Same for y
        for i in range(n):
            idx = 3 * i
            xi = x[idx]
            yi = x[idx + 1]
            ri = x[idx + 2]
            
            # x - ri >= 0  ->  ri - x <= 0
            cons.append({'type': 'ineq', 'fun': lambda v, k=idx: v[k] - v[k+2]}) # v[idx] - v[idx+2] >= 0
            # xi + ri - 1 <= 0 -> 1 - (xi + ri) >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, k=idx: 1.0 - (v[k] + v[k+2])})
            
            # y - ri >= 0 -> ri - y <= 0
            cons.append({'type': 'ineq', 'fun': lambda v, k=idx: v[k+1] - v[k+2]})
            # yi + ri - 1 <= 0 -> 1 - (yi + ri) >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, k=idx: 1.0 - (v[k+1] + v[k+2])})

        # 2. Overlap Constraints:
        # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
        # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_i = 3 * i
                idx_j = 3 * j
                
                def overlap_constraint(v, i=idx_i, j=idx_j):
                    xi, yi, ri = v[i], v[i+1], v[i+2]
                    xj, yj, rj = v[j], v[j+1], v[j+2]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    rad_sum_sq = (ri + rj)**2
                    return dist_sq - rad_sum_sq
                
                cons.append({'type': 'ineq', 'fun': overlap_constraint})
        
        return cons

    # Define bounds for variables
    # x, y in [0, 1], r >= 0
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (max possible radius is 0.5)

    # Generate constraints list
    # Note: Passing large number of constraints can be slow, but for 26 circles (325 overlaps) it is manageable.
    constraints = constraints_gen(n_circles)

    # Run Optimization
    # SLSQP is suitable for this type of non-linear optimization with constraints.
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )

    # Extract results
    best_x = result.x
    
    # Unpack centers and radii
    final_centers = np.zeros((n_circles, 2))
    final_radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        final_centers[i, 0] = best_x[3*i]
        final_centers[i, 1] = best_x[3*i + 1]
        final_radii[i] = best_x[3*i + 2]
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

    centers, radii, total = run_packing()
