def make_hexagonal_initial(n, row_sizes, r_base=0.095):

def penalty_objective(params, n, alpha=5000.0):
    """Objective function with penalty terms for constraints."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    obj = -np.sum(radii)

        x, y = centers[i]
        r = radii[i]

            obj += alpha * (x - r) ** 2
        if x + r > 1:
            obj += alpha * (x + r - 1) ** 2
        if y - r < 0:
            obj += alpha * (y - r) ** 2
        if y + r > 1:
            obj += alpha * (y + r - 1) ** 2

            min_dist = radii[i] + radii[j]

                gap = min_dist - dist

def validate_and_fix(centers, radii, n):
