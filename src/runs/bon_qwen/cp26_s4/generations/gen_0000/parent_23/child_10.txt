def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

    rows = []

    row_idx = 0
    
    while count < n:
        y = r_init + row_idx * np.sqrt(3) * r_init
        if y + r_init > 1.0:
            break
            
        # Determine number of circles in this row
        # If even row (0, 2, ...), centers at r, 3r, 5r...
        # If odd row (1, 3, ...), centers at 2r, 4r...
        
        if row_idx % 2 == 0:
            # Even row
            # Centers: r + 2*j*r
            # Max j such that r + 2*j*r + r <= 1 => 2r(j+1) <= 1 => j+1 <= 1/(2r)
            max_circles = int(1.0 / (2.0 * r_init))
        else:
            # Odd row
            # Centers: 2r + 2*j*r
            # Max j such that 2r + 2*j*r + r <= 1 => r(2j + 3) <= 1 => 2j+3 <= 1/r
            # Wait, center at 2r (for j=0). Last center 2r + 2*(m-1)r = 2mr.

            max_circles = int((1.0 / r_init - 1.0) / 2.0)
            
        # Adjust to not exceed remaining needed circles
        needed = n - count
        num_in_row = min(max_circles, needed)
        
        current_row_centers = []
        for j in range(num_in_row):
            if row_idx % 2 == 0:
                x = r_init + 2 * j * r_init
            else:
                x = 2 * r_init + 2 * j * r_init
            current_row_centers.append([x, y])
        
        rows.append(current_row_centers)
        count += num_in_row
        row_idx += 1
        
    # If we didn't get n circles, we might need to adjust r or pattern, 
    # but with r=0.09 we should get plenty.
    # Flatten centers
    centers_list = []
    for row in rows:
        centers_list.extend(row)
        
    centers = np.array(centers_list[:n])
    radii = np.full(n, r_init)
    
    return centers, radii
