def objective_with_penalty(v, centers_shape=None):
    """
    Computes the objective value (negative sum of radii + penalties).
    v is a flat array of size 3*N: [x1, y1, r1, x2, y2, r2, ...]
    """
    n = N_CIRCLES
    # Reshape
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    # Base objective: -sum(radii)
    obj = -np.sum(radii)
    
    # Penalties
    penalty_weight = 10000.0 # High weight to enforce constraints
    
    # 1. Boundary penalties
    # x - r >= 0 => r - x <= 0. If r > x, penalty.
    # We use max(0, violation)^2
    # Violation for x >= r: r - x. If positive, penalty.
    # But constraint is x >= r. So if x < r, violation is r - x.
    # Wait, x >= r means r - x <= 0.
    # Let violation = max(0, r - x).
    
    v_x_left = np.maximum(0, radii - centers[:, 0])
    v_x_right = np.maximum(0, radii - (1.0 - centers[:, 0]))
    v_y_bottom = np.maximum(0, radii - centers[:, 1])
    v_y_top = np.maximum(0, radii - (1.0 - centers[:, 1]))
    
    boundary_penalty = np.sum(v_x_left**2) + np.sum(v_x_right**2) + \
                       np.sum(v_y_bottom**2) + np.sum(v_y_top**2)
    
    # 2. Overlap penalties
    # Distance between i and j: dist
    # Constraint: dist >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)
    
    # Vectorized computation of pairwise distances
    # Centers shape (N, 2)
    # Diff shape (N, N, 2)
    # dist shape (N, N)
    
    # Optimization: only compute upper triangle
    # But for simplicity and numpy speed, full matrix is ok for N=26
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (N, N, 2)
    dists = np.sqrt(np.sum(diff**2, axis=2)) # (N, N)
    
    # Radii sum matrix (N, N)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violation
    overlaps = np.maximum(0, r_sum - dists)
    
    # Sum of squared violations (upper triangle to avoid double counting, though symmetry handles it)
    # Actually sum over all pairs. 
    overlap_penalty = np.sum(overlaps**2)
    
    obj += penalty_weight * (boundary_penalty + overlap_penalty)
    
    return obj

def gradient_with_penalty(v):
    """
    Computes the gradient of the objective function with penalties.
    """
    n = N_CIRCLES
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    grad = np.zeros_like(v)
    
    # Gradient of -sum(radii)
    # d/dr_i = -1
    grad[2*n:] = -1.0
    
    penalty_weight = 10000.0
    
    # 1. Boundary gradients
    # Term: max(0, r - x)^2
    # d/dr = 2*max(0, r-x)*1
    # d/dx = 2*max(0, r-x)*(-1)
    
    # Left wall: r - x
    viol_xl = np.maximum(0, radii - centers[:, 0])
    grad[2*n:] += penalty_weight * 2 * viol_xl # dr
    grad[::3] += penalty_weight * 2 * viol_xl * (-1) # dx (x is at index 0, 3, 6...)
    
    # Right wall: r - (1-x) = r + x - 1
    # Wait, constraint x <= 1-r => 1-r-x <= 0 => x+r-1 <= 0?
    # Constraint: x + r <= 1. Violation x + r - 1 > 0.
    # My previous penalty used max(0, r - (1-x)) = max(0, r + x - 1). Correct.
    viol_xr = np.maximum(0, radii + centers[:, 0] - 1.0)
    grad[2*n:] += penalty_weight * 2 * viol_xr # dr
    grad[::3] += penalty_weight * 2 * viol_xr * (1) # dx
    
    # Bottom wall: r - y
    viol_yb = np.maximum(0, radii - centers[:, 1])
    grad[2*n:] += penalty_weight * 2 * viol_yb # dr
    grad[1::3] += penalty_weight * 2 * viol_yb * (-1) # dy (y is at 1, 4, 7...)
    
    # Top wall: r + y - 1
    viol_yt = np.maximum(0, radii + centers[:, 1] - 1.0)
    grad[2*n:] += penalty_weight * 2 * viol_yt # dr
    grad[1::3] += penalty_weight * 2 * viol_yt * (1) # dy
    
    # 2. Overlap gradients
    # Term for pair (i, j): max(0, r_i + r_j - dist_ij)^2
    # Let u = r_i + r_j - dist_ij
    # Term = max(0, u)^2
    # If u <= 0, grad is 0.
    # If u > 0, term = u^2.
    # d/dr_i = 2u * 1
    # d/dr_j = 2u * 1
    # d/dx_i = 2u * (-d(dist)/dx_i) = 2u * (-(x_i - x_j)/dist)
    # d/dx_j = 2u * (-d(dist)/dx_j) = 2u * (-(-(x_i - x_j)/dist)) = 2u * ((x_i - x_j)/dist)
    # d/dy_i = 2u * (-(y_i - y_j)/dist)
    # d/dy_j = 2u * ((y_i - y_j)/dist)
    
    # We iterate over pairs to accumulate gradients
    # This is O(N^2) which is fine for N=26 (325 pairs)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Avoid division by zero
    dists_safe = np.where(dists > 1e-9, dists, 1e-9)
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    u = r_sum - dists
    
    # Mask where u > 0
    active = u > 1e-12
    active = active * 1.0 # float mask
    
    # Gradient contribution for each pair (i, j)
    # We only need to compute for i < j and add to both?
    # Or just compute full matrix and sum?
    # Full matrix approach:
    # grad_dr[i] += sum_j (2 * u_ij * active_ij)
    # grad_dx[i] += sum_j (2 * u_ij * active_ij * (x_j - x_i) / dist_ij)
    
    # Vectorized accumulation
    
    # Factor 2*u*active
    factor = 2.0 * u * active
    
    # For radii
    grad_r_contrib = np.sum(factor, axis=1) # Sum over j
    grad[2*n:] += penalty_weight * grad_r_contrib
    
    # For centers
    # Gradient w.r.t x_i is sum over j of 2*u * (x_j - x_i)/dist
    # diff is (x_i - x_j). We need (x_j - x_i) = -diff.
    # So term is factor * (-diff) / dists_safe
    
    # Compute term matrix (N, N, 2)
    # factor shape (N, N)
    # diff shape (N, N, 2)
    # dists_safe shape (N, N)
    
    # term = factor[:, :, np.newaxis] * (-diff) / dists_safe[:, :, np.newaxis]
    
    # Handle dists=0 case (though centers shouldn't be identical)
    # diff/ dists is unit vector pointing from j to i?
    # (x_i - x_j)/dist is unit vector j->i.
    # We want (x_j - x_i)/dist = - unit vector.
    
    # Let's compute normalized diff
    # norm_diff = diff / dists_safe[..., np.newaxis] # (N, N, 2)
    # force = factor[..., np.newaxis] * (-norm_diff) # (N, N, 2)
    # grad_xy += sum over j of force
    
    # Optimization:
    # grad_x[i] += sum_j factor[i,j] * (x_j - x_i)/dist
    # grad_y[i] += sum_j factor[i,j] * (y_j - y_i)/dist
    
    # Let's do it explicitly to avoid memory blowup if N was larger, but N=26 is small.
    
    # Compute unit vectors from j to i
    # diff[i,j] = center[i] - center[j]
    # We want direction from i to j? 
    # d(dist)/dx_i = (x_i - x_j)/dist.
    # We want -d(dist)/dx_i = (x_j - x_i)/dist.
    # Which is - (diff[i,j]_x / dist).
    
    # Unit vector u_vec = diff / dists. Shape (N, N, 2).
    u_vec = diff / dists_safe[..., np.newaxis]
    
    # We want -u_vec * factor.
    # grad_xy[i] += sum_j ( - factor[i,j] * u_vec[i,j] )
    
    # Reshape u_vec to (N, N, 2)
    # factor is (N, N)
    # term = - factor[:, :, np.newaxis] * u_vec
    # grad_xy = sum_j term[i, j]
    
    term = - factor[:, :, np.newaxis] * u_vec # (N, N, 2)
    grad_xy = np.sum(term, axis=1) # (N, 2)
    
    # Add to gradient
    # x at indices 0, 3, 6... -> center[:, 0]
    # y at indices 1, 4, 7... -> center[:, 1]
    
    grad[::3] += penalty_weight * grad_xy[:, 0]
    grad[1::3] += penalty_weight * grad_xy[:, 1]
    
    return grad

def solve_packing(initial_centers, initial_radii):
    """
    Runs optimization starting from given centers and radii.
    Returns best valid (centers, radii, sum_radii).
    """
    n = N_CIRCLES
    # Flatten variables
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])
    
    # Bounds
    # x, y in [0, 1]
    # r >= 0
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (max possible 0.5)
    
    # Optimize
    # L-BFGS-B with analytic gradient
    res = minimize(objective_with_penalty, x0, jac=gradient_with_penalty, 
                   method='L-BFGS-B', bounds=bounds, 
                   options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 5000})
    
    # Extract solution
    v_opt = res.x
    centers_opt = v_opt[:2*n].reshape(n, 2)
    radii_opt = v_opt[2*n:]
    
    # Clean up tiny radii or negative (due to numerical issues)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    return centers_opt, radii_opt

def validate_and_score(centers, radii):
    """
    Checks validity and returns sum of radii if valid, else -1.
    """
    n = centers.shape[0]
    if n != N_CIRCLES:
        return -1.0
    
    # Check NaN
    if np.isnan(centers).any() or np.isnan(radii).any():
        return -1.0
    
    # Check radii non-negative
    if np.any(radii < -1e-9):
        return -1.0
    
    # Check boundaries
    # x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    # Allow small tolerance for check? The validation function uses 1e-12.
    # We should be safe.
    
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    if np.any(x - r < -1e-9) or np.any(x + r > 1 + 1e-9):
        return -1.0
    if np.any(y - r < -1e-9) or np.any(y + r > 1 + 1e-9):
        return -1.0
        
    # Check overlaps
    # dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # Check with tolerance
    
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = (centers[i,0] - centers[j,0])**2 + (centers[i,1] - centers[j,1])**2
            min_dist = radii[i] + radii[j]
            if dist_sq < (min_dist - 1e-9)**2: # Strict check
                 return -1.0
    
    return np.sum(radii)

def run_packing():
    guesses = get_initial_guesses()
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    print(f"Testing {len(guesses)} initial configurations...")
    
    for idx, (c, r) in enumerate(guesses):
        # Ensure shapes
        c = np.array(c, dtype=float).reshape(-1, 2)
        r = np.array(r, dtype=float).flatten()
        
        # Pad or trim if necessary
        if len(c) < N_CIRCLES:
            # Pad with small circles
            while len(c) < N_CIRCLES:
                c = np.vstack([c, [[0.5, 0.5]]])
                r = np.append(r, 0.01)
        elif len(c) > N_CIRCLES:
            c = c[:N_CIRCLES]
            r = r[:N_CIRCLES]
            
        # Try optimization
        try:
            opt_c, opt_r = solve_packing(c, r)
            score = validate_and_score(opt_c, opt_r)
            
            if score > best_sum:
                best_sum = score
                best_centers = opt_c
                best_radii = opt_r
                print(f"Guess {idx}: New best sum = {best_sum:.6f}")
            
            # Also try to perturb and optimize again?
            # Maybe run a second pass with higher penalty?
            # For now, one pass per guess.
            
        except Exception as e:
            print(f"Guess {idx} failed: {e}")

    # If we found a valid solution, return it.
    # If best_sum is still -1, return a fallback.
    if best_centers is None:
        # Fallback: small circles in grid
        fallback_c = []
        fallback_r = []
        for i in range(N_CIRCLES):
            x = 0.05 + (i % 5) * 0.2
            y = 0.05 + (i // 5) * 0.2
            fallback_c.append([x, y])
            fallback_r.append(0.05)
        best_centers = np.array(fallback_c)
        best_radii = np.array(fallback_r)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
