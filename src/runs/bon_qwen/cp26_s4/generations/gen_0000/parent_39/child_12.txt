import numpy as np
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
        sum_radii: float sum of all radii
    """
    n = 26
    
    # --- Initialization ---
    # Hexagonal grid initialization for better starting density
    centers = np.zeros((n, 2))
    # Approximate radius for initial hex packing (r ~ 0.08)
    # Spacing in hex grid: dx = 2r, dy = sqrt(3)r
    # We want to fit 26 circles. 
    # 6 rows: 5, 5, 5, 5, 5, 1 (or similar distribution)
    # Let's just distribute them roughly in a hex pattern
    
    # Estimated optimal r is around 0.1. Start slightly smaller.
    r_init = 0.06
    centers_list = []
    
    # Create a hexagonal layout
    row_idx = 0
    col_idx = 0
    x = r_init
    y = r_init
    step_x = 2 * r_init
    step_y = math.sqrt(3) * r_init
    
    current_r = r_init
    
    for i in range(n):
        centers_list.append([x, y])
        
        col_idx += 1
        x += step_x
        
        # If we reach the end of the row or wrap around
        # Check if next x would exceed 1 - current_r
        if x + current_r > 1.0:
            x = r_init + (0.5 * step_x if row_idx % 2 == 1 else 0)
            y += step_y
            row_idx += 1
            col_idx = 0
            # Reset x to start of new row
            # For hex, odd rows (1-indexed) are shifted
            if row_idx % 2 == 1:
                x = r_init + r_init # shift by r (0.5 * 2r)
            else:
                x = r_init
            
            # If y also exceeds, just reset x
            if y + current_r > 1.0:
                 # Fallback if grid is too tall, just place at bottom
                 y = r_init
                 x = r_init + i * 0.05 # spread out
    
    centers = np.array(centers_list[:n])
    radii = np.full(n, r_init)
    
    # --- Optimization Parameters ---
    num_iterations = 1500
    base_pressure = 0.0002
    repulsion_strength = 1.0
    learning_rate_init = 0.05
    learning_rate_min = 0.001
    
    # --- Optimization Loop ---
    for it in range(num_iterations):
        # 1. Grow radii (Pressure)
        # Increase radii slowly. The factor decreases slightly over time to allow settling.
        growth_factor = 1.0 + base_pressure
        radii = radii * growth_factor
        
        # 2. Calculate Forces
        forces = np.zeros_like(centers)
        
        # Compute pairwise repulsion
        # Vectorized operations for speed
        # Centers shape (n, 2)
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
        dists_sq = np.sum(diffs**2, axis=2) # (n, n)
        
        # Avoid sqrt on diagonal
        np.fill_diagonal(dists_sq, np.inf)
        dists = np.sqrt(dists_sq)
        
        # Radii sum matrix
        radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :] # (n, n)
        
        # Overlap amount: (r_i + r_j) - d_ij. Positive if overlapping.
        # We only care if dist < r_i + r_j
        overlap = radii_sum - dists
        overlap = np.maximum(overlap, 0.0)
        
        # Force direction vector (normalized diffs)
        # Handle division by zero for very close points
        safe_dists = np.maximum(dists, 1e-8)
        directions = diffs / safe_dists[:, :, np.newaxis] # (n, n, 2)
        
        # Repulsive force proportional to overlap
        # F = overlap * repulsion_strength
        # This creates a spring-like force pushing circles apart
        force_mag = overlap * repulsion_strength
        
        # Sum forces for each circle
        # forces[i] += sum_j (force_mag[i,j] * directions[i,j])
        # Note: direction[i,j] is from i to j. We want to push i away from j.
        # Actually, diffs[i,j] = center[i] - center[j]. 
        # If i overlaps j, we want to move i in direction (center[i] - center[j]).
        # So direction is correct.
        
        forces += np.sum(force_mag[:, :, np.newaxis] * directions, axis=1)
        
        # 3. Boundary Forces
        # Push circles away from walls if they are too close (overlap with wall)
        # Wall constraint: x >= r, x <= 1-r => x-r >= 0, 1-r-x >= 0
        
        # Left wall (x < r)
        left_overlap = radii - centers[:, 0]
        left_overlap = np.maximum(left_overlap, 0.0)
        forces[:, 0] += left_overlap * repulsion_strength
        
        # Right wall (x > 1-r)
        right_overlap = centers[:, 0] - (1.0 - radii)
        right_overlap = np.maximum(right_overlap, 0.0)
        forces[:, 0] -= right_overlap * repulsion_strength
        
        # Bottom wall (y < r)
        bottom_overlap = radii - centers[:, 1]
        bottom_overlap = np.maximum(bottom_overlap, 0.0)
        forces[:, 1] += bottom_overlap * repulsion_strength
        
        # Top wall (y > 1-r)
        top_overlap = centers[:, 1] - (1.0 - radii)
        top_overlap = np.maximum(top_overlap, 0.0)
        forces[:, 1] -= top_overlap * repulsion_strength
        
        # 4. Update Centers
        # Adaptive learning rate
        lr = learning_rate_init * (learning_rate_min / learning_rate_init) ** (it / num_iterations)
        
        # Clip forces to prevent explosion
        max_force = 1.0
        forces = np.clip(forces, -max_force, max_force)
        
        centers += lr * forces
        
        # Clamp centers to [0, 1] strictly to prevent divergence, 
        # though forces should handle it. 
        # Actually, keeping them inside is good practice.
        centers[:, 0] = np.clip(centers[:, 0], 0, 1)
        centers[:, 1] = np.clip(centers[:, 1], 0, 1)
        
        # Optional: If forces are very small, we might want to increase growth?
        # But simple steady growth works.

    # --- Final Validation and Cleanup ---
    # Ensure no negative radii
    radii = np.maximum(radii, 0.0)
    
    # Final strict clamping of centers relative to radii
    # If a circle is cut off by boundary, shrink it or move it?
    # The problem requires circles to be inside.
    # If center is at 0.05 and r=0.1, it's outside.
    # We should adjust centers to be valid.
    # A simple fix: shift center to be at least r from wall.
    
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        
        # Adjust x
        if x - r < 0:
            x = r
        if x + r > 1:
            x = 1 - r
        # Adjust y
        if y - r < 0:
            y = r
        if y + r > 1:
            y = 1 - r
            
        centers[i] = [x, y]
        
        # If even after shifting, r is too large (e.g. r > 0.5), clamp r
        if r > 0.5:
            radii[i] = 0.5
            # Re-adjust center to 0.5
            centers[i] = [0.5, 0.5] # Or keep valid x,y

    # One last pass to resolve any remaining tiny overlaps due to discrete steps
    # Simple iterative relaxation without growing
    for _ in range(100):
        forces = np.zeros_like(centers)
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists_sq = np.sum(diffs**2, axis=2)
        np.fill_diagonal(dists_sq, np.inf)
        dists = np.sqrt(dists_sq)
        radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = np.maximum(radii_sum - dists, 0.0)
        safe_dists = np.maximum(dists, 1e-8)
        directions = diffs / safe_dists[:, :, np.newaxis]
        force_mag = overlap * repulsion_strength
        forces += np.sum(force_mag[:, :, np.newaxis] * directions, axis=1)
        
        # Boundary forces
        forces[:, 0] += np.maximum(radii - centers[:, 0], 0.0) * repulsion_strength
        forces[:, 0] -= np.maximum(centers[:, 0] - (1.0 - radii), 0.0) * repulsion_strength
        forces[:, 1] += np.maximum(radii - centers[:, 1], 0.0) * repulsion_strength
        forces[:, 1] -= np.maximum(centers[:, 1] - (1.0 - radii), 0.0) * repulsion_strength
        
        centers += 0.01 * forces
        centers[:, 0] = np.clip(centers[:, 0], 0, 1)
        centers[:, 1] = np.clip(centers[:, 1], 0, 1)

    # Final strict adjustment for boundaries
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
        # If r is too big for the square (r > 0.5), it's impossible to fit.
        # But with 26 circles, r will be small.

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
