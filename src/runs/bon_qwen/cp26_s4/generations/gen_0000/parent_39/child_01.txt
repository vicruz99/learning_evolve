import numpy as np
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')


def compute_penalty(vars, n, weight):
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]

    penalty = 0.0

    # Overlap penalty (quadratic)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            min_dist = radii[i] + radii[j]
            if dist_sq < min_dist * min_dist:
                dist = np.sqrt(dist_sq)
                overlap = min_dist - dist
                penalty += overlap * overlap

    # Boundary penalty
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            penalty += (r - x) ** 2
        if y < r:
            penalty += (r - y) ** 2
        if x > 1 - r:
            penalty += (x - (1 - r)) ** 2
        if y > 1 - r:
            penalty += (y - (1 - r)) ** 2

    # Negative radius penalty
    for i in range(n):
        if radii[i] < 0:
            penalty += 100 * radii[i] ** 2

    return weight * penalty


def objective_func(vars, n, weight):
    radii = vars[2 * n:]
    base = -np.sum(radii)
    pen = compute_penalty(vars, n, weight)
    return base + pen


def validate_solution(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r - 1e-9 or y < r - 1e-9 or x > 1 - r + 1e-9 or y > 1 - r + 1e-9:
            return False
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True


def try_optimization(centers_init, n, init_r, bounds, penalty_weights):
    radii_init = np.ones(n) * init_r
    vars_init = np.concatenate([centers_init.flatten(), radii_init])

    best_local_sum = 0
    best_local_centers = None
    best_local_radii = None

    for pw in penalty_weights:
        result = minimize(
            objective_func,
            vars_init.copy(),
            args=(n, pw),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50000, 'ftol': 1e-15, 'gtol': 1e-12}
        )

        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]

        if validate_solution(centers_opt, radii_opt, n):
            s = np.sum(radii_opt)
            if s > best_local_sum:
                best_local_sum = s
                best_local_centers = centers_opt.copy()
                best_local_radii = radii_opt.copy()

    # Also try with lower penalty starting from previous best
    if best_local_centers is not None:
        for pw in [50, 100, 200]:
            vars_refine = np.concatenate([best_local_centers.flatten(), best_local_radii])
            result = minimize(
                objective_func,
                vars_refine,
                args=(n, pw),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 30000, 'ftol': 1e-15}
            )
            centers_refine = result.x[:2 * n].reshape(n, 2)
            radii_refine = result.x[2 * n:]
            if validate_solution(centers_refine, radii_refine, n):
                s = np.sum(radii_refine)
                if s > best_local_sum:
                    best_local_sum = s
                    best_local_centers = centers_refine.copy()
                    best_local_radii = radii_refine.copy()

    return best_local_centers, best_local_radii, best_local_sum
