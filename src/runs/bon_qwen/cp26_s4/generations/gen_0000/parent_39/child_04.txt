import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_hex_grid_centers(n, width=1.0, height=1.0):
    """
    Generates n centers in a hexagonal grid pattern inside [0, width] x [0, height].
    """
    centers = []
    # Estimate row height for hex packing: sqrt(3) * spacing
    # We want to fit n circles. Let's guess a spacing.
    # Area per circle approx 2*sqrt(3)*r^2. 
    # n * 2*sqrt(3)*r^2 approx 1 => r approx sqrt(1/(2*sqrt(3)*n))
    # spacing = 2r
    r_est = np.sqrt(1.0 / (2 * np.sqrt(3) * n))
    spacing = 2 * r_est * 1.2  # Add some margin
    
    y = spacing
    row = 0
    while len(centers) < n:
        x = spacing if row % 2 == 1 else 0  # Shift odd rows
        while x < width - spacing + 1e-9 and len(centers) < n:
            # Adjust x to be within [0, 1]
            cx = x
            cy = y
            if 0 <= cx <= width and 0 <= cy <= height:
                centers.append([cx, cy])
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
        
    return np.array(centers[:n])

        # vars: [x1, y1, r1, ..., xn, yn, rn]
        # We want to maximize sum of radii, so minimize negative sum
        radii_part = vars[2::3]
        return -np.sum(radii_part)

    def penalty_function(vars, lam):
        # vars structure: every 3 elements are [x, y, r] for a circle
        centers_opt = vars[0::3], vars[1::3]
        radii_opt = vars[2::3]
        
        c_x = np.array(centers_opt[0])
        c_y = np.array(centers_opt[1])

            r - c_x,          # Left

        return penalty
