import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum_r = 0.0
    best_centers = None
    best_radii = None

    # Parameters for the force-directed layout
    num_iterations = 2000
    initial_r = 0.01
    r_growth_rate = 1.5e-4
    learning_rate = 0.05
    damping = 0.9
    
    # Number of random restarts to avoid local minima
    n_restarts = 15

    for restart in range(n_restarts):
        # 1. Initialization
        np.random.seed(restart * 100 + 42)
        centers = np.random.rand(n, 2)
        # Ensure centers are valid for a very small radius initially
        r = initial_r
        
        # Velocity for momentum
        velocities = np.zeros_like(centers)

        # 2. Expansion and Repulsion Loop
        for _ in range(num_iterations):
            forces = np.zeros_like(centers)
            
            # Calculate pair-wise repulsion forces
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    
                    # If overlapping or too close (threshold for stability)
                    overlap_threshold = 2 * r + 1e-4
                    if dist < overlap_threshold and dist > 1e-9:
                        # Repulsive force magnitude proportional to overlap
                        # Force pushes i away from j
                        force_mag = (overlap_threshold - dist) * 10.0
                        forces[i] += (diff / dist) * force_mag
                        forces[j] -= (diff / dist) * force_mag
            
            # Calculate boundary repulsion forces
            for i in range(n):
                # Left wall
                if centers[i, 0] < r:
                    forces[i, 0] += (r - centers[i, 0]) * 20.0
                # Right wall
                if centers[i, 0] > 1 - r:
                    forces[i, 0] -= (centers[i, 0] - (1 - r)) * 20.0
                # Bottom wall
                if centers[i, 1] < r:
                    forces[i, 1] += (r - centers[i, 1]) * 20.0
                # Top wall
                if centers[i, 1] > 1 - r:
                    forces[i, 1] -= (centers[i, 1] - (1 - r)) * 20.0

            # Update velocities with damping
            velocities = velocities * damping + learning_rate * forces
            # Update centers
            centers += velocities

            # Clip centers to stay within [0, 1] to prevent divergence during dynamics
            # The boundary forces above should ideally keep them in [r, 1-r], 
            # but clipping adds safety.
            centers = np.clip(centers, 0, 1)

            r += r_growth_rate

            if r > 0.12: 
                break

        # We assume equal radii for the polish step for stability.
        
        def objective(vars):
            # vars: [x1, y1, ..., x26, y26, r]
            cx = vars[:2*n]
            cy = vars[2*n:3*n]
            r_val = vars[-1]
            centers_opt = np.column_stack((cx, cy))
            
            penalty = 0.0
            
            # Boundary penalties
            for i in range(n):
                x, y = centers_opt[i]
                if x < r_val: penalty += (r_val - x)**2
                if x > 1 - r_val: penalty += (x - (1 - r_val))**2
                if y < r_val: penalty += (r_val - y)**2
                if y > 1 - r_val: penalty += (y - (1 - r_val))**2
                
            # Overlap penalties
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.linalg.norm(centers_opt[i] - centers_opt[j])
                    min_dist = 2 * r_val
                    if dist < min_dist:
                        penalty += (min_dist - dist)**2
            
            # We want to maximize r, so minimize -r + penalty
            # Weight penalty heavily
            return -r_val + 100 * penalty

        initial_vars = np.concatenate([centers.flatten(), [r]])
        
        # Bounds for centers [0, 1], r [0, 0.5]
        bounds = [(0, 1)] * (2*n) + [(0, 0.5)]
        
        # Use L-BFGS-B for bound constraints
        res = minimize(objective, initial_vars, method='L-BFGS-B', bounds=bounds,

        final_vars = res.x

        final_r = final_vars[-1]

        # Check boundaries

            if x < final_r - tol or x > 1 - final_r + tol or y < final_r - tol or y > 1 - final_r + tol:
                valid = False
                break

                if not valid: break
