import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, sum_radii) for a packing of 26 circles in a unit square.
    """
    n = 26
    
    def objective(p):
        """Negate the sum of radii for minimization."""
        return -np.sum(p[2::3])

    def constraints(p):
        """
        Define constraints for centers and radii.
        p structure: [x0, y0, r0, x1, y1, r1, ...]
        """
        con = []
        
        # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
        for i in range(n):
            idx = 3 * i
            x, y, r = p[idx], p[idx+1], p[idx+2]
            con.append({'type': 'ineq', 'fun': lambda p, i=i: p[3*i+2]}) # r >= 0
            con.append({'type': 'ineq', 'fun': lambda p, i=i: p[3*i] - p[3*i+2]}) # x - r >= 0
            con.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[3*i] - p[3*i+2]}) # 1 - x - r >= 0
            con.append({'type': 'ineq', 'fun': lambda p, i=i: p[3*i+1] - p[3*i+2]}) # y - r >= 0
            con.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[3*i+1] - p[3*i+2]}) # 1 - y - r >= 0

        # Pairwise non-overlap constraints: dist >= r1 + r2
        # Using squared distance for smoother optimization
        # (x1-x2)^2 + (y1-y2)^2 >= (r1+r2)^2
        for i in range(n):
            for j in range(i + 1, n):
                idx_i, idx_j = 3 * i, 3 * j
                x_i, y_i, r_i = p[idx_i], p[idx_i+1], p[idx_i+2]
                x_j, y_j, r_j = p[idx_j], p[idx_j+1], p[idx_j+2]
                
                # We define a constraint function that returns the margin
                # dist^2 - (r1+r2)^2 >= 0
                # Note: We define it inside loop or use a closure-like structure for scipy
                # Since we need to pass specific indices, we use a lambda with default args
                
                def make_con(i_idx, j_idx):
                    return {'type': 'ineq', 'fun': lambda p: ((p[3*i_idx] - p[3*j_idx])**2 + 
                                                               (p[3*i_idx+1] - p[3*j_idx+1])**2 - 
                                                               (p[3*i_idx+2] + p[3*j_idx+2])**2)}
                con.append(make_con(i, j))
        
        return con

    def get_bounds():
        bounds = []
        for _ in range(n):
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((0, 0.5)) # r
        return bounds

    # Helper to create initial guesses
    def create_initial_guess(type):
        x = np.linspace(0.1, 0.9, 5)
        y = np.linspace(0.1, 0.9, 5)
        pts = []
        
        if type == 'grid_5x5_plus':
            # 5x5 grid
            for yi in y:
                for xi in x:
                    pts.append([xi, yi])
            # 26th point in center
            pts.append([0.5, 0.5])
        elif type == 'hex':
            # Hexagonal packing attempt
            # 5 rows, varying number of points
            pts = []
            row_y = [0.15, 0.33, 0.5, 0.67, 0.85]
            # Try to fit 26 points
            # 5, 5, 6, 5, 5
            counts = [5, 5, 6, 5, 5]
            current_y_idx = 0
            pts_added = 0
            for count in counts:
                ys = row_y[current_y_idx]
                current_y_idx += 1
                # Shift every other row
                xs = np.linspace(0.15, 0.85, count)
                if pts_added % 2 == 1: # Shifted rows
                     xs = xs + (0.85-0.15)/ (count*2) # approximate shift
                     xs = np.clip(xs, 0.05, 0.95)
                for xi in xs:
                    pts.append([xi, ys])
                pts_added += 1
        
        # If not enough points, fill with random
        while len(pts) < 26:
            pts.append([np.random.rand(), np.random.rand()])
        
        pts = np.array(pts[:26])
        
        p0 = np.zeros(78)
        for i in range(26):
            p0[3*i] = pts[i, 0]
            p0[3*i+1] = pts[i, 1]
            p0[3*i+2] = 0.09 # Initial radius estimate
            
        return p0

    # --- Stage 1: Optimize Equal Radii Baseline ---
    best_p = None
    best_obj = -np.inf
    
    # Try a few initial configurations for equal radii
    for guess_type in ['grid_5x5_plus', 'hex']:
        p0 = create_initial_guess(guess_type)
        # Set all radii equal for baseline
        radii_val = 0.09
        for i in range(26):
            p0[3*i+2] = radii_val
            
        # Fix radii to be equal for this stage? 
        # Actually, let's just optimize centers for a fixed r, or optimize both.
        # Let's optimize both with the general constraints, starting from equal radii.
        
        res = minimize(objective, p0, method='SLSQP', bounds=get_bounds(), 
                       constraints=constraints(p0), options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success or (-res.fun > best_obj):
            best_obj = -res.fun
            best_p = res.x
            
    # --- Stage 2: Refine with Unequal Radii ---
    # Run multiple perturbations from the best found configuration
    final_best_p = best_p
    final_best_obj = best_obj
    
    for _ in range(5): # 5 local searches
        p_start = best_p.copy()
        # Perturb radii and centers slightly
        p_start[2::3] += np.random.normal(0, 0.005, 26)
        p_start[0::3] += np.random.normal(0, 0.01, 26)
        p_start[1::3] += np.random.normal(0, 0.01, 26)
        
        # Clip to valid range
        p_start[0::3] = np.clip(p_start[0::3], 0.01, 0.99)
        p_start[1::3] = np.clip(p_start[1::3], 0.01, 0.99)
        p_start[2::3] = np.clip(p_start[2::3], 0.001, 0.49)
        
        res = minimize(objective, p_start, method='SLSQP', bounds=get_bounds(), 
                       constraints=constraints(p_start), options={'maxiter': 500, 'ftol': 1e-9})
        
        if -res.fun > final_best_obj:
            final_best_obj = -res.fun
            final_best_p = res.x
            
    # Extract results
    centers = np.column_stack((final_best_p[0::3], final_best_p[1::3]))
    radii = final_best_p[2::3]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
