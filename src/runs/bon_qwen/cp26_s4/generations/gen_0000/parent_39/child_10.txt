   - Subject to: r_i ≤ min(x_i, 1-x_i, y_i, 1-y_i) [boundary constraints]
   - Subject to: r_i + r_j ≤ distance(i,j) [non-overlap constraints]

3. **Phase 3 - Position Optimization**: Use coordinate descent with simulated annealing to iteratively improve center positions:
   - For each circle, try perturbing its position
   - Accept moves that improve the sum of radii
   - Use temperature cooling to escape local minima
   - Run multiple iterations with decreasing step sizes

4. **Multiple Restarts**: Try many different initial configurations (varying seeds, row arrangements) and keep the best result.

Key insight: The LP formulation is elegant because for fixed centers, the radius optimization is a simple LP with linear constraints. The challenge is optimizing the centers, which we do heuristically.

import numpy as np
from scipy.optimize import linprog
import random


def solve_radii_lp(centers, n=26):
    """Solve LP to maximize sum of radii given fixed centers."""
    c = -np.ones(n)
    A = []
    b = []

    # Boundary: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        mr = min(x, 1.0 - x, y, 1.0 - y)
        if mr < 0:
            mr = 0
        row = np.zeros(n)
        row[i] = 1.0
        A.append(row)
        b.append(mr)

    # Non-overlap: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            b.append(d)

    A = np.array(A)
    b = np.array(b)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0


def make_hex_grid(rows_config, n=26):
    """Create hexagonal grid placement given row configurations."""
    pts = []
    total_in_grid = sum(rows_config)
    actual_rows = []
    remaining = n
    for rc in rows_config:
        actual_rows.append(min(rc, remaining))
        remaining -= actual_rows[0] if len(actual_rows) == 1 else actual_rows[-1] - (sum(actual_rows[:-1]) if len(actual_rows) > 1 else 0)
        if remaining <= 0:
            break

    # Recalculate properly
    actual_rows = []
    remaining = n
    for rc in rows_config:
        take = min(rc, remaining)
        actual_rows.append(take)
        remaining -= take
        if remaining <= 0:
            break

    # Calculate spacing
    n_rows = len(actual_rows)
    max_cols = max(actual_rows)
    y_margin = 0.05
    x_margin = 0.05

    y_spacing = (1.0 - 2 * y_margin) / max(n_rows - 1, 1)
    x_spacing = (1.0 - 2 * x_margin) / max(max_cols - 1, 1)

    for row_idx, n_circles in enumerate(actual_rows):
        y = y_margin + row_idx * y_spacing
        for col_idx in range(n_circles):
            x = x_margin + col_idx * x_spacing
            if row_idx % 2 == 1:
                x += x_spacing / 2.0
            pts.append([x, y])

    return np.array(pts[:n])


def make_random_grid(n=26):
    """Create a random initial placement."""
    rng = np.random.RandomState()
    pts = []
    for i in range(n):
        x = rng.uniform(0.1, 0.9)
        y = rng.uniform(0.1, 0.9)
        pts.append([x, y])
    return np.array(pts)


def improve_positions_sa(centers, n=26, iters=80, init_step=0.02, temp=0.005):
    """Improve positions using simulated annealing with LP-based radius optimization."""
    best = centers.copy()
    rad, best_sum = solve_radii_lp(best, n)
    step = init_step
    current_temp = temp

    for iteration in range(iters):
        improved = False
        n_improvements = 0

        for i in range(n):
            old = best[i].copy()
            tried = False

            for _ in range(8):
                dx = np.random.uniform(-step, step)
                dy = np.random.uniform(-step, step)
                nc = np.clip(old + np.array([dx, dy]), 0.005, 0.995)
                best[i] = nc
                rad, s = solve_radii_lp(best, n)

                if s > best_sum:
                    best_sum = s
                    improved = True
                    n_improvements += 1
                    tried = True
                else:
                    # Simulated annealing acceptance
                    if current_temp > 0 and np.random.random() < np.exp((s - best_sum) / current_temp):
                        best_sum = s
                        n_improvements += 1
                        tried = True
                    else:
                        best[i] = old

            if not tried:
                best[i] = old

        # Reduce step size and temperature
        step *= 0.95
        current_temp *= 0.96

        if not improved and iteration > 10:
            break

    rad, best_sum = solve_radii_lp(best, n)
    return best, rad, best_sum
