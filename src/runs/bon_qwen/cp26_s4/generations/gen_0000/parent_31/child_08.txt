import numpy as np
from scipy.optimize import minimize
import math

def get_constraints(n):
    """
    Returns a list of constraint dictionaries for the SLSQP optimizer.
    Constraints ensure circles are inside the unit square and do not overlap.
    """
    constraints = []

    # 1. Boundary constraints: 0 <= x, y <= 1 and r <= min(x, 1-x, y, 1-y)
    # In SLSQP, inequality constraints are of the form fun(x) >= 0.
    # So we need: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    
    for i in range(n):
        # x coordinate index: 2*i, y coordinate index: 2*i + 1
        # radius index: 2*n + i
        
        # x - r >= 0
        def make_bound_x_minus_r(i):
            return lambda params: params[2*i] - params[2*n + i]
        
        # 1 - x - r >= 0
        def make_bound_1_minus_x_minus_r(i):
            return lambda params: 1.0 - params[2*i] - params[2*n + i]
            
        # y - r >= 0
        def make_bound_y_minus_r(i):
            return lambda params: params[2*i + 1] - params[2*n + i]
            
        # 1 - y - r >= 0
        def make_bound_1_minus_y_minus_r(i):
            return lambda params: 1.0 - params[2*i + 1] - params[2*n + i]

        constraints.append({'type': 'ineq', 'fun': make_bound_x_minus_r(i)})
        constraints.append({'type': 'ineq', 'fun': make_bound_1_minus_x_minus_r(i)})
        constraints.append({'type': 'ineq', 'fun': make_bound_y_minus_r(i)})
        constraints.append({'type': 'ineq', 'fun': make_bound_1_minus_y_minus_r(i)})

    # 2. Non-overlap constraints: dist(c_i, c_j) - (r_i + r_j) >= 0
    for i in range(n):
        for j in range(i + 1, n):
            def make_overlap_constraint(i, j):
                return lambda params: math.sqrt(
                    (params[2*i] - params[2*j])**2 + 
                    (params[2*i + 1] - params[2*j + 1])**2
                ) - (params[2*n + i] + params[2*n + j])
            
            constraints.append({'type': 'ineq', 'fun': make_overlap_constraint(i, j)})

    return constraints

def generate_hexagonal_initialization(n):

    # Approximate radius for 26 circles
    r_init = 0.09 
    
    y = r_init

        x_start = r_init if row_idx % 2 == 0 else r_init + r_init # shift by r

        # So odd rows start at x = 2r?

            x = r_init

def run_packing():
    n = 26
    
    # 1. Initialize centers
    centers_init = generate_hexagonal_initialization(n)
    
    # 2. Initialize radii
    # Start with a small radius to ensure valid start
    radii_init = np.full(n, 0.05)
    
    # 3. Formulate the optimization problem
    # Variables: [x0, y0, x1, y1, ..., x25, y25, r0, r1, ..., r25]
    # Total 2*n + n = 3*n variables
    
    initial_params = np.concatenate([centers_init.flatten(), radii_init])
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5] (max possible radius in unit square)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
    for i in range(n):
        bounds.append((0.0, 0.5)) # r

    def objective(params):
        radii = params[2*n:]
        return -np.sum(radii)

    # Get constraints
    constraints = get_constraints(n)
    
    # 4. Run optimization
    # Using SLSQP which supports constraints
    result = minimize(
        objective,
        initial_params,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )

    best_params = result.x
    best_centers = best_params[:2*n].reshape(n, 2)
    best_radii = best_params[2*n:]
    
    # Ensure radii are non-negative (though bounds enforce this)
    best_radii = np.maximum(best_radii, 0.0)
    
    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
