import numpy as np
from scipy.optimize import minimize

def generate_initial_hex_packing(n=26):
    """
    Generates an initial hexagonal packing for n circles.
    Returns centers and radii.
    """
    centers = []
    # Try to fit rows. For 26, a pattern like 5, 6, 5, 6, 4 works well in hex lattice.
    # Row lengths:
    row_lengths = [5, 6, 5, 6, 4]
    if sum(row_lengths) != n:
        # Fallback to grid if pattern doesn't match exactly (unlikely here)
        row_lengths = []
        rows = 5
        for i in range(rows):
            row_lengths.append(5)
        # Adjust to match n
        while sum(row_lengths) < n:
            row_lengths.append(5)
    
    current_n = 0
    r_init = 0.07 # Small initial radius to ensure fit
    sqrt3 = np.sqrt(3)
    
    for i, count in enumerate(row_lengths):
        y = r_init + i * sqrt3 * r_init * 2 # Vertical spacing sqrt(3)*diameter/2? No, sqrt(3)*r
        # Correct vertical spacing for hex packing is sqrt(3)*r (distance between rows)
        # Wait, distance between centers is sqrt(3)*r?
        # Triangle with side 2r. Height is sqrt(3)r. Yes.
        y = r_init + i * sqrt3 * r_init 
        
        # Horizontal offset for staggered rows
        offset = 0
        if i % 2 == 1:
            offset = r_init # Shift by r (half diameter)
        
        for j in range(count):
            x = r_init + offset + j * 2 * r_init
            centers.append([x, y])
            current_n += 1
            if current_n >= n:
                break
        if current_n >= n:
            break
            
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    return centers, radii

def solve_packing(centers, radii):
    """
    Optimizes the packing to maximize sum of radii.
    """
    n = len(radii)
    
    # Flatten variables: x0, y0, r0, x1, y1, r1, ...
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Objective: maximize sum of radii -> minimize negative sum
    def objective(x):
        r = x[2::3]
        return -np.sum(r)
    
    # Constraints
    constraints = []
    
    # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # x_i >= r_i  => x_i - r_i >= 0
    # 1 - x_i >= r_i => 1 - x_i - r_i >= 0
    # Same for y
    
    for i in range(n):
        idx = i * 3
        xi, yi, ri = x0[idx], x0[idx+1], x0[idx+2]
        
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*3] - x[i*3+2]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[i*3] - x[i*3+2]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*3+1] - x[i*3+2]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[i*3+1] - x[i*3+2]})
        
        # r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*3+2]})

    # 2. Non-overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = i * 3
            idx_j = j * 3
            
            def non_overlap(x, i=i, j=j, idx_i=idx_i, idx_j=idx_j):
                xi, yi, ri = x[idx_i], x[idx_i+1], x[idx_i+2]
                xj, yj, rj = x[idx_j], x[idx_j+1], x[idx_j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                rad_sum = ri + rj
                return dist_sq - rad_sum**2
            
            constraints.append({'type': 'ineq', 'fun': non_overlap})

    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5] (since diameter <= 1)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Optimization
    # SLSQP is suitable for constrained nonlinear optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
    
    if res.success:
        opt_x = res.x
    else:
        # If failed, try to use the best point found or fallback
        opt_x = res.x

    # Extract results
    centers_opt = np.zeros((n, 2))
    radii_opt = np.zeros(n)
    for i in range(n):
        idx = i * 3
        centers_opt[i] = [opt_x[idx], opt_x[idx+1]]
        radii_opt[i] = opt_x[idx+2]
        
    return centers_opt, radii_opt, np.sum(radii_opt)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Main function to run the packing optimization.
    """
    # Generate initial configuration
    centers, radii = generate_initial_hex_packing(26)
