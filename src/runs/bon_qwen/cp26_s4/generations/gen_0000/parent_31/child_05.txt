import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None

    def get_initial_params():
        """
        Generates initial parameters for centers and radii using a hexagonal-like grid.
        """
        # Hexagonal grid layout parameters
        spacing_x = 1.0 / 6.0
        spacing_y = spacing_x * np.sqrt(3) / 2.0
        
        centers = []
        r_init = 0.01
        
        # Generate grid points
        row = 0
        col = 0
        while len(centers) < n:
            cx = (col + 0.5) * spacing_x + (0.5 * spacing_x * (row % 2))
            cy = (row + 0.5) * spacing_y
            
            if 0 <= cx <= 1 and 0 <= cy <= 1:
                centers.append([cx, cy])
            col += 1
            if col > 6:
                col = 0
                row += 1
        
        centers = np.array(centers[:n])
        radii = np.full(n, r_init)
        
        # Pack into vector: [x1...xn, y1...yn, r1...rn]
        params = np.concatenate([centers[:, 0], centers[:, 1], radii])
        return params, centers, radii

    def objective_function(params):
        """
        Objective to minimize: -sum(radii) + penalties for violations.
        """
        x = params[:n]
        y = params[n:2*n]
        r = params[2*n:]
        
        # Primary objective: maximize sum of radii
        val = -np.sum(r)
        
        # Penalty weight (large enough to enforce constraints)
        mu = 2000.0
        
        # 1. Boundary Penalties
        # Constraints: r <= x <= 1-r  and  r <= y <= 1-r
        # Violations are squared to ensure smoothness and differentiability
        p_left = np.maximum(0, r - x)
        p_right = np.maximum(0, r + x - 1)
        p_down = np.maximum(0, r - y)
        p_up = np.maximum(0, r + y - 1)
        
        boundary_pen = np.sum(p_left**2) + np.sum(p_right**2) + np.sum(p_down**2) + np.sum(p_up**2)
        
        # 2. Overlap Penalties
        # Constraint: dist^2 >= (r_i + r_j)^2
        # Violation: max(0, (r_i + r_j)^2 - dist^2)^2
        # Using squared distance to avoid sqrt and ensure smooth gradients
        
        # Compute pairwise squared distances efficiently
        dx = x[:, None] - x[None, :]
        dy = y[:, None] - y[None, :]
        dist_sq = dx**2 + dy**2
        
        # Radius sums
        r_sum = r[:, None] + r[None, :]
        
        # Calculate violations
        # (r_i + r_j)^2 - dist_sq. If positive, circles overlap.
        violations = r_sum**2 - dist_sq
        
        # Zero out diagonal (self-distance is 0, but self-overlap check is irrelevant)
        # Actually, dist_sq diagonal is 0, r_sum diagonal is 2r.
        # Violation would be 4r^2. We don't want to penalize a circle overlapping itself.
        np.fill_diagonal(violations, 0)
        
        # Only penalize positive violations (overlaps)
        overlap_pen = np.sum(np.maximum(0, violations)**2)
        
        return val + mu * (boundary_pen + overlap_pen)

    # Define bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5] (radius cannot exceed 0.5 in unit square)
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    # Run optimization multiple times to find the best global minimum
    for _ in range(10):
        params, centers, radii = get_initial_params()
        
        # Add slight random noise to avoid symmetric local minima
        noise = np.random.uniform(-0.01, 0.01, size=params.shape)
        # Ensure noise keeps bounds valid roughly
        noise[:2*n] = np.clip(noise[:2*n], -0.05, 0.05) 
        params += noise
        
        # Optimize
        result = minimize(objective_function, params, method='L-BFGS-B', bounds=bounds)
        
        if result.success:
            final_params = result.x
            final_x = final_params[:n]
            final_y = final_params[n:2*n]
            final_r = final_params[2*n:]
            
            # Validate and fix small violations by shrinking radii slightly
            # Check boundary constraints
            min_margin = np.min(np.array([
                final_x - final_r,
                1 - final_x - final_r,
                final_y - final_r,
                1 - final_y - final_r
            ]), axis=0)
            
            # Check overlap constraints
            # dist >= r_i + r_j
            # dist^2 >= (r_i + r_j)^2
            # If violation, we need to reduce r.
            # A simple robust way is to scale down all radii by a safety factor if needed,
            # or just rely on the optimizer's convergence.
            
            # Let's check strict validity
            valid = True
            
            # Boundary check
            for i in range(n):
                if (final_x[i] - final_r[i] < -1e-9 or 
                    final_x[i] + final_r[i] > 1 + 1e-9 or
                    final_y[i] - final_r[i] < -1e-9 or 
                    final_y[i] + final_r[i] > 1 + 1e-9):
                    valid = False
                    break
            
            if valid:
                # Overlap check
                for i in range(n):
                    for j in range(i + 1, n):
                        dx = final_x[i] - final_x[j]
                        dy = final_y[i] - final_y[j]
                        dist = np.sqrt(dx*dx + dy*dy)
                        if dist < final_r[i] + final_r[j] - 1e-9:
                            valid = False
                            break
                    if not valid:
                        break
            
            if valid:
                current_sum = np.sum(final_r)
                if current_sum > best_sum_r:
                    best_sum_r = current_sum
                    best_centers = np.column_stack((final_x, final_y))
                    best_radii = final_r
            else:
                # If invalid, shrink radii slightly to satisfy constraints and store
                # This is a fallback, ideally we want a valid one from optimizer
                # But let's try to fix it
                shrink_factor = 1.0 - 1e-5
                final_r *= shrink_factor
                current_sum = np.sum(final_r)
                if current_sum > best_sum_r:
                    best_sum_r = current_sum
                    best_centers = np.column_stack((final_x, final_y))
                    best_radii = final_r

    return best_centers, best_radii, best_sum_r
