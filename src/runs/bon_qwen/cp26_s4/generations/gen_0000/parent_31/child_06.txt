import numpy as np
from scipy.optimize import linprog, minimize
import time

def solve_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximizes sum(radii) subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    
    # Calculate distances to boundaries: min(x, 1-x, y, 1-y)
    x, y = centers[:, 0], centers[:, 1]
    d_boundary = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    
    # Prepare constraints for linprog
    # Constraints: r_i + r_j <= dist_ij  (for all i < j)
    #              r_i <= d_boundary_i   (for all i)
    # Variables: r_0, ..., r_{n-1}
    
    # Number of pairwise constraints
    n_pairs = n * (n - 1) // 2
    n_constraints = n_pairs + n
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    # Pairwise constraints
    row = 0
    # Precompute squared differences for distance calculation
    # Vectorized distance calculation
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dists[i, j]
            row += 1
            
    # Boundary constraints
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = d_boundary[i]
        row += 1
        
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c_obj = np.ones(n) * -1.0
    
    # Bounds: r_i >= 0
    bounds = [(0, None)] * n
    
    try:
        # Use 'highs' method if available (faster), else 'simplex'
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    # Fallback
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='simplex')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass

    # If all else fails, return zeros
    return np.zeros(n), 0.0

def objective_function(centers_flat):
    """
    Objective function for the outer optimization.
    Minimizes negative sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    radii, sum_r = solve_radii(centers)
    return -sum_r

def generate_hexagonal_grid(n, margin=0.05):
    """
    Generates a hexagonal grid of n points inside [margin, 1-margin].
    """
    points = []
    # Estimate spacing
    # Area per point ~ 1/n. Spacing ~ sqrt(1/n).
    # Hexagonal packing density factor sqrt(3)/2 ~ 0.866
    # Rough estimate for spacing s: s^2 * sqrt(3)/2 approx 1/n => s approx sqrt(2/(n*sqrt(3)))
    # Let's just iterate rows
    
    s_x = 0.15 # Horizontal spacing estimate
    s_y = s_x * np.sqrt(3) / 2 # Vertical spacing
    
    # We need to fit n points.
    # Try to fill rows
    rows = []
    y = margin
    while len(points) < n:
        row = []
        x = margin
        while x <= 1 - margin:
            row.append([x, y])
            x += s_x
        rows.append(row)
        y += s_y
        
    # Flatten
    all_points = [p for row in rows for p in row]
    
    # If we have too many, trim. If too few, maybe spacing was too big.
    # For n=26, this might produce around 25-30 points.
    # We take the first n points.
    # To be more robust, let's select n points that are best spread?
    # Just taking first n is okay for initialization.
    
    if len(all_points) >= n:
        return np.array(all_points[:n])
    else:
        # If not enough, just pad or resize? 
        # Shouldn't happen with reasonable margin/spacing.
        # Fallback to random
        return np.random.rand(n, 2)

def generate_random_spread(n, steps=500):
    """
    Generates n points using a simple repulsion force simulation.
    """
    centers = np.random.rand(n, 2)
    # Repulsion simulation
    for _ in range(steps):
        forces = np.zeros_like(centers)
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff**2)
                if dist_sq < 1e-4:
                    dist_sq = 1e-4
                    # Random nudge to break symmetry
                    diff += np.random.randn(2) * 0.01
                dist = np.sqrt(dist_sq)
                force_mag = 1.0 / dist_sq
                forces[i] += diff / dist * force_mag
                forces[j] -= diff / dist * force_mag
        
        # Boundary repulsion
        for i in range(n):
            x, y = centers[i]
            if x < 0.05: forces[i, 0] += 1.0 / (x + 0.01)
            if x > 0.95: forces[i, 0] -= 1.0 / (1 - x + 0.01)
            if y < 0.05: forces[i, 1] += 1.0 / (y + 0.01)
            if y > 0.95: forces[i, 1] -= 1.0 / (1 - y + 0.01)
            
        centers += forces * 0.01
        centers = np.clip(centers, 0, 1)
        
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Collect best results
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Initial configurations to try
    initial_configs = []
    
    # 1. Hexagonal grid
    try:
        cfg = generate_hexagonal_grid(n)
        initial_configs.append(cfg)
    except:
        pass
        
    # 2. Random spread
    try:
        cfg = generate_random_spread(n, steps=200)
        initial_configs.append(cfg)
    except:
        pass
        
    # 3. Uniform grid (5x5 + 1)
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    gx, gy = np.meshgrid(grid_x, grid_y)
    grid_pts = np.column_stack((gx.ravel(), gy.ravel())) # 25 points
    # Add center
    grid_pts = np.vstack((grid_pts, [0.5, 0.5]))
    initial_configs.append(grid_pts)
    
    # Optimization loop
    # We will try to optimize each config
    # Using Powell method for derivative-free optimization
    
    max_iter = 1000 # Limit iterations to prevent timeout
    
    for i, centers in enumerate(initial_configs):
        if len(centers) != n:
            continue
            
        centers_flat = centers.flatten()
        
        # Try Powell
        try:
            res_powell = minimize(
                objective_function, 
                centers_flat, 
                method='Powell', 
                options={'maxiter': 200, 'ftol': 1e-6, 'xtol': 1e-6},
                bounds=[(0, 1)] * (2 * n)
            )
            if res_powell.fun < 0: # Valid solution found (since we minimize -sum)
                current_sum = -res_powell.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = res_powell.x.reshape(n, 2)
                    # Compute radii for best centers
                    radii, _ = solve_radii(best_centers)
                    best_radii = radii
        except Exception as e:
            print(f"Powell failed on config {i}: {e}")
            
        # Try Nelder-Mead (might be faster or find different local optima)
        try:
            res_nm = minimize(
                objective_function, 
                centers_flat, 
                method='Nelder-Mead', 
                options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-6}
            )
            # Clip results to [0, 1]
            res_nm.x = np.clip(res_nm.x, 0, 1)
            if res_nm.fun < 0:
                current_sum = -res_nm.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = res_nm.x.reshape(n, 2)
                    radii, _ = solve_radii(best_centers)
                    best_radii = radii
        except Exception as e:
            print(f"Nelder-Mead failed on config {i}: {e}")

    # If no optimization succeeded, fall back to initial config with LP
    if best_centers is None:
        # Pick the first config and solve radii
        centers = initial_configs[0]
        radii, s = solve_radii(centers)
        best_centers = centers
        best_radii = radii
        best_sum = s

    # Final validation check (just to be sure, though LP guarantees it)
    # The LP guarantees r_i + r_j <= dist and r_i <= boundary
    # So it should be valid.
    
    # One last polish: re-solve radii on best centers to ensure consistency
    radii, final_sum = solve_radii(best_centers)
    
    return best_centers, radii, final_sum
