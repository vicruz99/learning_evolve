import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a hybrid approach: 25 circles in a grid and 1 central circle.
    """
    
    # 1. Initialization: 5x5 Grid + 1 Center
    # Initial guess: Grid radius 0.09, Center radius 0.05
    r_grid_init = 0.09
    r_center_init = 0.05
    
    grid_centers = np.array([[r_grid_init + i * (2 * r_grid_init), r_grid_init + j * (2 * r_grid_init)] 
                             for i in range(5) for j in range(5)])
    center = np.array([0.5, 0.5])
    
    # Construct arrays
    centers = np.vstack([grid_centers, center])
    radii = np.array([r_grid_init] * 25 + [r_center_init])
    
    # 2. Define Optimization Objective and Constraints
    # Variables: x_0..x_25, y_0..y_25, r_0..r_25 (52 variables)
    # Constraints: 
    #   1. Boundaries: r_i <= x_i <= 1-r_i, r_i <= y_i <= 1-r_i
    #   2. Non-overlap: dist(c_i, c_j) >= r_i + r_j
    #   3. Non-negative radii
    
    def objective(x_vec):
        # Extract radii (last 26 elements)
        r = x_vec[46:]
        # Return negative sum (for minimization)
        return -np.sum(r)

    def penalty(x_vec):
        p = 0.0
        c = x_vec[:46].reshape(26, 2)
        r = x_vec[46:]
        
        # Boundary constraints
        for i in range(26):
            # x - r >= 0 -> r - x <= 0
            if r[i] - c[i, 0] > 0: p += 1e4 * (r[i] - c[i, 0])**2
            # x + r <= 1 -> x + r - 1 <= 0
            if c[i, 0] + r[i] - 1 > 0: p += 1e4 * (c[i, 0] + r[i] - 1)**2
            # y constraints
            if r[i] - c[i, 1] > 0: p += 1e4 * (r[i] - c[i, 1])**2
            if c[i, 1] + r[i] - 1 > 0: p += 1e4 * (c[i, 1] + r[i] - 1)**2
            # Non-negative radii
            if r[i] < 0: p += 1e4 * r[i]**2

        # Overlap constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((c[i] - c[j])**2))
                min_dist = r[i] + r[j]
                if dist < min_dist:
                    p += 1e4 * (min_dist - dist)**2
        
        return p

    # 3. Run Optimization
    x0 = np.hstack([centers.flatten(), radii])
    result = minimize(lambda x: objective(x) + penalty(x), x0, method='Nelder-Mead', 
                      options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})
    
    # 4. Extract Final Solution
    best_c = result.x[:46].reshape(26, 2)
    best_r = result.x[46:]
    
    # 5. Validate and Refine
    # Ensure no NaNs and valid constraints
    if np.isnan(best_c).any() or np.isnan(best_r).any():
        return np.zeros((26, 2)), np.zeros(26), 0.0

    # Ensure radii are non-negative
    best_r = np.maximum(best_r, 1e-9)
    
    # Adjust centers to be strictly within bounds if numerical error occurred
    for i in range(26):
        best_c[i, 0] = np.clip(best_c[i, 0], best_r[i], 1 - best_r[i])
        best_c[i, 1] = np.clip(best_c[i, 1], best_r[i], 1 - best_r[i])

    return best_c, best_r, np.sum(best_r)
