import numpy as np
from scipy.optimize import minimize

def compute_violation(centers, R):
    n = centers.shape[0]
    v = 0.0
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary violation penalties
    v += np.sum(np.maximum(0.0, R - x)**2)
    v += np.sum(np.maximum(0.0, x - (1.0 - R))**2)
    v += np.sum(np.maximum(0.0, R - y)**2)
    v += np.sum(np.maximum(0.0, y - (1.0 - R))**2)
    
    # Pairwise overlap violation penalties
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    i_idx, j_idx = np.tril_indices(n, -1)
    gaps = 2.0 * R - dists[i_idx, j_idx]
    v += np.sum(np.maximum(0.0, gaps)**2)
    return v

def run_packing():
    n = 26
    # Initial hexagonal layout generation
    R_init = 0.100
    dy = np.sqrt(3.0)/2.0 * 2.0 * R_init
    dx = 2.0 * R_init
    
    centers = []
    row_idx = 0
    while len(centers) < n:
        for col_idx in range(6):
            x = 0.05 + col_idx * dx + (0.5 * dx if row_idx % 2 else 0)
            y = 0.05 + row_idx * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
        row_idx += 1
    centers = np.array(centers[:n])
    
    best_R = R_init
    best_centers = centers.copy()
    R_curr = 0.095
    step = 0.0005
    
    # Continuation method to maximize R
    for _ in range(500):
        R_test = R_curr + step
        found = False
        for _ in range(5):
            c = best_centers + np.random.randn(n, 2) * 0.0005
            c = np.clip(c, R_test, 1.0 - R_test)
            bounds = [(R_test, 1.0 - R_test)] * (2 * n)
            
            res = minimize(compute_violation, c.ravel(), args=(R_test,), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            
            if res.fun < 1e-13:
                best_centers = res.x.reshape(n, 2)
                best_R = R_test
                found = True
                break
        if found:
            R_curr = R_test
        else:
            step /= 2.0
            if step < 1e-6:
                break
                
    radii = np.full(n, best_R)
    return best_centers, radii, np.sum(radii)
