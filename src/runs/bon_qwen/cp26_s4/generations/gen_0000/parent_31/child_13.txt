import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, sum_radii) for the optimal packing of 26 circles.
    """
    n_circles = 26
    
    def generate_hexagonal_start(n):
        """Generates a hexagonal grid starting configuration for n circles."""
        centers = []
        # Arrange in rows: 5, 4, 5, 4, 5, 3 = 26
        row_counts = [5, 4, 5, 4, 5, 3]
        if sum(row_counts) != n:
            # Fallback or adjust if n is different
            row_counts = [5, 5, 5, 5, 5, 1] # Just a safe fallback
            
        y_offset = 0.15
        row_y = 0.15
        for count in row_counts:
            # Calculate x start to center the row
            # Width of row with 'count' circles of diameter approx 0.2
            # Just centering them for now
            if count > 0:
                # Space them evenly in the row
                step = 0.25
                row_width = (count - 1) * step
                start_x = (1.0 - row_width) / 2.0
                
                for i in range(count):
                    x = start_x + i * step
                    centers.append([x, row_y])
            row_y += 0.2 # Vertical step
            
        return np.array(centers[:n])

    def optimize_packing(start_centers):
        n = len(start_centers)
        
        # Initial radii
        initial_radii = np.ones(n) * 0.08
        
        # Pack all variables into one vector: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.concatenate([start_centers.flatten(), initial_radii])
        
        # Bounds: x, y in [0, 1], r >= 0
        bounds = []
        for _ in range(n):
            bounds.extend([(0, 1), (0, 1), (0, 0.5)]) # x, y, r
            
        def objective(vars):
            # We want to maximize sum of radii, so minimize negative sum
            radii = vars[2::3] # Step 3: x, y, r
            return -np.sum(radii)

        def constraint_overlap(vars):
            """Distance between centers >= sum of radii"""
            centers = vars[:2*n].reshape(-1, 2)
            radii = vars[2::3]
            # Vectorized distance calculation
            # To keep memory low, we can check pairs
            dists = []
            sums_r = []
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.linalg.norm(centers[i] - centers[j])
                    dists.append(d)
                    sums_r.append(radii[i] + radii[j])
            
            # Constraint: d - (r_i + r_j) >= 0
            # We return an array of constraints. 
            # SLSQP handles inequality constraints g(x) >= 0
            return np.array(dists) - np.array(sums_r)

        # Constraints for boundary inclusion
        # x - r >= 0, 1 - x - r >= 0, etc.
        def constraint_boundary(vars):
            centers = vars[:2*n].reshape(-1, 2)
            radii = vars[2::3]
            con = []
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                con.append(x - r)       # Left
                con.append(1 - x - r)   # Right
                con.append(y - r)       # Bottom
                con.append(1 - y - r)   # Top
            return np.array(con)

        cons = [
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ]

        # Optimization
        # Using SLSQP as it handles constraints well
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
        
        if res.success or res.fun > -2.0: # Check if we got a reasonable result
            return res.x
        return x0 # Return initial if failed, or best so far

    # Strategy: Run optimization from a few different starting points
    best_result = None
    best_sum = -1.0

    starts = []
    starts.append(generate_hexagonal_start(n_circles))

    starts.append(perturbed)
    
    # Random start
    random_start = np.random.rand(n_circles, 2) * 0.8 + 0.1
    starts.append(random_start)

    for i, start in enumerate(starts):
        try:
            vars_opt = optimize_packing(start)
            centers = vars_opt[:2*n_circles].reshape(-1, 2)

            s = np.sum(radii)

            if validate_packing(centers, radii):

                    best_result = (centers, radii, s)
        except Exception:

    if best_result is None:
        # Fallback to simple grid if optimization fails
        # 5x5 grid is 25, we need 26. 
        # Just return a valid but suboptimal packing
        centers = generate_hexagonal_start(n_circles)
        radii = np.ones(n_circles) * 0.05
        return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    import numpy as np
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
