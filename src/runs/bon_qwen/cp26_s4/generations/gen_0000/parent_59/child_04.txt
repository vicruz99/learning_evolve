import numpy as np
from scipy.optimize import minimize

def objective(vars):
    # Maximize sum of radii -> Minimize negative sum
    return -np.sum(vars[2::3])

def constraint_func(vars):
    n = len(vars) // 3
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = []
    # Boundary constraints: circle inside [0,1]x[0,1]
    c.extend(x - r)          # x >= r
    c.extend(1.0 - x - r)    # x + r <= 1
    c.extend(y - r)          # y >= r
    c.extend(1.0 - y - r)    # y + r <= 1
    
    # Overlap constraints: distance^2 >= (r1 + r2)^2
    idx = np.triu_indices(n, k=1)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    c.extend(dx[idx]**2 + dy[idx]**2 - dr[idx]**2)
            
    return np.array(c)

def run_packing():
    n = 26
    
    # Initial configuration: Hexagonal lattice pattern
    pts = []
    y = 0.15
    row = 0
    while len(pts) < n:
        shift = 0.1 if row % 2 == 1 else 0.0
        x = 0.15 + shift
        while x <= 0.85 and len(pts) < n:
            pts.append([x, y])
            x += 0.2
        y += 0.17
        row += 1
        
    centers = np.array(pts[:n])
    radii = np.full(n, 0.04)
    
    # Flatten to optimization vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 3000, 'ftol': 1e-11, 'disp': False})
    
    final_centers = np.column_stack((res.x[0::3], res.x[1::3]))
    final_radii = res.x[2::3]
    final_radii = np.maximum(final_radii, 0.0) # Ensure non-negativity
    
    return final_centers, final_radii, np.sum(final_radii)
