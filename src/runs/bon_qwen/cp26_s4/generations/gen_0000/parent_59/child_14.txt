import numpy as np
from scipy.optimize import minimize
from functools import partial


def boundary_x_min(x, i, n):
    return x[2 * i] - x[2 * n + i]


def boundary_x_max(x, i, n):
    return 1.0 - x[2 * i] - x[2 * n + i]


def boundary_y_min(x, i, n):
    return x[2 * i + 1] - x[2 * n + i]


def boundary_y_max(x, i, n):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def boundary_r_min(x, i, n):
    return x[2 * n + i]


def overlap_constraint(x, i, j, n):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[2 * n + i] - x[2 * n + j]


def objective(x, n):
    radii = x[2 * n:]
    return -np.sum(radii)


def hexagonal_init(n, pattern=None):
    if pattern is None:
        pattern = [5, 6, 5, 6, 4]

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    r = 0.09
    h = np.sqrt(3) * r

    idx = 0
    for row_idx, ncols in enumerate(pattern):
        y = r + row_idx * h
        offset = (row_idx % 2) * r
        for col_idx in range(ncols):
            x = r + col_idx * 2 * r + offset
            if idx < n:
                centers[idx] = [x, y]
                radii[idx] = r
                idx += 1

    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = r * 0.3
        idx += 1

    return centers, radii


def random_init(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.06
    h = np.sqrt(3) * r
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = r + col * 2 * r + (row % 2) * r
            y = r + row * h
            centers[idx] = [x, y]
            radii[idx] = r
            idx += 1
    return centers, radii


def build_constraints(n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': partial(boundary_x_min, i=i, n=n)})
        cons.append({'type': 'ineq', 'fun': partial(boundary_x_max, i=i, n=n)})
        cons.append({'type': 'ineq', 'fun': partial(boundary_y_min, i=i, n=n)})
        cons.append({'type': 'ineq', 'fun': partial(boundary_y_max, i=i, n=n)})
        cons.append({'type': 'ineq', 'fun': partial(boundary_r_min, i=i, n=n)})

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': partial(overlap_constraint, i=i, j=j, n=n)})

    return cons


def resolve_overlaps(centers, radii, n, tolerance=1e-10):
    for _ in range(50):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                sum_r = radii[i] + radii[j]
                if dist < sum_r - tolerance:
                    reduction = sum_r - dist + tolerance
                    scale = max(0.5, 1.0 - reduction / (sum_r + 1e-15))
                    radii[i] *= scale
                    radii[j] *= scale
                    overlap_found = True

        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

        if not overlap_found:
            break
    return centers, radii


def optimize_from_init(x0, n, bounds, cons, maxiter=8000):
    result = minimize(
        objective, x0, args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={
            'maxiter': maxiter,
            'ftol': 1e-15,
            'disp': False,
            'eps': 1e-8
        }
    )
    return result.x


def run_packing():
    n = 26
    np.random.seed(42)

    patterns = [
        [5, 6, 5, 6, 4],
        [6, 5, 6, 5, 4],
        [5, 5, 6, 5, 5],
        [4, 5, 4, 5, 4, 4],
        [5, 4, 5, 4, 5, 3],
        [6, 6, 4, 5, 5],
        [5, 5, 5, 5, 6],
        [4, 4, 4, 4, 4, 4, 2],
        [5, 5, 5, 5, 4, 2],
        [3, 5, 5, 5, 5, 3],
    ]

    bounds = [(1e-8, 1 - 1e-8)] * (2 * n) + [(1e-10, 0.5)] * n
    cons = build_constraints(n)

    best_sum = 0
    best_centers = None
    best_radii = None

    all_configs = []

    for pattern in patterns:
        centers, radii = hexagonal_init(n, pattern)
        all_configs.append((centers, radii))

    for pattern in patterns:
        centers, radii = hexagonal_init(n, pattern)
        centers[0, 0] += 0.05
        centers[0, 1] -= 0.05
        all_configs.append((centers, radii))

    for trial in range(8):
        centers, radii = random_init(n)
        perturbation = np.random.normal(0, 0.02, (n, 2))
        centers = np.clip(centers + perturbation, 0.05, 0.95)
        all_configs.append((centers, radii))

    for config_idx, (centers, radii) in enumerate(all_configs):
        x0 = np.concatenate([centers.flatten(), radii])

        result = optimize_from_init(x0, n, bounds, cons, maxiter=6000)

        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
        radii_opt = np.maximum(radii_opt, 1e-10)

        centers_opt, radii_opt = resolve_overlaps(centers_opt, radii_opt, n)

        s = np.sum(radii_opt)
        if s > best_sum:
            best_sum = s
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Second round: use best solution as starting point with more iterations
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])

        for trial in range(5):
            perturbed = x0.copy()
            perturbed[:2 * n] += np.random.normal(0, 0.005, (2 * n,))
            perturbed[:2 * n] = np.clip(perturbed[:2 * n], 0.01, 0.99)
            perturbed[2 * n:] *= np.clip(1 + np.random.normal(0, 0.1, n), 0.5, 1.5)
            perturbed[2 * n:] = np.maximum(perturbed[2 * n:], 1e-10)

            result = optimize_from_init(perturbed, n, bounds, cons, maxiter=10000)

            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            radii_opt = np.maximum(radii_opt, 1e-10)

            centers_opt, radii_opt = resolve_overlaps(centers_opt, radii_opt, n)

            s = np.sum(radii_opt)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

        # Extra optimization from best without perturbation
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        result = optimize_from_init(x0, n, bounds, cons, maxiter=15000)
        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
        radii_opt = np.maximum(radii_opt, 1e-10)
        centers_opt, radii_opt = resolve_overlaps(centers_opt, radii_opt, n)
        s = np.sum(radii_opt)
        if s > best_sum:
            best_sum = s
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Final validation and cleanup
    for i in range(n):
        r = best_radii[i]
        best_centers[i, 0] = np.clip(best_centers[i, 0], r + 1e-13, 1 - r - 1e-13)
        best_centers[i, 1] = np.clip(best_centers[i, 1], r + 1e-13, 1 - r - 1e-13)

    return best_centers, best_radii, float(np.sum(best_radii))
