import numpy as np

def get_centers_radii(z):
    """
    Convert optimization variables z to centers and radii.
    z is structured as [r1, u1, v1, r2, u2, v2, ..., r26, u26, v26]
    x = r + (1 - 2r)u
    y = r + (1 - 2r)v
    """
    n = 26
    r = z[0::3]
    u = z[1::3]
    v = z[2::3]
    
    # Compute centers based on parameterization
    # This ensures x, y are within [r, 1-r] if u, v in [0, 1] and r in [0, 0.5]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    centers = np.column_stack((x, y))
    return centers, r

def get_constraints(z):
    """
    Compute non-overlap constraints.
    Returns array of values that must be >= 0.
    Constraint: dist_sq >= (r_i + r_j)^2
    => dist_sq - (r_i + r_j)^2 >= 0
    """
    centers, radii = get_centers_radii(z)
    n = centers.shape[0]
    
    # Vectorized distance squared calculation
    # centers shape: (n, 2)
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dist_sq = dx**2 + dy**2
    
    # Sum of radii squared
    rad_sum = radii[:, None] + radii[None, :]
    rad_sum_sq = rad_sum**2
    
    # Constraint values
    constraints = dist_sq - rad_sum_sq
    
    # Extract upper triangle (i < j)
    i, j = np.triu_indices(n, k=1)
    return constraints[i, j]

def objective(z):
    """
    Objective function to minimize: negative sum of radii.
    """
    _, radii = get_centers_radii(z)
    return -np.sum(radii)

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
