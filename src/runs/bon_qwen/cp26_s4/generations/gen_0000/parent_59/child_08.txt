2. **Constraint Formulation**: We enforce non-overlap ($\|c_i - c_j\| \ge r_i + r_j$) and boundary containment ($r_i \le x_i, y_i, 1-x_i, 1-y_i$) directly as inequality constraints. Squaring the distance constraint avoids square roots and improves numerical stability.

4. **Vectorized Constraints**: Pairwise distance constraints are computed using NumPy broadcasting to efficiently evaluate the 325 circle-pair constraints, keeping each optimizer iteration fast.
5. **Feasible Initialization**: Starting with a small, feasible radius (0.05) and well-spaced centers ensures the optimizer begins in the feasible region, allowing it to steadily expand radii until constraints bind optimally.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_initializations():
    """Generate diverse initial layouts to avoid local minima."""
    inits = []
    
    # 1. Hexagonal lattice (dense packing structure)
    c1 = np.zeros((N_CIRCLES, 2))
    idx = 0
    rows = 5
    cols_per_row = [5, 6, 5, 6, 4]
    dy = 0.17
    dx = 0.17
    for r in range(rows):
        num_c = cols_per_row[r]
        if idx + num_c > N_CIRCLES: 
            num_c = N_CIRCLES - idx
        offset_x = 0.085 if r % 2 == 1 else 0.0
        for c in range(num_c):
            x = 0.15 + c * dx + offset_x
            y = 0.15 + r * dy
            c1[idx] = [x, y]
            idx += 1
    inits.append(c1)
    
    # 2. Dense grid layout
    c2 = np.zeros((N_CIRCLES, 2))
    idx = 0
    for r in range(5):
        for c in range(5):
            if idx < N_CIRCLES:
                c2[idx] = [0.1 + c * 0.2, 0.1 + r * 0.2]
                idx += 1
    if idx < N_CIRCLES:
        c2[idx] = [0.5, 0.5]
    inits.append(c2)
    
    # 3. Random perturbations to escape grid symmetries
    np.random.seed(42)
    for _ in range(4):
        inits.append(np.random.uniform(0.2, 0.8, (N_CIRCLES, 2)))
        
    return inits

def objective(vars):
    """Negative sum of radii (minimization form)."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """
    Returns array of constraint values. All must be >= 0.
    Constraints:
    1. x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0 (boundary)
    2. dist^2 - (r_i + r_j)^2 >= 0 (non-overlap)
    """
    c = vars.reshape(-1, 3)[:, :2]
    r = vars[2::3]
    
    # Boundary constraints
    b_cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Pairwise non-overlap constraints (vectorized)
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    rad_sum = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((N_CIRCLES, N_CIRCLES), dtype=bool), k=1)
    p_cons = dist_sq[mask] - rad_sum[mask]**2
    
    return np.concatenate([b_cons, p_cons])

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    cons = {'type': 'ineq', 'fun': constraints}
    # Centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    for init_centers in get_initializations():
        x0 = np.zeros(3 * N_CIRCLES)
        for i in range(N_CIRCLES):
            x0[3 * i] = init_centers[i, 0]
            x0[3 * i + 1] = init_centers[i, 1]
            x0[3 * i + 2] = 0.05  # Start feasible with small radii
            
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons, 
                options={'maxiter': 2000, 'ftol': 1e-10}
            )
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x.reshape(-1, 3)[:, :2]
                best_radii = res.x.reshape(-1, 3)[:, 2]
        except Exception:
            continue
            
    if best_centers is None:
        return np.zeros((N_CIRCLES, 2)), np.zeros(N_CIRCLES), 0.0
        
    return best_centers, best_radii, best_sum
