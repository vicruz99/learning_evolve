import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
LAMBDA = 10000.0

def compute_obj_and_grad(vars):
    """
    Computes the objective function and its gradient.
    Objective: Maximize sum of radii (equal radii assumption: -26*r)
    Penalty: Squared violations of boundary and pairwise constraints.
    """
    centers = vars[:N_CIRCLES*2].reshape(-1, 2)
    r = vars[N_CIRCLES*2]
    
    penalty = 0.0
    grad_penalty = np.zeros(len(vars))
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(N_CIRCLES):
        idx_x = i*2
        idx_y = i*2 + 1
        
        # x >= r
        val = centers[i, 0] - r
        if val < 0:
            penalty += val*val
            grad_penalty[idx_x] += 2*val
            grad_penalty[-1] -= 2*val
            
        # x <= 1-r
        val = 1.0 - centers[i, 0] - r
        if val < 0:
            penalty += val*val
            grad_penalty[idx_x] -= 2*val
            grad_penalty[-1] -= 2*val
            
        # y >= r
        val = centers[i, 1] - r
        if val < 0:
            penalty += val*val
            grad_penalty[idx_y] += 2*val
            grad_penalty[-1] -= 2*val
            
        # y <= 1-r
        val = 1.0 - centers[i, 1] - r
        if val < 0:
            penalty += val*val
            grad_penalty[idx_y] -= 2*val
            grad_penalty[-1] -= 2*val

    # Pairwise constraints: dist(i,j) >= 2r
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < 1e-9:
                dist = 1e-9
                
            val = 2.0*r - dist
            if val > 0:
                penalty += val*val
                factor = 2.0 * val / dist
                grad_penalty[i*2] -= factor * dx
                grad_penalty[j*2] += factor * dx
                grad_penalty[i*2+1] -= factor * dy
                grad_penalty[j*2+1] += factor * dy
                grad_penalty[-1] += 4.0 * val
                
    obj = -26.0*r - LAMBDA*penalty
    grad_obj = np.zeros_like(vars)
    grad_obj[-1] = -26.0
    grad_obj -= LAMBDA * grad_penalty
    
    return obj, grad_obj

def run_packing():
    np.random.seed(42)
    best_res = None
    best_sum_r = -1.0
    
    # Run multiple trials with different initial perturbations
    for trial in range(3):

        centers = np.zeros((N_CIRCLES, 2))
        r_init = 0.05
        idx = 0
        for row in range(10):
            for col in range(10):
                if idx >= N_CIRCLES:
                    break
                y = r_init + row * (2 * r_init * np.sin(np.pi/3))
                x = r_init + col * (2 * r_init)
                if row % 2 == 1:
                    x += r_init
                centers[idx] = [x, y]
                idx += 1
            if idx >= N_CIRCLES:
                break
                
        # Add perturbation to break symmetry and explore space
        centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
        centers = np.clip(centers, r_init, 1 - r_init)
        
        vars0 = np.concatenate([centers.flatten(), [r_init]])
        bounds = [(0.0, 1.0)] * (N_CIRCLES*2) + [(0.0, 0.5)]
        
        res = minimize(compute_obj_and_grad, vars0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-12})
        
        curr_r = res.x[-1]
        curr_sum_r = 26.0 * curr_r
        
        if curr_sum_r > best_sum_r:
            best_sum_r = curr_sum_r
            best_res = res
            
    opt_centers = best_res.x[:N_CIRCLES*2].reshape(-1, 2)
    opt_r = best_res.x[-1]
    opt_radii = np.full(N_CIRCLES, opt_r)
    
    return opt_centers, opt_radii, 26.0 * opt_r
