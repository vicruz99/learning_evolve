import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(x):
    """Objective function: negative sum of radii to be minimized."""
    # Radii are at indices 2, 5, 8, ...
    return -np.sum(x[2::3])

def compute_constraints(x):
    """
    Returns a vector of inequality constraint values.
    All constraints are formulated as g(x) >= 0.
    """
    n = N_CIRCLES
    c = []
    
    for i in range(n):
        idx = 3 * i
        xi, yi, ri = x[idx], x[idx+1], x[idx+2]
        
        # Boundary constraints: x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
        c.append(xi - ri)
        c.append(1.0 - (xi + ri))
        c.append(yi - ri)
        c.append(1.0 - (yi + ri))
        
        # Pairwise non-overlap constraints
        for j in range(i + 1, n):
            jdx = 3 * j
            xj, yj, rj = x[jdx], x[jdx+1], x[jdx+2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            # Constraint: dist^2 - (r_i + r_j)^2 >= 0
            c.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(c)

def get_initial_guess():
    """Generates a hexagonal lattice initialization for 26 circles."""
    centers = []
    r_init = 0.08
    row = 0
    count = 0
    
    # Fill rows with hexagonal offset
    while count < N_CIRCLES:
        col = 0
        while count < N_CIRCLES and col < 6:
            # Horizontal spacing ~0.16, vertical ~0.14
            # Offset alternating rows by half spacing
            offset = 0.08 if row % 2 == 1 else 0.0
            x = 0.12 + col * 0.16 + offset
            y = 0.12 + row * 0.14
            centers.append([x, y, r_init])
            count += 1
            col += 1
        row += 1
        
    return np.array(centers).flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    
    best_solution = None
    best_objective = np.inf
    
    x0 = get_initial_guess()
    
    # Run multiple restarts to escape local optima
    np.random.seed(42)
    for attempt in range(5):
        x_curr = x0.copy()
        if attempt > 0:
            # Add controlled noise to positions and radii
            noise = np.random.normal(0, 0.015, size=len(x_curr))
            x_curr += noise
            # Project back to bounds
            for k in range(len(x_curr)):
                low, high = bounds[k]
                x_curr[k] = max(low, min(high, x_curr[k]))
                
        try:
            res = minimize(
                compute_objective,
                x_curr,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': compute_constraints},
                options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
            )
            
            current_obj = res.fun
            if current_obj < best_objective:
                best_objective = current_obj
                best_solution = res.x.copy()
        except Exception:
            continue
            
    if best_solution is None:
        # Fallback to initial guess if optimization completely fails
        best_solution = x0
        
    # Extract centers and radii
    centers = np.array([[best_solution[i], best_solution[i+1]] for i in range(0, 3 * N_CIRCLES, 3)])
    radii = np.array([best_solution[i+2] for i in range(0, 3 * N_CIRCLES, 3)])
    
    # Strict clamping to guarantee validity within numerical tolerance
    for i in range(N_CIRCLES):
        r = radii[i]
        # Ensure center respects r <= coord <= 1-r
        centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))
        # Ensure radii are non-negative
        radii[i] = max(0.0, radii[i])
        
    total_radius_sum = np.sum(radii)
    return centers, radii, total_radius_sum
