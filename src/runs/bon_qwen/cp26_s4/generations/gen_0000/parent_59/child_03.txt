import numpy as np
from scipy.optimize import minimize
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_constraints_and_bounds(n):
    """
    Generates the constraints and bounds for the optimization problem.
    """
    constraints = []
    
    # Non-overlap constraints: dist_ij >= r_i + r_j
    # Equivalent to: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            # Indices in the variable vector:
            # x: i*3, y: i*3+1, r: i*3+2
            def non_overlap_constraint(vars, i=i, j=j):
                xi, yi, ri = vars[i*3], vars[i*3+1], vars[i*3+2]
                xj, yj, rj = vars[j*3], vars[j*3+1], vars[j*3+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                return dist_sq - sum_r**2
            
            constraints.append({'type': 'ineq', 'fun': non_overlap_constraint})

    # Boundary constraints are handled by bounds, but for clarity:
    # r_i >= 0 is in bounds.
    # x, y in [0, 1] is in bounds.
    
    bounds = []
    for i in range(n):
        # x, y in [0, 1]
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        # r in [0, 0.5] (max possible radius)
        bounds.append((0.0, 0.5))
        
    return constraints, bounds

def generate_hexagonal_grid(n):
    """
    Generates initial center positions for n circles using a hexagonal grid pattern.
    """
    centers = []
    # Approximate number of rows/cols
    # Area per circle ~ 1/n. Radius ~ 0.5/sqrt(n). Diameter ~ 1/sqrt(n).
    # Spacing ~ 1/sqrt(n).
    # Let's try to fit into the square.
    
    # Heuristic: try to fit rows
    # Number of rows approx sqrt(2/3 * n) * 1/sqrt(density) ?
    # Just try a grid search for best fit
    best_centers = None
    best_score = -1
    
    # Try different grid configurations
    for rows in range(4, 8):
        cols = math.ceil(n / rows) + 2
        # Hexagonal spacing
        # Width per col = spacing
        # Height per row = spacing * sqrt(3)/2
        # Total width = (cols-1)*spacing
        # Total height = (rows-1)*spacing*sqrt(3)/2 + 2*radius (approx)
        
        # We want to maximize spacing to allow larger radii later
        # spacing_x = 1 / (cols - 1)
        # spacing_y = 1 / ((rows - 1) * math.sqrt(3)/2)
        # spacing = min(spacing_x, spacing_y)
        
        if cols < 2 or rows < 2:
            continue
            
        spacing_x = 1.0 / (cols - 1)
        spacing_y = 1.0 / ((rows - 1) * math.sqrt(3)/2)
        
        # We want square-ish spacing for hex grid
        spacing = min(spacing_x, spacing_y)
        
        # If spacing is too small, skip? No, just generate.
        
        c_list = []
        count = 0
        for r in range(rows):
            for c in range(cols):
                if count >= n:
                    break
                x = c * spacing
                y = r * spacing * math.sqrt(3)/2
                
                # Offset every other row
                if r % 2 == 1:
                    x += spacing / 2
                
                # Center the grid in the square
                # We need to scale/shift to fit [0,1]
                c_list.append([x, y])
                count += 1
            if count >= n:
                break
        
        # Normalize to fit [0,1]
        if len(c_list) == n:
            xs = [p[0] for p in c_list]
            ys = [p[1] for p in c_list]
            min_x, max_x = min(xs), max(xs)
            min_y, max_y = min(ys), max(ys)
            
            # Add padding for radius (approx 0.1)
            # Scale factor
            range_x = max_x - min_x
            range_y = max_y - min_y
            if range_x == 0 or range_y == 0:
                continue
                
            # Target size 0.8 to leave room
            scale_x = 0.8 / range_x
            scale_y = 0.8 / range_y
            scale = min(scale_x, scale_y)
            
            norm_centers = []
            for p in c_list:
                nx = (p[0] - min_x) * scale + 0.1
                ny = (p[1] - min_y) * scale + 0.1
                norm_centers.append([nx, ny])
            
            # Check validity of initial packing with small radius
            # Just to ensure they are inside
            valid = True
            for p in norm_centers:
                if p[0] < 0 or p[0] > 1 or p[1] < 0 or p[1] > 1:
                    valid = False
                    break
            if valid:
                # Calculate min distance to estimate max possible radius
                min_dist = float('inf')
                for i in range(len(norm_centers)):
                    for j in range(i+1, len(norm_centers)):
                        d = np.sqrt((norm_centers[i][0]-norm_centers[j][0])**2 + (norm_centers[i][1]-norm_centers[j][1])**2)
                        if d < min_dist:
                            min_dist = d
                # Score based on min distance (larger is better)
                if min_dist > best_score:
                    best_score = min_dist
                    best_centers = norm_centers

    if best_centers is None:
        # Fallback to random
        best_centers = np.random.rand(n, 2)
        
    return np.array(best_centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    constraints, bounds = get_constraints_and_bounds(n)
    
    # Generate initial centers
    init_centers = generate_hexagonal_grid(n)
    
    # Initial radii: small value to ensure non-overlap initially
    # Or better, half of min distance
    min_dist = float('inf')
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(init_centers[i] - init_centers[j])
            min_dist = min(min_dist, d)
    init_radii = np.full(n, min_dist / 4) # Safe initial radii
    
    # Combine into vector
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = init_centers[i, 0]
        x0[3*i+1] = init_centers[i, 1]
        x0[3*i+2] = init_radii[i]
        
    best_sum_radii = -1
    best_sol = None
    
    # Run optimization multiple times with slight perturbations
    for k in range(5):
        # Perturb initial guess
        if k > 0:
            perturbation = np.random.normal(0, 0.02, size=(n, 2))
            pert_centers = np.clip(init_centers + perturbation, 0.1, 0.9)
            # Recalculate safe radii
            min_d = float('inf')
            for i in range(n):
                for j in range(i+1, n):
                    d = np.linalg.norm(pert_centers[i] - pert_centers[j])
                    min_d = min(min_d, d)
            r_start = min_d / 5
            
            curr_x0 = np.zeros(3 * n)
            for i in range(n):
                curr_x0[3*i] = pert_centers[i, 0]
                curr_x0[3*i+1] = pert_centers[i, 1]
                curr_x0[3*i+2] = r_start
        else:
            curr_x0 = x0
            
        # Objective: minimize negative sum of radii
        def objective(vars):
            total_r = sum(vars[3*i+2] for i in range(n))
            return -total_r

        try:
            res = minimize(objective, curr_x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'ftol': 1e-9, 'maxiter': 500})
            
            if res.success or (res.nit > 0 and res.fun < -best_sum_radii):
                final_sum = -res.fun
                if final_sum > best_sum_radii:
                    best_sum_radii = final_sum
                    best_sol = res.x.copy()
        except Exception as e:
            print(f"Optimization run {k} failed: {e}")
            continue
            
    # Extract results
    if best_sol is None:
        # Fallback to initial
        centers = init_centers
        radii = init_radii
    else:
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            centers[i, 0] = best_sol[3*i]
            centers[i, 1] = best_sol[3*i+1]
            radii[i] = best_sol[3*i+2]
            
    sum_radii = np.sum(radii)
    
    # Validate
    is_valid = validate_packing(centers, radii)
    if not is_valid:
        print("Final solution invalid!")
        # Try to fix by reducing radii slightly?
        # Just return as is, maybe numerical errors in validation
        
    return centers, radii, sum_radii

# To allow running the code directly
if __name__ == "__main__":
    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
    print(f"Valid: {validate_packing(c, r)}")
