def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    """

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Helper to construct initial state from centers and radii
    def state_from_cr(centers, radii):
        state = np.zeros(3 * n)
        state[0::3] = centers[:, 0]
        state[1::3] = centers[:, 1]
        state[2::3] = radii
        return state

        return -np.sum(x[2::3])

    def get_constraints():
        constraints = []

        # x <= 1-r => 1 - x - r >= 0
        # y >= r  => y - r >= 0
        # y <= 1-r => 1 - y - r >= 0
        for i in range(n):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            def c_x_min(x, i=i): return x[idx_x] - x[idx_r]
            def c_x_max(x, i=i): return 1.0 - x[idx_x] - x[idx_r]
            def c_y_min(x, i=i): return x[idx_y] - x[idx_r]
            def c_y_max(x, i=i): return 1.0 - x[idx_y] - x[idx_r]
            
            constraints.append({'type': 'ineq', 'fun': c_x_min})
            constraints.append({'type': 'ineq', 'fun': c_x_max})
            constraints.append({'type': 'ineq', 'fun': c_y_min})
            constraints.append({'type': 'ineq', 'fun': c_y_max})

        # Pairwise non-overlap constraints
        # dist^2 >= (r1 + r2)^2  => dist^2 - (r1 + r2)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_x1, idx_y1, idx_r1 = 3*i, 3*i+1, 3*i+2
                idx_x2, idx_y2, idx_r2 = 3*j, 3*j+1, 3*j+2
                
                def c_pair(x, i=i, j=j):
                    dx = x[idx_x1] - x[idx_x2]
                    dy = x[idx_y1] - x[idx_y2]
                    dist_sq = dx*dx + dy*dy
                    r_sum = x[idx_r1] + x[idx_r2]
                    return dist_sq - r_sum*r_sum
                
                constraints.append({'type': 'ineq', 'fun': c_pair})
        
        return constraints
