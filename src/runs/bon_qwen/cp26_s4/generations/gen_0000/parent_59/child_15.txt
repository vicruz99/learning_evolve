import numpy as np

def _compute_forces(centers, target_r, k=50.0):
    """
    Compute repulsive forces to maximize packing density.
    Returns force vectors to add to centers.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Pairwise distances and vectors
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # Ignore self-interaction
    
    # Avoid division by zero
    safe_dists = np.maximum(dists, 1e-12)
    dirs = diffs / safe_dists[:, :, np.newaxis]
    
    # Target separation is 2 * target_r
    target_sep = 2.0 * target_r
    overlaps = target_sep - dists
    # Only apply force if too close
    mask = overlaps > 0
    force_magnitudes = np.where(mask, k * overlaps / safe_dists, 0.0)
    
    # Sum forces: F_ij acts on i in direction i->j
    forces += np.sum(force_magnitudes[:, :, np.newaxis] * dirs, axis=0)
    
    # Boundary repulsion
    # Left wall (x=0)
    left_dist = centers[:, 0] - target_r
    left_mask = left_dist < 0
    forces[left_mask, 0] -= k * left_dist[left_mask]
    
    # Right wall (x=1)
    right_dist = (1.0 - centers[:, 0]) - target_r
    right_mask = right_dist < 0
    forces[right_mask, 0] -= k * right_dist[right_mask]
    
    # Bottom wall (y=0)
    bot_dist = centers[:, 1] - target_r
    bot_mask = bot_dist < 0
    forces[bot_mask, 1] -= k * bot_dist[bot_mask]
    
    # Top wall (y=1)
    top_dist = (1.0 - centers[:, 1]) - target_r
    top_mask = top_dist < 0
    forces[top_mask, 1] -= k * top_dist[top_mask]
    
    return forces

def _compute_max_radius(centers):
    """
    Compute the maximum uniform radius that fits the current center configuration.
    """
    # Distance to boundaries
    dist_to_bound = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    min_bound = np.min(dist_to_bound)
    
    # Distance between circles
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pair = np.min(dists) / 2.0
    
    return min(min_bound, min_pair)

        x_start = (1.0 - row_width) / 2.0 + r_init

            centers[idx, 0] = x_start + i * dx
            centers[idx, 1] = y
            idx += 1

    target_r = 0.08
    lr = 0.04
    damping = 0.95
    
    # Run force-directed relaxation
    for step in range(8000):
