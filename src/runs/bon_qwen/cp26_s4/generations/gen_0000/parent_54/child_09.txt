

To maximize the sum of radii for 26 circles in a unit square, we model the problem as a non-linear programming (NLP) task. We define 78 variables (26 centers and 26 radii) and optimize to maximize $\sum r_i$ subject to non-overlap and boundary constraints.

### Strategy
1.  **Initialization**: Place 26 circles in a hexagonal grid (a subgrid of a 6x5 lattice) with a conservative initial radius ($r=0.08$) to ensure a feasible starting point.
2.  **Optimization**: Use `scipy.optimize.minimize` with the SLSQP method to simultaneously adjust centers and expand radii. This method handles the non-linear distance constraints and linear boundary constraints effectively.
3.  **Validation**: Run the `validate_packing` function to ensure the result satisfies all constraints before returning the maximum sum of radii.

```python
import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing():
    n = 26
    # Initial hexagonal grid placement (subset of 6x5)
    initial_r = 0.08
    centers = np.zeros((n, 2))
    radii = np.full(n, initial_r)
    
    count = 0
    for row in range(5):
        y = initial_r + row * initial_r * np.sqrt(3)
        for col in range(6):
            if count < n:
                x = initial_r + col * 2 * initial_r + (row % 2) * initial_r
                # Boundary correction for initial placement
                x = np.clip(x, initial_r, 1 - initial_r)
                y = np.clip(y, initial_r, 1 - initial_r)
                centers[count] = [x, y]
                count += 1

    def objective(z):
        r = z[52:]
        return -np.sum(r)

    def constraints_func(z):
        constraints = []
        r = z[52:]
        
        # Boundary constraints
        for i in range(n):
            x, y = z[2*i], z[2*i+1]
            constraints.append(x - r[i])
            constraints.append(1 - x - r[i])
            constraints.append(y - r[i])
            constraints.append(1 - y - r[i])
            
        # Pairwise distance constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = z[2*i] - z[2*j]
                dy = z[2*i+1] - z[2*j+1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (r[i] + r[j]))
                
        return np.array(constraints)

    # Initial variables: 52 centers + 26 radii
    z0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: Centers [0, 1], Radii [0, 0.5]
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26

    # Non-linear constraint
    constraints = {'type': 'ineq', 'fun': constraints_func}

    # Run optimization
    result = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'ftol': 1e-12, 'maxiter': 2000})

    if result.success:
        final_centers = result.x[:52].reshape(-1, 2)
        final_radii = result.x[52:]
        if validate_packing(final_centers, final_radii):
            return final_centers, final_radii, np.sum(final_radii)
            
    # Fallback to initial if optimizer failed validation
    return centers, radii, np.sum(radii)

# Execute and print results
centers, radii, sum_radii = run_packing()
print(f"Sum of radii: {sum_radii}")
```
