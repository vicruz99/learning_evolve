import numpy as np
from scipy.optimize import minimize
import math

def objective_function(theta, n, penalty_weight):
    """
    Objective function to minimize.
    Minimizes negative sum of radii plus penalties for overlaps and boundary violations.
    """
    # Unpack parameters
    x = theta[0::3]
    y = theta[1::3]
    r = theta[2::3]
    
    # Primary objective: maximize sum of radii -> minimize -sum(r)
    val = -np.sum(r)
    
    # Penalty for pairwise overlaps
    overlap_pen = 0.0
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, n):
            dx = xi - x[j]
            dy = yi - y[j]
            dist = math.hypot(dx, dy)
            # Constraint: dist >= ri + r[j]
            # Violation: ri + r[j] - dist
            violation = ri + r[j] - dist
            if violation > 0:
                overlap_pen += violation ** 2
                
    # Penalty for boundary violations
    bound_pen = 0.0
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        # Left: xi >= ri
        if ri > xi:
            bound_pen += (ri - xi) ** 2
        # Right: xi + ri <= 1
        if ri + xi > 1:
            bound_pen += (ri + xi - 1) ** 2
        # Bottom: yi >= ri
        if ri > yi:
            bound_pen += (ri - yi) ** 2
        # Top: yi + ri <= 1
        if ri + yi > 1:
            bound_pen += (ri + yi - 1) ** 2
            
    val += penalty_weight * (overlap_pen + bound_pen)
    return val

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Helper to run optimization
    def try_opt(init_theta, penalty_w):
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
            
        res = minimize(objective_function, init_theta, args=(n, penalty_w), 
                       method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10})
        return res.x

    # Strategy 1: Random initializations
    for trial in range(20):
        np.random.seed(trial * 17 + 5)
        centers = np.random.rand(n, 2)
        radii = np.full(n, 0.001) # Start with tiny valid radii
        
        theta = np.zeros(3 * n)
        theta[0::3] = centers[:, 0]
        theta[1::3] = centers[:, 1]
        theta[2::3] = radii
        
        res_theta = try_opt(theta, 50000.0)
        
        c = np.column_stack((res_theta[0::3], res_theta[1::3]))
        r = res_theta[2::3]
        
        # Validate and update best
        if is_valid_packing(c, r):
            s = np.sum(r)
            if s > best_sum:
                best_sum = s
                best_centers = c.copy()
                best_radii = r.copy()

    # Strategy 2: Grid initialization (dense packing start)
    # 6x5 grid gives 30 points, we pick 26
    grid_pts = []
    for i in range(6):
        for j in range(5):
            grid_pts.append(((i + 0.5) / 6.0, (j + 0.5) / 5.0))
    
    np.random.seed(42)
    np.random.shuffle(grid_pts)
    init_pts = np.array(grid_pts[:n])
    
    radii = np.full(n, 0.01)
    theta = np.zeros(3 * n)
    theta[0::3] = init_pts[:, 0]
    theta[1::3] = init_pts[:, 1]
    theta[2::3] = radii
    
    res_theta = try_opt(theta, 50000.0)
    c = np.column_stack((res_theta[0::3], res_theta[1::3]))
    r = res_theta[2::3]
    if is_valid_packing(c, r):
        s = np.sum(r)
        if s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()

    # Fallback if optimization failed (should not happen)
    if best_centers is None:
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.full(n, 0.001)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)

def is_valid_packing(centers, radii):
    """Internal validation helper"""
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    
    # Check radii and boundaries
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
            
    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
