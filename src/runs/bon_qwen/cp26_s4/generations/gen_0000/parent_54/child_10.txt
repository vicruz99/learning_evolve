import numpy as np
import scipy.optimize as opt
import math

def compute_overlap_penalty(centers, radii):
    """
    Computes a penalty for overlapping circles.
    Penalty is sum of squared overlaps: sum(max(0, r_i + r_j - dist)^2)
    """
    n = centers.shape[0]
    penalty = 0.0
    # Vectorized distance calculation
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # radii sum shape (n, n)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount
    overlap = r_sum - dist
    
    # We only care about positive overlaps (actual intersection)
    # Also ignore self-interaction (diagonal)
    np.fill_diagonal(overlap, 0)
    
    # Take max(0, overlap)
    positive_overlap = np.maximum(0, overlap)
    
    # Square and sum
    penalty = np.sum(positive_overlap**2)
    return penalty

def compute_boundary_penalty(centers, radii):
    """
    Computes penalty for circles being outside the unit square.
    Penalty is sum of squared violations.
    """
    x, y = centers[:, 0], centers[:, 1]
    r = radii
    
    # Left wall: x - r >= 0  => violation if r - x > 0
    pen_left = np.maximum(0, r - x)**2
    # Right wall: 1 - x - r >= 0 => violation if x + r - 1 > 0
    pen_right = np.maximum(0, x + r - 1)**2
    # Bottom wall: y - r >= 0
    pen_bottom = np.maximum(0, r - y)**2
    # Top wall: 1 - y - r >= 0
    pen_top = np.maximum(0, y + r - 1)**2
    
    return np.sum(pen_left) + np.sum(pen_right) + np.sum(pen_bottom) + np.sum(pen_top)

    Add penalty for constraint violations.

    # Unpack params: x1, y1, r1, x2, y2, r2, ...

        centers[i, 0] = params[3*i]
        centers[i, 1] = params[3*i + 1]
        radii[i] = params[3*i + 2]

    sum_radii = -np.sum(radii)
    
    # Penalties
    overlap_pen = compute_overlap_penalty(centers, radii)
    boundary_pen = compute_boundary_penalty(centers, radii)
    
    total_penalty = penalty_weight * (overlap_pen + boundary_pen)
    
    return sum_radii + total_penalty

def constraint_function(params, n_circles):
    """
    Returns an array of constraint values.
    For 'ineq' type, we need result >= 0.
    Constraints:
    1. Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    2. Non-overlap: dist >= r_i + r_j  => dist^2 - (r_i+r_j)^2 >= 0
       Using squared distance avoids sqrt and is differentiable.
    """
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        centers[i, 0] = params[3*i]
        centers[i, 1] = params[3*i + 1]
        radii[i] = params[3*i + 2]
        
    constraints = []
    
    # Boundary constraints
    # x - r >= 0
    constraints.extend(centers[:, 0] - radii)
    # 1 - x - r >= 0
    constraints.extend(1 - centers[:, 0] - radii)
    # y - r >= 0
    constraints.extend(centers[:, 1] - radii)
    # 1 - y - r >= 0
    constraints.extend(1 - centers[:, 1] - radii)
    
    # Pairwise constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # We only need to check i < j
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            val = dist_sq - r_sum*r_sum
            constraints.append(val)
            
    return np.array(constraints)
