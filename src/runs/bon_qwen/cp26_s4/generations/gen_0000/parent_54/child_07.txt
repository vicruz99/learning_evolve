import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

_N = 26
_TOTAL_VARS = _N * 3
_CONSTRAINT_SIZE = 4 * _N + (_N * (_N - 1)) // 2  # 429 constraints
_MASK = np.triu(np.ones((_N, _N), dtype=bool), k=1)

def objective_func(vars):
    return -np.sum(vars[2::3])

def constraint_func(vars):
    out = np.empty(_CONSTRAINT_SIZE)
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]

    idx = 0
    out[idx:idx+_N] = xs - rs              # x >= r
    idx += _N
    out[idx:idx+_N] = 1.0 - xs - rs        # x + r <= 1
    idx += _N
    out[idx:idx+_N] = ys - rs              # y >= r
    idx += _N
    out[idx:idx+_N] = 1.0 - ys - rs        # y + r <= 1
    idx += _N

    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    out[idx:] = dx[_MASK]**2 + dy[_MASK]**2 - dr[_MASK]**2  # dist^2 >= (r_i+r_j)^2
    return out

def run_packing():
    bounds = [(0.0, 1.0)] * (2*_N) + [(0.0, 0.5)] * _N
    constr = NonlinearConstraint(constraint_func, 0.0, np.inf)

    best_val = np.inf
    best_res = None
    rng = np.random.RandomState(2024)

    # Multi-start optimization from diverse grid-like configurations
    for seed in range(5):
        xs = np.linspace(0.12, 0.88, 6)
        ys = np.linspace(0.12, 0.88, 5)
        init_xs = []
        init_ys = []
        for y in ys:
            for x in xs:
                if len(init_xs) < _N:
                    init_xs.append(x)
                    init_ys.append(y)
        
        init_xs = np.array(init_xs) + rng.uniform(-0.015, 0.015, _N)
        init_ys = np.array(init_ys) + rng.uniform(-0.015, 0.015, _N)
        init_xs = np.clip(init_xs, 0.05, 0.95)
        init_ys = np.clip(init_ys, 0.05, 0.95)

        v0 = np.zeros(_TOTAL_VARS)
        v0[0::3] = init_xs
        v0[1::3] = init_ys
        v0[2::3] = 0.045  # Start with small feasible radii

        res = minimize(
            objective_func,
            v0,
            method='SLSQP',
            bounds=bounds,
            constraints=[constr],
            options={'maxiter': 1500, 'ftol': 1e-10, 'disp': False}
        )
        if res.fun < best_val:
            best_val = res.fun
            best_res = res

    if best_res is not None:
        centers = np.column_stack((best_res.x[0::3], best_res.x[1::3]))
        radii = best_res.x[2::3]
    else:
        centers = np.zeros((_N, 2))
        radii = np.zeros(_N)

    # Ensure strict validity for the checker
    radii = np.maximum(radii, 1e-9)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
