import numpy as np
from scipy.optimize import minimize
import math

def get_hexagonal_initialization(n):
    """
    Generates an initial hexagonal packing configuration for n circles.
    Pattern: 6, 5, 6, 5, 4 (total 26)
    """
    if n == 26:
        rows = [6, 5, 6, 5, 4]
    else:
        # Fallback for other n
        rows = []
        rem = n
        count = 6
        while rem > 0:
            rows.append(min(count, rem))
            rem -= count
            count -= 1
            if count < 1: count = 1 # prevent infinite loop if n is weird, though not needed for 26

    centers = []
    y_curr = 0.0
    vert_spacing = math.sqrt(3) / 2
    
    for i, count in enumerate(rows):
        # Horizontal shift for hexagonal packing: even rows 0, odd rows 0.5
        # But to center the pattern in [0,1], we need to calculate offsets.
        # First, let's generate relative coordinates assuming spacing 1.
        
        # Relative x positions
        if i % 2 == 0:
            # Even rows (0, 2, 4): 6, 6, 4 items. 
            # For 6 items: 0, 1, 2, 3, 4, 5. Range 5.
            # For 4 items: 1, 2, 3, 4. Range 3. (Centered relative to 6)
            if count == 6:
                xs = np.arange(6) 
            elif count == 4:
                xs = np.arange(1, 5) # 1, 2, 3, 4
            else:
                xs = np.arange(count)
        else:
            # Odd rows (1, 3): 5, 5 items.
            # Staggered by 0.5. 
            # Range should align with even rows.
            # 5 items: 0.5, 1.5, 2.5, 3.5, 4.5. Range 4.
            xs = np.arange(0.5, 5.0, 1.0)
            
        # Current y position
        ys = [y_curr] * count
        
        for x, y in zip(xs, ys):
            centers.append([x, y])
        
        y_curr += vert_spacing
            
    centers = np.array(centers)
    
    # Normalize to fit in unit square with some margin
    # Find bounding box
    min_x, min_y = centers.min(axis=0)
    max_x, max_y = centers.max(axis=0)
    
    width = max_x - min_x
    height = max_y - min_y
    
    # Scale to fit in [0.1, 0.9] initially to give optimizer room
    scale = 0.8 / max(width, height)
    centers = centers * scale
    
    # Center in square
    centers -= centers.mean(axis=0)
    centers += 0.5
    
    # Initial radius estimate: 1 / (2 * max_dimension_in_units) * 0.8?
    # Roughly 1/10
    radii = np.ones(n) * 0.09 
    
    return centers, radii

def objective(x_vec, n):
    """
    Objective function: minimize negative sum of radii.
    x_vec contains [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = x_vec[2::3]
    return -np.sum(radii)

def get_constraints(n):
    """
    Returns list of constraint dictionaries for scipy.optimize.
    """
    constraints = []
    
    # 1. Boundary constraints: r <= x <= 1-r  =>  x - r >= 0,  1 - r - x >= 0
    # Similarly for y
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x >= r  => 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })
        
        # Radius non-negative (handled by bounds, but added for safety)
        # bounds will handle this strictly
        
    # 2. Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2
            
            def dist_constraint(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                return dist_sq - sum_r**2
            
            constraints.append({
                'type': 'ineq',
                'fun': dist_constraint
            })
            
    return constraints

def run_packing():
    n = 26
    # Initialization
    centers, radii = get_hexagonal_initialization(n)
    
    # Combine into vector
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5] (cannot be larger than 0.5)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    constraints = get_constraints(n)
    
    # Optimization
    # Using SLSQP which handles constraints well
    res = minimize(
        objective, 
        x0, 
        args=(n,), 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    
    # Extract results
    final_x = res.x
    centers_opt = np.zeros((n, 2))
    radii_opt = np.zeros(n)
    
    for i in range(n):
        centers_opt[i, 0] = final_x[3*i]
        centers_opt[i, 1] = final_x[3*i+1]
        radii_opt[i] = final_x[3*i+2]
        
    # Post-processing to ensure strict validity (remove tiny negative values if any due to precision)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    # Clip centers to [0,1] just in case
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    # Check shapes
    n = centers.shape[0]
    if radii.shape[0] != n:
        return False

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

# Testing the solution
if __name__ == "__main__":
    centers, radii, s_r = run_packing()
    print(f"Sum of radii: {s_r}")
    is_valid = validate_packing(centers, radii)
    print(f"Valid: {is_valid}")
