import numpy as np
from scipy.optimize import minimize

def obj_func(v):
    """Objective: minimize negative sum of radii"""
    radii = v[2::3]
    return -np.sum(radii)

def bound_x_low(v, i):
    """x_i - r_i >= 0"""
    return v[3*i] - v[3*i+2]

def bound_x_upp(v, i):
    """1 - x_i - r_i >= 0"""
    return 1.0 - v[3*i] - v[3*i+2]

def bound_y_low(v, i):
    """y_i - r_i >= 0"""
    return v[3*i+1] - v[3*i+2]

def bound_y_upp(v, i):
    """1 - y_i - r_i >= 0"""
    return 1.0 - v[3*i+1] - v[3*i+2]

def sep_con(v, i, j):
    """Distance^2 - (r_i + r_j)^2 >= 0"""
    dx = v[3*i] - v[3*j]
    dy = v[3*i+1] - v[3*j+1]
    dr = v[3*i+2] + v[3*j+2]
    return dx*dx + dy*dy - dr*dr

def run_packing():
    n = 26
    r0 = 0.1
    centers = np.zeros((n, 2))
    radii = np.full(n, r0)
    
    # Initialize with hexagonal lattice
    idx = 0
    row = 0
    while idx < n:
        y = r0 + row * r0 * np.sqrt(3)
        offset = r0 if row % 2 == 1 else 0.0
        max_cols = int(np.floor((1.0 - 2*r0 - offset) / (2*r0))) + 1
        n_cols = min(n - idx, max(1, max_cols))
        
        for j in range(n_cols):
            if idx >= n:
                break
            x = r0 + j * 2*r0 + offset
            x = np.clip(x, r0, 1.0 - r0)
            y_val = np.clip(y, r0, 1.0 - r0)
            centers[idx] = [x, y_val]
            idx += 1
        row += 1
        
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0) for _ in range(3*n)]
    
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': bound_x_low, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': bound_x_upp, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': bound_y_low, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': bound_y_upp, 'args': (i,)})
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({'type': 'ineq', 'fun': sep_con, 'args': (i, j)})
            
    res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    for i in range(n):
        final_centers[i] = [res.x[3*i], res.x[3*i+1]]
        final_radii[i] = max(res.x[3*i+2], 1e-7)
        
    return final_centers, final_radii, np.sum(final_radii)
