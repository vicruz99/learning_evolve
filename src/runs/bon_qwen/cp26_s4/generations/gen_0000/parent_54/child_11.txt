def calculate_pairwise_distances(centers):
    """
    Calculate the matrix of pairwise distances between centers.
    Centers shape: (N, 2)
    Returns: (N, N) matrix of distances
    """
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    return dists
