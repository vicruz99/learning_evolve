import numpy as np
from scipy.optimize import minimize

def compute_constraints(p, n=26):
    """Computes all inequality constraints >= 0"""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    c = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c.extend(x - r)
    c.extend(1 - x - r)
    c.extend(y - r)
    c.extend(1 - y - r)
    # Non-negative radii
    c.extend(r)
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            c.append(dx**2 + dy**2 - (r[i] + r[j])**2)
    return np.array(c)

def objective_func(p, n=26):
    """Objective: minimize negative sum of radii"""
    return -np.sum(p[2::3])

def run_packing():
    n = 26
    best_sum_r = -np.inf
    best_centers = None
    best_radii = None
    
    # Bounds for x, y in [0,1] and r in [0, 0.5]
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    cons = [{'type': 'ineq', 'fun': compute_constraints}]
    
    initial_guesses = []
    
    # 1. 5x5 Grid layout + 1 center
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    p1 = np.zeros(n * 3)
    for i in range(n):
        if i < 25:
            p1[3*i] = grid_x[i % 5]
            p1[3*i+1] = grid_y[i // 5]
        else:
            p1[3*i] = 0.5
            p1[3*i+1] = 0.5
        p1[3*i+2] = 0.05
    initial_guesses.append(p1)
    
    # 2. Hexagonal-ish layout
    p2 = np.zeros(n * 3)
    idx = 0
    for row in range(6):
        cols = 5 if row % 2 == 0 else 4
        y_pos = 0.15 + row * 0.14
        for col in range(cols):
            if idx >= n: break
            x_pos = 0.15 + col * 0.16 + (0.08 if row % 2 == 1 else 0)
            p2[3*idx] = np.clip(x_pos, 0.1, 0.9)
            p2[3*idx+1] = np.clip(y_pos, 0.1, 0.9)
            p2[3*idx+2] = 0.04
            idx += 1
    initial_guesses.append(p2)
    
    # 3. Random feasible layouts
    for seed in range(5):
        np.random.seed(seed * 100 + 42)
        p_rand = np.random.uniform([0.15, 0.15, 0.03], [0.85, 0.85, 0.03], size=(n, 3)).flatten()
        initial_guesses.append(p_rand)
        
    # Optimize over all initial guesses
    for p0 in initial_guesses:
        try:
            res = minimize(objective_func, p0, args=(n,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False})
            if res.success:
                s_r = -res.fun
                if s_r > best_sum_r:
                    best_sum_r = s_r
                    best_centers = res.x[0::3].reshape(-1, 2)
                    best_radii = res.x[2::3]
        except Exception:
            continue
            
    # Ensure validity and clamp to boundaries
    if best_centers is not None:
        best_radii = np.maximum(best_radii, 0.0)
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1 - best_radii)
        best_sum_r = np.sum(best_radii)
    else:
        # Fallback to a safe grid configuration
        best_centers = np.column_stack([np.tile(grid_x, 5), np.repeat(grid_y, 5)])
        best_radii = np.full(n, 0.09)
        best_sum_r = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum_r
