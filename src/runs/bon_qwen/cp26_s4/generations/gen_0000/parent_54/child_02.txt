import numpy as np
from scipy.optimize import minimize

# Validation function provided in the prompt
def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def compute_cost_and_grad(z, N, LAMBDA):
    """
    Compute the objective function value and its gradient.
    Objective: minimize -sum(r_i) + LAMBDA * penalty
    """
    # Unpack variables
    centers = z.reshape((N, 3))[:, :2] # (N, 2)
    radii = z.reshape((N, 3))[:, 2]   # (N,)
    
    # Precompute squared distances for efficiency
    # d_sq[i, j] = ||c_i - c_j||^2
    # Using broadcasting
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (N, N, 2)
    d_sq = np.sum(c_diff**2, axis=2) # (N, N)
    
    # Distance matrix
    d = np.sqrt(d_sq + 1e-20) # Avoid division by zero
    
    # Objective value: -sum(r)
    cost = -np.sum(radii)
    
    # Penalty initialization
    penalty = 0.0
    
    # Gradient initialization
    # grad_z corresponds to [dx_i, dy_i, dr_i] for all i
    grad = np.zeros_like(z)
    # grad_centers: (N, 2), grad_radii: (N,)
    grad_centers = np.zeros_like(centers)
    grad_radii = np.zeros_like(radii)
    
    # Pairwise constraints
    # Indices i < j
    for i in range(N):
        for j in range(i + 1, N):
            r_sum = radii[i] + radii[j]
            dist = d[i, j]
            overlap = r_sum - dist
            
            if overlap > 0:
                # Penalty contribution
                penalty += overlap**2
                
                # Gradient terms
                # Deriv of (r_i + r_j - d_ij)^2
                # w.r.t r_i: 2 * overlap
                grad_radii[i] += 2 * overlap
                grad_radii[j] += 2 * overlap
                
                # w.r.t x_i, y_i: 2 * overlap * (-deriv of dist)
                # deriv dist / deriv x_i = (x_i - x_j) / dist
                inv_dist = 1.0 / dist if dist > 1e-12 else 0.0
                grad_centers[i, 0] -= 2 * overlap * (centers[i, 0] - centers[j, 0]) * inv_dist
                grad_centers[i, 1] -= 2 * overlap * (centers[i, 1] - centers[j, 1]) * inv_dist
                grad_centers[j, 0] += 2 * overlap * (centers[i, 0] - centers[j, 0]) * inv_dist
                grad_centers[j, 1] += 2 * overlap * (centers[i, 1] - centers[j, 1]) * inv_dist

    # Boundary constraints
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        
        # Left wall: r <= x -> r - x <= 0
        v_left = r - x
        if v_left > 0:
            penalty += v_left**2
            grad_radii[i] += 2 * v_left
            grad_centers[i, 0] -= 2 * v_left
            
        # Right wall: x + r <= 1 -> x + r - 1 <= 0
        v_right = x + r - 1
        if v_right > 0:
            penalty += v_right**2
            grad_radii[i] += 2 * v_right
            grad_centers[i, 0] += 2 * v_right
            
        # Bottom wall: r <= y -> r - y <= 0
        v_bot = r - y
        if v_bot > 0:
            penalty += v_bot**2
            grad_radii[i] += 2 * v_bot
            grad_centers[i, 1] -= 2 * v_bot
            
        # Top wall: y + r <= 1 -> y + r - 1 <= 0
        v_top = y + r - 1
        if v_top > 0:
            penalty += v_top**2
            grad_radii[i] += 2 * v_top
            grad_centers[i, 1] += 2 * v_top

    # Combine cost and gradient
    total_cost = cost + LAMBDA * penalty
    
    # Construct full gradient vector
    grad[:, 0] = LAMBDA * grad_centers[:, 0] # dx
    grad[:, 1] = LAMBDA * grad_centers[:, 1] # dy
    grad[:, 2] = -1.0 + LAMBDA * grad_radii # dr (derivative of -r is -1)
    
    return total_cost, grad
