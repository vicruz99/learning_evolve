import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def all_constraints(v, N):
    """Evaluate all boundary and non-overlap constraints efficiently."""
    c = v.reshape(N, 3)  # Columns: x, y, r
    
    # Boundary constraints: x-r >=0, 1-x-r >=0, y-r >=0, 1-y-r >=0
    bx = c[:,0] - c[:,2]
    bx2 = 1.0 - c[:,0] - c[:,2]
    by = c[:,1] - c[:,2]
    by2 = 1.0 - c[:,1] - c[:,2]
    bcons = np.concatenate([bx, bx2, by, by2])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i + r_j)^2 >= 0
    diff_xy = c[:, None, :2] - c[None, :, :2]  # Shape (N, N, 2)
    dist_sq = np.sum(diff_xy**2, axis=2)
    sum_r = c[:, None, 2] + c[None, :, 2]
    ov = dist_sq - sum_r**2
    
    # Extract upper triangle (i < j) to avoid duplicates
    idx = np.triu_indices(N, k=1)
    ocons = ov[idx]
    
    return np.concatenate([bcons, ocons])

def run_packing():
    N = 26
    best_sum_r = -1.0
    best_v = None
    
    # Initial configuration: 5x5 grid + 1 circle in a gap
    base_centers = []
    for i in range(5):
        for j in range(5):
            base_centers.append((0.1 + 0.2*i, 0.1 + 0.2*j))
    base_centers.append((0.5, 0.4))  # 26th circle
    
    bounds_list = [(0.0, 1.0)] * 2 * N + [(0.0, 0.5)] * N
    constraint = {'type': 'ineq', 'fun': all_constraints, 'args': (N,)}
    
    # Multiple restarts to find the best local optimum
    for restart in range(10):
        v0 = np.zeros(3 * N)
        for idx in range(N):
            v0[3*idx] = base_centers[idx][0] + (np.random.rand() - 0.5) * 0.04
            v0[3*idx+1] = base_centers[idx][1] + (np.random.rand() - 0.5) * 0.04
            v0[3*idx+2] = 0.098 + (np.random.rand() - 0.5) * 0.005
            
        res = minimize(objective, v0, method='SLSQP', bounds=bounds_list, 
                       constraints=constraint, options={'maxiter': 150, 'ftol': 1e-10, 'disp': False})
        
        if not np.isnan(res.fun):
            current_sum_r = -res.fun
            if current_sum_r > best_sum_r:
                best_sum_r = current_sum_r
                best_v = res.x.copy()
                
    # Fallback to initial grid if optimization fails
    if best_v is None:
        best_v = np.zeros(3 * N)
        for idx in range(N):
            best_v[3*idx] = base_centers[idx][0]
            best_v[3*idx+1] = base_centers[idx][1]
            best_v[3*idx+2] = 0.09
            
    centers = np.array([[best_v[3*i], best_v[3*i+1]] for i in range(N)])
    radii = np.array([best_v[3*i+2] for i in range(N)])
    
    return centers, radii, np.sum(radii)
