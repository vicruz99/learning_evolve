import numpy as np
from scipy.optimize import minimize

def compute_constraints(vars, n):
    """Computes constraint slacks for boundary and non-overlap conditions."""
    res = []
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i+1]
        r = np.exp(vars[3*i+2])
        
        # Boundary constraints: x-r >= 0, 1-(x+r) >= 0, y-r >= 0, 1-(y+r) >= 0
        res.append(x - r)
        res.append(1.0 - x - r)
        res.append(y - r)
        res.append(1.0 - y - r)
        
        # Overlap constraints: dist(i,j) - (ri + rj) >= 0
        for j in range(i+1, n):
            xj = vars[3*j]
            yj = vars[3*j+1]
            rj = np.exp(vars[3*j+2])
            dist = np.sqrt((x - xj)**2 + (y - yj)**2)
            res.append(dist - r - rj)
    return np.array(res)

def constraint_fun(vars):
    """Wrapper for SLSQP constraint function."""
    return compute_constraints(vars, 26)

def objective(vars):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(np.exp(vars[2::3]))

def run_optimization(seed):
    """Runs SLSQP optimization from a perturbed hexagonal initialization."""
    np.random.seed(seed)
    n = 26
    init_vars = np.zeros(3 * n)
    r0 = 0.08
    rho0 = np.log(r0)
    
    # Hexagonal grid initialization
    placed = 0
    row, col = 0, 0
    while placed < n:
        x = 0.15 + col * 0.22
        y = 0.15 + row * 0.19
        if col % 2 == 1:
            y += 0.095
        if 0.05 < x < 0.95 and 0.05 < y < 0.95:
            init_vars[3*placed] = x + np.random.uniform(-0.01, 0.01)
            init_vars[3*placed+1] = y + np.random.uniform(-0.01, 0.01)
            init_vars[3*placed+2] = rho0 + np.random.uniform(-0.3, 0.3)
            placed += 1
        col += 1
        if col > 5:
            col = 0
            row += 1

    cons = {'type': 'ineq', 'fun': constraint_fun}
    res = minimize(objective, init_vars, method='SLSQP', 
                   constraints=cons, options={'maxiter': 1200, 'ftol': 1e-10})
    
    if res.success:
        best_vars = res.x
    else:
        best_vars = init_vars
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = np.exp(best_vars[2::3])
    return centers, radii

def check_validity(centers, radii, n, tol=1e-9):
    """Fast feasibility check."""
    for i in range(n):
        if radii[i] < 0: return False
        if centers[i][0] - radii[i] < -tol or centers[i][0] + radii[i] > 1 + tol: return False
        if centers[i][1] - radii[i] < -tol or centers[i][1] + radii[i] > 1 + tol: return False
    for i in range(n):
        for j in range(i+1, n):
            dist = np.hypot(centers[i][0]-centers[j][0], centers[i][1]-centers[j][1])
            if dist < radii[i] + radii[j] - tol:
                return False
    return True

def run_packing():
    n = 26
    best_centers, best_radii, best_sum = None, None, -1.0
    
    for seed in range(20):
        centers, radii = run_optimization(seed)
        
        # Quick validity check
        if not check_validity(centers, radii, n, tol=1e-7):
            continue
            
        curr_sum = np.sum(radii)
        
        # Binary search to uniformly scale radii to the feasibility boundary
        low, high = 1.0, 1.6
        for _ in range(25):
            mid = (low + high) / 2.0
            if check_validity(centers, radii * mid, n, tol=1e-9):
                low = mid
            else:
                high = mid
        radii *= low
        curr_sum = np.sum(radii)
        
        if curr_sum > best_sum:
            best_centers, best_radii, best_sum = centers.copy(), radii.copy(), curr_sum
            
    # Final safety adjustment to strictly satisfy 1e-12 tolerance
    if best_radii is not None:
        scale = 1.0
        for _ in range(10):
            if check_validity(best_centers, best_radii, n, tol=1e-12):
                break
            scale *= 0.995
        best_radii *= scale
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
