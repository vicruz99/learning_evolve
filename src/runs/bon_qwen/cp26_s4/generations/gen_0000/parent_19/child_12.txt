import numpy as np
from scipy.optimize import minimize

def compute_cost(vars, P, n):
    centers = vars[:2*n].reshape(-1, 2)
    radii = vars[2*n:]
    
    # Objective: maximize sum of radii => minimize negative sum
    cost = -np.sum(radii)
    
    # Boundary penalties: ensure circles stay within [0,1]x[0,1]
    cost += P * np.sum(np.maximum(0.0, radii - centers[:, 0])**2)
    cost += P * np.sum(np.maximum(0.0, radii - centers[:, 1])**2)
    cost += P * np.sum(np.maximum(0.0, centers[:, 0] + radii - 1.0)**2)
    cost += P * np.sum(np.maximum(0.0, centers[:, 1] + radii - 1.0)**2)
    
    # Overlap penalties: ensure dist >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = r_sum - dists
    
    # Consider only unique pairs (upper triangle)
    tri = np.triu(np.ones((n, n), dtype=bool), k=1)
    cost += P * np.sum(np.maximum(0.0, overlap[tri])**2)
    
    return cost

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Initialization: 5x5 grid + 1 extra circle, with small random noise
    centers_init = []
    for i in range(5):
        for j in range(5):
            centers_init.append([0.1 + j*0.18, 0.1 + i*0.18])
    centers_init.append([0.05, 0.05])
    centers_init = np.array(centers_init)
    centers_init += np.random.normal(0, 0.005, centers_init.shape)
    
    radii_init = np.full(n, 0.04)
    x0 = np.concatenate([centers_init.ravel(), radii_init])
    
    # Bounds for centers in [0, 1] and radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    x_best = x0
    # Continuation method: gradually increase penalty to enforce constraints
    penalty_sequence = [50, 200, 1000, 5000, 20000, 50000]
    for P in penalty_sequence:
        res = minimize(compute_cost, x_best, args=(P, n), 
                       method='L-BFGS-B', bounds=bounds,
                       options={'tol': 1e-10, 'maxiter': 5000})
        x_best = res.x
        
    centers = x_best[:2*n].reshape(-1, 2)
    radii = x_best[2*n:]
    
    # Ensure strict non-negativity and boundary compliance
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    return centers, radii, float(np.sum(radii))
