import numpy as np
from scipy.optimize import minimize

def compute_overlap_penalty(centers, radii):
    """Compute sum of squared overlaps between all circle pairs."""
    n = centers.shape[0]
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0.0:
                penalty += overlap**2
    return penalty

def optimize_centers_objective(centers_flat, radii):
    """Objective function for scipy optimizer."""
    n = len(radii)
    centers = centers_flat.reshape(n, 2)
    return compute_overlap_penalty(centers, radii)

def run_trial(init_centers, init_radii, n_iter=40):
    """Run alternating optimization for a single trial."""
    centers = init_centers.copy()
    radii = init_radii.copy()
    n = 26
    
    for _ in range(n_iter):
        # Step 1: Optimize centers to minimize overlaps
        bounds = [(0.0, 1.0)] * (n * 2)
        res = minimize(
            optimize_centers_objective,
            centers.flatten(),
            args=(radii,),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 150, 'ftol': 1e-10, 'gtol': 1e-8}
        )
        centers = res.x.reshape(n, 2)
        
        # Step 2: Update radii to maximum possible given current centers
        new_radii = np.zeros(n)
        for i in range(n):
            # Boundary constraints
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            # Inter-circle constraints

            new_radii[i] = max(0.0, max_r)
        radii = new_radii
        
    return centers, radii
