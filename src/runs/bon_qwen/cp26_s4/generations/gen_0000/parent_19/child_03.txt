import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_random_packing(n):
    """Generate a random initial packing."""
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.01
    return centers, radii

def get_hex_packing(n):
    """Generate a hexagonal initial packing."""
    centers = []
    radii = []
    # Estimate radius for n circles in square
    # Density approx pi/sqrt(12) ~ 0.9069
    # n * pi * r^2 <= 0.9069 => r <= sqrt(0.9069 / (n*pi))
    r_est = np.sqrt(0.85 / (n * np.pi)) # slightly less conservative
    
    row = 0
    while len(centers) < n:
        y = r_est + row * (r_est * np.sqrt(3))
        if y + r_est > 1:
            break
        # Shift every other row
        x_start = r_est
        if row % 2 == 1:
            x_start += r_est
        
        x = x_start
        while x + r_est <= 1 + 1e-9 and len(centers) < n:
            centers.append([x, y])
            radii.append(r_est)
            x += 2 * r_est
        row += 1
        
    # If we didn't fit n, or fit too many, adjust
    # This is just an initial guess, radii will be optimized
    # Ensure we have exactly n
    while len(centers) < n:
        # Add random circles
        centers.append(np.random.rand(2))
        radii.append(0.01)
        
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    return centers, radii

def optimize_packing(centers_init, radii_init):
    """Optimize centers and radii to maximize sum of radii."""
    n = len(radii_init)
    
    # Flatten variables: x1, y1, r1, x2, y2, r2, ...
    x0 = np.hstack([centers_init.flatten(), radii_init])
    
    # Bounds: x,y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r
    
    # Constraints
    constraints = []
    
    # Boundary constraints
    # x - r >= 0 => r - x <= 0
    # x + r <= 1 => x + r - 1 <= 0
    # y - r >= 0 => r - y <= 0
    # y + r <= 1 => y + r - 1 <= 0
    def make_ineq_constraint(fun):
        return {'type': 'ineq', 'fun': fun}

    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        constraints.append(make_ineq_constraint(lambda v, i=i: v[idx_x] - v[idx_r]))
        # 1 - (x + r) >= 0
        constraints.append(make_ineq_constraint(lambda v, i=i: 1 - (v[idx_x] + v[idx_r])))
        # y - r >= 0
        constraints.append(make_ineq_constraint(lambda v, i=i: v[idx_y] - v[idx_r]))
        # 1 - (y + r) >= 0
        constraints.append(make_ineq_constraint(lambda v, i=i: 1 - (v[idx_y] + v[idx_r])))

    # Non-overlap constraints
    # dist >= r_i + r_j => dist^2 - (r_i + r_j)^2 >= 0
    # dist^2 = (xi-xj)^2 + (yi-yj)^2
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3 * i, 3 * i + 1, 3 * i + 2
            idx_xj, idx_yj, idx_rj = 3 * j, 3 * j + 1, 3 * j + 2
            
            def non_overlap(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                return dist_sq - sum_r**2
            
            constraints.append(make_ineq_constraint(non_overlap))

    # Objective: Maximize sum of radii => Minimize -sum(radii)
    def objective(v):
        radii = v[2::3]
        return -np.sum(radii)

    # Use SLSQP
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        if res.success or res.fun > -2.5: # Check if it improved or is reasonable
            centers_opt = np.resize(res.x[:2*n], (n, 2))
            radii_opt = res.x[2*n:]
            return centers_opt, radii_opt, -res.fun
    except Exception:
        pass
        
    return centers_init, radii_init, np.sum(radii_init)
