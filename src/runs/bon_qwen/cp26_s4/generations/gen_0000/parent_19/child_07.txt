        y = vars[1::3]
        r = vars[2::3]

            con.append(y[i] - r[i])           # y - r >= 0

        return con

        # Constraint: dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0

        y = vars[1::3]
