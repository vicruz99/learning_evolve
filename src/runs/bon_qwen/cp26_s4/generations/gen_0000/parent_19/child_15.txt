import numpy as np
from scipy.optimize import minimize

def objective(params, n=26, weight=10000.0):
    """
    Objective function for circle packing optimization.
    Minimizes negative sum of radii plus penalty for constraint violations.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Boundary penalties: ensure circles stay within [0,1]x[0,1]
    p_left   = np.sum(np.maximum(radii - centers[:,0], 0)**2)
    p_right  = np.sum(np.maximum(centers[:,0] + radii - 1, 0)**2)
    p_bottom = np.sum(np.maximum(radii - centers[:,1], 0)**2)
    p_top    = np.sum(np.maximum(centers[:,1] + radii - 1, 0)**2)
    bound_pen = p_left + p_right + p_bottom + p_top
    
    # Overlap penalties: ensure distance between centers >= sum of radii
    # Vectorized pairwise computation for efficiency
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(c_diff**2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap is positive when circles intersect
    overlap = r_sum - dist
    np.fill_diagonal(overlap, 0)
    overlap_pen = np.sum(np.maximum(overlap, 0)**2)
    
    # Objective: maximize sum of radii (minimize negative sum) + weighted penalties
    return -np.sum(radii) + weight * (bound_pen + overlap_pen)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes positions and radii of 26 circles in a unit square.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_fun = np.inf
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    
    # Multi-start optimization with diverse initial configurations
    for trial in range(6):
        if trial == 0:
            # Grid initialization (5x5 grid + 1 point) with small perturbation
            pts = []
            for i in range(5):
                for j in range(5):
                    pts.append([0.1 + i*0.2, 0.1 + j*0.2])
            pts.append([0.9, 0.9])
            centers_init = np.array(pts) + np.random.normal(0, 0.01, (n, 2))
        else:
            # Random initialization
            centers_init = np.random.rand(n, 2)
            
        radii_init = np.ones(n) * 0.05
        params_init = np.concatenate([centers_init.flatten(), radii_init])
        
        # Variable bounds: x,y in [0,1], r in [epsilon, 0.5]
        bounds = [(0, 1)] * (2*n) + [(1e-5, 0.5)] * n
        
        res = minimize(
            objective, 
            params_init, 
            args=(n, 10000.0), 
            method='L-BFGS-B', 
            bounds=bounds, 
            options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-8}
        )
        
        # Track the best result (lowest objective value)
        if res.fun < best_fun:
            best_fun = res.fun
            best_centers = res.x[:2*n].reshape(n, 2).copy()
            best_radii = res.x[2*n:].copy()
            
    # Post-processing to guarantee strict validity within numerical tolerance
    best_centers = np.clip(best_centers, 1e-7, 1.0 - 1e-7)
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        # Shrink radius if it touches boundaries too closely
        margin = 1e-9
        r = min(r, x - margin, 1.0 - x - margin, y - margin, 1.0 - y - margin)
        best_radii[i] = max(1e-9, r)
        
    total_radius = float(np.sum(best_radii))
    return best_centers, best_radii, total_radius
