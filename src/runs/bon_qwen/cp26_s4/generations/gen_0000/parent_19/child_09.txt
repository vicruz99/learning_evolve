import numpy as np
from scipy.optimize import minimize
import math

def compute_loss(params, n):
    """
    Computes the objective function: -sum(radii) + penalty for constraint violations.
    params: 1D array of shape [x1, y1, ..., xn, yn, r1, ..., rn]
    """
    centers = params[:2*n].reshape((n, 2))
    radii = params[2*n:]
    
    # Primary objective: maximize sum of radii
    loss = -np.sum(radii)
    penalty_weight = 300.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Distance to walls
        v_left = max(0.0, r - x)
        v_right = max(0.0, x + r - 1.0)
        v_bottom = max(0.0, r - y)
        v_top = max(0.0, y + r - 1.0)
        loss += penalty_weight * (v_left**2 + v_right**2 + v_bottom**2 + v_top**2)
        
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            overlap = max(0.0, radii[i] + radii[j] - dist)
            loss += penalty_weight * overlap**2
            
    return loss

def generate_initial_params(n, seed_offset=0):
    """Generates an initial configuration using a hexagonal lattice."""
    np.random.seed(seed_offset)
    # Hexagonal spacing
    row_spacing = 1.0 / 6.0
    col_spacing = math.sqrt(3.0) / 2.0 * row_spacing
    
    candidates = []
    for i in range(8):
        y = 0.1 + i * row_spacing
        for j in range(8):
            x = 0.1 + j * col_spacing + (0.5 * col_spacing if i % 2 == 1 else 0.0)
            if 0 <= x <= 1 and 0 <= y <= 1:
                candidates.append((x, y))
                
    # Shuffle and pick n
    np.random.shuffle(candidates)
    candidates = candidates[:n]
    
    centers = np.array(candidates)
    # Initial radii set to a safe small value allowing room to grow
    radii = np.full(n, 0.06)
    
    return np.concatenate([centers.ravel(), radii])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_loss = float('inf')
    best_params = None
    
    # Bounds: x,y in [0,1], r in [0.001, 0.5]
    bounds_centers = [(0.0, 1.0)] * (2 * n)
    bounds_radii = [(0.001, 0.5)] * n
    bounds = bounds_centers + bounds_radii
    
    # Multiple restarts to avoid local minima
    for k in range(15):
        params = generate_initial_params(n, seed_offset=k*42)
        try:
            res = minimize(
                compute_loss,
                params,
                args=(n,),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            if res.fun < best_loss:
                best_loss = res.fun
                best_params = res.x.copy()
        except Exception:
            continue
            
    # Extract results
    centers = best_params[:2*n].reshape((n, 2))
    radii = best_params[2*n:]
    
    # Post-processing: Ensure strict validity by slight uniform scaling if needed
    valid = False
    safety_iter = 0
    while not valid and safety_iter < 50:
        valid = True
        # Check boundaries
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or 
                y - r < -1e-12 or y + r > 1 + 1e-12):
                valid = False
                break
        if not valid:
            radii *= 0.98
            continue
            
        # Check overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.hypot(dx, dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if not valid:
            radii *= 0.99
            safety_iter += 1
            
    total_radius = np.sum(radii)
    return centers, radii, total_radius
