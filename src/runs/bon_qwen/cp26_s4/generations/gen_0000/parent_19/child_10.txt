import numpy as np


def compute_forces(centers, radii):
    """Compute repulsion and boundary forces for all circles."""
    n = centers.shape[0]
    forces = np.zeros((n, 2))
    
    for i in range(n):
        # Boundary forces
        for axis in range(2):
            r = radii[i]
            margin = 0.001
            if centers[i][axis] - r < margin:
                forces[i][axis] += (margin - (centers[i][axis] - r)) * 100
            if centers[i][axis] + r > 1 - margin:
                forces[i][axis] -= ((centers[i][axis] + r) - (1 - margin)) * 100
        
        # Repulsion forces from other circles
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq)
            min_dist = radii[i] + radii[j] + 0.002
            
            if dist < min_dist and dist > 1e-12:
                force_mag = (min_dist - dist) * 30
                fx = dx / dist * force_mag
                fy = dy / dist * force_mag
                forces[i][0] += fx
                forces[i][1] += fy
                forces[j][0] -= fx
                forces[j][1] -= fy
    
    return forces


def update_radii(centers, radii):
    """Update radii based on available space."""
    n = centers.shape[0]
    new_radii = radii.copy()
    
    for i in range(n):
        min_gap = 1.0
        
        # Distance to other circles
        for j in range(n):
            if i == j:
                continue
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            gap = dist - radii[j]
            if gap < min_gap:
                min_gap = gap
        
        # Distance to boundaries
        min_gap = min(min_gap, centers[i][0], 1 - centers[i][0],
                     centers[i][1], 1 - centers[i][1])
        
        # Gradually increase radius
        growth_factor = 1.0002
        target_r = radii[i] * growth_factor
        new_radii[i] = min(target_r, min_gap)
    
    return new_radii


def cleanup_packing(centers, radii):
    """Ensure all constraints are satisfied."""
    n = centers.shape[0]
    
    # Clip to boundaries
    for i in range(n):
        r_max = min(centers[i][0], 1 - centers[i][0],
                   centers[i][1], 1 - centers[i][1])
        radii[i] = min(radii[i], max(0, r_max - 1e-13))
    
    # Resolve overlaps iteratively
    for _ in range(500):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_required = radii[i] + radii[j] - 1e-12
                
                if dist < min_required:
                    # Reduce radii proportionally to maintain sum
                    total_r = radii[i] + radii[j]
                    if total_r > 0:
                        scale = dist / total_r
                        radii[i] = max(0, radii[i] * scale)
                        radii[j] = max(0, radii[j] * scale)
                    else:
                        radii[i] = 0
                        radii[j] = 0
                    changed = True
        if not changed:
            break
    
    # Final boundary check
    for i in range(n):
        r_max = min(centers[i][0], 1 - centers[i][0],
                   centers[i][1], 1 - centers[i][1])
        radii[i] = min(radii[i], max(0, r_max - 1e-13))
    
    return centers, radii


def hexagonal_init(n):
    """Create initial hexagonal packing configuration."""
    r0 = 0.095
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r0
    
    idx = 0
    row = 0
    while idx < n:
        cols = 5 if row % 2 == 0 else 4
        for col in range(cols):
            if idx >= n:
                break
            x = r0 + col * 2 * r0 + (row % 2) * r0
            y = r0 + row * np.sqrt(3) * r0
            if x + r0 <= 1.0 and y + r0 <= 1.0:
                centers[idx] = [x, y]
                idx += 1
        row += 1
    
    # Fill remaining circles
    while idx < n:
        placed = False
        for row_fill in range(8):
            if placed:
                break
            cols = 5 if row_fill % 2 == 0 else 4
            for col in range(cols):
                if idx >= n:
                    placed = True
                    break
                x = r0 + col * 2 * r0 + (row_fill % 2) * r0
                y = r0 + row_fill * np.sqrt(3) * r0
                if x + r0 <= 1.0 and y + r0 <= 1.0:
                    already_placed = False
                    for k in range(idx):
                        dx = centers[k][0] - x
                        dy = centers[k][1] - y
                        if dx * dx + dy * dy < (2 * r0) ** 2:
                            already_placed = True
                            break
                    if not already_placed:
                        centers[idx] = [x, y]
                        idx += 1
            if placed:
                break
    
    return centers, radii


def random_init(n):
    """Create random initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07
    
    for i in range(n):
        centers[i][0] = np.random.uniform(0.15, 0.85)
        centers[i][1] = np.random.uniform(0.15, 0.85)
        radii[i] = 0.06 + np.random.uniform(0, 0.04)
    
    return centers, radii


def grid_init(n):
    """Create grid-based initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09
    
    idx = 0
    for row in range(5):
        for col in range(5):
            if idx < n:
                centers[idx][0] = 0.1 + col * 0.2
                centers[idx][1] = 0.1 + row * 0.2
                idx += 1
    
    return centers, radii


def run_packing():
    np.random.seed(42)
    n = 26
    
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Try multiple initial configurations
    init_configs = [hexagonal_init(n), random_init(n), grid_init(n)]
    
    for restart in range(10):
        if restart < 3:
            centers, radii = init_configs[restart]
        else:
            centers, radii = random_init(n)
            np.random.seed(42 + restart)
        
        # Force-directed optimization
        for iteration in range(8000):
            forces = compute_forces(centers, radii)
            
            # Annealed learning rate
            learning_rate = 0.003 / (1 + iteration / 800)
            
            # Update positions
            centers += forces * learning_rate
            centers = np.clip(centers, 0.0005, 0.9995)
            
            # Update radii periodically
            if iteration % 5 == 0:
                radii = update_radii(centers, radii)
        
        # Cleanup
        centers, radii = cleanup_packing(centers, radii)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii, best_sum
