import numpy as np
from scipy.optimize import minimize

def calculate_violation_energy(centers, radii, penalty_weight=1000.0):
    """
    Calculates the penalty energy based on constraint violations.
    """
    n = centers.shape[0]
    violation = 0.0
    
    # Boundary violations: circle must be inside [0,1]x[0,1]
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x + r - 1 <= 0
    # y - r >= 0  => r - y <= 0
    # y + r <= 1  => y + r - 1 <= 0
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    # Violation terms (squared max of 0 and violation amount)
    # If r - x > 0, circle is sticking out left
    v_left = np.maximum(0, r - x)
    v_right = np.maximum(0, x + r - 1)
    v_bottom = np.maximum(0, r - y)
    v_top = np.maximum(0, y + r - 1)
    
    boundary_violation = np.sum(v_left**2) + np.sum(v_right**2) + \
                         np.sum(v_bottom**2) + np.sum(v_top**2)
    
    # Overlap violations: distance between centers >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # If (r_i + r_j)^2 - dist^2 > 0, there is overlap
    # We compute overlap depth: (r_i + r_j) - dist
    # To avoid sqrt in loops, we can work with squared distances, but 
    # calculating penetration depth is easier with linear distance for intuition.
    # However, for smoothness and vectorization:
    
    # Vectorized pairwise distance calculation
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # radii sum matrix shape (n, n)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # We only care about upper triangle to avoid double counting and self-interaction
    # dist >= r_sum
    # Violation = max(0, r_sum - dist)
    # We need dist, so sqrt(dist_sq). Add epsilon to avoid div by zero or sqrt(0) issues if needed,
    # but here dist_sq should be positive if centers distinct.
    dist = np.sqrt(np.maximum(dist_sq, 1e-12))
    
    overlap_depth = np.maximum(0, r_sum - dist)
    
    # Sum squared violations for upper triangle
    # Create mask for upper triangle (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap_violation = np.sum((overlap_depth[mask])**2)
    
    total_violation = boundary_violation + overlap_violation
    return penalty_weight * total_violation

def objective_function(params, n, penalty_weight=1000.0):
    """
    Objective function to minimize: -sum(radii) + penalty.
    """
    x = params[:n]
    y = params[n:2*n]
    r = params[2*n:]
    
    centers = np.column_stack((x, y))
    radii = r
    
    # Negative sum of radii (we want to maximize sum of radii)
    score = -np.sum(radii)
    
    # Add penalty for violations
    penalty = calculate_violation_energy(centers, radii, penalty_weight)
    
    return score + penalty
