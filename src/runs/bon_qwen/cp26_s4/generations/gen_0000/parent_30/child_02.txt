import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Maximizes the sum of radii.
    """
    N = 26
    
    # --- 1. Initialization: Hexagonal Grid ---
    # We aim for a configuration close to the densest packing.
    # A 6-row hexagonal pattern with row counts [5, 4, 5, 4, 5, 3] sums to 26.
    
    # Initial radius estimate. 
    # For 5 circles in width: 10*r <= 1 => r <= 0.1
    # For 6 rows in height: (2 + 5*sqrt(3))*r <= 1 => r <= 0.0938
    # We choose 0.09 to be safe.
    r_init = 0.09
    row_counts = [5, 4, 5, 4, 5, 3]
    
    centers_init = []
    radii_init = []
    
    for row_idx, count in enumerate(row_counts):
        # Vertical position: base r + row_index * vertical_spacing
        # Vertical spacing in hexagonal packing is sqrt(3)/2 * diameter = sqrt(3) * r
        y_pos = r_init + row_idx * (np.sqrt(3) * r_init)
        
        # Horizontal positions
        # Even rows (0, 2, 4) start at r_init.
        # Odd rows (1, 3, 5) are shifted by r_init (half diameter) to nest in gaps.
        x_start = r_init
        if row_idx % 2 == 1:
            x_start = 2 * r_init
            
        # Generate x coordinates with spacing 2*r (diameter)
        # x_k = x_start + k * 2*r
        x_coords = np.linspace(x_start, x_start + (count - 1) * 2 * r_init, count)
        
        for x in x_coords:
            centers_init.append([x, y_pos])
            radii_init.append(r_init)
    
    centers_init = np.array(centers_init)
    radii_init = np.array(radii_init)
    
    # Ensure we have exactly N circles
    if len(centers_init) < N:
        # Fallback: Fill with small circles in available spots if logic failed (unlikely)
        while len(centers_init) < N:
            centers_init = np.vstack([centers_init, [[0.5, 0.5]]])
            radii_init = np.append(radii_init, 0.01)
    
    centers_init = centers_init[:N]
    radii_init = radii_init[:N]
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * N)
    x0[0::3] = centers_init[:, 0]
    x0[1::3] = centers_init[:, 1]
    x0[2::3] = radii_init
    
    # Bounds: x, y in [0, 1], r in [0, 0.5] (0.5 is theoretical max for single circle)
    bounds = []
    for i in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])

    # --- 2. Objective Function ---
    def objective(vars):
        # vars is flattened array. Radii are at indices 2, 5, 8, ... (2::3)
        r = vars[2::3]
        return -np.sum(r) # Minimize negative sum = Maximize sum

    # --- 3. Constraints ---
    def constraints(vars):
        # Reshape to (N, 3) for easier access
        # Each row: [x, y, r]
        # But vars is flat. Let's extract arrays.
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        cons = []
        
        # Boundary Constraints:
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        cons.extend(x - r)
        cons.extend(1.0 - x - r)
        cons.extend(y - r)
        cons.extend(1.0 - y - r)
        
        # Overlap Constraints:
        # dist^2 >= (r_i + r_j)^2
        # dist^2 - (r_i + r_j)^2 >= 0
        # We compute this for all pairs (i, j) with i < j
        # To optimize speed, we can vectorize, but for N=26 loops are fine.
        for i in range(N):
            ri = r[i]
            xi = x[i]
            yi = y[i]
            for j in range(i + 1, N):
                rj = r[j]
                xj = x[j]
                yj = y[j]
                
                dx = xi - xj
                dy = yi - yj
                dr = ri + rj
                
                dist_sq = dx*dx + dy*dy
                dr_sq = dr*dr
                
                cons.append(dist_sq - dr_sq)
                
        return np.array(cons)

    # --- 4. Optimization ---
    # Using SLSQP for constrained non-linear optimization
    # We define constraints as a dictionary for scipy.optimize
    cons_dict = {
        'type': 'ineq',
        'fun': constraints
    }

    # Run optimization
    # We might run it a few times to avoid local minima, but one run from hexagonal grid is strong.
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons_dict, 
                      options={'maxiter': 1000, 'ftol': 1e-9})
    
    # If the first run didn't find a great solution (e.g. got stuck), 
    # we could try a perturbation, but the hexagonal start is usually very good.
    # Let's extract result
    final_vars = result.x
    
    final_centers = np.zeros((N, 2))
    final_centers[:, 0] = final_vars[0::3]
    final_centers[:, 1] = final_vars[1::3]
    
    final_radii = final_vars[2::3]
    
    sum_radii = np.sum(final_radii)
    
    # Ensure radii are non-negative (though bounds should handle it)
    final_radii = np.maximum(final_radii, 0.0)
    
    return final_centers, final_radii, sum_radii

# Helper for validation check if needed, though run_packing is the requirement.
