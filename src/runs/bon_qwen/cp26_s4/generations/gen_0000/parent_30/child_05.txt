import numpy as np
from scipy.optimize import linprog, minimize
import warnings

# Suppress warnings for cleaner output
warnings.filterwarnings('ignore')

def solve_radii_lp(centers):
    """
    Solves the LP to find the maximum sum of radii for a fixed set of centers.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        radii: np.array of shape (n)
        sum_radii: float
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0

    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: Maximize sum(r_i) <=> Minimize -sum(r_i)
    c_obj = -np.ones(n)

    # Constraints: A_ub * r <= b_ub
    # 1. Wall constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    # 2. Pairwise constraints: r_i + r_j <= dist(c_i, c_j)

    # We will build lists of rows for the matrix A_ub and values for b_ub
    rows = []
    bounds_vals = []

    # Wall constraints
    for i in range(n):
        x, y = centers[i]
        # Distance to boundaries
        d_wall = min(x, 1 - x, y, 1 - y)
        if d_wall < 0:
            d_wall = 0 # Center outside, radius must be 0 (or infeasible, but we handle gracefully)
        
        # Constraint: r_i <= d_wall
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        bounds_vals.append(d_wall)

    # Pairwise constraints
    # r_i + r_j <= dist(i, j)
    # Distance calculation
    # Using broadcasting for efficiency
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = dists[i, j]
            if d_ij < 0: 
                d_ij = 0
            
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            bounds_vals.append(d_ij)

    A_ub = np.array(rows)
    b_ub = np.array(bounds_vals)

    # Bounds for variables: 0 <= r_i < infinity
    x_bounds = [(0, None)] * n

    # Solve LP
    # 'highs' is a robust and fast solver available in recent scipy
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=x_bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback if LP fails (should be rare)
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def objective_function(centers_flat):
    """
    Objective function for center optimization.
    Minimizes negative sum of radii.
    """
    centers = centers_flat.reshape(26, 2)
    # Ensure centers are within [0, 1] strictly for validity, though bounds handle this
    centers = np.clip(centers, 0.0, 1.0)
    
    radii, sum_r = solve_radii_lp(centers)
    return -sum_r

def generate_hexagonal_grid(n_circles):
    """Generates a hexagonal packing layout for initialization."""
    centers = []
    # Approximate radius for hex packing in unit square ~ 0.08-0.1
    # Spacing 2r ~ 0.2
    dx = 0.2
    dy = dx * np.sqrt(3) / 2
    
    y = 0.1
    row = 0
    while len(centers) < n_circles:
        offset = (row % 2) * (dx / 2)
        x = 0.1 + offset
        while x <= 0.9 and len(centers) < n_circles:
            centers.append([x, y])
            x += dx
        y += dy
        row += 1
    return np.array(centers[:n_circles])

def generate_grid(n_circles):
    """Generates a square grid layout."""
    centers = []
    # 5x5 grid gives 25 points, spacing 0.2
    # We need 26.
    # Let's use a 5x6 grid but take first 26?
    # Or just perturb 5x5 + 1
    x_vals = np.linspace(0.1, 0.9, 5)
    y_vals = np.linspace(0.1, 0.9, 5)
    
    for x in x_vals:
        for y in y_vals:
            centers.append([x, y])
            if len(centers) >= 26:
                return np.array(centers[:26])
            
    # If still need more (unlikely for 25), add random
    while len(centers) < n_circles:
        centers.append([0.5, 0.5]) # Fallback
    return np.array(centers[:26])

def run_packing():
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    # Define bounds for centers: [0, 1] for each coordinate
    bounds = [(0, 1)] * (2 * n)

    # List of initial configurations to try
    initial_configs = []

    # 1. Square Grid (perturbed)
    cfg1 = generate_grid(n)
    cfg1 += np.random.uniform(-0.02, 0.02, size=cfg1.shape)
    initial_configs.append(cfg1)

    # 2. Hexagonal Grid (perturbed)
    cfg2 = generate_hexagonal_grid(n)
    cfg2 += np.random.uniform(-0.02, 0.02, size=cfg2.shape)
    initial_configs.append(cfg2)

    # 3. Random initialization
    np.random.seed(123)
    cfg3 = np.random.uniform(0.05, 0.95, size=(n, 2))
    initial_configs.append(cfg3)
    
    # 4. Another random seed
    cfg4 = np.random.uniform(0.05, 0.95, size=(n, 2))
    initial_configs.append(cfg4)

    # Optimization Loop
    for i, start_centers in enumerate(initial_configs):
        x0 = start_centers.flatten()
        
        # Use L-BFGS-B which supports bounds and is efficient
        # Options: maxiter limits function evaluations
        try:
            res = minimize(objective_function, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 500, 'ftol': 1e-8})
            
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x.reshape(n, 2)
        except Exception as e:
            continue

    # Final calculation of radii for the best centers found
    # Re-run LP to get exact radii
    final_radii, final_sum = solve_radii_lp(best_centers)
    
    # Clip centers to be strictly within [0,1] for safety against numerical drift
    best_centers = np.clip(best_centers, 0.0, 1.0)

    return best_centers, final_radii, final_sum
