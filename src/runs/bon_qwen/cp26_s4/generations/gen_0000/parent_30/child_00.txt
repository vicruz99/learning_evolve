import numpy as np

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # --- Initialization ---
    # Generate a hexagonal grid pattern for initial centers.
    # This provides a dense packing structure to start from.
    # We aim for roughly 5-6 rows.
    
    centers = np.zeros((n, 2))
    idx = 0
    
    # Create a hexagonal lattice
    # Spacing roughly 0.15 initially to allow room for growth
    spacing = 0.18 
    
    # We will generate points and pick the best fit or just fill rows
    # Let's try to fill rows with 6, 5, 6, 5, 4 circles = 26
    # Row shifts: even rows at x=0, odd rows at x=spacing/2
    
    rows_config = [6, 5, 6, 5, 4]
    y_curr = 0.1
    
    for r_idx, count in enumerate(rows_config):
        shift = (spacing / 2) if r_idx % 2 == 1 else 0
        x_start = 0.1 + shift
        
        for c_idx in range(count):
            if idx < n:
                x = x_start + c_idx * spacing
                # Ensure we stay within bounds roughly
                if x <= 0.9: 
                    centers[idx] = [x, y_curr]
                    idx += 1
        
        y_curr += spacing * np.sqrt(3) / 2
    
    # If we didn't fill 26, fill remaining randomly (should not happen with config)
    if idx < n:
        remaining = n - idx
        # Randomly place in remaining space
        for i in range(remaining):
            centers[idx + i] = np.random.rand(2) * 0.8 + 0.1

    # Initial radii
    radii = np.ones(n) * 0.05

    # --- Simulation Parameters ---
    dt = 1.0          # Time step (learning rate for position)
    dr = 0.0005       # Learning rate for radius growth
    max_iters = 15000 # Number of simulation steps
    repulsion_k = 100 # Stiffness of repulsion
    wall_k = 100      # Stiffness of wall repulsion
    growth_k = 50     # Pressure to grow radii
    damping = 0.95    # Velocity damping for stability
    
    velocities = np.zeros((n, 2))
    
    # Precompute pair indices for efficiency
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    pairs = np.array(pairs)

    # --- Optimization Loop ---
    for step in range(max_iters):
        forces = np.zeros((n, 2))
        radius_forces = np.zeros(n)
        
        # 1. Calculate Repulsive Forces between circles
        # Vectorized computation for speed
        i_indices = pairs[:, 0]
        j_indices = pairs[:, 1]
        
        ci = centers[i_indices]
        cj = centers[j_indices]
        
        # Displacement vectors (j - i)
        diff = cj - ci
        dists = np.linalg.norm(diff, axis=1)
        
        # Avoid division by zero
        dists = np.maximum(dists, 1e-9)
        
        # Sum of radii
        r_sum = radii[i_indices] + radii[j_indices]
        
        # Overlap amount
        overlap = r_sum - dists
        
        # Only apply force if overlapping
        mask = overlap > 0
        
        if np.any(mask):
            # Force magnitude proportional to overlap
            # F = k * overlap
            # Direction: i pushed away from j (-diff direction), j pushed away from i (+diff direction)
            
            force_mag = repulsion_k * overlap[mask]
            # Normalize direction
            norm_diff = diff[mask] / dists[mask].reshape(-1, 1)
            
            force_vec = force_mag.reshape(-1, 1) * norm_diff
            
            # Accumulate forces using np.add.at to handle multiple updates per index
            np.add.at(forces, i_indices[mask], -force_vec)
            np.add.at(forces, j_indices[mask], force_vec)
            
            # Penalty for radius growth if overlapping heavily? 
            # Maybe shrink radii slightly to help resolve?
            # radius_forces[i_indices[mask]] -= 0.1 * overlap[mask]
            # radius_forces[j_indices[mask]] -= 0.1 * overlap[mask]

        # 2. Calculate Boundary Forces
        # Left wall (x=0)
        left_penetration = radii - centers[:, 0]
        left_mask = left_penetration > 0
        if np.any(left_mask):
            forces[left_mask, 0] += wall_k * left_penetration[left_mask]
        
        # Right wall (x=1)
        right_penetration = centers[:, 0] + radii - 1
        right_mask = right_penetration > 0
        if np.any(right_mask):
            forces[right_mask, 0] -= wall_k * right_penetration[right_mask]
            
        # Bottom wall (y=0)
        bottom_penetration = radii - centers[:, 1]
        bottom_mask = bottom_penetration > 0
        if np.any(bottom_mask):
            forces[bottom_mask, 1] += wall_k * bottom_penetration[bottom_mask]
            
        # Top wall (y=1)
        top_penetration = centers[:, 1] + radii - 1
        top_mask = top_penetration > 0
        if np.any(top_mask):
            forces[top_mask, 1] -= wall_k * top_penetration[top_mask]

        # 3. Growth Force (Pressure to increase radii)
        # We want to maximize sum(r), so we add a constant positive force to each radius.
        # However, if a circle is blocked, it can't grow. 
        # The repulsion forces will push it out.
        radius_forces += growth_k

        # 4. Update Positions (Velocity Verlet-like integration)
        # a = F
        # v += a * dt
        # x += v * dt
        
        # Actually simple Euler integration with damping
        velocities += forces * dt
        velocities *= damping
        centers += velocities * dt
        
        # Update Radii
        radii += radius_forces * dr
        radii = np.maximum(radii, 0) # Ensure non-negative
        
        # Optional: Annealing - reduce forces over time to settle?
        # Not strictly necessary if damping is good.

    # --- Final Validation and Cleanup ---
    # The simulation might leave tiny overlaps due to numerical precision or dynamics.
    # We perform a final projection/correction step.
    
    # 1. Enforce boundaries strictly
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        # Clamp center to allow radius r
        centers[i, 0] = np.clip(x, r, 1 - r)
        centers[i, 1] = np.clip(y, r, 1 - r)
        
    # 2. Resolve remaining overlaps by shrinking radii if necessary
    # This is a safety net. Ideally, the simulation resolves them.
    # We check all pairs again.
    # If overlap, reduce radii proportionally to penetration.
    
    # Simple iterative shrink for safety
    for _ in range(100):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    penetration = r_sum - dist
                    # Shrink both radii slightly to fix overlap
                    # Distribute shrinkage
                    shrink = penetration * 0.51 
                    radii[i] -= shrink
                    radii[j] -= shrink
                    radii[i] = max(0, radii[i])
                    radii[j] = max(0, radii[j])
                    if penetration > max_violation:
                        max_violation = penetration
        
        if max_violation < 1e-6:
            break
            
    # Recalculate sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)
