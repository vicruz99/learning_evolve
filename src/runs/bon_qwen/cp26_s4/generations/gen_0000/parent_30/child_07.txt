import numpy as np
from scipy.optimize import minimize

def compute_loss_and_grad(vars, n, mu):
    """
    Compute the loss function and its gradient for the circle packing problem.
    Loss = -sum(radii) + mu * (overlap_penalties + boundary_penalties)
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Objective: minimize negative sum of radii
    loss = -np.sum(radii)
    grad = np.zeros_like(vars)
    grad[2*n:] = -1.0  # Gradient of -sum(radii)
    
    # Overlap penalties between all pairs
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            dist = max(dist, 1e-12)
            
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                # Penalty term: mu * overlap^2
                loss += mu * overlap * overlap
                
                # Gradient components
                g = 2.0 * mu * overlap
                
                # Centers: repulsion force
                grad[2*i]   -= g * dx / dist
                grad[2*i+1] -= g * dy / dist
                grad[2*j]   += g * dx / dist
                grad[2*j+1] += g * dy / dist
                
                # Radii: increase penalty when radii sum is too large
                grad[2*n+i] += g
                grad[2*n+j] += g

    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary: x - r >= 0
        v = x - r
        if v < 0:
            loss += mu * v * v
            gv = 2.0 * mu * v
            grad[2*i] += gv
            grad[2*n+i] -= gv
            
        # Right boundary: 1 - x - r >= 0
        v = 1.0 - x - r
        if v < 0:
            loss += mu * v * v
            gv = 2.0 * mu * v
            grad[2*i] -= gv
            grad[2*n+i] -= gv
            
        # Bottom boundary: y - r >= 0
        v = y - r
        if v < 0:
            loss += mu * v * v
            gv = 2.0 * mu * v
            grad[2*i+1] += gv
            grad[2*n+i] -= gv
            
        # Top boundary: 1 - y - r >= 0
        v = 1.0 - y - r
        if v < 0:
            loss += mu * v * v
            gv = 2.0 * mu * v
            grad[2*i+1] -= gv
            grad[2*n+i] -= gv
            
    return loss, grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initial configuration: random points within [0.05, 0.95] to avoid immediate boundary issues
    np.random.seed(42)
    centers = np.random.uniform(0.05, 0.95, size=(n, 2))
    radii = np.full(n, 0.08)
    
    vars = np.concatenate([centers.ravel(), radii])
    bounds = [(0, 1)] * 2*n + [(0, 0.5)] * n
    
    best_vars = vars.copy()
    mu = 50.0
    
    # Iterative penalty method
    for _ in range(40):
        res = minimize(compute_loss_and_grad, best_vars, args=(n, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-9})
        best_vars = res.x
        mu *= 1.3
        
        # Early exit if target is comfortably exceeded
        if np.sum(best_vars[2*n:]) > 2.64:
            break
            
    # Extract results
    centers_out = best_vars[:2*n].reshape(n, 2)
    radii_out = best_vars[2*n:]
    
    # Ensure non-negativity and bounds
    radii_out = np.clip(radii_out, 0, None)
    centers_out = np.clip(centers_out, 0, 1)
    
    # Post-processing: if any circle violates constraints due to numerical tolerance, shrink slightly
    # This is a safety net; the optimizer should have handled it.
    for i in range(n):
        x, y = centers_out[i]
        r = radii_out[i]
        margin = max(0, r - x, 0) + max(0, r - (1-x), 0) + max(0, r - y, 0) + max(0, r - (1-y), 0)
        if margin > 1e-9:
            radii_out[i] -= margin + 1e-9
            
    sum_r = np.sum(radii_out)
    return centers_out, radii_out, sum_r
