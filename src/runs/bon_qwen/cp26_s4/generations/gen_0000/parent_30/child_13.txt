import numpy as np
from scipy.optimize import minimize

def compute_objective(v, N):
    """
    Objective function to maximize sum of radii with penalty for constraint violations.
    v: flattened array [x0, y0, x1, y1, ..., r0, r1, ...]
    """
    cx = v[:2*N].reshape(N, 2)
    cr = v[2*N:]
    
    # Boundary penalties: circles must stay inside [0,1]x[0,1]
    p = np.sum(np.maximum(0, cr - cx[:, 0])**2)
    p += np.sum(np.maximum(0, cr - (1.0 - cx[:, 0]))**2)
    p += np.sum(np.maximum(0, cr - cx[:, 1])**2)
    p += np.sum(np.maximum(0, cr - (1.0 - cx[:, 1]))**2)
    
    # Overlap penalties: distance between centers >= sum of radii
    diff = cx[:, np.newaxis, :] - cx[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)  # Ignore self-distance
    
    rad_sum = cr[:, np.newaxis] + cr[np.newaxis, :]
    overlap = np.maximum(0, rad_sum - dist)
    p += np.sum(overlap**2)
    
    # High penalty weight ensures constraints are respected
    return -np.sum(cr) + 5000.0 * p

def run_packing():
    N = 26
    best_score = -1e9
    best_vars = None
    
    # Multiple restarts to avoid poor local optima
    for seed in range(8):
        np.random.seed(seed)
        
        # Hexagonal-like initialization
        cols = int(np.ceil(np.sqrt(N * np.sqrt(3) / 2)))
        rows = int(np.ceil(N / cols))
        dx = 1.0 / (cols + 1)
        dy = 1.0 / (rows + 1) * np.sqrt(3) / 2
        
        idx = 0
        init_x = np.zeros((N, 2))
        for r_idx in range(rows):
            shift = 0.5 * dx if r_idx % 2 == 1 else 0.0
            for c_idx in range(cols):
                if idx < N:
                    init_x[idx, 0] = dx * (c_idx + 1) + shift
                    init_x[idx, 1] = dy * (r_idx + 1)
                    idx += 1
                    
        # Add small random perturbation
        init_x += np.random.normal(0, 0.02, size=(N, 2))
        init_x = np.clip(init_x, 0.05, 0.95)
        
        init_r = np.ones(N) * 0.02
        vars0 = np.concatenate([init_x.flatten(), init_r])
        
        # Bounds: centers in [0,1], radii in [0, 0.5]
        bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        
        res = minimize(
            compute_objective, 
            vars0, 
            args=(N,), 
            method='L-BFGS-B', 
            bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-10}
        )
        
        if -res.fun > best_score:
            best_score = -res.fun
            best_vars = res.x
            
    cx = best_vars[:2*N].reshape(N, 2)
    cr = best_vars[2*N:]
    
    # Ensure non-negative radii (optimizer respects bounds, but safeguard)
    cr = np.maximum(cr, 0.0)
    
    return cx, cr, np.sum(cr)
