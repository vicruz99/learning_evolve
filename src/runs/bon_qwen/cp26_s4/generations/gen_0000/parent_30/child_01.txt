2.  **Optimization**: Use `scipy.optimize.minimize` with the `SLSQP` method.

    *   **Constraints**:

        *   Circles must not overlap ($d_{ij}^2 \ge (r_i + r_j)^2$).
    *   **Vectorization**: The constraints (especially pairwise non-overlap) are evaluated using vectorized NumPy operations for efficiency.
3.  **Refinement**: The optimizer will adjust centers and radii to push against the constraints, effectively "inflating" the circles until they are jammed against each other and the boundaries.
4.  **Fallback**: If optimization fails or yields an invalid result, a fallback hexagonal packing with a safe radius (e.g., 0.1) will be returned to ensure validity.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Initialize centers and radii
    # Generate a hexagonal lattice pattern for 26 points
    # We aim for a layout roughly like 5 rows with varying counts (e.g., 5, 5, 5, 5, 6)
    # to approximate a hexagonal packing density.
    
    row_counts = [5, 5, 5, 5, 6]
    temp_points = []
    
    # Use a base spacing of 1.0 for generation, will scale later
    spacing = 1.0
    
    for r_idx, count in enumerate(row_counts):
        y = r_idx * (np.sqrt(3) / 2) * spacing
        shift = (r_idx % 2) * (spacing / 2)
        for c_idx in range(count):
            x = c_idx * spacing + shift
            temp_points.append([x, y])
            
    temp_arr = np.array(temp_points)
    
    # Normalize and scale to fit within [0.15, 0.85] initially
    # This ensures we start well inside the bounds
    min_vals = temp_arr.min(axis=0)
    max_vals = temp_arr.max(axis=0)
    
    # Calculate scaling factor to fit within 0.15 to 0.85 range
    range_x = max_vals[0] - min_vals[0]
    range_y = max_vals[1] - min_vals[1]
    
    # Target range is 0.7 (from 0.15 to 0.85)
    scale_x = 0.7 / range_x
    scale_y = 0.7 / range_y
    scale = min(scale_x, scale_y)
    
    centers = (temp_arr - min_vals) * scale + 0.15
    
    # Initial radii: estimate based on minimum distance between points
    # We use a safe initial radius to ensure no overlaps initially
    dists = []
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            dists.append(d)
    min_dist = min(dists)
    init_r = min_dist / 2 * 0.95  # Slightly smaller than half min dist
    
    radii = np.full(n, init_r)
    
    # Initial state vector: [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
    # Actually, let's keep centers and radii separate in logic but flatten for solver
    # Order: x1, y1, r1, x2, y2, r2, ... is easier to index
    # But solver vars are usually flat. Let's do [x1, y1, ..., xn, yn, r1, ..., rn]
    
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5] (since diameter <= 1)
    # r can technically be up to 0.5 if it's the only circle, but with 26, it's much smaller.
    # Setting r bound to 0.5 is safe.
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
    # Note: x0 structure is [x1, y1, ..., xn, yn, r1, ..., rn]
    # My x0 construction was centers.flatten() then radii.
    # centers.flatten() gives x1, y1, x2, y2...
    # So bounds should match.
    # Let's reconstruct bounds to match x0 structure.
    
    bounds = []
    for _ in range(2 * n):
        bounds.append((0, 1)) # x and y
    for _ in range(n):
        bounds.append((0, 0.5)) # r
        
    # 2. Define objective and constraints
    
    def objective(vars):
        # vars: [x1, y1, ..., xn, yn, r1, ..., rn]
        radii_vals = vars[2 * n:]
        return -np.sum(radii_vals)  # Minimize negative sum
        
    def constraints(vars):
        # vars: [x1, y1, ..., xn, yn, r1, ..., rn]
        centers = vars[:2 * n].reshape(n, 2)
        radii_vals = vars[2 * n:]
        
        c_list = []
        
        # Boundary constraints:
        # x - r >= 0  => x - r
        # 1 - (x + r) >= 0 => 1 - x - r
        # y - r >= 0
        # 1 - (y + r) >= 0
        
        # Using numpy operations
        c_list.extend(centers[:, 0] - radii_vals)
        c_list.extend(1.0 - centers[:, 0] - radii_vals)
        c_list.extend(centers[:, 1] - radii_vals)
        c_list.extend(1.0 - centers[:, 1] - radii_vals)
        
        # Pairwise non-overlap constraints:
        # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
        # => dist_sq - (ri + rj)^2 >= 0
        
        # Vectorized distance squared calculation
        # centers shape (n, 2)
        # dx[i, j] = xi - xj
        dx = centers[:, 0, np.newaxis] - centers[:, 0, np.newaxis].T
        dy = centers[:, 1, np.newaxis] - centers[:, 1, np.newaxis].T
        dist_sq = dx**2 + dy**2
        
        # r_sum[i, j] = ri + rj
        r_sum = radii_vals[:, np.newaxis] + radii_vals[np.newaxis, :]
        
        # Constraint value
        pair_constraints = dist_sq - r_sum**2
        
        # We only need upper triangle (i < j)
        # Extract upper triangular part without diagonal
        # Using np.triu_indices
        triu_indices = np.triu_indices(n, k=1)
        c_list.extend(pair_constraints[triu_indices])
        
        return np.array(c_list)
        
    # 3. Run Optimization
    # Using SLSQP which handles bounds and constraints
    try:
        # Set options for convergence
        options = {'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        
        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraints},
            options=options
        )
        
        if result.success or (result.fun > -2.6): # If it found a good sum
            final_centers = result.x[:2 * n].reshape(n, 2)
            final_radii = result.x[2 * n:]
        else:
            # If optimization didn't succeed well, fall back to initial or refined initial
            # But usually SLSQP returns the best found even if not converged
            final_centers = result.x[:2 * n].reshape(n, 2)
            final_radii = result.x[2 * n:]
            
    except Exception as e:
        # Fallback if solver fails completely
        final_centers = centers
        final_radii = radii

    # 4. Validation and Clipping
    # Ensure non-negative radii and within bounds
    final_radii = np.maximum(final_radii, 0)
    # Clip centers to [0, 1] just in case
    final_centers = np.clip(final_centers, 0, 1)
    
    # Re-check and fix boundary violations by shrinking radii
    # If a circle is out of bounds, reduce radius
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        # Distance to boundaries
        min_dist_boundary = min(x, 1-x, y, 1-y)
        if r > min_dist_boundary:
            final_radii[i] = max(0, min_dist_boundary)
            
    # Re-check overlaps and resolve by shrinking smaller circle
    # Simple iterative resolution
    changed = True
    iterations = 0
    while changed and iterations < 10:
        changed = False
        iterations += 1
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(final_centers[i] - final_centers[j])
                sum_r = final_radii[i] + final_radii[j]
                if dist < sum_r - 1e-9:
                    # Overlap detected
                    # Shrink the smaller radius or both
                    # Heuristic: shrink proportionally to overlap
                    overlap = sum_r - dist
                    # Shrink both equally
                    shrink = overlap / 2 + 1e-6
                    final_radii[i] = max(0, final_radii[i] - shrink)
                    final_radii[j] = max(0, final_radii[j] - shrink)
                    changed = True

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
