1.  **Configuration via Linear Programming:** For any given set of 26 centers, the optimal radii that satisfy all non-overlap and boundary constraints can be found by solving a linear program (LP). This transforms the complex geometric constraints into a fast, solvable linear system.

import numpy as np

def get_sum_radii(centers):
    """
    Solves the LP for a fixed set of centers to find max sum of radii.
    Returns the sum of radii.
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0

    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        # Ensure boundary distance is non-negative (clamp just in case)
        dist = min(x, 1.0 - x, y, 1.0 - y)
        if dist < 0: dist = 0
        
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(dist)
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    # We only need to consider pairs that are reasonably close or all of them.
    # For n=26, 325 constraints is small enough for LP.
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff**2))
            
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun
        else:
            # If LP fails (e.g., infeasible, though shouldn't be with r>=0), return 0
            return 0.0
    except Exception:
        return 0.0
