import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
import itertools

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # ---------------------------------------------------------
    # Helper: Greedy Maximin Initialization
    # ---------------------------------------------------------
    def generate_maximin_centers(n, seed=42):
        rng = np.random.default_rng(seed)
        centers = []
        # Place first point at center or random
        centers.append([0.5, 0.5])
        
        for _ in range(1, n):
            # Sample random points to find the best spot
            candidates = rng.random((1000, 2))
            best_pos = None
            best_val = -1.0
            
            for c in candidates:
                x, y = c
                # Distance to boundary
                dist_boundary = min(x, 1-x, y, 1-y)
                
                # Distance to existing centers
                if len(centers) > 0:
                    dist_centers = np.min(np.sqrt(np.sum((c - centers)**2, axis=1)))
                else:
                    dist_centers = 1.0
                
                val = min(dist_boundary, dist_centers)
                if val > best_val:
                    best_val = val
                    best_pos = c
                    
            centers.append(best_pos)
        return np.array(centers)

    # ---------------------------------------------------------
    # Helper: Solve LP for radii given centers
    # ---------------------------------------------------------
    def solve_radii_lp(centers):
        """
        Solves: Max sum(r_i)
        s.t. r_i + r_j <= dist(i,j)
             2*r_i <= dist_to_boundary(i)
             r_i >= 0
        """
        n = centers.shape[0]
        
        # Objective: minimize -sum(r)
        c_obj = -np.ones(n)
        
        # Constraints matrix A_ub @ r <= b_ub
        # We collect all inequality constraints
        
        A_ub = []
        b_ub = []
        
        # Pairwise constraints: r_i + r_j <= d_ij
        # r_i coeff 1, r_j coeff 1
        # Indices
        indices = np.arange(n)
        pairs = list(itertools.combinations(indices, 2))
        
        # Precompute distances
        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=2))
        
        # Boundary constraints: 2*r_i <= dist_to_boundary
        x = centers[:, 0]
        y = centers[:, 1]
        dist_boundary = np.minimum(np.minimum(x, 1-x), np.minimum(y, 1-y))
        
        # Build A_ub and b_ub
        # Number of constraints: len(pairs) + n
        n_constraints = len(pairs) + n
        
        A_ub = np.zeros((n_constraints, n))
        b_ub = np.zeros(n_constraints)
        
        idx = 0
        for i, j in pairs:
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
        for i in range(n):
            A_ub[idx, i] = 2.0
            b_ub[idx] = dist_boundary[i]
            idx += 1
            
        # Bounds for r_i >= 0
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                return -res.fun, res.x
            else:
                return 0.0, np.zeros(n)
        except Exception:
            return 0.0, np.zeros(n)

    # ---------------------------------------------------------
    # Objective function for center optimization
    # ---------------------------------------------------------
    def objective_func(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        # Clip centers to [0, 1] just in case optimizer goes out
        centers = np.clip(centers, 1e-9, 1 - 1e-9)
        
        sum_r, radii = solve_radii_lp(centers)
        # We want to maximize sum_r, so minimize -sum_r
        return -sum_r

    # ---------------------------------------------------------
    # Main Optimization Loop

    best_sum_radii = -1.0
    best_centers = None
    best_radii = None

    inits = []

    inits.append(generate_maximin_centers(n, seed=42))
    inits.append(generate_maximin_centers(n, seed=123))

    # 5x5 grid is 25 points, add 1
    grid_x = np.linspace(0.1, 0.9, 5)

    grid_centers = []
    for x in grid_x:
        for y in grid_y:
            grid_centers.append([x, y])
    # Add one in a gap
    grid_centers.append([0.05, 0.05]) # Corner gap
    inits.append(np.array(grid_centers))
    
    # 3. Hexagonal-ish
    hex_centers = []
    r_est = 0.09
    y_curr = r_est
    row = 0
    while len(hex_centers) < n:
        if row % 2 == 0:
            x_start = r_est
            count = 6
        else:
            x_start = r_est + r_est # offset
            count = 5
        
        x_curr = x_start
        while len(hex_centers) < n and x_curr <= 1 - r_est:
            hex_centers.append([x_curr, y_curr])
            x_curr += 2 * r_est
        
        y_curr += np.sqrt(3) * r_est
        row += 1
    inits.append(np.array(hex_centers))

    # 4. Random
    inits.append(np.random.random((n, 2)))

    for init_centers in inits:
        # Perturb slightly
        centers = init_centers + np.random.normal(0, 0.001, init_centers.shape)
        centers = np.clip(centers, 1e-4, 1-1e-4)
        
        # Optimize centers using Nelder-Mead
        # Bounds for centers are implicit via clipping in objective, 
        # but Nelder-Mead doesn't support bounds well. 
        # We rely on the penalty/clipping in objective.
        
        x0 = centers.flatten()
        
        # Run optimization
        # Maxiter can be increased, but Nelder-Mead is slow in high dims. 
        # 52 variables.
        res = minimize(objective_func, x0, method='Nelder-Mead',

        opt_centers = res.x.reshape(-1, 2)
        opt_sum, opt_radii = solve_radii_lp(opt_centers)
        
        if opt_sum > best_sum_radii:
            best_sum_radii = opt_sum
            best_centers = opt_centers
            best_radii = opt_radii

    # Final Validation Check (Mental or print)
    # The problem requires returning the result.
    
    return best_centers, best_radii, best_sum_radii
