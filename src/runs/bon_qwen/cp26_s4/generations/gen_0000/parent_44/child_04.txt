import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty-based optimization approach with multiple random restarts to escape local minima.
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Penalty weight for constraint violations
    # We will use a large weight to enforce constraints strictly
    penalty_weight = 1000.0

    def calculate_cost(params):
        """
        Calculates the cost function to be minimized.
        Cost = -Sum(Radii) + Penalty(Constraint Violations)
        """
        # Reshape params to (n, 3)
        pts = params.reshape(n, 3)
        centers = pts[:, :2]
        radii = pts[:, 2]

        # Ensure radii are non-negative (soft constraint via penalty, but we can clip for stability if needed)
        # However, optimization should handle it.
        
        sum_radii = np.sum(radii)
        cost = -sum_radii

        # 1. Boundary Constraints
        # x - r >= 0  =>  r - x <= 0
        # x + r <= 1  =>  x + r - 1 <= 0
        # y - r >= 0  =>  r - y <= 0
        # y + r <= 1  =>  y + r - 1 <= 0
        
        violations = np.maximum(0, radii - centers[:, 0])
        violations += np.maximum(0, centers[:, 0] + radii - 1.0)
        violations += np.maximum(0, radii - centers[:, 1])
        violations += np.maximum(0, centers[:, 1] + radii - 1.0)
        cost += penalty_weight * np.sum(violations ** 2)

        # 2. Overlap Constraints
        # dist(i, j) >= r_i + r_j  =>  r_i + r_j - dist <= 0
        # Calculate pairwise distances
        # Efficient computation of distances
        # centers shape (n, 2)
        # diff shape (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff ** 2, axis=2)
        dist = np.sqrt(np.maximum(dist_sq, 0)) # Avoid negative due to precision
        
        # Lower triangle indices
        i_idx, j_idx = np.tril_indices(n, -1)
        
        # Radii sum for pairs
        r_sum = radii[i_idx] + radii[j_idx]
        
        # Violation: max(0, r_i + r_j - dist)
        overlaps = np.maximum(0, r_sum - dist[i_idx, j_idx])
        cost += penalty_weight * np.sum(overlaps ** 2)

        return cost

    def get_initial_guess(method):
        """Generates an initial configuration."""
        if method == 'grid':
            # Create a grid layout
            # 5x5 grid has 25 points. We need 26.
            # We can use a slightly perturbed grid or a 6x5 grid selection.
            # Let's try to fit 26 points in a grid-like structure.
            # 5 rows, varying columns?
            # Or just random placement in a grid.
            x = np.linspace(0.1, 0.9, 6) # 6 points
            y = np.linspace(0.1, 0.9, 5) # 5 points
            # 30 points. Pick 26.
            grid_points = np.array(np.meshgrid(x, y)).T.reshape(-1, 2)
            # Shuffle and pick 26
            np.random.shuffle(grid_points)
            centers = grid_points[:n]
            radii = np.full(n, 0.05) # Initial radius
        elif method == 'hex':
            # Hexagonal packing attempt
            centers = []
            r_init = 0.05
            # 5 rows
            for row in range(5):
                y = r_init + row * (r_init * np.sqrt(3))
                if row % 2 == 0:
                    x_start = r_init
                    count = 6 # Try to fit more
                else:
                    x_start = r_init + r_init # offset
                    count = 5
                
                for col in range(count):
                    x = x_start + col * (2 * r_init)
                    if x + r_init <= 1.0 and len(centers) < n:
                        centers.append([x, y])
            
            # If not enough, fill randomly
            while len(centers) < n:
                centers.append([np.random.rand(), np.random.rand()])
            centers = np.array(centers[:n])
            radii = np.full(n, 0.05)
        else: # 'random'
            centers = np.random.rand(n, 2)
            radii = np.full(n, 0.02)
        
        # Flatten
        params = np.hstack([centers, radii.reshape(-1, 1)])
        return params.flatten()

    bounds = [(0, 1) for _ in range(n)] + \
             [(0, 1) for _ in range(n)] + \
             [(0, 0.5) for _ in range(n)] # x, y, r bounds

    # Run optimization multiple times
    strategies = ['grid', 'hex', 'random', 'random', 'random']
    
    for strat in strategies:
        params0 = get_initial_guess(strat)
        
        # Use L-BFGS-B for speed, but it doesn't handle non-linear constraints well directly.
        # We rely on the penalty function in objective.
        # SLSQP might be better but slower. Let's try L-BFGS-B with high penalty.
        
        try:
            res = minimize(calculate_cost, params0, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-9})
            
            params_opt = res.x.reshape(n, 3)
            c_opt = params_opt[:, :2]
            r_opt = params_opt[:, 2]
            
            # Check validity roughly
            valid = True
            # Check bounds
            if np.any(r_opt < 0): valid = False
            if np.any(c_opt < -1e-9) or np.any(c_opt > 1 + 1e-9): valid = False
            if np.any(r_opt + c_opt > 1 + 1e-9) or np.any(c_opt - r_opt < -1e-9): valid = False # Boundary
            
            # Check overlaps (quick check)
            dists = np.sqrt(((c_opt[:, None, :] - c_opt[None, :, :]) ** 2).sum(axis=2))
            np.fill_diagonal(dists, np.inf)
            min_dist = dists.min()
            sum_r_pairs = r_opt[None, :] + r_opt[:, None]
            # We need dist >= sum_r
            # Check if any dist < sum_r - epsilon
            if np.any(dists < sum_r_pairs - 1e-7):
                valid = False
            
            if valid:
                current_sum = np.sum(r_opt)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
                    print(f"Strategy {strat}: Sum = {current_sum:.5f}")
        except Exception as e:
            print(f"Optimization failed for {strat}: {e}")

    # Fallback to a known good grid if optimization fails or is poor
    # 25 circles grid sum = 2.5. Target 2.636.
    # If we didn't reach target, try to refine.
    # But the optimizer should find better.
    
    # Let's do one more specific optimization starting from the best found so far
    # to polish the solution.
    if best_centers is not None:
        params_final = np.hstack([best_centers, best_radii.reshape(-1, 1)]).flatten()
        try:
            res_final = minimize(calculate_cost, params_final, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-12})
            params_opt = res_final.x.reshape(n, 3)
            c_opt = params_opt[:, :2]
            r_opt = params_opt[:, 2]
            
            # Validate again
            valid = True
            if np.any(r_opt < 0): valid = False
            if np.any(c_opt < -1e-9) or np.any(c_opt > 1 + 1e-9): valid = False
            if np.any(r_opt + c_opt > 1 + 1e-9) or np.any(c_opt - r_opt < -1e-9): valid = False
            
            dists = np.sqrt(((c_opt[:, None, :] - c_opt[None, :, :]) ** 2).sum(axis=2))
            np.fill_diagonal(dists, np.inf)
            sum_r_pairs = r_opt[None, :] + r_opt[:, None]
            if np.any(dists < sum_r_pairs - 1e-7):
                valid = False
            
            if valid:
                current_sum = np.sum(r_opt)
                best_sum = current_sum
                best_centers = c_opt
                best_radii = r_opt
                print(f"Refined Sum = {current_sum:.5f}")
        except:
            pass

    # If result is still not great, fallback to a heuristic 25-circle grid + 1 small circle?
    # But sum would be ~2.54.
    # The optimizer should handle variable radii.
    
    # Ensure no NaN
    if best_centers is None or np.isnan(best_centers).any() or np.isnan(best_radii).any():
        # Absolute fallback: 5x5 grid
        centers = []
        radii = []
        r = 0.1
        for i in range(5):
            for j in range(5):
                centers.append([0.1 + i*0.2, 0.1 + j*0.2])
                radii.append(r)
        # Add 26th circle?
        # Try to fit a tiny one in center? Center is (0.5, 0.5) occupied.
        # Maybe (0.15, 0.15) occupied.
        # Just return 25 circles? No, must be 26.
        # Shrink grid to fit 26?
        # If we can't find good solution, return valid 26 circles.
        # 26 circles of radius 0.09 in grid?
        # 5x5 grid is 25.
        # Let's just output the best found.
        pass

    return best_centers, best_radii, best_sum
