import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the circle packing problem for 26 circles in a unit square
    to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None
    
    # Helper to extract arrays from flattened variables
    def get_arrays(vars):
        centers = vars.reshape(n, 3)[:, :2]
        radii = vars.reshape(n, 3)[:, 2]
        return centers, radii

    # Objective: minimize negative sum of radii
    def objective(vars):
        radii = vars[2::3]
        return -np.sum(radii)

    # Constraints functions
    def create_constraints():
        constraints = []
        # Boundary and non-negativity constraints for each circle
        for i in range(n):
            # x_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, idx=i: v[3*idx] - v[3*idx+2]
            })
            # 1 - x_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, idx=i: 1.0 - v[3*idx] - v[3*idx+2]
            })
            # y_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, idx=i: v[3*idx+1] - v[3*idx+2]
            })
            # 1 - y_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, idx=i: 1.0 - v[3*idx+1] - v[3*idx+2]
            })
            # r_i >= 0 (though bounds handle this, good for safety)
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, idx=i: v[3*idx+2]
            })
        
        # Non-overlap constraints for each pair
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: 
                        (v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2 - (v[3*i+2] + v[3*j+2])**2
                })
        return constraints

    cons = create_constraints()

        bounds[3*i + 2] = (0, 0.5)

        nonlocal best_sum_radii, best_centers, best_radii
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,

                c, r = get_arrays(res.x)
                # Simple validation check
                valid = True
                if np.any(r < -1e-9): valid = False
                for i in range(n):
                    if c[i,0] - r[i] < -1e-7 or c[i,0] + r[i] > 1 + 1e-7: valid = False
                    if c[i,1] - r[i] < -1e-7 or c[i,1] + r[i] > 1 + 1e-7: valid = False
                if valid:
                    s = np.sum(r)
                    if s > best_sum_radii:
                        best_sum_radii = s
                        best_centers = c.copy()
                        best_radii = r.copy()
        except Exception:
            pass

            x0_grid[3*idx+1] = y
            x0_grid[3*idx+2] = 0.095 # Slightly less to allow movement

    x0_grid[3*25+1] = 0.5
    x0_grid[3*25+2] = 0.01
    try_optimization(x0_grid)

    for _ in range(5):
        x0_rand = np.random.uniform(0.05, 0.95, size=3 * n)
        x0_rand[2::3] = np.random.uniform(0.02, 0.05, size=n) # Small radii
        try_optimization(x0_rand)

    x0_hex = np.zeros(3 * n)
    idx = 0
    r_init = 0.09
    y = r_init
    while idx < n:

        shift = (row_idx % 2) * r_init
        
        x_start = r_init + shift
        # Adjust to fit
        if x_start + (k-1)*2*r_init + r_init > 1.0:
            # Reduce k or adjust
            pass
        
        for i in range(k):
            if idx >= n: break
            x = x_start + i * 2 * r_init
            if x + r_init > 1.0: break
            if x - r_init < 0.0: continue
            
            x0_hex[3*idx] = x
            x0_hex[3*idx+1] = y
            x0_hex[3*idx+2] = 0.08 # Start slightly smaller
            idx += 1
        
        y += np.sqrt(3) * r_init
        
    # Fill remaining if any
    while idx < n:
        x0_hex[3*idx] = 0.5
        x0_hex[3*idx+1] = 0.5
        x0_hex[3*idx+2] = 0.01
        idx += 1

        try_optimization(x0_rand2)

    # Final fallback if optimization failed (should not happen)
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        for i in range(n):
            best_centers[i, 0] = 0.5
            best_centers[i, 1] = 0.5
            best_radii[i] = 0.001
    
    # Ensure return types are correct
    best_centers = np.array(best_centers)
    best_radii = np.array(best_radii)
    sum_radii = float(np.sum(best_radii))
    
    return best_centers, best_radii, sum_radii
