def get_constraints(centers, radii):

    Helper to compute constraint values for scipy.
    Returns a list of dicts for constraints.
    """

    # x_i - r_i >= 0
    # 1 - (x_i + r_i) >= 0
    # y_i - r_i >= 0
    # 1 - (y_i + r_i) >= 0

    def boundary_lower_x(vars_vec, idx):
        x = vars_vec[3 * idx]
        r = vars_vec[3 * idx + 2]
        return x - r

        min_dist = r1 + r2
        return dist_sq - min_dist**2
