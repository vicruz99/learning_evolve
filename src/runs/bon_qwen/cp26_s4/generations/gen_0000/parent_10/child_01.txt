from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty method with L-BFGS-B.

    N = 26

    r_init = 0.04
    centers = []

    y = r_init
    row_count = 0
    while len(centers) < N:

        while x + r_init <= 1.0 and len(centers) < N:
            centers.append([x, y])
            x += 2 * r_init + 0.01 # small gap
        y += 2 * r_init + 0.01
        row_count += 1
        
    centers = np.array(centers[:N])
    radii = np.full(N, r_init)
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for L-BFGS-B
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        return x, y, r

        _, _, r = get_coords_radii(vars)
        # We want to maximize sum(r), so minimize -sum(r)
        return -np.sum(r)

    def penalty(vars):
        x, y, r = get_coords_radii(vars)
        p = 0.0
        
        # Boundary penalties: max(0, violation)^2
        # x >= r  => r - x <= 0
        viol_x_min = np.maximum(0, r - x)
        # x <= 1-r => x + r - 1 <= 0
        viol_x_max = np.maximum(0, x + r - 1.0)
        # y >= r
        viol_y_min = np.maximum(0, r - y)
        # y <= 1-r
        viol_y_max = np.maximum(0, y + r - 1.0)
        
        p += np.sum(viol_x_min**2) + np.sum(viol_x_max**2)
        p += np.sum(viol_y_min**2) + np.sum(viol_y_max**2)
        
        # Overlap penalties
        # dist^2 >= (r_i + r_j)^2
        # violation = (r_i + r_j)^2 - dist^2
        # Use vectorized operations for speed

                dy = y[i] - y[j]
                dist_sq = dx*dx + dy*dy
                r_sum = r[i] + r[j]

                if viol > 0:
                    p += viol**2
        return p

            return objective(vars) + mu * penalty(vars)
        
        # Run optimizer
