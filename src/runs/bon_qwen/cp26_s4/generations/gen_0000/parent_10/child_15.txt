import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def initialize_hexagonal():
    """Initialize circles in hexagonal pattern"""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    
    idx = 0
    row = 0
    while idx < n:
        cols = 6 if row % 2 == 0 else 5
        for col in range(cols):
            if idx >= n:
                break
            x = 0.05 + col * 0.16
            y = 0.05 + row * 0.14
            if row % 2 == 1:
                x += 0.08
            centers[idx] = [x, y]
            idx += 1
        row += 1
    
    return centers

def force_based_optimization(centers, radii, max_iter=3000):
    """Use force-directed method to optimize positions and grow radii"""
    n = len(centers)
    
    for iteration in range(max_iter):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq)
                if dist < 1e-12:
                    dist = 1e-12
                    dx, dy = 1e-6, 1e-6
                
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    strength = overlap * 50
                    fx = (dx / dist) * strength
                    fy = (dy / dist) * strength
                    forces[i] -= np.array([fx, fy])
                    forces[j] += np.array([fx, fy])
            
            r = radii[i]
            if centers[i, 0] < r:
                forces[i, 0] += (r - centers[i, 0]) * 50
            if centers[i, 0] > 1 - r:
                forces[i, 0] -= (centers[i, 0] - (1 - r)) * 50
            if centers[i, 1] < r:
                forces[i, 1] += (r - centers[i, 1]) * 50
            if centers[i, 1] > 1 - r:
                forces[i, 1] -= (centers[i, 1] - (1 - r)) * 50
        
        lr = 0.005 * max(0.1, 1 - iteration / max_iter)
        centers += lr * forces
        
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-9, 1 - radii[i] - 1e-9)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-9, 1 - radii[i] - 1e-9)
        
        radii *= 1.0003
    
    return centers, radii

def gradient_based_optimization(centers, radii):
    """Use SLSQP to optimize positions and radii"""
    n = N_CIRCLES
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(2 * n):
        bounds.append((0, 1))
    for i in range(n):
        bounds.append((0, 0.5))
    
    def obj(x):
        return -np.sum(x[2 * n:])
    
    def cons(x):
        n_loc = N_CIRCLES
        c_locs = x[:2 * n_loc].reshape(n_loc, 2)
        r_locs = x[2 * n_loc:]
        
        result = []
        for i in range(n_loc):
            xi, yi = c_locs[i]
            ri = r_locs[i]
            result.append(xi - ri)
            result.append(1 - xi - ri)
            result.append(yi - ri)
            result.append(1 - yi - ri)
        
        for i in range(n_loc):
            for j in range(i + 1, n_loc):
                dx = c_locs[i, 0] - c_locs[j, 0]
                dy = c_locs[i, 1] - c_locs[j, 1]
                dsq = dx * dx + dy * dy
                mdist = r_locs[i] + r_locs[j]
                result.append(dsq - mdist * mdist)
        
        for i in range(n_loc):
            result.append(r_locs[i])
        
        return np.array(result)
    
    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                     constraints={'type': 'ineq', 'fun': cons},
                     options={'maxiter': 4000, 'ftol': 1e-15})
    
    copts = result.x[:2 * n].reshape(n, 2)
    ropts = result.x[2 * n:]
    ropts = np.maximum(ropts, 0)
    
    for i in range(n):
        copts[i, 0] = np.clip(copts[i, 0], ropts[i] + 1e-9, 1 - ropts[i] - 1e-9)
        copts[i, 1] = np.clip(copts[i, 1], ropts[i] + 1e-9, 1 - ropts[i] - 1e-9)
    
    return copts, ropts

def local_improvement(centers, radii):
    """Try to locally improve by adjusting individual circles"""
    n = N_CIRCLES
    improved = True
    
    while improved:
        improved = False
        for i in range(n):
            best_r = radii[i]
            best_pos = centers[i].copy()
            
            for dr in [0.002, 0.005, 0.01, 0.02]:
                test_r = radii[i] + dr
                test_pos = centers[i].copy()
                
                valid = True
                for j in range(n):
                    if j == i:
                        continue
                    dx = test_pos[0] - centers[j, 0]
                    dy = test_pos[1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < test_r + radii[j] - 1e-10:
                        valid = False
                        break
                
                if valid:
                    if test_pos[0] - test_r < 0 or test_pos[0] + test_r > 1:
                        valid = False
                    if test_pos[1] - test_r < 0 or test_pos[1] + test_r > 1:
                        valid = False
                
                if valid and test_r > best_r:
                    best_r = test_r
                    best_pos = test_pos.copy()
            
            if best_r > radii[i] + 1e-8:
                radii[i] = best_r
                centers[i] = best_pos
                improved = True
    
    return centers, radii

def run_packing():
    n = N_CIRCLES
    
    best_centers = None
    best_radii = None
    best_sum = 0
    
    for trial in range(8):
        centers = initialize_hexagonal()
        if trial > 0:
            noise_scale = 0.04 * np.exp(-trial / 4)
            centers += np.random.randn(n, 2) * noise_scale
            centers = np.clip(centers, 0.05, 0.95)
        
        radii = np.ones(n) * (0.07 + trial * 0.01)
        
        centers, radii = force_based_optimization(centers, radii, max_iter=2500)
        
        centers, radii = gradient_based_optimization(centers, radii)
        
        centers, radii = local_improvement(centers, radii)
        
        if np.sum(radii) > best_sum:
            best_sum = np.sum(radii)
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii, best_sum
