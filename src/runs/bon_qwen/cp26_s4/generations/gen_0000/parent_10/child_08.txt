import numpy as np
from scipy.optimize import minimize

def packing_objective(vars, n, penalty_weight):
    centers = np.array([(vars[3*i], vars[3*i+1]) for i in range(n)])
    radii = np.array([vars[3*i+2] for i in range(n)])
    
    obj = -np.sum(radii)
    penalty = 0.0
    p_grad = np.zeros_like(vars)
    
    # Boundary penalties
    for i in range(n):
        x, y, r = centers[i][0], centers[i][1], radii[i]
        
        # x >= r  => r - x <= 0
        v1 = r - x
        if v1 > 0:
            penalty += v1**2
            p_grad[3*i+2] += 2.0 * v1
            p_grad[3*i] -= 2.0 * v1
            
        # x <= 1 - r => r + x - 1 <= 0
        v2 = r + x - 1
        if v2 > 0:
            penalty += v2**2
            p_grad[3*i+2] += 2.0 * v2
            p_grad[3*i] += 2.0 * v2
            
        # y >= r => r - y <= 0
        v3 = r - y
        if v3 > 0:
            penalty += v3**2
            p_grad[3*i+2] += 2.0 * v3
            p_grad[3*i+1] -= 2.0 * v3
            
        # y <= 1 - r => r + y - 1 <= 0
        v4 = r + y - 1
        if v4 > 0:
            penalty += v4**2
            p_grad[3*i+2] += 2.0 * v4
            p_grad[3*i+1] += 2.0 * v4
            
    # Overlap penalties
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist_sq = dx*dx + dy*dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-12 else 1e-12
            ri, rj = radii[i], radii[j]
            
            v = ri + rj - dist
            if v > 0:
                penalty += v**2
                g = 2.0 * v / dist
                p_grad[3*i+2] += 2.0 * v
                p_grad[3*j+2] += 2.0 * v
                p_grad[3*i]   -= g * dx
                p_grad[3*i+1] -= g * dy
                p_grad[3*j]   += g * dx
                p_grad[3*j+1] += g * dy
                
    obj += penalty_weight * penalty
    grad = penalty_weight * p_grad
    for i in range(n):
        grad[3*i+2] -= 1.0
        
    return obj, grad

def run_packing():
    n = 26
    best_obj = np.inf
    best_vars = None
    
    # Bounds: x, y in [0, 1], r in [1e-4, 0.5]
    bounds = [(0.0, 1.0) for _ in range(3*n)]
    for i in range(n):
        bounds[3*i+2] = (1e-4, 0.5)
        
    penalty_weight = 2000.0
    
    # Run multiple starts to avoid local minima
    for seed in range(7):
        np.random.seed(seed)
        
        # Initialize with hexagonal lattice pattern
        init_vars = np.zeros(3*n)
        idx = 0
        rows_y = [0.1, 0.25, 0.40, 0.55, 0.70, 0.85]
        
        for r_idx, y in enumerate(rows_y):
            if idx >= n: break
            offset = 0.075 if r_idx % 2 == 1 else 0.0
            x = 0.1 + offset
            while x <= 0.9 and idx < n:
                init_vars[3*idx] = x
                init_vars[3*idx+1] = y
                init_vars[3*idx+2] = 0.085
                idx += 1
                x += 0.16
                
        # Fill remaining slots randomly if needed
        while idx < n:
            init_vars[3*idx] = np.random.uniform(0.1, 0.9)
            init_vars[3*idx+1] = np.random.uniform(0.1, 0.9)
            init_vars[3*idx+2] = 0.05
            idx += 1
            
        # Optimize
        res = minimize(packing_objective, init_vars, args=(n, penalty_weight), 
                       jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-9})
        
        if res.fun < best_obj:
            best_obj = res.fun
            best_vars = res.x.copy()
            
    # Extract final configuration
    centers = np.array([(best_vars[3*i], best_vars[3*i+1]) for i in range(n)])
    radii = np.array([best_vars[3*i+2] for i in range(n)])
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
