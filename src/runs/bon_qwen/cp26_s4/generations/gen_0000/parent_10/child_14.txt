2.  **Decoupling Approach**: We decouple the problem into two steps:
    *   **Inner Loop (LP)**: For fixed centers, the problem of maximizing radii is a Linear Programming (LP) problem. We maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and boundary bounds.
    *   **Outer Loop (Heuristic Search)**: We adjust the centers to improve the objective. We use a force-directed relaxation method where circles push each other apart when they are "touching" (active constraints in the LP).
3.  **Initialization**: A hexagonal lattice pattern is used to generate initial centers, as it approximates the densest packing for equal circles.

    *   Iteratively solve the LP to find optimal radii for current centers.
    *   Compute repulsive forces based on active constraints (circles that are touching or near walls).
    *   Update centers by moving them in the direction of the forces.
    *   Decay the step size to converge to a stable configuration.

import numpy as np
from scipy.optimize import linprog
import math
import random

def solve_radii(centers):
    """
    Solves the Linear Programming problem to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to r_i + r_j <= dist(i, j) and boundary constraints.
    """
    n = centers.shape[0]
    
    # Bounds for radii: 0 <= r_i <= distance to nearest wall
    bounds = []
    for i in range(n):
        x, y = centers[i]
        d_wall = min(x, 1-x, y, 1-y)
        bounds.append((0, max(0, d_wall)))
    
    # Constraints: r_i + r_j <= dist(i, j) for all i < j
    # A_ub * r <= b_ub
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            # Calculate distance between centers i and j
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = math.sqrt(dx*dx + dy*dy)
            
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Solve LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback if LP fails (e.g., infeasible due to numerical issues)
            # Return small radii to maintain validity
            return np.full(n, 1e-6)
    except Exception:
        return np.full(n, 1e-6)

def compute_forces(centers, radii, threshold=1e-4):
    """
    Computes forces to push centers apart if constraints are active (tight).
    Active constraints indicate that circles are limiting each other's size.
    Moving them apart allows for larger radii.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]
        
        # Check wall constraints
        # If r_i is close to wall distance, push away from wall
        if r_i > x_i - threshold:
            forces[i, 0] += 1.0 # Push right
        if r_i > (1 - x_i) - threshold:
            forces[i, 0] -= 1.0 # Push left
        if r_i > y_i - threshold:
            forces[i, 1] += 1.0 # Push up
        if r_i > (1 - y_i) - threshold:
            forces[i, 1] -= 1.0 # Push down
            
        # Check pairwise constraints
        for j in range(i + 1, n):
            r_j = radii[j]
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = math.sqrt(dx*dx + dy*dy)
            
            # If circles are touching (r_i + r_j approx dist), push apart
            # Use a slightly looser threshold to prevent sticking
            if r_i + r_j > dist - threshold:
                if dist > 1e-9:
                    fx = dx / dist
                    fy = dy / dist
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
    
    return forces

def run_packing():
    n = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Helper to run optimization from a starting configuration
    def optimize(centers_init, iterations=300, initial_step=0.05):
        centers = centers_init.copy()
        current_step = initial_step
        
        # Initial radii calculation
        radii = solve_radii(centers)
        current_sum = np.sum(radii)
        
        best_local_sum = current_sum
        best_local_centers = centers.copy()
        best_local_radii = radii.copy()
        
        for _ in range(iterations):
            # Recompute radii for current centers
            radii = solve_radii(centers)
            current_sum = np.sum(radii)
            
            if current_sum > best_local_sum:
                best_local_sum = current_sum
                best_local_centers = centers.copy()
                best_local_radii = radii.copy()
            
            # Compute forces
            forces = compute_forces(centers, radii)
            
            # Normalize forces to unit length approximately to control step size
            # Or just scale by step
            # To prevent exploding forces if many constraints active, normalize?
            # Simple scaling is usually fine for this problem size.
            
            # Update centers
            centers = centers + current_step * forces
            
            # Project back to [0, 1]
            centers = np.clip(centers, 0, 1)
            
            # Decay step size
            current_step *= 0.995
            
        return best_local_centers, best_local_radii, best_local_sum

    # Strategy: Multiple restarts with different initial configurations

            # Target 2.636.

    return best_centers, best_radii, float(best_sum_radii)
