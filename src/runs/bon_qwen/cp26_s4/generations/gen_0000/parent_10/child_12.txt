To maximize the sum of radii for 26 circles in a unit square, we need to find a configuration that balances boundary constraints and inter-circle non-overlap constraints. This is a non-convex optimization problem. 

Strategy:
1. **Formulation**: We treat the centers $(x_i, y_i)$ and radii $r_i$ as optimization variables. The objective is to maximize $\sum r_i$, which is equivalent to minimizing $-\sum r_i$.
2. **Constraints**: 
   - Boundary: $x_i - r_i \ge 0$, $1 - x_i - r_i \ge 0$, $y_i - r_i \ge 0$, $1 - y_i - r_i \ge 0$.
   - Non-overlap: $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$ for all $i < j$.
3. **Solver**: We use `scipy.optimize.minimize` with the `SLSQP` method, which handles bound and inequality constraints effectively. 
4. **Initialization**: We use multiple random restarts starting from perturbed grid configurations to escape local minima. Initial radii are set small (0.05) to ensure a feasible starting point.
5. **Post-processing**: After optimization, we slightly shrink radii to strictly satisfy the validation tolerance (`1e-12`) and handle any residual numerical violations. A deterministic grid fallback ensures a valid result even if optimization fails.
6. **Constraints Handling**: We define a single vector-valued constraint function at the top level to comply with the "no closures" rule, computing all boundary and pairwise constraints efficiently.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii"""
    return -np.sum(vars[2::3])

def constraint_fun(vars):
    """Compute all inequality constraints as a vector >= 0"""
    n = N_CIRCLES
    out = []
    
    # Boundary constraints
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        out.append(xi - ri)
        out.append(1.0 - xi - ri)
        out.append(yi - ri)
        out.append(1.0 - yi - ri)
        
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            out.append(dist_sq - (ri + rj)**2)
            
    return np.array(out)

def run_packing():
    global N_CIRCLES
    N_CIRCLES = 26
    n = N_CIRCLES
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': constraint_fun}
    
    def optimize_once(seed_val):
        np.random.seed(seed_val)
        # Initial guess: perturbed 5x5 grid
        xs = np.sort(np.random.uniform(0.05, 0.95, 5))
        ys = np.sort(np.random.uniform(0.05, 0.95, 5))
        pts = [[x, y] for x in xs for y in ys]
        while len(pts) < n:
            pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = pts[i][0]
            x0[3*i+1] = pts[i][1]
            x0[3*i+2] = 0.04
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 500, 'ftol': 1e-9})
            return res.x
        except Exception:
            return None

    # Run multiple restarts
    for seed in [42, 101, 202, 303, 404]:
        res_x = optimize_once(seed)
        if res_x is not None:
            centers = res_x[:2*n].reshape(n, 2)
            radii = res_x[2*n:]
            
            # Quick validity check
            valid = True
            for i in range(n):
                x, y, r = centers[i, 0], centers[i, 1], radii[i]
                if x - r < -1e-7 or x + r > 1.0 + 1e-7 or y - r < -1e-7 or y + r > 1.0 + 1e-7:
                    valid = False; break
                for j in range(i + 1, n):
                    d = np.hypot(x - centers[j, 0], y - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-7:
                        valid = False; break
                if not valid:
                    break
                    
            if valid:
                s = np.sum(radii)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers.copy()
                    best_radii = radii.copy()

    # Fallback deterministic packing if optimization didn't find a good solution
    if best_centers is None or best_sum < 2.55:
        grid = []
        for i in range(5):
            for j in range(5):
                grid.append([0.1 + 0.2 * i, 0.1 + 0.2 * j])
        grid.append([0.5, 0.5])
        best_centers = np.array(grid[:n])
        best_radii = np.full(n, 0.1)
        best_sum = 2.6

    # Final safety shrinkage to strictly satisfy validator tolerances
    if best_centers is not None:
        eps = 1e-9
        for i in range(n):
            x, y = best_centers[i]
            r = best_radii[i]
            # Boundary
            r = min(r, x - eps, 1.0 - x - eps, y - eps, 1.0 - y - eps)
            # Overlaps
            for j in range(i + 1, n):
                d = np.hypot(x - best_centers[j, 0], y - best_centers[j, 1])
                r = min(r, d - best_radii[j] - eps)
            best_radii[i] = max(0.0, r)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
