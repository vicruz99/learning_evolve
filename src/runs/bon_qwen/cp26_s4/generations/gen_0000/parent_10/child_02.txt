import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Assumes equal radii for optimal packing density.
    Returns centers, radii, and sum of radii.
    """
    n_circles = 26
    
    # --- 1. Initial Guess Construction (Hexagonal Lattice) ---
    # We arrange circles in rows with alternating offsets to mimic hexagonal packing.
    # Approximate radius r ~ 0.1. 
    # We estimate grid dimensions to hold 26 circles.
    # A 5x5 grid has 25 circles. Hexagonal allows packing slightly more or tighter.
    # Let's try a configuration with rows of size 6, 5, 6, 5, 4 (Total 26)
    # Or just a dense grid and let optimizer move them.
    
    # Let's generate a hexagonal grid with a reasonable initial spacing.
    # Target density: Area ~ 26 * pi * r^2. If r=0.1, Area ~ 0.8.
    # Square area 1. Density ~ 0.8.
    
    # Initial r estimate
    r_init = 0.10
    spacing_x = 2.0 * r_init
    spacing_y = np.sqrt(3) * r_init
    
    centers_init = []
    row_counts = [6, 5, 6, 5, 4] # Sum = 26
    
    current_y = r_init # Start near bottom boundary
    row_idx = 0
    
    while len(centers_init) < n_circles and row_idx < len(row_counts):
        count = row_counts[row_idx]
        # Center the row horizontally
        row_width = (count - 1) * spacing_x
        start_x = (1.0 - row_width) / 2.0
        
        for i in range(count):
            if len(centers_init) >= n_circles:
                break
            x = start_x + i * spacing_x
            # If row index is odd, shift x by spacing_x / 2 (staggered)
            if row_idx % 2 == 1:
                x += spacing_x / 2.0
            
            centers_init.append([x, current_y])
        
        current_y += spacing_y
        row_idx += 1
        
    centers_init = np.array(centers_init)
    # If we have fewer than 26 (unlikely with this config), pad
    while len(centers_init) < n_circles:
        centers_init = np.vstack([centers_init, [0.5, 0.5]]) # Fallback center
    # If more, trim
    centers_init = centers_init[:n_circles]
    
    # --- 2. Optimization Setup ---
    # Variables: x1, y1, x2, y2, ..., x26, y26, r
    # Total variables: 2 * n_circles + 1 = 53
    
    def objective(vars):
        # We want to maximize r, so minimize -r
        # r is the last variable
        r = vars[-1]
        return -r
    
    def boundary_constraints(vars):
        # Returns array of constraint values >= 0
        # Constraints: x_i >= r, x_i <= 1-r => x_i - r >= 0, 1 - x_i - r >= 0
        #               y_i >= r, y_i <= 1-r => y_i - r >= 0, 1 - y_i - r >= 0
        r = vars[-1]
        centers = vars[:-1].reshape(-1, 2)
        
        cons = []
        # x - r >= 0
        cons.extend(centers[:, 0] - r)
        # 1 - x - r >= 0
        cons.extend(1.0 - centers[:, 0] - r)
        # y - r >= 0
        cons.extend(centers[:, 1] - r)
        # 1 - y - r >= 0
        cons.extend(1.0 - centers[:, 1] - r)
        
        return np.array(cons)

    def overlap_constraints(vars):
        # Returns array of constraint values >= 0
        # Distance^2 >= (2r)^2  => Distance^2 - 4r^2 >= 0
        r = vars[-1]
        centers = vars[:-1].reshape(-1, 2)
        
        cons = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                cons.append(dist_sq - 4 * r**2)
        return np.array(cons)

    # Combine constraints
    def all_constraints(vars):
        c1 = boundary_constraints(vars)
        c2 = overlap_constraints(vars)
        return np.concatenate([c1, c2])

    # Initial guess vector
    x0 = np.hstack([centers_init.flatten(), [r_init]])
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
    bounds.append((0.0, 0.5)) # r

    # Optimization
    # SLSQP is suitable for this type of problem
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': all_constraints},
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    
    if result.success:
        final_vars = result.x
    else:
        # Fallback to initial if optimization fails (unlikely)
        final_vars = x0
        
    final_centers = final_vars[:-1].reshape(-1, 2)
    final_r = final_vars[-1]
    final_radii = np.full(n_circles, final_r)
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
