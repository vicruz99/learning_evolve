2. Boundary containment: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
3. Non-negativity: $r_i \ge 0$.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_constraints_26(x):
    """
    Computes all non-overlap and boundary constraints.
    Returns a 1D array where each element must be >= 0.
    """
    coords = x.reshape(N_CIRCLES, 3)
    pts = coords[:, :2]
    rs = coords[:, 2]
    
    # Pairwise distances using broadcasting: (n, 1, 2) - (1, n, 2) -> (n, n, 2)
    diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    triu_idx = np.triu_indices(N_CIRCLES, k=1)
    pair_vals = dists[triu_idx] - rs[triu_idx[0]] - rs[triu_idx[1]]
    
    xs = coords[:, 0]
    ys = coords[:, 1]
    bound_vals = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    return np.concatenate([pair_vals, bound_vals])

def objective_func(x):
    """Negative sum of radii (minimize negative sum = maximize sum)"""
    return -np.sum(x[2::3])

def run_packing():
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': compute_constraints_26}
    
    np.random.seed(42)
    
    # Run multiple optimizations from different starting points
    for trial in range(5):
        # Grid initialization with random perturbations
        grid_x = np.linspace(0.1, 0.9, 6)
        grid_y = np.linspace(0.1, 0.9, 5)
        x0 = np.zeros((n, 3))
        idx = 0
        for r in range(5):
            for c in range(6):
                if idx >= n: break
                x0[idx, 0] = grid_x[c] + np.random.uniform(-0.02, 0.02)
                x0[idx, 1] = grid_y[r] + np.random.uniform(-0.02, 0.02)
                x0[idx, 2] = 0.04 + np.random.uniform(0, 0.01)
                idx += 1
        
        # Ensure initial positions are safely inside the square
        x0[:, 0] = np.clip(x0[:, 0], 0.05, 0.95)
        x0[:, 1] = np.clip(x0[:, 1], 0.05, 0.95)
        
        x_init = x0.flatten()
        
        try:
            res = minimize(objective_func, x_init, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 1500, 'ftol': 1e-10})
            cur_sum = -res.fun
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = res.x.reshape(n, 3)[:, :2].copy()
                best_radii = res.x.reshape(n, 3)[:, 2].copy()
        except Exception:
            continue
            
    # Fallback if optimization fails completely
    if best_radii is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        
    # Safety clamp for numerical noise
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, best_sum
