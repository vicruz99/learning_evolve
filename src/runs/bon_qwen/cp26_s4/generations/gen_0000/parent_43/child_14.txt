import numpy as np
from scipy.optimize import minimize

def get_initial_guess(n):
    """
    Generates a hexagonal lattice initialization for n circles.
    Arranges circles in rows to fit tightly in a square.
    """
    centers = []
    radii = []
    
    # We will try to fit circles in a hexagonal pattern.
    # Rows strategy: 6, 5, 6, 5, 4 (Total 26)
    # Or just fill a grid-like pattern.
    # Let's try to fit as many as possible in rows.
    
    # Let's estimate a starting radius. 
    # For 26 circles, average radius is around 0.1.
    # Let's start with a smaller safe radius to ensure no overlaps initially.
    r_start = 0.08 
    
    # Hexagonal packing parameters
    # Horizontal spacing = 2*r, Vertical spacing = sqrt(3)*r
    h_step = 2.0 * r_start
    v_step = np.sqrt(3) * r_start
    
    # We'll place rows. 
    # Row 0: 6 circles
    # Row 1: 5 circles (shifted)
    # Row 2: 6 circles
    # Row 3: 5 circles (shifted)
    # Row 4: 4 circles
    
    row_counts = [6, 5, 6, 5, 4]
    
    y_offset = r_start
    
    for row_idx, count in enumerate(row_counts):
        # Shift for odd rows (hexagonal pattern)
        x_start = r_start
        if row_idx % 2 == 1:
            x_start = r_start + r_start # Shift by r (center to center distance shift)
            # Actually in hex packing, odd rows are shifted by r horizontally relative to even rows
            # Centers at x, x+2r... vs x+r, x+3r...
            # But we must keep them inside [0,1].
            # If we start at x_start, the last center is x_start + (count-1)*2r.
            # We need last_center + r <= 1.
            
        # Let's adjust x_start to center the row if possible, or just pack from left
        # For simplicity and validity, pack from left with margin r_start
        
        # Recalculate x_start for this specific count to ensure it fits
        # Width required = (count - 1) * 2r_start + 2r_start = count * 2r_start
        # Wait, width of 1 circle is 2r. Width of k circles touching is 2r + (k-1)2r = 2kr.
        # But we have gaps or shifts.
        
        # Let's just place them. If they don't fit, the optimizer will shrink/expand.
        # Just need a valid starting point.
        
        # Reset y for the row
        y_curr = y_offset + row_idx * v_step
        
        # Determine x start
        # For even rows (0, 2, 4), align with left margin r_start
        # For odd rows (1, 3), shift by r_start (so center is at 2*r_start)
        
        if row_idx % 2 == 0:
            x_curr = r_start
        else:
            x_curr = r_start + r_start # 2*r_start
            
        for _ in range(count):
            centers.append([x_curr, y_curr])
            radii.append(r_start)
            x_curr += h_step
            
        if len(centers) >= n:
            break
            
    # Trim to n
    centers = centers[:n]
    radii = radii[:n]
    
    return np.array(centers), np.array(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialize
    centers, radii = get_initial_guess(n)
    
    # Flatten to a single vector for optimization: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Define Objective: Maximize sum of radii -> Minimize -sum(radii)
    def objective(vars):
        # vars structure: [x_0, y_0, r_0, x_1, y_1, r_1, ...]
        # r_i are at indices 2, 5, 8, ... (3*i + 2)
        # Actually indices 2, 5, 8 ... is 3*i + 2
        r = vars[2::3]
        return -np.sum(r)

    # 3. Define Constraints
    # Inequality constraints: fun(vars) >= 0
    
    def constraints(vars):
        constraints_list = []
        
        # Reshape vars
        c = vars[:2*n].reshape(n, 2) # Centers (x, y)
        r = vars[2*n:]              # Radii
        
        # Boundary Constraints
        # x - r >= 0  => r - x <= 0 (ineq fun >= 0 means x - r >= 0)
        # Wait, scipy inequality is fun(x) >= 0.
        # So we want x - r >= 0 => fun = x - r
        
        # x_i - r_i >= 0
        constraints_list.append(c[:, 0] - r)
        # (1 - x_i) - r_i >= 0
        constraints_list.append((1.0 - c[:, 0]) - r)
        # y_i - r_i >= 0
        constraints_list.append(c[:, 1] - r)
        # (1 - y_i) - r_i >= 0
        constraints_list.append((1.0 - c[:, 1]) - r)
        
        # Pairwise Non-overlap Constraints
        # dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        
        # Vectorized calculation for all pairs
        # We only need upper triangle i < j to avoid duplicates
        
        # Compute pairwise squared distances
        # diff shape (N, N, 2)
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2) # Shape (N, N)
        
        # Compute sum of radii matrix
        r_mat = r[:, np.newaxis] + r[np.newaxis, :] # Shape (N, N)
        
        # Constraint value
        val = dist_sq - r_mat**2
        
        # Extract upper triangle (excluding diagonal)
        # indices
        rows, cols = np.triu_indices(n, k=1)
        pairwise_constraints = val[rows, cols]
        
        constraints_list.append(pairwise_constraints)
        
        # Flatten all constraints into a single array
        return np.concatenate(constraints_list)

    # 4. Bounds
    # x, y in [0, 1], r in [0, 0.5]
    # vars order: x0, y0, r0, x1, y1, r1 ...
    bounds = []
    for _ in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r
    
    # 5. Run Optimization
    # SLSQP supports bounds and constraints
    try:
        res = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraints},
            options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
        )
        
        if res.success:
            x_opt = res.x
        else:
            # Fallback to initial guess if optimization fails
            x_opt = x0
            
    except Exception as e:
        print(f"Optimization failed: {e}")
        x_opt = x0

    # 6. Extract results
    centers_opt = x_opt[:2*n].reshape(n, 2)
    radii_opt = x_opt[2*n:]
    
    # Ensure radii are non-negative (though bounds should handle it)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
