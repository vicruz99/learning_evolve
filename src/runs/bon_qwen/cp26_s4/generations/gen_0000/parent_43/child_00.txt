import numpy as np
from scipy.optimize import minimize
import math


def create_hexagonal_layout(n, r_init=0.08):
    """Create a hexagonal/lattice arrangement of n circles."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_init
    
    # Try different row distributions that sum to 26
    # Hexagonal packing: staggered rows
    row_configs = [
        [6, 5, 6, 5, 4],
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [4, 5, 6, 5, 6],
        [5, 5, 5, 5, 6],
        [6, 6, 6, 5, 3],
        [7, 6, 6, 7],
        [5, 6, 7, 6, 2],
        [8, 6, 6, 6],
    ]
    
    idx = 0
    for row_counts in row_configs:
        if sum(row_counts) == n:
            y_spacing = 1.0 / (len(row_counts) + 1)
            for row_i, count in enumerate(row_counts):
                y = (row_i + 1) * y_spacing
                if row_i % 2 == 1:
                    x_start = 0.5 * y_spacing * 0.866  # offset by sqrt(3)/2 * spacing
                else:
                    x_start = 0.0
                x_spacing = 1.0 / (count + 1)
                for col in range(count):
                    x = x_start + col * x_spacing
                    centers[idx] = [x, y]
                    idx += 1
            break
    
    return centers, radii


def compute_forces(centers, radii, k_repulse=50.0, k_expand=0.1):
    """Compute repulsive forces between circles and expansion force."""
    n = len(radii)
    forces = np.zeros_like(centers)
    
    for i in range(n):
        # Repulsion from boundaries
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary
        if x - r < 0:
            forces[i, 0] -= k_repulse * (r - x)
        # Right boundary
        if x + r > 1:
            forces[i, 0] += k_repulse * (x + r - 1)
        # Bottom boundary
        if y - r < 0:
            forces[i, 1] -= k_repulse * (r - y)
        # Top boundary
        if y + r > 1:
            forces[i, 1] += k_repulse * (y + r - 1)
        
        # Repulsion from other circles
        for j in range(i + 1, n):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist and dist > 1e-10:
                overlap = min_dist - dist
                # Force proportional to overlap, directed along the line connecting centers
                fx = (dx / dist) * k_repulse * overlap
                fy = (dy / dist) * k_repulse * overlap
                forces[i, 0] -= fx
                forces[i, 1] -= fy
                forces[j, 0] += fx
                forces[j, 1] += fy
    
    return forces


def expand_and_relax(centers, radii, n, max_iter=5000, expansion_rate=0.001):
    """Iteratively expand radii and relax positions to resolve overlaps."""
    centers = centers.copy()
    radii = radii.copy()
    
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    k_repulse = 100.0
    k_expand = 0.0005
    decay = 0.9995
    
    for iteration in range(max_iter):
        # Expand all radii
        radii *= (1 + k_expand)
        
        # Clamp radii to reasonable max
        radii = np.minimum(radii, 0.5)
        
        # Compute forces
        forces = compute_forces(centers, radii, k_repulse=k_repulse)
        
        # Update positions
        learning_rate = 0.01 / (1 + iteration * 0.001)
        centers += learning_rate * forces
        
        # Clamp centers to valid range
        centers = np.clip(centers, radii[:, np.newaxis], 1 - radii[:, np.newaxis])
        
        # Check if valid
        valid = True
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    dist = math.sqrt(sum((centers[i] - centers[j]) ** 2))
                    if dist < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        # Decay expansion rate
        k_expand *= decay
        k_repulse *= 1.001  # Increase repulsion strength over time
    
    return best_centers, best_radii, best_sum


def optimize_with_scipy(centers, radii, n):
    """Use scipy to optimize positions and radii."""
    def objective(x):
        """Negative sum of radii (we minimize negative to maximize sum)."""
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        
        # Penalty for boundary violations
        penalty = 0
        for i in range(n):
            x_c, y_c = c[i]
            r_i = r[i]
            if x_c - r_i < 0:
                penalty += 1000 * (x_c - r_i) ** 2
            if x_c + r_i > 1:
                penalty += 1000 * (x_c + r_i - 1) ** 2
            if y_c - r_i < 0:
                penalty += 1000 * (y_c - r_i) ** 2
            if y_c + r_i > 1:
                penalty += 1000 * (y_c + r_i - 1) ** 2
        
        # Penalty for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = math.sqrt(sum((c[i] - c[j]) ** 2))
                min_dist = r[i] + r[j]
                if dist < min_dist:
                    penalty += 1000 * (min_dist - dist) ** 2
        
        return -np.sum(r) + penalty
    
    def constraint_boundary(i, x):
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        return np.array([
            c[i, 0] - r[i],       # x >= r (left boundary)
            1 - (c[i, 0] + r[i]), # x + r <= 1 (right boundary)
            c[i, 1] - r[i],       # y >= r (bottom boundary)
            1 - (c[i, 1] + r[i]), # y + r <= 1 (top boundary)
        ])
    
    def constraint_overlap(i, j, x):
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        dist = math.sqrt(sum((c[i] - c[j]) ** 2))
        return dist - r[i] - r[j]
    
    # Initial state
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    # Try L-BFGS-B
    try:
        result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 1000, 'ftol': 1e-12})
        if result.success or result.fun < 0:
            c_opt = result.x[:2 * n].reshape(n, 2)
            r_opt = result.x[2 * n:]
            if r_opt.min() > 0:
                return c_opt, r_opt, np.sum(r_opt)
    except:
        pass
    
    return centers, radii, np.sum(radii)


def create_random_layout(n):
    """Create a random initial layout."""
    centers = np.random.uniform(0, 1, (n, 2))
    radii = np.random.uniform(0.03, 0.12, n)
    return centers, radii


def refine_packing(centers, radii, n, iterations=2000):
    """Refine packing using gradient-like updates."""
    centers = centers.copy()
    radii = radii.copy()
    
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    lr_pos = 0.005
    lr_rad = 0.001
    
    for t in range(iterations):
        # Try to increase each radius slightly
        for i in range(n):
            # Compute max possible radius for circle i given current positions
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dist = math.sqrt(sum((centers[i] - centers[j]) ** 2))
                    max_r = min(max_r, dist - radii[j])
            
            # Gradually increase radius
            target_r = min(max_r, radii[i] * 1.001)
            radii[i] = radii[i] * 0.999 + target_r * 0.001
        
        # Compute repulsion forces for positions
        forces = np.zeros_like(centers)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                forces[i, 0] -= 100 * (r - x)
            if x + r > 1:
                forces[i, 0] += 100 * (x + r - 1)
            if y - r < 0:
                forces[i, 1] -= 100 * (r - y)
            if y + r > 1:
                forces[i, 1] += 100 * (y + r - 1)
            
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    fx = (dx / dist) * 100 * overlap
                    fy = (dy / dist) * 100 * overlap
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
        
        centers += lr_pos * forces
        centers = np.clip(centers, 1e-6, 1 - 1e-6)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        lr_pos *= 0.999
        lr_rad *= 0.999
    
    return best_centers, best_radii, best_sum


def create_grid_layout(n, rows=None, cols=None):
    """Create a grid-based layout."""
    if rows is None or cols is None:
        # Find best grid dimensions
        best_rows = None
        best_cols = None
        best_ratio = 0
        for r in range(3, 8):
            for c in range(3, 8):
                if r * c >= n:
                    ratio = min(r / c * 1, c / r)
                    if ratio > best_ratio:
                        best_ratio = ratio
                        best_rows = r
                        best_cols = c
        rows, cols = best_rows, best_cols
    
    centers = np.zeros((n, 2))
    idx = 0
    y_spacing = 1.0 / (rows + 1)
    x_spacing = 1.0 / (cols + 1)
    
    for r in range(rows):
        for c in range(cols):
            if idx < n:
                centers[idx] = [(c + 1) * x_spacing, (r + 1) * y_spacing]
                idx += 1
    
    # Fill remaining
    while idx < n:
        centers[idx] = centers[idx % len(centers)].copy()
        centers[idx, 0] += 0.001
        idx += 1
    
    radii = np.ones(n) * 0.07
    return centers, radii
