import numpy as np
from scipy.optimize import linprog, differential_evolution
import math

def solve_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximize sum(r_i) subject to packing constraints.
    """
    n = centers.shape[0]
    # Objective: Minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Bounds for r_i >= 0
    bounds = [(0, None)] * n
    
    # Build A_ub and b_ub for constraints A*r <= b
    constraints = []
    rhs = []
    
    for i in range(n):
        x, y = centers[i]
        # Boundary constraints: r_i <= x, 1-x, y, 1-y
        for val in [x, 1-x, y, 1-y]:
            row = np.zeros(n)
            row[i] = 1.0
            constraints.append(row)
            rhs.append(val)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints.append(row)
            rhs.append(dist)
            
    A_ub = np.array(constraints)
    b_ub = np.array(rhs)
    
    # Solve LP
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        # Fallback for infeasible configurations (centers outside or extremely close)
        return 0.0, np.zeros(n)

def objective(centers_flat):
    """Objective function for the center optimizer."""
    centers = centers_flat.reshape((26, 2))
    sum_r, _ = solve_radii(centers)
    return -sum_r  # Minimize negative sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Optimization bounds: centers must be in [0, 1]
    bnds = [(0.0, 1.0)] * 52
    
    # Run global optimization
    result = differential_evolution(
        objective, 
        bnds, 
        popsize=30, 
        maxiter=1000, 
        tol=1e-5, 
        mutation=(0.5, 1.5), 
        recombination=0.7, 
        seed=42
    )
    
    best_centers = result.x.reshape((26, 2))
    
    # Final calculation of radii for the best centers found
    _, best_radii = solve_radii(best_centers)
    sum_radii = np.sum(best_radii)
    
    # Ensure centers are valid (clip just in case of numerical drift)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    return best_centers, best_radii, sum_radii
