## Strategy for Packing 26 Circles in a Unit Square

### Problem Analysis
- Need to pack 26 circles in [0,1]×[0,1] to maximize sum of radii
- Target: 2.636 (average radius ≈ 0.1014 per circle)
- For equal circles with hexagonal packing density π/(2√3) ≈ 0.9069: theoretical max sum ≈ 2.74

### Approach
1. **Multiple Initialization Strategies**:
   - Hexagonal grid (most efficient packing geometry)
   - Random initializations with different seeds
   - Each followed by optimization

2. **Two-Phase Optimization**:
   - **Phase 1 - Force Simulation**: Physics-based simulation that gradually expands circles while resolving overlaps through repulsive forces. This handles the non-convex nature of the problem and finds good configurations.
   - **Phase 2 - SLSQP Optimization**: Scipy's SLSQP solver with explicit boundary and non-overlap constraints for fine-tuning.

3. **Key Design Decisions**:
   - Repulsive forces between all close circles (not just overlapping) to create breathing room for expansion
   - Boundary forces to keep circles inside the square
   - Decreasing step size in force simulation for convergence
   - Multiple random restarts to avoid local optima
   - SLSQP with 104 boundary + 325 non-overlap constraints

4. **Constraint Formulation**:
   - Boundary: x-r ≥ 0, x+r ≤ 1, y-r ≥ 0, y+r ≤ 1 (linear)
   - Non-overlap: ||cᵢ-cⱼ||² ≥ (rᵢ+rⱼ)² (non-linear, non-convex)

### Implementation Notes
- All helper functions are top-level (no nesting/closures)
- No lambda functions used
- Robust error handling for optimization failures
- Best solution across all attempts is returned

import numpy as np
from scipy.optimize import minimize

N = 26


def neg_sum_radii(params):
    """Objective: negative sum of radii (minimize negative = maximize sum)"""
    return -np.sum(params[2 * N:])


def boundary_constraints(params):
    """Boundary constraints: each circle must be inside the unit square"""
    centers = params[:2 * N].reshape(N, 2)
    radii = params[2 * N:]
    result = np.zeros(4 * N)
    for i in range(N):
        result[4 * i] = centers[i, 0] - radii[i]
        result[4 * i + 1] = 1.0 - centers[i, 0] - radii[i]
        result[4 * i + 2] = centers[i, 1] - radii[i]
        result[4 * i + 3] = 1.0 - centers[i, 1] - radii[i]
    return result


def overlap_constraints(params):
    """Non-overlap constraints: distance between centers >= sum of radii"""
    centers = params[:2 * N].reshape(N, 2)
    radii = params[2 * N:]
    num_pairs = N * (N - 1) // 2
    result = np.zeros(num_pairs)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]
            result[idx] = dist_sq - r_sum * r_sum
            idx += 1
    return result


def make_hex_init():
    """Create hexagonal grid initialization"""
    centers = np.zeros((N, 2))
    radii = np.ones(N) * 0.04
    idx = 0
    s = 0.17
    for row in range(8):
        for col in range(8):
            if idx >= N:
                break
            x = s / 2 + col * s
            y = s * np.sqrt(3) / 2 * row
            if row % 2 == 1:
                x += s / 2
            x = max(0.06, min(0.94, x))
            y = max(0.06, min(0.94, y))
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def make_grid_init():
    """Create regular grid initialization"""
    centers = np.zeros((N, 2))
    radii = np.ones(N) * 0.04
    idx = 0
    spacing = 0.2
    offset = 0.1
    for row in range(7):
        for col in range(7):
            if idx >= N:
                break
            x = offset + col * spacing
            y = offset + row * spacing
            x = max(0.06, min(0.94, x))
            y = max(0.06, min(0.94, y))
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def make_random_init(seed_val):
    """Create random initialization with given seed"""
    rng = np.random.RandomState(seed_val)
    centers = rng.uniform(0.12, 0.88, (N, 2))
    radii = np.ones(N) * 0.03
    return centers, radii


def force_simulation(centers, radii, steps=3000, expand_per_step=1.00012):
    """Force-based simulation to expand circles while maintaining feasibility"""
    centers = centers.copy().astype(np.float64)
    radii = radii.copy().astype(np.float64)
    
    for step in range(steps):
        # Gradually expand all radii
        radii *= expand_per_step
        
        forces = np.zeros_like(centers)
        
        # Repulsive forces between circle pairs
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-14:
                    dist = 1e-14
                    dx = 1e-14
                    dy = 0.0
                
                min_dist = radii[i] + radii[j]
                
                # Strong repulsion for overlaps
                if dist < min_dist:
                    overlap = min_dist - dist
                    f = overlap / dist * 10.0
                # Mild repulsion for close circles (creates space for growth)
                elif dist < min_dist * 1.4:
                    gap = min_dist * 1.4 - dist
                    f = gap / dist * 0.5
                else:
                    f = 0.0
                
                forces[i, 0] -= f * dx
                forces[i, 1] -= f * dy
                forces[j, 0] += f * dx
                forces[j, 1] += f * dy
        
        # Boundary repulsion forces
        for i in range(N):
            for d in range(2):
                if centers[i, d] < radii[i] + 1e-8:
                    forces[i, d] += 15.0 * (radii[i] + 1e-8 - centers[i, d])
                if centers[i, d] > 1.0 - radii[i] - 1e-8:
                    forces[i, d] -= 15.0 * (centers[i, d] - (1.0 - radii[i] - 1e-8))
        
        # Apply forces with annealing step size
        alpha = 0.04 * max(0.1, 1.0 - step / steps)
        centers = centers + alpha * forces
        
        # Hard clamp to maintain feasibility
        for i in range(N):
            for d in range(2):
                centers[i, d] = max(radii[i] + 1e-10, min(1.0 - radii[i] - 1e-10, centers[i, d]))
    
    return centers, radii


def do_optimize_slsqp(centers, radii):
    """Run SLSQP optimization starting from given configuration"""
    params = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.005, 0.995)] * (2 * N) + [(0.005, 0.49)] * N
    
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]
    
    try:
        result = minimize(
            neg_sum_radii,
            params,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
        )
        new_centers = result.x[:2 * N].reshape(N, 2)
        new_radii = result.x[2 * N:]
        return new_centers, new_radii, result.success
    except Exception:
        return centers, radii, False


def do_force_then_optimize(centers, radii, sim_steps=3000):
    """Run force simulation followed by SLSQP optimization"""
    centers, radii = force_simulation(centers, radii, steps=sim_steps)
    centers, radii, success = do_optimize_slsqp(centers, radii)
    return centers, radii, success


def run_packing():
    """Main function: try multiple strategies and return best packing"""
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Strategy 1: Hexagonal grid initialization + force sim + SLSQP
    centers, radii = make_hex_init()
    centers, radii, _ = do_force_then_optimize(centers, radii, sim_steps=4000)
    s = np.sum(radii)
    if s > best_sum:
        best_sum = s
        best_centers = centers.copy()
        best_radii = radii.copy()
    
    # Strategy 2: Regular grid initialization
    centers, radii = make_grid_init()
    centers, radii, _ = do_force_then_optimize(centers, radii, sim_steps=4000)
    s = np.sum(radii)
    if s > best_sum:
        best_sum = s
        best_centers = centers.copy()
        best_radii = radii.copy()
    
    # Strategy 3-8: Random initializations with different seeds
    for seed in [42, 123, 456, 789, 1024, 2048]:
        centers, radii = make_random_init(seed)
        centers, radii, _ = do_force_then_optimize(centers, radii, sim_steps=3500)
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Final polishing: run another SLSQP on the best found so far
    if best_centers is not None and best_radii is not None:
        centers, radii, _ = do_optimize_slsqp(best_centers, best_radii)
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Verify and fix any numerical issues
    if best_centers is not None:
        for i in range(N):
            for d in range(2):
                best_centers[i, d] = max(best_radii[i] + 1e-12, 
                                          min(1.0 - best_radii[i] - 1e-12, 
                                              best_centers[i, d]))
        for i in range(N):
            best_radii[i] = max(0.001, min(0.49, best_radii[i]))
    
    return best_centers, best_radii, float(np.sum(best_radii))
