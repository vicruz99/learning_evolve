import numpy as np
from scipy.optimize import minimize
import math

def check_constraints(params, n):
    """
    Checks if the current packing satisfies the constraints.
    Returns a penalty value (0 if valid, positive if invalid).
    """
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]
    
    penalty = 0.0
    tol = 1e-8

    # Boundary constraints
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        if x - r < 0: penalty += (x - r) ** 2
        if x + r > 1: penalty += (x + r - 1) ** 2
        if y - r < 0: penalty += (y - r) ** 2
        if y + r > 1: penalty += (y + r - 1) ** 2
        
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist - tol:
                penalty += (min_dist - dist) ** 2
                
    return penalty

def objective_function(params, n):
    """
    Objective function to minimize.
    We want to maximize sum(radii), so we minimize negative sum.
    We add a penalty for constraint violations.
    """
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]
    
    # Negative sum of radii
    obj = -np.sum(radii)
    
    # Add penalty for violations
    # A high weight ensures constraints are prioritized
    penalty = check_constraints(params, n)
    obj += 1000.0 * penalty
    
    return obj

def generate_initial_packing(n, angle=0.0):
    """
    Generates an initial packing using a hexagonal grid rotated by 'angle'.
    """
    centers = []
    radii = []
    
    # Estimate radius for initial placement to ensure it fits
    # Rough estimate based on area density
    r_est = 0.1 
    
    # Grid dimensions
    rows = int(math.ceil(math.sqrt(n)))
    cols = int(math.ceil(n / rows))
    
    # Adjust to ensure we generate enough points
    if rows * cols < n:
        cols += 1
        
    count = 0
    y = 0
    while count < n:
        x = 0
        while x < cols and count < n:
            # Hexagonal offset
            x_coord = x * 2 * r_est + (y % 2) * r_est
            y_coord = y * math.sqrt(3) * r_est
            
            # Rotate
            cos_a = math.cos(angle)
            sin_a = math.sin(angle)
            rot_x = x_coord * cos_a - y_coord * sin_a
            rot_y = x_coord * sin_a + y_coord * cos_a
            
            # Center in unit square [0, 1]
            # We can just place them and let optimizer move them
            # Or scale to fit. For now, place in [0, 1] with some scaling
            scale_x = 0.8 # Initial scale to leave room
            scale_y = 0.8
            
            cx = rot_x * scale_x + (1 - scale_x) / 2
            cy = rot_y * scale_y + (1 - scale_y) / 2
            
            centers.append([cx, cy])
            radii.append(r_est)
            count += 1
            x += 1
        y += 1
        
    return np.array(centers), np.array(radii)

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Define bounds for optimization
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)]) # x, y
        bounds.append((0.0, 0.5)) # r
        
    # Try multiple initial configurations with different angles
    angles = [0.0, math.pi/8, math.pi/6, math.pi/4, 0.1, 0.5, 1.0]
    
    for angle in angles:
        # Generate initial guess
        centers, radii = generate_initial_packing(n, angle)
        
        # Flatten parameters
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Optimize
        try:
            res = minimize(
                objective_function, 
                x0, 
                args=(n,), 
                method='SLSQP', 
                bounds=bounds, 
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if res.success or res.fun < -2.0: # Check if we got a valid solution
                params = res.x

                r = params[2*n:]
                
                # Validate constraints explicitly
                penalty = check_constraints(params, n)
                if penalty < 1e-6: # Acceptable tolerance
                    current_sum = np.sum(r)
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = c
                        best_radii = r
        except Exception as e:
            print(f"Optimization failed for angle {angle}: {e}")
            continue

    # If no good result found, fallback to a simple grid
    if best_centers is None:
        print("Warning: Optimization failed to find valid packing, using grid fallback.")
        # 5x5 grid with one extra

        radii = []
        r = 0.095
        count = 0

                if count >= n:
                    break

                radii.append(r)
                count += 1
        best_centers = np.array(centers)
        best_radii = np.array(radii)
        best_sum = np.sum(best_radii)
