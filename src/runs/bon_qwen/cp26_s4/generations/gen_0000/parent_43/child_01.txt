def calculate_margin(centers, r):
    """
    Calculates the minimum margin for a given set of centers and a scalar radius r.
    Margins are: distance to walls - r, and distance between circles - 2r.
    """
    n = centers.shape[0]
    # Boundary margins
    left = centers[:, 0] - r
    right = 1 - centers[:, 0] - r
    bottom = centers[:, 1] - r
    top = 1 - centers[:, 1] - r
    
    boundaries = np.array([left, right, bottom, top])
    min_boundary = np.min(boundaries, axis=0) # (n,)
    
    # Inter-circle margins
    # Compute pairwise distances
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2)) # (n, n)
    
    # We only care about i < j, but we can just mask the diagonal
    np.fill_diagonal(dists, np.inf)
    min_inter = np.min(dists) - 2 * r
    
    return min(min(min_boundary), min_inter)
