import numpy as np
import scipy.optimize as opt
import math

def generate_hexagonal_initialization(n_circles, seed=None):
    """
    Generates an initial configuration of circles using a hexagonal packing pattern.
    Specifically tailored for n_circles=26 in a unit square.
    """
    if seed is not None:
        np.random.seed(seed)
    
    # Parameters for hexagonal packing
    # We aim for a pattern like 5-4-5-4-5-4 which sums to 27. 
    # We will generate 27 and drop one, or just generate 26 directly.
    # 6 rows.
    
    # Initial radius guess. 
    # For 6 rows, height ~ r(2 + 5*sqrt(3)) <= 1 => r <= 0.0938
    # For 5 circles width ~ 10*r <= 1 => r <= 0.1
    # Safe r = 0.08
    
    r_init = 0.085
    
    centers = []
    
    # Row configuration: number of circles in each row
    # 5, 4, 5, 4, 5, 4 -> 27 circles. We need 26.
    # Let's do 5, 4, 5, 4, 5, 3? Or just drop last.
    # Let's stick to 5, 4, 5, 4, 5, 4 and take first 26.
    row_counts = [5, 4, 5, 4, 5, 4]
    
    y_pos = r_init
    row_idx = 0
    
    for count in row_counts:
        if len(centers) >= n_circles:
            break
            
        # Horizontal shift for alternating rows
        shift = 0
        if row_idx % 2 == 1:
            shift = r_init # Shift by radius (half diameter)
        
        # Calculate x positions for this row
        # We want to center the row in [0, 1]
        # Width occupied by 'count' circles is count * 2 * r_init
        # But with shift, effective bounds might differ.
        # Let's just place them centered.

        # (count - 1) * 2*r + 2*r = 2*r*count

        # But for initialization, exact fit isn't critical, just valid.

                break
            x = start_x + i * (2 * r_init)
            centers.append([x, y_pos])
            
        # Next row y position
        y_pos += r_init * math.sqrt(3)
        row_idx += 1
        
    # If we didn't get enough (unlikely with this pattern), fill with random
    while len(centers) < n_circles:
        centers.append([0.5, 0.5]) # Fallback
        
    centers = np.array(centers[:n_circles])
    radii = np.full(n_circles, r_init)

    centers += perturbation

    # Since r is small, simple clipping might be okay, but better to just ensure centers in [r, 1-r]
    # Actually, if centers go out of bounds, the optimizer will fix it if valid.
    # But let's ensure they are inside [0,1] at least.
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers, radii

def validate_and_fix(centers, radii, n):
    """
    Quick check to ensure basic validity.
    """
    # Ensure non-negative radii
    radii = np.maximum(radii, 1e-9)
    
    # Ensure centers are within [0,1]
    centers = np.clip(centers, 0.0, 1.0)
    
    # Ensure circles are inside square
    # r <= x <= 1-r => x >= r, x <= 1-r
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        if x < r: centers[i, 0] = r
        if x > 1 - r: centers[i, 0] = 1 - r
        if y < r: centers[i, 1] = r
        if y > 1 - r: centers[i, 1] = 1 - r
        
    return centers, radii

def objective_function(vars, n):
    """
    Objective function: minimize negative sum of radii.
    vars = [x0, y0, r0, x1, y1, r1, ...]
    """
    radii = vars[2::3]
    return -np.sum(radii)

def constraints_list(n):
    """
    Generates constraints for scipy optimizer.
    """
    cons = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0

        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2

            def constr(vars):
                return vars[3*i] - vars[3*i + 2]
            return constr
        cons.append({'type': 'ineq', 'fun': make_bound_x(i)})
        
        # 1 - x - r >= 0
        def make_bound_x_upper(i):
            def constr(vars):
                return 1.0 - vars[3*i] - vars[3*i + 2]
            return constr
        cons.append({'type': 'ineq', 'fun': make_bound_x_upper(i)})
        
        # y - r >= 0
        def make_bound_y(i):
            def constr(vars):
                return vars[3*i + 1] - vars[3*i + 2]
            return constr
        cons.append({'type': 'ineq', 'fun': make_bound_y(i)})
        
        # 1 - y - r >= 0
        def make_bound_y_upper(i):
            def constr(vars):
                return 1.0 - vars[3*i + 1] - vars[3*i + 2]
            return constr
        cons.append({'type': 'ineq', 'fun': make_bound_y_upper(i)})
            
    # Non-overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0

                    return dist_sq - rad_sum_sq
                return constr
            
            cons.append({'type': 'ineq', 'fun': make_overlap(i, j)})
            
    return cons
