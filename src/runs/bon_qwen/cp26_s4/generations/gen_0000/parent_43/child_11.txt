from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n_circles = 26

    # Rows pattern: [5, 6, 5, 6, 4] = 26 circles
    rows_counts = [5, 6, 5, 6, 4]
    centers = []
    y_current = 0.1
    
    for i, count in enumerate(rows_counts):
        x_start = 0.1 + (i % 2) * 0.1
        spacing = 0.2
        for j in range(count):
            x = x_start + j * spacing
            # Ensure we stay within [0, 1] during initialization
            x = max(0.1, min(0.9, x))
            centers.append([x, y_current])
        
        # Move y up for the next row (vertical distance in hex packing is sqrt(3)/2 * 2r)
        y_current += 0.1732
        
    centers = np.array(centers)
    radii = np.ones(n_circles) * 0.08 # Start with a feasible radius

    # 3. Define Objective and Constraints for Optimization
    def objective(params):
        radii = params[n_circles:]
        return -np.sum(radii) # Minimize negative sum (Maximize sum)

    def inequality_constraints(params):
        centers = params[:n_circles * 2].reshape(-1, 2)
        radii = params[n_circles:]
        cons = []
        
        # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        # Equivalent to: x-r >= 0, 1-(x+r) >= 0, etc.
        cons.extend(centers[:, 0] - radii)          # x >= r
        cons.extend(1 - (centers[:, 0] + radii))    # x <= 1-r
        cons.extend(centers[:, 1] - radii)          # y >= r
        cons.extend(1 - (centers[:, 1] + radii))    # y <= 1-r

        for i in range(n_circles):

                dx = centers[i, 0] - centers[j, 0]

                dist_sq = dx**2 + dy**2
                radii_sum = radii[i] + radii[j]
                cons.append(dist_sq - radii_sum**2)
                
        return np.array(cons)

    constraint_def = {'type': 'ineq', 'fun': inequality_constraints}

    opt_radii = np.maximum(0, opt_radii)
