To maximize the sum of radii for 26 circles in a unit square, we employ a "Grow-and-Push" optimization algorithm. This approach iteratively expands circles and resolves collisions by pushing them apart, gradually increasing the total radius sum.

1.  **Initialization:** We initialize 26 circles with small radii (0.01) at random positions within the unit square.
2.  **Radial Growth:** In each iteration, every circle attempts to grow its radius. The maximum possible radius for each circle is determined by the nearest boundary and the nearest neighbor (distance between centers minus the neighbor's radius).
3.  **Collision Resolution (Push):** If growth is restricted by an overlap, the circle is pushed away from the nearest obstructing circle or boundary. This movement frees up space, allowing for further growth in subsequent iterations.
4.  **Iterative Refinement:** The algorithm runs multiple times from different random seeds to avoid local minima. The configuration yielding the highest sum of radii is returned.
5.  **Validation:** The final output is validated using the provided `validate_packing` function to ensure no overlaps or boundary violations exist.

import numpy as np
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def calculate_max_radii(centers, radii, n):
    """Calculates the maximum allowable radius for each circle based on current positions."""
    max_r = np.full(n, np.inf)
    
    # Boundary constraints
    max_r = np.minimum(max_r, centers[:, 0])
    max_r = np.minimum(max_r, 1 - centers[:, 0])
    max_r = np.minimum(max_r, centers[:, 1])
    max_r = np.minimum(max_r, 1 - centers[:, 1])
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            # Constraint: r_i + r_j <= dist  =>  r_i <= dist - r_j
            possible_r_i = dist - radii[j]
            if possible_r_i < max_r[i]:
                max_r[i] = possible_r_i
            possible_r_j = dist - radii[i]
            if possible_r_j < max_r[j]:
                max_r[j] = possible_r_j
                
    return max_r

def resolve_collisions(centers, radii, n):
    """Attempts to resolve collisions by pushing circles away from each other."""
    delta = 0.005
    pushed = False
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            required_dist = radii[i] + radii[j]
            
            if dist < required_dist - 1e-7:
                # Collision detected
                if dist < 1e-12:
                    # Circles are at same spot, push randomly
                    angle = np.random.uniform(0, 2 * np.pi)
                    centers[i] += [delta * math.cos(angle), delta * math.sin(angle)]
                    centers[j] -= [delta * math.cos(angle), delta * math.sin(angle)]
                else:
                    overlap = required_dist - dist
                    push_vec = (diff / dist) * (overlap / 2)
                    centers[i] += push_vec
                    centers[j] -= push_vec
                    pushed = True

    # Boundary resolution
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-7: centers[i, 0] = r + 1e-7
        if x + r > 1 + 1e-7: centers[i, 0] = 1 - r - 1e-7
        if y - r < -1e-7: centers[i, 1] = r + 1e-7
        if y + r > 1 + 1e-7: centers[i, 1] = 1 - r - 1e-7

    return centers, pushed

def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Run multiple seeds to find a robust packing
    for seed in range(20):
        np.random.seed(seed)
        
        # 1. Initialize with small radii at random positions
        centers = np.random.uniform(0.05, 0.95, (n, 2))
        radii = np.full(n, 0.02)
        
        # 2. Iterative expansion loop
        max_iters = 1000
        for _ in range(max_iters):
            # Step A: Try to grow all circles
            max_allowed = calculate_max_radii(centers, radii, n)
            growth = np.zeros(n)
            for i in range(n):
                if max_allowed[i] > radii[i]:
                    # Grow slightly towards limit
                    growth[i] = min(0.002, max_allowed[i] - radii[i])
            
            radii += growth
            
            # Step B: Resolve any collisions created by growth
            centers, pushed = resolve_collisions(centers, radii, n)
            
            # Step C: If no growth and no pushes, break early
            if np.max(growth) < 1e-7 and not pushed:
                break
                
        # Final cleanup pass to ensure strict validity
        for _ in range(100):
            centers, _ = resolve_collisions(centers, radii, n)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            if validate_packing(centers, radii):
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
            else:
                # Fallback if validation fails due to numerical issues, try to fix
                centers, _ = resolve_collisions(centers, radii, n)
                if validate_packing(centers, radii):
                    best_sum = np.sum(radii)
                    best_centers = centers.copy()
                    best_radii = radii.copy()

    # Final robust validation and return
    if validate_packing(best_centers, best_radii):
        return best_centers, best_radii, np.sum(best_radii)
    else:
        # Should not happen if logic is sound, but return fallback if it does
        centers = np.random.rand(n, 2) * 0.9 + 0.05
        radii = np.full(n, 0.01)
        return centers, radii, 0.0
