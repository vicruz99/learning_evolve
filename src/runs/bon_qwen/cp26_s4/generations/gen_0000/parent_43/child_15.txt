import numpy as np
import scipy.optimize as opt
import itertools

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n_circles = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    # Helper to solve LP for radii given centers
    def solve_radii(centers):
        n = centers.shape[0]
        # Variables: r_0, ..., r_{n-1}
        # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
        c_obj = -np.ones(n)

        # Constraints:
        # 1. Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
        #    => r_i + 0 <= bound_i
        # 2. Overlap constraints: r_i + r_j <= dist_ij
        # 3. Non-negativity: r_i >= 0

        A_ub = []
        b_ub = []

        # Boundary constraints
        # r_i <= x_i  => 1*r_i <= x_i
        # r_i <= 1-x_i => 1*r_i <= 1-x_i
        # r_i <= y_i
        # r_i <= 1-y_i
        for i in range(n):
            x, y = centers[i]
            bounds = [x, 1-x, y, 1-y]
            max_r = max(0, min(bounds)) # Radius cannot be negative
            
            # We add constraints r_i <= bound.
            # If bound is very small (circle outside), LP might be infeasible if we force r_i >= 0?
            # Actually, if center is outside [0,1], max_r is negative.
            # We should ensure centers are valid first, or allow r_i=0.
            # Here we assume centers are roughly valid.
            
            row = np.zeros(n)
            row[i] = 1.0
            # We only add constraint if bound is reasonable? 
            # Actually, linear constraints are fine.
            # But if bound < 0, it implies r_i must be negative, which conflicts with r_i >= 0.
            # So we clip bound to at least 0 to avoid infeasibility, effectively forcing r_i=0 if center is bad.
            # But better to ensure centers are valid.
            
            val = max(0.0, max_r)
            A_ub.append(row)
            b_ub.append(val)

            # No, LP handles multiple constraints. But for efficiency, 
            # r_i <= min(x, 1-x, y, 1-y) is equivalent to 4 inequalities.
            # However, adding 4 rows per circle is 104 rows.
            # We can just add the single tightest constraint: r_i <= min(...)
            # But to be safe and correct, let's just add the minimum bound.
            # r_i <= min(x, 1-x, y, 1-y) is sufficient because if r_i satisfies this, 
            # it satisfies all 4? Yes, because min is the smallest.
            # Wait, if r_i <= 0.1, and bounds are 0.2, 0.2, 0.2, 0.2. 
            # Then r_i <= 0.1 is tighter.
            # If bounds are 0.05, 0.9, ... min is 0.05. r_i <= 0.05 implies r_i <= 0.9.
            # So yes, single constraint r_i <= min_dist_to_wall is sufficient.
            
        # Re-evaluating boundary constraints logic for efficiency
        # Reset lists
        A_ub = []
        b_ub = []
        
        for i in range(n):
            x, y = centers[i]
            # Distance to walls
            d_walls = min(x, 1-x, y, 1-y)
            if d_walls < 0:
                d_walls = 0 # Center outside, radius must be 0
            
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(d_walls)
            
        # Overlap constraints
        # r_i + r_j <= dist_ij
        # This involves 2 variables.
        # Number of pairs is n*(n-1)/2 = 26*25/2 = 325.
        # 325 constraints is fine.

            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)
                
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Bounds for variables r_i >= 0
        bounds = [(0, None) for _ in range(n)]
        
        try:
            res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                return -res.fun, res.x # fun is minimized, so -fun is max sum
            else:
                # Infeasible (shouldn't happen if centers valid)
                return 0, np.zeros(n)
        except Exception:
            return 0, np.zeros(n)

    # Helper to compute forces to move centers
    def compute_forces(centers, radii):
        forces = np.zeros_like(centers)
        n = len(centers)
        
        # Pairwise repulsion
        # If r_i + r_j > dist, push apart.
        # Also apply some repulsion if close to touching to prevent sticking.
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff**2))

                    dist = 1e-9
                    diff = np.random.rand(2) * 0.01 # Prevent division by zero

                    force_mag = overlap * 10.0

                    # No, centers position determines capacity.

                    force_mag = 0.0

                # Force on i is away from j
                if dist > 0:
                    dir_vec = diff / dist
                else:
                    dir_vec = np.zeros(2)
                    
                forces[i] += dir_vec * force_mag
                forces[j] -= dir_vec * force_mag

        # Wall repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x < r:
                forces[i, 0] += (r - x) * 10.0
            # Right wall
            if x > 1 - r:
                forces[i, 0] -= (x - (1 - r)) * 10.0
            # Bottom wall
            if y < r:
                forces[i, 1] += (r - y) * 10.0
            # Top wall
            if y > 1 - r:
                forces[i, 1] -= (y - (1 - r)) * 10.0
                
        return forces

        sum_r, radii = solve_radii(centers)

            forces = compute_forces(centers, radii)

            centers += lr * forces

            # Re-solve radii every few steps? Or every step?

            if step % 10 == 0:
                sum_r, radii = solve_radii(centers)
                
        sum_r, radii = solve_radii(centers)
        return centers, radii, sum_r

    # Run multiple trials with different initializations
    np.random.seed(42)
    n_trials = 20
    
    for trial in range(n_trials):
        # Initialization: Hexagonal grid perturbed
        # We want to pack 26 circles. 
        # Approximate hex grid.
        
        # Strategy: Fill rows.
        # Row 0: 6 circles? Width ~ 6*2r. If r~0.1, width 1.2. Too wide.
        # Maybe 5 rows of 5? 25 circles. + 1.
        # Let's try a generic random perturbation of a grid.
        
        # Create a grid of points
        # 6x5 grid = 30 points. Pick 26 best? Or just random subset?
        # Better: Start with a grid that fits 25, and add 1 in a gap?
        # Or just random positions.
        
        # Let's try a dense random initialization within a slightly shrunken square
        init_centers = np.random.rand(n_circles, 2) * 0.9 + 0.05
        
        # Add some noise to break symmetry
        init_centers += np.random.randn(n_circles, 2) * 0.05
        init_centers = np.clip(init_centers, 0.01, 0.99)
        
        # Run optimizer
        # Use multiple learning rates or just one good one
        c, r, s = optimize_iteration(init_centers, steps=1000, lr=0.02)
        
        if s > best_sum_radii:
            best_sum_radii = s
            best_centers = c.copy()
            best_radii = r.copy()
            
        # Print progress
        if (trial + 1) % 5 == 0:
            print(f"Trial {trial+1}: Sum Radii = {s:.5f}")

    # Final validation and cleanup
    # Ensure strict validity
    # Sometimes numerical errors leave tiny overlaps.
    # We can slightly shrink radii to be safe.
    
    # Check validity
    valid = True
    n = best_centers.shape[0]
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if r < 0: valid = False
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            valid = False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
            if dist < best_radii[i] + best_radii[j] - 1e-9:
                valid = False

    # If not valid due to numerical issues, shrink slightly
    if not valid:
        # Find max overlap
        max_violation = 0
        for i in range(n):
             for j in range(i + 1, n):
                dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
                viol = (best_radii[i] + best_radii[j]) - dist
                if viol > max_violation:
                    max_violation = viol
        
        for i in range(n):
            x, y = best_centers[i]
            r = best_radii[i]
            # Wall violations
            wall_viol = 0
            if x - r < 0: wall_viol = max(wall_viol, -x + r)

        # Shrink all radii by a bit more than max violation to be safe

        best_radii -= shrink_factor
        best_radii = np.maximum(best_radii, 0)
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
