import numpy as np
from scipy.optimize import minimize

def compute_energy(xs, ys, rs, lam):
    n = len(xs)
    pen = 0.0
    # Penalty for circle overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            d = np.sqrt(max(0.0, dx * dx + dy * dy))
            r_sum = rs[i] + rs[j]
            if d < r_sum:
                pen += (r_sum - d) ** 2
                
    # Penalty for boundary violations
    for i in range(n):
        if xs[i] - rs[i] < 0.0:
            pen += (xs[i] - rs[i]) ** 2
        if xs[i] + rs[i] > 1.0:
            pen += (xs[i] + rs[i] - 1.0) ** 2
        if ys[i] - rs[i] < 0.0:
            pen += (ys[i] - rs[i]) ** 2
        if ys[i] + rs[i] > 1.0:
            pen += (ys[i] + rs[i] - 1.0) ** 2
            
    return -np.sum(rs) + lam * pen

def objective_func(v, n, lam):
    xs = v[:n]
    ys = v[n:2 * n]
    rs = v[2 * n:]
    return compute_energy(xs, ys, rs, lam)

def run_packing():
    n = 26
    
    # Initial configuration: 5x5 grid + 1 center circle
    pts = []
    x_grid = np.linspace(0.15, 0.85, 5)
    y_grid = np.linspace(0.15, 0.85, 5)
    for x in x_grid:
        for y in y_grid:
            pts.append([x, y])
    pts.append([0.5, 0.5])
    pts = np.array(pts)
    
    # Add small random noise to break symmetry and help escape local minima
    np.random.seed(42)
    pts += np.random.normal(0, 0.02, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    xs = pts[:, 0].copy()
    ys = pts[:, 1].copy()
    rs = np.full(n, 0.02)
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    v = np.concatenate([xs, ys, rs])
    
    # Staged optimization with increasing penalty weight
    # This helps the optimizer first arrange circles and grow radii, 
    # then strictly enforce non-overlap and boundary constraints.
    penalty_schedule = [100, 500, 2000, 5000]
    
    for lam in penalty_schedule:
        res = minimize(objective_func, v, args=(n, lam), bounds=bounds, 
                       method='L-BFGS-B', options={'maxiter': 2000, 'ftol': 1e-12})
        v = res.x
        xs = v[:n]
        ys = v[n:2 * n]
        rs = v[2 * n:]
        
    # Ensure non-negative radii (should already be satisfied by bounds)
    rs = np.maximum(rs, 0.0)
    
    centers = np.column_stack([xs, ys])
    return centers, rs, np.sum(rs)
