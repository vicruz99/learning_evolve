def objective_function(params, n):
    """
    Objective function: maximize sum of radii.
    We minimize the negative sum.
    params: array of size 3*n. First 2*n are centers (x,y), next n are radii.
    """
    radii = params[2*n:]
    return -np.sum(radii)

def constraint_function(params, n):
    """
    Constraints:
    1. Boundary: r <= x <= 1-r, r <= y <= 1-r
    2. Non-overlap: dist(i,j) >= r_i + r_j
    Returns an array of constraint values. All must be >= 0.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    constraints = []
    
    # Boundary constraints
    # x - r >= 0  =>  x - r
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r)
        constraints.append(1.0 - x - r)
        constraints.append(y - r)
        constraints.append(1.0 - y - r)
        
    # Overlap constraints
    # dist(i, j) - r_i - r_j >= 0
    # We can compute this efficiently
    # But for N=26, a loop is acceptable
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                             (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist - radii[i] - radii[j])
            
    return np.array(constraints)
