import numpy as np
from scipy.optimize import minimize


def constraint_boundary_x_min(x, i, n):
    return x[2 * i] - x[2 * n + i]


def constraint_boundary_x_max(x, i, n):
    return 1.0 - x[2 * i] - x[2 * n + i]


def constraint_boundary_y_min(x, i, n):
    return x[2 * i + 1] - x[2 * n + i]


def constraint_boundary_y_max(x, i, n):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def constraint_radius_nonneg(x, i, n):
    return x[2 * n + i]


def constraint_no_overlap(x, i, j, n):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy + 1e-14)
    return dist - x[2 * n + i] - x[2 * n + j]


def objective_func(x, n):
    return -np.sum(x[2 * n:])


def build_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_x_min, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_x_max, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_y_min, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_y_max, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': constraint_radius_nonneg, 'args': (i, n)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_no_overlap, 'args': (i, j, n)})
    return constraints


def make_hex_init(n):
    centers = np.zeros((n, 2))
    idx = 0
    spacing = 0.135
    for row in range(8):
        for col in range(7):
            if idx >= n:
                break
            x = 0.07 + col * spacing + (row % 2) * spacing * 0.5
            y = 0.07 + row * spacing * np.sqrt(3) * 0.5
            if x <= 0.93 and y <= 0.93:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    return centers


def make_grid_init(n):
    centers = np.zeros((n, 2))
    idx = 0
    rows = 5
    cols = 6
    spacing_x = 1.0 / (cols + 1)
    spacing_y = 1.0 / (rows + 1)
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = (col + 1) * spacing_x
            y = (row + 1) * spacing_y
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    return centers


def make_staggered_init(n):
    centers = np.zeros((n, 2))
    idx = 0
    rows = 6
    for row in range(rows):
        n_in_row = 5 if row % 2 == 0 else 4
        if row == 5:
            n_in_row = 2
        spacing = 1.0 / (n_in_row + 1)
        for col in range(n_in_row):
            if idx >= n:
                break
            x = (col + 1) * spacing
            y = (row + 1) * (1.0 / (rows + 1))
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    return centers


def run_single_opt(centers_init, radii_init, n, constraints):
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))
    result = minimize(
        objective_func, x0, method='SLSQP', bounds=bounds,
        constraints=constraints, args=(n,),
        options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
    )
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(result.x[2 * n:], 0.0)
    return centers_opt, radii_opt


def run_packing():
    n = 26
    constraints = build_constraints(n)
    best_centers = None
    best_radii = None
    best_sum = -1.0

    init_configs = []

    c1 = make_hex_init(n)
    init_configs.append((c1, np.ones(n) * 0.045))

    c2 = make_grid_init(n)
    init_configs.append((c2, np.ones(n) * 0.04))

    c3 = make_staggered_init(n)
    init_configs.append((c3, np.ones(n) * 0.042))

    np.random.seed(123)
    for seed in [0, 1, 2]:
        np.random.seed(seed + 100)
        rc = np.random.uniform(0.1, 0.9, size=(n, 2))
        rr = np.random.uniform(0.03, 0.06, size=n)
        init_configs.append((rc, rr))

    for centers_init, radii_init in init_configs:
        try:
            centers_opt, radii_opt = run_single_opt(centers_init, radii_init, n, constraints)
            current_sum = np.sum(radii_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
        except Exception:
            pass

    if best_centers is None:
        best_centers = make_hex_init(n)
        best_radii = np.ones(n) * 0.04
        best_sum = np.sum(best_radii)

    for _ in range(8):
        centers_opt, radii_opt = run_single_opt(best_centers, best_radii, n, constraints)
        current_sum = np.sum(radii_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)

    return best_centers, best_radii, float(np.sum(best_radii))
