import numpy as np
from scipy.optimize import minimize

def compute_energy_and_grad(centers_flat, r):
    centers = centers_flat.reshape(-1, 2)
    n = centers.shape[0]
    energy = 0.0
    grad = np.zeros_like(centers)
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # left: x - r >= 0
        val = x - r
        if val < 0:
            energy += val**2
            grad[i, 0] += 2.0 * val
        # right: 1 - (x + r) >= 0
        val = 1.0 - x - r
        if val < 0:
            energy += val**2
            grad[i, 0] += -2.0 * val
        # bottom: y - r >= 0
        val = y - r
        if val < 0:
            energy += val**2
            grad[i, 1] += 2.0 * val
        # top: 1 - (y + r) >= 0
        val = 1.0 - y - r
        if val < 0:
            energy += val**2
            grad[i, 1] += -2.0 * val

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            min_dist = 2.0 * r
            if dist < min_dist:
                val = min_dist - dist
                energy += val**2
                if dist > 1e-8:
                    fx = val * dx / dist
                    fy = val * dy / dist
                    grad[i, 0] += -2.0 * fx
                    grad[i, 1] += -2.0 * fy
                    grad[j, 0] += 2.0 * fx
                    grad[j, 1] += 2.0 * fy
                else:
                    # Deterministic push along x-axis if exactly coincident
                    grad[i, 0] += -2.0 * min_dist
                    grad[j, 0] += 2.0 * min_dist
    return energy, grad.flatten()

def objective_func(centers_flat, r):
    return compute_energy_and_grad(centers_flat, r)[0]

def jacobian_func(centers_flat, r):
    return compute_energy_and_grad(centers_flat, r)[1]

def get_initial_guess(n, r):
    centers = []
    row = 0
    col = 0
    while len(centers) < n:
        dx = 0.22
        dy = 0.19
        x = col * dx + max(r, 0.1)
        y = row * dy + max(r, 0.1)
        if row % 2 == 1:
            x += dx / 2.0
        if x < 1.0 - r and y < 1.0 - r:
            centers.append([x, y])
        col += 1
        if col > 6:
            col = 0
            row += 1
    return np.array(centers[:n]).flatten()

def run_packing():
    n = 26
    r_low, r_high = 0.05, 0.15
    best_centers = None
    best_r = r_low
    
    np.random.seed(42)
    
    # Binary search for maximum feasible radius
    for _ in range(25):
        r_mid = (r_low + r_high) / 2.0
        x0 = get_initial_guess(n, r_mid)
        x0 = np.clip(x0, r_mid, 1.0 - r_mid)
        
        bounds = [(r_mid, 1.0 - r_mid)] * (2 * n)
        
        res = minimize(
            objective_func, x0, args=(r_mid,),
            jac=jacobian_func,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        
        # Retry with perturbation if stuck
        if res.fun > 1e-3:
            x0_noisy = x0 + np.random.normal(0, 0.01, size=x0.shape)
            x0_noisy = np.clip(x0_noisy, r_mid, 1.0 - r_mid)
            res2 = minimize(
                objective_func, x0_noisy, args=(r_mid,),
                jac=jacobian_func,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            if res2.fun < res.fun:
                res = res2
                
        if res.fun < 1e-4:
            best_r = r_mid
            best_centers = res.x.reshape(n, 2)
            r_low = r_mid
        else:
            r_high = r_mid
            
    if best_centers is None:
        best_r = 0.10
        best_centers = get_initial_guess(n, best_r).reshape(n, 2)
        
    # Final high-precision polish
    bounds_final = [(best_r, 1.0 - best_r)] * (2 * n)
    res_final = minimize(
        objective_func, best_centers.flatten(), args=(best_r,),
        jac=jacobian_func,
        method='L-BFGS-B',
        bounds=bounds_final,
        options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12}
    )
    final_centers = res_final.x.reshape(n, 2)
    radii = np.full(n, best_r)
    
    return final_centers, radii, best_r * n
