import numpy as np
from scipy.optimize import minimize


def obj_func(x, n):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2 * n :])


def constr_xl(x, n, i):
    """Left boundary: x_i - r_i >= 0."""
    return x[2 * i] - x[2 * n + i]


def constr_xr(x, n, i):
    """Right boundary: 1 - x_i - r_i >= 0."""
    return 1.0 - x[2 * i] - x[2 * n + i]


def constr_yb(x, n, i):
    """Bottom boundary: y_i - r_i >= 0."""
    return x[2 * i + 1] - x[2 * n + i]


def constr_yt(x, n, i):
    """Top boundary: 1 - y_i - r_i >= 0."""
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def constr_r(x, n, i):
    """Non-negative radius: r_i >= 0."""
    return x[2 * n + i]


def constr_sep(x, n, i, j):
    """Separation: dist(i,j) >= r_i + r_j."""
    xi, yi = 2 * i, 2 * i + 1
    xj, yj = 2 * j, 2 * j + 1
    ri, rj = 2 * n + i, 2 * n + j
    dx = x[xi] - x[xj]
    dy = x[yi] - x[yj]
    dist_sq = dx * dx + dy * dy
    if dist_sq < 1e-20:
        return -x[ri] - x[rj]
    dist = np.sqrt(dist_sq)
    return dist - x[ri] - x[rj]


def build_constraints(n):
    """Build all constraint dictionaries for the optimization."""
    cons = []
    for i in range(n):
        cons.append({"type": "ineq", "fun": constr_xl, "args": (n, i)})
        cons.append({"type": "ineq", "fun": constr_xr, "args": (n, i)})
        cons.append({"type": "ineq", "fun": constr_yb, "args": (n, i)})
        cons.append({"type": "ineq", "fun": constr_yt, "args": (n, i)})
        cons.append({"type": "ineq", "fun": constr_r, "args": (n, i)})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({"type": "ineq", "fun": constr_sep, "args": (n, i, j)})
    return cons


def init_hex_centers(n, base_spacing, row_factor=1.0):
    """Initialize centers on a hexagonal lattice."""
    centers = np.zeros((n, 2))
    idx = 0
    row_spacing = base_spacing * np.sqrt(3) / 2 * row_factor

    for row in range(12):
        for col in range(12):
            if idx >= n:
                break
            x = base_spacing / 2 + col * base_spacing
            y = base_spacing / 2 + row * row_spacing
            if row % 2 == 1:
                x += base_spacing / 2
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break

    # Normalize to fit in unit square with margin
    centers -= centers.min(axis=0)
    if centers.max(axis=0)[0] > 0:
        centers[:, 0] *= 0.92 / centers.max(axis=0)[0]
    if centers.max(axis=0)[1] > 0:
        centers[:, 1] *= 0.92 / centers.max(axis=0)[1]
    centers += 0.04

    return centers


def make_feasible(centers, radii, n, max_iter=2000):
    """Make configuration feasible by resolving collisions and boundary violations."""
    for iteration in range(max_iter):
        improved = False

        # Fix boundary violations first
        for i in range(n):
            r = radii[i]
            x, y = centers[i]

            new_x = max(r + 1e-8, min(1 - r - 1e-8, x))
            new_y = max(r + 1e-8, min(1 - r - 1e-8, y))

            if abs(new_x - x) > 1e-12 or abs(new_y - y) > 1e-12:
                centers[i] = [new_x, new_y]
                improved = True

            max_r = min(new_x - 1e-8, 1 - new_x - 1e-8, new_y - 1e-8, 1 - new_y - 1e-8)
            if radii[i] > max_r and max_r > 0:
                radii[i] = max_r
                improved = True

        # Fix overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j] + 1e-8

                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    fx = overlap * dx / dist / 2
                    fy = overlap * dy / dist / 2
                    centers[i] += [fx, fy]
                    centers[j] -= [fx, fy]
                    improved = True
                elif dist < 1e-10:
                    # Separated almost overlapping circles
                    centers[i] += [0.01, 0.01]
                    centers[j] -= [0.01, 0.01]
                    improved = True

        # Re-clamp after pushing
        for i in range(n):
            radii[i] = min(radii[i], min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]) - 1e-8)

        if not improved:
            break

    return centers, radii


def force_expand(centers, radii, n, steps=3000, expansion_rate=0.002):
    """Expand radii using force-directed method."""
    for step in range(steps):
        forces = np.zeros((n, 2))

        # Repulsion between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]

                if dist < min_dist + 0.02 and dist > 1e-10:
                    push = max(0, min_dist - dist + 0.005)
                    strength = push * 50
                    fx = strength * dx / dist
                    fy = strength * dy / dist
                    forces[i] += [fx, fy]
                    forces[j] -= [fx, fy]

        # Boundary repulsion
        for i in range(n):
            margin = 0.001
            if centers[i, 0] - radii[i] < margin:
                forces[i, 0] += (margin - (centers[i, 0] - radii[i])) * 200
            if 1 - centers[i, 0] - radii[i] < margin:
                forces[i, 0] -= (margin - (1 - centers[i, 0] - radii[i])) * 200
            if centers[i, 1] - radii[i] < margin:
                forces[i, 1] += (margin - (centers[i, 1] - radii[i])) * 200
            if 1 - centers[i, 1] - radii[i] < margin:
                forces[i, 1] -= (margin - (1 - centers[i, 1] - radii[i])) * 200

        # Apply forces
        centers += forces * 0.05

        # Expand radii slightly
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]) - 1e-8
            radii[i] = min(radii[i] * (1 + expansion_rate), max_r)

        # Keep centers in bounds
        for i in range(n):
            centers[i, 0] = max(radii[i] + 1e-8, min(1 - radii[i] - 1e-8, centers[i, 0]))
            centers[i, 1] = max(radii[i] + 1e-8, min(1 - radii[i] - 1e-8, centers[i, 1]))

    return centers, radii


def validate_packing(centers, radii):
    """Validate that circles don't overlap and are inside the unit square."""
    n = centers.shape[0]

    if np.isnan(centers).any():
        return False

    if np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True


def run_packing():
    n = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Different hexagonal lattice configurations to try
    configs = []

    # Standard hexagonal spacings
    for base_s in [0.16, 0.17, 0.18, 0.19, 0.20]:
        for rf in [0.9, 1.0, 1.1]:
            centers = init_hex_centers(n, base_s, rf)
            if np.all(centers >= 0) and np.all(centers <= 1):
                configs.append(centers.copy())

    # Add some random configurations
    for seed in range(20):
        rng = np.random.RandomState(seed)
        centers = rng.uniform(0.15, 0.85, (n, 2))
        configs.append(centers)

    cons = build_constraints(n)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    for cfg_idx, centers_init in enumerate(configs):
        # Small random perturbation
        rng = np.random.RandomState(cfg_idx * 31 + 7)
        centers = centers_init + rng.randn(n, 2) * 0.005
        centers = np.clip(centers, 0.05, 0.95)

        # Estimate initial radius
        est_r = 0.09
        radii = np.full(n, est_r)

        # Make feasible
        centers, radii = make_feasible(centers, radii, n)

        # Force expand to get closer to optimum
        centers, radii = force_expand(centers, radii, n, steps=2000, expansion_rate=0.003)

        # Re-make feasible after expansion
        centers, radii = make_feasible(centers, radii, n)

        x0 = np.concatenate([centers.flatten(), radii])

        try:
            result = minimize(
                obj_func,
                x0,
                method="SLSQP",
                constraints=cons,
                args=(n,),
                bounds=bounds,
                options={"maxiter": 5000, "ftol": 1e-14, "disp": False},
            )

            current_sum = -result.fun

            # Extract and validate solution
            c = result.x[: 2 * n].reshape(n, 2)
            r = result.x[2 * n :]

            # Clamp radii to be feasible
            for i in range(n):
                r[i] = min(r[i], min(c[i, 0], 1 - c[i, 0], c[i, 1], 1 - c[i, 1]) - 1e-9)
                c[i, 0] = max(r[i] + 1e-9, min(1 - r[i] - 1e-9, c[i, 0]))
                c[i, 1] = max(r[i] + 1e-9, min(1 - r[i] - 1e-9, c[i, 1]))

            if validate_packing(c, r):
                current_sum_valid = np.sum(r)
                if current_sum_valid > best_sum:
                    best_sum = current_sum_valid
                    best_centers = c.copy()
                    best_radii = r.copy()
        except Exception:
            continue

    # If SLSQP didn't find anything valid, use the best force-expanded result
    if best_centers is None:
        best_sum = np.sum(radii)
        best_centers = centers
        best_radii = radii

    return best_centers, best_radii, best_sum
