import numpy as np
from scipy.optimize import minimize, linprog

# --- Constants and Helper Functions ---

N_CIRCLES = 26
PENALTY_WEIGHT = 10000.0
BOUND = 0.5 # Max possible radius

def compute_overlap_penalty(centers, radii):
    """
    Computes the quadratic penalty for circle-circle and circle-boundary overlaps.
    Returns the total penalty value.
    """
    n = len(radii)
    penalty = 0.0
    
    # --- Pairwise Overlaps ---
    # Compute distance matrix efficiently
    # centers shape: (n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Compute sum of radii matrix
    sum_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount: max(0, r_i + r_j - dist_ij)
    # We only need to check upper triangle to avoid double counting
    overlaps = np.maximum(0, sum_radii - dists)
    penalty += PENALTY_WEIGHT * np.sum(overlaps[np.triu_indices(n, k=1)]**2)
    
    # --- Boundary Overlaps ---
    # Circle i must satisfy: x_i - r_i >= 0, x_i + r_i <= 1, etc.
    # Violations: max(0, r_i - x_i), max(0, r_i - (1 - x_i)), etc.
    
    # Left and Right boundaries (X)
    penalty += PENALTY_WEIGHT * np.sum(np.maximum(0, radii - centers[:, 0])**2)
    penalty += PENALTY_WEIGHT * np.sum(np.maximum(0, radii - (1.0 - centers[:, 0]))**2)
    
    # Bottom and Top boundaries (Y)
    penalty += PENALTY_WEIGHT * np.sum(np.maximum(0, radii - centers[:, 1])**2)
    penalty += PENALTY_WEIGHT * np.sum(np.maximum(0, radii - (1.0 - centers[:, 1]))**2)
    
    return penalty

def objective_function(x):
    """
    Objective function for scipy.optimize.minimize.
    We minimize ( -Sum(Radii) + Penalty ).
    """
    # Reshape flat vector x to (N, 3) -> [x, y, r] for each circle
    pts = x.reshape(N_CIRCLES, 3)
    centers = pts[:, :2]
    radii = pts[:, 2]
    
    # Primary goal: Maximize sum of radii (so minimize negative sum)
    obj_val = -np.sum(radii)
    
    # Add penalty for constraints
    penalty = compute_overlap_penalty(centers, radii)
    
    return obj_val + penalty

def run_lp_refinement(centers):
    """
    Given fixed centers, solve a Linear Programming problem to find the
    maximal valid radii. This guarantees a valid packing and maximizes
    the sum of radii for the given geometry.
    
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= dist(c_i, c_j) for all i < j
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints matrix A_ub * r <= b_ub
    # We will build lists of rows for A_ub and b_ub
    a_rows = []
    b_values = []
    
    # 1. Pairwise constraints: r_i + r_j <= dist_ij
    # Compute distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            a_rows.append(row)
            b_values.append(d)
            
    # 2. Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row = np.zeros(n); row[i] = 1.0; a_rows.append(row); b_values.append(x)
        # r_i <= 1 - x
        row = np.zeros(n); row[i] = 1.0; a_rows.append(row); b_values.append(1.0 - x)
        # r_i <= y
        row = np.zeros(n); row[i] = 1.0; a_rows.append(row); b_values.append(y)
        # r_i <= 1 - y
        row = np.zeros(n); row[i] = 1.0; a_rows.append(row); b_values.append(1.0 - y)
        
    A_ub = np.array(a_rows)
    b_ub = np.array(b_values)
    
    # Bounds for r: [0, infinity) - practically [0, 0.5]
    bounds = [(0, BOUND) for _ in range(n)]
    
    # Solve LP
    # method='highs' is fast and robust
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback if LP fails (should not happen)
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def get_initial_configurations():

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Main function to pack 26 circles in a unit square.
    """
    # Define bounds for optimization variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, BOUND)])
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple restarts
    initial_configs = get_initial_configurations()
    
    for i, x0 in enumerate(initial_configs):
        try:
            # Run Optimization
            res = minimize(
                objective_function,
                x0,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-9}
            )
            
            if res.success:
                # Extract results
                pts = res.x.reshape(N_CIRCLES, 3)
                centers_opt = pts[:, :2]
                
                # Refine with LP to get valid radii
                radii_valid, sum_r = run_lp_refinement(centers_opt)
                
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_centers = centers_opt
                    best_radii = radii_valid
        except Exception:
            continue

    # Fallback if nothing worked (unlikely)
    if best_centers is None:
        best_centers = np.random.rand(N_CIRCLES, 2) * 0.5 + 0.25
        best_radii = np.full(N_CIRCLES, 0.01)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
