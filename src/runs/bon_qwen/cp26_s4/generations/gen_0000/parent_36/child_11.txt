

<strategy>
I'll solve this circle packing problem using a hybrid approach combining linear programming for radius optimization and gradient-based optimization for center positioning. The key insight is that for fixed centers, finding optimal radii is a linear program, while for fixed radii, finding good centers can be done with gradient descent.

Strategy:
1. Create multiple initial hexagonal lattice configurations with different row arrangements and offsets
2. For each configuration, alternate between:
   - Optimizing radii given centers using LP (scipy linprog)
   - Optimizing centers given radii using gradient descent with repulsion forces
3. Run multiple restarts from different initial positions to escape local optima
4. Return the best valid packing found

The hexagonal arrangement is optimal for dense circle packing, and allowing unequal radii lets us fill gaps more efficiently than equal-radius packing.

import numpy as np
from scipy.optimize import linprog, minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    # Try many different initial configurations
    all_results = []
    
    for trial in range(40):
        centers = create_hex_initial(n, trial)
        radii = np.full(n, 0.085)
        
        # Run the alternating optimization
        centers, radii = alternating_optimize(centers, radii, n, iters=200)
        
        current_sum = np.sum(radii)
        all_results.append((centers.copy(), radii.copy(), current_sum))
    
    # Pick best valid result
    for centers, radii, s in all_results:
        if s > best_sum and is_valid(centers, radii):
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii, best_sum


def create_hex_initial(n: int, trial: int) -> np.ndarray:
    centers = np.zeros((n, 2))
    idx = 0
    
    # Different row arrangements to explore diverse configurations
    if trial < 8:
        row_counts = [6, 5, 6, 5, 4]
    elif trial < 16:
        row_counts = [5, 6, 5, 6, 4]
    elif trial < 24:
        row_counts = [6, 6, 5, 5, 4]
    elif trial < 32:
        row_counts = [5, 5, 6, 5, 5]
    else:
        row_counts = [6, 5, 5, 5, 5]
    
    # Vary spacing and offsets per trial
    spacing_x = 0.17 + (trial % 10) * 0.003
    spacing_y = 0.148 + (trial % 7) * 0.004
    offset_x = ((trial * 3) % 20 - 10) * 0.004
    offset_y = ((trial * 7) % 20 - 10) * 0.004
    
    for row_idx, count in enumerate(row_counts):
        y = 0.12 + offset_y + row_idx * spacing_y
        x_start = (1 - (count - 1) * spacing_x) / 2 + offset_x
        
        for col in range(count):
            hex_offset = (spacing_x / 2) * (row_idx % 2)
            x = x_start + col * spacing_x + hex_offset
            x = np.clip(x, 0.02, 0.98)
            y_clipped = np.clip(y, 0.02, 0.98)
            centers[idx] = [x, y_clipped]
            idx += 1
    
    return centers


def alternating_optimize(centers: np.ndarray, radii: np.ndarray, n: int, iters: int = 200) -> tuple[np.ndarray, np.ndarray]:
    for it in range(iters):
        # Step 1: Optimize radii using LP given fixed centers
        radii_new = optimize_radii_lp(centers, n)
        if radii_new is None:
            break
        radii = radii_new
        
        # Step 2: Optimize centers given fixed radii using force-based method
        centers = optimize_centers_force(centers, radii, n, step=0.002 / (1 + it * 0.02))
    
    return centers, radii


def optimize_radii_lp(centers: np.ndarray, n: int) -> np.ndarray:
    """Maximize sum of radii given fixed centers using linear programming."""
    c = -np.ones(n)  # minimize -sum(r) = maximize sum(r)
    
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= distance to each boundary
    for i in range(n):
        x, y = centers[i]
        max_r_boundary = min(x, 1 - x, y, 1 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(max_r_boundary)
    
    # Non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    bounds = [(0.0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None


def optimize_centers_force(centers: np.ndarray, radii: np.ndarray, n: int, step: float) -> np.ndarray:
    """Optimize centers using repulsive forces between overlapping/near circles."""
    new_centers = centers.copy()
    forces = np.zeros((n, 2))
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = new_centers[i] - new_centers[j]
            dist = np.linalg.norm(diff)
            
            if dist < 1e-10:
                dist = 1e-10
                diff = np.array([0.01, 0.0])
            
            required_dist = radii[i] + radii[j]
            gap = dist - required_dist
            
            # Strong repulsion when overlapping, weaker when close
            if gap < 0.05:
                force_mag = max(0, -gap * 50 + 0.5)
                if dist > 1e-10:
                    force_dir = diff / dist
                    forces[i] += force_dir * force_mag
                    forces[j] -= force_dir * force_mag
    
    # Also add gentle repulsion for all pairs to spread out
    for i in range(n):
        for j in range(i + 1, n):
            diff = new_centers[i] - new_centers[j]
            dist = np.linalg.norm(diff)
            if dist < 0.3 and dist > 1e-10:
                repulse = 0.01 / (dist ** 2 + 0.001)
                force_dir = diff / dist
                forces[i] += force_dir * repulse
                forces[j] -= force_dir * repulse
    
    # Boundary forces: push inward if too close to boundary
    for i in range(n):
        x, y = new_centers[i]
        r = radii[i]
        
        # Left boundary
        if x - r < 0.01:
            forces[i][0] += (0.01 - (x - r)) * 20
        # Right boundary
        if x + r > 0.99:
            forces[i][0] -= (x + r - 0.99) * 20
        # Bottom boundary
        if y - r < 0.01:
            forces[i][1] += (0.01 - (y - r)) * 20
        # Top boundary
        if y + r > 0.99:
            forces[i][1] -= (y + r - 0.99) * 20
    
    # Apply forces
    max_step = step * 0.05
    for i in range(n):
        force_norm = np.linalg.norm(forces[i])
        if force_norm > 0:
            actual_step = min(max_step, force_norm * step)
            new_centers[i] += (forces[i] / force_norm) * actual_step
    
    # Clip to valid range
    for i in range(n):
        new_centers[i][0] = np.clip(new_centers[i][0], radii[i] + 1e-8, 1 - radii[i] - 1e-8)
        new_centers[i][1] = np.clip(new_centers[i][1], radii[i] + 1e-8, 1 - radii[i] - 1e-8)
    
    return new_centers


def is_valid(centers: np.ndarray, radii: np.ndarray) -> bool:
    """Check if packing is valid (no overlaps, inside square)."""
    n = centers.shape[0]
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0 or np.isnan(r):
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    all_candidates = []
    
    # Generate many initial configurations
    for trial in range(50):
        centers = create_hex_initial(n, trial)
        radii = np.full(n, 0.085)
        
        # Alternating optimization
        centers, radii = alternating_optimize(centers, radii, n, iters=300)
        
        current_sum = np.sum(radii)
        
        # Refine with more iterations if promising
        if current_sum > 2.0:
            centers, radii = alternating_optimize(centers, radii, n, iters=100)
            current_sum = np.sum(radii)
        
        all_candidates.append((centers.copy(), radii.copy(), current_sum))
    
    # Select best valid solution
    for centers, radii, s in sorted(all_candidates, key=lambda x: -x[2]):
        if is_valid(centers, radii) and s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
            if best_sum > 2.5:
                break  # Good enough, stop searching
    
    return best_centers, best_radii, best_sum
