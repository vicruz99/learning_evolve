    Tries to fit n circles into a hexagonal grid pattern scaled to fit in unit square.
    """

    r_start = 0.09

    radii = []

    # Col width: 2 * r
    row_height = r_start * math.sqrt(3)
    col_width = 2 * r_start
    
    y = r_start
    row_idx = 0
    
    while len(centers) < n:
        # Alternate starting x for staggered rows
        x_offset = r_start if row_idx % 2 == 0 else r_start + r_start
        x = x_offset
        
        while x + r_start <= 1.0:
            if len(centers) < n:
                centers.append([x, y])
                radii.append(r_start)
            x += col_width
        
        y += row_height
        row_idx += 1
        if y + r_start > 1.0:
            break
            
    return np.array(centers), np.array(radii)

def get_random_initial_guess(n):

    Generates a random initial guess.
    """
    centers = []
    radii = []
    r_start = 0.08
    attempts = 0
    while len(centers) < n and attempts < 10000:
        x = np.random.uniform(r_start, 1.0 - r_start)
        y = np.random.uniform(r_start, 1.0 - r_start)

                break
        
        if valid:
            centers.append([x, y])
            radii.append(r_start)
        attempts += 1
        
    return np.array(centers), np.array(radii)

def objective(vars_vec, n):
    """
    Objective function to maximize sum of radii.
    We minimize negative sum.
    """
    radii = vars_vec[2::3]
    return -np.sum(radii)

def constraint_overlap(vars_vec, i, j, n):
    """
    Constraint: dist(i, j) >= r_i + r_j
    Equivalent to: dist^2 - (r_i + r_j)^2 >= 0
    """
    x_i, y_i, r_i = vars_vec[3*i], vars_vec[3*i+1], vars_vec[3*i+2]
    x_j, y_j, r_j = vars_vec[3*j], vars_vec[3*j+1], vars_vec[3*j+2]
    
    dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
    sum_r_sq = (r_i + r_j)**2
    
    return dist_sq - sum_r_sq

def constraint_boundary_x(vars_vec, i, n):
    """
    Constraint: r <= x <= 1-r
    Two inequalities: x - r >= 0 and 1 - x - r >= 0
    """
    x = vars_vec[3*i]
    r = vars_vec[3*i+2]
    # Return value that must be >= 0
    # We need to return an array for scipy? Or handle separately.

    return np.array([x - r, 1.0 - x - r])

def constraint_boundary_y(vars_vec, i, n):
    """
    Constraint: r <= y <= 1-r
    """
    y = vars_vec[3*i+1]
    r = vars_vec[3*i+2]
    return np.array([y - r, 1.0 - y - r])
