def objective(vars):

    Objective function: Minimize negative sum of radii.
    vars: [x1, x2, ..., x26, y1, ..., y26, r1, ..., r26]

def constraints(vars):
    """
    Computes inequality constraints: g(vars) >= 0.
    1. Boundary constraints
    2. Pairwise non-overlap constraints
    """
    n = 26
    cx = vars[:26]
    cy = vars[26:52]
    cr = vars[52:]
    
    # Boundary constraints
    # x_i >= r_i  => x_i - r_i >= 0
    # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i
    # y_i <= 1 - r_i
    
    cons = []
    cons.extend(cx - cr)
    cons.extend(1.0 - cx - cr)
    cons.extend(cy - cr)
    cons.extend(1.0 - cy - cr)
    
    # Pairwise non-overlap constraints
    # dist_sq >= (r_i + r_j)^2
    # dist_sq - (r_i + r_j)^2 >= 0
    
    # Vectorized computation
    # Create matrices of differences
    # Shape (n, n)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx*dx + dy*dy
    
    r_sum = cr[:, None] + cr[None, :]
    r_sum_sq = r_sum * r_sum
    
    # Constraint matrix
    constraint_matrix = dist_sq - r_sum_sq
    
    # We only need constraints for i < j (upper triangle excluding diagonal)
    # Diagonal is 0 - 4r^2 which is negative (invalid), so ignore diagonal.
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pairwise_cons = constraint_matrix[mask]
    
    cons.extend(pairwise_cons)
    
    return np.array(cons)
