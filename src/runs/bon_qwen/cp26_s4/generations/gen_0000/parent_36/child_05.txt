def generate_initial(n, seed):

    strategy = seed % 4

    if strategy == 0:
        centers = hexagonal_layout(n)

def hexagonal_layout(n):
    centers = np.zeros((n, 2))
    row_counts = [6, 5, 6, 5, 4]
    num_rows = len(row_counts)
    row_y = np.linspace(0.15, 0.85, num_rows)
    idx = 0
    for row_idx, count in enumerate(row_counts):
        y = row_y[row_idx]
        x_positions = np.linspace(0.08, 0.92, count)
        if row_idx % 2 == 1:
            spacing = x_positions[1] - x_positions[0]
            x_positions = x_positions + spacing / 2
        for x in x_positions:
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    return centers


def grid_layout(n):
    centers = np.zeros((n, 2))
    cols = 5
    rows = 5
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx < n:
                centers[idx] = [0.1 + j * 0.2, 0.1 + i * 0.2]
                idx += 1
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    return centers


def corner_biased_layout(n):
    centers = np.zeros((n, 2))
    corner_r = 0.22
    centers[0] = [corner_r, corner_r]
    centers[1] = [1 - corner_r, corner_r]
    centers[2] = [corner_r, 1 - corner_r]
    centers[3] = [1 - corner_r, 1 - corner_r]
    edge_r = 0.13
    centers[4] = [0.5, edge_r]
    centers[5] = [edge_r, 0.5]
    centers[6] = [1 - edge_r, 0.5]
    centers[7] = [0.5, 1 - edge_r]
    idx = 8
    remaining = n - idx
    grid_n = remaining
    gs = int(np.ceil(np.sqrt(grid_n)))
    spacing = 0.4 / (gs + 1)
    for i in range(gs):
        for j in range(gs):
            if idx < n:
                centers[idx] = [0.3 + j * spacing * 2, 0.3 + i * spacing * 2]
                idx += 1
    return centers


def random_layout(n):
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i] = [np.random.rand(), np.random.rand()]
    return centers


def force_based_relaxation(centers, n, steps=4000):
    centers = centers.copy()

    for step in range(steps):
        lr = 0.015 * (0.998 ** step)
        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < 1e-10:
                    dist = 1e-10
                    diff = np.random.rand(2) * 1e-5

                target_sep = 0.16
                if dist < target_sep:
                    repulsion = (target_sep - dist) / (dist + 1e-10) * 3
                    forces[i] += repulsion * diff
                    forces[j] -= repulsion * diff

            for axis in range(2):
                if centers[i, axis] < 0.03:
                    forces[i, axis] += (0.03 - centers[i, axis]) * 80
                if centers[i, axis] > 0.97:
                    forces[i, axis] -= (centers[i, axis] - 0.97) * 80

        centers += lr * forces
        centers = np.clip(centers, 0.01, 0.99)

    return centers


def compute_clearance_radii(centers, n):
    radii = np.zeros(n)
    for i in range(n):
        min_clearance = min(
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1]
        )
        for j in range(n):
            if i != j:
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < min_clearance:
                    min_clearance = dist
        radii[i] = max(0.0, min_clearance / 2.0 - 1e-12)
    return radii
