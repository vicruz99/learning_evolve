import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    # Helper function to flatten centers and radii
    def flatten(centers, radii):
        return np.concatenate([centers.flatten(), radii])

    # Helper function to unflatten
    def unflatten(x):
        centers = x[:2 * n_circles].reshape((n_circles, 2))
        radii = x[2 * n_circles:]
        return centers, radii

    # Objective function: maximize sum of radii -> minimize -sum(radii)
    def objective(x):
        _, radii = unflatten(x)
        return -np.sum(radii)

    # Constraints
    # 1. Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # x - r >= 0  =>  x - r >= 0
    # 1 - x - r >= 0 => 1 - x - r >= 0
    # Same for y
    
    # 2. Non-overlap: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0

    def get_constraints():
        constraints = []

        # Boundary constraints
        for i in range(n_circles):
            idx_x = 2 * i
            idx_y = 2 * i + 1
            idx_r = 2 * n_circles + i
            
            # x >= r  => x - r >= 0
            def c_x_lower(x, i=i):
                return x[2 * i] - x[2 * n_circles + i]
            
            # 1 - x - r >= 0
            def c_x_upper(x, i=i):
                return 1.0 - x[2 * i] - x[2 * n_circles + i]
            
            # y >= r
            def c_y_lower(x, i=i):
                return x[2 * i + 1] - x[2 * n_circles + i]
            
            # 1 - y - r >= 0
            def c_y_upper(x, i=i):
                return 1.0 - x[2 * i + 1] - x[2 * n_circles + i]

            constraints.append({'type': 'ineq', 'fun': c_x_lower})
            constraints.append({'type': 'ineq', 'fun': c_x_upper})
            constraints.append({'type': 'ineq', 'fun': c_y_lower})
            constraints.append({'type': 'ineq', 'fun': c_y_upper})

        # Non-overlap constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_xi = 2 * i
                idx_yi = 2 * i + 1
                idx_ri = 2 * n_circles + i
                
                idx_xj = 2 * j
                idx_yj = 2 * j + 1
                idx_rj = 2 * n_circles + j
                
                def c_overlap(x, i=i, j=j):
                    xi = x[2 * i]
                    yi = x[2 * i + 1]
                    ri = x[2 * n_circles + i]
                    
                    xj = x[2 * j]
                    yj = x[2 * j + 1]
                    rj = x[2 * n_circles + j]
                    
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r = ri + rj
                    return dist_sq - sum_r**2

                constraints.append({'type': 'ineq', 'fun': c_overlap})

        return constraints

    constraints = get_constraints()

    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5] (max possible radius)
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
    for _ in range(n_circles):
        bounds.append((0.0, 0.5)) # r

    # Run optimization multiple times with different initializations
    num_trials = 20
    
    # Try a structured initialization first (Hexagonal grid)
    # Then random perturbations
    
    for trial in range(num_trials):
        # Initial guess
        # Option 1: Grid/Hex packing
        if trial == 0:
            # Create a hexagonal packing pattern
            centers_init = []
            # Approximate radius 0.1
            r_init = 0.1
            rows = 5
            cols = 6 # Try to fit more
            # Let's just place them in a grid and let optimizer fix it
            # Grid 6x5 = 30 spots, take 26
            # Or 5x5 = 25, plus 1
            # Let's do a dense random pack
            # Better: Hexagonal lattice
            pts = []
            y = r_init
            row = 0
            while len(pts) < n_circles:
                x = r_init
                if row % 2 == 1:
                    x = r_init + r_init # Shift by r
                while x <= 1.0 - r_init and len(pts) < n_circles:
                    pts.append([x, y])
                    x += 2 * r_init
                y += r_init * np.sqrt(3)
                row += 1
            
            # If not enough, just fill remaining randomly
            while len(pts) < n_circles:
                pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
            
            centers_init = np.array(pts[:n_circles])
            radii_init = np.full(n_circles, r_init)
        else:
            # Random initialization with small radii
            centers_init = np.random.rand(n_circles, 2) * 0.8 + 0.1 # Keep away from edges initially
            radii_init = np.full(n_circles, 0.05)

        x0 = flatten(centers_init, radii_init)

        try:
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                               options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                centers_opt, radii_opt = unflatten(res.x)
                sum_r = np.sum(radii_opt)

                        best_sum_radii = sum_r
                        best_centers = centers_opt.copy()
                        best_radii = radii_opt.copy()
        except Exception:
            continue

    # If no valid solution found from optimization, return a valid fallback (e.g. small circles)
    if best_centers is None:
        best_centers = np.random.rand(n_circles, 2) * 0.8 + 0.1
        best_radii = np.full(n_circles, 0.05)
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum_radii)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    import numpy as np
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        # print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        # print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            # print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            # print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            # print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                # print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
