

<strategy>
The problem is a continuous non-convex optimization task: maximize $\sum r_i$ subject to circle non-overlap and boundary constraints. 

1. **Formulation**: Variables are $(x_i, y_i, r_i)$ for $i=1..26$. Constraints: $r_i \le x_i \le 1-r_i$, $r_i \le y_i \le 1-r_i$, and $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ for all $i < j$.
2. **Initialization**: A hexagonal lattice provides higher packing density than a square grid. We generate 26 points on a hex grid, scale/translate them to lie safely inside $[0.1, 0.9]$, and set initial radii to $0.08$. This guarantees a strictly feasible starting point.
3. **Optimization**: We use `scipy.optimize.minimize` with the SLSQP algorithm, which is designed for constrained non-linear problems.
4. **Performance & Accuracy**: We provide analytical gradients (Jacobians) for the objective and all constraints. This avoids slow/numerical gradient approximations and significantly improves convergence speed and precision. Constraint evaluations are vectorized using NumPy slicing.
5. **Symmetry Breaking**: The inherent offset in hexagonal rows prevents the optimizer from getting stuck in symmetric local optima, guiding it toward a tighter, near-optimal configuration where circles mutually touch and press against boundaries.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize

N = 26
SEP_IDX = list(zip(*np.triu_indices(N, k=1)))

def objective(x):
    # Maximize sum of radii <=> minimize negative sum
    return -np.sum(x[2::3])

def objective_jac(x):
    jac = np.zeros_like(x)
    jac[2::3] = -1.0
    return jac

def boundary_constraints(x):
    val = np.empty(4*N)
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    val[0::4] = xc - r
    val[1::4] = 1.0 - xc - r
    val[2::4] = yc - r
    val[3::4] = 1.0 - yc - r
    return val

def boundary_jac(x):
    jac = np.zeros((4*N, 3*N))
    jac[0::4, 0::3] = 1.0
    jac[0::4, 2::3] = -1.0
    jac[1::4, 0::3] = -1.0
    jac[1::4, 2::3] = -1.0
    jac[2::4, 1::3] = 1.0
    jac[2::4, 2::3] = -1.0
    jac[3::4, 1::3] = -1.0
    jac[3::4, 2::3] = -1.0
    return jac

def separation_constraints(x):
    val = np.empty(len(SEP_IDX))
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    for k, (i, j) in enumerate(SEP_IDX):
        dx = xc[i] - xc[j]
        dy = yc[i] - yc[j]
        rs = r[i] + r[j]
        val[k] = dx*dx + dy*dy - rs*rs
    return val

def separation_jac(x):
    jac = np.zeros((len(SEP_IDX), 3*N))
    xc = x[0::3]
    yc = x[1::3]
    r = x[2::3]
    for k, (i, j) in enumerate(SEP_IDX):
        dx = xc[i] - xc[j]
        dy = yc[i] - yc[j]
        rs = r[i] + r[j]
        jac[k, 3*i] = 2*dx
        jac[k, 3*i+1] = 2*dy
        jac[k, 3*i+2] = -2*rs
        jac[k, 3*j] = -2*dx
        jac[k, 3*j+1] = -2*dy
        jac[k, 3*j+2] = -2*rs
    return jac

def run_packing():
    # Initial hexagonal lattice guess
    r_init = 0.08
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    pts = []
    j = 0
    while len(pts) < N:
        i = 0
        while len(pts) < N:
            x = i * dx + (j % 2) * r_init
            y = j * dy
            pts.append([x, y])
            i += 1
        j += 1
        
    pts = np.array(pts[:N])
    # Normalize to safely fit inside [0.1, 0.9]
    pts = (pts - pts.min(axis=0)) / (pts.max(axis=0) - pts.min(axis=0)) * 0.8 + 0.1
    
    x0 = np.zeros(N * 3)
    for i in range(N):
        x0[3*i] = pts[i, 0]
        x0[3*i+1] = pts[i, 1]
        x0[3*i+2] = r_init
        
    bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N
    
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints, 'jac': boundary_jac},
        {'type': 'ineq', 'fun': separation_constraints, 'jac': separation_jac}
    ]
    
    res = minimize(objective, x0, method='SLSQP', jac=objective_jac, 
                   bounds=bounds, constraints=cons, 
                   options={'maxiter': 5000, 'ftol': 1e-12})
    
    centers = res.x[:2*N].reshape(N, 2)
    radii = res.x[2*N:]
    return centers, radii, np.sum(radii)
```
