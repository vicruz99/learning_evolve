import numpy as np
from scipy.optimize import minimize

def compute_energy_and_grad(centers_flat, n=26):
    """
    Computes the repulsive energy and its gradient for a given configuration of centers.
    Energy is sum of inverse squared distances between pairs and boundaries.
    """
    c = centers_flat.reshape(-1, 2)
    E = 0.0
    grad = np.zeros_like(c)
    
    # Pairwise repulsion
    for i in range(n):
        for j in range(i + 1, n):
            diff = c[i] - c[j]
            dist_sq = np.dot(diff, diff)
            if dist_sq < 1e-8:
                dist_sq = 1e-8
            dist = np.sqrt(dist_sq)
            inv_d3 = 1.0 / (dist ** 3)
            inv_d2 = 1.0 / dist_sq
            E += inv_d2
            grad[i] -= 2.0 * diff * inv_d3
            grad[j] += 2.0 * diff * inv_d3
            
    # Boundary repulsion
    for i in range(n):
        x, y = c[i]
        for k, val in enumerate([x, y]):
            # Distance to left/bottom wall
            d0 = val
            if d0 < 1e-4:
                d0 = 1e-4
            E += 1.0 / (d0 ** 2)
            grad[i, k] -= 2.0 / (d0 ** 3)
            
            # Distance to right/top wall
            d1 = 1.0 - val
            if d1 < 1e-4:
                d1 = 1e-4
            E += 1.0 / (d1 ** 2)
            grad[i, k] += 2.0 / (d1 ** 3)
            
    return E, grad.flatten()

def objective_fun(x):
    return compute_energy_and_grad(x, 26)[0]

def gradient_fun(x):
    return compute_energy_and_grad(x, 26)[1]

def compute_max_equal_radius(centers):
    """
    Computes the maximum possible equal radius for a given set of centers
    such that no circles overlap and all stay inside the unit square.
    """
    n = centers.shape[0]
    # Distance to boundaries
    min_d = min(np.min(centers[:, 0]), np.min(1.0 - centers[:, 0]),
                np.min(centers[:, 1]), np.min(1.0 - centers[:, 1]))
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs ** 2, axis=2))
    np.fill_diagonal(dists, 1.0)
    min_d = min(min_d, np.min(dists) / 2.0)
    
    return min_d

def run_packing():
    n = 26
    best_r = 0.0
    best_centers = None
    
    # Generate multiple initial configurations to avoid local minima
    inits = []
    for seed in range(50):
        np.random.seed(seed)
        inits.append(np.random.rand(n, 2))
        
    # Add a structured grid initialization
    pts = []
    for i in range(6):
        for j in range(5):
            pts.append((0.1 + i * 0.15, 0.1 + j * 0.18))
    np.random.shuffle(pts)
    inits.append(np.array(pts[:n]))
    
    bounds = [(0.001, 0.999)] * 52
    
    # Optimize each initialization
    for init in inits:
        try:
            res = minimize(objective_fun, init.flatten(), jac=gradient_fun, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-12})
            fc = res.x.reshape(-1, 2)
            r = compute_max_equal_radius(fc)
            if r > best_r:
                best_r = r
                best_centers = fc.copy()
        except Exception:
            continue
            
    # Local perturbation refinement on the best found configuration
    if best_centers is not None:
        for _ in range(10):
            pert = best_centers + np.random.normal(0, 0.005, size=best_centers.shape)
            pert = np.clip(pert, 0.001, 0.999)
            try:
                res = minimize(objective_fun, pert.flatten(), jac=gradient_fun, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 500, 'ftol': 1e-12})
                fc = res.x.reshape(-1, 2)
                r = compute_max_equal_radius(fc)
                if r > best_r:
                    best_r = r
                    best_centers = fc.copy()
            except Exception:
                pass
                
    radii = np.full(n, best_r)
    return best_centers, radii, 26 * best_r
