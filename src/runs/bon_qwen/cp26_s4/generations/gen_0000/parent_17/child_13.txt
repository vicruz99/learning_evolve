import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Function to solve the packing problem
    def solve_packing(init_centers, init_radii):
        # Variable vector: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
        # Size: 3 * n_circles
        x0 = np.hstack([init_centers, init_radii.reshape(-1, 1)]).flatten()
        
        # Bounds for variables
        # x, y in [0, 1], r >= 0
        # Actually, x, y must be in [r, 1-r], but SLSQP handles constraints.
        # We set loose bounds [0, 1] for x,y and [0, 0.5] for r.
        bounds = []
        for _ in range(n_circles):
            bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
            
        # Constraints
        constraints = []
        
        # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        # x - r >= 0  => r - x <= 0
        # x + r <= 1  => x + r - 1 <= 0
        # Same for y
        for i in range(n_circles):
            idx = i * 3
            xi, yi, ri = x0[idx], x0[idx+1], x0[idx+2]
            
            # Constraint: x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[i*3] - v[i*3+2]
            })
            # Constraint: 1 - (x + r) >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - (v[i*3] + v[i*3+2])
            })
            # Constraint: y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[i*3+1] - v[i*3+2]
            })
            # Constraint: 1 - (y + r) >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - (v[i*3+1] + v[i*3+2])
            })

        # 2. Non-overlap constraints: dist(i,j) >= ri + rj
        # dist^2 >= (ri+rj)^2  => dist^2 - (ri+rj)^2 >= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: \
                        (v[i*3] - v[j*3])**2 + (v[i*3+1] - v[j*3+1])**2 - (v[i*3+2] + v[j*3+2])**2
                })

        # Objective: Maximize sum of radii => Minimize negative sum
        def objective(v):
            radii_sum = sum(v[i*3+2] for i in range(n_circles))
            return -radii_sum

        try:
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                               options={'maxiter': 1000, 'ftol': 1e-9})
            if res.success or res.fun < -best_sum:
                return res.x, -res.fun
        except Exception:
            pass
        return None, -best_sum

    # Generate initial configurations
    # Strategy 1: 5x5 Grid of r=0.1, plus one small circle in the center gap
    # Grid centers at 0.1, 0.3, 0.5, 0.7, 0.9
    grid_coords = []
    radii_init = []
    coords = np.linspace(0.1, 0.9, 5)
    for x in coords:
        for y in coords:
            grid_coords.append([x, y])
            radii_init.append(0.1)
            
    # 25 circles placed. Add 26th.
    # The center of the square is (0.5, 0.5), but occupied.
    # A good spot for a small circle is between 4 circles, e.g., (0.2, 0.2) is occupied?
    # Grid is (0.1, 0.1), (0.1, 0.3)...
    # Gap at (0.2, 0.2). Distance to (0.1, 0.1) is sqrt(2)*0.1 ~ 0.141.
    # Max radius at gap = 0.141 - 0.1 = 0.041.
    grid_coords.append([0.2, 0.2])
    radii_init.append(0.04)
    
    init_centers = np.array(grid_coords)
    init_radii = np.array(radii_init)
    
    # Solve
    res_x, sum_val = solve_packing(init_centers, init_radii)
    if res_x is not None:
        best_sum = sum_val
        best_centers = res_x[:, :2].reshape(n_circles, 2)
        best_radii = res_x[:, 2]
        
    # Strategy 2: Perturb the grid and try again
    # Randomly perturb centers slightly
    perturbed_centers = init_centers + np.random.normal(0, 0.02, size=init_centers.shape)
    # Clip to [0,1]
    perturbed_centers = np.clip(perturbed_centers, 0.01, 0.99)
    # Slightly reduce radii to ensure validity initially
    perturbed_radii = init_radii * 0.95
    
    res_x2, sum_val2 = solve_packing(perturbed_centers, perturbed_radii)
    if res_x2 is not None and sum_val2 > best_sum:
        best_sum = sum_val2
        best_centers = res_x2[:, :2].reshape(n_circles, 2)
        best_radii = res_x2[:, 2]

    # Strategy 3: Hexagonal packing attempt
    # 5 rows, staggered
    hex_centers = []
    hex_radii = []
    r_est = 0.09
    # 5 rows
    for row in range(5):
        y = 0.1 + row * 0.1732 # approx sqrt(3)/2 * 0.2
        # Stagger every other row
        shift = 0.1 if row % 2 == 1 else 0.0
        num_cols = 5 if row % 2 == 0 else 5 # Try to fit 5 in each
        # Actually for hex, width might be tighter.
        # Let's just place 5 in each row, adjusted x
        for col in range(5):
            x = 0.1 + col * 0.2 + shift
            if x + r_est <= 1.0 and x - r_est >= 0.0:
                hex_centers.append([x, y])
                hex_radii.append(r_est)
    
    # If we have less than 26, add more or adjust. 
    # This initial guess might be valid but small radii.
    # The optimizer should grow them.
    while len(hex_centers) < 26:
        # Fill remaining randomly in empty space?
        # Just pick random spots not too close
        found = False
        for _ in range(100):
            cx, cy = np.random.uniform(0.1, 0.9, 2)
            # Check distance to existing
            dists = np.sqrt(np.sum((np.array(hex_centers) - [cx, cy])**2, axis=1))
            min_dist = np.min(dists) if len(dists) > 0 else 1.0
            if min_dist > 0.2: # Rough check
                hex_centers.append([cx, cy])
                hex_radii.append(0.05)
                found = True
                break
        if not found:
            break
            
    if len(hex_centers) >= 26:
        hex_centers = np.array(hex_centers[:26])
        hex_radii = np.array(hex_radii[:26])
        res_x3, sum_val3 = solve_packing(hex_centers, hex_radii)
        if res_x3 is not None and sum_val3 > best_sum:
            best_sum = sum_val3
            best_centers = res_x3[:, :2].reshape(n_circles, 2)
            best_radii = res_x3[:, 2]

    # Strategy 4: Random start with small circles
    random_centers = np.random.uniform(0.1, 0.9, size=(26, 2))
    random_radii = np.full(26, 0.02)
    res_x4, sum_val4 = solve_packing(random_centers, random_radii)
    if res_x4 is not None and sum_val4 > best_sum:
        best_sum = sum_val4
        best_centers = res_x4[:, :2].reshape(n_circles, 2)
        best_radii = res_x4[:, 2]

    # Final validation and cleanup
    if best_centers is None:
        # Fallback
        best_centers = init_centers
        best_radii = init_radii
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
