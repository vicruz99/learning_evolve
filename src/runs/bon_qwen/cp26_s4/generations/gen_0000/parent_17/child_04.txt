from scipy.optimize import minimize

def run_packing():

    cols = 6
    rows = 5
    # Total 30 points, we will pick 26
    
    # Spacing

    margin = 0.1
    x_coords = np.linspace(margin, 1 - margin, cols)
    y_coords = np.linspace(margin, 1 - margin, rows)
    
    pts = []
    for y in y_coords:
        for x in x_coords:
            pts.append([x, y])
            if len(pts) >= n:
                break
        if len(pts) >= n:
            break
            
    centers_init = np.array(pts[:n])
    
    # Add small random jitter to break symmetry and help escape local minima
    np.random.seed(42)
    jitter = np.random.uniform(-0.05, 0.05, size=(n, 2))
    centers_init += jitter
    
    # Clamp centers to valid range (must be at least radius away from boundary)
    # Initial radius will be small, say 0.02
    r_init = 0.02
    centers_init[:, 0] = np.clip(centers_init[:, 0], r_init + 1e-6, 1.0 - r_init - 1e-6)
    centers_init[:, 1] = np.clip(centers_init[:, 1], r_init + 1e-6, 1.0 - r_init - 1e-6)
    
    radii_init = np.full(n, r_init)
    
    # --- 2. Optimization Setup ---
    # Variables: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]

        bounds.append((0.0, 0.5)) # r

    def boundary_constraints(vars):
        # vars is shape (3n,)
        # Reshape to (n, 3) -> [x, y, r]
        # Actually vars is ordered x's, y's, r's
        # So we need to reshape carefully or just index
        
        xs = vars[0::3]
        ys = vars[1::3]
        rs = vars[2::3]
        
        c1 = xs - rs          # x >= r
        c2 = 1.0 - (xs + rs)  # 1 >= x + r
        c3 = ys - rs          # y >= r
        c4 = 1.0 - (ys + rs)  # 1 >= y + r
        
        return np.concatenate([c1, c2, c3, c4])
    
    def overlap_constraints(vars):
        xs = vars[0::3]
        ys = vars[1::3]
        rs = vars[2::3]
        
        # We need constraints for all pairs i < j
        # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2

        r_sum_sq = dr**2
        
        return dist_sq - r_sum_sq

    constraints = [

    ]

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
