import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple:
    n = 26

    # Create initial hexagonal configuration
    # 5 rows: 5, 6, 5, 6, 4 circles = 26 total
    centers_list = []
    idx = 0
    row_specs = [(5, 0.12), (6, 0.06), (5, 0.12), (6, 0.06), (4, 0.21)]

    for row_idx, (ncols, x_offset) in enumerate(row_specs):
        y = 0.1 + row_idx * 0.2
        for col in range(ncols):
            if idx < n:
                x = x_offset + col * 0.18
                centers_list.append([x, y])
                idx += 1

    centers_init = np.array(centers_list)
    initial_radii = np.full(n, 0.08)

    def objective(x):
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]

        obj = -np.sum(r)
        penalty = 0.0
        w = 10000.0

        for i in range(n):
            xi, yi = c[i]
            ri = r[i]
            if ri > xi:
                penalty += w * (ri - xi) ** 2
            if ri > 1 - xi:
                penalty += w * (ri - (1 - xi)) ** 2
            if ri > yi:
                penalty += w * (ri - yi) ** 2
            if ri > 1 - yi:
                penalty += w * (ri - (1 - yi)) ** 2

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                if r[i] + r[j] > dist:
                    penalty += w * (r[i] + r[j] - dist) ** 2

        return obj + penalty

    x0 = np.concatenate([centers_init.flatten(), initial_radii])

    best_c = centers_init.copy()
    best_r = initial_radii.copy()
    best_sum = np.sum(initial_radii)
    best_valid = False

    def is_valid(c, r):
        for i in range(n):
            xi, yi = c[i]
            ri = r[i]
            if ri < -1e-12:
                return False
            if ri > xi + 1e-12:
                return False
            if ri > 1 - xi + 1e-12:
                return False
            if ri > yi + 1e-12:
                return False
            if ri > 1 - yi + 1e-12:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                if r[i] + r[j] > dist + 1e-12:
                    return False
        return True

    for seed in range(50):
        np.random.seed(seed)
        x0_perturbed = x0.copy()
        x0_perturbed[:2 * n] += np.random.randn(2 * n) * 0.03
        x0_perturbed[2 * n:] += np.random.randn(n) * 0.01
        x0_perturbed[2 * n:] = np.maximum(x0_perturbed[2 * n:], 0.005)

        result = minimize(
            objective,
            x0_perturbed,
            method="Nelder-Mead",
            options={"maxiter": 100000, "xatol": 1e-12, "fatol": 1e-12},
        )

        c = result.x[:2 * n].reshape(n, 2)
        r = result.x[2 * n:]

        valid = is_valid(c, r)
        s = np.sum(r)

        if valid and s > best_sum:
            best_c = c.copy()
            best_r = r.copy()
            best_sum = s
            best_valid = True
        elif not best_valid and s > best_sum:
            best_c = c.copy()
            best_r = r.copy()
            best_sum = s

    # Post-process: ensure boundary constraints
    for i in range(n):
        xi, yi = best_c[i]
        max_ri = min(xi, 1 - xi, yi, 1 - yi)
        best_r[i] = min(best_r[i], max_ri - 1e-10)

    # Post-process: ensure no overlap by iteratively reducing radii
    for _ in range(2000):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((best_c[i] - best_c[j]) ** 2))
                if best_r[i] + best_r[j] > dist - 1e-12:
                    excess = best_r[i] + best_r[j] - dist + 1e-12
                    best_r[i] -= excess / 2
                    best_r[j] -= excess / 2
                    changed = True
        if not changed:
            break

    # Ensure non-negative radii
    best_r = np.maximum(best_r, 0)

    # Re-verify boundary constraints after overlap reduction
    for i in range(n):
        xi, yi = best_c[i]
        max_ri = min(xi, 1 - xi, yi, 1 - yi)
        best_r[i] = min(best_r[i], max_ri - 1e-10)
    best_r = np.maximum(best_r, 0)

    return (best_c, best_r, np.sum(best_r))
