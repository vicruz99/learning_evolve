import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Returns centers, radii, and sum of radii.

    best_result = None
    best_score = -np.inf

    n_restarts = 20

        s = 0.2 
        h = s * np.sqrt(3) / 2

        # We'll try to fit them in a way that roughly covers the square
        # Offset x by s/2 for alternating rows

        # Then scale if needed, but optimizer will handle positions.
        
        row_counts = [5, 6, 5, 6, 4] # Sum = 26
        y_coords = []
        # Calculate Y positions to center them
        # 5 rows. Heights: 0, h, 2h, 3h, 4h.
        # We want them roughly in [0.1, 0.9].
        # Let's just place them starting from a low y and shift later?
        # Or let the optimizer move them. Initial validity not strictly required for SLSQP, 
        # but helps.
        
        curr_y = 0.1
        for count in row_counts:
            row_x = []
            # Center the row horizontally
            row_width = (count - 1) * s
            start_x = 0.5 - row_width / 2
            for k in range(count):
                x = start_x + k * s
                row_x.append(x)

        scale_x = 0.9 / max_x if max_x > 0 else 1

        centers_arr[:, 0] *= scale
        centers_arr[:, 1] *= scale
        
        # Add small random noise to avoid perfect symmetry which might cause issues or local minima
        centers_arr += rng.uniform(-0.01, 0.01, size=centers_arr.shape)

        centers_arr = np.clip(centers_arr, 0.01, 0.99)
