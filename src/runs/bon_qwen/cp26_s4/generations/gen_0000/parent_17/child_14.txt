import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns centers (26, 2), radii (26,), and sum_radii.
    """
    n = 26
    best_res = None
    best_sum = -np.inf
    
    # Helper to construct bounds
    def get_bounds():
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
        return bounds

    # Helper to construct constraints and Jacobian
    def get_constraints_and_jac():
        # Total constraints: 4*n (boundary) + n*(n-1)/2 (overlap)
        num_constraints = 4 * n + n * (n - 1) // 2
        
        def con_fun(x):
            vals = np.zeros(num_constraints)
            idx = 0
            # Boundary constraints
            for i in range(n):
                xi = x[3*i]
                yi = x[3*i+1]
                ri = x[3*i+2]
                vals[idx] = xi - ri             # x >= r
                vals[idx+1] = 1.0 - xi - ri     # x + r <= 1
                vals[idx+2] = yi - ri           # y >= r
                vals[idx+3] = 1.0 - yi - ri     # y + r <= 1
                idx += 4
            
            # Overlap constraints
            for i in range(n):
                xi = x[3*i]
                yi = x[3*i+1]
                ri = x[3*i+2]
                for j in range(i + 1, n):
                    xj = x[3*j]
                    yj = x[3*j+1]
                    rj = x[3*j+2]
                    
                    dx = xi - xj
                    dy = yi - yj
                    dist_sq = dx*dx + dy*dy
                    sum_r = ri + rj
                    vals[idx] = dist_sq - sum_r*sum_r
                    idx += 1
            return vals

        def con_jac(x):
            jac = np.zeros((num_constraints, 3 * n))
            idx = 0
            # Boundary constraints Jacobian
            for i in range(n):
                # d/dx_i = 1, d/dr_i = -1
                jac[idx, 3*i] = 1.0
                jac[idx, 3*i+2] = -1.0
                idx += 1
                
                # d/dx_i = -1, d/dr_i = -1
                jac[idx, 3*i] = -1.0
                jac[idx, 3*i+2] = -1.0
                idx += 1
                
                # d/dy_i = 1, d/dr_i = -1
                jac[idx, 3*i+1] = 1.0
                jac[idx, 3*i+2] = -1.0
                idx += 1
                
                # d/dy_i = -1, d/dr_i = -1
                jac[idx, 3*i+1] = -1.0
                jac[idx, 3*i+2] = -1.0
                idx += 1
            
            # Overlap constraints Jacobian
            for i in range(n):
                xi = x[3*i]
                yi = x[3*i+1]
                ri = x[3*i+2]
                for j in range(i + 1, n):
                    xj = x[3*j]
                    yj = x[3*j+1]
                    rj = x[3*j+2]
                    
                    dx = xi - xj
                    dy = yi - yj
                    sum_r = ri + rj
                    
                    # d/dx_i = 2*dx, d/dx_j = -2*dx
                    jac[idx, 3*i] = 2 * dx
                    jac[idx, 3*j] = -2 * dx
                    
                    # d/dy_i = 2*dy, d/dy_j = -2*dy
                    jac[idx, 3*i+1] = 2 * dy
                    jac[idx, 3*j+1] = -2 * dy
                    
                    # d/dr_i = -2*sum_r, d/dr_j = -2*sum_r
                    jac[idx, 3*i+2] = -2 * sum_r
                    jac[idx, 3*j+2] = -2 * sum_r
                    
                    idx += 1
            return jac

        return {
            'type': 'ineq',
            'fun': con_fun,
            'jac': con_jac
        }

    bounds = get_bounds()
    cons = get_constraints_and_jac()
    
    # Objective: minimize negative sum of radii
    def obj(x):
        return -np.sum(x[2::3])

    # Run multiple optimization restarts to avoid local minima
    # 10 restarts with different random seeds
    for seed in range(10):
        np.random.seed(seed)
        
        # Initialize centers randomly in the middle of the square
        # Radii initialized small to ensure initial feasibility
        centers = np.random.uniform(0.2, 0.8, (n, 2))
        radii = np.full(n, 0.05)
        
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        try:
            res = minimize(
                obj,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-12, 'maxiter': 500, 'disp': False}
            )
            
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_res = res
        except Exception:
            continue
            
    # Extract best result
    if best_res is not None:
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            centers[i, 0] = best_res.x[3*i]
            centers[i, 1] = best_res.x[3*i+1]
            radii[i] = best_res.x[3*i+2]
        return centers, radii, np.sum(radii)
    else:
        # Fallback (should not happen with valid logic)
        centers = np.random.rand(n, 2) * 0.5 + 0.25
        radii = np.full(n, 0.1)
        return centers, radii, 2.6
