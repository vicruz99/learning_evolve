import numpy as np
from scipy.optimize import minimize


def compute_penalty(centers, radii, n):
    """Compute penalty for constraint violations."""
    penalty = 0.0
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        penalty += max(0, r - x) ** 2
        penalty += max(0, r - y) ** 2
        penalty += max(0, x + r - 1) ** 2
        penalty += max(0, y + r - 1) ** 2
    
    # Negative radius penalty
    for r in radii:
        if r < 0:
            penalty += r ** 2
    
    return penalty


def objective(params, n):
    """Objective function: minimize negative sum of radii plus penalties."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    penalty = compute_penalty(centers, radii, n)
    
    return -np.sum(radii) + 1000.0 * penalty


def compute_forces(centers, radii, n, softness=100.0):
    """Compute repulsion forces between overlapping circles."""
    forces = np.zeros((n, 2))
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = np.sum(diff ** 2)
            dist = np.sqrt(dist_sq)
            
            if dist < 1e-12:
                dist = 1e-12
                diff = np.array([1e-12, 0.0])
            
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist:
                overlap = min_dist - dist
                force_magnitude = softness * overlap / dist
                forces[i] += diff * force_magnitude
                forces[j] -= diff * force_magnitude
    
    return forces


def init_hexagonal(n):
    """Initialize circles in a hexagonal packing pattern."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    r = 0.085
    
    # Pattern: 5, 4, 5, 4, 5, 3 = 26 circles
    row_pattern = [5, 4, 5, 4, 5, 3]
    num_rows = len(row_pattern)
    
    idx = 0
    y_spacing = r * np.sqrt(3.0)
    
    # Calculate vertical offset to center the pattern
    total_height = 2 * r + (num_rows - 1) * y_spacing
    y_offset = (1.0 - total_height) / 2.0
    
    for row_idx in range(num_rows):
        count = row_pattern[row_idx]
        y = y_offset + r + row_idx * y_spacing
        
        if row_idx % 2 == 0:
            # Even rows: full width
            if count == 1:
                x_positions = np.array([0.5])
            else:
                x_positions = np.linspace(r, 1.0 - r, count)
        else:
            # Odd rows: offset
            if count == 1:
                x_positions = np.array([0.5])
            else:
                x_positions = np.linspace(2.0 * r, 1.0 - 2.0 * r, count)
        
        for col in range(count):
            centers[idx] = np.array([x_positions[col], y])
            radii[idx] = r
            idx += 1
    
    return centers, radii


def repulsion_optimize(centers, radii, n, iterations=2000, dt=0.005):
    """Optimize positions using repulsion forces."""
    centers = centers.copy()
    radii = radii.copy()
    
    for it in range(iterations):
        forces = compute_forces(centers, radii, n)
        
        # Adaptive step size
        max_force = np.max(np.linalg.norm(forces, axis=1))
        if max_force < 1e-10:
            break
        
        step = min(dt, 0.1 / (1.0 + max_force))
        centers += step * forces
        
        # Project to valid region
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    return centers


def grow_and_repulse(centers, radii, n, target_factor=1.2, steps=500):
    """Gradually grow circles while maintaining packing."""
    centers = centers.copy()
    radii = radii.copy()
    
    for step in range(steps):
        # Grow radii
        growth = 1.0 + (target_factor - 1.0) / steps
        radii *= growth
        
        # Run repulsion optimization
        centers = repulsion_optimize(centers, radii, n, iterations=50, dt=0.01)
    
    return centers, radii


def local_optimize(centers, radii, n, max_iter=20000):
    """Local optimization using L-BFGS-B."""
    params = np.concatenate([centers.flatten(), radii])
    
    # Bounds: centers in [0, 1], radii >= 0
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, None)] * n
    
    result = minimize(
        objective,
        params,
        method='L-BFGS-B',
        bounds=bounds,
        options={
            'maxiter': max_iter,
            'ftol': 1e-14,
            'gtol': 1e-10,
            'iprint': -1
        }
    )
    
    opt_centers = result.x[:2 * n].reshape(n, 2)
    opt_radii = result.x[2 * n:]
    
    # Ensure non-negative radii
    opt_radii = np.maximum(opt_radii, 1e-10)
    
    # Final projection
    for i in range(n):
        opt_centers[i, 0] = np.clip(opt_centers[i, 0], opt_radii[i], 1.0 - opt_radii[i])
        opt_centers[i, 1] = np.clip(opt_centers[i, 1], opt_radii[i], 1.0 - opt_radii[i])
    
    return opt_centers, opt_radii


def run_packing():
