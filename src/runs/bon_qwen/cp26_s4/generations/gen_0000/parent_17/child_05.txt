import numpy as np
import scipy.optimize as opt
import scipy.spatial.distance as dist

def solve_radii(centers):
    """
    Solves for the optimal radii given fixed centers to maximize sum of radii.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        radii: np.array of shape (n)
        sum_radii: float
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = np.ones(n) * -1.0
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    # 2. Distance constraints: r_i + r_j <= dist(i, j)
    
    A_ub = []
    b_ub = []
    
    # Precompute distances
    # Using cdist for efficiency
    dists = dist.cdist(centers, centers, metric='euclidean')
    
    # Boundary constraints
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    # We can combine these or add separately. 
    # A_ub will be sparse, but for n=26 dense is fine.
    
    # 4 constraints per circle
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        # Constraint: r_i <= max_r  =>  1*r_i <= max_r
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max_r)
        
    # Distance constraints
    # r_i + r_j <= dist_ij
    # Iterate over unique pairs i < j
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: 0 <= r_i <= 1 (loose upper bound)
    bounds = [(0, 1)] * n
    
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            return radii, -res.fun
        else:
            # Fallback if LP fails (should not happen for valid centers)
            # Return small radii
            radii = np.zeros(n)
            return radii, 0.0
    except Exception:
        return np.zeros(n), 0.0

    centers = np.clip(centers, 1e-9, 1 - 1e-9)
    
    _, sum_radii = solve_radii(centers)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
