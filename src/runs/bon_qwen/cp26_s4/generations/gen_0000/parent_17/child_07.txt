import numpy as np
import scipy.optimize as opt

def calculate_overlap_loss(centers, radii):
    """
    Calculates a penalty value for overlaps and boundary violations.
    Positive value indicates violation.
    """
    n = len(radii)
    loss = 0.0
    
    # Boundary violations
    # x - r >= 0  => r - x <= 0
    # 1 - x - r >= 0 => x + r - 1 <= 0
    # We sum the squares of positive violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left
        if x - r < 0:
            loss += (x - r)**2
        # Right
        if x + r > 1:
            loss += (x + r - 1)**2
        # Bottom
        if y - r < 0:
            loss += (y - r)**2
        # Top
        if y + r > 1:
            loss += (y + r - 1)**2
            
    # Circle-circle overlaps
    # dist >= r_i + r_j
    # dist - (r_i + r_j) >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                loss += (dist - min_dist)**2
                
    return loss
