import numpy as np
from scipy.optimize import minimize

def obj_func(x):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum of radii)"""
    return -np.sum(x[2::3])

def boundary_left(x, i):
    """Constraint: cx - r >= 0"""
    return x[3*i] - x[3*i+2]

def boundary_right(x, i):
    """Constraint: 1 - cx - r >= 0"""
    return 1.0 - x[3*i] - x[3*i+2]

def boundary_bottom(x, i):
    """Constraint: cy - r >= 0"""
    return x[3*i+1] - x[3*i+2]

def boundary_top(x, i):
    """Constraint: 1 - cy - r >= 0"""
    return 1.0 - x[3*i+1] - x[3*i+2]

def pair_dist(x, pair):
    """Constraint: ||ci - cj|| - (ri + rj) >= 0"""
    i, j = pair
    dx = x[3*i] - x[3*j]
    dy = x[3*i+1] - x[3*j+1]
    return np.hypot(dx, dy) - (x[3*i+2] + x[3*j+2])

def run_packing():
    n = 26
    
    # Initialize centers in a dense hexagonal-like arrangement
    rows = [5, 6, 6, 5, 4]
    centers = []
    y_spacing = 0.18
    y_base = 0.12
    
    for k, count in enumerate(rows):
        y = y_base + k * y_spacing
        x_spacing = 1.0 / (count + 1)
        for m in range(count):
            x = (m + 1) * x_spacing
            # Offset odd rows for hexagonal packing
            if k % 2 == 1:
                x += x_spacing / 2.0
            centers.append([x, y])
            
    centers = np.array(centers[:n])
    radii = np.ones(n) * 0.03  # Small initial radii to ensure feasibility
    
    # Flatten variables: [c1x, c1y, r1, c2x, c2y, r2, ..., c26x, c26y, r26]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds for variables
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
    
    # Construct constraints
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': boundary_left, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': boundary_right, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': boundary_bottom, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': boundary_top, 'args': (i,)})
        
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': pair_dist, 'args': ((i, j),)})
            
    # Run optimization
    res = minimize(
        obj_func, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
    )
    
    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = res.x[2*n:]
    
    return final_centers, final_radii, np.sum(final_radii)
