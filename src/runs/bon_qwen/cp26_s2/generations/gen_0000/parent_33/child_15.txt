def generate_initial_centers(n):

            x += 0.2
        y += 0.1 * np.sqrt(3)
        row_idx += 1
        
    return np.array(centers[:n])

def validate_and_adjust(centers, radii):

        1.0 - centers[:, 0],
        centers[:, 1],
        1.0 - centers[:, 1]
    ], axis=1), axis=1)
    radii = np.minimum(radii, r_min - 1e-9)
    radii = np.maximum(radii, 0)
    return radii

    n = centers.shape[0]

            dist = np.linalg.norm(centers[i] - centers[j])

            if overlap > 0:
                penalty += overlap * overlap
    return penalty
