import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- Helper Function 1: Solve Radii via LP ---
    def solve_radii_lp(centers):
        """
        Solves the Linear Program to find radii that maximize sum(r_i)
        subject to non-overlap and boundary constraints for fixed centers.
        """
        # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
        c_obj = np.ones(n) * -1.0
        
        # Inequality Constraints: A_ub @ r <= b_ub
        # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
        # 2. Pairwise constraints: r_i + r_j <= dist(i, j)
        
        A_ub = []
        b_ub = []
        
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            # r_i <= x_i
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(x)
            
            # r_i <= 1 - x_i
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(1.0 - x)
            
            # r_i <= y_i
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(y)
            
            # r_i <= 1 - y_i
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(1.0 - y)
            
        # Pairwise distance constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < 1e-9: 
                    # If centers are identical, radius must be 0
                    row = np.zeros(n)
                    row[i] = 1.0
                    row[j] = 1.0
                    A_ub.append(row)
                    b_ub.append(0.0)
                    continue
                    
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Bounds for r_i: [0, infinity)
        bounds = [(0, None)] * n
        
        try:
            res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                radii = res.x
                return radii, -res.fun
            else:
                # Fallback if LP fails
                return np.full(n, 1e-5), 26e-5
        except Exception:
            return np.full(n, 1e-5), 26e-5

    # --- Initialization: Perturbed Hexagonal Grid ---
    # We use a staggered grid which has higher density than square grids.
    centers = np.zeros((n, 2))
    idx = 0
    
    # Approximate parameters for hexagonal packing of 26 circles
    # 5 rows seems viable. 4-5-4-5-4-4 distribution? 
    # Let's try a dense packing initialization
    # We want to fill the square. 
    
    # Initial layout: 5 rows
    # Row counts: 5, 5, 5, 5, 6 (Total 26) - wait, 6 is hard for width.
    # Let's try 5, 5, 5, 5, 4, 2? 
    # Or simply random within a valid range? 
    # A perturbed 5x5 grid + 1 is safer.
    
    # Base 5x5 grid
    base_coords = []
    for i in range(5):
        for j in range(5):
            base_coords.append([0.1 + i*0.2, 0.1 + j*0.2])
            
    # Add one point in the middle of a gap (0.2, 0.2)
    base_coords.append([0.2, 0.2])
    
    # Apply small random jitter to break symmetry and help optimization
    centers = np.array(base_coords)
    centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
    
    # Project centers back into valid range [0.01, 0.99] to ensure LP feasibility
    centers = np.clip(centers, 0.01, 0.99)

    # --- Optimization Loop: Hill Climbing ---
    best_centers = centers.copy()
    _, best_sum = solve_radii_lp(best_centers)
    
    # Initial step size
    step_size = 0.05
    patience = 0
    
    # We iterate to refine positions
    for iteration in range(500):
        improved = False
        
        # Try to perturb each circle
        for i in range(n):
            # Save current best
            temp_centers = best_centers.copy()
            
            # Generate random perturbation
            dx = np.random.uniform(-step_size, step_size)
            dy = np.random.uniform(-step_size, step_size)
            
            # Apply perturbation
            temp_centers[i, 0] = np.clip(temp_centers[i, 0] + dx, 0.01, 0.99)
            temp_centers[i, 1] = np.clip(temp_centers[i, 1] + dy, 0.01, 0.99)
            
            # Evaluate
            _, current_sum = solve_radii_lp(temp_centers)
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = temp_centers.copy()
                improved = True
                
        if not improved:
            patience += 1
            step_size *= 0.9 # Reduce step size if no improvement
            if patience > 20:
                break
        else:
            patience = 0
            
    # --- Final Validation & Correction ---
    final_radii, final_sum = solve_radii_lp(best_centers)
    
    # Ensure radii are non-negative
    final_radii = np.maximum(final_radii, 0.0)
    
    # Final check for validity (optional but good for safety)
    # The LP guarantees constraints, but numerical errors might occur.
    # We trust the LP solution.
    
    return best_centers, final_radii, final_sum
