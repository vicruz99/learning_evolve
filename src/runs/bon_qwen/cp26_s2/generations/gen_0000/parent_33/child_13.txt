import numpy as np
from scipy.optimize import minimize
import warnings

# Suppress optimization warnings for cleaner output
warnings.filterwarnings("ignore")

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n_circles = 26
    best_result = None
    best_sum = 0.0

    # Multiple restarts to find a good local optimum
    num_restarts = 5
    
    for _ in range(num_restarts):
        # 1. Initialize with a hexagonal grid + small random perturbation
        centers = np.zeros((n_circles, 2))
        
        # Hexagonal packing parameters
        # Try to fit 26 circles in a roughly 6x5 or similar arrangement
        # A 5x5 grid is 25. We need 1 more.
        # Let's use a grid density estimate.
        # If r ~ 0.1, spacing ~ 0.2. 
        # 1/0.2 = 5. So 5x5 grid.
        
        # Generate a grid
        rows = 6
        cols = 5
        # Approximate spacing
        spacing_x = 1.0 / cols
        spacing_y = 1.0 / rows
        
        idx = 0
        for r in range(rows):
            for c in range(cols):
                if idx >= n_circles:
                    break
                # Hexagonal shift for alternating rows
                shift = spacing_x / 2.0 if r % 2 == 1 else 0.0
                x = c * spacing_x + spacing_x / 2.0 + shift
                y = r * spacing_y + spacing_y / 2.0
                
                # Clamp to [0, 1] roughly, though grid logic keeps it inside
                x = np.clip(x, 0.05, 0.95)
                y = np.clip(y, 0.05, 0.95)
                
                # Add small random noise to break symmetry and help optimization
                x += np.random.normal(0, 0.01)
                y += np.random.normal(0, 0.01)
                
                centers[idx] = [x, y]
                idx += 1
            if idx >= n_circles:
                break
        
        # Initial radii (small enough to be valid)
        initial_radii = np.full(n_circles, 0.05)
        
        # Concatenate variables: [x1, y1, ..., x26, y26, r1, ..., r26]
        x0 = np.concatenate([centers.flatten(), initial_radii])
        
        # 2. Define Objective Function
        def objective(vars):
            r = vars[2 * n_circles:]
            return -np.sum(r)

        # 3. Define Constraints
        constraints = []

        # Boundary constraints: 0 <= x - r, x + r <= 1, same for y
        # x_i >= r_i  =>  x_i - r_i >= 0
        # x_i <= 1 - r_i => 1 - x_i - r_i >= 0

            xi_idx = 2 * i
            yi_idx = 2 * i + 1
            ri_idx = 2 * n_circles + i
            
            # x >= r
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[xi_idx] - v[ri_idx]
            })
            # x <= 1 - r  =>  1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[xi_idx] - v[ri_idx]
            })
            # y >= r
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[yi_idx] - v[ri_idx]
            })
            # y <= 1 - r
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[yi_idx] - v[ri_idx]
            })
            
        # Non-overlap constraints: dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
        
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                xi_idx = 2 * i
                yi_idx = 2 * i + 1
                ri_idx = 2 * n_circles + i
                
                xj_idx = 2 * j
                yj_idx = 2 * j + 1
                rj_idx = 2 * n_circles + j
                
                def non_overlap(v, i=i, j=j):
                    xi, yi = v[xi_idx], v[yi_idx]
                    xj, yj = v[xj_idx], v[yj_idx]
                    ri, rj = v[ri_idx], v[rj_idx]
                    
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r_sq = (ri + rj)**2
                    return dist_sq - sum_r_sq
                
                constraints.append({
                    'type': 'ineq',
                    'fun': non_overlap
                })

        # 4. Bounds

        # r >= 0
        bounds = []
        for _ in range(2 * n_circles):
            bounds.append((0.0, 1.0))
        for _ in range(n_circles):
            bounds.append((0.0, 1.0)) # Max radius 0.5 effectively, but 1.0 is safe bound

        # 5. Run Optimization
        try:
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9}
            )

                        valid = False

                            break

            continue

    if best_result is None:
