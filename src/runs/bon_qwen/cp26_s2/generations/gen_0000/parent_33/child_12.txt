import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a combination of heuristic initialization (hexagonal packing) and 
    numerical optimization (SLSQP).
    """
    n_circles = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Helper to create a hexagonal packing initialization
    def get_hex_initialization(rows_pattern):
        """
        rows_pattern: list of integers, number of circles in each row.
        Returns centers and initial radii.
        """
        centers = []
        # Estimate initial radius to fit the pattern roughly
        # Width constraint: max(width) * 2r <= 1 => r <= 1 / (2 * max_width)
        # Height constraint: (rows-1)*sqrt(3)r + 2r <= 1 => r <= 1 / (2 + (rows-1)sqrt(3))
        
        # Let's pick a conservative r to start
        r_start = 0.08 
        
        current_y = r_start
        for r_idx, n_in_row in enumerate(rows_pattern):
            # Shift x for odd rows to create hexagonal pattern
            shift = 0.0
            if r_idx % 2 == 1:
                shift = r_start * np.sqrt(3) * 0.5 # Actually shift is r in x if vertical spacing is sqrt(3)r?
                # Standard hex: vertical dist sqrt(3)r, horizontal shift r.
                # Wait, if vertical spacing is sqrt(3)r, then x-shift is r.
                # Let's stick to simple shift r.
                shift = r_start
            
            row_width = (n_in_row - 1) * 2 * r_start
            # Center the row
            start_x = (1.0 - row_width) / 2.0 - shift
            
            for i in range(n_in_row):
                x = start_x + i * 2 * r_start
                y = current_y
                centers.append([x, y])
            
            current_y += r_start * np.sqrt(3)
            
        return np.array(centers), np.full(n_circles, r_start)

    # Possible patterns for 26 circles
    # Sum must be 26
    patterns = [
        [5, 4, 5, 4, 5, 3], # 6 rows
        [5, 5, 5, 5, 4, 2], # 6 rows
        [4, 5, 4, 5, 4, 4], # 6 rows
        [5, 6, 5, 6, 4],    # 5 rows
        [5, 5, 6, 5, 5],    # 5 rows
        [4, 4, 4, 4, 4, 4, 4, 0] # fallback
    ]
    
    # Filter valid patterns summing to 26
    valid_patterns = [p for p in patterns if sum(p) == 26 and all(c > 0 for c in p)]
    # Ensure we have at least one
    if not valid_patterns:
        valid_patterns = [[5, 4, 5, 4, 5, 3]]

    # Optimization function
    def objective(vars):
        # vars: [x0, y0, r0, x1, y1, r1, ...]
        radii = vars[2::3]
        return -np.sum(radii) # Minimize negative sum

    def constraint_overlap(vars):
        # dist >= r_i + r_j  =>  dist^2 >= (r_i + r_j)^2
        # We want dist^2 - (r_i + r_j)^2 >= 0
        # Return array of values >= 0
        n = len(vars) // 3
        dists = []
        for i in range(n):
            xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
            for j in range(i+1, n):
                xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                min_dist = ri + rj
                dists.append(dist_sq - min_dist**2)
        return np.array(dists)

    def constraint_boundary(vars):
        # x - r >= 0, x + r <= 1 => r <= x <= 1-r
        # Similarly for y
        # Constraints: x - r >= 0, 1 - (x + r) >= 0 => 1 - x - r >= 0
        con = []
        for i in range(n_circles):
            xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
            con.append(xi - ri)           # x >= r
            con.append(1.0 - xi - ri)     # x + r <= 1
            con.append(yi - ri)           # y >= r
            con.append(1.0 - yi - ri)     # y + r <= 1
        return np.array(con)

    def constraint_radius_positive(vars):
        return vars[2::3] # r >= 0

    n_vars = n_circles * 3
    bounds = [(0.0, 1.0)] * n_vars # Loose bounds for x, y; r handled by constraints
    # Actually r can be up to 0.5, but bounded by box.
    # Let's tighten bounds for x, y to [0, 1]
    # r bounds [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_radius_positive}
    ]

    best_sum = 0.0
    
    # Try multiple starts
    np.random.seed(42)
    
    for p_idx, pattern in enumerate(valid_patterns):
        for trial in range(5): # 5 random seeds per pattern
            try:
                # Generate initial guess
                # Perturb the hex pattern
                centers, radii = get_hex_initialization(pattern)
                
                # Add noise
                centers += np.random.normal(0, 0.01, centers.shape)
                radii += np.random.normal(0, 0.005, radii.shape)
                
                # Clip to valid range for initialization
                radii = np.clip(radii, 0.01, 0.4)
                centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
                centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

                x0 = np.zeros(n_vars)
                for i in range(n_circles):
                    x0[3*i] = centers[i, 0]
                    x0[3*i+1] = centers[i, 1]
                    x0[3*i+2] = radii[i]

                # Optimization
                res = opt.minimize(
                    objective, 
                    x0, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 1000, 'ftol': 1e-9}
                )
                
                if res.success or res.fun < -best_sum: # Better objective
                    # Extract solution
                    c_sol = res.x[0::3]
                    r_sol = res.x[2::3]
                    sum_r = np.sum(r_sol)
                    
                    # Validate quickly
                    valid = True
                    for i in range(n_circles):
                        if r_sol[i] < -1e-6 or c_sol[i,0] < -1e-6 or c_sol[i,0] > 1+1e-6:
                            valid = False
                            break
                    # Check overlaps
                    for i in range(n_circles):
                        for j in range(i+1, n_circles):
                            d = np.sqrt((c_sol[i,0]-c_sol[j,0])**2 + (c_sol[i,1]-c_sol[j,1])**2)
                            if d < r_sol[i] + r_sol[j] - 1e-7:
                                valid = False
                                break
                        if not valid: break
                    
                    if valid and sum_r > best_sum:
                        best_sum = sum_r
                        best_centers = c_sol
                        best_radii = r_sol
                        # print(f"Pattern {p_idx}, Trial {trial}: New best sum {best_sum}")

            except Exception as e:
                pass

    # Fallback if optimization failed or sum is low
    # Try a simple grid-based expansion
    if best_sum < 2.5:
        # Try 5x5 grid + 1
        c = np.zeros((26, 2))
        r = np.full(26, 0.0)
        
        # 5x5 grid
        idx = 0
        for i in range(5):
            for j in range(5):
                c[idx, 0] = 0.2 * i + 0.1
                c[idx, 1] = 0.2 * j + 0.1
                r[idx] = 0.1
                idx += 1
        
        # 26th circle in corner gap
        # Gap at (0.2, 0.2) roughly?
        # Center of gap between (0.1,0.1), (0.3,0.1), (0.1,0.3), (0.3,0.3) is (0.2, 0.2)
        # Dist to (0.1, 0.1) is sqrt(0.02) ~ 0.1414
        # Max radius ~ 0.1414 - 0.1 = 0.0414
        c[25, 0] = 0.2
        c[25, 1] = 0.2
        r[25] = 0.041
        
        # Optimize this specific config
        x0 = np.zeros(n_vars)
        for i in range(n_circles):
            x0[3*i] = c[i, 0]
            x0[3*i+1] = c[i, 1]
            x0[3*i+2] = r[i]
        
        try:
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000})
            c_sol = res.x[0::3]
            r_sol = res.x[2::3]
            sum_r = np.sum(r_sol)
            
            # Validate
            valid = True
            for i in range(n_circles):
                for j in range(i+1, n_circles):
                    d = np.sqrt((c_sol[i,0]-c_sol[j,0])**2 + (c_sol[i,1]-c_sol[j,1])**2)
                    if d < r_sol[i] + r_sol[j] - 1e-7:
                        valid = False
            if valid and sum_r > best_sum:
                best_sum = sum_r
                best_centers = c_sol
                best_radii = r_sol
        except:
            pass

    # Final fallback: simple equal circles hex packing
    if best_sum < 2.5:
        # Just return a valid packing, e.g. 5x5 grid radius 0.1
        c = np.zeros((26, 2))
        r = np.full(26, 0.0)
        idx = 0
        for i in range(5):
            for j in range(5):
                c[idx, 0] = 0.2 * i + 0.1
                c[idx, 1] = 0.2 * j + 0.1
                r[idx] = 0.1
                idx += 1
        # Place 26th tiny circle
        c[25, :] = [0.5, 0.5] # Center
        r[25] = 1e-6
        best_centers = c
        best_radii = r
        best_sum = np.sum(r)

    return best_centers, best_radii, best_sum
