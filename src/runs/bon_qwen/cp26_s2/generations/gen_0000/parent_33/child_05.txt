        pass

    for i in range(n):
        idx = 3 * i
        # x - r >= 0
        def c1(v, i=i): return v[idx] - v[idx + 2]
        cons.append({'type': 'ineq', 'fun': c1})
        
        # 1 - x - r >= 0
        def c2(v, i=i): return 1 - v[idx] - v[idx + 2]

        # y - r >= 0

        def c4(v, i=i): return 1 - v[idx + 1] - v[idx + 2]
