import numpy as np
from scipy.optimize import minimize

N = 26

def compute_objective(vars):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2*N:])

def compute_constraints(vars):
    """Returns array of constraint values that must be >= 0."""
    x = vars[:N]
    y = vars[N:2*N]
    r = vars[2*N:]
    c = []
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N):
        for j in range(i + 1, N):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            c.append(dx*dx + dy*dy - (r[i] + r[j])**2)
            
    return np.array(c)

def run_packing():
    np.random.seed(42)
    
    # Initial guess: Hexagonal-like staggered packing
    centers = []
    y = 0.12
    count = 0
    for row in range(5):
        x = 0.12
        n_in_row = 6 if row % 2 == 0 else 5
        if row == 4 and count + n_in_row > N:
            n_in_row = N - count
        for col in range(n_in_row):
            cx = x + (0.12 if row % 2 == 1 else 0.0)
            centers.append([cx, y])
            x += 0.24
            count += 1
        y += 0.21
        
    centers = np.array(centers)
    r_init = np.full(N, 0.08)
    
    x0 = np.concatenate([centers[:, 0], centers[:, 1], r_init])
    # Small perturbation to break symmetry and aid convergence
    x0 += np.random.randn(*x0.shape) * 0.005
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Main optimization
    res = minimize(
        fun=compute_objective,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 15000, 'ftol': 1e-11}
    )
    
    best_centers = np.column_stack((res.x[:N], res.x[N:2*N]))
    best_radii = res.x[2*N:]
    best_sum = np.sum(best_radii)
    
    # Iterative expansion phase to push radii to physical limits
    for _ in range(15):
        best_radii *= 1.008
        cur_x0 = np.concatenate([best_centers[:, 0], best_centers[:, 1], best_radii])
        
        res2 = minimize(
            fun=compute_objective,
            x0=cur_x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 3000, 'ftol': 1e-11}
        )
        
        new_radii = res2.x[2*N:]
        new_sum = np.sum(new_radii)
        
        if res2.success and new_sum > best_sum:
            best_centers = np.column_stack((res2.x[:N], res2.x[N:2*N]))
            best_radii = new_radii
            best_sum = new_sum
        else:
            break
            
    # Safety clamp
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, best_sum
