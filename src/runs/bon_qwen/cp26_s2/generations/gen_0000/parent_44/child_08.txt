import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, lam):
    """
    Computes negative sum of radii plus penalty for overlaps.
    vars: flattened array [x1, y1, r1, x2, y2, r2, ...]
    n: number of circles
    lam: penalty weight
    """
    C = vars.reshape(n, 3)
    # Compute pairwise differences using broadcasting
    dx = C[:, 0, None] - C[None, 0, :]
    dy = C[:, 1, None] - C[None, 1, :]
    dr = C[:, 2, None] + C[None, 2, :]
    
    # Euclidean distance matrix
    dist = np.sqrt(np.maximum(dx*dx + dy*dy, 0))
    
    # Overlap amount (positive if circles intersect)
    overlap = dr - dist
    
    # Penalty: sum of squared overlaps for all unique pairs
    # Factor 0.5 accounts for symmetric matrix and self-pairs (which are zero)
    pen = 0.5 * np.sum(np.maximum(overlap, 0)**2)
    
    # Objective: maximize sum of radii -> minimize negative sum + penalty
    return -np.sum(C[:, 2]) + lam * pen

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    # 1. Initialize with a perturbed hexagonal grid
    centers = []
    r0 = 0.08
    y = r0
    row = 0
    while len(centers) < n and y + r0 < 1.0:
        x = r0 + (row % 2) * r0
        while x + r0 < 1.0 and len(centers) < n:
            cx = x + np.random.uniform(-0.002, 0.002)
            cy = y + np.random.uniform(-0.002, 0.002)
            # Clamp to safe interior initially
            centers.append([max(0.01, min(0.99, cx)), max(0.01, min(0.99, cy))])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
        
    centers = np.array(centers[:n])
    radii = np.full(n, r0)
    
    # Flatten to optimization variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds: x,y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    
    cur_x = x0.copy()
    
    # 2. Homotopy Penalty Optimization
    # Gradually increase penalty weight to enforce non-overlap strictly
    for k in range(35):
        lam = 10.0 * (2.0 ** k)
        res = minimize(
            compute_objective, 
            cur_x, 
            args=(n, lam), 
            method='L-BFGS-B', 
            bounds=bounds, 
            options={'maxiter': 300, 'ftol': 1e-14}
        )
        cur_x = res.x
        
    # 3. Extract and return results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    for i in range(n):
        final_centers[i] = [cur_x[3*i], cur_x[3*i+1]]
        final_radii[i] = cur_x[3*i+2]
        
    return final_centers, final_radii, float(np.sum(final_radii))
