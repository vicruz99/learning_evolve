import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    to maximize the sum of radii.

    def get_hex_grid_points(n_points):
        points = []

        # Area per point approx 1/n. Hex area factor sqrt(3)/2 * s^2
        # s^2 approx 2 / (n * sqrt(3))
        s = np.sqrt(2.0 / (n_points * np.sqrt(3)))

        col = 0
        while len(points) < n_points:
            # y coordinate with hexagonal offset
            y = row * s * np.sqrt(3) / 2.0
            # x coordinate with shift for odd rows
            x_offset = (s / 2.0) if row % 2 == 1 else 0
            
            # Generate points in this row
            # We add points until we might exceed bounds, then check
            # Simple strategy: fill a grid and trim, but here we just generate enough
            # To be safe, let's just generate a grid of sufficient size
            # and then we can shift/scale or just let optimizer move them.
            # A dense generation:
            for c in range(int(n_points/5) + 5): 
                x = c * s + x_offset
                if x <= 1.0 and y <= 1.0:
                     points.append([x, y])
                    if len(points) >= n_points:
                        break
            row += 1
            if row > 20: break # safety
        
        return np.array(points[:n_points])

        # y_i - r_i >= 0

            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3*i] - x[3*i+2]}) # 1 - x - r >= 0
