import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """Minimize negative sum of radii"""
    return -np.sum(vars[2::3])

def constr_boundary(vars, n):
    """Ensure circles stay within [0,1]x[0,1]"""
    X = vars[0::3]
    Y = vars[1::3]
    R = vars[2::3]
    return np.concatenate([
        X - R,
        1.0 - (X + R),
        Y - R,
        1.0 - (Y + R)
    ])

def constr_overlap(vars, n):
    """Ensure pairwise non-overlap: dist^2 >= (r_i + r_j)^2"""
    X = vars[0::3]
    Y = vars[1::3]
    R = vars[2::3]
    
    dx = X[:, None] - X[None, :]
    dy = Y[:, None] - Y[None, :]
    dr = R[:, None] + R[None, :]
    
    dist_sq = dx**2 + dy**2
    # Select upper triangle (i < j) to avoid duplicates and self-comparisons
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return dist_sq[mask] - dr[mask]**2

def run_packing():
    n = 26
    r_init = 0.06
    np.random.seed(42)  # Reproducibility
    
    # 1. Generate hexagonal lattice initialization
    centers_init = []
    row = 0
    while len(centers_init) < n:
        col = 0
        while len(centers_init) < n:
            x = col * 2 * r_init + (row % 2) * r_init + r_init
            y = row * np.sqrt(3) * r_init + r_init
            if x <= 1.0 and y <= 1.0:
                centers_init.append([x, y])
            col += 1
        row += 1
        
    centers_init = np.array(centers_init[:n])
    
    best_sum = -np.inf
    best_vars = None
    
    # 2. Optimizer setup
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = [
        {'type': 'ineq', 'fun': constr_boundary, 'args': (n,)},
        {'type': 'ineq', 'fun': constr_overlap, 'args': (n,)}
    ]
    
    # 3. Multiple trials with perturbations to avoid local minima
    for trial in range(3):
        perturbation = np.random.uniform(-0.02, 0.02, size=centers_init.shape)
        current_centers = np.clip(centers_init + perturbation, 0.05, 0.95)
        
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = current_centers[i, 0]
            x0[3*i+1] = current_centers[i, 1]
            x0[3*i+2] = r_init
            
        res = minimize(objective, x0, method='SLSQP', args=(n,), 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
        
        if res.fun < best_sum:
            best_sum = res.fun
            best_vars = res.x.copy()
            
    # 4. Extract and sanitize results
    final_vars = best_vars
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = final_vars[3*i]
        centers[i, 1] = final_vars[3*i+1]
        radii[i] = final_vars[3*i+2]
        
    # Enforce strict validity within numerical tolerance
    radii = np.maximum(radii, 0.0)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    sum_r = np.sum(radii)
    return centers, radii, sum_r
