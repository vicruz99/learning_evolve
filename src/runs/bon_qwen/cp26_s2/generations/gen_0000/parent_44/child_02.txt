import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_constraints(centers, radii):
    """Calculate constraint violations (negative means valid)"""
    n = len(radii)
    constraints = []
    
    # Boundary constraints: 4 per circle (x-r >= 0, r-x+1 >= 0, etc.)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r)
        constraints.append(1 - (x + r))
        constraints.append(y - r)
        constraints.append(1 - (y + r))

    # Overlap constraints: (x_i-x_j)^2 + (y_i-y_j)^2 >= (r_i+r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j])**2)
            rad_sum = radii[i] + radii[j]
            constraints.append(dist_sq - rad_sum**2)
            
    return np.array(constraints)

def generate_hexagonal_init(n):
    centers = []
    r = 0.1 # Initial small radius
    row = 0
    y = r
    while len(centers) < n:
        if row % 2 == 0:
            x_start = r
        else:
            x_start = r + r # Offset
            
        x = x_start
        while x <= 1 - r and len(centers) < n:
            centers.append([x, y])
            x += 2 * r
            if x + r > 1:
                break
        row += 1
        y += r * np.sqrt(3)
        if y + r > 1:
            break
    return np.array(centers[:n])

def objective(vars, n):
    # vars contains [x1, y1, r1, x2, y2, r2, ...]
    # Objective: minimize negative sum of radii
    radii = vars[2::3]
    return -np.sum(radii)

def constraints_func(vars, n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i] = [vars[3*i], vars[3*i+1]]
        radii[i] = vars[3*i+2]
        
    c = []
    # Boundary
    for i in range(n):
        c.append(vars[3*i] - vars[3*i+2])
        c.append(1 - (vars[3*i] + vars[3*i+2]))
        c.append(vars[3*i+1] - vars[3*i+2])
        c.append(1 - (vars[3*i+1] + vars[3*i+2]))
        
    # Overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = (vars[3*i] - vars[3*j])**2 + (vars[3*i+1] - vars[3*j+1])**2
            rad_sum = vars[3*i+2] + vars[3*j+2]
            c.append(dist_sq - rad_sum**2)
            
    return np.array(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization
    init_centers = generate_hexagonal_init(n)
    init_radii = np.full(n, 0.09) # Start slightly below 0.1
    
    # Ensure initial validity
    while not validate_packing(init_centers, init_radii):
        init_radii *= 0.9
        
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = init_centers[i, 0]
        x0[3*i+1] = init_centers[i, 1]
        x0[3*i+2] = init_radii[i]
        
    # 2. Optimization
    bounds = [(0, 1) if i % 3 != 2 else (1e-6, 0.5) for i in range(3 * n)]
    
    cons = {'type': 'ineq', 'fun': lambda x: constraints_func(x, n)}
    
    try:
        res = minimize(objective, x0, args=(n,), method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'ftol': 1e-8, 'maxiter': 1000})
        x_opt = res.x
    except:
        x_opt = x0

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [x_opt[3*i], x_opt[3*i+1]]
        radii[i] = x_opt[3*i+2]

    # 3. Refinement (Incremental Growth)
    # Try to squeeze more radius by iterative expansion
    for _ in range(20):
        radii *= 1.005
        # Re-optimize centers to fit the new radii
        x_start = np.zeros(3 * n)
        for i in range(n):
            x_start[3*i] = centers[i, 0]
            x_start[3*i+1] = centers[i, 1]
            x_start[3*i+2] = radii[i]
            
        try:
            res_local = minimize(lambda x: 0, x_start, args=(), method='SLSQP', bounds=bounds, 
                                 constraints={'type': 'ineq', 'fun': lambda x: constraints_func(x, n)},
                                 options={'ftol': 1e-10, 'maxiter': 500})
            if res_local.success:
                # Extract optimized centers, keep radii
                new_centers = np.zeros((n, 2))
                for i in range(n):
                    new_centers[i] = [res_local.x[3*i], res_local.x[3*i+1]]
                centers = new_centers
        except:
            pass
            
    # Extract final results
    centers_final = np.zeros((n, 2))
    radii_final = np.zeros(n)
    for i in range(n):
        centers_final[i] = centers[i]
        radii_final[i] = radii[i]
        
    # Ensure validity one last time
    if not validate_packing(centers_final, radii_final):
        # Fallback to equal radii if optimization failed
        r_eq = 0.095
        grid_x = np.linspace(r_eq, 1-r_eq, 6)[:5] # 5 points
        grid_y = np.linspace(r_eq, 1-r_eq, 6)[:5]
        centers_final = []
        radii_final = []
        for y in grid_y:
            for x in grid_x:
                if len(centers_final) < n:
                    centers_final.append([x, y])
                    radii_final.append(r_eq)
        centers_final = np.array(centers_final)
        radii_final = np.array(radii_final)

    sum_radii = np.sum(radii_final)
    return centers_final, radii_final, float(sum_radii)
