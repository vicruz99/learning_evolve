import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    """
    Computes a differentiable penalty for boundary violations and overlaps.
    centers: (n, 2) array
    radii: (n,) array
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary penalties: r <= x <= 1-r and r <= y <= 1-r
    # Equivalent to max(0, r - x)^2 + max(0, x - (1-r))^2 + ...
    penalty += np.sum(np.maximum(0, radii - centers[:, 0])**2)
    penalty += np.sum(np.maximum(0, radii - (1 - centers[:, 0]))**2)
    penalty += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    penalty += np.sum(np.maximum(0, radii - (1 - centers[:, 1]))**2)
    
    # Overlap penalties: d_ij >= r_i + r_j
    # Vectorized pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Only consider upper triangle (i < j) to avoid double counting and self-distance
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = np.maximum(0, radii[:, None] + radii[None, :] - dists)
    penalty += np.sum(overlaps[mask]**2)
    
    return penalty

def objective(z, n, lam):
    """
    Objective function to minimize: -sum(radii) + lambda * penalty
    """
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return -np.sum(radii) + lam * compute_penalty(centers, radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Penalty weight for constraint enforcement
    lam = 5000.0
    
    # Run multiple optimizations with different seeds to escape local minima
    for seed in range(5):
        np.random.seed(seed)
        
        # Initial configuration: structured grid with jitter
        centers = np.zeros((n, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n:
                    centers[idx] = [0.15 + j * 0.18, 0.15 + i * 0.18]
                    idx += 1
        if idx < n:
            centers[idx] = [0.5, 0.9]
            idx += 1
            
        # Add random perturbation to break symmetry
        centers += np.random.uniform(-0.05, 0.05, (n, 2))
        centers = np.clip(centers, 0.05, 0.95)
        
        # Initial small radii (feasible for the spacing)
        radii = np.full(n, 0.04)
        
        # Flatten for optimizer
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Bounds: centers in [0, 1], radii in [1e-4, 0.4]
        bounds = [(0.0, 1.0) for _ in range(2*n)] + [(1e-4, 0.4) for _ in range(n)]
        
        # Optimize
        res = minimize(
            objective, x0, args=(n, lam),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 20000, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        
        opt_centers = res.x[:2*n].reshape((n, 2))
        opt_radii = res.x[2*n:]
        
        # Projection step to guarantee strict validity
        final_radii = np.copy(opt_radii)
        for i in range(n):
            # Max radius allowed by boundaries
            max_r = min(
                opt_centers[i, 0],
                1.0 - opt_centers[i, 0],
                opt_centers[i, 1],
                1.0 - opt_centers[i, 1]
            )
            # Max radius allowed by other circles
            for j in range(n):
                if i != j:

            final_radii[i] = max(max_r, 0.0)
            
        current_sum = np.sum(final_radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = opt_centers
            best_radii = final_radii
            
    return best_centers, best_radii, float(best_sum)
