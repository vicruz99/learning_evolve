import numpy as np
from scipy.optimize import minimize
import warnings

# Suppress warnings for cleaner output during optimization steps
warnings.filterwarnings('ignore')

def generate_initial_state(N):
    """
    Generate an initial valid configuration for N=26 circles
    based on a hexagonal lattice pattern (6, 5, 6, 5, 4).
    """
    # Pattern for 26 circles: 5 rows with counts 6, 5, 6, 5, 4
    row_counts = [6, 5, 6, 5, 4]
    
    # Initial radius guess. 
    # For 26 circles, a radius of ~0.05 is a safe lower bound for a loose grid,
    # allowing the optimizer to expand them.
    r_guess = 0.04 
    
    centers = []
    
    # Y-coordinates for 5 rows distributed evenly within [0, 1] with some margin
    # We place centers in the range [0.1, 0.9] initially to be safe
    y_levels = np.linspace(0.1, 0.9, 5)
    
    for i, count in enumerate(row_counts):
        y = y_levels[i]
        
        # Determine X-coordinates
        # For hexagonal packing, odd rows (1, 3) are shifted
        # We distribute them across the width [0.1, 0.9]
        x_range_start = 0.1
        x_range_end = 0.9
        
        # Shift for hexagonal staggering
        if i % 2 != 0:
            shift = (x_range_end - x_range_start) / (2 * (count + 1)) 
            # Adjust start for shifted rows
            x_start = x_range_start + shift
            x_end = x_range_end - shift
        else:
            x_start = x_range_start
            x_end = x_range_end
            
        if count == 0:
            continue
            
        # Linearly space x coordinates
        if count == 1:
            xs = np.array([0.5])
        else:
            xs = np.linspace(x_start, x_end, count)
            
        for x in xs:
            centers.append([x, y, r_guess])
            
    # If we have fewer than N due to logic error (shouldn't happen with fixed counts),
    # or more (padding), ensure we have exactly N
    if len(centers) > N:
        centers = centers[:N]
    elif len(centers) < N:
        # Fallback random fill (unlikely needed)
        while len(centers) < N:
            centers.append([np.random.uniform(0, 1), np.random.uniform(0, 1), r_guess])
            
    return np.array(centers)

def transform_vars(vars_vec, N):
    """
    Transform the optimization variables [r1, ux1, uy1, r2, ux2, uy2, ...]
    back to physical coordinates [x1, y1, r1, ...]
    Transformation: x = r + (1 - 2r) * ux
    """
    N_circles = len(vars_vec) // 3
    r = vars_vec[0::3]
    ux = vars_vec[1::3]
    uy = vars_vec[2::3]
    
    # Ensure r is not negative (though bounds handle this, for safety in calc)
    r = np.maximum(r, 0)
    
    # Calculate x, y from normalized coordinates
    # x = r * (1 - 2*ux) + ux = r - 2*r*ux + ux
    x = r - 2 * r * ux + ux
    y = r - 2 * r * uy + uy
    
    return x, y, r

def objective(vars_vec):
    """
    Objective function: Minimize negative sum of radii.
    """
    _, _, r = transform_vars(vars_vec, len(vars_vec) // 3)
    return -np.sum(r)

def constraints(vars_vec):
    """
    Returns the inequality constraints.
    Constraint: dist_ij^2 >= (r_i + r_j)^2
    Returned value must be >= 0.
    """
    x, y, r = transform_vars(vars_vec, len(vars_vec) // 3)
    n = len(r)
    
    # Vectorized pairwise distance calculation
    # x_diff shape (n, n)
    x_diff = x[:, None] - x[None, :]
    y_diff = y[:, None] - y[None, :]
    
    dist_sq = x_diff**2 + y_diff**2
    
    # r_sum_sq shape (n, n)
    r_sum_sq = (r[:, None] + r[None, :])**2
    
    # Constraint value
    val = dist_sq - r_sum_sq
    
    # Extract upper triangular part (excluding diagonal)
    # There are n*(n-1)/2 constraints
    triu_indices = np.triu_indices(n, k=1)
    return val[triu_indices]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_vars = 3 * N
    
    # 1. Initial State
    # Generate centers
    init_centers_data = generate_initial_state(N)
    
    # Convert to optimization variables
    # We need to map (x, y) to (r, ux, uy)

    vars_list = []

        x = init_centers_data[i, 0]
        y = init_centers_data[i, 1]
        r = init_centers_data[i, 2]
        
        # Avoid division by zero if r=0.5, though bounds prevent this
        denom = 1 - 2 * r
        if abs(denom) < 1e-9:
            denom = 1e-9 
            
        ux = (x - r) / denom
        uy = (y - r) / denom
        
        vars_list.extend([r, ux, uy])
        
    x0 = np.array(vars_list)
    
    # 2. Bounds
    # r in [0, 0.5]
    # ux, uy in [0, 1]
    bounds = []

        bounds.append((0, 0.5))   # r
        bounds.append((0, 1.0))   # ux
        bounds.append((0, 1.0))   # uy
        
    # 3. Constraints
    cons = ({'type': 'ineq', 'fun': constraints})
    
    # 4. Optimization
    best_res = None
    best_sum = -np.inf

    # The problem is non-convex.
    n_restarts = 5

        # Perturb initial guess slightly
        if k == 0:
            x0_curr = x0
        else:

            noise[1::3] *= 0.05  # Larger noise to positions

                best_res = res

            continue

        radii = r

        centers = np.column_stack((x, y))
        radii = r
        
    # Final sum
    final_sum = np.sum(radii)
    
    return centers, radii, float(final_sum)
