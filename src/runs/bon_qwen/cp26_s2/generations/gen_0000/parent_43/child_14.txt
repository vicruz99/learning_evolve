import numpy as np
from scipy.optimize import minimize
import warnings

# Suppress optimization warnings for cleaner output
warnings.filterwarnings('ignore')

def compute_objective(vars, n=26, lam=1000.0):
    """
    Computes the objective function: -sum(radii) + lambda * penalty
    Penalty enforces boundary and non-overlap constraints.
    """
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Primary objective: maximize sum of radii (minimize negative sum)
    obj = -np.sum(rs)
    penalty = 0.0
    
    # 1. Boundary penalties: circles must stay inside [0,1]x[0,1]
    # Constraints: r <= x <= 1-r  and  r <= y <= 1-r
    low_x = np.minimum(0.0, xs - rs)
    high_x = np.minimum(0.0, 1.0 - xs - rs)
    low_y = np.minimum(0.0, ys - rs)
    high_y = np.minimum(0.0, 1.0 - ys - rs)
    penalty += np.sum(low_x**2 + high_x**2 + low_y**2 + high_y**2)
    
    # 2. Inter-circle non-overlap penalties
    # Constraint: distance(i,j) >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # Add epsilon to prevent sqrt(0) gradient issues
    dist = np.sqrt(dx*dx + dy*dy + 1e-12)
    overlap = np.maximum(0.0, dr - dist)
    
    # Only sum upper triangle to avoid double counting pairs
    idx = np.triu_indices(n, k=1)
    penalty += np.sum(overlap[idx]**2)
    
    return obj + lam * penalty

def run_packing():
    n = 26
    best_vars = None
    best_obj = np.inf
    
    # Run multiple trials with different random seeds to escape local minima
    for trial in range(10):
        np.random.seed(trial * 17 + 3)
        
        # Generate hexagonal grid initial guess
        pts = []
        cols = 6
        rows = 5
        spacing_x = 1.0 / (cols + 1)
        spacing_y = spacing_x * np.sqrt(3) / 2
        
        cx, cy = 0.5, 0.5
        for r in range(rows):
            for c in range(cols):
                if len(pts) >= n:
                    break
                x = cx + (c - (cols-1)/2) * spacing_x
                y = cy + (r - (rows-1)/2) * spacing_y
                if r % 2 == 1:
                    x += spacing_x / 2
                pts.append([x, y])
                
        centers = np.array(pts[:n])
        # Add controlled jitter to break symmetry
        centers += np.random.randn(n, 2) * 0.02
        centers = np.clip(centers, 0.08, 0.92)
        
        radii = np.full(n, 0.085)
        
        # Flatten to 1D array for scipy: [x1, y1, r1, x2, y2, r2, ...]
        vars0 = np.zeros(3*n)
        for i in range(n):
            vars0[3*i] = centers[i, 0]
            vars0[3*i+1] = centers[i, 1]
            vars0[3*i+2] = radii[i]
            
        current_vars = vars0.copy()
        
        # Continuation method: gradually increase penalty weight
        lambdas = [10, 50, 200, 1000, 5000, 20000, 50000]
        for lam in lambdas:
            res = minimize(compute_objective, current_vars, args=(n, lam),
                           method='L-BFGS-B',
                           bounds=[(0.01, 0.99)]*(2*n) + [(0.005, 0.45)]*n,
                           options={'maxiter': 2000, 'ftol': 1e-10})
            current_vars = res.x
            
        if res.fun < best_obj:
            best_obj = res.fun
            best_vars = res.x.copy()
            
    # Extract best solution
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Final safety check & slight shrink if needed to satisfy 1e-12 tolerance
    valid = True
    for i in range(n):
        # Boundary check
        if (centers[i,0] - radii[i] < -1e-9 or centers[i,0] + radii[i] > 1 + 1e-9 or
            centers[i,1] - radii[i] < -1e-9 or centers[i,1] + radii[i] > 1 + 1e-9):
            valid = False
            break
        # Overlap check
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-9:
                valid = False
                break
        if not valid:
            break
            
    if not valid:
        # Conservatively shrink radii to guarantee validity
        radii *= 0.998
        
    # Ensure non-negative
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
