import numpy as np
from scipy.optimize import minimize

def loss_and_grad(params, N=26, lam=5000.0):
    """
    Computes the objective (negative sum of radii) and its gradient,
    including quadratic penalties for overlap and boundary violations.
    """
    pts = params.reshape(N, 3)
    x = pts[:, 0]
    y = pts[:, 1]
    r = pts[:, 2]
    
    loss = -np.sum(r)
    grad = np.zeros_like(params)
    grad[:, 2] = -1.0
    
    # Pairwise non-overlap constraints
    for i in range(N):
        for j in range(i + 1, N):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            d = np.hypot(dx, dy)
            if d < 1e-14:
                d = 1e-14
            overlap = r[i] + r[j] - d
            if overlap > 0:
                loss += lam * overlap**2
                g = 2.0 * lam * overlap
                fx = dx / d
                fy = dy / d
                grad[i, 0] -= g * fx
                grad[i, 1] -= g * fy
                grad[i, 2] += g
                grad[j, 0] += g * fx
                grad[j, 1] += g * fy
                grad[j, 2] += g
                
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    for i in range(N):
        val = r[i] - x[i]
        if val > 0:
            loss += lam * val**2
            g = 2.0 * lam * val
            grad[i, 0] -= g
            grad[i, 2] += g
        val = x[i] + r[i] - 1.0
        if val > 0:
            loss += lam * val**2
            g = 2.0 * lam * val
            grad[i, 0] += g
            grad[i, 2] += g
        val = r[i] - y[i]
        if val > 0:
            loss += lam * val**2
            g = 2.0 * lam * val
            grad[i, 1] -= g
            grad[i, 2] += g
        val = y[i] + r[i] - 1.0
        if val > 0:
            loss += lam * val**2
            g = 2.0 * lam * val
            grad[i, 1] += g
            grad[i, 2] += g
            
    return loss, grad

def run_packing():
    N = 26
    best_params = None
    best_score = -np.inf
    
    # Multi-start optimization to explore different basin of attraction
    for trial in range(20):
        np.random.seed(trial * 13 + 5)
        # Initialize in the bulk to avoid immediate boundary penalties
        x = np.random.uniform(0.2, 0.8, N)
        y = np.random.uniform(0.2, 0.8, N)
        r = np.full(N, 0.03)
        params = np.column_stack((x, y, r)).ravel()
        
        bounds = [(0.0, 1.0) for _ in range(3 * N)]
        res = minimize(loss_and_grad, params, jac=True, args=(N, 5000.0), 
                       method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-9})
        
        final_params = res.x.reshape(N, 3)
        
        # Strict violation check
        pts = final_params
        fx, fy, fr = pts[:, 0], pts[:, 1], pts[:, 2]
        max_viol = 0.0
        
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(fx[i] - fx[j], fy[i] - fy[j])
                ov = fr[i] + fr[j] - d
                if ov > max_viol:
                    max_viol = ov
            # Check boundaries
            for coord in (fx[i], fy[i]):
                if fr[i] - coord > max_viol:
                    max_viol = fr[i] - coord
                if coord + fr[i] - 1.0 > max_viol:
                    max_viol = coord + fr[i] - 1.0
                    
        # Minimal shrinkage to guarantee hard validity
        if max_viol > 1e-9:
            final_params[:, 2] -= max_viol * 1.01
            fr = final_params[:, 2]
            
        score = np.sum(fr)
        if score > best_score:
            best_score = score
            best_params = final_params.copy()
            
        # Early termination heuristic if we significantly beat target
        if best_score > 2.64:
            continue

    centers = best_params[:, :2]
    radii = best_params[:, 2]
    return centers, radii, np.sum(radii)
