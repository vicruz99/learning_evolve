from scipy.optimize import minimize

def compute_constraints(v):

    c = v[:52].reshape(n, 2)

        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Pairwise distance constraints: dist^2 >= (r_i + r_j)^2
    d_con = np.empty(325)

            rsq = (r[i] + r[j])**2
            d_con[idx] = d2 - rsq
            idx += 1
            
    return np.concatenate([b_con, d_con])

def compute_objective(v):
    return -np.sum(v[52:])
