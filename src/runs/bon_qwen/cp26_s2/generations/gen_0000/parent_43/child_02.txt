import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    def objective(x):
        """Objective to maximize sum of radii (minimize negative sum)."""
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        
        obj = -np.sum(radii)
        
        # Large penalty for constraint violations
        P = 1e8
        
        # Boundary penalties: x >= r, x <= 1-r, y >= r, y <= 1-r
        obj += P * np.sum(np.maximum(0, radii - centers[:, 0])**2)
        obj += P * np.sum(np.maximum(0, centers[:, 0] + radii - 1)**2)
        obj += P * np.sum(np.maximum(0, radii - centers[:, 1])**2)
        obj += P * np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2)
        
        # Overlap penalties: dist(i,j) >= r_i + r_j
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        obj += P * np.sum(np.maximum(0, r_sum[mask] - dists[mask])**2)
        
        return obj
    
    def init_hex(n, spacing=0.21, seed=0):
        """Hexagonal packing initialization."""
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.09
        
        idx = 0
        row = 0
        while idx < n:
            col = 0
            while idx < n:
                x = spacing/2 + col * spacing + (spacing/2 if row % 2 == 1 else 0)
                y = spacing * np.sqrt(3)/2 * row + spacing/2
                if x < 1 - spacing/2 and y < 1 - spacing/2:
                    centers[idx] = [x, y]
                    idx += 1
                col += 1
            row += 1
        
        centers += np.random.uniform(-0.008, 0.008, size=centers.shape)
        centers = np.clip(centers, 0.04, 0.96)
        return centers, radii
    
    def init_grid(n, seed=0):
        """Grid initialization with slight randomness."""
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.09
        
        # Arrange in roughly 5x5 pattern
        spacing_x = 1.0 / 6
        spacing_y = 1.0 / 6
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n:
                    centers[idx] = [spacing_x * (i + 1), spacing_y * (j + 1)]
                    idx += 1
        
        centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
        centers = np.clip(centers, 0.04, 0.96)
        return centers, radii
    
    def init_mixed(n, seed=0):
        """Mixed initialization: larger circles in corners, smaller in center."""
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        
        # 4 corner circles
        corners = [[0.14, 0.14], [0.86, 0.14], [0.14, 0.86], [0.86, 0.86]]
        for i, (x, y) in enumerate(corners):
            centers[i] = [x, y]
            radii[i] = 0.14
        
        # 8 edge circles
        edges = [
            [0.5, 0.11], [0.25, 0.11], [0.75, 0.11],
            [0.5, 0.89], [0.25, 0.89], [0.75, 0.89],
            [0.11, 0.5], [0.89, 0.5]
        ]
        for i in range(4, 12):
            centers[i] = edges[i-4]
            radii[i] = 0.10
        
        # 14 interior circles
        idx = 12
        row = 0
        while idx < n:
            col = 0
            while idx < n:
                x = 0.32 + col * 0.10 + (0.05 if row % 2 == 1 else 0)
                y = 0.32 + row * 0.09 * np.sqrt(3)/2
                if 0.15 < x < 0.85 and 0.15 < y < 0.85:
                    centers[idx] = [x, y]
                    radii[idx] = 0.08
                    idx += 1
                col += 1
            row += 1
        
        centers += np.random.uniform(-0.005, 0.005, size=centers.shape)
        centers = np.clip(centers, 0.02, 0.98)
        return centers, radii
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    bounds = [(0.001, 0.999) for _ in range(2*n)] + [(0.001, 0.5) for _ in range(n)]
    
    # Try multiple initial configurations
    init_configs = [
        [(lambda n, s: init_hex(n, spacing=0.21, seed=s), 15)],
        [(lambda n, s: init_hex(n, spacing=0.23, seed=s), 15)],
        [(lambda n, s: init_grid(n, seed=s), 10)],
        [(lambda n, s: init_mixed(n, seed=s), 10)],
    ]
    
    all_seeds = []
    for configs in init_configs:
        for init_func, num_trials in configs:
            for seed in range(num_trials):
                all_seeds.append((init_func, seed))
    
    trial = 0
    for init_func, seed in all_seeds:
        centers, radii = init_func(n, seed)
        x0 = np.concatenate([centers.flatten(), radii])
        
        try:
            result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                            options={'maxiter': 15000, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})
            
            c = result.x[:2*n].reshape(n, 2)
            r = result.x[2*n:]
            s = np.sum(r)
            
            # Check validity of solution
            valid = True
            for i in range(n):
                if r[i] < 1e-10:
                    valid = False
                    break
                if c[i][0] - r[i] < -1e-10 or c[i][0] + r[i] > 1 + 1e-10:
                    valid = False
                    break
                if c[i][1] - r[i] < -1e-10 or c[i][1] + r[i] > 1 + 1e-10:
                    valid = False
                    break
            if valid:
                for i in range(n):
                    for j in range(i+1, n):
                        d = np.sqrt(np.sum((c[i]-c[j])**2))
                        if d < r[i] + r[j] - 1e-10:
                            valid = False
                            break
                    if not valid:
                        break
            
            if valid and s > best_sum:
                best_sum = s
                best_centers = c.copy()
                best_radii = r.copy()
                trial += 1
        except:
            pass
    
    if best_centers is None:
        centers, radii = init_hex(n, spacing=0.22, seed=42)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)
    
    return best_centers, best_radii, best_sum
