import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def optimize_packing(n_circles, initial_centers, initial_radii):
    """
    Optimizes the positions and radii of circles to maximize the sum of radii.
    """
    n = n_circles
    
    # Flattened variables: [x0, y0, r0, x1, y1, r1, ..., x_n-1, y_n-1, r_n-1]
    # Size: 3 * n
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])
    
    # Bounds for variables: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)]) # x, y, r

    # Constraints
    constraints = []

    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # x - r >= 0  => r - x <= 0
    # 1 - x - r >= 0 => x + r - 1 <= 0
    # y - r >= 0 => r - y <= 0
    # 1 - y - r >= 0 => y + r - 1 <= 0
    for i in range(n):
        idx = 3 * i
        # x - r >= 0
        cons_x_min = lambda v, i=i: v[idx] - v[idx + 2] # x - r
        constraints.append({'type': 'ineq', 'fun': cons_x_min})
        
        # 1 - (x + r) >= 0
        cons_x_max = lambda v, i=i: 1.0 - (v[idx] + v[idx + 2])
        constraints.append({'type': 'ineq', 'fun': cons_x_max})
        
        # y - r >= 0
        cons_y_min = lambda v, i=i: v[idx + 1] - v[idx + 2] # y - r
        constraints.append({'type': 'ineq', 'fun': cons_y_min})
        
        # 1 - (y + r) >= 0
        cons_y_max = lambda v, i=i: 1.0 - (v[idx + 1] + v[idx + 2])
        constraints.append({'type': 'ineq', 'fun': cons_y_max})

    # Non-overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            
            def cons_overlap(v, i=i, j=j, idx_i=idx_i, idx_j=idx_j):
                xi, yi, ri = v[idx_i], v[idx_i+1], v[idx_i+2]
                xj, yj, rj = v[idx_j], v[idx_j+1], v[idx_j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                return dist_sq - sum_r**2

            constraints.append({'type': 'ineq', 'fun': cons_overlap})

    # Objective: Maximize sum of radii => Minimize negative sum
    def objective(v):
        radii_vals = v[2::3]
        return -np.sum(radii_vals)

    try:
        # Use SLSQP which handles bounds and non-linear constraints
        res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-9})
        
        if res.success:
            optimal_centers = res.x[:2*n].reshape((n, 2))
            optimal_radii = res.x[2*n:]
            return optimal_centers, optimal_radii, -res.fun
        else:
            # Return initial if optimization failed
            return initial_centers, initial_radii, np.sum(initial_radii)
            
    except Exception as e:
        print(f"Optimization error: {e}")
        return initial_centers, initial_radii, np.sum(initial_radii)
