import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Generate initial configurations
    # We'll try a few different starting layouts to ensure we find a good packing
    configs = []

    # Configuration 1: Hexagonal grid scaled to unit square
    # We generate more points than needed and pick the first 26, or arrange them carefully
    rows = 6
    cols = 6
    centers_1 = []
    count = 0
    for r in range(rows):
        for c in range(cols):
            if count >= n_circles:
                break
            # Hexagonal offset
            x = c * 2 + (0.5 if r % 2 == 1 else 0)
            y = r * math.sqrt(3)
            centers_1.append([x, y])
            count += 1
    # Scale to fit in [0, 1]x[0, 1] with some margin for initial radii
    centers_1 = np.array(centers_1[:n_circles])
    # Normalize to [0, 1]
    centers_1 = centers_1 / np.max(centers_1) * 0.8 + 0.1
    configs.append(centers_1)

    # Configuration 2: Random perturbation of a grid
    centers_2 = np.zeros((n_circles, 2))
    # 5x5 grid + 1
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n_circles:
                centers_2[idx] = [ (2*i + 1)/10.0, (2*j + 1)/10.0 ]
                idx += 1
    if idx < n_circles:
        # Place last one in center
        centers_2[idx] = [0.5, 0.5]
    # Add noise
    centers_2 += np.random.uniform(-0.05, 0.05, size=centers_2.shape)
    # Clip to valid range
    centers_2 = np.clip(centers_2, 0.05, 0.95)
    configs.append(centers_2)

    # Configuration 3: Random uniform
    np.random.seed(42)
    centers_3 = np.random.uniform(0.1, 0.9, size=(n_circles, 2))
    configs.append(centers_3)

    def objective(params):
        # params contains [x0, y0, r0, x1, y1, r1, ...]
        # But it's easier to reshape
        centers = params[:2 * n_circles].reshape(n_circles, 2)
        radii = params[2 * n_circles:]
        return -np.sum(radii) # Minimize negative sum

    def get_constraints(centers_init):
        cons = []
        
        # Boundary constraints
        for i in range(n_circles):
            idx_x = 2 * i
            idx_y = 2 * i + 1
            idx_r = 2 * n_circles + i
            
            # x - r >= 0  => x - r - 0 >= 0
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[idx_x] - p[idx_r]})
            # 1 - (x + r) >= 0
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[idx_x] - p[idx_r]})
            # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[idx_y] - p[idx_r]})
            # 1 - (y + r) >= 0
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[idx_y] - p[idx_r]})
            # r >= 0
            cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[idx_r]})

        # Overlap constraints: dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # dist^2 - (r_i + r_j)^2 >= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_xi = 2 * i
                idx_yi = 2 * i + 1
                idx_ri = 2 * n_circles + i
                idx_xj = 2 * j
                idx_yj = 2 * j + 1
                idx_rj = 2 * n_circles + j

                def dist_constraint(p, i=i, j=j, 
                                    xi=idx_xi, yi=idx_yi, ri=idx_ri,
                                    xj=idx_xj, yj=idx_yj, rj=idx_rj):
                    dx = p[xi] - p[xj]
                    dy = p[yi] - p[yj]
                    d_sq = dx*dx + dy*dy
                    r_sum = p[ri] + p[rj]
                    return d_sq - r_sum*r_sum

                cons.append({'type': 'ineq', 'fun': dist_constraint})
        
        return cons

    constraints = get_constraints(np.zeros((n_circles, 2))) # Constraints depend on indices, not values

        try:
            res = minimize(objective, init_params, method='SLSQP', 
                           constraints=constraints, 
                           options={'maxiter': 2000, 'ftol': 1e-12})
            
            if res.success or (not np.isnan(res.fun) and np.sum(res.x[2*n_circles:]) > best_sum):
                # Extract result
                c = res.x[:2*n_circles].reshape(n_circles, 2)

                            if dist < r[i] + r[j] - 1e-9:
                                is_valid = False
                                break
                        if not is_valid: break
                
                if is_valid:
                    s = np.sum(r)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c
                        best_radii = r

        except Exception:
            continue

    # If no good solution found from optimization (rare with SLSQP), fallback to grid
    if best_centers is None:
        best_centers = np.zeros((n_circles, 2))
        best_radii = np.zeros(n_circles)
        # 5x5 grid

                    idx += 1
        if idx < n_circles:
            best_centers[idx] = [0.5, 0.5]
            best_radii[idx] = 0.01 # Small filler
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
