import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform

def get_optimal_radii_and_value(centers):
    """
    Solves the LP to maximize sum of radii given fixed centers.
    Variables: r_0, ..., r_25
    Maximize sum(r_i) => Minimize sum(-r_i)
    Constraints:
    1. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    2. r_i + r_j <= d_{ij} for all pairs
    3. r_i >= 0
    """
    n = centers.shape[0]
    c = np.ones(n) * -1  # Minimize -sum(r)

    # Constraints matrix A_ub * r <= b_ub
    # 1. Boundary constraints: 4 per circle
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x_i
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - x)
        
        # r_i <= y_i
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y_i
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - y)

    # 2. Overlap constraints: r_i + r_j <= d_ij
    # There are n*(n-1)/2 pairs
    # To keep matrix size manageable, we construct it carefully
    # A_ub will be (4n + n(n-1)/2, n)
    
    # Pre-allocate
    total_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((total_constraints, n))
    b_ub = np.zeros(total_constraints)
    
    idx = 0
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        # r_i <= 1-x
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        # r_i <= y
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        # r_i <= 1-y
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1

    # Overlap constraints
    # Compute pairwise distances
    dists = squareform(pdist(centers))
    
    for i in range(n):
        for j in range(i + 1, n):
            # r_i + r_j <= dist
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1

    bounds = [(0, None) for _ in range(n)]
    
    # Use 'highs' solver for speed
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        # Fallback if LP fails (should rarely happen)
        return 0.0, np.zeros(n)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
