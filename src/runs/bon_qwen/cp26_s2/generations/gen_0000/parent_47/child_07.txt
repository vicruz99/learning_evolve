import numpy as np
from scipy.optimize import minimize

def calculate_penalty(centers_flat, r, n):
    """
    Calculates the penalty for a given configuration of centers and radius r.
    Penalty is sum of squared overlaps between circles and with boundaries.
    """
    centers = centers_flat.reshape(n, 2)
    
    # 1. Inter-circle overlaps
    # Calculate pairwise distances
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Overlap is (2r - dist) if positive
    # We want to penalize when dist < 2r
    # overlap = max(0, 2r - dist)
    # Since dists[i,i] = 0, overlap is 2r, which is large. We ignore diagonal.
    # We can mask the diagonal or just accept it, but it adds constant 2r^2 * n.
    # Better to ignore diagonal.
    
    # Create a mask for upper triangle to avoid double counting and self
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    # Overlap amount
    overlap = 2.0 * r - dists
    # Set negative overlaps to 0 (no penalty)
    overlap[overlap < 0] = 0.0
    
    # Sum squared overlaps
    penalty = np.sum(overlap[mask]**2)
    
    # 2. Boundary constraints
    # Circle i with center (x, y) and radius r must satisfy:
    # x >= r  => r - x <= 0
    # x <= 1 - r => x - (1 - r) <= 0
    # Same for y
    
    xs = centers[:, 0]
    ys = centers[:, 1]
    
    # Violations
    # Left wall: r - x
    viol_left = np.maximum(0, r - xs)
    # Right wall: x - (1 - r)
    viol_right = np.maximum(0, xs - (1.0 - r))
    # Bottom wall: r - y
    viol_bottom = np.maximum(0, r - ys)
    # Top wall: y - (1 - r)
    viol_top = np.maximum(0, ys - (1.0 - r))
    
    penalty += np.sum(viol_left**2)
    penalty += np.sum(viol_right**2)
    penalty += np.sum(viol_bottom**2)
    penalty += np.sum(viol_top**2)
    
    return penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initial configuration: Random points centered in the square to avoid immediate boundary issues
    # Using a fixed seed for reproducibility
    rng = np.random.RandomState(42)
    # Start with small radius
    r = 0.01

    centers = 0.1 + 0.8 * rng.rand(n, 2)

    # though with r=0.01 they are likely fine.
    x0 = centers.flatten()
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    
    # Helper for minimize
    def obj_func(x):
        return calculate_penalty(x, r, n)

    # Minimize penalty for initial r
    res = minimize(obj_func, x0, bounds=bounds, method='L-BFGS-B', 
                   options={'ftol': 1e-12, 'gtol': 1e-10, 'maxiter': 2000})
    centers = res.x.reshape(n, 2)
    
    # Iterative expansion of radius
    # We try to increase r and resolve overlaps
    step = 0.0005
    max_iters = 2000 # Safety break
    
    # We track the best valid r and centers
    best_r = r
    best_centers = centers.copy()
    
    # Loop to grow radius
    # We can try to increase r as long as we can find a valid configuration
    # We use a while loop with a counter
    
    iters = 0
    consecutive_failures = 0
    
    # Target sum is ~2.636 => r ~ 0.1014
    # Start from 0.01, need to grow ~0.09
    # With step 0.0005, ~180 steps.
    
    while iters < max_iters:
        iters += 1
        
        # Try to increase radius
        new_r = r + step
        
        # Check current penalty with new radius
        # We don't need to re-optimize immediately if penalty is low
        # But let's check
        current_penalty = calculate_penalty(centers.flatten(), new_r, n)
        
        if current_penalty < 1e-8:
            # Valid configuration found for new_r
            r = new_r
            best_r = r
            best_centers = centers.copy()
            consecutive_failures = 0
            # Maybe increase step slightly to speed up
            step = min(step * 1.02, 0.002)

                return calculate_penalty(x, new_r, n)

    # Final radii array

    return best_centers, radii, sum_radii
