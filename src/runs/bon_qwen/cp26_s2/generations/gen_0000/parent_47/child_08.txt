import numpy as np
from scipy.optimize import minimize

def generate_initial_packing(n=26):
    """
    Generates an initial hexagonal packing of n circles.
    """
    # Configuration: 5 rows with counts [6, 5, 6, 5, 4]
    # This sums to 26 and fits well in a square aspect ratio.
    row_counts = [6, 5, 6, 5, 4]
    
    # Hexagonal packing parameters
    # We scale the grid to fit in [0,1]x[0,1]
    # Width constraint: max(2*c_A, 2*c_B + 1) * r <= 1
    # Height constraint: (2 + (rows-1)*sqrt(3)) * r <= 1
    
    # Estimate max radius for this grid
    max_c_a = 6
    max_c_b = 5
    # Width constraints
    r_width_a = 1.0 / (2 * max_c_a)
    r_width_b = 1.0 / (2 * max_c_b + 1)
    # Height constraint (5 rows)
    r_height = 1.0 / (2 + (5 - 1) * np.sqrt(3))
    
    # The limiting factor for the grid geometry
    r_max = min(r_width_a, r_width_b, r_height)
    
    # We scale down slightly to ensure valid start for optimization
    r_scale = 0.95 * r_max
    
    centers = []
    y_curr = r_scale
    
    for i, count in enumerate(row_counts):
        x_start = r_scale + (i % 2) * r_scale
        for j in range(count):
            x = x_start + j * 2 * r_scale
            centers.append([x, y_curr])
        y_curr += r_scale * np.sqrt(3)
        
    return np.array(centers), np.full(n, r_scale)

    for scale in scale_factors:

            # 1. Generate Initial Guess
            init_centers, init_radii = generate_initial_packing(n)
            # Scale the radii to start from a valid configuration
            current_radii = init_radii * scale
            
            # 2. Define Objective Function: Maximize sum of radii (Minimize negative sum)
            def objective(x):
                centers = x[:2 * n].reshape(n, 2)
                radii = x[2 * n:]
                return -np.sum(radii)

            # 3. Define Constraints
            constraints = []
            
            # Boundary Constraints: r_i <= x_i, r_i <= 1 - x_i, etc.
            # r_i - x_i <= 0  => x_i - r_i >= 0
            # 1 - x_i - r_i >= 0 => x_i + r_i <= 1
            for i in range(n):
                # x_i - r_i >= 0
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*n + i]})
                # 1 - x_i - r_i >= 0
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i] - x[2*n + i]})
                # y_i - r_i >= 0
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i + 1] - x[2*n + i]})
                # 1 - y_i - r_i >= 0
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i + 1] - x[2*n + i]})
                
                # r_i >= 0
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*n + i]})

            # Overlap Constraints: dist >= r_i + r_j
            # sqrt((x_i - x_j)^2 + (y_i - y_j)^2) - (r_i + r_j) >= 0

            # due to the sum of radii on RHS, but direct distance is fine with careful implementation.
            for i in range(n):
                for j in range(i + 1, n):
                    constraints.append({
                        'type': 'ineq',

                    })

            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
