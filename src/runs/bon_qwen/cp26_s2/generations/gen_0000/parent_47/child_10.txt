import numpy as np
from scipy.optimize import linprog

def get_max_radii(centers):
    """
    Solves the Linear Program to find maximal radii for fixed centers.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Boundary constraints:
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    # 2. Pairwise constraints: r_i + r_j <= dist(i, j)
    
    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        # r_i <= 1 - x
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        # r_i <= y
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        # r_i <= 1 - y
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            # r_i + r_j <= dist
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve using Highs solver (usually fastest and most robust)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    
    # Fallback to other methods if Highs fails
    for method in ['interior-point', 'simplex']:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
            if res.success:
                return res.x
        except:
            continue
            
    return np.zeros(n)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Initialization: Hexagonal Lattice ---
    centers = np.zeros((n, 2))
    
    # Estimate a reasonable initial radius to fit 26 circles
    # Square grid of 25 circles has r=0.1. 
    # Hex grid is denser, so r ~ 0.09 might fit 26.
    r_guess = 0.09
    dx = 2 * r_guess
    dy = np.sqrt(3) * r_guess
    
    points = []
    y = r_guess
    row_idx = 0
    
    # Generate hexagonal points
    while len(points) < n:
        # Offset logic for hex grid
        # Even rows (0, 2, ...): start at r_guess
        # Odd rows (1, 3, ...): start at 2*r_guess (shifted right by r_guess)
        if row_idx % 2 == 0:
            x_start = r_guess
        else:
            x_start = 2 * r_guess
            
        x = x_start
        while x <= 1 - r_guess and len(points) < n:
            points.append([x, y])
            x += dx
        
        y += dy
        row_idx += 1
        
    centers = np.array(points[:n])
    
    # --- 2. Optimization Loop ---
    best_centers = centers.copy()
    best_radii = get_max_radii(centers)
    best_sum_radii = np.sum(best_radii)
    
    # Optimization parameters
    max_iters = 800
    force_strength = 0.02
    jitter_magnitude = 0.005
    
    for iteration in range(max_iters):
        # 2a. Solve LP for maximal radii
        radii = get_max_radii(centers)
        current_sum = np.sum(radii)
        
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        # 2b. Compute forces based on active constraints
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < 1e-12:
                    dist = 1e-12
                
                slack = dist - (radii[i] + radii[j])
                
                # If slack is small, constraint is active/tight
                if slack < 0.01:
                    dir_vec = (centers[i] - centers[j]) / dist
                    # Stronger force if very tight
                    f_mag = force_strength if slack > 1e-6 else force_strength * 2
                    forces[i] += f_mag * dir_vec
                    forces[j] -= f_mag * dir_vec

        # Wall repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left
            if x - r < 0.01:
                forces[i, 0] += force_strength * 1.5
            # Right
            if (1.0 - x) - r < 0.01:
                forces[i, 0] -= force_strength * 1.5
            # Bottom
            if y - r < 0.01:
                forces[i, 1] += force_strength * 1.5
            # Top
            if (1.0 - y) - r < 0.01:
                forces[i, 1] -= force_strength * 1.5
            
        # 2c. Update centers
        centers += forces
        centers = np.clip(centers, 0.001, 0.999)
        
        # 2d. Periodic jitter to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            centers += np.random.randn(n, 2) * jitter_magnitude
            centers = np.clip(centers, 0.001, 0.999)
            
    return best_centers, best_radii, best_sum_radii
