import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def create_hex_init(n_circles=26, r_init=0.08):
    centers = []
    # Approximate rows for hex packing
    # 5, 4, 5, 4, 5, 3 = 26 circles
    row_counts = [5, 4, 5, 4, 5, 3]
    y_curr = r_init
    row_idx = 0
    
    for count in row_counts:
        if row_idx > 0:
            y_curr += r_init * np.sqrt(3)
        x_start = r_init if row_idx % 2 == 0 else (2 * r_init) # 0 shift or r shift (center to center)
        # Correction: center-to-center horizontal distance in hex is r? No, it's 2r in same row, r in staggered
        # If we shift by r horizontally, vertical shift is sqrt(3)*r
        # Row 0: r, 3r, 5r...
        # Row 1: 2r, 4r, 6r...
        # So start x depends on shift.
        
        current_x = r_init if row_idx % 2 == 0 else (2 * r_init)
        for _ in range(count):
            centers.append([current_x, y_curr])
            current_x += 2 * r_init
        row_idx += 1
        
    return np.array(centers[:n_circles]), np.full(n_circles, r_init)

def get_cost_and_grad(vars, n_circles):
    # vars is flattened array of [x1, y1, r1, x2, y2, r2, ...]
    centers = vars[:2 * n_circles].reshape(n_circles, 2)
    radii = vars[2 * n_circles:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    cost = -np.sum(radii)
    
    # Penalty for boundary violations
    # x - r >= 0  => r - x <= 0
    # 1 - x - r >= 0 => x + r - 1 <= 0
    # same for y
    margin = 1e-4
    pen = 0.0
    
    # Boundary penalties (smooth)
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        # Lower bounds
        d_x1 = r - x
        d_x2 = x + r - 1.0
        d_y1 = r - y
        d_y2 = y + r - 1.0
        
        for d in [d_x1, d_x2, d_y1, d_y2]:
            if d > margin:
                pen += (d - margin)**2
                
    # Overlap penalties
    # dist >= r1 + r2  => r1 + r2 - dist <= 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            dist = np.sqrt(max(0, dist_sq))
            req_dist = radii[i] + radii[j]
            
            violation = req_dist - dist
            if violation > margin:
                pen += (violation - margin)**2
                
    # Add penalty to cost with a weight. 
    # We need the weight to be high enough to enforce constraints, but not so high it breaks gradient flow.
    # Adaptive weight is hard here, let's use a static high weight.
    penalty_weight = 100.0
    return cost + penalty_weight * pen

def run_packing():

    best_sum_radii = -1.0

    # and different penalty weights/scales to find the global optimum.

    inits = []

    c1, r1 = create_hex_init(n_circles, 0.09)

        inits.append(np.concatenate([c.flatten(), r]))

        # Bounds for variables
        # x, y in [0, 1], r in [0, 0.5]
        bounds = []

        # We run minimize twice: once to find feasible, once to refine
        # Actually, one run with high penalty weight should suffice if started valid-ish
        
        # To make the problem easier, we can scale the penalty dynamically, 
        # but for simplicity, we'll run a few passes.
        
        # Pass 1: Low penalty to find a general shape, Pass 2: High penalty
        weights = [10.0, 50.0, 200.0]
        
        current_vars = init_vars.copy()
        
        for w in weights:
            # Redefine cost function with current weight
            def obj(vars):
                centers = vars[:2 * n_circles].reshape(n_circles, 2)
                radii = vars[2 * n_circles:]
                cost = -np.sum(radii)
                pen = 0.0
                margin = 1e-6 # Tighter margin as we go
                
                # Boundary
                for i in range(n_circles):
                    x, y = centers[i]
                    r = radii[i]
                    for d in [r - x, x + r - 1.0, r - y, y + r - 1.0]:
                        if d > margin:
                            pen += (d - margin)**2
                            
                # Overlap
                for i in range(n_circles):
                    for j in range(i + 1, n_circles):
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = np.sqrt(dx*dx + dy*dy)
                        req = radii[i] + radii[j]
                        viol = req - dist
                        if viol > margin:
                            pen += (viol - margin)**2
                return cost + w * pen

            try:
                res = opt.minimize(obj, current_vars, method='SLSQP', bounds=bounds,

                pass
        
        # Evaluate result

        radii = current_vars[2 * n_circles:]

        if validate_packing(centers, radii):

                break
            factor -= 0.01
            
    return best_centers, best_radii, best_sum_radii
