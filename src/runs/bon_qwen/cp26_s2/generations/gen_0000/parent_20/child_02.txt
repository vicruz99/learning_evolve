import numpy as np
from scipy.optimize import minimize

def objective(params):
    # Objective: Minimize negative sum of radii (equivalent to maximizing sum of radii)
    # params layout: [x0, y0, r0, x1, y1, r1, ...]
    radii = params[2::3]
    return -np.sum(radii)

def bounds():
    # Bounds for each variable: x, y in [0, 1], r in [0, 0.5]
    n = 26
    b = []
    for _ in range(n):
        b.append((0.0, 1.0)) # x
        b.append((0.0, 1.0)) # y
        b.append((1e-6, 0.5)) # r (positive)
    return b

def constraints(params):
    n = 26
    centers = np.column_stack((params[0::3], params[1::3]))
    radii = params[2::3]
    
    cons = []
    
    # Boundary constraints: x - r >= 0, x + r <= 1, etc.
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[3*i] - p[3*i+2]}) # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - (p[3*i] + p[3*i+2])}) # x + r <= 1
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[3*i+1] - p[3*i+2]}) # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - (p[3*i+1] + p[3*i+2])}) # y + r <= 1

    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # Equivalent to: (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2 >= 0

                    (p[3*i] - p[3*j])**2 + (p[3*i+1] - p[3*j+1])**2 - (p[3*i+2] + p[3*j+2])**2

            })

def get_hexagonal_start():
    # Generate a hexagonal grid packing as a starting point
    n = 26
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08 # Initial guess radius
    
    idx = 0
    row = 0
    col = 0
    # Approximate spacing for hex grid
    spacing_x = 0.20
    spacing_y = spacing_x * np.sqrt(3) / 2
    
    # Center the grid in the square
    offset_x = 0.1
    offset_y = 0.1
    
    while idx < n:
        x = offset_x + col * spacing_x
        if row % 2 == 1:
            x += spacing_x / 2
        y = offset_y + row * spacing_y
        
        if x + 0.1 <= 1.0 and y + 0.1 <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            col += 1
        else:
            row += 1
            col = 0
            
    # Flatten to params array
    params = np.zeros(3 * n)
    for i in range(n):
        params[3*i] = centers[i, 0]
        params[3*i+1] = centers[i, 1]
        params[3*i+2] = radii[i]
        
    return params

    radii = best_params[2::3]
