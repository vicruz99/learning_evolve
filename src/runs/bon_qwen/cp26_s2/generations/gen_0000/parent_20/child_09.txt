import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # 1. Initialize centers and radii
    # Start with a 5x5 grid (25 circles) plus one extra circle at the center 
    # (shifted slightly to avoid perfect overlap if center was occupied, 
    # though here center is (0.5, 0.5) which is in the grid).
    # Actually, a 5x5 grid has points at 0.1, 0.3, 0.5, 0.7, 0.9.
    # (0.5, 0.5) is the exact center.
    # To fit 26 circles, we can start with a slightly perturbed grid or a denser layout.
    # Let's use a 6x5 grid (30 points) but only keep 26, or simply place 
    # 26 points on a 5x5 grid structure where the last one is squeezed.
    # Better: Initialize with a valid small radius configuration that is dense.
    # A 5x5 grid of radius 0.1 is valid. We need 26th circle.
    # Let's place 25 circles in a 5x5 grid and the 26th near the center 
    # with a small radius, then let the optimizer expand them.
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    idx = 0
    # 5x5 Grid points
    grid_coords = np.linspace(0.1, 0.9, 5)
    for x in grid_coords:
        for y in grid_coords:
            if idx < n - 1:
                centers[idx] = [x, y]
                radii[idx] = 0.05 # Start with a valid small radius
                idx += 1
    
    # Place the 26th circle at the center (0.5, 0.5) which is a hole in the grid 
    # relative to the 5x5 grid if we consider grid points at 0.1, 0.3...
    # Wait, 0.5 is in grid_coords. So (0.5, 0.5) is already taken.
    # We need to displace the 26th circle slightly or perturb the whole grid.
    # Let's perturb the grid slightly to make room.
    
    if idx < n:
        # Place last circle at center, slightly offset
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.01
    
    # Apply a small random perturbation to avoid symmetry issues and singular gradients
    np.random.seed(42) # Reproducibility
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    # Ensure centers are within [0, 1] after perturbation
    centers = np.clip(centers, 0.01, 0.99)
    
    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds: x, y in [0, 1], r >= 0
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    # Objective function with penalty
    def objective(vars):
        c_x = vars[0::3]
        c_y = vars[1::3]
        r = vars[2::3]
        
        sum_radii = np.sum(r)
        
        # Penalty parameters
        penalty_overlap = 1000.0
        penalty_boundary = 1000.0
        
        penalty = 0.0
        
        # Check overlaps and boundary constraints
        # Using vectorized operations for speed
        # Compute all pairwise distances
        # dist_ij^2 = (xi - xj)^2 + (yi - yj)^2
        # We need sum over i < j
        
        # For performance, we iterate. N=26 is small.
        for i in range(n):
            # Boundary constraints
            xi, yi = c_x[i], c_y[i]
            ri = r[i]
            
            dist_to_boundary = min(xi, 1.0 - xi, yi, 1.0 - yi)
            if ri > dist_to_boundary:
                diff = ri - dist_to_boundary
                penalty += penalty_boundary * (diff ** 2)
            
            # Overlap constraints with j > i
            for j in range(i + 1, n):
                xj, yj = c_x[j], c_y[j]
                rj = r[j]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                # Avoid sqrt for speed and numerical stability, compare squared
                # (ri + rj)^2 <= dist_sq
                sum_r = ri + rj
                if sum_r**2 > dist_sq:
                    diff = sum_r - np.sqrt(dist_sq)
                    penalty += penalty_overlap * (diff ** 2)
        
        return -sum_radii + penalty

    # Optimize
    # Use L-BFGS-B as it handles bounds well and is efficient for this size
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-8})
    
    # Extract results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = result.x[3*i]
        final_centers[i, 1] = result.x[3*i+1]
        final_radii[i] = result.x[3*i+2]
        
    # Clip radii to 0 just in case
    final_radii = np.maximum(final_radii, 0.0)
    
    # Calculate sum
    total_sum = np.sum(final_radii)
    
    # Verify validity (optional check for debugging, not required for return)
    # validate_packing(final_centers, final_radii)
    
    return final_centers, final_radii, float(total_sum)
