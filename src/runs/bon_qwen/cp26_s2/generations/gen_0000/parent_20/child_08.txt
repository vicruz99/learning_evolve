import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initial Configuration
    # Start with a 5x5 grid of 25 circles with radius 0.1 (tight packing)
    # and place the 26th circle in a gap with a small radius.
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Grid parameters
    grid_size = 5
    coords = np.linspace(0.1, 0.9, grid_size)
    
    idx = 0
    for r in range(grid_size):
        for c in range(grid_size):
            centers[idx] = [coords[c], coords[r]]
            radii[idx] = 0.1
            idx += 1
    
    # Place 26th circle in a gap, e.g., (0.2, 0.2)
    # Nearest grid points are (0.1, 0.1), (0.1, 0.3), (0.3, 0.1), (0.3, 0.3)
    # Distance is sqrt(0.02) approx 0.1414.
    # With neighbors at r=0.1, max radius for new circle is 0.0414.
    # We start slightly smaller to be safe, but close.
    centers[idx] = [0.2, 0.2]
    radii[idx] = 0.04
    
    # Initial state vector: [x0, y0, x1, y1, ..., r0, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Define Objective Function
    # We want to maximize sum(radii), so minimize -sum(radii)
    def objective(vars):
        r = vars[52:] # Last 26 elements are radii
        return -np.sum(r)
    
    # 3. Define Constraints
    # Constraints must return value >= 0
    
    constraints = []

    # x - r >= 0  => x - r >= 0
    # 1 - x - r >= 0 => x + r <= 1
    # Same for y
    
    for i in range(n):
        x_idx = 2 * i
        y_idx = 2 * i + 1
        r_idx = 52 + i
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda vars, i=i: vars[x_idx] - vars[r_idx]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda vars, i=i: 1.0 - vars[x_idx] - vars[r_idx]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda vars, i=i: vars[y_idx] - vars[r_idx]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda vars, i=i: 1.0 - vars[y_idx] - vars[r_idx]
        })
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            x_i, y_i = 2 * i, 2 * i + 1
            x_j, y_j = 2 * j, 2 * j + 1
            r_i, r_j = 52 + i, 52 + j
            
            def dist_constraint(vars, i=i, j=j):
                dx = vars[x_i] - vars[x_j]
                dy = vars[y_i] - vars[y_j]
                dr = vars[r_i] + vars[r_j]
                return dx**2 + dy**2 - dr**2
            
            constraints.append({
                'type': 'ineq',
                'fun': dist_constraint
            })

    # 4. Bounds
    # x, y in [0, 1], r in [0, 1] (loose upper bound)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (cannot exceed 0.5 in unit square)

    # 5. Run Optimization
    # SLSQP is suitable for constrained non-linear optimization
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )

        optimal_vars = result.x

    # y coords are at 1, 3, ..., 51
    # r coords are at 52, ..., 77

    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = optimal_vars[2 * i]
        centers[i, 1] = optimal_vars[2 * i + 1]
        radii[i] = optimal_vars[52 + i]
        
    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)
