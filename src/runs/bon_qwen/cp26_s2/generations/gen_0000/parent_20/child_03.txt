import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n):
    """
    Computes the negative sum of radii plus a penalty for constraint violations.
    vars: array of [x1, y1, ..., xn, yn, r1, ..., rn]
    """
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    obj = -np.sum(r)
    
    # Penalty weight
    w = 1e5
    
    # Boundary penalties: ensure 0 <= x-r, x+r <= 1, etc.
    # Violations are when constraint is < 0. We penalize max(0, -constraint)^2
    # x - r >= 0  => r - x <= 0
    b1 = np.maximum(0, r - c[:, 0])
    b2 = np.maximum(0, c[:, 0] + r - 1)
    b3 = np.maximum(0, r - c[:, 1])
    b4 = np.maximum(0, c[:, 1] + r - 1)
    pen = w * np.sum(b1**2 + b2**2 + b3**2 + b4**2)
    
    # Overlap penalties: dist >= r_i + r_j
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)  # Ignore self-distance
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    viol = np.maximum(r_sum - dist, 0)
    
    # Sum only upper triangle to avoid double counting
    tri = np.triu_indices(n, k=1)
    pen += w * np.sum(viol[tri]**2)
    
    return obj + pen

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = -1.0
    best_sol = None
    
    # Initial configurations
    inits = []
    
    # 1. Hexagonal lattice pattern
    pts = []
    y = 0.1
    row = 0
    while len(pts) < n:
        x = 0.1 if row % 2 == 0 else 0.15
        while x < 0.9 and len(pts) < n:
            pts.append([x, y])
            x += 0.15
        y += 0.13
        row += 1
    inits.append(np.array(pts[:n]))
    
    # 2. Random initialization
    inits.append(np.random.rand(n, 2))
    
    # 3. Structured grid
    cols = 6
    rows = 5
    grid = np.mgrid[0:rows, 0:cols].reshape(2, -1).T
    grid = grid[:n]
    grid = grid / np.array([rows-1, cols-1]) * 0.8 + 0.1
    inits.append(grid)
    
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    for pts in inits:
        x0 = np.concatenate([pts.flatten(), np.full(n, 0.09)])
        
        res = minimize(compute_objective, x0, args=(n,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-12})
        
        cur_c = res.x[:2*n].reshape(n, 2)
        cur_r = res.x[2*n:]
        
        # Quick validation check to ensure penalty drove constraints to ~0
        valid = True
        # Check boundaries
        if np.any(cur_r - cur_c[:,0] > 1e-6) or np.any(cur_c[:,0] + cur_r - 1 > 1e-6):
            valid = False
        if np.any(cur_r - cur_c[:,1] > 1e-6) or np.any(cur_c[:,1] + cur_r - 1 > 1e-6):
            valid = False
            
        # Check overlaps
        if valid:
            diff = cur_c[:, np.newaxis, :] - cur_c[np.newaxis, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2))
            np.fill_diagonal(dist, np.inf)
            r_sum = cur_r[:, np.newaxis] + cur_r[np.newaxis, :]
            if np.any(r_sum - dist > 1e-6):
                valid = False
                
        if valid:
            cur_sum = np.sum(cur_r)
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_sol = (cur_c.copy(), cur_r.copy())
                
    # Fallback if optimization fails to find valid (should not happen with penalty)
    if best_sol is None:
        centers = np.random.rand(n, 2)
        radii = np.full(n, 0.04)
        best_sol = (centers, radii)
        
    return best_sol[0], best_sol[1], np.sum(best_sol[1])
