import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # Bounds: r in [1e-4, 0.49], s in [0, 1]
    lb = np.zeros(3*n)
    ub = np.ones(3*n)
    lb[0::3] = 1e-4
    ub[0::3] = 0.49
    lb[1::3] = 0.0
    ub[1::3] = 1.0
    lb[2::3] = 0.0
    ub[2::3] = 1.0
    bounds = list(zip(lb, ub))
    
    mu = 5000.0
    
    def objective(params):
        p = params.reshape(n, 3)
        r = p[:, 0, None]
        sx = p[:, 1, None]
        sy = p[:, 2, None]
        
        # Transform to actual coordinates
        x = r * (1 - 2*sx) + sx
        y = r * (1 - 2*sy) + sy
        
        # Vectorized pairwise distance and overlap calculation
        dx = x - x.T
        dy = y - y.T
        dist = np.sqrt(dx**2 + dy**2)
        
        r_sum = r + r.T
        overlap = np.maximum(0.0, r_sum - dist)
        np.fill_diagonal(overlap, 0.0)
        
        penalty = np.sum(overlap**2)
        return -np.sum(r.ravel()) + mu * penalty

    best_params = None
    best_score = np.inf
    
    # Initialization 1: Hexagonal lattice (dense packing structure)
    row_counts = [5, 4, 5, 4, 5, 3]
    centers_init = []
    r0 = 0.05
    dx0 = 2.2 * r0
    dy0 = np.sqrt(3) * r0
    y_pos = 0.5
    for i, cnt in enumerate(row_counts):
        offset = (1 if i%2==1 else 0) * (dx0/2)
        x_start = 0.5 - (cnt-1)*dx0/2 + offset
        for j in range(cnt):
            centers_init.append([x_start + j*dx0, y_pos])
        y_pos -= dy0
    centers_init = np.array(centers_init)
    centers_init[:, 1] -= np.mean(centers_init[:, 1]) - 0.5
    centers_init[:, 0] -= np.mean(centers_init[:, 0]) - 0.5
    centers_init = np.clip(centers_init, 0.1, 0.9)
    
    p0_hex = np.zeros(3*n)
    p0_hex[0::3] = r0
    for i in range(n):
        p0_hex[3*i+1] = (centers_init[i, 0] - r0) / (1 - 2*r0)
        p0_hex[3*i+2] = (centers_init[i, 1] - r0) / (1 - 2*r0)
        
    inits = [p0_hex]
    
    # Initialization 2: Random starts to escape local minima
    for _ in range(5):
        p0 = np.random.rand(3*n)
        p0[0::3] = np.random.uniform(0.03, 0.08, n)
        inits.append(p0)
        
    # Run optimization for each initialization
    for p0 in inits:
        res = minimize(objective, p0, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 5000, 'ftol': 1e-10})
        if res.fun < best_score:
            best_score = res.fun
            best_params = res.x.copy()
            
    # Extract best packing
    p_best = best_params.reshape(n, 3)
    r = p_best[:, 0].copy()
    sx = p_best[:, 1].copy()
    sy = p_best[:, 2].copy()
    x = r * (1 - 2*sx) + sx
    y = r * (1 - 2*sy) + sy
    centers = np.column_stack((x, y))
    
    # Safety check: if any overlap exceeds tolerance, scale radii down slightly
    # Shrinking radii strictly improves boundary satisfaction and overlap conditions
    max_overlap = 0.0
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt((x[i]-x[j])**2 + (y[i]-y[j])**2)
            ov = r[i] + r[j] - d
            if ov > max_overlap:
                max_overlap = ov
                
    if max_overlap > 1e-10:
        scale = 1.0 / (1.0 + max_overlap)
        r *= scale
        
    return centers, r, np.sum(r)
