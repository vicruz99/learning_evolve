import numpy as np
from scipy.optimize import minimize

N = 26

def compute_loss(vars, n_circles):
    """
    Objective function to minimize.
    Minimizes: -Sum(radii) + Penalty(overlaps, boundaries)
    
    Args:
        vars: 1D array of variables [x1, y1, r1, x2, y2, r2, ...]
        n_circles: Number of circles
    """
    centers = vars[:2 * n_circles].reshape(n_circles, 2)
    radii = vars[2 * n_circles:]
    
    # Objective: Maximize sum of radii (Minimize negative sum)
    loss = -np.sum(radii)
    
    # Penalty weight
    # High weight ensures constraints are respected
    penalty_weight = 10000.0
    penalty = 0.0
    
    # Boundary penalties
    # Circles must be inside [0, 1] x [0, 1]
    # Constraints: r <= x <= 1-r  =>  x - r >= 0  and  x + r <= 1
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        
        # Violation: x - r < 0
        if x - r < 0:
            penalty += (x - r)**2
        # Violation: x + r > 1
        if x + r > 1:
            penalty += (x + r - 1)**2
        # Violation: y - r < 0
        if y - r < 0:
            penalty += (y - r)**2
        # Violation: y + r > 1
        if y + r > 1:
            penalty += (y + r - 1)**2
            
    # Overlap penalties
    # Circles must not overlap: dist(i, j) >= r_i + r_j
    # Violation: r_i + r_j - dist > 0
    for i in range(n_circles):
        r_i = radii[i]
        c_i = centers[i]
        for j in range(i + 1, n_circles):
            r_j = radii[j]
            c_j = centers[j]
            dist = np.sqrt(np.sum((c_i - c_j)**2))
            overlap = r_i + r_j - dist
            if overlap > 0:
                penalty += overlap**2
                
    return loss + penalty_weight * penalty

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    # 1. Initialization
    # Start with a valid 5x5 grid configuration (25 circles)
    # and place the 26th circle in a gap.
    
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    
    idx = 0
    # 5x5 grid with radius 0.1 fits perfectly in [0,1]
    # Centers at 0.1, 0.3, 0.5, 0.7, 0.9
    for r in range(5):
        for c in range(5):
            centers[idx] = [0.1 + 0.2 * c, 0.1 + 0.2 * r]
            radii[idx] = 0.1
            idx += 1
    
    # Place 26th circle in the largest gap.
    # The grid creates holes at (0.2, 0.2), (0.4, 0.2), etc.
    # Distance from (0.2, 0.2) to nearest centers (0.1,0.1), (0.3,0.1)... is sqrt(0.02) approx 0.1414.
    # Available radius is approx 0.1414 - 0.1 = 0.0414.
    centers[idx] = [0.2, 0.2]
    radii[idx] = 0.04
    
    # Flatten variables for optimizer
    x0_init = np.concatenate([centers.flatten(), radii])
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    best_sol = None
    best_loss = np.inf
    
    # 2. Optimization Loop with Multi-start
    # Run multiple times with perturbations to escape local minima
    n_attempts = 10
    
    for k in range(n_attempts):
        # Create a perturbed copy of the initial solution
        if k == 0:
            x0 = x0_init.copy()
        else:
            # Perturb centers slightly
            noise = np.random.uniform(-0.01, 0.01, size=(N, 2))
            centers_pert = centers + noise
            # Clip to ensure centers are strictly inside [0, 1] with margin
            centers_pert = np.clip(centers_pert, 0.01, 0.99)
            
            # Reduce radii slightly to ensure validity after perturbation
            # This helps the optimizer start from a feasible state
            radii_pert = radii * 0.95 
            
            x0 = np.concatenate([centers_pert.flatten(), radii_pert])
        
        # Run L-BFGS-B optimizer
        # maxiter set high to allow convergence
        res = minimize(
            compute_loss, 
            x0, 
            args=(N,), 
            method='L-BFGS-B', 
            bounds=bounds, 
            options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        
        # Track best result
        # res.fun is the value of objective (loss)
        if res.fun < best_loss:
            best_loss = res.fun
            best_sol = res.x
            
    # 3. Extract and Validate Solution
    if best_sol is not None:
        final_centers = best_sol[:2 * N].reshape(N, 2)
        final_radii = best_sol[2 * N:]
    else:
        # Fallback to initial
        final_centers = centers
        final_radii = radii
        
    # Post-processing to ensure strict validity
    # The penalty method might leave tiny overlaps. 
    # We will shrink radii until valid.
    
    # Helper to check validity
    def is_valid(c, r):
        # Boundary check
        for i in range(N):
            x, y = c[i]
            rad = r[i]
            if x - rad < -1e-12 or x + rad > 1 + 1e-12 or \
               y - rad < -1e-12 or y + rad > 1 + 1e-12:
                return False
        # Overlap check
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.sqrt(np.sum((c[i] - c[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return False
        return True

    # If valid, return
    if is_valid(final_centers, final_radii):
        return final_centers, final_radii, np.sum(final_radii)

    shrink_factor = 1.0
    while not is_valid(final_centers, final_radii):
        shrink_factor *= 0.999
        final_radii = final_radii * shrink_factor
        # Safety break
        if shrink_factor < 0.5:
            break
            
    return final_centers, final_radii, np.sum(final_radii)
