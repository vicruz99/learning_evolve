import numpy as np
from scipy.optimize import minimize, linprog

def get_max_sum_radii(centers):
    """
    Given centers of 26 circles, solve the LP to find the maximum sum of radii
    such that circles do not overlap and stay within the unit square.
    
    Args:
        centers: np.array of shape (26, 2)
        
    Returns:
        optimal_sum: float
        optimal_radii: np.array of shape (26,)
    """
    n = 26
    c = np.ones(n) * -1.0  # Maximize sum of radii => Minimize -sum(r_i)
    
    # Prepare inequality constraints A_ub @ r <= b_ub
    # Constraints:
    # 1. r_i + r_j <= distance(centers[i], centers[j]) for all i < j
    # 2. r_i <= dist(centers[i], boundary)
    
    # Calculate pairwise distances
    # Using broadcasting for efficiency
    # centers shape (n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    dists = np.sqrt(np.sum(diff**2, axis=2)) # (n, n)
    
    # Prepare constraints
    # We need to construct A_ub and b_ub
    # Since n=26, number of pair constraints is 26*25/2 = 325
    # Number of wall constraints is 26
    # Total rows ~ 351
    
    # We can construct the matrix sparsely or just list constraints.
    # For linprog, we can pass A_ub as a matrix.
    
    # Let's build the matrix explicitly.
    # Number of constraints
    num_pair_constraints = n * (n - 1) // 2
    num_wall_constraints = n
    total_constraints = num_pair_constraints + num_wall_constraints
    
    A_ub = np.zeros((total_constraints, n))
    b_ub = np.zeros(total_constraints)
    
    # Pair constraints: r_i + r_j <= d_ij
    row_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dists[i, j]
            row_idx += 1
            
    # Wall constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        margin = min(x, 1 - x, y, 1 - y)
        # Ensure margin is not negative due to numerical errors, though centers should be in [0,1]
        # If center is outside, margin < 0, LP might fail or return 0. 
        # We assume valid centers.
        if margin < 0: margin = 0.0 
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = margin
        row_idx += 1
        
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # If LP fails, return 0 and zeros
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_func(centers_flat):
    """
    Objective function for optimizer.
    Minimizes negative sum of radii.
    """
    centers = centers_flat.reshape((26, 2))
    # Clip centers to [0, 1] to ensure validity during optimization steps
    # Although bounds are set, numerical drift can happen.
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_r, _ = get_max_sum_radii(centers)
    return -sum_r
