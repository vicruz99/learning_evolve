import numpy as np
from scipy.optimize import linprog
import itertools

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum_r = 0.0

    # 1. Initial Configuration: 5x5 grid
    # This provides a solid starting point for 25 circles, leaving space for the 26th
    for i in range(n):
        r = i // 5
        c = i % 5
        best_centers[i, 0] = 0.1 + c * 0.2
        best_centers[i, 1] = 0.1 + r * 0.2
        best_radii[i] = 0.09

    # 2. Helper function to solve for optimal radii given fixed centers
    def solve_radii(centers):
        n = centers.shape[0]
        c = np.zeros(n)  # Objective: Maximize sum(r) => Minimize -sum(r)
        c[:] = -1.0

        A_ub = []
        b_ub = []

        # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        # r - x <= 0 => 1*r_i + 0*others <= x
        # 0*r_i ... -1*r_i <= -x => -r_i <= -x => r_i >= x (Wait, constraint is r <= boundary)
        # Actually: r <= x  =>  r - x <= 0.
        
        # Format for linprog: A_ub @ r <= b_ub
        # Boundary: 1 * r_i <= x_i
        for i in range(n):
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(centers[i, 0])
            
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(1.0 - centers[i, 0])

            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(centers[i, 1])
            
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(1.0 - centers[i, 1])

        # Non-overlap constraints: r_i + r_j <= dist_ij
        # 1*r_i + 1*r_j <= dist_ij
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)

        # Bounds: r_i >= 0
        bounds = [(0, None) for _ in range(n)]

        # Solve LP
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                radii = res.x
                sum_r = np.sum(radii)
                return radii, sum_r
        except Exception:
            pass
        return None, 0.0

    # 3. Local Search with Simulated Annealing
    current_centers = best_centers.copy()
    current_radii, current_sum_r = solve_radii(current_centers)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum_r = current_sum_r

    temperature = 0.05
    decay = 0.995
    iterations = 2000

    for _ in range(iterations):
        # Pick a random circle to move
        idx = np.random.randint(n)
        direction = np.random.normal(0, 0.02, 2)
        new_center = current_centers[idx] + direction
        
        # Keep center inside [0.1, 0.9] initially to avoid boundary issues
        new_center = np.clip(new_center, 0.05, 0.95)
        
        test_centers = current_centers.copy()
        test_centers[idx] = new_center
        
        new_radii, new_sum_r = solve_radii(test_centers)
        
        if new_radii is not None and new_sum_r > best_sum_r - 1e-7:
            diff = new_sum_r - current_sum_r
            # Accept if better, or with probability based on temperature
            if diff > 0 or np.random.random() < np.exp(diff / temperature):
                current_centers = test_centers
                current_radii = new_radii
                current_sum_r = new_sum_r
                
                if current_sum_r > best_sum_r:
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
                    best_sum_r = current_sum_r

        temperature *= decay

    # 4. Final refinement with smaller steps
    for _ in range(500):
        idx = np.random.randint(n)
        direction = np.random.normal(0, 0.005, 2)
        new_center = current_centers[idx] + direction
        new_center = np.clip(new_center, 0.01, 0.99)
        
        test_centers = current_centers.copy()
        test_centers[idx] = new_center
        
        new_radii, new_sum_r = solve_radii(test_centers)
        
        if new_radii is not None and new_sum_r > best_sum_r - 1e-7:
            diff = new_sum_r - current_sum_r
            if diff > 0 or np.random.random() < np.exp(diff / 0.01):
                current_centers = test_centers
                current_radii = new_radii
                current_sum_r = new_sum_r
                
                if current_sum_r > best_sum_r:
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
                    best_sum_r = current_sum_r

    # Ensure no negative radii or NaNs
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, best_sum_r
