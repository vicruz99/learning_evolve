from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    """

    rng = np.random.RandomState(42)

    initial_centers = []

    dy = 0.23    # Vertical spacing

        x_start = (1.0 - row_width) / 2.0
        
        for i in range(count):
            x = x_start + i * 0.25
            # Add slight random jitter to avoid perfect symmetry traps
            x += rng.uniform(-0.01, 0.01)
            y = y_pos + rng.uniform(-0.01, 0.01)
            initial_centers.append([x, y])
        y_pos += dy

    initial_centers = np.array(initial_centers)
    initial_radii = np.full(n, 0.08) # Start with small radius to ensure feasibility

    # 2. Optimization Function
    def objective(vars_):
        # vars_ layout: [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(r), so we minimize -sum(r)
        r = vars_[2::3]
        return -np.sum(r)

    def constraints(vars_):
        centers = np.column_stack((vars_[0::3], vars_[1::3]))
        r = vars_[2::3]
        
        cons = []
        # Boundary constraints: x-r >= 0, 1-(x+r) >= 0, y-r >= 0, 1-(y+r) >= 0
        for i in range(n):
            x, y = centers[i]
            ri = r[i]
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[idx*3] - v[idx*3+2]})
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1 - (v[idx*3] + v[idx*3+2])})
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[idx*3+1] - v[idx*3+2]})
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1 - (v[idx*3+1] + v[idx*3+2])})
            
        # Non-overlap constraints: dist_ij >= ri + rj  =>  dist_ij - ri - rj >= 0
        for i in range(n):
            for j in range(i + 1, n):
                def overlap_constraint(v, i=i, j=j):
                    c_i = np.array([v[i*3], v[i*3+1]])
                    c_j = np.array([v[j*3], v[j*3+1]])
                    dist = np.sqrt(np.sum((c_i - c_j)**2))
                    return dist - v[i*3+2] - v[j*3+2]
                cons.append({'type': 'ineq', 'fun': overlap_constraint})

    best_sol = None

                        valid = False; break

                        best_sol = (c, r)
