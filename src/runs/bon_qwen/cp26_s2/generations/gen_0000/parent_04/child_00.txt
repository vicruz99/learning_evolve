import numpy as np
from scipy.optimize import minimize, linprog

def solve_lp(centers):
    """
    For a fixed set of centers, solves the LP to find the maximum sum of radii
    satisfying non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    # Objective: minimize -sum(r_i) -> maximize sum(r_i)
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= distance to each edge
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n); row[i] = 1.0
        A_ub.append(row); b_ub.append(x)
        # r_i <= 1 - x
        row = np.zeros(n); row[i] = 1.0
        A_ub.append(row); b_ub.append(1.0 - x)
        # r_i <= y
        row = np.zeros(n); row[i] = 1.0
        A_ub.append(row); b_ub.append(y)
        # r_i <= 1 - y
        row = np.zeros(n); row[i] = 1.0
        A_ub.append(row); b_ub.append(1.0 - y)
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                           (centers[i, 1] - centers[j, 1])**2)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None) for _ in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    else:
        # Fallback for infeasibility (e.g., centers on boundary causing r<=0 conflict)
        return np.zeros(n), 0.0

def objective(centers_flat):
    """Objective function for the outer optimizer."""
    centers = centers_flat.reshape(26, 2)
    _, sum_radii = solve_lp(centers)
    return -sum_radii

def run_packing():
    n = 26
    points = []
    r_init = 0.1
    
    # Generate initial hexagonal grid
    for i in range(7): 
        y = r_init + i * np.sqrt(3) * r_init
        if y + r_init > 1.0: break

        x = start_x
        while x + r_init <= 1.0 + 1e-9:
             points.append([x, y])
             x += 2 * r_init
        if len(points) >= n: break

    if len(points) < n:

                for p in points:
                    if np.hypot(p[0]-x, p[1]-y) < 0.05:
                        is_new = False; break
                if is_new: points.append([x, y])
            if len(points) >= n: break
            
    points = points[:n]
    centers0 = np.array(points).flatten()
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    
    best_res = None
    best_val = float('inf')
    
    # Run multiple trials with jitter to avoid local minima
    for trial in range(5):
        current_x0 = centers0.copy()

            current_x0 += np.random.uniform(-0.05, 0.05, size=current_x0.shape)
            current_x0 = np.clip(current_x0, 0, 1)

                       options={'maxiter': 2000, 'ftol': 1e-10})

            best_res = res
            
    final_centers = best_res.x.reshape(n, 2)

    return final_centers, final_radii, final_sum
