import numpy as np
import math
from scipy.optimize import minimize
import random

def compute_overlap(centers, radii):
    """
    Calculates the energy function for a packing of circles.
    Returns a scalar representing the sum of squared overlaps and boundary violations.
    """
    n = centers.shape[0]
    energy = 0.0
    
    # Pre-calculate radii to avoid repeated lookups
    radii_flat = radii.flat
    
    for i in range(n):
        r_i = radii_flat[i]
        c_i = centers[i]
        
        # Boundary violation checks
        if c_i[0] < r_i:
            energy += (c_i[0] - r_i) ** 2
        elif c_i[0] > 1 - r_i:
            energy += (c_i[0] - (1 - r_i)) ** 2
        if c_i[1] < r_i:
            energy += (c_i[1] - r_i) ** 2
        elif c_i[1] > 1 - r_i:
            energy += (c_i[1] - (1 - r_i)) ** 2
        
        # Inter-circle overlap checks
        for j in range(i + 1, n):
            r_j = radii_flat[j]
            c_j = centers[j]
            dx = c_i[0] - c_j[0]
            dy = c_i[1] - c_j[1]
            dist_sq = dx * dx + dy * dy
            sum_radii = r_i + r_j
            
            # Check for overlap using squared distance to avoid sqrt if possible, 
            # but for the energy function we need the distance difference.
            # dist = math.sqrt(dist_sq) 
            # We can approximate or just compute it.
            dist = math.sqrt(dist_sq)
            
            if dist < sum_radii:
                energy += (sum_radii - dist) ** 2
                
    return energy

def get_initializations(n):

    hex_grid = hex_grid[:26]
    inits.append(np.array(hex_grid))
