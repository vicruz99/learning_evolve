from scipy.optimize import minimize


def compute_violation(centers, radii, n):
    """Compute total constraint violation."""
    v = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v += max(0.0, r - x) ** 2 + max(0.0, r - (1.0 - x)) ** 2
        v += max(0.0, r - y) ** 2 + max(0.0, r - (1.0 - y)) ** 2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - d
            if overlap > 0:
                v += overlap ** 2
    return v


def objective_func(params, n, penalty_weight):
    """Objective: maximize sum of radii with penalty for violations."""
    centers = params[: 2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n :], 1e-10)
    obj = -np.sum(radii)
    violation = compute_violation(centers, radii, n)
    return obj + penalty_weight * violation


def gradient_func(params, n, penalty_weight):
    """Numerical gradient approximation."""
    grad = np.zeros_like(params)
    eps = 1e-7
    base = objective_func(params, n, penalty_weight)
    for k in range(len(params)):
        params_plus = params.copy()
        params_plus[k] += eps
        grad[k] = (objective_func(params_plus, n, penalty_weight) - base) / eps
    return grad

def force_relax(centers, radii, n, iterations=5000, learning_rate=0.005):
    """Force-directed relaxation to reduce overlaps."""
    centers = centers.copy()
    radii = radii.copy()
    
    for it in range(iterations):
        lr = learning_rate * (1.0 - it / iterations)
        lr = max(lr, 0.0001)

            # Growth force: try to grow radius

                        centers[i, 1], 1 - centers[i, 1])

                    continue

                    forces[i, 0] += dx * f
                    forces[i, 1] += dy * f

            radii[i] = min(radii[i] * 1.0002, max_r * 0.99)
