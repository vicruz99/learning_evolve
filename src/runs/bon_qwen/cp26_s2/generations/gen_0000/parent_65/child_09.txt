import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26

def _objective(vars):
    """Objective function: maximize t (minimize -t)"""
    return -vars[-1]

def _constraint_func(vars):
    """
    Constraint function returning an array of values that must be >= 0.
    Constraints:
    1. Boundary: x_i - t >= 0, 1 - x_i - t >= 0, y_i - t >= 0, 1 - y_i - t >= 0
    2. Overlap: dist_ij^2 - 4*t^2 >= 0
    """
    x = vars[0::2]
    y = vars[1::2]
    t = vars[-1]
    
    # Boundary constraints
    c_boundary = np.concatenate([x - t, 1 - x - t, y - t, 1 - y - t])
    
    # Overlap constraints (vectorized)
    # Compute squared distance matrix
    xx = x[:, None] - x[None, :]
    yy = y[:, None] - y[None, :]
    dist_sq = xx**2 + yy**2
    
    # Extract lower triangle (strict) to avoid duplicate and self-constraints
    mask = np.tril(np.ones((NUM_CIRCLES, NUM_CIRCLES), dtype=bool), k=-1)
    c_overlap = dist_sq[mask] - 4 * t**2
    
    return np.concatenate([c_boundary, c_overlap])

def _get_initial_guess(seed=0):
    """Generate a valid initial configuration using a hexagonal grid."""
    np.random.seed(seed)
    r0 = 0.08
    pts = []
    
    # Hexagonal packing pattern
    for i in range(15):
        for j in range(15):
            y = r0 + i * r0 * np.sqrt(3)
            x = r0 + j * 2 * r0 + (i % 2) * r0
            if 0 < x < 1 and 0 < y < 1:
                pts.append([x, y])
    
    # Fill remaining if needed
    while len(pts) < NUM_CIRCLES:
        px, py = np.random.uniform(r0, 1 - r0, 2)
        pts.append([px, py])
        
    # Shuffle to break symmetry and select required number
    np.random.shuffle(pts)
    pts = pts[:NUM_CIRCLES]
    
    # Assemble variables: [x1, y1, ..., x26, y26, t]
    vars_ = np.zeros(2 * NUM_CIRCLES + 1)
    vars_[0::2] = [p[0] for p in pts]
    vars_[1::2] = [p[1] for p in pts]
    vars_[-1] = r0
    return vars_

def run_packing():
    """
    Optimizes circle packing to maximize sum of radii.
    Returns centers, radii, and sum_radii.
    """
    # Variable bounds: x,y in [0,1], t in [0, 0.5]
    bounds = [(0, 1)] * (2 * NUM_CIRCLES) + [(0, 0.5)]
    constraints = {'type': 'ineq', 'fun': _constraint_func}
    
    best_t = 0
    best_vars = None
    
    # Multi-start optimization
    for seed in range(50):
        x0 = _get_initial_guess(seed)
        try:
            res = minimize(
                _objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds,
                constraints=constraints, 
                options={'maxiter': 3000, 'ftol': 1e-10, 'iprint': -1}
            )
            if res.success and res.x[-1] > best_t:
                best_t = res.x[-1]
                best_vars = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_vars is None:
        best_vars = _get_initial_guess(0)
        best_t = best_vars[-1]
        
    # Extract results
    centers = np.column_stack((best_vars[0::2], best_vars[1::2]))
    radii = np.full(NUM_CIRCLES, best_t)
    
    return centers, radii, 26 * best_t
