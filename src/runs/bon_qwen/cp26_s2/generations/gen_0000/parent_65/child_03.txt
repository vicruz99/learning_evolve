from scipy.optimize import minimize

def initialize_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08

    row_configs = [5, 5, 5, 5, 4, 2]
    row_spacing = 0.18
    col_spacing = 0.18
    offset = 0.09

    row_idx = 0
    circle_idx = 0

    for num_in_row in row_configs:
        for col in range(num_in_row):
            base_x = col * col_spacing + 0.5 * col_spacing + (row_idx % 2) * offset
            base_y = row_idx * row_spacing + 0.1

            centers[circle_idx, 0] = np.clip(base_x + np.random.uniform(-0.015, 0.015), 0.06, 0.94)
            centers[circle_idx, 1] = np.clip(base_y + np.random.uniform(-0.015, 0.015), 0.06, 0.94)
            circle_idx += 1
        row_idx += 1

    return centers, radii


def optimize_packing(n, centers, radii):
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))

    constraints = build_all_constraints(n)

    result = minimize(
        objective_func,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 15000, 'ftol': 1e-15, 'disp': False}
    )

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    radii_opt = np.maximum(radii_opt, 0)

    return centers_opt, radii_opt, np.sum(radii_opt)


def objective_func(x):
    n = len(x) // 3
    return -np.sum(x[2 * n:])


def build_all_constraints(n):
    constraints = []

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_x_low, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': constraint_x_high, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_low, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_high, 'args': (i, n)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap_sq, 'args': (i, j, n)})

    return constraints


def constraint_x_low(x, i, n):
    return x[2 * i] - x[2 * n + i]


def constraint_x_high(x, i, n):
    return 1.0 - x[2 * i] - x[2 * n + i]


def constraint_y_low(x, i, n):
    return x[2 * i + 1] - x[2 * n + i]


def constraint_y_high(x, i, n):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def constraint_overlap_sq(x, i, j, n):
    xi = x[2 * i]
    yi = x[2 * i + 1]
    xj = x[2 * j]
    yj = x[2 * j + 1]
    ri = x[2 * n + i]
    rj = x[2 * n + j]

    dx = xi - xj
    dy = yi - yj

    return dx * dx + dy * dy - (ri + rj) ** 2
