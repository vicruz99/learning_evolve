import numpy as np
import math
import scipy.optimize

def run_packing():
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Maximizes the sum of radii using force-directed relaxation followed by LP refinement.
    """
    n = 26
    np.random.seed(42)  # For reproducibility
    
    # 1. Initialization: Hexagonal Grid
    # Start with a reasonable radius that fits easily to ensure a valid starting configuration.
    r_init = 0.06
    row_height = math.sqrt(3) * r_init
    col_width = 2 * r_init
    
    centers_list = []
    y = r_init
    row_idx = 0
    
    # Generate hexagonal lattice points
    while y + r_init <= 1.0 + 1e-9 and len(centers_list) < n:
        x = r_init
        # Shift odd rows for hexagonal packing
        if row_idx % 2 == 1:
            x += col_width / 2.0
        
        # Fill the row
        while x + r_init <= 1.0 + 1e-9:
            if x - r_init >= -1e-9:
                centers_list.append([x, y])
            x += col_width
        y += row_height
        row_idx += 1

    # If not enough points generated, fill with random points inside the square
    while len(centers_list) < n:
        centers_list.append([np.random.rand() * 0.9 + 0.05, np.random.rand() * 0.9 + 0.05])
    
    # Take first n centers
    centers = np.array(centers_list[:n])
    radii = np.full(n, r_init)
    
    # 2. Force-Directed Optimization
    # Iteratively expand radii and resolve overlaps using repulsive forces.
    num_iterations = 5000
    learning_rate = 0.01
    expansion_rate = 0.0002
    
    for step in range(num_iterations):
        forces = np.zeros_like(centers)
        
        # Boundary repulsion forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Push away from left wall
            if x < r:
                forces[i, 0] += learning_rate * (r - x) * 10.0
            # Push away from right wall
            elif x > 1 - r:
                forces[i, 0] -= learning_rate * (x - (1 - r)) * 10.0
            # Push away from bottom wall
            if y < r:
                forces[i, 1] += learning_rate * (r - y) * 10.0
            # Push away from top wall
            elif y > 1 - r:
                forces[i, 1] -= learning_rate * (y - (1 - r)) * 10.0
        
        # Inter-circle repulsion forces
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                
                if dist_sq < 1e-12:
                    # If centers coincide, push apart randomly
                    dx = np.random.rand() - 0.5
                    dy = np.random.rand() - 0.5
                    dist = math.sqrt(dx*dx + dy*dy)

                    dist = math.sqrt(dist_sq)
                
                sum_r = radii[i] + radii[j]

                    overlap = sum_r - dist
