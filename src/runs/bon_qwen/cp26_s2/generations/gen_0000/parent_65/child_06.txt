import numpy as np
from scipy.optimize import minimize


def compute_penalty(x, n=26, penalty_weight=10000.0):
    """Compute the penalty-based objective function."""
    centers = x[:2 * n].reshape((n, 2))
    radii = np.maximum(0, x[2 * n:])

    penalty = 0.0

    # Boundary violations
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]

        if x_i - r_i < 0:
            penalty += (x_i - r_i) ** 2
        if x_i + r_i > 1:
            penalty += (x_i + r_i - 1) ** 2
        if y_i - r_i < 0:
            penalty += (y_i - r_i) ** 2
        if y_i + r_i > 1:
            penalty += (y_i + r_i - 1) ** 2

    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-15)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2

    return -np.sum(radii) + penalty_weight * penalty


def make_hexagonal_init(n=26, margin=0.06):
    """Create a hexagonal-like initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05

    row_counts = [6, 5, 6, 5, 4]
    n_rows = len(row_counts)

    y_positions = np.linspace(margin + 0.05, 1 - margin - 0.05, n_rows)

    idx = 0
    for row in range(n_rows):
        n_in_row = row_counts[row]
        x_positions = np.linspace(margin, 1 - margin, n_in_row)

        if row % 2 == 1:
            shift = (x_positions[1] - x_positions[0]) / 2.0 if n_in_row > 1 else 0
            x_positions = x_positions + shift
            x_positions = np.clip(x_positions, margin, 1 - margin)

        for col in range(n_in_row):
            if idx < n:
                centers[idx] = [x_positions[col], y_positions[row]]
                idx += 1

    return centers, radii


def make_square_init(n=26, margin=0.07):
    """Create a square-grid-like initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04

    grid_size = 5
    spacing = (1 - 2 * margin) / (grid_size - 1)

    idx = 0
    for i in range(grid_size):
        for j in range(grid_size):
            if idx < n:
                centers[idx] = [margin + i * spacing, margin + j * spacing]
                idx += 1

    if n > 25:
        centers[25] = [margin + spacing * 2, margin + spacing * 2]

    return centers, radii


def make_random_init(n=26, seed=42):
    """Create a random initial configuration."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.ones(n) * 0.035
    return centers, radii


def make_concentric_init(n=26):
    """Create a concentric-like initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04

    # Place circles in rings
    ring_configs = [
        (8, 0.35),   # 8 circles at radius 0.35
        (9, 0.25),   # 9 circles at radius 0.25
        (6, 0.15),   # 6 circles at radius 0.15
        (3, 0.08),   # 3 circles at radius 0.08
    ]

    idx = 0
    for n_in_ring, ring_r in ring_configs:
        angles = np.linspace(0, 2 * np.pi, n_in_ring, endpoint=False)
        for angle in angles:
            if idx < n:
                centers[idx] = [0.5 + ring_r * np.cos(angle), 0.5 + ring_r * np.sin(angle)]
                idx += 1

    return centers, radii


def make_staggered_init(n=26):

def validate_packing_internal(centers, radii, n=26):
    """Internal validation function."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True


def optimize_packing(x0, n=26, penalty_weight=10000.0, max_iter=8000):
    """Optimize a packing configuration."""
    result = minimize(
        compute_penalty,
        x0,
        args=(n, penalty_weight),
        method='L-BFGS-B',
        bounds=[(0, 1)] * (2 * n) + [(0, 0.5)] * n,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'eps': 1e-8},
    )
    return result
