import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """Minimize negative sum of radii."""
    # Radii are at indices 2, 5, 8, ...
    r_vals = vars[2::3]
    return -np.sum(r_vals)

def constraint_func(vars, n):
    """Compute constraint values >= 0."""
    pos = vars.reshape(n, 3)
    xs, ys, rs = pos[:, 0], pos[:, 1], pos[:, 2]
    
    num_constraints = 4 * n + n * (n - 1) // 2
    c = np.empty(num_constraints)
    idx = 0
    
    # Boundary constraints
    c[idx:idx+n] = xs - rs
    idx += n
    c[idx:idx+n] = 1.0 - xs - rs
    idx += n
    c[idx:idx+n] = ys - rs
    idx += n
    c[idx:idx+n] = 1.0 - ys - rs
    idx += n
    
    # Overlap constraints

            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dr = rs[i] + rs[j]
            c[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
            
    return c

def generate_initial_configs(n):

    configs = []
    
    # 1. Grid initialization
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)

            if idx < n:

            idx += 1

        x0 = np.hstack([init_centers.ravel(), init_radii])
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP',
