import numpy as np
import scipy.optimize as opt

def get_constraints(centers, radii):
    """
    Returns the list of constraint dictionaries for scipy.optimize.
    """
    n = centers.shape[0]
    constraints = []

    # Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1, r >= 0
    # x - r >= 0  => x - r >= 0
    # x + r <= 1  => 1 - (x + r) >= 0
    # y - r >= 0  => y - r >= 0
    # y + r <= 1  => 1 - (y + r) >= 0
    # r >= 0

    for i in range(n):
        # x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[2*i] - v[2*n + i]
        })
        # 1 - (x_i + r_i) >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - (v[2*i] + v[2*n + i])
        })
        # y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[2*i + 1] - v[2*n + i]
        })
        # 1 - (y_i + r_i) >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - (v[2*i + 1] + v[2*n + i])
        })
        # r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[2*n + i]
        })

    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: 
                    (v[2*i] - v[2*j])**2 + (v[2*i+1] - v[2*j+1])**2 - (v[2*n + i] + v[2*n + j])**2
            })
            
    return constraints
