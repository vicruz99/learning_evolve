The LP formulation:
- Maximize Σ r_i
- Subject to: r_i + r_j ≤ d_ij for all pairs (non-overlap)
- Subject to: r_i ≤ min(x_i, 1-x_i, y_i, 1-y_i) for all i (boundary)
- Subject to: r_i ≥ 0

- This is more effective than equal-radius packing because circles can adapt their sizes
- Corner circles can be larger, interior circles smaller
- The LP exactly captures the trade-off between circle sizes

import numpy as np
from scipy.optimize import linprog


def solve_radii_lp(centers, n=26):
    """Solve LP to find optimal radii for given fixed centers.
    
    Maximizes sum of radii subject to non-overlap and boundary constraints.
    """
    c = -np.ones(n)
    
    constraints = []
    rhs = []
    
    # Pairwise constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints.append(row)
            rhs.append(dist)
    
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        bound = min(
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1],
        )
        row = np.zeros(n)
        row[i] = 1.0
        constraints.append(row)
        rhs.append(max(1e-12, bound))
    
    A_ub = np.array(constraints)
    b_ub = np.array(rhs)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)


def hex_init(n=26):
    """Create hexagonal lattice initial placement."""
    pts = []
    r0 = 0.09
    row = 0
    while len(pts) < n:
        off = (row % 2) * r0
        for col in range(8):
            x = col * 2 * r0 + off + r0
            y = row * r0 * np.sqrt(3) + r0
            if x <= 1.0 - r0 and y <= 1.0 - r0:
                pts.append([x, y])
            if len(pts) == n:
                break
        row += 1
    return np.array(pts[:n])


def hex_init_denser(n=26):
    """Create denser hexagonal lattice initial placement."""
    pts = []
    r0 = 0.075
    row = 0
    while len(pts) < n:
        off = (row % 2) * r0
        for col in range(10):
            x = col * 2 * r0 + off + r0
            y = row * r0 * np.sqrt(3) + r0
            if x <= 1.0 - r0 and y <= 1.0 - r0:
                pts.append([x, y])
            if len(pts) == n:
                break
        row += 1
    return np.array(pts[:n])


def grid_init(n=26):
    """Create grid initial placement."""
    cols, rows = 6, 5
    pts = []
    for i in range(rows):
        for j in range(cols):
            if len(pts) < n:
                x = (j + 0.5) / cols
                y = (i + 0.5) / rows
                pts.append([x, y])
    return np.array(pts[:n])


def corners_init(n=26):
    """Create corner-focused initial placement."""
    pts = []
    # Place 4 larger circles near corners
    pts.append([0.18, 0.18])
    pts.append([0.82, 0.18])
    pts.append([0.18, 0.82])
    pts.append([0.82, 0.82])
    
    # Fill remaining with a hex-ish pattern in the center
    r0 = 0.08
    row = 0
    while len(pts) < n:
        off = (row % 2) * r0
        for col in range(6):
            x = 0.35 + col * 2 * r0 + off
            y = 0.35 + row * r0 * np.sqrt(3)
            if x <= 0.85 and y <= 0.85:
                pts.append([x, y])
                if len(pts) == n:
                    break
        row += 1
    return np.array(pts[:n])


def random_init(n=26, seed=42):
    """Create random initial placement."""
    rng = np.random.RandomState(seed)
    pts = []
    for _ in range(n):
        x = rng.uniform(0.08, 0.92)
        y = rng.uniform(0.08, 0.92)
        pts.append([x, y])
    return np.array(pts)


def stratified_random_init(n=26, seed=99):
    """Create stratified random initial placement."""
    rng = np.random.RandomState(seed)
    pts = []
    cols, rows = 6, 5
    for i in range(rows):
        for j in range(cols):
            if len(pts) < n:
                x = (j + 0.3) / cols + rng.uniform(-0.03, 0.03)
                y = (i + 0.3) / rows + rng.uniform(-0.03, 0.03)
                x = np.clip(x, 0.05, 0.95)
                y = np.clip(y, 0.05, 0.95)
                pts.append([x, y])
    return np.array(pts[:n])


def hill_climb(centers, n=26, max_iter=25000):
    """Hill climbing optimization with adaptive step size."""
    centers = np.copy(centers)
    current_sum, current_radii = solve_radii_lp(centers, n)
    
    best_sum = current_sum
    best_centers = centers.copy()
    best_radii = current_radii.copy()
    
    step_size = 0.02
    
    for iteration in range(max_iter):
        # Perturb a random circle
        i = np.random.randint(n)
        perturbation = np.random.randn(2) * step_size
        new_centers = centers.copy()
        new_centers[i] += perturbation
        new_centers[i] = np.clip(new_centers[i], 0.005, 0.995)
        
        new_sum, new_radii = solve_radii_lp(new_centers, n)
        
        if new_sum > current_sum:
            current_sum = new_sum
            current_radii = new_radii.copy()
            centers = new_centers
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = current_radii.copy()
        
        # Adaptive step size reduction
        if iteration > 0 and iteration % 3000 == 0:
            step_size *= 0.8
    
    return best_centers, best_radii, best_sum


def local_refine(centers, n=26):
    """Local refinement using coordinate-wise optimization."""
    centers = np.copy(centers)
    current_sum, current_radii = solve_radii_lp(centers, n)
    
    improved = True
    while improved:
        improved = False
        for i in range(n):
            best_local_sum = current_sum
            best_local_x = centers[i, 0]
            best_local_y = centers[i, 1]
            
            # Try moving in each direction
            for dx in [-0.01, -0.005, -0.001, 0.001, 0.005, 0.01]:
                for dy in [-0.01, -0.005, -0.001, 0.001, 0.005, 0.01]:
                    test_centers = centers.copy()
                    test_centers[i, 0] = np.clip(centers[i, 0] + dx, 0.005, 0.995)
                    test_centers[i, 1] = np.clip(centers[i, 1] + dy, 0.005, 0.995)
                    
                    test_sum, _ = solve_radii_lp(test_centers, n)
                    
                    if test_sum > best_local_sum + 1e-10:
                        best_local_sum = test_sum
                        best_local_x = test_centers[i, 0]
                        best_local_y = test_centers[i, 1]
            
            if best_local_sum > current_sum + 1e-10:
                centers[i, 0] = best_local_x
                centers[i, 1] = best_local_y
                current_sum, current_radii = solve_radii_lp(centers, n)
                improved = True
    
    return centers, current_radii, current_sum


def run_packing():
