import numpy as np
from scipy.optimize import minimize

def constraint_func(vars, n):
    """
    Evaluates inequality constraints for the packing problem.
    Returns an array of values that must be >= 0.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # 1. Boundary constraints: 0 <= x,y <= 1 and r >= 0 implies:
    # x - r >= 0
    # y - r >= 0
    # 1 - x - r >= 0
    # 1 - y - r >= 0
    b1 = centers[:, 0] - radii
    b2 = centers[:, 1] - radii
    b3 = 1.0 - centers[:, 0] - radii
    b4 = 1.0 - centers[:, 1] - radii
    
    # 2. Pairwise non-overlap constraints: dist(i,j) - r_i - r_j >= 0
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-12)
    
    # Extract upper triangle indices for i < j
    i_idx, j_idx = np.triu_indices(n, k=1)
    pair_cons = dists[i_idx, j_idx] - radii[i_idx] - radii[j_idx]
    
    # Concatenate all constraints
    return np.concatenate([b1, b2, b3, b4, pair_cons])

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    r_init = 0.085  # Initial radius guess
    
    # Generate a hexagonal lattice initial configuration
    centers = []
    rows = 6
    cols = 5
    dx = 2.0 * r_init
    dy = np.sqrt(3) * r_init
    
    y_curr = r_init
    for r in range(rows):
        x_curr = r_init
        if r % 2 == 1:
            x_curr += dx / 2.0
        for c in range(cols):
            if len(centers) >= n:
                break
            centers.append([x_curr, y_curr])
            x_curr += dx
        y_curr += dy
        if len(centers) >= n:
            break
            
    centers = np.array(centers[:n])
    radii = np.ones(n) * r_init
    
    # Flatten variables for optimizer: [x1, y1, ..., xn, yn, r1, ..., rn]
    x0 = np.hstack([centers.flatten(), radii])
    
    # Define bounds for each variable
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    # Setup constraints
    cons = {'type': 'ineq', 'fun': lambda x: constraint_func(x, n)}
        
    # Objective: maximize sum(radii) => minimize -sum(radii)
    def objective(x):
        return -np.sum(x[2::3])
        
    # Run optimization
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={
            'maxiter': 3000, 
            'ftol': 1e-12, 
            'disp': False
        }
    )
    
    # Extract results
    x_opt = res.x
    centers_opt = x_opt[:2*n].reshape(n, 2)
    radii_opt = x_opt[2*n:]
    
    # Ensure numerical feasibility
    radii_opt = np.maximum(radii_opt, 0.0)
    centers_opt[:, 0] = np.clip(centers_opt[:, 0], radii_opt, 1.0 - radii_opt)
    centers_opt[:, 1] = np.clip(centers_opt[:, 1], radii_opt, 1.0 - radii_opt)
    
    total_sum = np.sum(radii_opt)
    
    return centers_opt, radii_opt, total_sum
