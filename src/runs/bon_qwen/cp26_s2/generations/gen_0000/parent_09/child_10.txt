To maximize the sum of radii for 26 circles in a unit square, we aim for a dense packing configuration, ideally approaching hexagonal lattice density. Since the objective is to maximize $\sum r_i$ rather than the minimum radius, uniform radii are generally near-optimal due to the convexity of area constraints.

Strategy:
1. **Penalty-Based Optimization**: We formulate the problem as an unconstrained optimization over centers and radii, using a penalty function for boundary and overlap constraints. This avoids the complexity of handling hard constraints in non-convex solvers.
2. **Loss Function**: The loss combines $-\sum r_i$ (to be minimized) with a quadratic penalty for constraint violations. A high penalty weight ensures solutions respect the geometric limits.
3. **Multi-Start Initialization**: We initialize the optimizer with multiple configurations (hexagonal grid, uniform grid, and random) to escape local minima. L-BFGS-B is used for efficient bound-constrained optimization.
4. **Validation & Correction**: After optimization, a post-processing step slightly shrinks radii if any microscopic violations remain due to numerical precision, ensuring the final output strictly passes the validation function.
5. **Vectorization**: Distance and overlap calculations are vectorized for performance and stability.

import numpy as np
from scipy.optimize import minimize

def compute_loss(params, n=26, penalty_weight=20000.0):
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: maximize sum of radii => minimize negative sum
    loss = -np.sum(radii)
    p = 0.0
    
    # Boundary constraints
    # x >= r  => r - x <= 0
    viol = np.maximum(radii - centers[:, 0], 0)
    p += np.sum(viol**2)
    # x <= 1-r => x+r <= 1
    viol = np.maximum(radii + centers[:, 0] - 1, 0)
    p += np.sum(viol**2)
    # y >= r
    viol = np.maximum(radii - centers[:, 1], 0)
    p += np.sum(viol**2)
    # y <= 1-r
    viol = np.maximum(radii + centers[:, 1] - 1, 0)
    p += np.sum(viol**2)
    
    # Inter-circle constraints (vectorized)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Upper triangle indices to check each pair once
    i, j = np.triu_indices(n, k=1)
    overlaps = radii[i] + radii[j] - dists[i, j]
    p += np.sum(np.maximum(overlaps, 0)**2)
        
    return loss + penalty_weight * p

def run_packing():
    n = 26
    best_loss = float('inf')
    best_params = None
    
    # Generate initial configurations
    configs = []
    
    # 1. Hexagonal lattice pattern
    hex_cfg = []
    y_step = 0.17
    x_step = 0.18
    row = 0
    while len(hex_cfg) < 30:
        y_c = 0.1 + row * y_step
        if y_c > 0.9:
            row += 1
            continue
        col = 0
        x_start = 0.1 if row % 2 == 0 else 0.1 + x_step / 2
        while True:
            x_c = x_start + col * x_step
            if x_c > 0.9:
                break
            hex_cfg.append([x_c, y_c])
            col += 1
        row += 1
    np.random.shuffle(hex_cfg)
    configs.append(np.array(hex_cfg[:n]))
    
    # 2. Uniform Grid pattern
    x_grid = np.linspace(0.08, 0.92, 6)
    y_grid = np.linspace(0.08, 0.92, 5)
    grid_cfg = []
    for xi in x_grid:
        for yi in y_grid:
            grid_cfg.append([xi, yi])
    np.random.shuffle(grid_cfg)
    configs.append(np.array(grid_cfg[:n]))
    
    # 3. Multiple random restarts
    for _ in range(5):
        configs.append(np.random.rand(n, 2))
    
    # Optimization loop
    for cfg in configs:
        radii_init = np.full(n, 0.03)
        # Add small jitter to radii to break symmetry
        radii_init += np.random.uniform(-0.001, 0.001, n)
        radii_init = np.clip(radii_init, 0.01, 0.1)
        
        params = np.concatenate([cfg.ravel(), radii_init])
        
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        res = minimize(compute_loss, params, args=(n, 20000.0), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-9})
        
        if res.fun < best_loss:
            best_loss = res.fun
            best_params = res.x.copy()
            
    if best_params is None:
        centers = np.random.rand(n, 2)
        radii = np.full(n, 0.01)
        return centers, radii, np.sum(radii)
        
    centers = best_params[:2*n].reshape(n, 2)
    radii = best_params[2*n:]
    
    # Validation and safety correction
    # Shrink radii slightly if numerical violations exist
    for _ in range(1000):
        valid = True
        
        # Check boundaries
        for i in range(n):
            for dim in range(2):
                if centers[i, dim] - radii[i] < -1e-11:
                    valid = False
                if centers[i, dim] + radii[i] > 1 + 1e-11:
                    valid = False
                    
        if valid:
            # Check overlaps
            i, j = np.triu_indices(n, k=1)
            diffs = centers[i] - centers[j]
            dists = np.sqrt(np.sum(diffs**2, axis=1))
            overlaps = radii[i] + radii[j] - dists
            if np.any(overlaps > 1e-11):
                valid = False
                
        if valid:
            break
            
        # Gradually shrink radii to enforce validity
        radii *= 0.99995
        
    return centers, radii, np.sum(radii)
