    Overlaps and boundary violations are penalized.

    """
    n = centers.shape[0]

    # Boundary violations
    # x - r >= 0, x + r <= 1 => r <= x <= 1-r

        if y - r < 0:
            violations += penalty_weight * (-(y - r))**2

    return violations
