import math

# These functions must take the variable vector v and return the constraint value.
# Constraints are of the form fun(v) >= 0.

def constraint_boundary_x(v, i):
    """Constraint: x_i >= r_i  =>  x_i - r_i >= 0"""
    return v[3 * i] - v[3 * i + 2]

def constraint_boundary_x_max(v, i):
    """Constraint: 1 - x_i >= r_i  =>  1 - x_i - r_i >= 0"""
    return 1.0 - v[3 * i] - v[3 * i + 2]

def constraint_boundary_y(v, i):
    """Constraint: y_i >= r_i  =>  y_i - r_i >= 0"""
    return v[3 * i + 1] - v[3 * i + 2]

def constraint_boundary_y_max(v, i):
    """Constraint: 1 - y_i >= r_i  =>  1 - y_i - r_i >= 0"""
    return 1.0 - v[3 * i + 1] - v[3 * i + 2]

def constraint_overlap_sq(v, i, j):
    """Constraint: dist(i, j)^2 >= (r_i + r_j)^2
       Equivalent to: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
       Using squared distance avoids square root singularities and is smooth.

    dy = v[3 * i + 1] - v[3 * j + 1]
    r_sum = v[3 * i + 2] + v[3 * j + 2]

    # Generate points within the square
    while y + r_guess <= 1.0:
        # Determine x start based on row parity (shifted rows for hex packing)
        if row % 2 == 0:
            x_start = r_guess
        else:
            x_start = r_guess + r_guess  # Shift by r
        
        x = x_start
        while x + r_guess <= 1.0:
            points.append([x, y])
            x += dx
        
        y += dy
        row += 1
        
    # Select N_CIRCLES points that are spread out
    selected_points = select_best_points(points, N_CIRCLES)
    return selected_points

# --- Main Packing Function ---

def run_packing():
    # 1. Generate initial centers
    init_centers = generate_initial_configuration()
    
    # 2. Prepare initial vector for optimizer
    # Vector layout: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    # Total size: 3 * 26 = 78
    x0 = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0[3 * i] = init_centers[i][0]
        x0[3 * i + 1] = init_centers[i][1]
        x0[3 * i + 2] = 0.05  # Initial small radius to satisfy constraints
    
    # 3. Define bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES)
    bounds += [(0.0, 0.5)] * N_CIRCLES
    
    # 4. Define constraints
    constraints = []
    
    # Boundary constraints for each circle
    for i in range(N_CIRCLES):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_x, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_x_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_y, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_y_max, 'args': (i,)})
        
    # Overlap constraints for each pair
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap_sq, 'args': (i, j)})
            
    # 5. Define objective function
    # We want to maximize sum of radii, so we minimize negative sum
    def objective(v):
        total_r = 0.0
        for i in range(N_CIRCLES):
            total_r += v[3 * i + 2]
        return -total_r

    # 6. Run optimization
    # Using SLSQP method which handles non-linear constraints
    options = {
        'ftol': 1e-10,
        'maxiter': 5000,
        'disp': False
    }
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options=options)
    
    # 7. Extract results
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.zeros(N_CIRCLES)
    
    for i in range(N_CIRCLES):
        centers[i] = [result.x[3 * i], result.x[3 * i + 1]]
        radii[i] = result.x[3 * i + 2]
        
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
