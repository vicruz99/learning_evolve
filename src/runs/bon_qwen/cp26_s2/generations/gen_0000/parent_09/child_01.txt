import numpy as np


def compute_effective_radii(centers, radii):
    """Compute the maximum feasible radii given current positions."""
    n = centers.shape[0]
    new_radii = np.copy(radii)
    
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0],
                    centers[i, 1], 1 - centers[i, 1])
        for j in range(n):
            if j != i:
                dist = np.linalg.norm(centers[i] - centers[j])
                max_r = min(max_r, dist - new_radii[j])
        new_radii[i] = max(0, max_r)
    
    return new_radii
