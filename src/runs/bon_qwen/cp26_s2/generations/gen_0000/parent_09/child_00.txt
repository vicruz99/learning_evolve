import numpy as np
from scipy.optimize import minimize

def compute_loss(v, n, penalty):
    """
    Computes the objective function: -sum(radii) + penalty * constraint_violations
    """
    cx = v[:2*n].reshape((n, 2))
    r = v[2*n:]
    
    loss = 0.0
    # Boundary constraints penalties
    loss += np.sum(np.maximum(0.0, r - cx[:, 0])**2)
    loss += np.sum(np.maximum(0.0, cx[:, 0] + r - 1.0)**2)
    loss += np.sum(np.maximum(0.0, r - cx[:, 1])**2)
    loss += np.sum(np.maximum(0.0, cx[:, 1] + r - 1.0)**2)
    
    # Overlap constraints penalties
    for i in range(n):
        for j in range(i + 1, n):
            diff = cx[i] - cx[j]
            dist = np.sqrt(np.sum(diff**2))
            violation = r[i] + r[j] - dist
            if violation > 0.0:
                loss += violation**2
                
    return -np.sum(r) + penalty * loss

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Hexagonal initialization for high density

    r = res.x[2*n:]

    for _ in range(30):

    return cx, r, np.sum(r)
