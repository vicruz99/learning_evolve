To maximize the sum of radii for 26 circles in a unit square, we need to solve a non-linear optimization problem. The constraints are boundary conditions ($x \pm r \in [0,1], y \pm r \in [0,1]$) and non-overlap ($\|c_i - c_j\| \ge r_i + r_j$).

Strategy:
1. **Initialization**: Generate multiple diverse initial configurations to avoid poor local minima. These will include:
   - A $5 \times 5$ grid layout (which naturally accommodates 25 circles) with one extra circle placed optimally.
   - A hexagonal lattice arrangement, which is denser than square packing.
   - Random perturbations to explore the space.
2. **Optimization Method**: Use `scipy.optimize.minimize` with the `L-BFGS-B` algorithm. It handles bounds efficiently and is robust.
3. **Constraint Handling**: Instead of passing hundreds of constraints to the solver, we will use a penalty method. We add a large penalty term to the objective function for any constraint violation (boundary overflow or circle overlap). The penalty coefficient is increased iteratively to enforce strict feasibility.
4. **Refinement**: Run the optimizer from each initialization, progressively increasing the penalty weight. The best valid configuration (highest sum of radii) will be selected.
5. **Validation**: Ensure the final result strictly satisfies all constraints before returning.

```python
import numpy as np
from scipy.optimize import minimize
import math

# Global constant for number of circles
N_CIRCLES = 26

def compute_objective_and_penalty(vars_flat, lam):
    """
    Computes the objective (negative sum of radii) plus a penalty for constraint violations.
    vars_flat: numpy array of shape (3 * N_CIRCLES)
               [x0, y0, r0, x1, y1, r1, ...]
    lam: penalty weight
    """
    n = N_CIRCLES
    centers = vars_flat[:2 * n].reshape(n, 2)
    radii = vars_flat[2 * n:]
    
    # Objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary constraints
    # x - r >= 0
    v = radii - centers[:, 0]
    penalty += np.sum(np.maximum(0.0, v) ** 2)
    # x + r <= 1
    v = centers[:, 0] + radii - 1.0
    penalty += np.sum(np.maximum(0.0, v) ** 2)
    # y - r >= 0
    v = radii - centers[:, 1]
    penalty += np.sum(np.maximum(0.0, v) ** 2)
    # y + r <= 1
    v = centers[:, 1] + radii - 1.0
    penalty += np.sum(np.maximum(0.0, v) ** 2)
    
    # Overlap constraints
    # Compute pairwise distances efficiently
    # centers shape: (n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff ** 2, axis=2)
    dist = np.sqrt(np.maximum(dist_sq, 1e-12))
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    v = r_sum - dist
    
    # Only consider upper triangle (i < j)
    mask = np.triu(np.ones((n, n)), k=1)
    v = v * mask
    penalty += np.sum(np.maximum(0.0, v) ** 2)
    
    return obj + lam * penalty

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    dim = 3 * n
    
    # Prepare initial configurations
    inits = []
    
    # 1. 5x5 Grid + 1 extra circle
    grid_coords = np.linspace(0.1, 0.9, 5)
    grid_pts = np.array(np.meshgrid(grid_coords, grid_coords)).T.reshape(-1, 2)
    init1 = np.zeros(dim)
    init1[:2*n-2] = grid_pts.flatten() # First 25
    init1[2*n-2] = 0.5; init1[2*n-1] = 0.05 # 26th center
    init1[2*n:] = 0.05 # Initial radii
    init1[-1] = 0.02 # Smaller radius for 26th to start feasible
    inits.append(init1)
    
    # 2. Hexagonal-like arrangement
    hex_pts = []
    r_est = 0.08
    row_lengths = [5, 6, 5, 6, 4]
    y_curr = r_est
    for rl in row_lengths:
        x_start = (1.0 - rl * 2.0 * r_est) / 2.0
        for k in range(rl):
            hex_pts.append([x_start + k * 2.0 * r_est, y_curr])
        y_curr += math.sqrt(3) * r_est
    while len(hex_pts) < n:
        hex_pts.append([0.5, 0.5])
    hex_arr = np.array(hex_pts[:n])
    init2 = np.hstack([hex_arr.flatten(), np.full(n, r_est)])
    inits.append(init2)
    
    # 3. Random initialization
    init3 = np.random.rand(dim)
    init3[2*n:] = np.random.uniform(0.02, 0.08, n)
    inits.append(init3)
    
    best_sum = 0.0
    best_vars = None
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.25)] * n
    
    # Optimization loop with increasing penalty
    penalty_weights = [100.0, 500.0, 2000.0, 8000.0]
    
    for idx, init in enumerate(inits):
        # Project initial radii to be strictly inside boundaries
        for i in range(n):
            x, y, r = init[i*3], init[i*3+1], init[i*3+2]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            init[i*3+2] = min(r, max_r * 0.95)
            
        curr_vars = init.copy()
        
        for lam in penalty_weights:
            # Define closure-free objective for this lam
            def obj_func(v):
                return compute_objective_and_penalty(v, lam)
                
            res = minimize(
                obj_func,
                curr_vars,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-6}
            )
            curr_vars = res.x
            
        # Evaluate final result
        c_curr = curr_vars[:2*n].reshape(n, 2)
        r_curr = curr_vars[2*n:]
        curr_sum = np.sum(r_curr)
        
        # Strict validation
        valid = True
        for i in range(n):
            if r_curr[i] < 0:
                valid = False; break
            if c_curr[i,0] < r_curr[i] or c_curr[i,0] > 1.0 - r_curr[i] or \
               c_curr[i,1] < r_curr[i] or c_curr[i,1] > 1.0 - r_curr[i]:
                valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((c_curr[i] - c_curr[j]) ** 2))
                    if dist < r_curr[i] + r_curr[j] - 1e-9:
                        valid = False; break
                if not valid: break
                
        if valid and curr_sum > best_sum:
            best_sum = curr_sum
            best_vars = curr_vars.copy()
            
    # Fallback if optimization failed
    if best_vars is None:
        best_vars = inits[0].copy()
        best_sum = np.sum(best_vars[2*n:])
        
    final_centers = best_vars[:2*n].reshape(n, 2)
    final_radii = best_vars[2*n:]
    
    return final_centers, final_radii, best_sum
```
