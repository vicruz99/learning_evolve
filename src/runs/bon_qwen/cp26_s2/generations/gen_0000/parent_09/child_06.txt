import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26

    # 1. Initialization
    # Place circles in a 6x5 grid to ensure a good initial spread
    cols = 5
    rows = 6
    # Use margins slightly larger than initial radius to ensure valid start
    dx = 1.0 / (cols + 1)
    dy = 1.0 / (rows + 1)
    
    centers = []
    count = 0
    for r in range(rows):
        for c in range(cols):
            if count < n:
                centers.append([ (c + 1) * dx, (r + 1) * dy ])
                count += 1
    
    # Convert to numpy array and flatten for the optimizer
    # Structure: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i, pos in enumerate(centers):
        x0[3*i] = pos[0]
        x0[3*i+1] = pos[1]
        x0[3*i+2] = 0.05  # Start with small radii

    # Bounds for variables
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # 2. Penalty-based Objective Function
    def objective(vars, penalty_weight):
        cost = 0.0
        
        # Penalty for boundary violations: max(0, dist - r)^2
        for i in range(n):
            x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
            # Left, Right, Bottom, Top
            cost += penalty_weight * max(0, r - x)**2
            cost += penalty_weight * max(0, r - (1.0 - x))**2
            cost += penalty_weight * max(0, r - y)**2
            cost += penalty_weight * max(0, r - (1.0 - y))**2

        # Penalty for overlaps: max(0, (r1+r2) - dist)^2
        for i in range(n):
            x1, y1, r1 = vars[3*i], vars[3*i+1], vars[3*i+2]
            for j in range(i + 1, n):
                x2, y2, r2 = vars[3*j], vars[3*j+1], vars[3*j+2]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                overlap = (r1 + r2) - dist
                if overlap > 0:
                    cost += penalty_weight * overlap**2

        # Maximize sum of radii -> minimize negative sum
        return -np.sum(vars[2::3]) + cost

    # 3. Iterative Optimization
    current_vars = x0.copy()
    p = 100.0  # Initial penalty weight

    for _ in range(20):
        # Run SLSQP with current penalty
        res = minimize(
            objective, current_vars, 
            args=(p,),
            method='SLSQP', 
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
        )
        current_vars = res.x
        p *= 2.0 # Increase penalty to enforce stricter constraints

    # 4. Extract and Return Results
    centers_out = np.zeros((n, 2))
    radii_out = np.zeros(n)
    for i in range(n):
        centers_out[i, 0] = current_vars[3*i]
        centers_out[i, 1] = current_vars[3*i+1]
        radii_out[i] = current_vars[3*i+2]

    sum_radii = np.sum(radii_out)
    return centers_out, radii_out, sum_radii
