def penalty(centers, radii):
    """
    Calculates the penalty for overlapping circles and circles outside the unit square.
    Uses a soft constraint approach with quadratic penalties.
    """
    n = len(radii)
    pen = 0.0
    
    # Boundary penalties
    # Centers must be in [r, 1-r]. 
    # Penalty if x < r or x > 1-r (similarly for y)
    # We use a slack of 0 to make it strict at the limit, or a small epsilon for numerical stability.
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left/Right boundaries
        if x - r < 0:
            pen += 100.0 * (x - r)**2

            pen += 100.0 * (x + r - 1)**2

        if y - r < 0:
            pen += 100.0 * (y - r)**2
        if y + r > 1:
            pen += 100.0 * (y + r - 1)**2

    # Overlap penalties
    for i in range(n):

                pen += 1000.0 * overlap**2
                
    return pen
