import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    best_result = None
    best_score = -1.0

    initial_configs = []

    # 1. Hexagonal Packings
    # Rows of 6, 5, 6, 5, 4
    configs = []

    for rows in [row_counts, list(reversed(row_counts))]:
        centers = []
        r_est = 0.09 
        y_curr = r_est
        for r_idx, count in enumerate(rows):
            x_spacing = (1 - 2 * r_est) / (count - 1) if count > 1 else 0
            x_start = r_est + x_spacing * (0.5 if r_idx % 2 == 1 else 0)
            for c_idx in range(count):
                x = x_start + c_idx * x_spacing
                centers.append([x, y_curr])
            y_curr += r_est * 1.732
        initial_configs.append(centers)

    # 2. Square Grid Perturbations
    grid = np.linspace(0.1, 0.9, 5)

                    count += 1

                if count >= n: break
            initial_configs.append(centers)

    # 3. Random Starts
    for _ in range(5):
        centers = np.random.uniform(0.2, 0.8, (n, 2))
        initial_configs.append(centers)

    # Constraint Definitions for SciPy
    def obj_func(vars):
        # vars: [x1, y1, r1, ..., x26, y26, r26]
        # Maximize sum(r) => Minimize -sum(r)
        r = vars[2::3]
        return -np.sum(r)

    def constraint_boundary(vars):
        # r <= x, r <= 1-x, r <= y, r <= 1-y
        cons = []
        for i in range(n):
            x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
            cons.append(x - r)       # r <= x
            cons.append((1 - x) - r) # r <= 1-x
            cons.append(y - r)       # r <= y
            cons.append((1 - y) - r) # r <= 1-y
        return np.array(cons)

    def constraint_overlap(vars):
        # dist >= r1 + r2  =>  dist - r1 - r2 >= 0
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi = vars[3*i], vars[3*i+1]
                xj, yj = vars[3*j], vars[3*j+1]
                ri, rj = vars[3*i+2], vars[3*j+2]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                cons.append(dist - ri - rj)
        return np.array(cons)

    # Optimization Loop
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 1) for _ in range(n)]

    for config in initial_configs:
        # Ensure config is float array and shape (n, 2)

            continue

        r_init = np.full(n, 0.09)

                obj_func, 
                x0, 
                method='SLSQP', 
                bounds=bounds,
                constraints=[
                    {'type': 'ineq', 'fun': constraint_boundary},
                    {'type': 'ineq', 'fun': constraint_overlap}

                options={'ftol': 1e-9, 'maxiter': 1000}

                                    break

                        pass
