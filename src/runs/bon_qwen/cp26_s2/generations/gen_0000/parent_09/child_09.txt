import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def objective(params, n):
    # Objective: minimize negative sum of radii
    r = params[2::3]
    return -np.sum(r)

def constraints_list(params, n):

    centers = np.empty((n, 2))
    radii = np.empty(n)

        # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        c.append({'type': 'ineq', 'fun': lambda p, i=i: p[3*i+2] - p[3*i], 'jac': _jac_bound}) # r - x <= 0 -> x - r >= 0?
        # SLSQP ineq: fun >= 0.
        # x - r >= 0 -> p[3*i] - p[3*i+2] >= 0
        c.pop() # Remove wrong one
        c.append({'type': 'ineq', 'fun': lambda p, i=i: p[3*i] - p[3*i+2]}) 
        c.append({'type': 'ineq', 'fun': lambda p, i=i: (1 - p[3*i]) - p[3*i+2]})
        c.append({'type': 'ineq', 'fun': lambda p, i=i: p[3*i+1] - p[3*i+2]})
        c.append({'type': 'ineq', 'fun': lambda p, i=i: (1 - p[3*i+1]) - p[3*i+2]})

    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            def overlap_fn(p, i=i, j=j):
                dx = p[3*i] - p[3*j]
                dy = p[3*i+1] - p[3*j+1]
                sum_r = p[3*i+2] + p[3*j+2]
                return dx*dx + dy*dy - sum_r*sum_r
            c.append({'type': 'ineq', 'fun': overlap_fn})
            
    return c

def _jac_bound(p, i):
    # Not strictly needed if not provided, but for speed
    jac = np.zeros_like(p)
    jac[3*i] = 1
    jac[3*i+2] = -1
    return jac

            max_r = min(x, 1-x, y, 1-y)
