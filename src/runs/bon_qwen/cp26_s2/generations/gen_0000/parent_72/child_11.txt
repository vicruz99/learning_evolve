import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # Objective: maximize sum of radii -> minimize negative sum
    def objective(x):
        # x layout: [x0, y0, r0, x1, y1, r1, ...]
        radii = x[2::3]
        return -np.sum(radii)

    # Bounds: 0 <= x, y <= 1, 0 <= r <= 0.5
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # Constraints function
    def get_constraints(x):
        cx = x[0::3]
        cy = x[1::3]
        cr = x[2::3]
        
        cons = []
        
        # Boundary constraints
        # x - r >= 0
        cons.append(cx - cr)
        # 1 - x - r >= 0
        cons.append(1.0 - cx - cr)
        # y - r >= 0
        cons.append(cy - cr)
        # 1 - y - r >= 0
        cons.append(1.0 - cy - cr)
        
        # Overlap constraints
        # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
        dx = cx[:, np.newaxis] - cx[np.newaxis, :]
        dy = cy[:, np.newaxis] - cy[np.newaxis, :]
        dr = cr[:, np.newaxis] + cr[np.newaxis, :]
        
        dist_sq = dx**2 + dy**2
        req_sq = dr**2
        
        # We only need to check unique pairs (i < j)
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        diffs = dist_sq[mask] - req_sq[mask]
        cons.append(diffs)
        
        return np.concatenate(cons)

    best_sum_r = -1.0
    best_centers = None
    best_radii = None

    # Multiple restarts to find global optimum
    n_restarts = 15
    
    for restart in range(n_restarts):
        # Initialize centers
        if restart % 3 == 0:
            # Hexagonal grid initialization
            centers = []
            y = 0.1
            row = 0
            count = 0
            while count < n:
                x = 0.1 + (row % 2) * 0.15 # Shift every other row
                while x <= 0.9 and count < n:
                    centers.append([x, y])
                    x += 0.3
                    count += 1
                y += 0.26
                row += 1
            cx_init = np.array([c[0] for c in centers])
            cy_init = np.array([c[1] for c in centers])
        else:
            cx_init = np.random.rand(n)
            cy_init = np.random.rand(n)

        # Grow radii to find a dense valid starting point
        cr_init = np.zeros(n)
        for i in range(n):
            dist_to_boundary = min(cx_init[i], 1 - cx_init[i], cy_init[i], 1 - cy_init[i])
            min_dist_to_center = np.inf
            for j in range(n):
                if i != j:
                    dist = np.sqrt((cx_init[i] - cx_init[j])**2 + (cy_init[i] - cy_init[j])**2)
                    if dist < min_dist_to_center:
                        min_dist_to_center = dist
            
            # Radius is limited by boundary and half the distance to nearest neighbor
            # Use a factor slightly less than 0.5 to ensure no overlap initially
            max_r = min(dist_to_boundary, 0.49 * min_dist_to_center)
            cr_init[i] = max(0.01, max_r)

        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = cx_init[i]
            x0[3*i+1] = cy_init[i]
            x0[3*i+2] = cr_init[i]

        # Optimization
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': get_constraints},
                       options={'maxiter': 2000, 'ftol': 1e-13})

        if res.success or (np.sum(res.fun) < -best_sum_r): # res.fun is negative sum
            current_sum = -res.fun
            if current_sum > best_sum_r:
                best_sum_r = current_sum
                best_centers = np.array([[res.x[3*i], res.x[3*i+1]] for i in range(n)])
                best_radii = res.x[2::3]

    return best_centers, best_radii, best_sum_r
