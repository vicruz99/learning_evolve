    Uses a pressure expansion simulation with repulsive forces.

        x += x_offset

            points.append([x, y])
            x += grid_spacing

        row_idx += 1

    pressure_step = 1e-4
    repulsion_strength = 100.0

    for step in range(num_iterations):
        # Gradually decrease pressure step for finer tuning
        current_pressure = pressure_step * (1.0 - step / num_iterations)
        
        # 1. Expand radii (Pressure)
        radii += current_pressure

            for j in range(i + 1, N):
                vec = centers[i] - centers[j]

                    overlap = sum_radii - dist

            r = radii[i]

            if x - r < 0:

                displacement[i, 0] -= repulsion_strength * (x + r - 1) * dt

                displacement[i, 1] += repulsion_strength * (r - y) * dt
