import numpy as np
from scipy.optimize import minimize
import math


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    def energy(vars, penalty=2000):
        centers = vars[:2 * n].reshape(n, 2)
        radii = np.maximum(vars[2 * n:], 0.001)

        obj = -np.sum(radii)
        penalty_sum = 0

        # Boundary penalties
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # x - r >= 0
            v = r - x
            if v > 0:
                penalty_sum += v * v
            # x + r <= 1
            v = x + r - 1
            if v > 0:
                penalty_sum += v * v
            # y - r >= 0
            v = r - y
            if v > 0:
                penalty_sum += v * v
            # y + r <= 1
            v = y + r - 1
            if v > 0:
                penalty_sum += v * v

        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    penalty_sum += overlap * overlap

        return obj + penalty * penalty_sum

    best_sum = 0.0
    best_centers = None
    best_radii = None

                    idx += 1

        centers[:, 1] = np.clip(centers[:, 1], 0.06, 0.94)

        for i in range(n):
