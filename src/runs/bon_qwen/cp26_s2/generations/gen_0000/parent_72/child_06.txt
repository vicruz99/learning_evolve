import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import cdist


def generate_hexagonal_packing(rows, r_scale=0.08):
    """Generate hexagonal packing with given row sizes."""
    centers = []
    r = r_scale
    for row_idx, num_cols in enumerate(rows):
        y = r + row_idx * r * np.sqrt(3)
        x_offset = r if row_idx % 2 == 1 else 0
        x_start = r + x_offset
        x_spacing = 2 * r
        for col_idx in range(num_cols):
            x = x_start + col_idx * x_spacing
            centers.append([x, y])
    return np.array(centers)


def optimize_radii_lp(centers):
    """Given fixed positions, find optimal radii using LP."""
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    A_ub_list = []
    b_ub_list = []
    
    # Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        e = np.zeros(n)
        e[i] = 1
        A_ub_list.append(e)
        b_ub_list.append(x)
        A_ub_list.append(e)
        b_ub_list.append(1 - x)
        A_ub_list.append(e)
        b_ub_list.append(y)
        A_ub_list.append(e)
        b_ub_list.append(1 - y)
    
    # Non-overlap: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            e = np.zeros(n)
            e[i] = 1
            e[j] = 1
            A_ub_list.append(e)
            b_ub_list.append(dist)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return np.full(n, 0.01)


def compute_repulsion_force(centers, radii, k_repulse=10.0):
    """Compute repulsion forces between circles."""
    n = centers.shape[0]
    force = np.zeros_like(centers)
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff ** 2))
            if dist < 1e-10:
                dist = 1e-10
            required = radii[i] + radii[j]
            if dist < required * 1.1:
                strength = k_repulse * (required - dist) / dist
                force[i] += diff * strength
                force[j] -= diff * strength
    
    return force


def compute_boundary_force(centers, radii, k_bound=50.0):
    """Compute forces pushing circles away from boundaries."""
    n = centers.shape[0]
    force = np.zeros_like(centers)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        margin = r + 0.001
        
        if x - r < 0.001:
            force[i, 0] += k_bound * (margin - (x - r))
        if x + r > 0.999:
            force[i, 0] -= k_bound * (margin - (1 - x - r))
        if y - r < 0.001:
            force[i, 1] += k_bound * (margin - (y - r))
        if y + r > 0.999:
            force[i, 1] -= k_bound * (margin - (1 - y - r))
    
    return force


def simulate_forces(centers, radii, num_iters=200, dt=0.005):
    """Simulate force-directed layout to spread circles apart."""
    centers = centers.copy()
    radii = radii.copy()
    
    for _ in range(num_iters):
        force = compute_repulsion_force(centers, radii) + compute_boundary_force(centers, radii)
        
        # Limit force magnitude
        max_f = 0.5
        norms = np.linalg.norm(force, axis=1, keepdims=True)
        norms = np.maximum(norms, 1e-10)
        force = np.where(norms > max_f, force * max_f / norms, force)
        
        centers += dt * force
        
        # Clamp to valid region
        for i in range(len(centers)):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r + 1e-6, 1 - r - 1e-6)
            centers[i, 1] = np.clip(centers[i, 1], r + 1e-6, 1 - r - 1e-6)
    
    return centers


def try_configuration(rows, num_iters_force=150):
    """Try a specific row configuration and optimize."""
    n = sum(rows)
    if n != 26:
        return None, None, -1
    
    # Generate initial packing with various scales

    best_centers = None
    best_radii = None

            continue
        
        # Initial radii optimization
        radii = optimize_radii_lp(centers)
        
        # Force-directed optimization

            centers = simulate_forces(centers, radii, num_iters=num_iters_force, dt=0.003)
            radii = optimize_radii_lp(centers)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii, best_sum


def optimize_with_slsqp(centers_init, radii_init, n):
    """Fine-tune using SLSQP optimizer."""
    
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers_init[i, 0]
        x0[3 * i + 1] = centers_init[i, 1]
        x0[3 * i + 2] = radii_init[i]
    
    def obj(vars):
        return -np.sum(vars[2::3])
    
    def boundary_constraints(vars):
        cons = np.zeros(5 * n)
        for i in range(n):
            idx = 3 * i
            x, y, r = vars[idx], vars[idx + 1], vars[idx + 2]
            cons[5 * i] = x - r
            cons[5 * i + 1] = 1 - x - r
            cons[5 * i + 2] = y - r
            cons[5 * i + 3] = 1 - y - r
            cons[5 * i + 4] = r
        return cons
    
    def overlap_constraints(vars):
        n_pairs = n * (n - 1) // 2
        cons = np.zeros(n_pairs)
        k = 0
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi, ri = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
                xj, yj, rj = vars[3 * j], vars[3 * j + 1], vars[3 * j + 2]
                dist = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2)
                cons[k] = dist - ri - rj
                k += 1
        return cons
    
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]
    
    result = minimize(
        obj, x0, method='SLSQP',
        constraints=constraints,
        options={'maxiter': 5000, 'ftol': 1e-14}
    )
    
    centers_opt = np.zeros((n, 2))
    radii_opt = np.zeros(n)
    for i in range(n):
        centers_opt[i, 0] = result.x[3 * i]
        centers_opt[i, 1] = result.x[3 * i + 1]
        radii_opt[i] = max(0, result.x[3 * i + 2])
    
    return centers_opt, radii_opt, np.sum(radii_opt)


def perturb_and_optimize(centers, radii, n, num_perturbations=20):
    """Try random perturbations and optimize each."""
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    for _ in range(num_perturbations):
        perturbed = centers.copy()
        scale = 0.02
        for i in range(n):
            perturbed[i, 0] += np.random.uniform(-scale, scale)
            perturbed[i, 1] += np.random.uniform(-scale, scale)
            perturbed[i, 0] = np.clip(perturbed[i, 0], radii[i] + 1e-6, 1 - radii[i] - 1e-6)
            perturbed[i, 1] = np.clip(perturbed[i, 1], radii[i] + 1e-6, 1 - radii[i] - 1e-6)
        
        radii_pert = optimize_radii_lp(perturbed)
        sum_pert = np.sum(radii_pert)
        
        if sum_pert > best_sum:
            best_centers = perturbed.copy()
            best_radii = radii_pert.copy()
            best_sum = sum_pert
    
    return best_centers, best_radii, best_sum
