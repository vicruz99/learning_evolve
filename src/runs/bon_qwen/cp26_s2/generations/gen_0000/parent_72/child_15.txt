import numpy as np
import scipy.optimize as opt
import math
import sys

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to find max radii.
    Maximize sum(r_i)
    Subject to:
    r_i + r_j <= dist(i, j) for all i < j
    r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i
    r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    c = -np.ones(n)
    
    # Constraints matrix A_ub * r <= b_ub
    constraints = []
    bounds_rhs = []
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            # r_i + r_j <= dist
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints.append(row)
            bounds_rhs.append(dist)
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    for i in range(n):
        x, y = centers[i]
        limits = [x, 1 - x, y, 1 - y]
        # We only need the tightest bound, but adding all is fine.
        # Actually, r_i <= min(limits).
        # In LP: r_i <= L_k for each k.
        for limit in limits:
            row = np.zeros(n)
            row[i] = 1.0
            constraints.append(row)
            bounds_rhs.append(limit)
            
    if not constraints:
        return np.zeros(n)

    A_ub = np.array(constraints)
    b_ub = np.array(bounds_rhs)
    
    # Bounds for r_i >= 0
    bounds = [(0, None)] * n
    
    # Solve LP
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback if LP fails (rare)
            return np.zeros(n)
    except Exception:
        return np.zeros(n)

def compute_forces(centers, radii, strength=0.01):
    """
    Compute repulsive forces on centers based on active constraints.
    If r_i + r_j is close to dist(i,j), push apart.
    If r_i is close to boundary, push inward.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    eps = 1e-4
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            sum_r = radii[i] + radii[j]
            
            if dist < 1e-9: dist = 1e-9 # Prevent div by zero
            
            # If constraint is active (sum_r approx dist), repel
            # The "slack" is dist - sum_r. If slack is small, force is high.
            # We want to increase dist.
            # Force vector direction: i away from j.
            if sum_r > 0:
                # Proportionality: how tight is the constraint?
                # If sum_r > dist, overlap (should not happen with valid radii), huge force.
                # If sum_r ~ dist, significant force.
                # If sum_r << dist, small force.
                
                # Simple heuristic: Force ~ (sum_r / dist) * 1 / dist ?
                # Or just push if touching.
                if sum_r >= dist - eps:
                    diff = centers[i] - centers[j]
                    norm_diff = np.sqrt(np.sum(diff**2))
                    if norm_diff > 1e-9:
                        # Unit vector from j to i
                        u = diff / norm_diff
                        # Force magnitude. 
                        # If we are touching, we want to push apart to allow larger radii.
                        # But we can't push infinitely.
                        # Let's apply a constant repulsive force if touching.
                        forces[i] += u * strength
                        forces[j] -= u * strength

        # Boundary forces
        # r_i <= x_i => if r_i approx x_i, push x_i positive
        # r_i <= 1-x_i => if r_i approx 1-x_i, push x_i negative
        # etc.
        
        x, y = centers[i]
        r = radii[i]
        
        # Left wall
        if r >= x - eps:
            forces[i, 0] += strength
        # Right wall
        if r >= (1 - x) - eps:
            forces[i, 0] -= strength
        # Bottom wall
        if r >= y - eps:
            forces[i, 1] += strength
        # Top wall
        if r >= (1 - y) - eps:
            forces[i, 1] -= strength
            
    return forces

def run_optimization(initial_centers, iterations=500):
    """
    Run the optimization loop.
    """
    centers = initial_centers.copy()
    best_sum_r = 0
    best_centers = None
    best_radii = None
    
    for step in range(iterations):
        # 1. Solve LP for radii
        radii = solve_radii_lp(centers)
        current_sum = np.sum(radii)
        
        if current_sum > best_sum_r:
            best_sum_r = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        # 2. Compute forces and update centers
        # Decrease step size over time
        strength = 0.05 * (1.0 - step / iterations)
        if strength < 1e-6: break
        
        forces = compute_forces(centers, radii, strength)
        
        # Update centers
        centers += forces
        
        # Clip to valid range [0, 1]
        # Actually centers can be anywhere, but radii will be 0 if outside?
        # Better to keep centers in [0,1] or slightly outside?
        # If center is at -0.1, radius must be <= x - (-0.1)? No.
        # Constraint: x - r >= 0 => r <= x. If x < 0, r <= negative => r=0.
        # So keeping centers in [0,1] is safe.
        centers = np.clip(centers, 1e-9, 1 - 1e-9)
        
    # Final check
    radii = solve_radii_lp(best_centers)
    return best_centers, radii, np.sum(radii)

def generate_hexagonal_grid(n):
    """
    Generate a hexagonal grid configuration.
    """
    # Estimate radius to fit n circles roughly
    # Area ~ n * pi * r^2 <= 1 => r ~ 1/sqrt(n*pi)
    # But packing density ~ 0.9.
    # r_approx = 0.5 * sqrt(1 / (n * sqrt(3)/2)) ? 
    # Simple guess: r=0.1 works for 25.
    r_guess = 0.1
    spacing_x = 2 * r_guess
    spacing_y = math.sqrt(3) * r_guess
    
    rows = []
    count = 0
    r_idx = 0
    y = r_guess
    
    while count < n:
        row_circles = []
        x_start = r_guess
        if r_idx % 2 == 1:
            x_start += r_guess # Shift by r_guess for hexagonal
            # Actually shift by spacing_x / 2 = r_guess
            
        x = x_start
        while x <= 1 - r_guess and count < n:
            row_circles.append([x, y])
            count += 1
            x += spacing_x
        rows.append(row_circles)
        y += spacing_y
        r_idx += 1
        
    centers = np.array([c for row in rows for c in row])
    return centers
