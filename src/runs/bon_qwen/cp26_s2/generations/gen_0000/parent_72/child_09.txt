import numpy as np
from scipy.optimize import minimize

def compute_objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[52:])

def compute_constraints(x):
    """
    Constraint function: returns values that must be >= 0.
    x layout: [x1, y1, ..., x26, y26, r1, ..., r26]
    """
    n = 26
    cx = x[:52].reshape(n, 2)
    cr = x[52:]
    
    vals = []
    # Boundary constraints: circle inside [0,1]x[0,1]
    for i in range(n):
        vals.append(cx[i, 0] - cr[i])           # x - r >= 0
        vals.append(1.0 - cx[i, 0] - cr[i])     # 1 - x - r >= 0
        vals.append(cx[i, 1] - cr[i])           # y - r >= 0
        vals.append(1.0 - cx[i, 1] - cr[i])     # 1 - y - r >= 0
        
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    diff_x = cx[:, 0] - cx[:, 0].T
    diff_y = cx[:, 1] - cx[:, 1].T
    dist_sq = diff_x**2 + diff_y**2
    rad_sum_sq = (cr[:, None] + cr[None, :])**2
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    pair_vals = dist_sq[i_idx, j_idx] - rad_sum_sq[i_idx, j_idx]
    
    return np.concatenate((vals, pair_vals))

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)

    for restart in range(5):

        cols = 6

            shift = 0.06 if i % 2 == 1 else 0.0
            for j in range(cols):
                if idx < n:
                    x = shift + j * 0.18
                    cx_init.append(x)
                    cy_init.append(y)
                    idx += 1
                    
        centers0 = np.column_stack((cx_init, cy_init))
        # Add small random perturbation to break symmetry
        centers0 += np.random.randn(n, 2) * 0.005
        centers0 = np.clip(centers0, 0.05, 0.95)

        x0 = np.concatenate([centers0.flatten(), r0])

        cons = {'type': 'ineq', 'fun': compute_constraints}
        
        res = minimize(fun=compute_objective,
                       x0=x0,
                       method='SLSQP',

    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    return best_centers, best_radii, best_sum
