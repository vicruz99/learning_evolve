def solve_radii_lp(centers):
    """
    Given fixed centers, find radii that maximize sum of radii
    using Linear Programming.
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum(r) -> Minimize -sum(r)

    # Constraints:
    # 1. Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    #    r_i - x_i <= 0  =>  r_i <= x_i
    #    r_i + x_i <= 1  =>  r_i <= 1 - x_i
    #    etc.
    
    # 2. Non-overlap: r_i + r_j <= dist(i, j)
    
    # We construct A_ub and b_ub such that A_ub @ r <= b_ub
    
    n_constraints = 0
    n_dist = n * (n - 1) // 2
    
    # Boundary constraints: 4 per circle
    A_ub = np.zeros((4 * n + n_dist, n))
    b_ub = np.zeros(4 * n + n_dist)
    
    row = 0
    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i  =>  1*r_i <= x_i
        A_ub[row, i] = 1
        b_ub[row] = x
        row += 1
        
        # r_i <= 1 - x_i => 1*r_i <= 1 - x_i
        A_ub[row, i] = 1
        b_ub[row] = 1 - x
        row += 1
        
        # r_i <= y_i
        A_ub[row, i] = 1
        b_ub[row] = y
        row += 1
        
        # r_i <= 1 - y_i
        A_ub[row, i] = 1
        b_ub[row] = 1 - y
        row += 1
        
    # Distance constraints: r_i + r_j <= dist
    idx = 0
    for i in range(n):

            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[row, i] = 1
            A_ub[row, j] = 1
            b_ub[row] = dist
            row += 1
            
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:

    else:
        return 0, np.zeros(n)

        # r in [0, 0.5]
        bounds.append((0, 0.5))
        
    # Optimize
    # SLSQP might struggle with 379 constraints.
    # Let's try to reduce constraints or use a simpler method.
    # But let's try it.
    
    # To improve performance, we can limit the number of constraints checked?
    # No, SLSQP handles them.
    
    # However, 78 vars and 379 constraints might be slow or fail to converge.
    # Let's try with a smaller number of iterations or simpler init.
    
    try:
        res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-9})
        if res.success:
            centers_opt = res.x[:52].reshape(n, 2)
            radii_opt = res.x[52:]
            return centers_opt, radii_opt, np.sum(radii_opt)
    except Exception as e:
        print("Optimization failed:", e)

    best_sum = 0
    best_centers = None
    best_radii = None
    
    for restart in range(3):

            centers = best_centers.copy()

            forces = np.zeros_like(centers)
            for i in range(n):

                # Boundary

                if x + r > 1 - 1e-5: forces[i, 0] -= 1
                if y - r < 1e-5: forces[i, 1] += 1
                if y + r > 1 - 1e-5: forces[i, 1] -= 1
