import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n_circles = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Helper function to compute constraints
    def get_constraints(vars):
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        constraints = []
        
        # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        # x - r >= 0
        constraints.append(x - r)
        # 1 - x - r >= 0
        constraints.append(1.0 - x - r)
        # y - r >= 0
        constraints.append(y - r)
        # 1 - y - r >= 0
        constraints.append(1.0 - y - r)
        
        # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        # We compute these for all pairs i < j
        pair_constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dr = r[i] + r[j]
                pair_constraints.append(dx*dx + dy*dy - dr*dr)
        
        constraints.append(np.array(pair_constraints))
        
        # Flatten all constraints into a single array
        return np.concatenate(constraints)

    def objective(vars):
        r = vars[2::3]
        return -np.sum(r)

    # Initial guess generation
    def generate_initial_guess(seed):
        rng = np.random.default_rng(seed)
        vars = np.zeros(3 * n_circles)
        
        # Strategy: Try a hexagonal-like packing or random grid
        # Let's try placing them in a grid first and adding randomness
        
        # Approximate spacing for 26 circles: sqrt(1/26) approx 0.2
        # Let's try a 5x5 grid (25) + 1
        
        # Grid positions
        grid_spacing = 0.22 # Slightly larger than 0.2 to allow space
        offset = 0.1
        
        count = 0
        # Place 5x5 grid
        for row in range(5):
            for col in range(5):
                if count < n_circles:
                    idx = count
                    x_pos = offset + col * grid_spacing
                    y_pos = offset + row * grid_spacing
                    # Add small jitter
                    x_pos += rng.uniform(-0.02, 0.02)
                    y_pos += rng.uniform(-0.02, 0.02)
                    
                    # Clamp to valid range for initial feasibility (small radius)
                    x_pos = np.clip(x_pos, 0.05, 0.95)
                    y_pos = np.clip(y_pos, 0.05, 0.95)
                    
                    vars[3*idx] = x_pos
                    vars[3*idx+1] = y_pos
                    vars[3*idx+2] = 0.01 # Small initial radius
                    count += 1
        
        # If we have fewer than 26 (shouldn't with 5x5=25), fill remaining
        while count < n_circles:
            idx = count
            vars[3*idx] = rng.uniform(0.1, 0.9)
            vars[3*idx+1] = rng.uniform(0.1, 0.9)
            vars[3*idx+2] = 0.01
            count += 1
            
        return vars

    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Run optimization multiple times with different seeds
    seeds = [42, 123, 456, 789, 1001, 2023, 999, 0, 1, 2]
    
    for seed in seeds:
        try:
            x0 = generate_initial_guess(seed)
            
            # Constraints definition for scipy
            # SLSQP expects constraints as a list of dict or NonlinearConstraint
            # But passing a function that returns array is tricky with list of dicts
            # We can use a single constraint function returning an array
            
            # However, scipy.optimize.minimize with method='SLSQP' accepts 
            # constraints as a sequence of Mapping objects or a single callable 
            # if we use the older syntax? 
            # Actually, standard way is list of {'type': 'ineq', 'fun': ...}
            # But our constraint function returns a large array. 
            # We can define a wrapper.
            
            cons = {
                'type': 'ineq',
                'fun': get_constraints,
                'jac': '2-point' # Numerical Jacobian
            }
            
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'ftol': 1e-9, 'maxiter': 500})
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_sum_radii:
                    # Validate locally before updating best
                    centers_res = np.column_stack((res.x[0::3], res.x[1::3]))
                    radii_res = res.x[2::3]
                    
                    # Quick validation check (simplified)
                    # Check bounds and overlaps
                    valid = True
                    for i in range(n_circles):
                        if radii_res[i] < -1e-9 or radii_res[i] > 0.5 + 1e-9:
                            valid = False; break
                        if centers_res[i, 0] < radii_res[i] - 1e-9 or centers_res[i, 0] > 1 - radii_res[i] + 1e-9:
                            valid = False; break
                        if centers_res[i, 1] < radii_res[i] - 1e-9 or centers_res[i, 1] > 1 - radii_res[i] + 1e-9:
                            valid = False; break
                    
                    if valid:
                        # Check overlaps

                            for j in range(i+1, n_circles):
                                dist = np.sqrt(np.sum((centers_res[i] - centers_res[j])**2))
                                if dist < radii_res[i] + radii_res[j] - 1e-7:
                                    valid = False; break
                            if not valid: break
                        
                        if valid:
                            best_sum_radii = sum_r
                            best_centers = centers_res.copy()
                            best_radii = radii_res.copy()
                            
        except Exception as e:
            # print(f"Optimization failed for seed {seed}: {e}")
            pass

    # If no valid solution found from optimizer (unlikely), fallback to a simple grid
    if best_centers is None:
        centers_fallback = np.zeros((n_circles, 2))
        radii_fallback = np.zeros(n_circles)
        # 5x5 grid
        r_fallback = 0.1
        count = 0
        for r in range(5):
            for c in range(5):
                if count < n_circles:
                    centers_fallback[count] = [0.1 + c * 0.2, 0.1 + r * 0.2]
                    radii_fallback[count] = 0.1
                    count += 1
        # Last one
        if count < n_circles:
            centers_fallback[count] = [0.5, 0.5]
            radii_fallback[count] = 0.0
        best_centers = centers_fallback
        best_radii = radii_fallback
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
