def run_packing():

    # 1. Initialization: Grid

    centers = np.array(coords)
    
    # 2. Force-directed expansion (Vectorized)
    max_iter = 500
    dr = 5e-4
    k_rep = 20.0
    k_wall = 50.0
    dt = 0.05
    
    # Preallocate
    # centers shape (26, 2)
    
    for step in range(max_iter):
        radii += dr
        
        # Relaxation steps
        for _ in range(5):
            # Compute pairwise differences
            # diff[i, j] = centers[i] - centers[j]
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            # dist_sq[i, j] = ||centers[i] - centers[j]||^2
            dist_sq = np.sum(diff**2, axis=2)

            dists = np.sqrt(dist_sq)
            
            # Sum of radii matrix
            r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]

            # Force direction (unit vector)

            forces_wall[mask, 1] -= k_wall * (centers[mask, 1] - (1 - radii[mask]))
            
            centers += (forces_pair + forces_wall) * dt
            centers = np.clip(centers, 0, 1)

    # 3. Refinement using scipy.optimize
    # We refine the solution to strictly satisfy constraints and maximize sum of radii.
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)] * 2*n + [(0, 1)] * n
    
    res = minimize(scipy_obj, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 500, 'ftol': 1e-9})
    
    centers_opt = res.x[:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]
    
    # Ensure non-negative radii
    radii_opt = np.maximum(radii_opt, 0)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
