import scipy.optimize as opt

def run_packing():

    Pack 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    # This mimics a hexagonal lattice.
    row_counts = [6, 5, 6, 5, 4]

    raw_points = []
    y_curr = 0
    for i, count in enumerate(row_counts):
        # Offset odd rows by 1 unit (which is radius in this scale)
        offset = 1.0 if (i % 2 == 1) else 0.0
        for j in range(count):
            x = offset + j * 2.0
            raw_points.append([x, y_curr])
        y_curr += np.sqrt(3) # Vertical spacing for hex lattice with radius 1 is sqrt(3)
    
    raw_points = np.array(raw_points)
    
    # Normalize to fit in [0, 1] square with a safety margin
    # We want to map the bounding box of raw_points to [margin, 1-margin]
    # Initially, we don't know optimal r, but we know r will be roughly 1/scale.
    # Let's map raw x range [min_x, max_x] to [0, 1] first, then we can shrink if needed.
    min_x, max_x = raw_points[:, 0].min(), raw_points[:, 0].max()
    min_y, max_y = raw_points[:, 1].min(), raw_points[:, 1].max()

    # Actually, we want centers to be roughly in the middle.
    # Let's scale such that the extent is 0.9, leaving 0.05 margin.
    width = max_x - min_x
    height = max_y - min_y
    scale = 0.9 / max(width, height)

    centers_init -= [centers_init[:, 0].min(), centers_init[:, 1].min()] # Shift to 0
    
    # Estimate initial radius based on scale. 
    # In raw coords, radius was 1. Scaled radius is `scale`.

    # But we must leave room for boundary. 
    # A safe initial radius is slightly less than half the min spacing to boundary or other circles.

    radii_init = np.full(n, r_init)

    vars0 = np.hstack([centers_init[:, 0], centers_init[:, 1], radii_init])

        # Maximize sum of radii -> Minimize negative sum
        r = vars[2::3]
        return -np.sum(r)

    for i in range(n):

        idx_y = 3 * i + 1
        idx_r = 3 * i + 2

        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - (x + r) >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - (v[idx_x] + v[idx_r])
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - (y + r) >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - (v[idx_y] + v[idx_r])
        })

        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_r]
        })

        vars_opt = vars0

        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)

                        dir_vec = dir_vec / d

            if centers[i, 1] < r: centers[i, 1] = r
            if centers[i, 1] > 1 - r: centers[i, 1] = 1 - r
        if not moved:
            break

    # Recalculate sum radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
