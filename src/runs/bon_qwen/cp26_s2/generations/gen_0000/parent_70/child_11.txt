import numpy as np
import scipy.optimize as opt
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_initial_configuration():
    """
    Generates an initial configuration of 26 circles in a hexagonal-like pattern.
    Layout: 6 rows with counts 5, 4, 5, 4, 5, 3.
    """
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial guess radius
    r_guess = 0.09
    
    rows_counts = [5, 4, 5, 4, 5, 3]
    idx = 0
    for row_idx, count in enumerate(rows_counts):
        y = r_guess + row_idx * r_guess * math.sqrt(3)
        
        # Center the row horizontally
        # Width of row with 'count' circles is count * 2 * r_guess? 
        # Actually spacing is 2r. Total width = 2r + (count-1)*2r = 2r*count.
        # To center in [0, 1], start x = (1 - 2r*count) / 2 + r

        shift = 0
        if row_idx % 2 == 1:
            shift = r_guess # Shift by one radius to nest in gaps
            
        start_x = (1 - (count * 2 * r_guess)) / 2 + r_guess + shift
        
        for col in range(count):
            x = start_x + col * 2 * r_guess
            centers[idx] = [x, y]
            radii[idx] = r_guess
            idx += 1
            
    return centers, radii

def objective_function(params, n):
    """
    Objective function to minimize.
    We want to maximize sum of radii, so we minimize -sum(radii).
    We add penalties for constraint violations.
    
    Params: flattened array [x0, y0, r0, x1, y1, r1, ...]
    """
    centers = params[0:2*n].reshape((n, 2))
    radii = params[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    score = -np.sum(radii)
    
    # Penalty parameters
    pen_weight = 1000.0
    
    # Boundary penalties
    # x - r >= 0 => r - x <= 0
    # x + r <= 1 => x + r - 1 <= 0
    # Same for y
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Boundary violations
        v1 = max(0, r - x)
        v2 = max(0, x + r - 1)
        v3 = max(0, r - y)
        v4 = max(0, y + r - 1)
        
        # Radius must be positive (handled by bounds, but just in case)
        v5 = max(0, -r)
        
        score += pen_weight * (v1**2 + v2**2 + v3**2 + v4**2 + v5**2)
        
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            req_dist = radii[i] + radii[j]
            if dist < req_dist:
                score += pen_weight * (req_dist - dist)**2
                
    return score
