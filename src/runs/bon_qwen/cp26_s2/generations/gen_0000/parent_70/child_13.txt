import numpy as np
from scipy.optimize import minimize
import math

def compute_penalty(centers, radii):
    """Computes the violation penalty for boundaries and overlaps."""
    n = centers.shape[0]
    pen = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Boundary constraints
        if x - r < 0:
            pen += (x - r)**2
        if x + r > 1:
            pen += (x + r - 1)**2
        if y - r < 0:
            pen += (y - r)**2
        if y + r > 1:
            pen += (y + r - 1)**2
            
        # Overlap constraints
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                pen += overlap**2
    return pen

    n = 26
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    penalty = compute_penalty(centers, radii)
    # Weight for penalty ensures constraints are respected during optimization
    return -np.sum(radii) + 5000.0 * penalty
