import numpy as np
from scipy.optimize import linprog, minimize

def get_max_sum_radii(centers):
    """
    Given fixed centers, solve the LP to find optimal radii maximizing sum.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        float: max sum of radii, or 0 if infeasible/invalid
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Pairwise non-overlap: r_i + r_j <= dist(i, j)
    # 2. Boundary constraints: r_i <= dist(i, boundary)
    
    constraints_matrix = []
    constraints_vector = []
    
    # Precompute distances
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            # Row with 1 at i and 1 at j
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            constraints_matrix.append(row)
            constraints_vector.append(dist)
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r <= x
        row = np.zeros(n)
        row[i] = 1
        constraints_matrix.append(row)
        constraints_vector.append(x)
        
        # r <= 1 - x
        row = np.zeros(n)
        row[i] = 1
        constraints_matrix.append(row)
        constraints_vector.append(1 - x)
        
        # r <= y
        row = np.zeros(n)
        row[i] = 1
        constraints_matrix.append(row)
        constraints_vector.append(y)
        
        # r <= 1 - y
        row = np.zeros(n)
        row[i] = 1
        constraints_matrix.append(row)
        constraints_vector.append(1 - y)

    A_ub = np.array(constraints_matrix)
    b_ub = np.array(constraints_vector)
    
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            return -res.fun
        else:
            return 0.0
    except Exception:
        return 0.0

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
