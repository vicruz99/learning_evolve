

<strategy>
I'll use a constrained optimization approach with scipy.optimize.minimize (SLSQP method) to pack 26 circles in a unit square.

Key design decisions:

1. **Initial Configuration**: Start with a structured hexagonal grid arrangement. This is crucial because hexagonal packing is the densest circle packing in 2D. For 26 circles, I'll arrange them in 5-6 rows with alternating offsets.

2. **Variable Encoding**: Each circle i has 3 variables: x_i, y_i, r_i. Total 78 variables.

3. **Objective**: Maximize sum of radii (minimize negative sum).

4. **Constraints**:
   - Boundary: x_i - r_i >= 0, x_i + r_i <= 1, y_i - r_i >= 0, y_i + r_i <= 1 for each circle
   - Non-negativity: r_i >= 0
   - Non-overlap: distance(centers[i], centers[j]) >= r_i + r_j for each pair

5. **Multiple Restarts**: Run optimization from different starting configurations to escape local minima.

6. **Helper Functions**: All top-level, no closures, no lambda functions. Use scipy's `args` parameter to pass indices to constraint functions.

7. **Refinement**: After optimization, run a post-processing step that further pushes circles apart to increase radii where possible, using a local search approach.

import numpy as np
from scipy.optimize import minimize


def objective(x, n):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    return -np.sum(x[2::3])


def boundary_x_lower(x, i):
    """Constraint: x_i - r_i >= 0 (circle not below left boundary)."""
    return x[3 * i] - x[3 * i + 2]


def boundary_x_upper(x, i):
    """Constraint: 1 - x_i - r_i >= 0 (circle not above right boundary)."""
    return 1.0 - x[3 * i] - x[3 * i + 2]


def boundary_y_lower(x, i):
    """Constraint: y_i - r_i >= 0 (circle not below bottom boundary)."""
    return x[3 * i + 1] - x[3 * i + 2]


def boundary_y_upper(x, i):
    """Constraint: 1 - y_i - r_i >= 0 (circle not above top boundary)."""
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def radius_nonnegative(x, i):
    """Constraint: r_i >= 0."""
    return x[3 * i + 2]


def non_overlap(x, i, j):
    """Constraint: distance(i,j) >= r_i + r_j (no overlap)."""
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    dist_sq = dx * dx + dy * dy
    min_dist = x[3 * i + 2] + x[3 * j + 2]
    return dist_sq - min_dist * min_dist


def make_hexagonal_initial(n, r_guess=0.095):
    """Create initial hexagonal packing configuration for n circles."""
    centers = np.zeros((n, 2))
    radii = np.full(n, r_guess)
    
    idx = 0
    row = 0
    
    while idx < n:
        if row % 2 == 0:
            num_in_row = 5
        else:
            num_in_row = 5 if idx + 5 <= n else 4
        
        if idx + num_in_row > n:
            num_in_row = n - idx
        
        spacing_x = 0.19
        spacing_y = 0.165
        offset = spacing_x * 0.5 if row % 2 == 1 else 0.0
        
        base_x = 0.5 - (num_in_row - 1) * spacing_x * 0.5 + offset
        base_y = 0.08 + row * spacing_y
        
        for col in range(num_in_row):
            centers[idx, 0] = base_x + col * spacing_x
            centers[idx, 1] = base_y
            idx += 1
        
        row += 1
    
    return centers, radii


def build_constraints(n):
    """Build all constraints for the optimization problem."""
    constraints = []
    
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': boundary_x_lower, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_x_upper, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_lower, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_upper, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': radius_nonnegative, 'args': (i,)})
    
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': non_overlap, 'args': (i, j)})
    
    return constraints


def build_bounds(n):
    """Build variable bounds."""
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))
    return bounds


def extract_solution(result, n):
    """Extract centers and radii from optimization result."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = result.x[3 * i]
        centers[i, 1] = result.x[3 * i + 1]
        radii[i] = result.x[3 * i + 2]
    return centers, radii


def perturb_initial(centers, radii, scale=0.02):
    """Add small random perturbation to initial configuration."""
    np.random.seed(42)
    perturbed_centers = centers.copy()
    perturbed_centers += np.random.randn(*centers.shape) * scale
    perturbed_centers = np.clip(perturbed_centers, 0.05, 0.95)
    return perturbed_centers, radii.copy()


def local_radius_increase(centers, radii, n, steps=200, step_size=0.001):
    """Try to increase radii by pushing circles to boundaries."""
    centers = centers.copy()
    radii = radii.copy()
    
    for step in range(steps):
        improved = False
        for i in range(n):
            for direction in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
                new_centers = centers.copy()
                new_centers[i, 0] += direction[0] * step_size * (step + 1) * 0.1
                new_centers[i, 1] += direction[1] * step_size * (step + 1) * 0.1
                new_centers = np.clip(new_centers, 0, 1)
                
                valid = True
                for j in range(n):
                    if i == j:
                        continue
                    dx = new_centers[i, 0] - new_centers[j, 0]
                    dy = new_centers[i, 1] - new_centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                
                if valid and np.all(new_centers[i] - radii[i] >= -1e-10) and np.all(new_centers[i] + radii[i] <= 1 + 1e-10):
                    centers[i] = new_centers[i]
                    improved = True
        
        if not improved:
            break
    
    # Try increasing radii
    for iteration in range(100):
        max_possible = 0.5
        for i in range(n):
            # Check boundary limits
            r_max = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            
            # Check overlap limits
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                r_max = min(r_max, dist - radii[j])
            
            r_max = max(0, r_max)
            
            if r_max > radii[i] + 1e-10:
                increment = min(r_max - radii[i], 0.001)
                radii[i] += increment * 0.5
                max_possible = max(max_possible, radii[i])
        
        if max_possible - np.max(radii) < 1e-12:
            break
    
    return centers, radii


def run_packing():
    n = 26
    
    constraints = build_constraints(n)
    bounds = build_bounds(n)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple initial configurations
    for trial in range(4):
        centers_init, radii_init = make_hexagonal_initial(n, r_guess=0.095)
        
        if trial > 0:
            centers_init, radii_init = perturb_initial(centers_init, radii_init, scale=0.015 * trial)
        
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3 * i] = centers_init[i, 0]
            x0[3 * i + 1] = centers_init[i, 1]
            x0[3 * i + 2] = radii_init[i]
        
        result = minimize(
            objective, x0, args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
        )
        
        if result.fun < 0:
            centers_opt, radii_opt = extract_solution(result, n)
            current_sum = np.sum(radii_opt)
            
            # Local refinement
            centers_opt, radii_opt = local_radius_increase(centers_opt, radii_opt, n)
            current_sum = np.sum(radii_opt)
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
    
    # If optimization failed, use a safe fallback
    if best_centers is None or best_sum < 2.0:
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.085)
        idx = 0
        for row in range(5):
            for col in range(5):
                if idx < n:
                    centers[idx] = [0.11 + col * 0.18, 0.11 + row * 0.18]
                    idx += 1
        while idx < n:
            centers[idx] = [0.5 + 0.15 * (idx - 25), 0.9]
            idx += 1
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)
    
    # Final safety check - ensure no NaN
    if np.any(np.isnan(best_centers)) or np.any(np.isnan(best_radii)):
        best_centers = np.ones((n, 2)) * 0.5
        best_radii = np.zeros(n)
        best_sum = 0.0
    
    return best_centers, best_radii, best_sum
