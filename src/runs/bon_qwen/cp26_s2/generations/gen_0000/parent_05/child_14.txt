import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def boundary_constraints(vars):
    """Constraints ensuring circles stay within the unit square."""
    n = N_CIRCLES
    cons = np.empty(4 * n)
    for i in range(n):
        idx = 3 * i
        x, y, r = vars[idx], vars[idx+1], vars[idx+2]
        cons[4*i] = x - r
        cons[4*i+1] = 1.0 - x - r
        cons[4*i+2] = y - r
        cons[4*i+3] = 1.0 - y - r
    return cons

def pairwise_constraints(vars):
    """Constraints ensuring no two circles overlap."""
    n = N_CIRCLES
    m = n * (n - 1) // 2
    cons = np.empty(m)
    idx = 0
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            dx, dy = xi - xj, yi - yj
            cons[idx] = dx*dx + dy*dy - (ri + rj)**2
            idx += 1
    return cons

def run_packing():
    n = N_CIRCLES
    n_vars = 3 * n
    bnds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': pairwise_constraints}
    ]
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Multiple restarts to escape local optima
    for seed in range(15):

        vars_init = np.zeros(n_vars)

            vars_init[0::3] = np.tile(np.array([0.2, 0.4, 0.6, 0.8, 0.3]), 6)[:n]

            vars_init[2::3] = 0.03

            vars_init[0::3] = np.random.uniform(0.1, 0.9, n)
            vars_init[1::3] = np.random.uniform(0.1, 0.9, n)
            vars_init[2::3] = np.random.uniform(0.02, 0.05, n)
            
        try:
            res = minimize(objective, vars_init, method='SLSQP', bounds=bnds,

            if res.success:
                s = -res.fun
                if s > best_sum:

                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            continue
            
    if best_centers is not None:
        return best_centers, best_radii, best_sum
        
    # Fallback solution
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.ones(n) * 0.02
    return centers, radii, 0.52
