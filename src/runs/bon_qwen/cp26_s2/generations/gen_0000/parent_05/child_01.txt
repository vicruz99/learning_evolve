import numpy as np
from scipy.optimize import minimize, linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the positions of 26 circles in a unit square to maximize the sum of radii.
    Uses a nested optimization approach:
    1. Outer loop: Optimizes centers using Nelder-Mead.
    2. Inner function: Solves a Linear Programming problem to find max radii for fixed centers.
    """
    n = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    # Helper function to solve for max radii given fixed centers
    def solve_radii(centers):
        # Variables: r_0, ..., r_25 (26 variables)
        # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
        c = -np.ones(n)

        # Constraints matrices
        # We construct A_ub and b_ub for constraints: A_ub @ r <= b_ub
        
        # 1. Boundary constraints:
        # r_i <= x_i  =>  r_i - x_i <= 0  (Wait, x_i is constant here)
        # Actually: r_i <= x_i  =>  1*r_i <= x_i
        # r_i <= 1 - x_i =>  1*r_i <= 1 - x_i
        # r_i <= y_i
        # r_i <= 1 - y_i
        # Total 4 * n constraints
        
        # 2. Pairwise constraints:
        # r_i + r_j <= dist(centers[i], centers[j])
        # 1*r_i + 1*r_j <= dist
        # Total n*(n-1)/2 constraints
        
        num_constraints = 4 * n + (n * (n - 1)) // 2
        A_ub = np.zeros((num_constraints, n))
        b_ub = np.zeros(num_constraints)
        
        idx = 0
        for i in range(n):
            xi, yi = centers[i]
            
            # r_i <= xi
            A_ub[idx, i] = 1.0
            b_ub[idx] = xi
            idx += 1
            
            # r_i <= 1 - xi
            A_ub[idx, i] = 1.0
            b_ub[idx] = 1.0 - xi
            idx += 1
            
            # r_i <= yi
            A_ub[idx, i] = 1.0
            b_ub[idx] = yi
            idx += 1
            
            # r_i <= 1 - yi
            A_ub[idx, i] = 1.0
            b_ub[idx] = 1.0 - yi
            idx += 1

        # Pairwise constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                A_ub[idx, i] = 1.0
                A_ub[idx, j] = 1.0
                b_ub[idx] = dist
                idx += 1
                
        # Bounds for r_i >= 0
        bounds = [(0, None) for _ in range(n)]
        
        # Solve LP
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                return -res.fun, res.x
            else:
                return 0.0, np.zeros(n)
        except Exception:
            return 0.0, np.zeros(n)

        return -sum_r

                centers.append([x, y])

        centers.append([0.2, 0.2]) 
        
        return np.array(centers).flatten()
