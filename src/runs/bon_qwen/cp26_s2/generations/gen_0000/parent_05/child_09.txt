import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_initial_positions(n):

    Initialize circle centers in a dense hexagonal pattern.
    Uses a row count of [5, 5, 5, 5, 5, 1] to handle 26 circles efficiently.
    """
    centers = np.zeros((n, 2))
    r_init = 0.105 # Initial radius estimate
    
    # Hexagonal vertical spacing
    dy = np.sqrt(3) * r_init
    dx = 2 * r_init
    
    rows = [5, 5, 5, 5, 5, 1]
    idx = 0
    
    # Start position to center the packing
    y_base = 0.05
    
    for row_idx, count in enumerate(rows):
        # Offset odd rows for hexagonal packing
        x_offset = r_init if row_idx % 2 == 1 else r_init

        # Total width of circles in row: (count-1)*2r + 2r = 2*count*r
        # We want the center of the row to be at 0.5
        row_width = count * 2 * r_init
        x_start = 0.5 - row_width / 2 + r_init
        
        for k in range(count):
            if idx < n:
                centers[idx, 0] = x_start + k * 2 * r_init
                centers[idx, 1] = y_base + row_idx * dy
                idx += 1
                
    # Normalize or adjust if needed, but this places them well inside [0,1]
    return centers

def objective(variables, n):
    """
    Objective function to maximize sum of radii (minimize negative sum).
    Variables: [x_0, y_0, r_0, x_1, y_1, r_1, ...]
    """
    radii = variables[2::3]
    return -np.sum(radii)

def constraints_factory(n):
    """
    Factory for optimization constraints.
    1. Circle non-overlap
    2. Boundary containment
    """
    cons = []
    
    # 1. Overlap constraints: dist >= r_i + r_j  => dist - (r_i + r_j) >= 0
    # To speed up, we can just add a penalty or rely on bounds, 
    # but for strict validity, explicit constraints are better.
    # However, SLSQP handles many constraints poorly. 
    # We will use a penalty approach for overlaps if constraints fail, 
    # but let's try standard constraints first. 
    # Given n=26, there are 325 pairs. This is heavy.
    # Alternative: Use a barrier method or just rely on the optimizer's ability.
    # Let's stick to the prompt's request for a robust solution.
    # A common trick in packing is to not list all constraints but use a soft penalty 
    # in the objective if the optimizer is robust, OR list them.
    # Let's try listing them.
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = i * 3
            idx_j = j * 3
            
            def make_overlap_constraint(i, j):
                def overlap_constraint(variables):
                    xi, yi = variables[idx_i], variables[idx_i+1]
                    ri = variables[idx_i+2]
                    xj, yj = variables[idx_j], variables[idx_j+1]
                    rj = variables[idx_j+2]
                    
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    # We want dist >= ri + rj
                    # Constraint function must be >= 0
                    # Return dist - (ri + rj). Note: sqrt is expensive, 
                    # but necessary for correct distance.
                    # Using dist_sq - (ri+rj)^2 >= 0 is also valid and faster.
                    return dist_sq - (ri + rj)**2
                return overlap_constraint
            
            cons.append({'type': 'ineq', 'fun': make_overlap_constraint(i, j)})

    # 2. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # x - r >= 0 => x >= r
    # 1 - x - r >= 0 => x + r <= 1
    for i in range(n):
        idx = i * 3
        # x >= r
        cons.append({'type': 'ineq', 'fun': lambda v, idx=idx: v[idx] - v[idx+2]})
        # y >= r
        cons.append({'type': 'ineq', 'fun': lambda v, idx=idx: v[idx+1] - v[idx+2]})
        # x + r <= 1 => 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=idx: 1.0 - v[idx] - v[idx+2]})
        # y + r <= 1 => 1 - y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=idx: 1.0 - v[idx+1] - v[idx+2]})
        
    return cons
