    *   **Position Update:** Move centers based on the net forces.

import numpy as np

def solve_packing():
    """
    Runs a force-directed simulation to pack 26 circles in a unit square,
    maximizing the sum of radii.
    """
    n_circles = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Number of random restarts to find a good global optimum
    num_restarts = 20 
    
    # Simulation parameters
    # Stiffness of the repulsion forces
    stiffness = 20.0 
    # Initial radius
    init_radius = 0.001
    # Growth rate per step (decayed over time)
    initial_growth = 0.001
    # Time step for integration
    dt = 0.05
    # Number of iterations
    max_iter = 3000
    
    for restart in range(num_restarts):
        # Initialize centers randomly inside [0.1, 0.9] to avoid immediate boundary issues
        centers = np.random.uniform(0.1, 0.9, size=(n_circles, 2))
        radii = np.ones(n_circles) * init_radius
        
        # Current growth rate
        current_growth = initial_growth

        for step in range(max_iter):
            forces = np.zeros_like(centers)
            
            # 1. Calculate repulsive forces between circles
            # Vectorized calculation for efficiency
            # Centers shape: (N, 2)
            # Compute all pairwise differences: (N, N, 2)
            diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists_sq = np.sum(diffs**2, axis=2) # (N, N)
            
            # Avoid division by zero for i==j
            np.fill_diagonal(dists_sq, np.inf)
            dists = np.sqrt(dists_sq)
            
            # Minimum required distance
            min_dists = radii[:, np.newaxis] + radii[np.newaxis, :]
            
            # Overlap amount: positive if overlapping
            overlaps = min_dists - dists
            overlaps[overlaps < 0] = 0
            
            # Direction vector (normalized)
            # Handle zero distance (should not happen due to inf diag)
            with np.errstate(divide='ignore', invalid='ignore'):
                directions = diffs / dists[:, :, np.newaxis]
            
            # Force magnitude proportional to overlap
            # F = stiffness * overlap
            force_magnitudes = stiffness * overlaps
            
            # Apply forces to the force accumulator
            # forces[i] += sum over j of (force_mag * direction)
            # Note: directions[i, j] points from j to i
            # We want to push i away from j, so direction is correct.
            # However, since we sum over all j, we need to be careful with indices.
            # Actually, for pair (i,j), force on i is towards i (away from j).
            # diffs[i, j] = c[i] - c[j]. Direction is correct.
            
            # Sum forces along axis 1 (sum over j)
            # force_magnitudes is (N, N), directions is (N, N, 2)
            # Element-wise multiply
            forces += np.sum(force_magnitudes[:, :, np.newaxis] * directions, axis=1)

            # 2. Boundary forces
            # If circle i is too close to x=0, push right (positive x)
            # Violation amount: r_i - x_i
            left_violations = radii - centers[:, 0]
            left_violations[left_violations < 0] = 0
            forces[:, 0] += stiffness * 2.0 * left_violations # Factor 2 for stronger boundary push
            
            # If circle i is too close to x=1, push left (negative x)
            right_violations = (centers[:, 0] + radii) - 1.0
            right_violations[right_violations < 0] = 0
            forces[:, 0] -= stiffness * 2.0 * right_violations
            
            # Same for y
            bottom_violations = radii - centers[:, 1]
            bottom_violations[bottom_violations < 0] = 0
            forces[:, 1] += stiffness * 2.0 * bottom_violalties
            
            top_violations = (centers[:, 1] + radii) - 1.0
            top_violations[top_violations < 0] = 0
            forces[:, 1] -= stiffness * 2.0 * top_violations

            # 3. Update centers
            centers += dt * forces
            
            # Clamp centers strictly to valid range [r, 1-r] to prevent numerical drift
            # Although forces should handle it, clamping ensures validity
            centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
            centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
            
            # 4. Grow radii
            # Decay growth rate slightly to settle
            radii += current_growth
            if step % 100 == 0:
                current_growth *= 0.98 

        # After simulation, check validity and sum
        # Since we clamped and used forces, it should be valid, but verify radii sum
        current_sum = np.sum(radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii, sum_radii = solve_packing()
    return centers, radii, sum_radii
