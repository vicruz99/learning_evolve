import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initialization: 5x5 Grid
    # Start with a grid that is known to be a strong baseline for 25 circles.
    # We reduce the radius slightly to create a valid "starting point" for the optimizer
    # and to leave room for the 26th circle to be integrated.
    r_start = 0.095
    x_pos = np.linspace(r_start, 1 - r_start, 5)
    y_pos = np.linspace(r_start, 1 - r_start, 5)
    
    grid_centers = []
    for x in x_pos:
        for y in y_pos:
            grid_centers.append([x, y])
            
    # Add the 26th circle at the center of the square
    grid_centers.append([0.5, 0.5])
    grid_centers = np.array(grid_centers)
    
    # 2. Define Variables and Optimization Function
    # Variables vector: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    n_vars = 3 * n
    x0 = np.zeros(n_vars)
    for i in range(n):
        x0[3 * i] = grid_centers[i, 0]
        x0[3 * i + 1] = grid_centers[i, 1]
        x0[3 * i + 2] = r_start

    def objective(vars_vec):
        # Return negative sum of radii (to minimize)
        return -np.sum(vars_vec[2::3])

    def boundary_constraints(vars_vec):
        # x_i - r_i >= 0  =>  r_i - x_i <= 0
        # x_i + r_i <= 1  =>  x_i + r_i - 1 <= 0
        # Same for y
        cons = []
        for i in range(n):
            idx = 3 * i
            r = vars_vec[idx + 2]
            x = vars_vec[idx]
            y = vars_vec[idx + 1]
            cons.append(r - x)       # Left boundary
            cons.append(x + r - 1)   # Right boundary
            cons.append(r - y)       # Bottom boundary
            cons.append(y + r - 1)   # Top boundary
        return np.array(cons)

    def non_overlap_constraints(vars_vec):
        # dist_ij >= r_i + r_j  =>  (r_i + r_j)^2 - dist_ij^2 <= 0
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                idx_i = 3 * i
                idx_j = 3 * j
                r_i = vars_vec[idx_i + 2]
                r_j = vars_vec[idx_j + 2]
                
                dist_sq = (vars_vec[idx_i] - vars_vec[idx_j])**2 + \
                          (vars_vec[idx_i + 1] - vars_vec[idx_j + 1])**2
                
                # Constraint: r_i + r_j <= sqrt(dist_sq)
                # Squared: (r_i + r_j)^2 - dist_sq <= 0
                cons.append((r_i + r_j)**2 - dist_sq)
        return np.array(cons)

    # 3. Setup Constraints for SLSQP
    # SLSQP requires inequality constraints to be >= 0, so we negate them.
    boundary_con = {
        'type': 'ineq',
        'fun': lambda x: -boundary_constraints(x)
    }
    
    overlap_con = {
        'type': 'ineq',
        'fun': lambda x: -non_overlap_constraints(x)
    }

    constraints = [boundary_con, overlap_con]

    # 4. Run Optimization
    # Method 'SLSQP' handles non-linear constraints well.
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        constraints=constraints,
        options={'maxiter': 2000, 'ftol': 1e-9}
    )

    if not result.success:
        print("Warning: Optimization did not fully converge.")

    # 5. Format Output
    optimal_vars = result.x
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        idx = 3 * i
        centers[i, 0] = optimal_vars[idx]
        centers[i, 1] = optimal_vars[idx + 1]
        radii[i] = optimal_vars[idx + 2]
        
    # Enforce non-negative radii (numerical safety)
    radii = np.maximum(radii, 0)

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
