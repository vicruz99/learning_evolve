import numpy as np
import scipy.optimize

def compute_objective(x):
    """
    Objective function for optimization.
    Minimizes negative sum of radii + penalty for overlaps and boundary violations.
    """
    N = 26
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    obj = -np.sum(cr)
    penalty = 0.0
    lam = 800.0
    
    # Pairwise overlap penalty
    for i in range(N):
        for j in range(i + 1, N):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dist = np.sqrt(dx*dx + dy*dy)
            min_dist = cr[i] + cr[j]
            if dist < min_dist:
                penalty += (min_dist - dist)**2
                
    # Boundary violation penalty
    for i in range(N):
        r = cr[i]
        x_c, y_c = cx[i], cy[i]
        d = min(x_c, 1.0 - x_c, y_c, 1.0 - y_c)
        if r > d:
            penalty += (r - d)**2
            
    return obj + lam * penalty

def run_packing():
    N = 26
    centers = np.zeros((N, 2))
    idx = 0
    
    # Hexagonal lattice initialization
    for i in range(6):
        y = (i + 1) / 7.0
        for j in range(5):
            if idx >= N: break
            x = (j + 1) / 6.0
            if i % 2 == 1:
                x += 1.0 / 12.0
            centers[idx] = [x, y]
            idx += 1
            
    # Flatten initial state: x1, y1, r1, x2, y2, r2, ...
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = 0.005  # Small initial radius
        
    # Bounds: x,y in [0,1], r in [1e-6, 0.5]
    bounds = []
    for i in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
        
    # Optimize
    res = scipy.optimize.minimize(compute_objective, x0, bounds=bounds, method='L-BFGS-B',
                                  options={'maxiter': 15000, 'ftol': 1e-13, 'gtol': 1e-13})
                                  
    final_centers = res.x[0::3].reshape(-1, 2)
    final_radii = res.x[2::3]
    
    # Post-processing: Strict constraint enforcement
    for i in range(N):
        final_centers[i, 0] = np.clip(final_centers[i, 0], 0.0, 1.0)
        final_centers[i, 1] = np.clip(final_centers[i, 1], 0.0, 1.0)
        d = min(final_centers[i, 0], 1.0 - final_centers[i, 0],
                final_centers[i, 1], 1.0 - final_centers[i, 1])
        final_radii[i] = min(final_radii[i], d)
        
    # Iteratively resolve residual overlaps
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                dx = final_centers[i, 0] - final_centers[j, 0]
                dy = final_centers[i, 1] - final_centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                sum_r = final_radii[i] + final_radii[j]
                if dist < sum_r - 1e-12:
                    overlap = sum_r - dist
                    final_radii[i] -= overlap * 0.5
                    final_radii[j] -= overlap * 0.5
                    changed = True
        if not changed:
            break
            
    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
