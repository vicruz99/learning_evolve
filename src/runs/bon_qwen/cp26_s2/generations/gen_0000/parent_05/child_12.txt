To maximize the sum of radii for 26 circles in a unit square, we will employ a multi-stage optimization strategy.

1.  **Equal Radii Baseline:** We first optimize for a configuration where all circles have the same radius. This simplifies the problem to finding the optimal placement of 26 points to maximize the minimum separation distance (and distance to boundaries). We will initialize this with a dense hexagonal lattice pattern, as it provides the highest theoretical packing density. We will use `scipy.optimize.minimize` (SLSQP method) to find the maximum possible equal radius.
2.  **Variable Radii Refinement:** Using the positions found in the first stage as a warm start, we relax the equal radii constraint. We allow each circle to have its own radius $r_i$ and optimize the sum $\sum r_i$ directly. This allows the algorithm to exploit irregular gaps or boundary corners to potentially squeeze out more area.
3.  **Robustness:** To avoid local minima, we will run the optimization multiple times with different random perturbations to the initial hexagonal grid and keep the best result found.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Helper to create initial hexagonal grid
    def get_initial_centers(method='hex'):
        centers = np.zeros((n, 2))
        if method == 'hex':
            # Hexagonal packing initialization
            # Estimate spacing to fit 26 circles
            # Area per circle approx 1/26. Side of hex cell s ~ sqrt(1/26 * 2/sqrt(3))
            s = 0.22 
            k = 0
            for row in range(10):
                y = row * s * np.sqrt(3) / 2
                if y > 0.9: break
                x_start = 0.1 if row % 2 == 0 else 0.2
                for col in range(10):
                    x = x_start + col * s
                    if x > 0.9: break
                    centers[k] = [x, y]
                    k += 1
                    if k >= n: break
                if k >= n: break
        else:
            # Random initialization inside [0,1]
            centers = np.random.rand(n, 2)
        # Clip to safe zone initially
        centers = np.clip(centers, 0.05, 0.95)
        return centers

    def objective(vars):
        # vars layout: x1, y1, r1, x2, y2, r2, ...
        radii = vars[2::3]
        return -np.sum(radii)

    def get_constraints(centers_shape):
        cons = []
        n = centers_shape[0]
        
        # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
        # Inequality form: fun(vars) >= 0
        for i in range(n):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]})
            # 1 - x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]})
            # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - v[idx_r]})
            # 1 - y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]})
            
        # Overlap constraints: dist^2 >= (r1 + r2)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
                idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
                
                def make_cons(i, j):
                    def fun(v):
                        dx = v[idx_xi] - v[idx_xj]
                        dy = v[idx_yi] - v[idx_yj]
                        dr = v[idx_ri] + v[idx_rj]
                        return (dx**2 + dy**2) - (dr**2)
                    return fun

                cons.append({'type': 'ineq', 'fun': make_cons(i, j)})
        
        return cons

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    # Run optimization multiple times to escape local minima
    for seed in range(5):
        np.random.seed(seed)
        centers_init = get_initial_centers('hex')
        # Add some noise
        centers_init += np.random.uniform(-0.02, 0.02, centers_init.shape)
        centers_init = np.clip(centers_init, 0.02, 0.98)
        
        # Initial radii: small value
        r_init = np.full(n, 0.05)
        
        # Construct initial vector
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = r_init[i]

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=get_constraints(centers_init.shape),
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = np.zeros((n, 2))
                best_radii = np.zeros(n)
                for i in range(n):
                    best_centers[i, 0] = res.x[3*i]
                    best_centers[i, 1] = res.x[3*i+1]
                    best_radii[i] = res.x[3*i+2]
        except Exception as e:
            continue

    # Fallback if optimization failed (should not happen with good init)
    if best_sum < 0:
        centers = np.random.rand(n, 2)
        radii = np.full(n, 0.01)
        return centers, radii, np.sum(radii)

    return best_centers, best_radii, best_sum
