import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

    max_iter = 5000
    dt = 0.01
    decay = 0.9995
    repulsion_strength = 10.0
    expansion_rate = 0.001
    
    # Run simulation
    for step in range(max_iter):
        force = np.zeros_like(centers)
        radii_change = np.zeros(n_circles)
        
        # 1. Calculate repulsive forces from overlaps
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                req_dist = radii[i] + radii[j]
                
                if dist < req_dist and dist > 1e-9:
                    # Overlap detected
                    overlap = req_dist - dist
                    # Force direction
                    f_vec = diff / dist
                    # Magnitude proportional to overlap
                    f_mag = overlap * repulsion_strength
                    force[i] += f_vec * f_mag
                    force[j] -= f_vec * f_mag
        
        # 2. Calculate boundary forces
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x - r < 0:
                force[i, 0] += (-(x - r)) * repulsion_strength
            # Right wall
            if x + r > 1:
                force[i, 0] -= (x + r - 1) * repulsion_strength
            # Bottom wall
            if y - r < 0:
                force[i, 1] += (-(y - r)) * repulsion_strength
            # Top wall
            if y + r > 1:
                force[i, 1] -= (y + r - 1) * repulsion_strength
                
        # 3. Update positions
        centers += dt * force
        dt *= decay
        
        # 4. Expand radii if no significant overlap/boundary violation
