import numpy as np
from scipy.optimize import minimize

def compute_constraints(v):
    n = 26
    r = v[-1]
    c = v[:-1].reshape(n, 2)
    vals = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    vals.extend(c[:, 0] - r)
    vals.extend(1 - c[:, 0] - r)
    vals.extend(c[:, 1] - r)
    vals.extend(1 - c[:, 1] - r)
    # Pairwise non-overlap: dist^2 >= 4r^2
    for i in range(n):
        for j in range(i + 1, n):
            vals.append(np.sum((c[i] - c[j]) ** 2) - 4 * r * r)
    return np.array(vals)

def obj_func(v):
    return -v[-1]

def run_packing():
    np.random.seed(42)
    n = 26
    best_result = None
    best_sum_r = -1.0
    
    # Multiple restarts to escape local minima and find the globally densest packing
    for seed in range(20):
        np.random.seed(seed)
        # Initial random placement inside a shrunken square to ensure strict feasibility at start
        centers = np.random.rand(n, 2) * 0.6 + 0.2
        r0 = 0.04
        v0 = np.concatenate([centers.flatten(), [r0]])

        cons = {'type': 'ineq', 'fun': compute_constraints}

    radii = np.full(n, r)

    return centers, radii, np.sum(radii)
