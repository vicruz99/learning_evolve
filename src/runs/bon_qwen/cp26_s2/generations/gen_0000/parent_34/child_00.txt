import numpy as np
from scipy.optimize import differential_evolution, minimize

N_CIRCLES = 26

def packing_objective(vars):
    """Objective function to minimize: -sum(radii) + penalties for violations."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n = N_CIRCLES
    
    # Boundary penalties
    pen = 0.0
    pen += np.sum(np.maximum(0.0, r - x)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    pen += np.sum(np.maximum(0.0, r - y)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap penalties (vectorized)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.hypot(dx, dy)
    r_sum = r[:, None] + r[None, :]
    
    # Only consider upper triangle pairs
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap = r_sum[mask] - dists[mask]
    pen += np.sum(np.maximum(0.0, overlap)**2)
    
    return -np.sum(r) + 2000.0 * pen

def run_packing():
    n = N_CIRCLES
    # Bounds: x, y in [0, 1], r in [0.01, 0.25]
    bounds = [(0.0, 1.0) for _ in range(3 * n)]
    for i in range(n):
        bounds[2 * i + 2] = (0.01, 0.25)
        
    # Global search with Differential Evolution
    de_res = differential_evolution(
        packing_objective, 
        bounds, 
        popsize=15, 
        maxiter=250,
        mutation=(0.5, 1.2), 
        recombination=0.9, 
        seed=42,
        tol=1e-7, 
        updating='deferred', 
        polish=False
    )
    
    # Local refinement with L-BFGS-B
    local_res = minimize(
        packing_objective, 
        de_res.x, 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    
    best_vars = local_res.x
    
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Project to strictly feasible region
    radii = np.clip(radii, 1e-6, 0.5)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    # Apply tiny margin to pass validator's 1e-12 tolerance robustly
    radii -= 1e-9
    radii = np.maximum(radii, 1e-6)
    
    sum_r = np.sum(radii)
    return centers, radii, sum_r
