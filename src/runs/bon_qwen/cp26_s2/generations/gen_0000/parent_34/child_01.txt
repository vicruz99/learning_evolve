import numpy as np
from scipy.optimize import minimize

def objective(params, n, mu):
    """
    Computes the objective: -sum(radii) + mu * (overlap_violations + boundary_violations)
    """
    params = params.reshape(n, 3)
    cost = 0.0
    
    # Pairwise overlap penalties
    for i in range(n):
        xi, yi, ri = params[i]
        for j in range(i + 1, n):
            xj, yj, rj = params[j]
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            overlap = ri + rj - dist
            if overlap > 0:
                cost += mu * overlap * overlap
                
    # Boundary constraint penalties
    for i in range(n):
        xi, yi, ri = params[i]
        # Left boundary
        val = xi - ri
        if val < 0:
            cost += mu * val * val
        # Right boundary
        val = xi + ri - 1.0
        if val > 0:
            cost += mu * val * val
        # Bottom boundary
        val = yi - ri
        if val < 0:
            cost += mu * val * val
        # Top boundary
        val = yi + ri - 1.0
        if val > 0:
            cost += mu * val * val
            
    return -np.sum(params[:, 2]) + cost

def is_valid(centers, radii, n):
    """
    Strictly checks if circles are inside the unit square and non-overlapping.
    Uses a tolerance of 1e-9 for numerical safety.
    """
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    # Boundary check
    if np.any(x - r < -1e-9) or np.any(x + r > 1.0 + 1e-9) or \
       np.any(y - r < -1e-9) or np.any(y + r > 1.0 + 1e-9):
        return False
        
    # Overlap check
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.linalg.norm(diff, axis=2)
    r_sum = radii[:, None] + radii[None, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return np.all(dists[mask] >= r_sum[mask] - 1e-9)

def run_packing():
    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: centers in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    num_restarts = 40
    mu_start = 80.0
    mu_factor = 2.8
    mu_steps = 14
    
    for _ in range(num_restarts):
        # Random initialization
        rng = np.random.default_rng(np.random.randint(0, 1000000))
        x_init = rng.uniform(0.05, 0.95, (n, 2))
        r_init = rng.uniform(0.02, 0.04, n)
        x0 = np.hstack([x_init, r_init.reshape(-1, 1)]).flatten()
        
        current_x = x0.copy()
        current_mu = mu_start
        
        # Penalty annealing
        for _ in range(mu_steps):
            try:
                res = minimize(objective, x0=current_x, args=(n, current_mu),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 600, 'ftol': 1e-14, 'gtol': 1e-12})
                current_x = res.x
            except Exception:
                pass
            current_mu *= mu_factor
            
        params = current_x.reshape(n, 3)
        centers = params[:, :2]
        radii = params[:, 2]
        sum_r = np.sum(radii)
        
        # Keep best valid configuration
        if is_valid(centers, radii, n):
            if sum_r > best_sum_r:
                best_sum_r = sum_r
                best_centers = centers.copy()
                best_radii = radii.copy()
        elif best_centers is None:
            # Fallback if no valid packing found yet
            best_sum_r = sum_r
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, best_sum_r
