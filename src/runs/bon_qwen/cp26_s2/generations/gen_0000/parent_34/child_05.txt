import numpy as np
from scipy.optimize import linprog
import warnings

warnings.filterwarnings("ignore")

def solve_radii_lp(centers):
    """Solve LP to maximize sum of radii given fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)  # Minimize -sum(r) => Maximize sum(r)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        for bound_val in [centers[i, 0], 1.0 - centers[i, 0], 
                          centers[i, 1], 1.0 - centers[i, 1]]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(bound_val)
            
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i,j)
    dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=2))
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x if res.success else np.full(n, 1e-6)
