import numpy as np
from scipy.optimize import minimize

def compute_available_radii(centers):
    """Compute the maximum possible radius for each circle given fixed centers."""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    min_dist_other = np.min(dist, axis=1)
    
    dist_bound = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1 - centers[:, 1]))
    return 0.5 * np.minimum(min_dist_other, dist_bound)

def objective(centers_flat):
    """Objective to minimize: negative sum of available radii."""
    centers = centers_flat.reshape(26, 2)
    radii = compute_available_radii(centers)
    return -np.sum(radii)

def run_simulation(centers_init):
    """Repulsion-based packing simulation to find a good local configuration."""
    n = centers_init.shape[0]
    centers = centers_init.copy()
    radii = np.full(n, 0.02)
    
    for step in range(3000):
        radii *= 1.0004
        radii = np.clip(radii, 1e-6, 0.49)
        
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dist, np.inf)
        
        overlap = radii[:, np.newaxis] + radii[np.newaxis, :] - dist
        push_mag = np.where(overlap > 0, overlap * 0.5, 0.0)
        
        dir_vec = np.where(overlap > 0, diff / (dist[:, :, np.newaxis] + 1e-12), 0.0)
        
        force = np.sum(push_mag[:, :, np.newaxis] * dir_vec, axis=1)
        centers -= force
        
        # Enforce boundary constraints
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
        
        # Periodic jitter to escape local minima
        if step % 50 == 0:
            noise = np.random.uniform(-1e-3, 1e-3, centers.shape)
            centers += noise
            centers = np.clip(centers, radii, 1 - radii)
            
    return centers

def run_packing():

    candidates.append(np.random.uniform(0.1, 0.9, (n, 2)))
    
    # 3. Uniform grid initialization
    c3 = np.zeros((n, 2))
    idx = 0
    for i in range(5):

                idx += 1
    if idx < n:
        for i in range(idx, n):
            c3[i] = np.random.uniform(0.1, 0.9, 2)
    candidates.append(c3)

    best_sum = -np.inf

        c_opt = run_simulation(c_init)

        bounds = [(0.001, 0.999) for _ in range(2*n)]
        try:
