The problem of packing 26 circles in a unit square to maximize the sum of radii is a continuous, non-convex optimization problem. The theoretical maximum density for equal circles approaches $\pi/\sqrt{12} \approx 0.9069$, suggesting an optimal equal radius around $0.101$ (sum $\approx 2.636$). 

Strategy:
1. **Initialization**: Use a hexagonal lattice grid. This is the densest packing pattern for circles in the plane and provides an excellent feasible starting point close to the optimum.
2. **Optimization Formulation**: 
   - Variables: $x_i, y_i, r_i$ for each circle $i=1,\dots,26$. Total 78 variables.
   - Objective: Maximize $\sum r_i$ (minimize $-\sum r_i$).
   - Constraints: 
     - Boundary: $r_i \le x_i \le 1-r_i$, $r_i \le y_i \le 1-r_i$
     - Non-overlap: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ for all $i<j$
   - These are smooth inequality constraints suitable for `scipy.optimize`.
3. **Solver**: Use `SLSQP` (Sequential Least Squares Programming) which handles nonlinear constraints and bounds efficiently. Constraints are vectorized using NumPy broadcasting for performance.
4. **Robustness**: Run the optimizer 5 times with different random perturbations to the initial hex grid to escape local minima. The best result is retained.
5. **Compliance**: All helper functions are defined at the top level, no closures or lambdas are used, and the function signature matches requirements.

import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def compute_constraints(x, n):
    """Compute boundary and non-overlap constraints for n circles."""
    xc = x[0::3]
    yc = x[1::3]
    rc = x[2::3]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c_boundary = np.concatenate([
        xc - rc, 1 - xc - rc, yc - rc, 1 - yc - rc
    ])
    
    # Vectorized overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dr = rc[:, None] + rc[None, :]
    
    dist_sq = dx**2 + dy**2
    rad_sum_sq = dr**2
    
    # Lower triangle indices (excluding diagonal)
    idx = np.tril_indices(n, -1)
    c_overlap = dist_sq[idx] - rad_sum_sq[idx]
    
    return np.concatenate([c_boundary, c_overlap])

def objective_func(x, n):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(x[2::3])

def gen_hex_grid(n, r):
    """Generate initial positions on a hexagonal lattice."""
    pts = []
    y = r
    row = 0
    while len(pts) < n:
        shift = (row % 2) * r
        x = r + shift
        while x <= 1 - r:
            pts.append([x, y])
            x += 2 * r
            if len(pts) == n: break
        y += np.sqrt(3) * r
        row += 1
        if len(pts) >= n: break
    return np.array(pts[:n])

# Top-level wrappers to avoid closures/lambdas
def constr_26(x):
    return compute_constraints(x, 26)

def obj_26(x):
    return objective_func(x, 26)

def run_packing():
    n = 26
    best_val = -np.inf
    best_x = None
    
    # Initial radius guess based on area density approx
    r_init = 0.095
    centers_base = gen_hex_grid(n, r_init)
    
    # Bounds: x,y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0)]*n + [(0.0, 1.0)]*n + [(1e-6, 0.5)]*n
    
    nl_constr = NonlinearConstraint(constr_26, 0, np.inf)
    
    # Multi-start optimization to avoid local minima
    for i in range(5):
        np.random.seed(i * 17 + 123)
        noise = np.random.uniform(-0.015, 0.015, size=(n, 3))
        
        x0 = np.zeros(3*n)
        x0[0::3] = np.clip(centers_base[:, 0] + noise[:, 0], 0.0, 1.0)
        x0[1::3] = np.clip(centers_base[:, 1] + noise[:, 1], 0.0, 1.0)
        x0[2::3] = np.clip(r_init + noise[:, 2], 1e-6, 0.5)
        
        try:
            res = minimize(obj_26, x0, method='SLSQP', bounds=bounds, 
                          constraints=nl_constr, options={'maxiter': 5000, 'ftol': 1e-14})
            if -res.fun > best_val:
                best_val = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_x is None:
        best_x = np.zeros(3*n)
        best_x[0::3] = centers_base[:, 0]
        best_x[1::3] = centers_base[:, 1]
        best_x[2::3] = r_init
        best_val = n * r_init
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    return centers, radii, best_val
