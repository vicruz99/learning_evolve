import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Solves the circle packing problem for 26 circles in a unit square.
    Returns centers, radii, and the sum of radii.

    radii = np.zeros(n)

    dx = 2 * r_init
    dy = math.sqrt(3) * r_init
    
    points = []
    y = r_init
    row = 0
    while True:
        if len(points) >= n:
            break
        x = r_init
        col = 0
        # Shift odd rows
        if row % 2 == 1:
            x += r_init # shift by r, which is dx/2

            points.append([x, y])
            if len(points) >= n:
                break
            x += dx
            col += 1
        y += dy
        row += 1
        
    # If we have enough points, take the first n
    if len(points) >= n:
        centers = np.array(points[:n])
    else:
        # Fallback to grid if hex generation failed (unlikely)
        # Fill remaining with grid
        pass
        
    # Initial radii
    radii[:] = r_init
    
    # 2. Optimization
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 3 * n = 78
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    # x, y in [0, 1]
    # r >= 0 (practically we can set a small lower bound like 1e-5 to avoid zero)
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 1.0)])

    # 1. Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # Equivalent to: r <= x <= 1-r
    # r - x <= 0
    # x + r - 1 <= 0
    # r - y <= 0
    # y + r - 1 <= 0
    
    # 2. Non-overlap: dist(i, j) >= r_i + r_j
    # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
    # -(xi - xj)^2 - (yi - yj)^2 + (ri + rj)^2 <= 0
    
    def constraint_bound_x_less_r(idx):
        # x_i - r_i >= 0 => -x_i + r_i <= 0
        return {
            'type': 'ineq',
            'fun': lambda v, i=idx: v[3*i] - v[3*i+2] # x - r >= 0
        }

    def constraint_bound_x_plus_r(idx):
        # x_i + r_i <= 1 => x_i + r_i - 1 <= 0 => 1 - x_i - r_i >= 0
        return {
            'type': 'ineq',
            'fun': lambda v, i=idx: 1.0 - v[3*i] - v[3*i+2]
        }

    def constraint_bound_y_less_r(idx):
        # y_i - r_i >= 0
        return {
            'type': 'ineq',
            'fun': lambda v, i=idx: v[3*i+1] - v[3*i+2]
        }

    def constraint_bound_y_plus_r(idx):
        # y_i + r_i <= 1
        return {
            'type': 'ineq',
            'fun': lambda v, i=idx: 1.0 - v[3*i+1] - v[3*i+2]
        }

    def constraint_overlap(i, j):
        # dist^2 - (r_i + r_j)^2 >= 0
        def fun(v):
            xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
            xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            rad_sum_sq = (ri + rj)**2
            return dist_sq - rad_sum_sq
        return {
            'type': 'ineq',
            'fun': fun
        }

    for i in range(n):

        constraints.append(constraint_bound_x_plus_r(i))
        constraints.append(constraint_bound_y_less_r(i))
        constraints.append(constraint_bound_y_plus_r(i))

        for j in range(i + 1, n):
            constraints.append(constraint_overlap(i, j))

    def objective(v):
        s = 0.0
        for i in range(n):
            s -= v[3*i+2]
        return s

    res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,

        final_radii[i] = res.x[3*i+2]
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
