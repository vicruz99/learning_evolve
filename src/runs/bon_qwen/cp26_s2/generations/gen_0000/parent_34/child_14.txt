import numpy as np
from scipy.optimize import minimize

# Number of circles
N_CIRCLES = 26

def objective_function(params):
    """
    Objective function to minimize: negative sum of radii.
    params shape: (3 * N_CIRCLES,) -> [x1, y1, r1, x2, y2, r2, ...]
    """
    # Radii are at indices 2, 5, 8, ...
    radii = params[2::3]
    return -np.sum(radii)

def constraint_function(params):
    """
    Constraint function returning an array of values that must be >= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    n = len(params) // 3
    # Reshape to (n, 3) where columns are x, y, r
    p = params.reshape((n, 3))
    x = p[:, 0]
    y = p[:, 1]
    r = p[:, 2]
    
    constraints = []
    
    # 1. Boundary Constraints
    # Circle i must be inside [0, 1] x [0, 1]
    # x - r >= 0  => r <= x
    # 1 - x - r >= 0 => r <= 1 - x
    # y - r >= 0 => r <= y
    # 1 - y - r >= 0 => r <= 1 - y
    constraints.append(x - r)
    constraints.append(1 - x - r)
    constraints.append(y - r)
    constraints.append(1 - y - r)
    
    # 2. Pairwise Non-overlap Constraints
    # Distance^2 >= (r_i + r_j)^2
    # We compute this for all pairs i < j
    # Using broadcasting for efficiency
    # dx shape: (n, n), dy shape: (n, n)
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    
    dist_sq = dx**2 + dy**2
    sum_r_sq = dr**2
    
    # We only need constraints for i < j (upper triangle)
    # Create a mask for upper triangle (excluding diagonal)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    # The constraint is dist_sq - sum_r_sq >= 0
    constraints.append(dist_sq[mask] - sum_r_sq[mask])
    
    # Concatenate all constraints into a single vector
    return np.concatenate(constraints)

def run_packing():
    """
    Runs the optimization to pack 26 circles in a unit square.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    # Structure: [x1, y1, r1, x2, y2, r2, ...]
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # Constraint definition for SLSQP
    cons = {'type': 'ineq', 'fun': constraint_function}
    
    best_sum_radii = 0.0
    best_params = None
    
    # Run optimization with multiple random restarts to avoid local minima
    n_restarts = 10
    
    for i in range(n_restarts):
        # Generate random initial positions for centers
        # Use a slightly perturbed grid or random points
        # Random points with very small initial radii ensure feasibility
        np.random.seed(i * 42 + 123)
        init_centers = np.random.rand(N_CIRCLES, 2)
        init_radii = np.full(N_CIRCLES, 1e-4) # Small radius to start feasible
        
        # Interleave x, y, r
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = init_centers[:, 0]
        x0[1::3] = init_centers[:, 1]
        x0[2::3] = init_radii
        
        try:
            res = minimize(
                objective_function,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'disp': False, 'maxiter': 1000}
            )
            
            if res.success or -res.fun > best_sum_radii:
                current_sum = -res.fun
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_params = res.x.copy()
        except Exception as e:
            # Continue if one restart fails
            continue
            
    if best_params is None:
        # Fallback if optimization failed completely
        # Return a valid trivial packing
        centers = np.zeros((N_CIRCLES, 2))
        radii = np.zeros(N_CIRCLES)
        return centers, radii, 0.0

    # Extract results
    p = best_params.reshape((N_CIRCLES, 3))
    centers = p[:, :2]
    radii = p[:, 2]
    
    # Ensure radii are non-negative (numerical safety)
    radii = np.maximum(radii, 0.0)
    
    # Recalculate sum to be precise
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

# Final program block
