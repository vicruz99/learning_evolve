    *   **Penalty Method:** Since explicit inequality constraints are numerous (325 pairwise non-overlap constraints), I will incorporate them into the objective function as a penalty term. The penalty will heavily weight any overlap or boundary violation, forcing the optimizer to find a valid packing.
3.  **Validation & Selection:** After optimization, each candidate solution will be strictly validated against the non-overlap and boundary constraints. The valid configuration with the highest sum of radii will be returned.
4.  **Refinement:** The process will be repeated with random perturbations to fine-tune the result.

This approach combines geometric intuition (initial configurations) with robust numerical optimization to discover high-density packings.

import numpy as np
import math
from scipy.optimize import minimize

def compute_loss(x, n):
    """
    Computes the objective function (negative sum of radii) plus penalty terms
    for constraint violations.
    
    Args:
        x: np.array of shape (3*n,) containing [x1, y1, r1, x2, y2, r2, ...]
        n: number of circles
        
    Returns:
        loss: scalar value
    """
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Objective: minimize -sum(radii)
    loss = -np.sum(radii)
    
    # Penalty weight - needs to be large enough to enforce constraints
    W = 10000.0
    
    # Boundary constraints penalties
    # Circle i must satisfy: r_i <= x_i <= 1 - r_i  and  r_i <= y_i <= 1 - r_i
    
    # Left wall: r - x <= 0
    v = radii - centers[:, 0]
    loss += W * np.sum(np.maximum(0, v)**2)
    
    # Right wall: x + r - 1 <= 0
    v = centers[:, 0] + radii - 1
    loss += W * np.sum(np.maximum(0, v)**2)
    
    # Bottom wall: r - y <= 0
    v = radii - centers[:, 1]
    loss += W * np.sum(np.maximum(0, v)**2)
    
    # Top wall: y + r - 1 <= 0
    v = centers[:, 1] + radii - 1
    loss += W * np.sum(np.maximum(0, v)**2)
    
    # Overlap constraints penalties
    # For all i < j: dist(i, j) >= r_i + r_j  =>  r_i + r_j - dist(i, j) <= 0
    
    # Compute pairwise distances efficiently
    # centers shape: (n, 2)
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Compute sum of radii for each pair
    # rad_sum shape: (n, n)
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Calculate overlap amount (positive means overlap)
    overlap = rad_sum - dists
    
    # The matrix is symmetric, so we sum all positive overlaps and divide by 2
    # Diagonal is 0, so it doesn't contribute
    pos_overlap = np.maximum(0, overlap)
    loss += W * 0.5 * np.sum(pos_overlap**2)
    
    return loss

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Helper to validate a packing strictly
    def is_valid(centers, radii):
        # Check NaN
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False
        # Check radii non-negative
        if np.any(radii < 0):
            return False
        # Check boundaries
        # x - r >= 0 => r <= x
        if np.any(radii > centers[:, 0] + 1e-9): return False
        # x + r <= 1 => r <= 1 - x
        if np.any(radii > 1 - centers[:, 0] + 1e-9): return False
        # y - r >= 0 => r <= y
        if np.any(radii > centers[:, 1] + 1e-9): return False
        # y + r <= 1 => r <= 1 - y
        if np.any(radii > 1 - centers[:, 1] + 1e-9): return False
        
        # Check overlaps
        # dist >= r_i + r_j
        # Using broadcasting
        # centers: (n, 2)
        dists = np.sqrt(np.sum((centers[:, None] - centers[None, :])**2, axis=2))
        sums_r = radii[:, None] + radii[None, :]
        # Check if any distance is strictly less than sum of radii (with tolerance)
        # If dist < sum_r - epsilon, it's an overlap
        if np.any(dists < sums_r - 1e-9):
            return False
        return True

    # Generate initial configurations
    starts = []
    
    # 1. Grid 5x5 (25 circles) + 1 in a gap
    # 5x5 grid of radius 0.1 fits exactly in [0,1]^2
    # Centers at 0.1, 0.3, 0.5, 0.7, 0.9
    c1 = []
    r1 = []
    for i in range(5):
        for j in range(5):
            c1.append([0.1 + i*0.2, 0.1 + j*0.2])
            r1.append(0.1)
    # Add 26th circle in a gap. Center of first cell is (0.2, 0.2).
    # Distance to (0.1, 0.1) is sqrt(0.02) ~ 0.1414.
    # Available radius ~ 0.0414.
    c1.append([0.2, 0.2])
    r1.append(0.04)
    starts.append((np.array(c1), np.array(r1)))
    
    # 2. Hexagonal packing approximation
    c2 = []
    r2 = []
    y = 0.1
    row = 0
    while len(c2) < n:
        if row % 2 == 0:
            # Even row: centers at 0.1, 0.3, ...
            xs = np.arange(0.1, 1.0, 0.2)
        else:
            # Odd row: shifted by 0.1? No, hex shift is 0.1 * sqrt(3) ~ 0.173 in y
            # x-shift should be 0.1 (half of 0.2 spacing)
            xs = np.arange(0.2, 1.0, 0.2)
        
        for x in xs:
            if len(c2) < n:
                # Check if point is roughly inside
                if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                    c2.append([x, y])
                    r2.append(0.09) # Start slightly smaller
            else:
                break
        y += 0.1 * math.sqrt(3)
        if y + 0.1 > 1.0:
            break
        row += 1
    
    # Fill remaining if needed
    while len(c2) < n:
        c2.append([0.5, 0.5])
        r2.append(0.01)
    starts.append((np.array(c2), np.array(r2)))
    
    # 3. Random initialization
    c3 = np.random.rand(n, 2) * 0.8 + 0.1
    r3 = np.full(n, 0.05)
    starts.append((c3, r3))

    # Define bounds for optimization
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r

    # Run optimization for each start configuration
    for c_init, r_init in starts:
        x0 = np.concatenate([c_init.flatten(), r_init])
        
        # Perform multiple attempts with small perturbations
        for _ in range(5):
            x0_perturbed = x0 + np.random.normal(0, 0.001, x0.shape)
            
            try:
                res = minimize(compute_loss, x0_perturbed, args=(n,), 
                               method='L-BFGS-B', bounds=bounds, 
                               options={'maxiter': 5000, 'ftol': 1e-15})
                
                c_res = res.x[:2*n].reshape(n, 2)
                r_res = res.x[2*n:]
                
                # Validate the result
                if is_valid(c_res, r_res):
                    s = np.sum(r_res)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_res.copy()
                        best_radii = r_res.copy()
            except Exception:
                pass

    # Fallback if no valid packing found (unlikely given grid init)
    if best_centers is None:
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.05)
        k = 0
        for i in range(5):
            for j in range(5):
                if k < n:
                    centers[k] = [0.1 + i*0.2, 0.1 + j*0.2]
                    radii[k] = 0.1
                    k += 1
        if k < n:
            centers[k] = [0.5, 0.5]
            radii[k] = 0.01
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
