import numpy as np
from scipy.optimize import minimize
import math
import random


def neg_sum_radii(params, n):
    """Objective: negative sum of radii (to be minimized)"""
    return -np.sum(params[2 * n:])


def bound_left(params, n, i):
    """Constraint: x_i - r_i >= 0"""
    return params[2 * i] - params[2 * n + i]


def bound_right(params, n, i):
    """Constraint: 1 - x_i - r_i >= 0"""
    return 1.0 - params[2 * i] - params[2 * n + i]


def bound_bottom(params, n, i):
    """Constraint: y_i - r_i >= 0"""
    return params[2 * i + 1] - params[2 * n + i]


def bound_top(params, n, i):
    """Constraint: 1 - y_i - r_i >= 0"""
    return 1.0 - params[2 * i + 1] - params[2 * n + i]


def overlap_constraint(params, n, i, j):
    """Constraint: distance(i,j) >= r_i + r_j"""
    x1 = params[2 * i]
    y1 = params[2 * i + 1]
    x2 = params[2 * j]
    y2 = params[2 * j + 1]
    r1 = params[2 * n + i]
    r2 = params[2 * n + j]
    dx = x1 - x2
    dy = y1 - y2
    return math.sqrt(dx * dx + dy * dy) - r1 - r2


def make_constraints(n):
    """Create all constraints for the optimization problem"""
    constraints = []
    
    # Boundary constraints (4 per circle)
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': bound_left, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': bound_right, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': bound_bottom, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': bound_top, 'args': (n, i)})
    
    # Overlap constraints (n*(n-1)/2 pairs)
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': overlap_constraint, 'args': (n, i, j)})
    
    return constraints


def make_bounds(n):
    """Create bounds for all parameters"""
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))  # x coordinate
        bounds.append((0.0, 1.0))  # y coordinate
    for _ in range(n):
        bounds.append((1e-10, 0.5))  # radius (small positive lower bound)
    return bounds


def hex_init(n, r):
    """Initialize circles in a hexagonal lattice pattern"""
    centers = []
    row = 0
    while len(centers) < n:
        col = 0
        while len(centers) < n:
            x = r + col * 2.0 * r
            y = r + row * r * math.sqrt(3.0)
            if col % 2 == 1:
                y += r * math.sqrt(3.0) / 2.0
            if x + r <= 1.0 and y + r <= 1.0:
                centers.append([x, y])
            col += 1
            if col > 20:
                break
        row += 1
        if row > 20:
            break
    
    while len(centers) < n:
        centers.append([0.5, 0.5])
    
    return np.array(centers[:n]), np.full(n, r)


def grid_init(n, r):
    """Initialize circles in a square grid pattern"""
    centers = []
    side = int(math.ceil(math.sqrt(n)))
    spacing_x = (1.0 - 2.0 * r) / max(side - 1, 1)
    spacing_y = (1.0 - 2.0 * r) / max(side - 1, 1)
    
    row = 0
    while len(centers) < n:
        col = 0
        while len(centers) < n:
            x = r + col * spacing_x
            y = r + row * spacing_y
            if x + r <= 1.0 and y + r <= 1.0:
                centers.append([x, y])
            col += 1
        row += 1
        if row > 20:
            break
    
    while len(centers) < n:
        centers.append([0.5, 0.5])
    
    return np.array(centers[:n]), np.full(n, r)


def perturbed_init(n, base_centers, sigma):
    """Create a perturbed version of an initial configuration"""
    centers = base_centers.copy()
    for i in range(n):
        centers[i, 0] += np.random.normal(0, sigma)
        centers[i, 1] += np.random.normal(0, sigma)
        centers[i, 0] = np.clip(centers[i, 0], 0.01, 0.99)
        centers[i, 1] = np.clip(centers[i, 1], 0.01, 0.99)
    return centers


def run_optimization(n, centers, radii, maxiter=3000):
    """Run SLSQP optimization from a given initial configuration"""
    params0 = np.concatenate([centers.flatten(), radii])
    constraints = make_constraints(n)
    bounds = make_bounds(n)
    
    result = minimize(
        neg_sum_radii,
        params0,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    
    if result.success or result.fun < 0:
        optimal = result.x
        opt_centers = optimal[:2 * n].reshape(n, 2)
        opt_radii = optimal[2 * n:]
        return opt_centers, opt_radii, -result.fun
    
    return centers, radii, np.sum(radii)


def refine_with_lp(centers, n):
    """Given positions, compute optimal radii using linear programming approach"""
    from scipy.optimize import linprog
    
    # Maximize sum of radii = minimize negative sum
    c = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    A_ub_rows = []
    b_ub_rows = []
    
    for i in range(n):
        x, y = centers[i]
        unit_i = np.zeros(n)
        unit_i[i] = 1.0
        
        # r_i <= x_i
        A_ub_rows.append(unit_i.copy())
        b_ub_rows.append(x)
        # r_i <= 1 - x_i
        A_ub_rows.append(unit_i.copy())
        b_ub_rows.append(1.0 - x)
        # r_i <= y_i
        A_ub_rows.append(unit_i.copy())
        b_ub_rows.append(y)
        # r_i <= 1 - y_i
        A_ub_rows.append(unit_i.copy())
        b_ub_rows.append(1.0 - y)
    
    # Overlap: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_rows.append(dist)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return np.full(n, 0.05)


def iterative_refine(centers, radii, n, num_iter=20):
    """Iteratively refine: optimize radii with LP, then repel overlapping circles"""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(num_iter):
        # Step 1: Compute optimal radii for current positions
        radii = refine_with_lp(centers, n)
        
        # Step 2: Apply repulsive forces to separate overlapping circles
        lr = 0.002 / (1.0 + iteration * 0.1)
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-10:
                    dist = 1e-10
                    diff = np.array([1e-6, 0.0])
                
                desired_dist = radii[i] + radii[j] + 1e-6
                if dist < desired_dist:
                    # Repulsive force proportional to overlap
                    force_mag = (desired_dist - dist) / (dist * dist + 1e-12)
                    force = force_mag * diff / dist
                    forces[i] += force
                    forces[j] -= force
            
            # Boundary repulsion
            x, y = centers[i]
            r = radii[i]
            if x < r:
                forces[i, 0] += (r - x) * 50.0
            if x > 1.0 - r:
                forces[i, 0] -= (x - (1.0 - r)) * 50.0
            if y < r:
                forces[i, 1] += (r - y) * 50.0
            if y > 1.0 - r:
                forces[i, 1] -= (y - (1.0 - r)) * 50.0
        
        centers += lr * forces
        
        # Clip to valid range
        for i in range(n):
            r = max(radii[i], 0.01)
            centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
        
        # Recompute radii after movement
        radii = refine_with_lp(centers, n)
    
    return centers, radii
