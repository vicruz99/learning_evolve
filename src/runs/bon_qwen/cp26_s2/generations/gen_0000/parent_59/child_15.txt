import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to find maximum valid radii.
    Maximize sum(r_i)
    Subject to:
        r_i + r_j <= dist(centers[i], centers[j])
        r_i <= dist(centers[i], boundary)
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum(r) -> Minimize -sum(r)
    
    # Constraints for linprog: A_ub @ r <= b_ub
    rows = []
    rhs = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        
        # r_i <= x
        rows.append(row.copy())
        rhs.append(x)
        
        # r_i <= 1 - x
        rows.append(row.copy())
        rhs.append(1.0 - x)
        
        # r_i <= y
        rows.append(row.copy())
        rhs.append(y)
        
        # r_i <= 1 - y
        rows.append(row.copy())
        rhs.append(1.0 - y)
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dist)
            
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return np.zeros(n)

def objective(vars, n):
    """
    Objective function: Negative sum of radii.
    vars layout: [x1, y1, r1, x2, y2, r2, ...]
    """
    total_r = 0.0
    for i in range(n):
        total_r += vars[3*i + 2]
    return -total_r

    Inequality constraints >= 0.
    """

        yi = vars[3*i + 1]
        ri = vars[3*i + 2]

        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[3*idx] - v[3*idx + 2]})
        # r <= y => y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx + 1] - v[3*idx + 2]})
        # r <= 1-y => 1 - y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[3*idx + 1] - v[3*idx + 2]})
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            # (ri + rj)^2 <= dist^2  => dist^2 - (ri + rj)^2 >= 0
            # dist^2 = (xi-xj)^2 + (yi-yj)^2

                (v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2 - (v[3*i+2] + v[3*j+2])**2})
                
    return cons

def run_packing():

    best_sum = -1.0
