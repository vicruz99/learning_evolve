import numpy as np
from scipy.optimize import minimize


N_CIRCLES = 26


def objective(x):
    """Negative sum of radii to maximize."""
    n = N_CIRCLES
    return -np.sum(x[2 * n:])


def constraint_array(x):
    """All inequality constraints as g(x) >= 0."""
    n = N_CIRCLES
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    c = []
    
    # Non-overlap: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-16 else 1e-8
            c.append(dist - radii[i] - radii[j])
    
    # Boundary: circle inside [0,1]x[0,1]
    for i in range(n):
        c.append(centers[i, 0] - radii[i])
        c.append(1.0 - centers[i, 0] - radii[i])
        c.append(centers[i, 1] - radii[i])
        c.append(1.0 - centers[i, 1] - radii[i])
    
    return np.array(c)


def hexagonal_init(n, shift_x=0, shift_y=0):
    """Create hexagonal lattice initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    
    row = 0
    max_rows = 8
    while idx < n and row < max_rows:
        cols = 6 if row % 2 == 0 else 5
        for col in range(cols):
            if idx >= n:
                break
            x = 0.06 + col * 0.165 + (row % 2) * 0.0825 + shift_x
            y = 0.06 + row * 0.155 + shift_y
            centers[idx] = [np.clip(x, 0.03, 0.97), np.clip(y, 0.03, 0.97)]
            idx += 1
        row += 1
    
    return centers


def corner_biased_init(n):

    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, final_sum
