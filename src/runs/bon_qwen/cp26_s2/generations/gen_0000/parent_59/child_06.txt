import numpy as np
import math
from scipy.optimize import differential_evolution

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns centers, radii, and the sum of radii.
    """
    n = 26
    
    # --- Phase 1: Initialization (Hexagonal Lattice) ---
    # We create a hexagonal packing pattern to start with a dense configuration.
    # We generate points on a hexagonal lattice and keep the first 26 that fit.
    
    points = []
    r_init = 0.05 # Start small to ensure points fit
    
    # Generate hexagonal lattice points
    # Rows are shifted by r horizontally, separated by r*sqrt(3) vertically
    # We try to fit them in [0, 1]x[0, 1]
    
    y = r_init
    row = 0
    while len(points) < n:
        x_start = r_init + (row % 2) * r_init  # Shift every other row
        x = x_start
        while x < 1.0 - r_init:
            if len(points) < n:
                points.append((x, y))
            x += 2 * r_init
        
        y += r_init * math.sqrt(3)
        row += 1
        
    # If we didn't get enough points (unlikely with small r), fill with random
    while len(points) < n:
        points.append((0.5, 0.5))

    # Convert to numpy array
    centers_init = np.array(points[:n])
    # Scale to center in the square roughly
    # Find bounding box of points
    min_x, min_y = np.min(centers_init, axis=0)
    max_x, max_y = np.max(centers_init, axis=0)
    width = max_x - min_x
    height = max_y - min_y
    
    # Center the configuration
    cx, cy = 0.5, 0.5
    centers_init -= np.array([min_x + width/2, min_y + height/2])
    centers_init += np.array([cx, cy])
    
    # Calculate a safe initial radius based on min distance between centers
    # and distance to boundaries
    min_dist = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers_init[i] - centers_init[j])
            if d < min_dist:
                min_dist = d
        # Distance to boundary
        d_min = min(centers_init[i][0], 1-centers_init[i][0], 
                    centers_init[i][1], 1-centers_init[i][1])
        if d_min < min_dist:
            min_dist = d_min
            
    r_initial = min_dist / 2.0 * 0.9 # 90% of tightest fit
    
    radii_init = np.full(n, r_initial)

    # --- Phase 2: Optimization ---
    # We will optimize a single vector containing [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 3 * n = 78

        """
        Objective function to minimize.
        We want to maximize sum of radii, so we minimize negative sum.
        We add penalties for constraint violations.
        """
        centers = params[0:2*n].reshape((n, 2))
        radii = params[2*n:3*n]

        if np.any(radii < 0):
            return 1e12 # Huge penalty
            
        current_sum_r = -np.sum(radii)
        
        penalty = 0.0
        
        # Boundary constraints: r <= x <= 1-r  =>  x >= r AND x <= 1-r
        # i.e., x - r >= 0 AND x + r <= 1
        # Same for y
        
        # Check boundaries
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Violation amounts
            v_x1 = max(0, r - x)
            v_x2 = max(0, x + r - 1)
            v_y1 = max(0, r - y)
            v_y2 = max(0, y + r - 1)
            
            penalty += (v_x1 + v_x2 + v_y1 + v_y2) ** 2
            
        # Overlap constraints: dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                sum_r = radii[i] + radii[j]
                if dist < sum_r:
                    overlap = sum_r - dist
                    penalty += overlap ** 2
                    
        return current_sum_r + 1e6 * penalty

    # Bounds for optimization
    # x, y in [0, 1]
    # r in [0, 0.5] (max possible radius)
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)]) # x, y
        bounds.extend([(0.0, 0.5)])             # r

        objective,
        bounds,
        x0=x0,

        maxiter=200,

        recombination=0.9,
        seed=42,
        polish=True, # Use L-BFGS-B for local optimization at the end
        disp=False
    )

    sum_radii = np.sum(radii_opt)

                    radii_opt[j] = max(0, radii_opt[j])
                    valid = False

            break
            
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, float(sum_radii)
