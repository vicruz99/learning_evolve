import numpy as np
from scipy.optimize import minimize


def objective_function(params, n):
    c = params[:2 * n].reshape(n, 2)
    r = np.maximum(params[2 * n:], 1e-10)

    penalty = 0.0

    # Boundary violations
    for i in range(n):
        penalty += max(0, r[i] - c[i, 0]) ** 2
        penalty += max(0, c[i, 0] + r[i] - 1) ** 2
        penalty += max(0, r[i] - c[i, 1]) ** 2
        penalty += max(0, c[i, 1] + r[i] - 1) ** 2

    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = r[i] + r[j] - dist
            if overlap > 0:
                penalty += overlap ** 2

    return -np.sum(r) + 10000 * penalty


def force_relaxation(centers, radii, n, iterations=5000):
    for iteration in range(iterations):
        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq)
                if dist < 1e-10:
                    dist = 1e-10
                    dx = 1e-10
                    dy = 0

                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    force = (min_dist - dist) / dist
                    fx = dx * force
                    fy = dy * force
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy

            x, y = centers[i]
            r = radii[i]
            if x < r:
                forces[i, 0] += (r - x) * 20
            if x > 1 - r:
                forces[i, 0] -= (x + r - 1) * 20
            if y < r:
                forces[i, 1] += (r - y) * 20
            if y > 1 - r:
                forces[i, 1] -= (y + r - 1) * 20

        alpha = 0.5 / (1 + iteration * 0.005)
        centers = centers + forces * alpha

        radii = radii * (1 + 0.0008)

        for i in range(n):
            r = radii[i]
            centers[i, 0] = min(max(centers[i, 0], r), 1 - r)
            centers[i, 1] = min(max(centers[i, 1], r), 1 - r)

    return centers, radii


def generate_hexagonal_grid(n, rows, cols):
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.04)

    y_spacing = 1.0 / (rows + 1)
    x_spacing = 1.0 / (cols + 1)

    count = 0
    for row in range(rows):
        y = y_spacing * (row + 1)
        for col in range(cols):
            if count >= n:
                break
            x = x_spacing * (col + 1)
            if row % 2 == 1:
                x += x_spacing / 2
            x = max(0.02, min(0.98, x))
            y = max(0.02, min(0.98, y))
            centers[count] = [x, y]
            count += 1

    return centers[:n], radii[:n]


def optimize_packing(centers, radii, n):
    centers_relaxed, radii_relaxed = force_relaxation(
        centers.copy(), radii.copy(), n, iterations=5000
    )

    params = np.concatenate([centers_relaxed.flatten(), radii_relaxed])

    result = minimize(
        objective_function,
        params,
        args=(n,),
        method="Nelder-Mead",
        options={"maxiter": 200000, "xatol": 1e-12, "fatol": 1e-12},
    )

    opt_centers = result.x[: 2 * n].reshape(n, 2)
    opt_radii = np.maximum(result.x[2 * n :], 0)

    for i in range(n):
        r = opt_radii[i]
        opt_centers[i, 0] = min(max(opt_centers[i, 0], r), 1 - r)
        opt_centers[i, 1] = min(max(opt_centers[i, 1], r), 1 - r)

    return opt_centers, opt_radii


def run_packing():
    n = 26

    best_centers = None
    best_radii = None
    best_sum = -1

    configs = [
        (5, 6),
        (4, 7),
        (6, 5),
        (7, 4),
        (5, 7),
        (4, 8),
    ]

    for rows, cols in configs:
        centers, radii = generate_hexagonal_grid(n, rows, cols)
        opt_centers, opt_radii = optimize_packing(centers, radii, n)

        valid = True
        for i in range(n):
            x, y = opt_centers[i]
            r = opt_radii[i]
            if x < r - 1e-8 or x > 1 - r + 1e-8 or y < r - 1e-8 or y > 1 - r + 1e-8:
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(
                        np.sum((opt_centers[i] - opt_centers[j]) ** 2)
                    )
                    if dist < opt_radii[i] + opt_radii[j] - 1e-8:
                        valid = False
                        break
                if not valid:
                    break

        if valid:
            sum_radii = np.sum(opt_radii)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = opt_centers.copy()
                best_radii = opt_radii.copy()

    if best_centers is None:
        centers, radii = generate_hexagonal_grid(n, 5, 6)
        best_centers, best_radii = optimize_packing(centers, radii, n)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
