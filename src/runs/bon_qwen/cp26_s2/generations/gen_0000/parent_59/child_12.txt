    penalty_weight = 2000.0

        # Unpack variables
        c = vars[:2*n].reshape((n, 2)) # Centers (n, 2)
        r = vars[2*n:]                 # Radii (n,)

        # Constraint: distance(c_i, c_j) >= r_i + r_j
        # Calculate pairwise distances
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        
        # Indices for unique pairs (lower triangle, excluding diagonal)
        i, j = np.tril_indices(n, -1)
        
        # Calculate violation: if dist < r_i + r_j, violation = (r_i + r_j) - dist
        violations = (r[i] + r[j]) - dist[i, j]
