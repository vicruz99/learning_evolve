                centers[idx] = [x, y]
                idx += 1

            if np.max(overlap) < 1e-9:
                break

            i_idx, j_idx = np.where(mask)
            for k in range(len(i_idx)):

                centers[i, 1] += dy * push
                centers[j, 0] -= dx * push
                centers[j, 1] -= dy * push
