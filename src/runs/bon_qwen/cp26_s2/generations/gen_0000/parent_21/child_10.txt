import numpy as np
from scipy.optimize import minimize

def run_packing():
    N = 26
    np.random.seed(42)
    
    # --- Initialization ---
    # Place circles on a 5x6 grid pattern, taking the first 26 positions
    rows, cols = 5, 6
    x_vals = np.linspace(0.1, 0.9, cols)
    y_vals = np.linspace(0.1, 0.9, rows)
    
    init_centers = []
    for y in y_vals:
        for x in x_vals:
            if len(init_centers) >= N:
                break
            init_centers.append([x, y])
        if len(init_centers) >= N:
            break
            
    init_centers = np.array(init_centers[:N])
    
    # Initial radius: chosen to be safely inside boundaries and non-overlapping
    # Grid spacing is ~0.18, so r=0.07 is safe
    r_init = 0.07
    radii_init = np.full(N, r_init)
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([init_centers.flatten(), radii_init])
    
    # Add small random perturbation to break symmetry and avoid grid traps
    x0[:2*N] += np.random.randn(2*N) * 0.015
    # Clamp to valid range
    x0[:2*N] = np.clip(x0[:2*N], 0.0, 1.0)
    
    # --- Optimization Setup ---
    def objective(vars_flat):
        # Maximize sum of radii <=> minimize negative sum
        return -np.sum(vars_flat[2::3])
        
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    for i in range(N):
        idx = 3 * i
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]})
        
    # Non-overlap constraints: dist^2 >= (ri + rj)^2
    for i in range(N):
        for j in range(i + 1, N):
            def make_cons(ii, jj):
                def fun(v):
                    xi, yi, ri = v[3*ii], v[3*ii+1], v[3*ii+2]
                    xj, yj, rj = v[3*jj], v[3*jj+1], v[3*jj+2]
                    return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
                return fun
            constraints.append({'type': 'ineq', 'fun': make_cons(i, j)})
            
    # Variable bounds
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    
    # --- Run Optimizer ---
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
                   
    # Extract results
    centers = res.x[:2*N].reshape((N, 2))
    radii = res.x[2*N:]
    
    # Ensure numerical validity
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
