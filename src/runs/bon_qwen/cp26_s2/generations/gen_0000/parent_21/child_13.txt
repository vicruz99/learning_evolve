    def objective(x_flat):
        centers = x_flat[:2*n].reshape(n, 2)
        radii = x_flat[2*n:]
        
        # Objective: Maximize sum of radii -> Minimize -sum
        current_sum = np.sum(radii)
        
        # Penalty for constraints
        penalty = 0.0
        
        # Boundary constraints
        # x - r >= 0 => r - x <= 0
        # x + r <= 1 => x + r - 1 <= 0
        # Same for y
        
        # We use a soft penalty: if violated, add (violation)^2
        # Violation is max(0, constraint_violation)
        
        # Boundary
        # x_i >= r_i  => r_i - x_i <= 0
        viol_x_left = np.maximum(0, radii - centers[:, 0])
        penalty += np.sum(viol_x_left**2)
        
        # x_i <= 1 - r_i => x_i + r_i - 1 <= 0
        viol_x_right = np.maximum(0, centers[:, 0] + radii - 1.0)
        penalty += np.sum(viol_x_right**2)
        
        # y_i >= r_i
        viol_y_bottom = np.maximum(0, radii - centers[:, 1])
        penalty += np.sum(viol_y_bottom**2)
        
        # y_i <= 1 - r_i

        penalty += np.sum(viol_y_top**2)

def validate_packing_safe(centers, radii):
    """
    Safe validation similar to provided validate_packing but returns bool, msg
    """
    n = centers.shape[0]
    
    # Check NaN
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN detected"
        
    for i in range(n):
        if radii[i] < 0:
            return False, f"Negative radius {i}"
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False, f"Circle {i} out of bounds"
            
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Overlap {i} {j}"
                
    return True, "Valid"
