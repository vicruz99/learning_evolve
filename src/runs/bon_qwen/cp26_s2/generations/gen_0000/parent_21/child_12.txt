import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns:
        centers: (26, 2) array of (x, y)
        radii: (26,) array of radii
        sum_radii: float
    """
    n_circles = 26
    
    # --- Helper Functions ---
    
    def generate_hexagonal_grid(n, r):
        """Generate initial positions for n circles in a hexagonal lattice."""
        centers = []
        rows = []
        
        # Estimate number of rows for hexagonal packing
        # Area approx: n * pi * r^2 approx 0.9 * 1 (density)
        # Roughly sqrt(n) rows
        num_rows = int(np.ceil(np.sqrt(n * np.sqrt(3) / 2)))
        
        row_idx = 0
        circle_idx = 0
        
        # Adjust number of circles per row to fit 'n'
        circles_per_row = int(np.ceil(n / num_rows))
        
        # Vertical spacing for hexagonal packing: sqrt(3) * r
        # Horizontal spacing: 2 * r
        v_step = np.sqrt(3) * r
        h_step = 2 * r
        
        y = r
        while circle_idx < n and row_idx < num_rows + 2:
            # Determine how many circles to place in this row
            remaining = n - circle_idx
            # Shift odd rows by r to nest circles
            shift = r if (row_idx % 2 == 1) else 0
            x_start = r + shift
            
            count_in_row = 0
            x = x_start
            while circle_idx < n and x - r >= 0: # Ensure within left boundary
                centers.append([x, y])
                circle_idx += 1
                count_in_row += 1
                x += h_step
                
            rows.append(count_in_row)
            y += v_step
            row_idx += 1
            
        return np.array(centers[:n])

    def get_constraints(centers, radii):
        """
        Returns the list of constraints for the optimizer.
        1. Boundary constraints (circles inside [0,1]x[0,1])
        2. Non-overlap constraints (distance >= sum of radii)
        """
        n = len(radii)
        cons = []
        
        # Flatten variables for easier indexing if needed, but here we pass separate arrays
        # We need to map back from flattened vector to (x_i, y_i, r_i)
        
        # Boundary Constraints
        # x_i - r_i >= 0  =>  r_i - x_i <= 0
        # 1 - x_i - r_i >= 0 => x_i + r_i - 1 <= 0
        # Same for y
        
        for i in range(n):
            # x >= r => x - r >= 0 => -(x-r) <= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i] - v[2*i + n]})
            # x <= 1-r => x + r <= 1 => 1 - x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[2*i] - v[2*i + n]})
            
            # y >= r => y - r >= 0 => -(y-r) <= 0
            # y index is i + n
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[i + n] - v[2*i + n]})
            # y <= 1-r => y + r <= 1 => 1 - y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[i + n] - v[2*i + n]})
            
        # Non-overlap Constraints
        # ||c_i - c_j|| >= r_i + r_j
        # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        
        for i in range(n):
            for j in range(i + 1, n):
                # Indices in vector v:
                # x_i = v[2*i]
                # y_i = v[i + n]
                # r_i = v[2*i + n]
                
                xi, xj = 2*i, 2*j
                yi, yj = i + n, j + n
                ri, rj = 2*i + n, 2*j + n
                
                # Define constraint function
                def dist_constraint(v, i=i, j=j, xi=xi, xj=xj, yi=yi, yj=yj, ri=ri, rj=rj):
                    dx = v[xi] - v[xj]
                    dy = v[yi] - v[yj]
                    sum_r = v[ri] + v[rj]
                    return dx**2 + dy**2 - sum_r**2
                
                cons.append({'type': 'ineq', 'fun': dist_constraint})
                
        return cons

    def objective_function(v):
        """Maximize sum of radii -> Minimize negative sum of radii."""
        n = len(v) // 3
        radii = v[2*n:]
        return -np.sum(radii)

    # --- Setup ---
    
    # Initial radius estimation
    # For N=26, approx r ~ 0.1. Let's start a bit smaller to ensure feasibility.
    init_r = 0.08
    
    # Generate initial centers
    centers_init = generate_hexagonal_grid(n_circles, init_r)
    
    # Flatten variables: [x1, x2, ..., xn, y1, y2, ..., yn, r1, r2, ..., rn]
    x0 = np.concatenate([centers_init[:, 0], centers_init[:, 1], np.full(n_circles, init_r)])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # y
    for i in range(n_circles):
        bounds.append((0.0, 0.5)) # r
    
    # Get constraints
    constraints = get_constraints(centers_init, np.full(n_circles, init_r))
    
    # --- Optimization ---
    
    # Use SLSQP which handles non-linear constraints
    # Options for better convergence
    options = {
        'maxiter': 1000,
        'ftol': 1e-12,
        'disp': False
    }
    
    try:
        res = minimize(objective_function, x0, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options=options)
        
        if res.success:
            v_opt = res.x
        else:
            # Fallback to initial if optimization fails or returns worse result
            v_opt = x0
            
    except Exception as e:
        print(f"Optimization failed: {e}")
        v_opt = x0

    # --- Extract Results ---
    
    n = n_circles
    centers_opt = np.column_stack((v_opt[:n], v_opt[n:2*n]))
    radii_opt = v_opt[2*n:]
    
    # Validate and fix potential numerical issues (e.g., tiny negative radii)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    # Ensure centers are within bounds based on radii
    for i in range(n):
        r = radii_opt[i]
        x, y = centers_opt[i]
        centers_opt[i, 0] = np.clip(x, r, 1.0 - r)
        centers_opt[i, 1] = np.clip(y, r, 1.0 - r)

    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, float(sum_radii)

if __name__ == "__main__":
    centers, radii, total_sum = run_packing()
    print(f"Sum of radii: {total_sum}")
