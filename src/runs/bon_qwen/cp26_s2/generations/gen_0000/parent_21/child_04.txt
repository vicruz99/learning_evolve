import numpy as np
from scipy.optimize import minimize

def compute_constraints(v, n):
    """Compute inequality constraints: wall boundaries and circle overlaps."""
    m = 4 * n + n * (n - 1) // 2
    c = np.empty(m)
    idx = 0
    for i in range(n):
        x, y, r = v[i*3], v[i*3+1], v[i*3+2]
        c[idx] = x - r; idx += 1
        c[idx] = 1.0 - x - r; idx += 1
        c[idx] = y - r; idx += 1
        c[idx] = 1.0 - y - r; idx += 1
        
    for i in range(n):
        xi, yi, ri = v[i*3], v[i*3+1], v[i*3+2]
        for j in range(i+1, n):
            xj, yj, rj = v[j*3], v[j*3+1], v[j*3+2]
            dx = xi - xj
            dy = yi - yj
            c[idx] = dx*dx + dy*dy - (ri + rj)**2
            idx += 1
    return c

def obj_func(v):
    """Objective: negative sum of radii (to be minimized)."""
    return -np.sum(v[2::3])

def run_packing():
    n = 26
    best_sum = -np.inf
    best_x = None
    
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': lambda v: compute_constraints(v, n)}
    
    np.random.seed(42)
    
    # Strategy 1: Multiple random restarts to escape local minima
    for i in range(15):

        r0 = np.full(n, 0.03)  # Start small to ensure initial feasibility
        v0 = np.concatenate([pts.ravel(), r0])
        
        res = minimize(obj_func, v0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_x = res.x.copy()

    r_init = 0.095
    y = 0.5
    row = 0

        row += 1
    while len(pts) < n:

        best_sum = -res_hex.fun
        best_x = res_hex.x.copy()
        
    # Extract and return results
    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]
    return centers, radii, np.sum(radii)
