import numpy as np

import math

def check_validity(centers, radii):
    """
    Check if the current packing is valid.
    Returns True if valid, False otherwise.
    """
    n = centers.shape[0]
    
    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0: return False
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
            
    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            if dist_sq < (r_sum - 1e-9)**2:
                return False
    return True

def get_max_radius_for_circle(centers, radii, i):
    """
    Calculate the maximum possible radius for circle i given fixed centers and other radii.
    """
    n = centers.shape[0]
    x, y = centers[i]
    
    # Distance to boundaries
    min_r = min(x, 1.0 - x, y, 1.0 - y)
    
    # Distance to other circles
    for j in range(n):
        if i == j: continue
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = math.sqrt(dx*dx + dy*dy)
        # Constraint: r_i + r_j <= dist  =>  r_i <= dist - r_j
        r_limit = dist - radii[j]
        if r_limit < min_r:
            min_r = r_limit
            
    return max(0.0, min_r)
