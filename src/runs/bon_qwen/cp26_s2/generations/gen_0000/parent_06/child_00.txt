import numpy as np
from scipy.optimize import minimize


def objective(p, n):
    """Objective: minimize negative sum of radii"""
    return -np.sum(p[2 * n:])


def boundary_constraints(p, n):
    """Boundary constraints: each circle inside [0,1]^2"""
    centers = p[:2 * n].reshape(n, 2)
    radii = p[2 * n:]
    result = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        result.append(x - r)
        result.append(1.0 - x - r)
        result.append(y - r)
        result.append(1.0 - y - r)
    return np.array(result)


def overlap_constraints(p, n):
    """Non-overlap constraints: dist(i,j) >= r_i + r_j"""
    centers = p[:2 * n].reshape(n, 2)
    radii = p[2 * n:]
    result = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            result.append(dist - radii[i] - radii[j])
    return np.array(result)


def create_hexagonal_init(n, spacing_factor=0.16):
    """Create initial hexagonal grid configuration"""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)

    idx = 0
    x_spacing = spacing_factor
    y_spacing = x_spacing * np.sqrt(3) / 2.0

    for row in range(6):
        for col in range(5):
            if idx < n:
                x_offset = (row % 2) * x_spacing / 2.0
                centers[idx, 0] = 0.09 + col * x_spacing + x_offset
                centers[idx, 1] = 0.09 + row * y_spacing
                idx += 1

    return centers, radii


def create_alternative_init(n, variant=0):
    """Create alternative initial configurations"""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)

    if variant == 0:
        # 5x5 grid + 1 extra
        idx = 0
        for row in range(5):
            for col in range(5):
                if idx < n:
                    centers[idx, 0] = 0.1 + col * 0.18
                    centers[idx, 1] = 0.1 + row * 0.18
                    idx += 1
        # Extra circle in center
        if idx < n:
            centers[idx, 0] = 0.5
            centers[idx, 1] = 0.5
            idx += 1

    elif variant == 1:
        # 6x5 hexagonal with different spacing
        idx = 0
        x_spacing = 0.15
        y_spacing = x_spacing * np.sqrt(3) / 2.0
        for row in range(5):
            for col in range(6):
                if idx < n:
                    x_offset = (row % 2) * x_spacing / 2.0
                    centers[idx, 0] = 0.08 + col * x_spacing + x_offset
                    centers[idx, 1] = 0.08 + row * y_spacing
                    idx += 1

    elif variant == 2:
        # Random perturbed grid
        idx = 0
        for row in range(5):
            for col in range(5):
                if idx < n:
                    centers[idx, 0] = 0.1 + col * 0.18 + np.random.uniform(-0.01, 0.01)
                    centers[idx, 1] = 0.1 + row * 0.18 + np.random.uniform(-0.01, 0.01)
                    idx += 1
        if idx < n:
            centers[idx, 0] = 0.5 + np.random.uniform(-0.01, 0.01)
            centers[idx, 1] = 0.5 + np.random.uniform(-0.01, 0.01)
            idx += 1

    elif variant == 3:
        # Compact hexagonal
        idx = 0
        x_spacing = 0.155
        y_spacing = x_spacing * np.sqrt(3) / 2.0
        for row in range(6):
            ncols = 5 if row < 3 else 4 if row < 5 else 3
            for col in range(ncols):

                    x_offset = (row % 2) * x_spacing / 2.0
                    centers[idx, 0] = 0.1 + col * x_spacing + x_offset
                    centers[idx, 1] = 0.1 + row * y_spacing
                    idx += 1

    return centers, radii


def refine_with_simulation(centers, radii, n, n_iters=500):
    """Refine packing using repulsive force simulation"""
    centers = centers.copy()
    radii = radii.copy()

    lr = 0.005
    growth = 1.00005

    for iteration in range(n_iters):
        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-12:
                    dist = 1e-12
                    dx = 1e-6
                    dy = 0

                required = radii[i] + radii[j]
                if dist < required:
                    overlap = (required - dist) / dist
                    fx = overlap * dx
                    fy = overlap * dy
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x < r + 1e-10:
                forces[i, 0] += (r - x + 1e-8) * 10.0
            if x > 1.0 - r - 1e-10:
                forces[i, 0] -= (x - (1.0 - r) + 1e-8) * 10.0
            if y < r + 1e-10:
                forces[i, 1] += (r - y + 1e-8) * 10.0
            if y > 1.0 - r - 1e-10:
                forces[i, 1] -= (y - (1.0 - r) + 1e-8) * 10.0

        centers += lr * forces

        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r + 1e-10, min(1.0 - r - 1e-10, centers[i, 0]))
            centers[i, 1] = max(r + 1e-10, min(1.0 - r - 1e-10, centers[i, 1]))

        if iteration % 50 == 0:
            new_radii = radii * growth
            valid = True
            for i in range(n):
                x, y = centers[i]
                r = new_radii[i]
                if x < r or x > 1.0 - r or y < r or y > 1.0 - r:
                    valid = False
                    break

            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                        if dist < new_radii[i] + new_radii[j] - 1e-12:
                            valid = False
                            break
                    if not valid:
                        break

            if valid:
                radii = new_radii
            else:
                lr *= 0.99

    return centers, radii


def run_packing():
    n = 26

    constraints = [
        {'type': 'ineq', 'fun': lambda p: boundary_constraints(p, n)},
        {'type': 'ineq', 'fun': lambda p: overlap_constraints(p, n)},
    ]

    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n

    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Try multiple initial configurations
    init_configs = []

    # Hexagonal
    c, r = create_hexagonal_init(n, 0.16)
    init_configs.append((c, r))
    c, r = create_hexagonal_init(n, 0.17)
    init_configs.append((c, r))
    c, r = create_hexagonal_init(n, 0.15)
    init_configs.append((c, r))

    # Alternative inits
    for variant in range(4):
        c, r = create_alternative_init(n, variant)
        init_configs.append((c, r))

    for ci, (centers0, radii0) in enumerate(init_configs):
        params0 = np.concatenate([centers0.flatten(), radii0])

        # Multiple random perturbations
        for trial in range(3):
            if trial > 0:
                noisy = params0.copy()
                noisy[:2 * n] += np.random.uniform(-0.015, 0.015, 2 * n)
                noisy[:2 * n] = np.clip(noisy[:2 * n], 0.01, 0.99)
                noisy[2 * n:] += np.random.uniform(-0.005, 0.005, n)
                noisy[2 * n:] = np.clip(noisy[2 * n:], 1e-6, 0.4)
                current_params = noisy
            else:
                current_params = params0.copy()

            try:
                result = minimize(
                    objective,
                    current_params,
                    args=(n,),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
                )

                centers_opt = result.x[:2 * n].reshape(n, 2)
                radii_opt = result.x[2 * n:]

                # Validate
                valid = True
                for i in range(n):
                    x, y = centers_opt[i]
                    r = radii_opt[i]
                    if x < r - 1e-10 or x > 1.0 - r + 1e-10 or y < r - 1e-10 or y > 1.0 - r + 1e-10:
                        valid = False
                        break

                if valid:
                    for i in range(n):
                        for j in range(i + 1, n):
                            dist = np.sqrt((centers_opt[i, 0] - centers_opt[j, 0]) ** 2 + (centers_opt[i, 1] - centers_opt[j, 1]) ** 2)
                            if dist < radii_opt[i] + radii_opt[j] - 1e-10:
                                valid = False
                                break
                        if not valid:
                            break

                if valid:
                    current_sum = np.sum(radii_opt)
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = centers_opt.copy()
                        best_radii = radii_opt.copy()

                        # Try to refine with simulation
                        refined_centers, refined_radii = refine_with_simulation(best_centers, best_radii, n, n_iters=800)
                        refined_sum = np.sum(refined_radii)

                        if refined_sum > best_sum:
                            best_sum = refined_sum
                            best_centers = refined_centers.copy()
                            best_radii = refined_radii.copy()

            except Exception:
                pass

    return best_centers, best_radii, best_sum

def objective(p, n):
    """Objective: minimize negative sum of radii"""
    return -np.sum(p[2 * n:])


def boundary_constraints(p, n):
    """Boundary constraints: each circle inside [0,1]^2"""
    centers = p[:2 * n].reshape(n, 2)
    radii = p[2 * n:]
    result = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        result.append(x - r)
        result.append(1.0 - x - r)
        result.append(y - r)
        result.append(1.0 - y - r)
    return np.array(result)


def overlap_constraints(p, n):
    """Non-overlap constraints: dist(i,j) >= r_i + r_j"""
    centers = p[:2 * n].reshape(n, 2)
    radii = p[2 * n:]
    result = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            result.append(dist - radii[i] - radii[j])
    return np.array(result)
