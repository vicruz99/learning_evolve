import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    # 1. Hexagonal lattice initialization for high density
    pts = []
    r_init = 0.08
    rows = [6, 5, 6, 5, 4]
    y = 0.1
    dy = r_init * np.sqrt(3) * 1.05
    
    for i, count in enumerate(rows):
        x_start = 0.1 + (i % 2) * r_init
        for j in range(count):
            x = x_start + j * 2 * r_init
            pts.append([x, y, r_init])
        y += dy
        
    # Scale and center to fit comfortably inside [0,1]^2
    xs = [p[0] for p in pts]
    ys = [p[1] for p in pts]
    x_min, x_max = min(xs), max(xs)
    y_min, y_max = min(ys), max(ys)
    dim_max = max(x_max - x_min, y_max - y_min)
    scale = 0.8 / dim_max
    
    x0 = np.zeros(3 * n)
    for k, p in enumerate(pts):
        x0[3*k]   = 0.1 + (p[0] - x_min) * scale + rng.uniform(-0.01, 0.01)
        x0[3*k+1] = 0.1 + (p[1] - y_min) * scale + rng.uniform(-0.01, 0.01)
        x0[3*k+2] = p[2]
        
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    def objective(vars):
        pts = vars.reshape(n, 3)
        r = pts[:, 2]
        loss = -np.sum(r)
        
        mu = 1500.0
        x, y = pts[:, 0], pts[:, 1]
        
        # Boundary penalties
        pen = np.maximum(0, r - x)**2 + np.maximum(0, r - (1-x))**2 + \
              np.maximum(0, r - y)**2 + np.maximum(0, r - (1-y))**2
        loss += mu * np.sum(pen)
        
        # Overlap penalties (vectorized)
        diff = pts[:, :2][:, np.newaxis] - pts[:, :2][np.newaxis, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        overlaps = np.maximum(0, r_sum - dists)
        loss += mu * np.sum(overlaps[mask]**2)
        
        return loss

    # 2. Optimization
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-8})
    
    vars_opt = res.x
    centers = vars_opt.reshape(n, 3)[:, :2]
    radii = vars_opt.reshape(n, 3)[:, 2]
    
    # 3. Strict feasibility post-processing
    radii = np.maximum(radii, 0.0)
    
    # Clamp to boundaries
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1-x, y, 1-y) - 1e-7
        radii[i] = min(radii[i], max_r)
        
    # Resolve any remaining overlaps by proportional shrinking
    for _ in range(5):
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < radii[i] + radii[j] - 1e-9:
                    factor = (d + 1e-9) / (radii[i] + radii[j])
                    radii[i] *= factor
                    radii[j] *= factor
                    
    radii = np.maximum(radii, 0.0)
    return centers, radii, np.sum(radii)
