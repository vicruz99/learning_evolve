

To maximize the sum of radii for 26 circles in a unit square, we must balance the trade-off between circle size and geometric packing density. 

The optimal strategy is to maximize the area utilization of the square. While the target sum of radii (2.636) corresponds to an average radius of $\approx 0.1014$, which is slightly larger than the maximum possible radius for a strictly uniform grid, we can achieve it by:
1.  **Hexagonal Packing:** Using a staggered hexagonal lattice, which has a packing density of $\pi/\sqrt{12} \approx 0.9069$, allowing for tighter fitting of circles than a square grid.
2.  **Radius Optimization:** Solving a Linear Programming (LP) problem to find the maximum radii for a given set of centers. This ensures that for any configuration, the circles are expanded to their maximum possible size (touching boundaries or neighbors).
3.  **Center Optimization:** Iteratively adjusting circle centers to open up space for larger radii.

By combining a hexagonal lattice initialization with a force-directed "repulsion" adjustment, we can achieve a valid and efficient packing.

import numpy as np
from scipy.optimize import linprog

def get_max_radii(centers):
    """
    Solves an LP to find the maximum possible radii for fixed centers.
    Maximizes sum(r) subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0

    # Variables: r_0, r_1, ..., r_n-1
    # Objective: maximize sum(r_i) -> minimize -sum(r_i)
    c_obj = np.ones(n) * -1

    # Constraints
    A_ub = []
    b_ub = []

    # 1. Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        # r_i <= x  =>  r_i - x <= 0  =>  1*r_i <= x
        A_ub.append([1.0 if j == i else 0.0 for j in range(n)])
        b_ub.append(x)
        
        # r_i <= 1-x
        A_ub.append([1.0 if j == i else 0.0 for j in range(n)])
        b_ub.append(1.0 - x)
        
        # r_i <= y
        A_ub.append([1.0 if j == i else 0.0 for j in range(n)])
        b_ub.append(y)
        
        # r_i <= 1-y
        A_ub.append([1.0 if j == i else 0.0 for j in range(n)])
        b_ub.append(1.0 - y)

    # 2. Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = [0.0] * n
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            return radii, -res.fun
        else:
            # Fallback to small radii if LP fails
            return np.full(n, 1e-9), 0.0
    except Exception:
        return np.full(n, 1e-9), 0.0

def run_packing():
    # 1. Initialize centers using a hexagonal lattice pattern
    centers = []
    r_init = 0.105 # Slightly larger than target avg to force optimization
    
    # Hexagonal rows
    rows = 6
    cols = 5
    y_start = r_init
    y_step = r_init * np.sqrt(3)
    
    count = 0
    for i in range(rows):
        if count >= 26:
            break
        y = y_start + i * y_step
        # Shift odd rows
        x_offset = r_init if i % 2 == 1 else 0
        x_start = r_init + x_offset
        
        for j in range(cols):
            if count >= 26:
                break
            x = x_start + j * (2 * r_init)
            if x + r_init <= 1.0:
                centers.append([x, y])
                count += 1
    
    # Pad if fewer than 26 (shouldn't happen with these params)
    while len(centers) < 26:
        centers.append([0.5, 0.5])
    centers = np.array(centers[:26])

    # 2. Iterative Refinement (Repulsion / Hill Climbing)
    # We perturb centers and re-solve the LP for radii
    best_sum = 0
    best_r = None
    
    # Initial evaluation
    radii, current_sum = get_max_radii(centers)
    best_r = radii.copy()
    best_sum = current_sum

    # Simple hill climbing with random perturbations
    # 2000 iterations should be sufficient for N=26
    for _ in range(2000):
        # Pick a random circle to move
        idx = np.random.randint(0, 26)
        old_pos = centers[idx].copy()
        
        # Propose new position (random step)
        step = 0.02 * (np.random.rand(2) - 0.5)
        new_pos = old_pos + step
        
        # Keep within bounds (with margin)
        margin = 0.01
        new_pos[0] = np.clip(new_pos[0], margin, 1.0 - margin)
        new_pos[1] = np.clip(new_pos[1], margin, 1.0 - margin)
        
        centers[idx] = new_pos
        
        # Evaluate
        radii, new_sum = get_max_radii(centers)
        
        if new_sum > best_sum:
            best_sum = new_sum
            best_r = radii.copy()
        else:
            # Revert move if worse
            centers[idx] = old_pos

    # Final evaluation with best centers found
    # Note: The centers variable currently holds the state of the last successful move or revert.
    # To be safe, re-run get_max_radii on the current centers state which should be the best.
    # However, we need to return the centers corresponding to best_r. 
    # The loop logic above updates best_r only if new_sum > best_sum, but centers is updated in place.
    # If we revert, centers is restored. So centers should be consistent with best_sum.
    
    # One final check to ensure radii match the centers
    final_radii, final_sum = get_max_radii(centers)
    
    # Sort indices for consistent output (optional, but good practice)
    # The problem doesn't require sorted, just arrays of shape (26,2) and (26,)
    
    return centers, final_radii, final_sum
