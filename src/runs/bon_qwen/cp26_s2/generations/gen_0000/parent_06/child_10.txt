import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def objective(vars):
    radii = vars[2::3]
    return -np.sum(radii)


def constraint_funcs(vars):
    constraints = []
    n = N_CIRCLES
    for i in range(n):
        x = vars[3 * i]
        y = vars[3 * i + 1]
        r = vars[3 * i + 2]
        constraints.append(x - r)
        constraints.append(1.0 - x - r)
        constraints.append(y - r)
        constraints.append(1.0 - y - r)
        constraints.append(r)
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1, r1 = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
            x2, y2, r2 = vars[3 * j], vars[3 * j + 1], vars[3 * j + 2]
            dx = x1 - x2
            dy = y1 - y2
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-30 else 1e-15
            constraints.append(dist - r1 - r2)
    return np.array(constraints)


def make_initial_config(r_init):
    n = N_CIRCLES
    centers = np.zeros((n, 2))

    # 6-5-6-5-4 row pattern for hexagonal packing = 26 circles
    rows = [6, 5, 6, 5, 4]

    row_height = np.sqrt(3.0) * r_init
    col_width = 2.0 * r_init

    idx = 0
    for row_idx, num_cols in enumerate(rows):
        offset = (row_idx % 2) * col_width * 0.5
        for col in range(num_cols):
            x = col * col_width + offset
            y = row_idx * row_height
            centers[idx] = [x, y]
            idx += 1

    # Center and scale to fit in unit square with margin
    centers -= centers.min(axis=0)
    max_dim = centers.max(axis=0)
    scale = 0.85 / max(max_dim[0], max_dim[1], 1e-10)
    centers *= scale
    centers += (1.0 - scale * max_dim) * 0.5

    radii = np.ones(n) * r_init * scale

    vars_out = np.zeros(3 * n)
    for i in range(n):
        vars_out[3 * i] = centers[i, 0]
        vars_out[3 * i + 1] = centers[i, 1]
        vars_out[3 * i + 2] = radii[i]

    return vars_out


def run_packing():
    n = N_CIRCLES

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': constraint_funcs}

    best_result = None
    best_sum = -1.0

    # Try multiple initial radii to explore different configurations
    for r_init in [0.04, 0.05, 0.06, 0.07, 0.08, 0.09]:
        x0 = make_initial_config(r_init)

        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 5000, 'ftol': 1e-12, 'iprint': -1}
        )

        sum_radii = -result.fun
        if sum_radii > best_sum:
            best_result = result
            best_sum = sum_radii

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_result.x[3 * i]
        centers[i, 1] = best_result.x[3 * i + 1]
        radii[i] = best_result.x[3 * i + 2]

    return centers, radii, float(np.sum(radii))
