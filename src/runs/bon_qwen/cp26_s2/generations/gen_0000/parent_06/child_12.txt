import numpy as np
from scipy.optimize import minimize

def generate_initial_guess(n):
    """Generates a valid initial hexagonal packing for n circles."""
    points = []
    # Generate a dense hexagonal grid
    for i in range(15):
        for j in range(15):
            x = j * np.sqrt(3) / 2 + (i % 2) * np.sqrt(3) / 4
            y = i * 0.75
            points.append(np.array([x, y]))
    points = np.array(points)
    
    # Find the bounding box of the points
    min_x, min_y = np.min(points, axis=0)
    max_x, max_y = np.max(points, axis=0)
    width = max_x - min_x
    height = max_y - min_y
    
    # Center the points at (0.5, 0.5) and scale to fit in [0, 1]x[0, 1]
    points = (points - np.array([min_x, min_y])) / np.array([width, height])
    scale = 0.95 / max(width, height) # Leave a small margin
    points = points * scale + (1 - scale) / 2
    
    # Select the n points closest to the center of the square
    center = np.array([0.5, 0.5])
    dists = np.linalg.norm(points - center, axis=1)
    indices = np.argsort(dists)[:n]
    initial_centers = points[indices]
    
    # Initial radius slightly smaller than min distance to boundary
    r_init = 0.05
    return initial_centers, np.full(n, r_init)

def optimize_packing(n, initial_centers, initial_radii):
    """Optimizes the packing using SLSQP."""
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])
    
    def objective(p):
        return -np.sum(p[2*n::3]) # Minimize negative sum of radii

    def constraints_func(p):
        centers = p[:2*n].reshape(n, 2)
        radii = p[2*n::3]
        cons = []
        
        # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
        cons.append(centers[:, 0] - radii)
        cons.append(1 - centers[:, 0] - radii)
        cons.append(centers[:, 1] - radii)
        cons.append(1 - centers[:, 1] - radii)
        
        # Pairwise non-overlap constraints: (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
        diff = centers[:, None, :] - centers[None, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        r_sum = radii[:, None] + radii[None, :]
        # Only check upper triangle to avoid duplicates and self-comparison
        triu_indices = np.triu_indices(n, k=1)
        pairwise_cons = dist_sq[triu_indices] - r_sum[triu_indices]**2
        cons.append(pairwise_cons)
        
        return np.concatenate(cons)

    constraints = {'type': 'ineq', 'fun': constraints_func}
    bounds = [(0, 1)] * (2*n) + [(0, 1)] * n
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 500, 'ftol': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_result = None
    best_sum = -np.inf
    
    # Run optimization multiple times with different initial seeds/configurations
    for seed in range(5):
        np.random.seed(seed)
        centers_init, radii_init = generate_initial_guess(n)
        
        # Add slight random perturbation to avoid symmetry issues
        perturbation = np.random.uniform(-0.01, 0.01, size=centers_init.shape)
        centers_perturbed = np.clip(centers_init + perturbation, 0.1, 0.9)
        
        optimized = optimize_packing(n, centers_perturbed, radii_init)
        centers = optimized[:2*n].reshape(n, 2)
        radii = optimized[2*n::3]
        
        # Ensure radii are non-negative
        radii = np.maximum(radii, 0)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_result = (centers, radii)
            
    centers, radii = best_result
    return centers, radii, float(np.sum(radii))
