import numpy as np
import math


def hexagonal_init(n, n_cols=6, n_rows=5):
    """Generate hexagonal initial configuration."""
    r = 0.09
    centers = []
    idx = 0
    for row in range(n_rows):
        for col in range(n_cols):
            if idx >= n:
                break
            x = r + col * 2 * r
            y = r + row * math.sqrt(3) * r
            if row % 2 == 1:
                x += r
            centers.append([x, y])
            idx += 1
    return np.array(centers[:n]), np.full(n, r)


def resolve_overlaps(centers, radii, max_iter=500):
    """Resolve overlaps by pushing centers apart."""
    n = len(radii)
    for _ in range(max_iter):
        max_violation = 0

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = math.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]

                if dist < min_dist - 1e-12 and dist > 1e-12:
                    overlap = min_dist - dist
                    dx_n = dx / dist
                    dy_n = dy / dist
                    centers[i, 0] += dx_n * overlap / 2
                    centers[i, 1] += dy_n * overlap / 2
                    centers[j, 0] -= dx_n * overlap / 2
                    centers[j, 1] -= dy_n * overlap / 2
                    max_violation = max(max_violation, overlap)

        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

        if max_violation < 1e-10:
            break

    return centers


def check_validity(centers, radii):
    """Check if the packing is valid."""
    n = len(radii)
    for i in range(n):
        if radii[i] < 0:
            return False
        if centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1 - radii[i] + 1e-10:
            return False
        if centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1 - radii[i] + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def fix_violations(centers, radii, max_iter=100):
    """Fix remaining violations by reducing radii minimally."""
    n = len(radii)
    for _ in range(max_iter):
        any_violation = False

        for i in range(n):
            for j in range(i + 1, n):
                dist = math.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    reduction = (radii[i] + radii[j] - dist + 1e-12) / 2
                    radii[i] -= reduction
                    radii[j] -= reduction
                    radii[i] = max(1e-10, radii[i])
                    radii[j] = max(1e-10, radii[j])
                    any_violation = True

        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

        if not any_violation:
            break

    return centers, radii


def grow_circles_equal(centers, radii, steps=3000, initial_step=0.001):
    """Grow all circles equally while resolving overlaps."""
    n = len(radii)
    step = initial_step

    for _ in range(steps):
        radii += step
        centers = resolve_overlaps(centers, radii, max_iter=200)

        # Check if valid
        valid = True
        for i in range(n):
            for j in range(i + 1, n):
                dist = math.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                if dist < radii[i] + radii[j] - 1e-10:
                    valid = False
                    break
            if not valid:
                break
            if centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1 - radii[i] + 1e-10:
                valid = False
            if centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1 - radii[i] + 1e-10:
                valid = False

        if not valid:
            radii -= step
            step *= 0.97
            if step < 1e-10:
                break

    return centers, radii


def grow_circles_variable(centers, radii, steps=2000, step_size=0.00002):
    """Grow circles with variable radii while resolving overlaps."""
    n = len(radii)

    for _ in range(steps):
        # Increase radii, favoring smaller ones to even out
        for i in range(n):
            radii[i] += step_size * (1.5 - radii[i] * 5)

        centers = resolve_overlaps(centers, radii, max_iter=200)

                    valid = False
                    break

                break
            if centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1 - radii[i] + 1e-10:
                valid = False
            if centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1 - radii[i] + 1e-10:
                valid = False

        if not valid:
            for i in range(n):
                radii[i] -= step_size * (1.5 - radii[i] * 5)
            step_size *= 0.95
            if step_size < 1e-12:
                break

    return centers, radii


def try_config(n, centers_init, radii_init):
    """Try to optimize from a given initial configuration."""
    centers = centers_init.copy()
    radii = radii_init.copy()

    # Phase 1: Grow equal circles
    centers, radii = grow_circles_equal(centers, radii, steps=3000, initial_step=0.001)

    # Phase 2: Allow variable radii
    centers, radii = grow_circles_variable(centers, radii, steps=2000, step_size=0.00002)

    # Phase 3: Fine adjustment - try to increase each radius individually
    for _ in range(20):
        for i in range(n):
            radii[i] += 0.00001
            centers = resolve_overlaps(centers, radii, max_iter=100)
            # Quick validity check
            valid = True
            for j in range(n):
                if j == i:
                    continue
                dist = math.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                if dist < radii[i] + radii[j] - 1e-10:
                    valid = False
                    break
            if not valid:
                radii[i] -= 0.00001

    # Ensure boundary constraints
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

    # Final fix of any remaining violations
    centers, radii = fix_violations(centers, radii)

    return centers, radii


def run_packing():

    best_radii = None

            best_radii = radii.copy()

        centers = best_centers.copy()
        radii = best_radii.copy()

        def objective(x):
            r = x[:n]
            c = x[n:].reshape(n, 2)
            obj = -np.sum(r)
            penalty = 0.0
            w = 50000.0

            for i in range(n):
                penalty += w * max(0, r[i] - c[i, 0]) ** 2
                penalty += w * max(0, r[i] - c[i, 1]) ** 2
                penalty += w * max(0, c[i, 0] + r[i] - 1) ** 2
                penalty += w * max(0, c[i, 1] + r[i] - 1) ** 2

            for i in range(n):
                for j in range(i + 1, n):
                    d = math.sqrt((c[i, 0] - c[j, 0]) ** 2 + (c[i, 1] - c[j, 1]) ** 2)
                    penalty += w * max(0, r[i] + r[j] - d) ** 2

            return obj + penalty

        x0 = np.concatenate([radii, centers.flatten()])
        bounds = [(0, 0.5)] * n + [(0, 1)] * (2 * n)

        result = minimize(
            objective,
            x0,
            method='L-BFGS-B',

    return best_centers, best_radii, best_sum
