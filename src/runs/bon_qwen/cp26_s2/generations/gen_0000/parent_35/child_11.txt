Algorithm:
1. Initialize centers randomly.
2. Initialize radii to a small value (e.g., 0.05).
3. Iteratively:
   a. Increase all radii by a small amount.

      - If circle touches boundary, move it away from boundary.

It's easier to implement and often finds good packings.

        # Subject to r_i + r_j <= dist_ij and 0 <= r_i <= bound_i
        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_vals, bounds=bounds_lp, method='highs', options={'disp': False})
            if res.success:
                return res.x, np.sum(res.x)
        except Exception:
            pass
        return np.zeros(n), 0.0

    cur_radii, cur_sum = solve_radii(cur_centers)
    best_sum = cur_sum
    best_centers = cur_centers.copy()
    best_radii = cur_radii.copy()
    
    for iteration in range(n_iter):
        current_step = step_size * (0.99 ** iteration)
        
        idx = np.random.randint(n)
        
        new_centers = cur_centers.copy()
        dx = np.random.uniform(-current_step, current_step)
        dy = np.random.uniform(-current_step, current_step)
        new_centers[idx, 0] += dx
        new_centers[idx, 1] += dy
        np.clip(new_centers, 0.0, 1.0, out=new_centers)
        
        new_radii, new_sum = solve_radii(new_centers)
        
        if new_sum > cur_sum:
            cur_centers = new_centers
            cur_radii = new_radii
            cur_sum = new_sum
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = cur_centers.copy()
                best_radii = cur_radii.copy()
                
    return best_centers, best_radii, best_sum
