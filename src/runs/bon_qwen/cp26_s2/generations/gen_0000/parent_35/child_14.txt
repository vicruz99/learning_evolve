import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.spatial.distance import pdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    Uses Differential Evolution to optimize centers, and Linear Programming 
    to determine optimal radii for fixed centers.
    """
    n = 26
    
    # Precompute the structure of the pairwise constraint matrix A_ub
    # Constraints: r_i + r_j <= dist_ij for all i < j
    # Number of pairs
    n_pairs = n * (n - 1) // 2
    
    # A_ub is a (n_pairs, n) matrix. 
    # Each row has two 1s corresponding to indices i and j.
    # We build this once.
    A_ub = np.zeros((n_pairs, n))
    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            row += 1
            
    # Objective coefficients for LP: minimize -sum(r_i)
    c_lp = -np.ones(n)

    def objective(centers_flat):
        """
        Calculates the maximum sum of radii for given centers.
        centers_flat: 1D array of length 52 (26 circles * 2 coords)
        Returns: negative sum of radii (for minimization)
        """
        centers = centers_flat.reshape(n, 2)
        
        # Calculate pairwise Euclidean distances
        # pdist returns the upper triangular part of the distance matrix
        dists = pdist(centers)
        
        # Boundary constraints determine the maximum possible radius for each circle
        # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        # This implies r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
        # We incorporate this into the variable bounds for linprog
        max_r_boundary = np.minimum(
            np.minimum(centers[:, 0], 1 - centers[:, 0]),
            np.minimum(centers[:, 1], 1 - centers[:, 1])
        )
        
        # Variable bounds for radii: 0 <= r_i <= max_r_boundary[i]
        bounds = [(0, max_r_boundary[i]) for i in range(n)]
        
        # Solve LP
        # Maximize sum(r) <=> Minimize -sum(r)
        # Constraints: A_ub * r <= dists
        try:
            res = linprog(c_lp, A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
            
            if res.success:
                # Return negative of the objective value (which was -sum(r))
                # So we return sum(r). But DE minimizes, so we return -sum(r).
                return -res.fun
            else:
                # If LP fails (unlikely given feasible r=0), return a penalty
                return -100.0 
        except Exception:
            return -100.0

    # Bounds for centers: each coordinate in [0, 1]
    # Actually, centers can be anywhere, but optimal ones will be within [0,1].
    # If a center is outside, the boundary constraints will force radius to 0 or negative (clamped).
    # But keeping search space [0,1] is logical.
    bounds_centers = [(0, 1) for _ in range(2 * n)]

    # Run Differential Evolution
    # strategy='best1bin', popsize=15, maxiter=1000, tol=1e-7
    # seed for reproducibility
    result = differential_evolution(
        objective,
        bounds_centers,
        seed=42,
        maxiter=200, # Increase if time allows, 200 is usually enough for n=26
        popsize=20,
        tol=1e-7,
        mutation=(0.5, 1),
        recombination=0.7,
        polish=True # Run a local minimizer on the best solution found
    )

    best_centers = result.x.reshape(n, 2)
    
    # Solve LP one last time to get final radii
    dists = pdist(best_centers)
    max_r_boundary = np.minimum(
        np.minimum(best_centers[:, 0], 1 - best_centers[:, 0]),
        np.minimum(best_centers[:, 1], 1 - best_centers[:, 1])
    )
    bounds = [(0, max_r_boundary[i]) for i in range(n)]
    
    res_final = linprog(c_lp, A_ub=A_ub, b_ub=dists, bounds=bounds, method='highs')
    radii = res_final.x
    sum_radii = np.sum(radii)
    
    # Clean up radii (handle any tiny negative due to numerics, though bounds prevent it)
    radii = np.maximum(radii, 0)
    
    # Re-calculate sum radii after clipping
    sum_radii = np.sum(radii)
    
    return best_centers, radii, sum_radii
