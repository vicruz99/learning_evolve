import numpy as np
from scipy.optimize import minimize

def compute_cost(vars, n):
    """
    Compute the negative sum of radii.
    vars: array of shape (3*n,), [x1, y1, r1, x2, y2, r2, ...]
    """
    # Extract radii
    # radii are at indices 2, 5, 8, ...
    radii = vars[2::3]
    return -np.sum(radii)

def get_constraints(n):
    """
    Generate a list of constraint dictionaries for scipy.optimize.minimize.
    Constraints:
    1. Circle inside square: r <= x, r <= 1-x, r <= y, r <= 1-y
    2. Non-overlap: dist(i, j) >= r_i + r_j
    """
    constraints = []
    
    # Helper to index variables
    # x_i is at 3*i, y_i at 3*i+1, r_i at 3*i+2

    for i in range(n):
        r_idx = 3 * i + 2
        x_idx = 3 * i
        y_idx = 3 * i + 1
        
        # r <= x  => x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx_x=x_idx, idx_r=r_idx: v[idx_x] - v[idx_r]
        })
        # r <= 1 - x => x + r - 1 <= 0 => 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx_x=x_idx, idx_r=r_idx: 1.0 - v[idx_x] - v[idx_r]
        })
        # r <= y => y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx_y=y_idx, idx_r=r_idx: v[idx_y] - v[idx_r]
        })
        # r <= 1 - y => 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx_y=y_idx, idx_r=r_idx: 1.0 - v[idx_y] - v[idx_r]
        })
        # r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx_r=r_idx: v[idx_r]
        })

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            idx_x_i = 3 * i
            idx_y_i = 3 * i + 1
            idx_r_i = 3 * i + 2
            
            idx_x_j = 3 * j
            idx_y_j = 3 * j + 1
            idx_r_j = 3 * j + 2
            
            # Constraint: sqrt((xi-xj)^2 + (yi-yj)^2) >= ri + rj
            # To avoid non-smoothness at 0 distance and sqrt issues, 
            # we can use the squared form if we were minimizing squared error,
            # but here we need the inequality to hold.
            # f(v) = sqrt(...) - (ri + rj) >= 0
            
            def dist_constraint(v, i=i, j=j):
                xi = v[3*i]
                yi = v[3*i+1]
                ri = v[3*i+2]
                xj = v[3*j]
                yj = v[3*j+1]
                rj = v[3*j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist = np.sqrt(dx*dx + dy*dy)
                return dist - (ri + rj)
            
            constraints.append({
                'type': 'ineq',
                'fun': dist_constraint
            })
            
    return constraints
