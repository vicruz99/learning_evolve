import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a numerical optimization approach with a physics-inspired initialization.
    """
    n = 26
    
    # --- 1. Initialization ---
    # Start with a 5x5 grid for 25 circles, radius 0.09 (smaller than 0.1 to allow movement)
    # This provides a stable, dense starting point.
    coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    centers_25 = np.array(np.meshgrid(coords, coords)).T.reshape(-1, 2)
    
    # Add the 26th circle at the center. 
    # We start it with a very small radius to avoid immediate huge overlaps.
    # The center (0.5, 0.5) is actually a grid point for 5x5, so we perturb it slightly
    # to a nearby gap or just rely on the optimizer to separate them.
    # Let's place it at (0.5, 0.5) but the optimizer will separate it.
    # To be safer, place it at a local void. 
    # Actually, just random perturbation of the 26th point works well.
    center_26 = np.array([0.5, 0.51]) # Slight offset
    
    centers = np.vstack([centers_25, [center_26]])
    radii = np.ones(n) * 0.05 # Small initial radius to ensure valid start

    # Flatten for optimizer: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])

    # --- 2. Optimization Setup ---
    
    def objective(vars):
        # Minimize negative sum of radii (Maximize sum of radii)
        r = vars[2::3]
        return -np.sum(r)

    def constraint_boundary(vars):
        # Constraints: r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
        # Similarly for y
        res = []
        for i in range(n):
            idx = i * 3
            x = vars[idx]
            y = vars[idx+1]
            r = vars[idx+2]
            res.append(x - r)
            res.append(1.0 - x - r)
            res.append(y - r)
            res.append(1.0 - y - r)
        return np.array(res)

    def constraint_overlap(vars):
        # Constraints: dist(i, j) >= r_i + r_j
        res = []
        for i in range(n):
            for j in range(i + 1, n):
                idx_i = i * 3
                idx_j = j * 3
                xi, yi, ri = vars[idx_i:idx_i+3]
                xj, yj, rj = vars[idx_j:idx_j+3]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                # Add a small epsilon to avoid sqrt(0) gradient issues if they coincide
                # though with valid start they shouldn't.
                res.append(dist - (ri + rj))
        return np.array(res)

    # Define constraints for SLSQP
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Bounds: x, y in [0, 1], r >= 0
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 1.0)) # r

    # --- 3. Run Optimizer ---
    # SLSQP is suitable for this small-scale non-linear problem
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
        final_vars = res.x
    except Exception:
        # Fallback if optimizer fails
        final_vars = x0

    # --- 4. Post-Processing (Heuristic Refinement) ---
    # The optimizer might get stuck in local optima. 
    # We can try a quick "push to corners" refinement.
    # However, the constraints are strict, so we must ensure validity.
    # A simple way is to just trust the optimizer but maybe run it twice?
    # Given the complexity, a single run with good initialization is best.
    
    # Extract results
    centers_final = np.zeros((n, 2))
    radii_final = np.zeros(n)
    
    for i in range(n):
        idx = i * 3
        centers_final[i] = [final_vars[idx], final_vars[idx+1]]
        radii_final[i] = final_vars[idx+2]

    # Validation check (internal)
    # If the result is invalid, we might need to adjust. 
    # But SLSQP should satisfy constraints.
    
    return centers_final, radii_final, np.sum(radii_final)
