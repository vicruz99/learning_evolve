def smooth_max(x, alpha=1e5):
    """Smooth differentiable approximation of max(0, x)"""
    eps = 1.0 / alpha
    return 0.5 * (x + np.sqrt(x * x + eps))


def compute_objective_and_grad(params, n, alpha):
    """Compute objective value for circle packing optimization"""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Non-overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy + 1e-16
            dist = np.sqrt(dist_sq)
            overlap = radii[i] + radii[j] - dist
            penalty += alpha * smooth_max(overlap, alpha)
    
    # Boundary penalties
    for i in range(n):
        penalty += alpha * smooth_max(radii[i] - centers[i, 0], alpha)
        penalty += alpha * smooth_max(centers[i, 0] + radii[i] - 1.0, alpha)
        penalty += alpha * smooth_max(radii[i] - centers[i, 1], alpha)
        penalty += alpha * smooth_max(centers[i, 1] + radii[i] - 1.0, alpha)
        penalty += alpha * smooth_max(-radii[i], alpha)
    
    return obj + penalty


def hexagonal_init(n, row_counts, r_init):
    """Create hexagonal initial arrangement"""
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    
    idx = 0
    y = r_init
    for row_idx, count in enumerate(row_counts):
        if row_idx % 2 == 1:
            x_start = 2 * r_init
        else:
            x_start = r_init
        
        for col_idx in range(count):
            x = x_start + col_idx * 2 * r_init
            centers[idx] = np.array([x, y])
            idx += 1
        y += np.sqrt(3) * r_init
    
    return centers, radii


def make_perturbed_init(centers, radii, n, seed):

def resolve_overlaps(centers, radii, n, max_iter=2000):

    radii = radii.copy()

        max_overlap = -1
        best_i, best_j = 0, 1

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-16)
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
                    best_i, best_j = i, j
        
        if max_overlap < 1e-12:
            break
        
        i, j = best_i, best_j
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx * dx + dy * dy + 1e-16)
        
        push = max_overlap * 0.5
        scale = push / dist
        
        centers[i, 0] += dx * scale
        centers[i, 1] += dy * scale
        centers[j, 0] -= dx * scale
        centers[j, 1] -= dy * scale
    
    # Clip to boundaries

        radii[i] = max(0, radii[i])

        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    return centers, radii
