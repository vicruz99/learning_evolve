import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26

    best_sum = 0.0
    best_c = None
    best_r = None

    for trial in range(40):
        c, r = _init(n, trial)
        result = _optimize(c, r, n)
        if result is not None:
            c_opt, r_opt = result
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_c = c_opt
                best_r = r_opt

    if best_c is None:
        best_c, best_r = _init(n, 0)
        best_sum = np.sum(best_r)

    best_c, best_r = _cleanup(best_c, best_r, n)

    return best_c, best_r, float(np.sum(best_r))


def _init(n, trial):
    np.random.seed(trial)

    rows = [6, 5, 6, 5, 4]
    centers = []

    r = 0.099 + 0.003 * np.random.rand()

    y_step = r * np.sqrt(3.0)
    x_step = 2.0 * r

    y0 = r + 0.015 + 0.01 * np.random.rand()
    x0 = r + 0.015 + 0.01 * np.random.rand()

    idx = 0
    for row_i, cnt in enumerate(rows):
        y = y0 + row_i * y_step
        x_off = x0
        if row_i % 2 == 1:
            x_off += r

        for col_i in range(cnt):
            x = x_off + col_i * x_step
            centers.append([
                x + np.random.uniform(-0.003, 0.003),
                y + np.random.uniform(-0.003, 0.003)
            ])
            idx += 1
            if idx >= n:
                break
        if idx >= n:
            break

    centers = np.array(centers[:n])
    radii = np.full(n, r)

    return centers, radii


def _loss(params, n):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    radii = np.maximum(radii, 1e-8)

    loss = -np.sum(radii)
    penalty = 0.0

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        margins = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
        for m in margins:
            if m < 0:
                penalty += m * m

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            req = radii[i] + radii[j]
            gap = req - dist
            if gap > 0:
                penalty += gap * gap

    penalty *= 25000.0

    return loss + penalty


def _optimize(centers, radii, n):
    params = np.concatenate([centers.flatten(), radii])

    b_center = (0.01, 0.99)
    b_radius = (0.01, 0.3)
    bounds = [b_center] * (2 * n) + [b_radius] * n

    try:
        res = minimize(
            _loss, params, args=(n,), bounds=bounds,
            method="L-BFGS-B",
            options={"maxiter": 30000, "ftol": 1e-16, "gtol": 1e-12}
        )

        if res.fun < -2.5:
            c = res.x[:2 * n].reshape(n, 2)
            r = np.maximum(res.x[2 * n:], 0.001)

            if _is_valid(c, r, n):
                return c, r
    except Exception:
        pass

    return None


def _is_valid(centers, radii, n):
    eps = 1e-9
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -eps or x + r > 1.0 + eps:
            return False
        if y - r < -eps or y + r > 1.0 + eps:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if d < radii[i] + radii[j] - eps:
                return False

    return True


def _cleanup(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        r = radii[i]

        r = min(r, x - 1e-10, 1.0 - x - 1e-10, y - 1e-10, 1.0 - y - 1e-10)

        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r = min(r, d - radii[j] - 1e-10)

        radii[i] = max(0.0, r)

    return centers, radii
