import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initial Configuration: Staggered grid approximating hexagonal packing
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(6):
        for col in range(6):
            if idx >= n:
                break
            x = col * 0.18 + (0.09 if row % 2 == 1 else 0.0)
            y = row * 0.18
            centers[idx] = [np.clip(x, 0.1, 0.9), np.clip(y, 0.1, 0.9)]
            idx += 1
        if idx >= n:
            break
            
    # Small random perturbation to break symmetry and aid optimization
    centers += np.random.uniform(-0.005, 0.005, size=centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
        
    # Flatten centers and initialize radius
    x0 = np.concatenate([centers.flatten(), [0.08]])
    
    def objective(vars):
        c = vars[:n*2].reshape(n, 2)
        r = vars[n*2]
        if r <= 0:
            return 1e6
        
        pen = 0.0
        # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
        for i in range(n):
            x, y = c[i]
            pen += np.maximum(0, r - x)**2
            pen += np.maximum(0, x - (1 - r))**2
            pen += np.maximum(0, r - y)**2
            pen += np.maximum(0, y - (1 - r))**2
            
        # Non-overlap constraints: distance >= 2r
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                pen += np.maximum(0, 2 * r - d)**2
                
        # Objective: maximize r (minimize -r) while keeping penalties low
        return -r + 5000 * pen

    # Bounds for centers and radius
    bounds = [(0.0, 1.0) for _ in range(n*2)]
    bounds.append((0.01, 0.5))
    
    # 2. Optimization
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False})
                   
    best_c = res.x[:n*2].reshape(n, 2)
    best_r = res.x[n*2]
    
    # 3. Post-processing: Ensure strict validity
    best_c = np.clip(best_c, best_r, 1.0 - best_r)
    radii = np.full(n, best_r)
    
    return best_c, radii, float(np.sum(radii))
