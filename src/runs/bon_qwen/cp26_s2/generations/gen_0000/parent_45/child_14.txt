import numpy as np
from scipy.optimize import minimize

def _penalty_obj(params, n, alpha):
    """
    Objective function: maximize sum of radii (minimize negative sum)
    subject to boundary and non-overlap constraints via penalty method.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    obj = -np.sum(radii)
    
    # Boundary constraints: circles must be inside [0,1]x[0,1]
    obj += alpha * np.sum(np.maximum(0, radii - centers[:, 0])**2)
    obj += alpha * np.sum(np.maximum(0, radii - (1.0 - centers[:, 0]))**2)
    obj += alpha * np.sum(np.maximum(0, radii - centers[:, 1])**2)
    obj += alpha * np.sum(np.maximum(0, radii - (1.0 - centers[:, 1]))**2)
    
    # Overlap constraints: circles must not overlap
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-14)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap = np.maximum(0, r_sum - dist)
    obj += alpha * np.sum(overlap[mask]**2)
    
    return obj

def run_packing():
    n = 26
    best_params = None
    best_sum = 0.0
    
    # Multiple restarts with different seeds to escape local minima
    for seed in range(30):
        rng = np.random.RandomState(seed)
        
        # Initial configuration: perturbed 5x5 grid
        centers = np.zeros((n, 2))
        idx = 0
        for i in range(5):

                if idx < n:

                    idx += 1

        centers = np.clip(centers, 0.05, 0.95)
        radii = np.ones(n) * 0.101
        
        x0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        curr = x0
        # Continuation method: increase penalty weight gradually
        for alpha in [50.0, 500.0, 5000.0, 50000.0]:
            res = minimize(_penalty_obj, curr, args=(n, alpha),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-12})
            curr = res.x
            
        centers = curr[:2*n].reshape(n, 2)
        radii = curr[2*n:]
        
        # Validate strictly
        valid = True

                valid = False
                break

                if d < radii[i] + radii[j] - 1e-10:
                    valid = False
                    break
