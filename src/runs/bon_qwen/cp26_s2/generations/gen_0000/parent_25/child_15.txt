

<strategy>
The problem is a non-convex nonlinear optimization task: maximize $\sum r_i$ subject to quadratic inequality constraints for non-overlap and linear constraints for boundaries. I will use SciPy's SLSQP algorithm, which handles inequality constraints well. To avoid poor local minima, I will employ multiple random restarts with a feasible initial guess (small radii well-separated). The optimizer will naturally expand the radii until the constraints become active, effectively finding a tight packing. A strict validation check ensures only feasible configurations are kept. A deterministic grid fallback guarantees a valid return even if optimization fails. This direct approach efficiently balances exploration and exploitation for circle packing.
</strategy>

import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Constraint function: returns array of constraint violations.
    All values must be >= 0 for feasibility.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    centers = v[:2*N].reshape(N, 2)
    radii = v[2*N:]
    
    m = 4*N + N*(N-1)//2
    c = np.empty(m)
    idx = 0
    
    # Boundary constraints: 0 <= x-r, x+r <= 1, 0 <= y-r, y+r <= 1
    for i in range(N):
        c[idx]   = centers[i, 0] - radii[i]
        c[idx+1] = 1.0 - centers[i, 0] - radii[i]
        c[idx+2] = centers[i, 1] - radii[i]
        c[idx+3] = 1.0 - centers[i, 1] - radii[i]
        idx += 4
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N):
        for j in range(i+1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            c[idx] = dx*dx + dy*dy - (radii[i] + radii[j])**2
            idx += 1
            
    return c

def run_packing():
    # Variable bounds: centers in [0, 1], radii in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_v = None
    best_sum = -1.0
    
    # Multiple restarts to escape local optima
    for seed in range(8):
        np.random.seed(seed)
        # Initial guess: random centers in center region, small radii to ensure feasibility
        centers = np.random.uniform(0.15, 0.85, size=(N, 2))
        radii = np.full(N, 0.03)
        x0 = np.concatenate([centers.flatten(), radii])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 3000, 'ftol': 1e-12})
            
            if np.isnan(res.x).any():
                continue
                
            c_centers = res.x[:2*N].reshape(N, 2)
            c_radii = res.x[2*N:]
            
            # Strict validation before accepting
            valid = True
            for i in range(N):
                if c_radii[i] < 0:
                    valid = False; break
                if (c_centers[i, 0] - c_radii[i] < -1e-9) or (c_centers[i, 0] + c_radii[i] > 1 + 1e-9):
                    valid = False; break
                if (c_centers[i, 1] - c_radii[i] < -1e-9) or (c_centers[i, 1] + c_radii[i] > 1 + 1e-9):
                    valid = False; break
                    
            if valid:
                for i in range(N):
                    for j in range(i+1, N):
                        d2 = np.sum((c_centers[i] - c_centers[j])**2)
                        if d2 < (c_radii[i] + c_radii[j])**2 - 1e-9:
                            valid = False; break
                    if not valid: break
                    
            if valid:
                curr_sum = np.sum(c_radii)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_v = res.x.copy()
                    
        except Exception:
            continue
            
    if best_v is not None:
        return best_v[:2*N].reshape(N, 2), best_v[2*N:], best_sum
        
    # Deterministic fallback packing
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.05)
    idx = 0
    for r in range(5):
        for c in range(5):
            if idx >= N: break
            centers[idx] = [0.1 + c*0.2, 0.1 + r*0.2]
            idx += 1
    return centers, radii, 0.05*N
