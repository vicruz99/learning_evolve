

To solve the circle packing problem for 26 circles in a unit square, I'll implement a force-directed optimization approach that:
1. Initializes circles in a hexagonal-like pattern (known to be space-efficient)
2. Uses repulsive forces to push overlapping circles apart and grow radii
3. Fine-tunes with scipy optimization for maximum sum of radii

<strategy>
1. Initialize circles in a 5-row hexagonal pattern (counts: 6, 5, 6, 5, 4 = 26)
2. Run force-directed simulation:
   - Repulsive forces between overlapping circles
   - Wall repulsion to keep circles inside [0,1]
   - Gradual radius growth
3. Use scipy.optimize.minimize with penalty functions for fine-tuning
4. Return the optimized configuration
</strategy>

```python
import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # Initialize centers in a hexagonal pattern
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)
    
    idx = 0
    # 5 rows with counts: 6, 5, 6, 5, 4 = 26
    row_specs = [(6, 0.0), (5, 0.08), (6, 0.0), (5, 0.08), (4, 0.12)]
    y = 0.06
    for count, x_off in row_specs:
        for col in range(count):
            x = 0.06 + x_off + col * 0.14
            centers[idx] = [x, y]
            idx += 1
        y += 0.17
        
    # Force-directed optimization parameters
    dt = 0.001
    repulsion = 100.0
    wall_rep = 200.0
    growth = 2e-5
    damping = 0.95
    
    vel = np.zeros((n, 2))
    
    # Run simulation
    for step in range(60000):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            
            # Wall repulsion forces
            fx, fy = 0.0, 0.0
            if xi < ri:
                fx += wall_rep * (ri - xi)
            if xi > 1 - ri:
                fx -= wall_rep * (xi - (1 - ri))
            if yi < ri:
                fy += wall_rep * (ri - yi)
            if yi > 1 - ri:
                fy -= wall_rep * (yi - (1 - ri))
            forces[i] = [fx, fy]
            
            # Pairwise repulsion
            for j in range(i + 1, n):
                dx = xi - centers[j, 0]
                dy = yi - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = ri + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    f = repulsion * overlap
                    fx = f * dx / dist
                    fy = f * dy / dist
                    forces[i] += [fx, fy]
                    forces[j] -= [fx, fy]
        
        vel = vel * damping + forces * dt
        centers += vel * dt
        
        # Gradually increase all radii
        radii += growth * np.ones(n)
    
    # Fine-tune with scipy optimization
    def objective(x):
        rs = x[2::3]
        return -np.sum(rs)
    
    def penalty(x):
        p = 0.0
        for i in range(n):
            xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
            # Boundary penalties
            if xi - ri < 0:
                p += 1000 * (xi - ri) ** 2
            if xi + ri > 1:
                p += 1000 * (xi + ri - 1) ** 2
            if yi - ri < 0:
                p += 1000 * (yi - ri) ** 2
            if yi + ri > 1:
                p += 1000 * (yi + ri - 1) ** 2
            
            for j in range(i + 1, n):
                xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
                dist = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2)
                if dist < ri + rj:
                    p += 1000 * (dist - ri - rj) ** 2
        return p
    
    def obj_penalized(x):
        return objective(x) + penalty(x)
    
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    bounds = [(0, 1)] * (2 * n) + [(0.001, 0.5)] * n
    
    result = minimize(obj_penalized, x0, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': 10000, 'ftol': 1e-15})
    
    final_centers = result.x[:2 * n].reshape(n, 2)
    final_radii = result.x[2 * n:]
    
    # Final clipping to ensure validity
    for i in range(n):
        final_centers[i, 0] = np.clip(final_centers[i, 0], final_radii[i], 1 - final_radii[i])
        final_centers[i, 1] = np.clip(final_centers[i, 1], final_radii[i], 1 - final_radii[i])
    
    return final_centers, final_radii, np.sum(final_radii)
```
