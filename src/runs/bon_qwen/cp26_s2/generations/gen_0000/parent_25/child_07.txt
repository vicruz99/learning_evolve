The problem requires packing 26 circles in a unit square to maximize the sum of radii. A target sum of 2.636 implies an average radius of approximately 0.101, which is slightly larger than the 0.1 radius of the optimal 25-circle square grid. This suggests that a hexagonal-like arrangement or a configuration with unequal radii is needed to achieve higher density.

I will implement a solution that:
1.  Initializes the circles in a hexagonal lattice pattern, which is denser than a square grid.
2.  Uses a penalty-based optimization approach with `scipy.optimize.minimize` (BFGS method) to refine the positions and radii.
3.  Runs multiple optimization attempts from different initializations to avoid local minima and selects the best valid packing found.
4.  Ensures all constraints (non-overlap, boundary containment) are satisfied with a tolerance.

The penalty function will penalize overlap and boundary violations quadratically. The optimization will try to minimize $-\sum r_i + \lambda \cdot \text{penalty}$.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Function to calculate penalty and objective
    def compute_penalty_and_obj(vars, lam=1000.0):
        centers = vars[:2*n].reshape((n, 2))
        radii = vars[2*n:]
        
        obj = -np.sum(radii)
        penalty = 0.0
        
        # Boundary penalties
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # x - r >= 0
            if x - r < 0:
                penalty += (x - r)**2
            # x + r <= 1
            if x + r > 1:
                penalty += (x + r - 1)**2
            # y - r >= 0
            if y - r < 0:
                penalty += (y - r)**2
            # y + r <= 1
            if y + r > 1:
                penalty += (y + r - 1)**2
        
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                dist = np.sqrt(dist_sq)
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    penalty += (r_sum - dist)**2
        
        return obj + lam * penalty

    # Generate hexagonal initialization
    def generate_hex_init():
        # Try to fit 26 circles in a hex pattern
        # Estimate radius ~ 0.1
        r_est = 0.1
        centers = []
        # Hex grid parameters
        # Rows
        for row in range(6):
            y = r_est + row * np.sqrt(3) * r_est
            if y + r_est > 1.0:
                break
            # Shift for even/odd rows
            shift = r_est if row % 2 == 1 else 0
            for col in range(6):
                x = r_est + col * 2 * r_est + shift
                if x + r_est > 1.0:
                    break
                centers.append([x, y])
                if len(centers) >= n:
                    break
            if len(centers) >= n:
                break
        
        # If we don't have 26, fill with random or adjust
        if len(centers) < n:
            # Fallback: random in valid area
            while len(centers) < n:
                centers.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        
        centers = centers[:n]
        radii = np.full(n, r_est)
        return np.array(centers), radii

    # Run multiple optimizations
    np.random.seed(42)
    for attempt in range(10):
        if attempt == 0:
            centers_init, radii_init = generate_hex_init()
        else:
            # Random perturbation or new random init
            centers_init = np.random.uniform(0.1, 0.9, (n, 2))
            radii_init = np.full(n, 0.08 + np.random.uniform(0, 0.05))
        
        vars_init = np.concatenate([centers_init.flatten(), radii_init])
        
        # Optimize
        res = minimize(compute_penalty_and_obj, vars_init, args=(500.0,), method='BFGS', options={'maxiter': 2000, 'ftol': 1e-12})
        
        if res.success or res.nit == 2000:
            centers_res = res.x[:2*n].reshape((n, 2))
            radii_res = res.x[2*n:]
            
            # Check validity
            valid = True
            # Check boundaries
            for i in range(n):
                x, y = centers_res[i]
                r = radii_res[i]
                if r < 0 or x - r < -1e-5 or x + r > 1 + 1e-5 or y - r < -1e-5 or y + r > 1 + 1e-5:
                    valid = False
                    break
            
            # Check overlaps
            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.sqrt(np.sum((centers_res[i] - centers_res[j])**2))
                        if dist < radii_res[i] + radii_res[j] - 1e-5:
                            valid = False
                            break
                    if not valid:
                        break
            
            if valid:
                current_sum = np.sum(radii_res)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers_res.copy()
                    best_radii = radii_res.copy()

    # If no valid solution found, fallback to a known simple packing (e.g. scaled hex)
    if best_centers is None:
        centers_fallback, radii_fallback = generate_hex_init()
        # Scale down to ensure validity
        scale = 0.95
        centers_fallback *= scale
        radii_fallback *= scale
        best_centers = centers_fallback
        best_radii = radii_fallback
        best_sum = np.sum(radii_fallback)

    return best_centers, best_radii, best_sum
