import numpy as np
from scipy.optimize import minimize

def compute_penalty_objective(v, N, lam):
    centers = v[:2*N].reshape(N, 2)
    radii = v[2*N:]
    obj = -np.sum(radii)
    pen = 0.0

    # Boundary constraints: circles must stay inside [0,1]^2
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        pen += max(0, r - x)**2
        pen += max(0, r - (1 - x))**2
        pen += max(0, r - y)**2
        pen += max(0, r - (1 - y))**2
        pen += max(0, -r)**2

    # Overlap constraints: distance between centers must be >= sum of radii
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            pen += max(0, radii[i] + radii[j] - dist)**2

    return obj + lam * pen

                idx += 1
    while idx < N:

        init_centers[idx, 1] = 0.5

    init_centers += np.arange(N).reshape(-1, 1) * 0.001

    v = v0
    # Gradually increase penalty weight to first fix violations, then maximize radii
    lams = [1000, 5000, 20000]
    for lam in lams:
        res = minimize(compute_penalty_objective, v, args=(N, lam),
                       method='L-BFGS-B', bounds=bounds,

        v = res.x

    centers = v[:2*N].reshape(N, 2)
    radii = v[2*N:]
    
    return centers, radii, np.sum(radii)
