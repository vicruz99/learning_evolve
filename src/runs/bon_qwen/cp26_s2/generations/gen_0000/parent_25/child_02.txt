import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # Helper function to calculate the cost (objective + penalties)
    def cost(v):
        # Unpack variables
        # v[0:52] -> centers (x, y) for 26 circles
        # v[52:78] -> radii for 26 circles
        c = v[:2*n].reshape(n, 2)
        r = v[2*n:]
        
        # Objective: Negative sum of radii (we minimize)
        obj = -np.sum(r)
        
        penalty = 0.0
        
        # Boundary penalties:
        # Circles must be inside [0, 1].
        # Constraints: r <= x <= 1-r  =>  r - x <= 0  and  x + r - 1 <= 0
        # Similarly for y.
        # Penalty is sum of squares of positive violations.
        
        # Left/Bottom: r - x >= 0 is violation? No, x >= r => r - x <= 0.
        # If r - x > 0, violation.
        penalty += np.sum(np.maximum(0, r - c[:, 0])**2) # x < r
        penalty += np.sum(np.maximum(0, r - c[:, 1])**2) # y < r
        
        # Right/Top: x + r - 1 >= 0 is violation? x <= 1-r => x + r - 1 <= 0.
        # If x + r - 1 > 0, violation.
        penalty += np.sum(np.maximum(0, c[:, 0] + r - 1)**2) # x > 1-r
        penalty += np.sum(np.maximum(0, c[:, 1] + r - 1)**2) # y > 1-r
        
        # Overlap penalties:
        # dist >= r_i + r_j  =>  dist - (r_i + r_j) >= 0
        # Violation if dist < r_i + r_j.
        # We calculate Euclidean distance squared to avoid sqrt, 
        # but penalty function is easier with linear distance for gradient?
        # Actually, dist - sum_r is smooth enough?
        # dist = sqrt(dx^2 + dy^2). Derivative is dx/dist.
        # At dist=0, gradient is undefined, but we handle that.
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                # Avoid division by zero or nan if dist is 0
                if dist < 1e-12:
                    dist = 1e-12
                
                overlap = r[i] + r[j] - dist
                if overlap > 0:
                    penalty += overlap**2
        
        # Weight for penalties needs to be high enough to enforce constraints
        # but not so high it dominates the objective completely.
        # 10000 is a safe large number.
        return obj + 10000.0 * penalty

    # Initialization
    # Start with a hexagonal-like grid distribution to help convergence
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05
    
    idx = 0
    # Try to fill rows
    # Row 0: 5 circles
    # Row 1: 6 circles (shifted)
    # Row 2: 5 circles
    # Row 3: 6 circles (shifted)
    # Row 4: 4 circles
    # Total 26
    
    # Simple grid init for robustness
    x_vals = np.linspace(0.1, 0.9, 6)
    y_vals = np.linspace(0.1, 0.9, 5)
    
    # Place 26 points
    # 6 columns x 5 rows = 30 spots. We pick 26.
    # Let's fill row by row.
    row_counts = [6, 5, 6, 5, 4] # Sum = 26
    
    current_idx = 0
    for r_idx, count in enumerate(row_counts):
        y = y_vals[r_idx]
        # Distribute x evenly for this row
        xs = np.linspace(0.15, 0.85, count)
        for k in range(count):
            centers[current_idx, 0] = xs[k]
            centers[current_idx, 1] = y
            current_idx += 1
            
    # Construct initial variable vector
    vars_init = np.concatenate([centers.flatten(), radii])
    
    # Bounds
    # x, y in [0, 1]
    # r in [0, 0.5] (max possible radius in unit square)
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    # Run Optimization
    # L-BFGS-B is good for bound constrained problems
    # We run it once from a good initialization. 
    # If needed, we could restart, but usually one good run suffices for N=26.
    
    res = minimize(cost, vars_init, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
    
    # Extract results
    c_opt = res.x[:2*n].reshape(n, 2)
    r_opt = res.x[2*n:]
    
    # Post-processing to ensure strict validity (handle tiny numerical errors)
    # The validator allows 1e-12 error, but let's be safe.
    # If any circle is out of bounds or overlapping, shrink slightly.
    
    # Check and fix boundaries
    for i in range(n):
        x, y = c_opt[i]
        r = r_opt[i]
        
        # Push inside if slightly out
        if x - r < 0:
            new_r = x # radius limited by left wall
            r_opt[i] = new_r
        elif x + r > 1:
            new_r = 1 - x
            r_opt[i] = new_r
            
        if y - r < 0:
            new_r = y
            r_opt[i] = new_r
        elif y + r > 1:
            new_r = 1 - y
            r_opt[i] = new_r
            
        # Ensure radius is non-negative
        r_opt[i] = max(0.0, r_opt[i])

    # Check and fix overlaps
    # Simple iterative shrinking of overlapping circles
    for _ in range(10): # Iterate a few times
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c_opt[i] - c_opt[j])**2))
                required_dist = r_opt[i] + r_opt[j]
                if dist < required_dist - 1e-12:
                    # Overlap detected. Reduce radii equally to fix.
                    excess = required_dist - dist
                    reduction = excess / 2.0
                    r_opt[i] -= reduction
                    r_opt[j] -= reduction
                    r_opt[i] = max(0.0, r_opt[i])
                    r_opt[j] = max(0.0, r_opt[j])
                    changed = True
        if not changed:
            break

    # Recalculate sum
    sum_radii = np.sum(r_opt)
    
    return c_opt, r_opt, float(sum_radii)
