

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii. The target is 2.636, meaning average radius ~0.1014.

2. **Optimization Strategy**: 
   - Use a penalty-based approach with L-BFGS-B to maximize sum of radii
   - Penalty term for non-overlap violations (smooth L2 penalty)
   - Bounds handle boundary constraints naturally
   - Iteratively increase penalty coefficient for better constraint satisfaction

3. **Key Insights**:
   - Hexagonal packing is optimal for equal circles, but variable radii may help near boundaries
   - The alternating LP for radii + position optimization can escape local minima
   - Multiple restarts from different initial configurations improve robustness
   - After initial optimization, a "squeeze" phase can push circles apart to make room for larger radii

4. **Implementation**:

   - Use scipy L-BFGS-B with penalty method for unconstrained optimization

   - Try multiple configurations and keep the best valid one

import numpy as np
from scipy.optimize import minimize
import math

def initialize_hexagonal(n, r):
    """Place circles in a hexagonal lattice pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r
    count = 0
    row = 0
    while count < n:
        y = r + row * r * math.sqrt(3)
        if y + r > 1.0 + 1e-10:
            row += 1
            continue
        row_offset = (row % 2) * r
        col = 0
        while count < n:
            x = r + row_offset + col * 2 * r
            if x + r > 1.0 + 1e-10:
                break
            centers[count] = [x, y]
            count += 1
            col += 1
        row += 1
    # Fill any remaining with small perturbation of existing positions
    while count < n:
        idx = count % (count) if count > 0 else 0
        centers[count] = centers[idx] + np.array([0.01, 0.01])
        count += 1
    return centers, radii

def initialize_grid(n, r):
    """Alternative grid initialization."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r
    cols = math.ceil(math.sqrt(n))
    rows = math.ceil(n / cols)
    spacing_x = (1 - 2 * r) / max(1, cols - 1)
    spacing_y = (1 - 2 * r) / max(1, rows - 1)
    count = 0
    for row in range(rows):
        for col in range(cols):
            if count >= n:
                break
            centers[count] = [r + col * spacing_x, r + row * spacing_y]
            count += 1
    return centers, radii

def compute_violation_penalty(centers, radii, n):
    """Compute L2 penalty for all constraint violations."""
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap * overlap
    return penalty

def objective_function(vars, n, penalty_coeff):
    """Objective: maximize sum of radii with penalty for overlaps."""
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    radii = np.maximum(radii, 1e-10)
    return -np.sum(radii) + penalty_coeff * compute_violation_penalty(centers, radii, n)

def check_validity(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r - 1e-10 or x > 1 - r + 1e-10 or y < r - 1e-10 or y > 1 - r + 1e-10:
            return False
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True

def project_to_feasible(centers, radii, n):
    """Project radii and centers to satisfy boundary constraints."""
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = min(radii[i], max(1e-10, max_r))
    return centers, radii

def optimize_positions_force(centers, radii, n, steps=500, lr=0.005):
    """Use force-directed method to optimize positions for given radii."""
    centers = centers.copy().astype(float)
    radii = radii.copy().astype(float)
    
    for step in range(steps):
        forces = np.zeros((n, 2))
        current_lr = lr * (1 - step / steps) ** 0.5
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                if dist < 1e-10:
                    dist = 1e-10
                    dx = 0.001
                    dy = 0.0
                    dist = math.sqrt(dx * dx + dy * dy)
                
                target_dist = radii[i] + radii[j]
                if dist < target_dist + 0.02:
                    overlap = max(0, target_dist + 0.01 - dist)
                    fx = (dx / dist) * overlap * 5.0
                    fy = (dy / dist) * overlap * 5.0
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
            
            # Boundary forces
            r = radii[i]
            margin = 0.005
            if centers[i, 0] < r + margin:
                forces[i, 0] += (r + margin - centers[i, 0]) * 10
            elif centers[i, 0] > 1 - r - margin:
                forces[i, 0] -= (centers[i, 0] - (1 - r - margin)) * 10
            if centers[i, 1] < r + margin:
                forces[i, 1] += (r + margin - centers[i, 1]) * 10
            elif centers[i, 1] > 1 - r - margin:
                forces[i, 1] -= (centers[i, 1] - (1 - r - margin)) * 10
        
        centers += forces * current_lr
        
        # Clamp to valid region
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 0]))
            centers[i, 1] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 1]))
    
    return centers, radii

def solve_radii_given_centers(centers, n, max_iter=50, tol=1e-8):
    """Iteratively compute maximum radii given fixed centers."""
    radii = np.ones(n) * 0.01
    
    for iteration in range(max_iter):
        new_radii = radii.copy()
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                max_r = min(max_r, dist - radii[j])
            new_radii[i] = max(1e-10, max_r)
        
        if np.max(np.abs(new_radii - radii)) < tol:
            break
        radii = new_radii.copy()
    
    return radii

def run_packing():
    n = 26
    
    # Bounds for optimization
    bounds = []
    for i in range(n):
        bounds.extend([(1e-8, 1 - 1e-8), (1e-8, 1 - 1e-8)])  # x, y
        bounds.append((1e-8, 0.45))  # radius
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Generate multiple initial configurations
    init_configs = []
    
    # Hexagonal with various starting radii
    for r in [0.05, 0.06, 0.07, 0.08, 0.09, 0.10]:
        c, r_arr = initialize_hexagonal(n, r)
        init_configs.append(np.concatenate([c.flatten(), r_arr]))
    
    # Grid initialization
    for r in [0.06, 0.08, 0.09]:
        c, r_arr = initialize_grid(n, r)
        init_configs.append(np.concatenate([c.flatten(), r_arr]))
    
    # Optimize each configuration
    for init_idx, init_vars in enumerate(init_configs):
        vars_curr = init_vars.copy()
        
        # Iterative penalty increase
        penalty = 100.0
        for it in range(60):
            res = minimize(
                objective_function,
                vars_curr,
                args=(n, penalty),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            vars_curr = res.x
            penalty *= 2.0
            
            # Project radii to be feasible with respect to boundaries
            centers_t = vars_curr[:2 * n].reshape(n, 2)
            radii_t = vars_curr[2 * n:]
            centers_t, radii_t = project_to_feasible(centers_t, radii_t, n)
            vars_curr[:2 * n] = centers_t.flatten()
            vars_curr[2 * n:] = radii_t
        
        centers = vars_curr[:2 * n].reshape(n, 2)
        radii = vars_curr[2 * n:]
        
        # Force-directed refinement
        centers, radii = optimize_positions_force(centers, radii, n, steps=300, lr=0.003)
        
        # Solve for optimal radii given positions
        radii = solve_radii_given_centers(centers, n, max_iter=100)
        
        # Another round of position optimization
        centers, radii = optimize_positions_force(centers, radii, n, steps=200, lr=0.002)
        radii = solve_radii_given_centers(centers, n, max_iter=100)
        
        # Final penalty-based optimization
        vars_curr = np.concatenate([centers.flatten(), radii])
        penalty = 500.0
        for it in range(30):
            res = minimize(
                objective_function,
                vars_curr,
                args=(n, penalty),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-15}
            )
            vars_curr = res.x
            penalty *= 2
        
        centers = vars_curr[:2 * n].reshape(n, 2)
        radii = vars_curr[2 * n:]
        centers, radii = project_to_feasible(centers, radii, n)
        
        # Verify and record best
        if check_validity(centers, radii, n):
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            # Try to fix by reducing radii slightly
            radii = radii * 0.99
            if check_validity(centers, radii, n):
                s = np.sum(radii)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers.copy()
                    best_radii = radii.copy()
    
    # Fallback if no valid solution found
    if best_centers is None:
        centers, radii = initialize_hexagonal(n, 0.07)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)
    
    return best_centers, best_radii, float(best_sum)
