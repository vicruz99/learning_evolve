import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Parameters
    n = 26
    
    # 1. Generate an initial hexagonal packing guess
    # Row distribution: 5, 6, 5, 6, 4 (Total 26)
    # This shape is somewhat hexagonal and fits well in a square.
    # We will compute approximate coordinates based on a target radius.
    # For N=26, a hex packing radius is approx 0.102.
    r_guess = 0.102
    h = np.sqrt(3) * r_guess  # Vertical spacing
    
    centers_guess = []
    current_row = 0
    y_pos = r_guess
    
    # Row counts
    row_counts = [5, 6, 5, 6, 4]
    
    for i, count in enumerate(row_counts):
        x_start = r_guess + (i % 2) * r_guess # Stagger odd rows
        # Center the row within the square
        row_width = (count - 1) * 2 * r_guess
        # Adjust start to center the row
        # Ideally, we want the row to be centered.
        # If centered, x coordinates would be: 0.5 - row_width/2 + r_guess ...
        # But let's just start at a standard position and let optimization fix it.
        # To help optimization, let's try to center them.
        x_start = 0.5 - (count * 2 * r_guess) / 2 + r_guess
        
        for j in range(count):
            x = x_start + j * 2 * r_guess
            centers_guess.append([x, y_pos])
        
        y_pos += h
    
    centers_guess = np.array(centers_guess)
    radii_guess = np.full(n, r_guess)
    
    # 2. Optimization Function
    # We want to maximize sum(radii).
    # Equivalent to minimizing -sum(radii).
    # Variables: centers (2n) and radii (n). Total 3n variables.
    # Constraints:
    # 1. r_i >= 0
    # 2. x_i - r_i >= 0, x_i + r_i <= 1, etc.
    # 3. dist(i,j) >= r_i + r_j
    
    # We will use a penalty method or SLSQP.
    # Given the non-convexity, a robust optimizer with a good initial guess is key.
    
    def objective(x):
        # x contains centers and radii
        # centers: x[0:2n], radii: x[2n:3n]
        # Actually, let's keep them separate for clarity in constraints
        # But scipy expects a 1D array.
        return -np.sum(x[2*n:])
    
    # Initial state vector
    x0 = np.concatenate([centers_guess.flatten(), radii_guess])
    
    # Bounds
    # Centers: [0, 1]
    # Radii: [0, 1]
    bounds = [(0, 1)] * (2 * n) + [(0, 1)] * n
    
    # Constraints
    constraints = []
    
    # Boundary constraints for each circle
    # x - r >= 0 => r - x <= 0
    # x + r <= 1 => x + r - 1 <= 0
    # Same for y
    # Inequality constraints must be >= 0 for SLSQP usually?
    # scipy.optimize.minimize constraints: fun(x) >= 0 for 'ineq'.
    
    # Helper to add constraints
    # We need:
    # x_i >= r_i  => x_i - r_i >= 0
    # 1 - x_i >= r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i
    # 1 - y_i >= r_i
    
    for i in range(n):
        idx_c_x = 2 * i
        idx_c_y = 2 * i + 1
        idx_r = 2 * n + i
        
        # x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx_c_x] - x[idx_r]
        })
        # 1 - x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1 - x[idx_c_x] - x[idx_r]
        })
        # y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx_c_y] - x[idx_r]
        })
        # 1 - y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1 - x[idx_c_y] - x[idx_r]
        })

    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    # dist(i,j) - r_i - r_j >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_c_x_i, idx_c_y_i = 2 * i, 2 * i + 1
            idx_c_x_j, idx_c_y_j = 2 * j, 2 * j + 1
            idx_r_i, idx_r_j = 2 * n + i, 2 * n + j
            
            # Distance squared constraint might be easier but sqrt is fine
            # To avoid sqrt, we could use squared, but constraint is linear in r.
            # Let's use sqrt.
            def overlap_constraint(x, i=i, j=j):
                dx = x[idx_c_x_i] - x[idx_c_x_j]
                dy = x[idx_c_y_i] - x[idx_c_y_j]
                dist = np.sqrt(dx*dx + dy*dy)
                return dist - (x[idx_r_i] + x[idx_r_j])
            
            constraints.append({
                'type': 'ineq',
                'fun': overlap_constraint
            })

    # Optimization
    # SLSQP handles non-linear constraints well.
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-12}
    )
    
    # Extract results
    centers_opt = result.x[:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:]
    
    # Post-processing: Ensure non-negative radii (though bounds handle this)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
