import numpy as np
import math
from scipy.optimize import minimize

def generate_initial_packing(n=26):
    """
    Generates an initial valid packing of n circles using a hexagonal lattice.
    """
    # Estimate an initial radius that allows fitting n circles in a hexagonal pattern
    # Area density of hex packing is pi/sqrt(12) ~ 0.9069
    # Approx radius for equal circles: r ~ sqrt(1 / (n * 2 * sqrt(3)))
    # For n=26, r ~ 0.105. Let's start slightly smaller to ensure validity.
    r_init = 0.08 
    
    centers = []
    
    # Generate hexagonal grid points
    # Vertical spacing: sqrt(3) * r
    # Horizontal spacing: 2 * r
    # Odd rows shifted by r
    
    dy = math.sqrt(3) * r_init
    dx = 2 * r_init
    
    y = r_init
    row_idx = 0
    
    while len(centers) < n:
        # Determine x-start based on row parity
        # Even rows (0, 2...): start at r_init
        # Odd rows (1, 3...): start at 2*r_init (shifted by r_init)
        # Actually, standard hex: 
        # Row 0: r, 3r, 5r...
        # Row 1: 2r, 4r, 6r...
        # Distance between (r, 0) and (2r, sqrt(3)r) is sqrt(r^2 + 3r^2) = 2r. Correct.
        
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
            
        x = x_start
        while x + r_init <= 1.0 + 1e-9:
            centers.append([x, y])
            if len(centers) == n:
                break
            x += dx
        
        if len(centers) < n:
            y += dy
            row_idx += 1
        else:
            break
            
    # If we still don't have enough (unlikely with r=0.08), add random points or refine
    # But with r=0.08, we should get plenty.
    # If we have more than n (loop condition handles this), trim.
    
    # Ensure we have exactly n
    if len(centers) > n:
        centers = centers[:n]
        
    # Convert to numpy array
    centers = np.array(centers)
    
    # Add small random perturbation to break symmetry
    np.random.seed(42)
    noise = np.random.uniform(-0.001, 0.001, centers.shape)
    centers += noise
    
    # Clip to valid range
    centers = np.clip(centers, r_init, 1.0 - r_init)
    
    return centers

def objective(x_flat, n):
    """
    Objective function: Minimize negative sum of radii.
    x_flat is [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = x_flat[2::3]
    return -np.sum(radii)

def get_constraints(n):
    """
    Returns list of constraint dictionaries for scipy.optimize.
    """
    constraints = []
    
    # 1. Boundary constraints
    # x_i - r_i >= 0
    # 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0
    # r_i >= 0 (handled by bounds)
    
    for i in range(n):
        x_idx = 3 * i
        y_idx = 3 * i + 1
        r_idx = 3 * i + 2
        
        # x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, idx=x_idx, r_idx=r_idx: x[idx] - x[r_idx]
        })
        # 1 - x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, idx=x_idx, r_idx=r_idx: 1.0 - x[idx] - x[r_idx]
        })
        # y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, idx=y_idx, r_idx=r_idx: x[idx] - x[r_idx]
        })
        # 1 - y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, idx=y_idx, r_idx=r_idx: 1.0 - x[idx] - x[r_idx]
        })
        
    # 2. Non-overlap constraints
    # dist(i, j) - (r_i + r_j) >= 0
    # sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            xi_idx = 3 * i
            yi_idx = 3 * i + 1
            ri_idx = 3 * i + 2
            
            xj_idx = 3 * j
            yj_idx = 3 * j + 1
            rj_idx = 3 * j + 2
            
            # Using squared distance might be numerically more stable or simpler?
            # But sqrt is needed for linear in r.
            # (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2 is equivalent but non-convex in r.
            # dist - ri - rj >= 0 is also non-convex but standard.
            
            def constraint_func(x, i_idx=xi_idx, j_idx=xj_idx, i_y=yi_idx, j_y=yj_idx, i_r=ri_idx, j_r=rj_idx):
                dx = x[i_idx] - x[j_idx]
                dy = x[i_y] - x[j_y]
                dist = math.sqrt(dx*dx + dy*dy)
                return dist - (x[i_r] + x[j_r])
            
            constraints.append({
                'type': 'ineq',
                'fun': constraint_func
            })
                
    return constraints

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Generate initial centers
    initial_centers = generate_initial_packing(n)
    
    # 2. Setup initial radii
    initial_radii = np.full(n, 0.05) # Start with small radius
    
    # 3. Construct initial vector [x1, y1, r1, ...]
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = initial_centers[i, 0]
        x0[3*i+1] = initial_centers[i, 1]
        x0[3*i+2] = initial_radii[i]
        
    # 4. Define bounds
    # x, y in [0, 1], r >= 0. 
    # Actually r is bounded by 0.5 theoretically, but constraints handle it.
    # Tighter bounds might help.
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # 5. Define constraints
    constraints = get_constraints(n)
    
    # 6. Run optimization
    # Using SLSQP
    try:
        result = minimize(
            objective, 
            x0, 
            args=(n,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
        )
        
        # Extract solution
        if result.success or result.fun < 0: # fun is negative sum
            final_x = result.x
        else:
            # Fallback to initial if optimization failed
            final_x = x0
            
    except Exception:
        final_x = x0

    # 7. Reconstruct centers and radii
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_x[3*i]
        final_centers[i, 1] = final_x[3*i+1]
        final_radii[i] = final_x[3*i+2]
        
    # 8. Clip radii to ensure non-negative (optimization might slightly violate due to tolerance)
    final_radii = np.maximum(final_radii, 0.0)
    
    # 9. Ensure centers are within bounds relative to radii
    # Optimization constraints should handle this, but just in case
    for i in range(n):
        r = final_radii[i]
        final_centers[i, 0] = np.clip(final_centers[i, 0], r, 1.0 - r)
        final_centers[i, 1] = np.clip(final_centers[i, 1], r, 1.0 - r)
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
