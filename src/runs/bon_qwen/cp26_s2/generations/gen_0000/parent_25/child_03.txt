import numpy as np
from scipy.optimize import minimize
import math


def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0

    for trial in range(8):
        centers, radii = init_packing(n, trial)
        centers, radii = force_relaxation(centers, radii, n, iterations=3000)
        centers, radii = sequential_expand(centers, radii, n, steps=2000)
        centers, radii = scipy_refine(centers, radii, n)

        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Ensure validity
    best_centers, best_radii = ensure_valid(best_centers, best_radii, n)

    return best_centers, best_radii, np.sum(best_radii)


def init_packing(n, seed):
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.095)

    idx = 0
    for row in range(6):
        if row % 2 == 0:
            num_in_row = 5
        else:
            num_in_row = 4
        if idx + num_in_row > n:
            num_in_row = n - idx

        y = 0.1 + row * 0.145
        offset = 0.1 if row % 2 == 1 else 0.0
        for col in range(num_in_row):
            x = 0.05 + col * 0.2 + offset + rng.uniform(-0.02, 0.02)
            y_actual = y + rng.uniform(-0.01, 0.01)
            x = max(0.05, min(0.95, x))
            y_actual = max(0.05, min(0.95, y_actual))
            centers[idx] = [x, y_actual]
            radii[idx] = 0.095 + rng.uniform(-0.005, 0.005)
            idx += 1

    return centers, radii


def force_relaxation(centers, radii, n, iterations=3000):
    centers = centers.copy()
    radii = radii.copy()
    dt = 0.05

    for it in range(iterations):
        forces = np.zeros_like(centers)

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]

                if dist < min_dist and dist > 1e-12:
                    overlap = min_dist - dist
                    fx = (dx / dist) * overlap
                    fy = (dy / dist) * overlap
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy

        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r:
                forces[i, 0] += (r - centers[i, 0]) * 2.0
            if centers[i, 0] > 1.0 - r:
                forces[i, 0] -= (centers[i, 0] - (1.0 - r)) * 2.0
            if centers[i, 1] < r:
                forces[i, 1] += (r - centers[i, 1]) * 2.0
            if centers[i, 1] > 1.0 - r:
                forces[i, 1] -= (centers[i, 1] - (1.0 - r)) * 2.0

        centers += forces * dt
        radii *= 1.0 + 0.0002

        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
            radii[i] = max(0.001, radii[i])

        if dt > 0.005:
            dt *= 0.999

    return centers, radii


def sequential_expand(centers, radii, n, steps=2000):
    centers = centers.copy()
    radii = radii.copy()

    for step in range(steps):
        radii *= 1.00015

        for iteration in range(200):
            max_force = 0.0
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    min_dist = radii[i] + radii[j]

                    if dist < min_dist and dist > 1e-12:
                        overlap = min_dist - dist
                        fx = (dx / dist) * overlap * 0.5
                        fy = (dy / dist) * overlap * 0.5
                        centers[i, 0] -= fx
                        centers[i, 1] -= fy
                        centers[j, 0] += fx
                        centers[j, 1] += fy
                        max_force = max(max_force, overlap)

            for i in range(n):
                r = radii[i]
                if centers[i, 0] < r:
                    centers[i, 0] = r
                if centers[i, 0] > 1.0 - r:
                    centers[i, 0] = 1.0 - r
                if centers[i, 1] < r:
                    centers[i, 1] = r
                if centers[i, 1] > 1.0 - r:
                    centers[i, 1] = 1.0 - r

            if max_force < 1e-10:
                break

    return centers, radii


def scipy_refine(centers, radii, n):
    x0 = np.concatenate([centers.flatten(), radii])

    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2 * i] - x[2 * n + i]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[2 * i] - x[2 * n + i]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2 * i + 1] - x[2 * n + i]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[2 * i + 1] - x[2 * n + i]})

    for i in range(n):
        for j in range(i + 1, n):
            def make_constraint(i_idx, j_idx):
                def constraint(x):
                    d = np.sqrt((x[2 * i_idx] - x[2 * j_idx]) ** 2 + (x[2 * i_idx + 1] - x[2 * j_idx + 1]) ** 2)
                    return d - x[2 * n + i_idx] - x[2 * n + j_idx]
                return constraint
            constraints.append({'type': 'ineq', 'fun': make_constraint(i, j)})

    def objective(x):
        return -np.sum(x[2 * n:])

    result = minimize(objective, x0, method='SLSQP', constraints=constraints,
                     options={'maxiter': 3000, 'ftol': 1e-15, 'disp': False})

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(result.x[2 * n:], 0)

    return centers_opt, radii_opt


def ensure_valid(centers, radii, n):
    centers = centers.copy()
    radii = radii.copy()

    for _ in range(100):
        valid = True
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r:
                centers[i, 0] = r
                valid = False
            if centers[i, 0] > 1.0 - r:
                centers[i, 0] = 1.0 - r
                valid = False
            if centers[i, 1] < r:
                centers[i, 1] = r
                valid = False
            if centers[i, 1] > 1.0 - r:
                centers[i, 1] = 1.0 - r
                valid = False

        if valid:
            break

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-12:
                    overlap = (min_dist - dist) / 2.0
                    fx = (dx / dist) * overlap
                    fy = (dy / dist) * overlap
                    centers[i, 0] -= fx
                    centers[i, 1] -= fy
                    centers[j, 0] += fx
                    centers[j, 1] += fy

    for i in range(n):
        r = radii[i]
        centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))

    return centers, radii
