# Helper function to calculate Euclidean distance
def dist(p1, p2):
    return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

# Helper function to calculate the max valid radius for a center given other circles
def calculate_max_radius(center, other_centers, other_radii, square_side=1.0):
    cx, cy = center
    r_bound = min(cx, square_side - cx, cy, square_side - cy)
    if r_bound < 0:
        return 0.0
    
    min_dist_to_other = r_bound
    for j in range(len(other_centers)):
        d = dist(center, other_centers[j])
        available_space = d - other_radii[j]
        if available_space < min_dist_to_other:
            min_dist_to_other = available_space
            
    return max(0.0, min_dist_to_other)
