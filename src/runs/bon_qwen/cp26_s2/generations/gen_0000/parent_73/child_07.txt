import numpy as np
from scipy.optimize import linprog, basinhopping, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    def get_max_sum_radii(centers):
        # Centers shape: (n, 2)
        # Maximize sum(r_i) => Minimize -sum(r_i)
        c = -np.ones(n)
        
        # Constraints:
        # 1. r_i <= x_i
        # 2. r_i <= 1 - x_i
        # 3. r_i <= y_i
        # 4. r_i <= 1 - y_i
        # 5. r_i + r_j <= dist(i, j)
        
        A_ub = []
        b_ub = []
        
        # Wall constraints
        for i in range(n):
            x, y = centers[i]
            
            row = np.zeros(n)
            row[i] = 1.0
            
            # r_i <= x
            A_ub.append(row)
            b_ub.append(x)
            
            # r_i <= 1 - x
            A_ub.append(row)
            b_ub.append(1.0 - x)
            
            # r_i <= y
            A_ub.append(row)
            b_ub.append(y)
            
            # r_i <= 1 - y
            A_ub.append(row)
            b_ub.append(1.0 - y)
            
        # Pairwise constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i][0] - centers[j][0])**2 + 
                               (centers[i][1] - centers[j][1])**2)
                
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        bounds = [(0, None) for _ in range(n)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                return -res.fun, res.x
        except Exception:
            pass
            
        return 0.0, np.zeros(n)

    def objective(x_flat):
        # x_flat is (2*n,) vector of centers
        centers = x_flat.reshape((n, 2))
        # Ensure centers are strictly inside for numerical stability in LP 
        # (though LP handles boundaries, 0 distance might cause singularities if not careful, 
        # but linprog is robust. Clamping helps).
        # We rely on the optimizer bounds to keep centers in [0,1].
        
        sum_r, radii = get_max_sum_radii(centers)
        # We want to maximize sum_r, so minimize -sum_r
        return -sum_r, radii

    def minimize_wrapper(x_flat):
        val, _ = objective(x_flat)
        return val

    # Initial guess: 5x5 grid + 1 point
    # Grid spacing 0.2, but centers need margin 0.1.
    # Let's create a grid that fits well.
    # 5x5 grid of centers at 0.1, 0.3, 0.5, 0.7, 0.9
    # This gives 25 points. We need 26.
    # Place the 26th at center (0.5, 0.5) slightly perturbed? 
    # Actually, just random perturbation is safer for global opt.
    
    best_result = None
    best_sum = -np.inf
    
    # Try a few different random seeds/configurations
    for _ in range(5):
        # Generate random centers in [0.05, 0.95] to avoid immediate wall issues
        x0 = np.random.uniform(0.05, 0.95, size=(2*n))
        
        # Local optimization to refine this random guess
        res = minimize(minimize_wrapper, x0, method='L-BFGS-B', 
                       bounds=[(0.0, 1.0)] * (2*n))
        
        val = res.fun
        if val < best_sum:
            best_sum = val
            best_result = res.x
            
        # Also try a BasinHopping search for global optimum
        # BasinHopping is good for these geometric problems
        # We use a min step to ensure we explore the space
        minimizer_kwargs = {"method": "L-BFGS-B", "bounds": [(0.0, 1.0)] * (2*n)}
        
        # A quick basin hop run
        bh = basinhopping(minimize_wrapper, res.x, 
                          niter=20, 
                          minimizer_kwargs=minimizer_kwargs,
                          stepsize=0.05)
        
        if bh.fun < best_sum:
            best_sum = bh.fun
            best_result = bh.x

    # Extract final centers and radii
    final_centers = best_result.reshape((n, 2))
    # Solve LP one last time to get optimal radii for these centers
    sum_radii, final_radii = get_max_sum_radii(final_centers)
    
    # Clamp radii to non-negative just in case
    final_radii = np.maximum(final_radii, 0)
    
    return final_centers, final_radii, sum_radii
