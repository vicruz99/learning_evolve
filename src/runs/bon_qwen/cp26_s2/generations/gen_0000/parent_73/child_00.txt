import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp_radii(centers):
    """
    Solves the Linear Programming problem to maximize sum of radii given fixed centers.
    """
    n = centers.shape[0]
    
    # 1. Determine variable bounds based on distance to square boundaries
    # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # So 0 <= r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1-x, y, 1-y)
        # Ensure non-negative bound (though centers should be in [0,1])
        max_r = max(0.0, max_r)
        bounds.append((0.0, max_r))

    # 2. Setup pairwise non-overlap constraints: r_i + r_j <= dist(c_i, c_j)
    # We need A_ub @ r <= b_ub
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    row_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dist
            row_idx += 1
            
    # 3. Objective: Maximize sum(r_i) <=> Minimize -sum(r_i)
    c = -np.ones(n)
    
    # 4. Solve
    # Using HiGHS method as it is robust and fast
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            # res.fun is min(-sum), so optimal sum is -res.fun
            return -res.fun, res.x
        else:
            # If LP fails (e.g., infeasible), return 0
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for center optimization.
    We want to maximize sum of radii, so we minimize negative sum.
    """
    centers = centers_flat.reshape(-1, 2)
    
    # Constraint: Centers must be within [0, 1] to have valid radii.
    # We clip them to ensure feasibility for the LP bounds calculation.
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_radii, _ = solve_lp_radii(centers)
    
    return -sum_radii

def get_initial_hex_centers(n_circles):
    """
    Generates an initial configuration of centers on a hexagonal lattice.
    """
    centers = []
    
    # Heuristic spacing for 26 circles in unit square
    # Area ~ 1, N=26. If density ~ 0.9, r ~ sqrt(1/(26*pi*0.9)) ~ 0.115
    # Spacing d = 2r ~ 0.23. Let's try a bit tighter 0.20 to fill space.
    spacing = 0.20
    
    row = 0
    while len(centers) < n_circles:
        # Hexagonal packing y-offset: row * spacing * sqrt(3)/2
        y = row * spacing * np.sqrt(3) / 2
        
        # x-offset for odd rows
        x_offset = (spacing / 2) * (row % 2)
        
        # Add points in this row
        x = 0.05 + x_offset # Small margin from left
        while x <= 0.95: # Small margin from right
            centers.append([x, y])
            if len(centers) >= n_circles:
                break
            x += spacing
        
        # Stop if we go beyond top margin
        if y > 0.95:
            break
            
        row += 1
        
    # If we still don't have enough (unlikely with this logic), pad with random
    while len(centers) < n_circles:
        centers.append([0.5 + np.random.rand() * 0.1 - 0.05, 0.5 + np.random.rand() * 0.1 - 0.05])
        
    return np.array(centers[:n_circles])

def run_packing():
    n_circles = 26
    
    # 1. Initialize centers
    # Seed for reproducibility
    np.random.seed(42)
    initial_centers = get_initial_hex_centers(n_circles)

    # Using Nelder-Mead (derivative-free) which works well for this type of landscape
    # Bounds for centers: [0, 1] for each coordinate

    res = minimize(
        objective_function, 
        x0=initial_centers.flatten(), 
        method='Nelder-Mead', 
        bounds=bounds_centers,
        options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-8}
    )

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any(): return False
    if np.isnan(radii).any(): return False
    for i in range(n):
        if radii[i] < 0: return False
        elif np.isnan(radii[i]): return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
