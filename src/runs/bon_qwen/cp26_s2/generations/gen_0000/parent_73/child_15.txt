import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization followed by SLSQP optimization.
    """
    n_circles = 26
    
    # --- 1. Generate Hexagonal Lattice Candidates ---
    # We aim for a density close to hexagonal packing.
    # Estimate radius for 26 circles: Area ~ 0.9069, pi*r^2*26 ~ 0.9069 -> r ~ 0.105
    # Lattice spacing 2r ~ 0.21. Vertical spacing sqrt(3)*r ~ 0.18.
    r_est = 0.11
    dy = np.sqrt(3) * r_est
    dx = 2 * r_est
    
    candidates = []
    row = 0
    while True:
        y = r_est + row * dy
        if y > 1 - r_est + 0.001: # Allow slight boundary exceedance for optimization to pull back
            break
        
        # Shift every other row
        x_start = r_est if row % 2 == 0 else r_est + dx / 2
        
        col = 0
        while True:
            x = x_start + col * dx
            if x > 1 - r_est + 0.001:
                break
            candidates.append([x, y])
            col += 1
        row += 1
        
    candidates = np.array(candidates)
    
    # --- 2. Select 26 Best Candidates ---
    # We select 26 points that maximize the minimum distance to other selected points and boundaries.
    selected_indices = []
    available = list(range(len(candidates)))
    
    # Initial radii estimate based on boundary distance
    boundary_dist = np.minimum(
        np.minimum(candidates[:, 0], 1 - candidates[:, 0]),
        np.minimum(candidates[:, 1], 1 - candidates[:, 1])
    )
    
    for _ in range(n_circles):
        best_idx = -1
        best_min_dist = -1
        
        for idx in available:
            c = candidates[idx]
            # Distance to boundary
            d_bound = boundary_dist[idx]
            
            # Distance to already selected circles
            d_neighbors = np.inf
            for s_idx in selected_indices:
                s = candidates[s_idx]
                dist = np.linalg.norm(c - s)
                if dist < d_neighbors:
                    d_neighbors = dist
            
            # The limiting factor is the min of boundary and neighbor distance
            # We want to pick points that are "roomy"
            current_min = min(d_bound, d_neighbors / 2) # /2 because d_neighbors is center-to-center
            
            if current_min > best_min_dist:
                best_min_dist = current_min
                best_idx = idx
        
        selected_indices.append(best_idx)
        available.remove(best_idx)
        
    centers = candidates[selected_indices]
    
    # --- 3. Optimize Centers and Radii ---
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(n_circles * 3)
    for i in range(n_circles):
        # Initialize radii to a safe value
        r_safe = 0.05
        # Check local distances
        for j in range(n_circles):
            if i != j:
                d = np.linalg.norm(centers[i] - centers[j])
                r_safe = min(r_safe, d / 2 - 0.001)
        # Check boundary
        x, y = centers[i]
        r_safe = min(r_safe, x - 0.001, 1 - x - 0.001, y - 0.001, 1 - y - 0.001)
        r_safe = max(r_safe, 0.01)
        
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = r_safe

    def objective(x):
        # Maximize sum of radii => Minimize negative sum
        radii = x[2::3]
        return -np.sum(radii)

    def constraints(x):
        cons = []
        # Boundary constraints: r <= x <= 1-r  => x-r >= 0, 1-x-r >= 0
        # Same for y
        for i in range(n_circles):
            xi = x[3*i]
            yi = x[3*i+1]
            ri = x[3*i+2]
            
            cons.append(xi - ri)
            cons.append(1 - xi - ri)
            cons.append(yi - ri)
            cons.append(1 - yi - ri)
            
            # Overlap constraints
            for j in range(i + 1, n_circles):
                xj = x[3*j]
                yj = x[3*j+1]
                rj = x[3*j+2]
                
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                cons.append(dist - ri - rj)
                
        return cons

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n_circles
    
    # Use SLSQP
    cons = opt.NonlinearConstraint(constraints, 0, np.inf)
    
    # Run optimization multiple times with slight perturbations to avoid local minima
    best_result = None
    best_loss = np.inf
    
    for trial in range(5):
        # Perturb initial guess slightly
        x_trial = x0.copy()
        if trial > 0:
            noise = np.random.randn(len(x0)) * 0.005
            x_trial = x_trial + noise
            x_trial = np.clip(x_trial, 0, 1) # Keep bounds loosely
            
        try:
            res = opt.minimize(objective, x_trial, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 2000, 'ftol': 1e-10})
            if res.success and res.fun < best_loss:
                best_loss = res.fun
                best_result = res
        except Exception:
            continue
            
    if best_result is None:
        # Fallback to simple grid if optimization fails
        best_result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})

    # Extract results
    final_centers = np.zeros((n_circles, 2))
    final_radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        final_centers[i] = best_result.x[3*i:3*i+2]
        final_radii[i] = best_result.x[3*i+2]
        
    # Ensure radii are non-negative
    final_radii = np.maximum(final_radii, 1e-9)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
