def calculate_overlap_penalty(centers, radii, eps=1e-4):
    """
    Calculates a penalty based on overlaps between circles and boundaries.
    Returns a positive value if there are violations, 0 otherwise.
    """
    penalty = 0.0
    n = centers.shape[0]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # Equivalent to: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    # We penalize if x < r, i.e., r - x > 0. Penalty max(0, r - x)^2
    
    # Left boundary
    violations = radii - centers[:, 0]
    penalty += np.sum(np.maximum(violations, 0)**2)
    
    # Right boundary
    violations = (radii + centers[:, 0]) - 1
    penalty += np.sum(np.maximum(violations, 0)**2)
    
    # Bottom boundary
    violations = radii - centers[:, 1]
    penalty += np.sum(np.maximum(violations, 0)**2)
    
    # Top boundary
    violations = (radii + centers[:, 1]) - 1
    penalty += np.sum(np.maximum(violations, 0)**2)
    
    # Inter-circle constraints: dist >= r_i + r_j
    # Violation if dist < r_i + r_j, i.e., r_i + r_j - dist > 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            req_dist = radii[i] + radii[j]
            violation = req_dist - dist
            if violation > 0:
                penalty += violation**2
                
    return penalty
