import numpy as np
from scipy.optimize import linprog
import math

def get_sum_radii(centers):
    """
    Solves an LP to find the maximum sum of radii for fixed centers.
    """
    n = centers.shape[0]
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = np.ones(n) * -1.0

    # Constraints: A_ub @ r <= b_ub
    # We will build these lists and convert to sparse/dense later
    A_rows = []
    b_cols = []

    # Circle-Circle constraints: r_i + r_j <= dist(c_i, c_j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = [0.0] * n
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_cols.append(dist)

    # Boundary constraints:
    # r_i <= x_i, r_i <= 1 - x_i, r_i <= y_i, r_i <= 1 - y_i
    for i in range(n):
        x, y = centers[i]
        row = [0.0] * n
        row[i] = 1.0
        A_rows.append(row)
        b_cols.append(x)
        
        A_rows.append(row)
        b_cols.append(1.0 - x)
        
        A_rows.append(row)
        b_cols.append(y)
        
        A_rows.append(row)
        b_cols.append(1.0 - y)

    # Lower bounds for r_i >= 0
    bounds = [(0, None) for _ in range(n)]

    # Solve LP
    res = linprog(c_obj, A_ub=A_rows, b_ub=b_cols, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun
    else:
        # Fallback if LP fails (rare), return a small valid sum or 0
        return 0.0
