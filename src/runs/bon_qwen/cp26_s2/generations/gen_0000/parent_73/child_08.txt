import numpy as np
from scipy.optimize import minimize


def hexagonal_init(n):
    """Initialize circles in a hexagonal lattice pattern"""
    centers = np.zeros((n, 2))
    idx = 0

    row = 0
    col = 0
    spacing = 0.13

    while idx < n:
        x = col * spacing * np.sqrt(3.0)
        y = row * spacing * 1.5
        if row % 2 == 1:
            x += spacing * np.sqrt(3.0) / 2.0

        centers[idx, 0] = x
        centers[idx, 1] = y
        idx += 1
        col += 1
        if col > 10:
            col = 0
            row += 1

    # Normalize to [0.05, 0.95]
    x_min, x_max = centers[:, 0].min(), centers[:, 0].max()
    y_min, y_max = centers[:, 1].min(), centers[:, 1].max()

    x_range = x_max - x_min
    y_range = y_max - y_min

    if x_range > 1e-10:
        centers[:, 0] = 0.05 + (centers[:, 0] - x_min) / x_range * 0.9
    else:
        centers[:, 0] = 0.5

    if y_range > 1e-10:
        centers[:, 1] = 0.05 + (centers[:, 1] - y_min) / y_range * 0.9
    else:
        centers[:, 1] = 0.5

    return centers


def compute_objective(params, n, penalty_weight):
    """Compute objective: negative sum of radii plus constraint penalties"""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    # Main objective
    obj = -np.sum(radii)

    # Constraint penalties
    penalty = 0.0

    # Circle-circle non-overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-15)
            min_dist = radii[i] + radii[j]
            gap = max(0.0, min_dist - dist)
            penalty += gap * gap

    # Circle-boundary
    for i in range(n):
        x, y = centers[i, 0], centers[i, 1]
        r = radii[i]
        penalty += max(0.0, r - x) ** 2
        penalty += max(0.0, x + r - 1.0) ** 2
        penalty += max(0.0, r - y) ** 2
        penalty += max(0.0, y + r - 1.0) ** 2

    # Non-negative radii
    for i in range(n):
        if radii[i] < 0:
            penalty += radii[i] * radii[i]

    obj += penalty_weight * penalty

    return obj


def check_validity(centers, radii, n):
    """Check if packing is valid"""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def optimize_packing(params0, n):
    """Optimize packing with adaptive penalty weights"""
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    # Progressive penalty increase
    penalty_weights = [50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0]

    params = params0.copy()
    for pw in penalty_weights:
        result = minimize(
            compute_objective,
            params,
            args=(n, pw),
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 15000, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        params = result.x

    return params


def coordinate_descent_refine(centers, radii, n, iterations=50):
    """Refine packing using coordinate descent: optimize one circle at a time"""
    centers = centers.copy()
    radii = radii.copy()

    for _ in range(iterations):
        for i in range(n):
            # Save original
            orig_center = centers[i].copy()
            orig_r = radii[i]

            def obj_local(p):
                x, y, r = p
                obj = -r
                penalty = 0.0

                # Check against all other circles
                for j in range(n):
                    if j == i:
                        continue
                    dx = x - centers[j, 0]
                    dy = y - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy + 1e-15)
                    min_dist = r + radii[j]
                    gap = max(0.0, min_dist - dist)
                    penalty += gap * gap

                # Boundary
                penalty += max(0.0, r - x) ** 2
                penalty += max(0.0, x + r - 1.0) ** 2
                penalty += max(0.0, r - y) ** 2
                penalty += max(0.0, y + r - 1.0) ** 2
                penalty += max(0.0, -r) ** 2

                obj += 10000.0 * penalty
                return obj

            p0 = [orig_center[0], orig_center[1], orig_r]
            bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)]

            result = minimize(obj_local, p0, bounds=bounds, method='L-BFGS-B',
                             options={'maxiter': 5000})

            centers[i, 0] = result.x[0]
            centers[i, 1] = result.x[1]
            radii[i] = result.x[2]

    return centers, radii


def simulated_annealing_refine(centers, radii, n, max_iter=50000):
    """Refine packing using simulated annealing"""
    centers = centers.copy()
    radii = radii.copy()

    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)

    temp = 0.01
    cooling_rate = 0.99995

    for iteration in range(max_iter):
        i = np.random.randint(n)

        new_centers = centers.copy()
        new_radii = radii.copy()

        # Perturb position
        new_centers[i, 0] += np.random.randn() * 0.01
        new_centers[i, 1] += np.random.randn() * 0.01

        # Perturb radius
        new_radii[i] += np.random.randn() * 0.003

        # Clip
        new_centers[i, 0] = np.clip(new_centers[i, 0], 0.0, 1.0)
        new_centers[i, 1] = np.clip(new_centers[i, 1], 0.0, 1.0)
        new_radii[i] = max(0.0, new_radii[i])

        # Quick validity check

        r = new_radii[i]

        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            valid = False

        if valid:
            for j in range(n):
                if j == i:
                    continue
                dx = x - new_centers[j, 0]
                dy = y - new_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < r + new_radii[j] - 1e-12:
                    valid = False
                    break

        if not valid:
            continue

        new_sum = np.sum(new_radii)
        delta = new_sum - best_sum

            radii = new_radii.copy()

                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum = new_sum

        temp *= cooling_rate

    return best_centers, best_radii, best_sum
