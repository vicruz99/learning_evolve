def constraint_non_overlap(params, n, i, j):
    """
    Constraint for non-overlap between circle i and j.
    dist >= r_i + r_j  =>  dist^2 >= (r_i + r_j)^2
    Returns value >= 0.
    """
    x_i, y_i, r_i = params[3*i], params[3*i+1], params[3*i+2]
