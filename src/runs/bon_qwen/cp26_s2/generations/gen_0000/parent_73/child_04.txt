def compute_penalty(centers, radii, penalty_weight=1000.0):
    """
    Computes the penalty for invalid packing (boundary violations and overlaps).
    """
    n = len(radii)
    penalty = 0.0
    
    # Boundary penalties (Vectorized)
    # Left: x >= r  =>  r - x <= 0
    val = radii - centers[:, 0]
    mask = val > 0
    if np.any(mask):
        penalty += penalty_weight * np.sum((val[mask] ** 2))
    
    # Right: x <= 1 - r  =>  x + r - 1 <= 0
    val = centers[:, 0] + radii - 1.0
    mask = val > 0
    if np.any(mask):
        penalty += penalty_weight * np.sum((val[mask] ** 2))
        
    # Bottom: y >= r
    val = radii - centers[:, 1]
    mask = val > 0
    if np.any(mask):
        penalty += penalty_weight * np.sum((val[mask] ** 2))
        
    # Top: y <= 1 - r
    val = centers[:, 1] + radii - 1.0
    mask = val > 0
    if np.any(mask):
