def objective_function(coords, r, n):
    """
    Calculates the penalty for overlapping circles and boundary violations.
    coords: flattened array of x1, y1, x2, y2, ...
    r: radius of circles
    n: number of circles
    """
    centers = coords.reshape(n, 2)
    
    penalty = 0.0
    
    # Boundary penalties
    # For each circle, check distance to walls
    # Constraint: r <= x <= 1-r  =>  x - r >= 0 and 1 - r - x >= 0
    # Violation if x < r or x > 1-r
    
    # Vectorized boundary check
    xs = centers[:, 0]
    ys = centers[:, 1]
    
    # Left and Right walls
    # If x < r, violation is (r - x)^2
    # If x > 1-r, violation is (x - (1-r))^2
    # Using max(0, val)^2 is smooth enough for optimizers if val is 0 derivative
    # Actually (r-x)^2 when x<r. Derivative 2(x-r). At x=r, 0.
    
    viol_x_left = np.maximum(0, r - xs)
    viol_x_right = np.maximum(0, xs - (1 - r))
    
    viol_y_bottom = np.maximum(0, r - ys)
    viol_y_top = np.maximum(0, ys - (1 - r))
    
    penalty += np.sum(viol_x_left**2)
    penalty += np.sum(viol_x_right**2)
    penalty += np.sum(viol_y_bottom**2)
    penalty += np.sum(viol_y_top**2)
    
    # Overlap penalties
    # Distance squared between i and j must be >= (2r)^2 = 4r^2
    # Violation if dist_sq < 4r^2 => 4r^2 - dist_sq > 0
    # Penalty (4r^2 - dist_sq)^2
    
    # Compute pairwise distance squared matrix
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # We only need upper triangle to avoid double counting and self
    # But summing all and dividing by 2 is fine, or just mask
    # Mask for i < j
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    violations = np.maximum(0, 4 * r**2 - dist_sq)
    # Only consider upper triangle
    violations = violations[mask]
    
    penalty += np.sum(violations**2)
    
    return penalty

def optimize_packing(n, target_radius):
    """
    Tries to find a valid configuration for n circles of radius target_radius.
    Returns (coords, min_penalty).
    """
    # Initial guess
    centers_init = generate_hexagonal_grid(n)
    coords_init = centers_init.flatten()
    
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    
    # Optimization
    # Use SLSQP or L-BFGS-B. SLSQP handles constraints well, but we use penalty method.
    # L-BFGS-B is fast for bound constraints.
    
    res = opt.minimize(
        objective_function,
        coords_init,
        args=(target_radius, n),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-6}
    )
    
    return res.x, res.fun
