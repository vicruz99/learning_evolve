import numpy as np
from scipy.optimize import minimize

def compute_loss(vars):
    """
    Computes the objective function: negative sum of radii plus penalty terms
    for boundary violations and circle overlaps.
    """
    n = 26
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    
    # Objective: maximize sum(radii) => minimize -sum(radii)
    loss = -np.sum(radii)
    
    # Penalty weight for constraints
    W = 50000.0
    
    # Boundary penalties: circles must be inside [0,1]x[0,1]
    # x - r >= 0  =>  r - x <= 0
    loss += W * np.sum(np.maximum(radii - centers[:, 0], 0.0)**2)
    # 1 - x - r >= 0  =>  r - (1 - x) <= 0
    loss += W * np.sum(np.maximum(radii - (1.0 - centers[:, 0]), 0.0)**2)
    # y - r >= 0
    loss += W * np.sum(np.maximum(radii - centers[:, 1], 0.0)**2)
    # 1 - y - r >= 0
    loss += W * np.sum(np.maximum(radii - (1.0 - centers[:, 1]), 0.0)**2)
    
    # Overlap penalties: distance between centers >= sum of radii
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dist = np.sqrt(dx**2 + dy**2)
    radii_sum = radii[:, None] + radii[None, :]
    overlap = radii_sum - dist
    
    # Only consider pairs i < j to avoid double counting and self-overlap
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    loss += W * np.sum(np.maximum(overlap[mask], 0.0)**2)
    
    return loss

def get_hex_initial(n):
    """Generates an initial configuration based on a hexagonal lattice."""
    r_start = 0.09
    centers = []
    row = 0
    while len(centers) < n:
        y = r_start + row * np.sqrt(3) * r_start
        if y + r_start > 1.05:
            if len(centers) >= n:
                break
            r_start -= 0.005
            continue
        
        col = 0
        while len(centers) < n:
            x = r_start + col * 2 * r_start + (row % 2) * r_start
            if x + r_start > 1.05:
                break
            centers.append([x, y])
            col += 1
        row += 1
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_start)
    return np.concatenate([centers.flatten(), radii])

def get_random_initial(n, seed):
    """Generates a random initial configuration."""
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.1, 0.9, size=(n, 2))
    radii = rng.uniform(0.05, 0.12, size=n)
    return np.concatenate([centers.flatten(), radii])

def run_optimization(x0):
    """Runs the L-BFGS-B optimizer from a given initial state."""
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
    res = minimize(compute_loss, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False})
    return res.x, res.fun

def run_packing():
    """
    Main function to pack 26 circles in a unit square maximizing sum of radii.
    Returns (centers, radii, sum_radii).
    """
    best_loss = np.inf
    best_vars = None
    
    # Try multiple hexagonal starts with small jitter to break symmetry
    for i in range(5):
        x0 = get_hex_initial(26)
        jitter = np.random.uniform(-0.001, 0.001, size=x0.shape)
        x0_jittered = x0 + jitter
        vars_opt, loss = run_optimization(x0_jittered)
        if loss < best_loss:
            best_loss = loss
            best_vars = vars_opt
            
    # Try random starts to escape potential local minima
    for i in range(3):
        x0 = get_random_initial(26, seed=i)
        vars_opt, loss = run_optimization(x0)
        if loss < best_loss:
            best_loss = loss
            best_vars = vars_opt
            
    centers = best_vars[:52].reshape(26, 2)
    radii = best_vars[52:]
    
    # Ensure strict feasibility against numerical drift
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers, radii, float(np.sum(radii))
