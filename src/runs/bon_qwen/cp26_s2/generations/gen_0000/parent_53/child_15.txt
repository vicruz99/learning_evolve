def calculate_penalty(centers, radii, weight):
    """
    Calculates the penalty for constraint violations.
    Penalties are for:
    1. Circles outside the unit square.
    2. Overlapping circles.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary penalties
    # x - r >= 0  => r - x <= 0
    # 1 - x - r >= 0 => r + x - 1 <= 0
    # y - r >= 0
    # 1 - y - r >= 0
    
    # Vectorized boundary checks
    # r - x
    viol_x_min = np.maximum(0, radii - centers[:, 0])
    # r + x - 1
    viol_x_max = np.maximum(0, radii + centers[:, 0] - 1.0)
    # r - y
    viol_y_min = np.maximum(0, radii - centers[:, 1])
    # r + y - 1
    viol_y_max = np.maximum(0, radii + centers[:, 1] - 1.0)
    
    boundary_violation = (viol_x_min**2 + viol_x_max**2 + 
                          viol_y_min**2 + viol_y_max**2)
    penalty += np.sum(boundary_violation)
    
    # Overlap penalties
    # dist^2 >= (r_i + r_j)^2
    # Violation: (r_i + r_j)^2 - dist^2 > 0
    
    # Compute pairwise squared distances and sum of radii squared
    # Using broadcasting for efficiency
    # centers: (n, 2)
    # dist_sq: (n, n)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    dist_sq = np.sum(diff**2, axis=2)
    
    # radii: (n,)
    # sum_radii_sq: (n, n)
    sum_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
    sum_radii_sq = sum_radii**2
    
    # We only need upper triangle to avoid double counting and self-overlap
    # But for penalty, summing all is fine as long as diagonal is 0 (dist=0, sum_r=r+r, penalty > 0?)
    # Wait, diagonal: dist_sq=0, sum_radii_sq = (2r)^2. Violation = 4r^2.
    # We must exclude diagonal.
    
    # Mask for upper triangle (i < j)
    i, j = np.triu_indices(n, k=1)
    
    # Violations for pairs
    viol = np.maximum(0, sum_radii_sq[i, j] - dist_sq[i, j])
    penalty += np.sum(viol**2) # Squared penalty for smoothness
    
    return weight * penalty

    # params[0] = x0, params[1] = y0, params[2] = r0
    # So centers[:, 0] = params[0::3]

    # radii = params[2::3]
    
    centers_x = params[0::3]
    centers_y = params[1::3]
    centers = np.column_stack((centers_x, centers_y))
    radii = params[2::3]
    
    # Clip radii to be non-negative (bounds should handle this, but safety)
    radii = np.maximum(radii, 0)
    
    obj_val = -np.sum(radii)
    obj_val += calculate_penalty(centers, radii, weight)
    
    return obj_val
