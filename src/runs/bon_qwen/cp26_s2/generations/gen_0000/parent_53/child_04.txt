import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the placement and radii of 26 circles in a unit square 
    to maximize the sum of radii using a penalty-based optimization.
    """
    n = 26
    penalty_weight = 1000.0
    n_iterations = 10  # Number of restarts with different initial configurations

    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None

    # Helper to validate and calculate sum
    def evaluate_state(centers, radii):
        # Boundary check
        if np.any(centers - radii[:, np.newaxis] < -1e-6):
            return -1e9
        if np.any(centers + radii[:, np.newaxis] > 1 + 1e-6):
            return -1e9
        if np.any(radii < -1e-6):
            return -1e9
            
        # Overlap check
        valid = True
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-6:
                    valid = False
                    break
            if not valid:
                break
        return np.sum(radii) if valid else -1e9

    for _ in range(n_iterations):
        # --- 1. Initialization: Hexagonal Lattice ---
        centers = np.zeros((n, 2))
        # Estimate radius based on hexagonal packing density for initialization
        # Area approx 0.9069 => 26 * pi * r^2 ~ 0.9 => r ~ 0.105
        init_r = 0.105
        
        count = 0
        row = 0
        col = 0
        
        # Fill rows with hexagonal offset
        while count < n:
            y = init_r + row * np.sqrt(3) * init_r
            x_start = init_r + (row % 2) * init_r
            
            while count < n:
                x = x_start + col * 2 * init_r
                if x <= 1 - init_r:
                    centers[count] = [x, y]
                    count += 1
                    col += 1
                else:
                    break
            col = 0
            row += 1
            # Break if y exceeds boundary too much to prevent infinite loop
            if y > 1 - init_r and count < n: 
                 # Fallback to random if hex grid doesn't fit 26 easily
                 centers = np.random.rand(n, 2) * 0.8 + 0.1 
                 break

        # Initial radii estimation
        radii = np.full(n, init_r)

        # --- 2. Optimization ---
        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.concatenate([centers.flatten(), radii])
        
        def objective(vars_flat):
            c = vars_flat[:52].reshape(-1, 2)
            r = vars_flat[52:]
            
            # Penalty for negative radii
            p1 = np.sum(np.maximum(0, -r)) * penalty_weight
            
            # Penalty for boundary violations
            # x - r < 0 or x + r > 1
            p2 = np.sum(np.maximum(0, -(c[:, 0] - r))) * penalty_weight
            p3 = np.sum(np.maximum(0, (c[:, 0] + r) - 1)) * penalty_weight
            p4 = np.sum(np.maximum(0, -(c[:, 1] - r))) * penalty_weight
            p5 = np.sum(np.maximum(0, (c[:, 1] + r) - 1)) * penalty_weight
            
            # Penalty for overlaps
            overlap_pen = 0
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.linalg.norm(c[i] - c[j])
                    if dist < r[i] + r[j]:
                        overlap_pen += (r[i] + r[j] - dist) * penalty_weight
            
            # Objective: Maximize sum of radii (Minimize negative sum)
            return -np.sum(r) + p1 + p2 + p3 + p4 + p5 + overlap_pen

        # Run optimizer
        res = minimize(objective, x0, method='SLSQP', options={'maxiter': 1000, 'ftol': 1e-9})
        
        final_centers = res.x[:52].reshape(-1, 2)
        final_radii = res.x[52:]
        
        # Clamp radii to non-negative
        final_radii = np.maximum(final_radii, 1e-6)
        
        # Clamp centers to valid range (if optimization drifted)
        for i in range(n):
            final_centers[i, 0] = np.clip(final_centers[i, 0], final_radii[i], 1 - final_radii[i])
            final_centers[i, 1] = np.clip(final_centers[i, 1], final_radii[i], 1 - final_radii[i])
            
        current_sum = evaluate_state(final_centers, final_radii)
        
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_centers = final_centers.copy()
            best_radii = final_radii.copy()

    return best_centers, best_radii, best_sum_radii
