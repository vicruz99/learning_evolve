

<strategy>
I'll use a two-stage optimization approach:

1. **LP-based radius optimization**: For any fixed set of centers, I can compute the maximum possible radii by solving a linear program. This is exact and efficient.

2. **Center optimization via Nelder-Mead**: I optimize the center positions using Nelder-Mead simplex method, which works well for non-smooth objectives. The objective is the negative sum of radii from the LP.

3. **Multiple restarts**: I'll try several initial configurations (hexagonal grid, uniform grid, random) with perturbations to escape local minima.

4. **Key insight**: The LP formulation has 26 variables and ~351 constraints (325 pairwise + 26 boundary). Using HiGHS solver makes this very fast per evaluation, allowing thorough center search.

5. **Post-processing refinement**: After finding a good solution, I'll run additional local optimization passes to fine-tune.

import numpy as np
from scipy.optimize import minimize, linprog


def solve_radii_lp(centers, n):
    """Given fixed centers, solve for maximum radii using LP.
    
    Maximizes sum(r_i) subject to:
    - r_i + r_j <= distance(i, j) for all i < j
    - r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i
    - r_i >= 0
    """
    c_obj = -np.ones(n)
    
    num_pairwise = n * (n - 1) // 2
    num_boundary = n
    num_constraints = num_pairwise + num_boundary
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist
            idx += 1
    
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        A_ub[idx, i] = 1
        b_ub[idx] = max_r
        idx += 1
    
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.zeros(n)


def objective(x):
    """Objective function: negative sum of radii for given centers."""
    n = 26
    centers = x.reshape((n, 2))
    radii = solve_radii_lp(centers, n)
    return -np.sum(radii)


def make_hexagonal_init(n, scale=1.0):
    """Create hexagonal grid initialization."""
    centers = []
    for row in range(8):
        for col in range(8):
            if len(centers) >= n:
                break
            x = (col + 0.5) / 6.0
            y = (row + 0.5) * np.sqrt(3) / 6.0
            if row % 2 == 1:
                x += 1.0 / 12.0
            centers.append([x, y])
    centers = np.array(centers[:n])
    # Center and scale
    c = centers.mean(axis=0)
    centers = (centers - c) * scale + 0.5
    centers = np.clip(centers, 0.08, 0.92)
    return centers


def make_grid_init(n, rows=5, cols=6):
    """Create uniform grid initialization."""
    centers = []
    for row in range(rows):
        for col in range(cols):
            if len(centers) >= n:
                break
            x = (col + 0.5) / cols
            y = (row + 0.5) / rows
            centers.append([x, y])
    return np.array(centers[:n])


def make_corner_heavy_init(n):
    """Initialize with emphasis on corners and edges."""
    centers = []
    
    # Place circles near corners
    corner_margin = 0.15
    corners = [
        [corner_margin, corner_margin],
        [1 - corner_margin, corner_margin],
        [corner_margin, 1 - corner_margin],
        [1 - corner_margin, 1 - corner_margin]
    ]
    for c in corners:
        centers.append(c)
    
    # Place along edges
    edge_positions = 0.3, 0.5, 0.7
    for p in edge_positions:
        if len(centers) >= n:
            break
        centers.append([p, corner_margin])
        if len(centers) >= n:
            break
        centers.append([1 - p, corner_margin])
        if len(centers) >= n:
            break
        centers.append([corner_margin, p])
        if len(centers) >= n:
            break
        centers.append([1 - corner_margin, p])
        if len(centers) >= n:
            break
        centers.append([p, 1 - corner_margin])
        if len(centers) >= n:
            break
        centers.append([1 - p, 1 - corner_margin])
    
    # Fill interior with hexagonal pattern
    while len(centers) < n:
        for row in range(4):
            for col in range(4):
                if len(centers) >= n:
                    break
                x = 0.3 + col * 0.15
                y = 0.3 + row * 0.15 * np.sqrt(3) / 2
                if row % 2 == 1:
                    x += 0.075
                centers.append([x, y])
            if len(centers) >= n:
                break
    
    return np.array(centers[:n])


def optimize_from_init(x0, max_iter=80000):
    """Run Nelder-Mead optimization from a given initial point."""
    result = minimize(
        objective,
        x0,
        method='Nelder-Mead',
        options={
            'maxiter': max_iter,
            'xatol': 1e-9,
            'fatol': 1e-9,
            'adaptive': True
        }
    )
    return result


def refine_solution(centers, radii, n=26):
    """Post-processing refinement: swap circles with small radii to better positions."""
    centers = centers.copy()
    radii = radii.copy()
    
    improved = True
    while improved:
        improved = False
        # Find circle with smallest radius
        min_idx = np.argmin(radii)
        min_r = radii[min_idx]
        
        if min_r < 0.02:
            # Try to relocate this circle
            best_pos = centers[min_idx].copy()
            best_new_r = min_r
            
            # Try random positions
            for _ in range(200):
                test_pos = np.array([
                    np.random.uniform(0.1, 0.9),
                    np.random.uniform(0.1, 0.9)
                ])
                
                test_centers = centers.copy()
                test_centers[min_idx] = test_pos
                test_radii = solve_radii_lp(test_centers, n)
                
                if test_radii[min_idx] > best_new_r:
                    best_new_r = test_radii[min_idx]
                    best_pos = test_pos.copy()
            
            if best_new_r > min_r * 1.1:
                centers[min_idx] = best_pos
                radii = solve_radii_lp(centers, n)
                improved = True
    
    return centers, radii


def run_packing():
    n = 26
    
    best_value = float('inf')
    best_x = None
    
    # Generate diverse initial configurations
    configs = []
    
    # Hexagonal packings with different scales
    for scale in [0.8, 1.0, 1.2, 1.4]:
        configs.append(make_hexagonal_init(n, scale=scale))
    
    # Grid packings
    configs.append(make_grid_init(n, rows=5, cols=6))
    configs.append(make_grid_init(n, rows=6, cols=5))
    
    # Corner-heavy
    configs.append(make_corner_heavy_init(n))
    
    # Random initializations
    for seed in [42, 123, 456]:
        np.random.seed(seed)
        configs.append(np.random.rand(n, 2) * 0.7 + 0.15)
    
    # Optimize from each configuration with perturbations
    for config in configs:
        for scale in [0.0, 0.01, 0.03]:
            if scale > 0:
                np.random.seed(None)
                x0 = (config + np.random.randn(n, 2) * scale).flatten()
            else:
                x0 = config.flatten()
            
            x0 = np.clip(x0, 0.03, 0.97)
            
            result = optimize_from_init(x0, max_iter=60000)
            
            if result.fun < best_value:
                best_value = result.fun
                best_x = result.x.copy()
    
    # Extract and refine
    centers_opt = best_x.reshape((n, 2))
    radii_opt = solve_radii_lp(centers_opt, n)
    
    # Additional refinement passes
    for _ in range(3):
        centers_opt, radii_opt = refine_solution(centers_opt, radii_opt, n)
        # Re-optimize centers after refinement
        result = optimize_from_init(centers_opt.flatten(), max_iter=30000)
        centers_opt = result.x.reshape((n, 2))
        radii_opt = solve_radii_lp(centers_opt, n)
    
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
