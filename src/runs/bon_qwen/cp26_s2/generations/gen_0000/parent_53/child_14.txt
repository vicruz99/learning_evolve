import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n_circles = 26
    
    # Objective function: Maximize sum of radii
    # We minimize negative sum
    def objective(vars):
        radii = vars[2::3] # indices 2, 5, 8, ... correspond to radii
        return -np.sum(radii)

    # Constraints
    def create_constraints(n):
        constraints = []
        
        # 1. Boundary constraints:
        # x - r >= 0  => r - x <= 0
        # x + r <= 1  => x + r - 1 <= 0
        # y - r >= 0  => r - y <= 0
        # y + r <= 1  => y + r - 1 <= 0
        # 2. Non-overlap constraints:
        # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
        # (r_i + r_j)^2 - dist^2 <= 0
        
        # Note: scipy constraints are of the form fun(x) >= 0 for 'ineq' or fun(x) = 0 for 'eq'.
        # So we need to negate the <= 0 forms.
        
        # Boundary constraints (4 per circle)
        # r - x <= 0  =>  x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        
        # We can express these as fun(vars) >= 0
        for i in range(n):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]})
            # 1 - x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]})
            # y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - v[idx_r]})
            # 1 - y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]})

        # Non-overlap constraints
        # dist^2 - (r_i + r_j)^2 >= 0

            for j in range(i + 1, n):
                idx_xi = 3 * i
                idx_yi = 3 * i + 1
                idx_ri = 3 * i + 2
                
                idx_xj = 3 * j
                idx_yj = 3 * j + 1
                idx_rj = 3 * j + 2
                
                def dist_sq_minus_sum_r_sq(v, i=i, j=j):
                    dx = v[idx_xi] - v[idx_xj]
                    dy = v[idx_yi] - v[idx_yj]
                    dist_sq = dx*dx + dy*dy
                    r_sum = v[idx_ri] + v[idx_rj]
                    return dist_sq - r_sum*r_sum
                
                constraints.append({'type': 'ineq', 'fun': dist_sq_minus_sum_r_sq})
        
        return constraints

    constraints = create_constraints(n_circles)
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    # We can tighten bounds for r to [0, 0.5]
    bounds = [(0, 1)] * (2 * n_circles) + [(0, 0.5)] * n_circles

    best_centers = None
    best_radii = None
    
    # Multiple restarts to avoid local minima
    n_restarts = 20

        centers_init = np.random.rand(n_circles, 2)

            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = radii_init[i]
            
        # Optimize
        try:
            res = opt.minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_centers = np.zeros((n_circles, 2))
                    best_radii = np.zeros(n_circles)

                        best_centers[i, 0] = res.x[3*i]

        except Exception:
            continue

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True
