import numpy as np
from scipy.optimize import minimize, differential_evolution


N = 26


def make_hex_grid_initial(seed):
    """Generate a hexagonal grid initial configuration."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.07)
    
    # 6 rows: 5+4+5+4+5+3 = 26
    row_counts = [5, 4, 5, 4, 5, 3]
    
    row_h = 1.0 / 7.0
    idx = 0
    for row in range(6):
        y = 0.08 + row * row_h * np.sqrt(3.0) * 0.5
        offset = row_h * 0.5 if row % 2 == 1 else 0.0
        for col in range(row_counts[row]):
            x = offset + col * row_h
            if x < 0.95 and idx < N:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
    
    centers += 0.003 * np.random.randn(N, 2)
    centers = np.clip(centers, 0.08, 0.92)
    return centers, radii


def make_dense_grid_initial(seed):
    """Generate a dense grid initial configuration."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.05)
    
    # Try to pack in ~5x6 grid
    idx = 0
    for row in range(5):
        for col in range(6):
            if idx < N:
                x = 0.1 + col * 0.15
                y = 0.1 + row * 0.18
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
    
    centers += 0.002 * np.random.randn(N, 2)
    centers = np.clip(centers, 0.05, 0.95)
    return centers, radii


def make_random_initial(seed):
    """Generate a random initial configuration."""
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (N, 2))
    radii = np.full(N, 0.04)
    return centers, radii


def penalty_objective(x):
    """Penalty-based objective: maximize sum of radii with penalties for violations."""
    radius_sum = 0.0
    penalty = 0.0
    w = 1e9
    
    for i in range(N):
        radius_sum += x[3 * i + 2]
    
    # Boundary violations
    for i in range(N):
        xi = x[3 * i]
        yi = x[3 * i + 1]
        ri = x[3 * i + 2]
        
        if xi - ri < 0:
            penalty += w * (xi - ri) * (xi - ri)
        if xi + ri > 1.0:
            v = xi + ri - 1.0
            penalty += w * v * v
        if yi - ri < 0:
            penalty += w * (yi - ri) * (yi - ri)
        if yi + ri > 1.0:
            v = yi + ri - 1.0
            penalty += w * v * v
    
    # Separation violations
    for i in range(N):
        for j in range(i + 1, N):
            dx = x[3 * i] - x[3 * j]
            dy = x[3 * i + 1] - x[3 * j + 1]
            dist = np.sqrt(dx * dx + dy * dy)
            ri_rj = x[3 * i + 2] + x[3 * j + 2]
            if dist < ri_rj:
                v = ri_rj - dist
                penalty += w * v * v
    
    return -radius_sum + penalty


def extract_solution(x):
    """Extract centers and radii from solution vector."""
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    for i in range(N):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = max(0.0, x[3 * i + 2])
    return centers, radii


def fix_solution(centers, radii):
    """Fix any minor constraint violations."""
    for i in range(N):
        ri = radii[i]
        centers[i, 0] = max(ri, min(1.0 - ri, centers[i, 0]))
        centers[i, 1] = max(ri, min(1.0 - ri, centers[i, 1]))
    
    # Reduce radii if overlapping
    for iteration in range(100):
        max_violation = 0.0
        max_i = -1
        max_j = -1
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                ri_rj = radii[i] + radii[j]
                if dist < ri_rj:
                    v = ri_rj - dist
                    if v > max_violation:
                        max_violation = v
                        max_i = i
                        max_j = j
        
        if max_i < 0:
            break
        
        # Reduce both radii slightly
        reduction = max_violation * 0.5
        radii[max_i] -= reduction
        radii[max_j] -= reduction
        radii[max_i] = max(0.0, radii[max_i])
        radii[max_j] = max(0.0, radii[max_j])
    
    return centers, radii


def optimize_with_slscp(x0):
    """Optimize using SLSQP with explicit constraints."""
    n = N
    
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        ii = i
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, idx=ii: x[3 * idx] - x[3 * idx + 2] - 1e-9
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, idx=ii: 1.0 - x[3 * idx] - x[3 * idx + 2] - 1e-9
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, idx=ii: x[3 * idx + 1] - x[3 * idx + 2] - 1e-9
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, idx=ii: 1.0 - x[3 * idx + 1] - x[3 * idx + 2] - 1e-9
        })
    
    # Separation constraints
    for i in range(n):
        for j in range(i + 1, n):
            ii = i
            jj = j
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=ii, j=jj: np.sqrt((x[3*i] - x[3*j])**2 + (x[3*i+1] - x[3*j+1])**2) - x[3*i+2] - x[3*j+2] - 1e-9
            })
    
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((1e-6, 0.5))
    
    def obj(x):
        s = 0.0
        for i in range(n):
            s += x[3 * i + 2]
        return -s
    
    try:
        result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                         constraints=constraints, options={'maxiter': 10000, 'ftol': 1e-15})
        return result.x, result.success
    except Exception:
        return x0, False


def expand_radii(centers, radii):
    """Try to expand radii by finding the maximum feasible radius for each circle."""
    centers = centers.copy()
    radii = radii.copy()
    
    improved = True
    while improved:
        improved = False
        for i in range(N):
            # Find the maximum radius for circle i given all others
            max_r = 0.5  # Theoretical max
            max_r = min(max_r, centers[i, 0])
            max_r = min(max_r, 1.0 - centers[i, 0])
            max_r = min(max_r, centers[i, 1])
            max_r = min(max_r, 1.0 - centers[i, 1])
            
            for j in range(N):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                max_r = min(max_r, dist - radii[j])
            
            max_r = max(0.0, max_r)
            if max_r > radii[i] + 1e-10:
                radii[i] = max_r
                improved = True
    
    return centers, radii


def perturbation_search(centers, radii, num_trials=50):
    """Try random perturbations to find better configurations."""
    centers = centers.copy()
    radii = radii.copy()
    best_sum = np.sum(radii)
    
    for trial in range(num_trials):
        # Randomly perturb positions
        trial_centers = centers.copy()
        trial_radii = radii.copy()
        
        # Perturb a few circles
        num_to_perturb = np.random.randint(1, 6)
        indices = np.random.choice(N, num_to_perturb, replace=False)
        
        for idx in indices:
            dx = np.random.uniform(-0.02, 0.02)
            dy = np.random.uniform(-0.02, 0.02)
            trial_centers[idx, 0] += dx
            trial_centers[idx, 1] += dy
            trial_centers[idx, 0] = max(trial_radii[idx], min(1.0 - trial_radii[idx], trial_centers[idx, 0]))
            trial_centers[idx, 1] = max(trial_radii[idx], min(1.0 - trial_radii[idx], trial_centers[idx, 1]))
        
        # Reduce radii slightly and re-expand
        trial_radii *= 0.9
        trial_centers, trial_radii = expand_radii(trial_centers, trial_radii)
        
        trial_sum = np.sum(trial_radii)
        if trial_sum > best_sum:
            centers = trial_centers
            radii = trial_radii
            best_sum = trial_sum
    
    return centers, radii
