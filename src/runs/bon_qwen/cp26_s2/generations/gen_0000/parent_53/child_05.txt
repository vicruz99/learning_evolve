import numpy as np
from scipy.optimize import minimize

N = 26

def hex_init():
    centers = []
    row = 0
    while len(centers) < N:
        col = 0
        while len(centers) < N:
            x = col * 0.20 + (0.10 if row % 2 == 1 else 0.0)
            y = row * 0.20 * np.sqrt(3)/2 + 0.10
            if x < 1.0 and y < 1.0:
                centers.append((x, y))
            col += 1
            if col > 7:
                break
        row += 1
    
    vars = np.zeros(3 * N)
    for i in range(N):
        vars[3*i] = np.clip(centers[i][0], 0.10, 0.90)
        vars[3*i+1] = np.clip(centers[i][1], 0.10, 0.90)
        vars[3*i+2] = 0.095
    return vars

def random_init():
    vars = np.zeros(3 * N)
    for i in range(N):
        vars[3*i] = np.random.uniform(0.15, 0.85)
        vars[3*i+1] = np.random.uniform(0.15, 0.85)
        vars[3*i+2] = 0.085 + np.random.uniform(-0.01, 0.01)
    return vars

def get_bounds():
    b = []
    for i in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.50)])
    return b

def eval_fn(vars, lam):
    n = N
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties (vectorized)
    pen += np.sum(np.maximum(0.0, r - x)**2)
    pen += np.sum(np.maximum(0.0, x + r - 1.0)**2)
    pen += np.sum(np.maximum(0.0, r - y)**2)
    pen += np.sum(np.maximum(0.0, y + r - 1.0)**2)
    
    # Overlap penalties
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i+1, n):
            dist = np.sqrt((xi - x[j])**2 + (yi - y[j])**2 + 1e-12)
            ov = ri + r[j] - dist
            if ov > 0:
                pen += ov**2
    
    return obj + lam * pen

def project_to_valid(centers, radii):
    n = N
    for _ in range(100):
        changed = False
        # Boundary constraints
        for i in range(n):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            new_r = min(r, x, 1.0 - x, y, 1.0 - y)
            if abs(new_r - r) > 1e-10:
                radii[i] = max(0.0, new_r)
                changed = True
        
        # Overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt((centers[i,0] - centers[j,0])**2 + (centers[i,1] - centers[j,1])**2)
                if radii[i] + radii[j] > dist + 1e-10:
                    factor = dist / (radii[i] + radii[j])
                    radii[i] *= factor
                    radii[j] *= factor
                    changed = True
        if not changed:
            break
    return centers, radii

def run_packing():
    best_sum = -1.0
    best_c = None
    best_r = None
    
    for trial in range(25):
        if trial < 10:
            x0 = hex_init()
        else:
            x0 = random_init()
        
        lam = 80.0
        curr = x0.copy()
        for step in range(7):
            res = minimize(eval_fn, curr, args=(lam,), method='L-BFGS-B',
                          bounds=get_bounds(), options={'maxiter': 400, 'ftol': 1e-15})
            curr = res.x
            lam *= 2.0
        
        centers = np.column_stack((curr[0::3], curr[1::3]))
        radii = curr[2::3].copy()
        
        centers, radii = project_to_valid(centers, radii)
        
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_c = centers.copy()
            best_r = radii.copy()
            
    return best_c, best_r, best_sum
