import numpy as np
from scipy.optimize import minimize

def obj_func(v, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def cons_x_min(v, i):
    """Constraint: x >= r"""
    return v[3*i] - v[3*i+2]

def cons_x_max(v, i):
    """Constraint: x <= 1 - r"""
    return 1.0 - v[3*i] - v[3*i+2]

def cons_y_min(v, i):
    """Constraint: y >= r"""
    return v[3*i+1] - v[3*i+2]

def cons_y_max(v, i):
    """Constraint: y <= 1 - r"""
    return 1.0 - v[3*i+1] - v[3*i+2]

def cons_overlap(v, i, j):
    """Constraint: dist^2 >= (r_i + r_j)^2"""
    dx = v[3*i] - v[3*j]
    dy = v[3*i+1] - v[3*j+1]
    dr = v[3*i+2] + v[3*j+2]
    return dx*dx + dy*dy - dr*dr

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: x in [0,1], y in [0,1], r in [0, 0.5]
    bnds = []
    for _ in range(n):
        bnds.append((0.0, 1.0))
        bnds.append((0.0, 1.0))
        bnds.append((0.0, 0.5))

    for seed in range(5):
        np.random.seed(seed)
        
        # Initialize centers and radii
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        
        # Grid initialization (5x5) + 1 extra
        idx = 0
        for r in range(5):

                cons.append({'type': 'ineq', 'fun': cons_overlap, 'args': (i, j)})

    return best_centers, best_radii, np.sum(best_radii)
