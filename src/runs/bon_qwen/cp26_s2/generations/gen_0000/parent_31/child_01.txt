import numpy as np
from scipy.optimize import minimize
import warnings

warnings.filterwarnings('ignore')

N = 26

def objective(x):
    # Maximize sum of radii -> minimize negative sum
    return -np.sum(x[4*N:])

def boundary_constraints(x):
    # Ensure circles stay within [0,1]x[0,1]
    c = x[:4*N].reshape((N, 2))
    r = x[4*N:]
    return np.concatenate([
        c[:, 0] - r,          # x >= r
        1 - c[:, 0] - r,      # x + r <= 1
        c[:, 1] - r,          # y >= r
        1 - c[:, 1] - r       # y + r <= 1
    ])

def overlap_constraints(x):
    # Ensure non-overlap: dist_sq >= (r_i + r_j)^2
    c = x[:4*N].reshape((N, 2))
    r = x[4*N:]
    ii, jj = np.triu_indices(N, k=1)
    diff = c[ii] - c[jj]
    return np.sum(diff**2, axis=1) - (r[ii] + r[jj])**2

def run_packing():
    np.random.seed(42)
    
    # Initialize with a hexagonal grid pattern for high packing density
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.08)
    idx = 0
    dx = 0.2
    dy = dx * np.sqrt(3) / 2
    
    for r_idx in range(6):
        for c_idx in range(5):
            if idx >= N: 
                break
            x = c_idx * dx + 0.1 + (r_idx % 2) * (dx / 2)
            y = r_idx * dy + 0.1
            # Clip to ensure initial feasibility
            centers[idx] = [np.clip(x, radii[idx], 1-radii[idx]), 
                            np.clip(y, radii[idx], 1-radii[idx])]
            idx += 1
        if idx >= N: 
            break
            
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1) for _ in range(4*N)] + [(0, 0.25) for _ in range(N)]
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]
    
    best_res = None
    best_val = -np.inf
    
    # Multiple restarts with perturbations to escape local optima
    for trial in range(5):
        x_start = x0.copy()
        if trial > 0:
            x_start[:4*N] += np.random.uniform(-0.01, 0.01, size=4*N)
            x_start[:4*N] = np.clip(x_start[:4*N], 0.02, 0.98)
            
        try:
            res = minimize(objective, x_start, method='SLSQP', bounds=bounds, 
                           constraints=constraints, 
                           options={'maxiter': 5000, 'ftol': 1e-11})
            current_val = -res.fun
            if current_val > best_val:
                best_val = current_val
                best_res = res
        except Exception:
            pass
            
    # Fallback if all trials failed
    if best_res is None:
        best_res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                            constraints=constraints, options={'maxiter': 5000})
                            
    final_centers = best_res.x[:4*N].reshape((N, 2))
    final_radii = best_res.x[4*N:]
    
    # Final safety clamping to guarantee validity within tolerance
    final_radii = np.maximum(final_radii, 1e-6)
    final_centers = np.clip(final_centers, final_radii[:, None], 1 - final_radii[:, None])
    
    return final_centers, final_radii, float(np.sum(final_radii))
