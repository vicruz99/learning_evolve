def compute_clearance(centers):

    n = centers.shape[0]
    # Boundary clearance: distance to the closest wall
    b = np.min(np.minimum(centers, 1 - centers))

    diff = centers[:, None, :] - centers[None, :, :]

    return min(b, p)
