import numpy as np
from scipy.optimize import minimize

def objective_func(v, n):
    """Negative sum of radii to be minimized."""
    return -np.sum(v[2*n:])

def constraint_func(v, n):
    """Returns array of inequality constraint values (must be >= 0)."""
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    
    cons = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.append(centers[:, 0] - radii)
    cons.append(1 - centers[:, 0] - radii)
    cons.append(centers[:, 1] - radii)
    cons.append(1 - centers[:, 1] - radii)
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    dist_sq = dx**2 + dy**2
    r_sum = radii[:, None] + radii[None, :]
    
    # Extract upper triangular part for unique pairs
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons.append((dist_sq - r_sum**2)[mask])
    
    return np.concatenate(cons)

def create_bounds(n):
    """Bounds for centers [0,1] and radii [0, 0.5]."""
    return [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n

def run_packing():
    n = 26
    bounds = create_bounds(n)
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    best_v = None
    best_sum = -np.inf
    
    starts = []
    
    # 1. Grid initialization
    c1 = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                c1[idx] = [0.12 + i*0.176, 0.12 + j*0.176]
                idx += 1
    if idx < n:
        c1[idx] = [0.5, 0.05]
    starts.append(np.concatenate([c1.flatten(), np.full(n, 0.08)]))
    
    # 2. Hexagonal initialization
    c2 = np.zeros((n, 2))
    idx = 0
    y = 0.15
    row = 0
    while idx < n and y < 0.9:
        x = 0.15
        if row % 2 == 1:
            x += 0.1
        while x < 0.9 and idx < n:
            c2[idx] = [x, y]
            idx += 1
            x += 0.2
        y += 0.13
        row += 1
    starts.append(np.concatenate([c2.flatten(), np.full(n, 0.08)]))
    
    # 3. Random initialization
    np.random.seed(42)
    c3 = np.random.rand(n, 2) * 0.6 + 0.2
    starts.append(np.concatenate([c3.flatten(), np.full(n, 0.07)]))
    
    # Run optimization from each start
    for v0 in starts:
        try:
            res = minimize(objective_func, v0, method='SLSQP', bounds=bounds, 
                           constraints=cons, args=(n,), options={'maxiter': 2000, 'ftol': 1e-10})
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        return np.zeros((n, 2)), np.full(n, 0.01), 0.0
        
    final_centers = best_v[:2*n].reshape((n, 2))
    final_radii = best_v[2*n:]
    
    # Post-processing to strictly satisfy validator tolerances
    final_radii = np.maximum(final_radii, 0.0)
    final_centers[:, 0] = np.clip(final_centers[:, 0], final_radii, 1.0 - final_radii)
    final_centers[:, 1] = np.clip(final_centers[:, 1], final_radii, 1.0 - final_radii)
    
    return final_centers, final_radii, np.sum(final_radii)
