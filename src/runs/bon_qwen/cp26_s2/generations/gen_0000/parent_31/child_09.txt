import numpy as np


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = 0.0
    best_centers = None
    best_radii = None

        if valid:

                best_sum = s
                best_centers = centers.copy()

    return best_centers, best_radii, best_sum


def init_packing(n: int) -> tuple[np.ndarray, np.ndarray]:
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.02
    
    idx = 0
    row = 0
    while idx < n:
        col = 0
        while idx < n:
            x = 0.05 + col * 0.28 + (row % 2) * 0.14
            y = 0.05 + row * 0.24
            
            if x < 0.95 and y < 0.95:
                centers[idx] = [x, y]
                idx += 1
            
            col += 1
            if col > 7:
                break
        row += 1
    
    return centers, radii


def validate_quick(centers: np.ndarray, radii: np.ndarray) -> bool:
    n = centers.shape[0]
    
    for i in range(n):
        if radii[i] < 0 or np.isnan(radii[i]):
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True


def optimize_packing(centers: np.ndarray, radii: np.ndarray, n: int, steps: int = 250000) -> tuple[np.ndarray, np.ndarray]:
    for step in range(steps):
        forces = np.zeros((n, 2))
        
        # Vectorized pairwise repulsion
        for i in range(n):
            dx = centers[:, 0] - centers[i, 0]
            dy = centers[:, 1] - centers[i, 1]
            dist_sq = dx * dx + dy * dy
            dist_sq = np.maximum(dist_sq, 1e-24)
            dist = np.sqrt(dist_sq)
            
            min_dist = radii + radii[i]
            overlap = min_dist - dist
            overlap = np.maximum(overlap, 0)
            
            # Mask out self-interaction
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            
            inv_dist = np.where(dist > 1e-12, 1.0 / dist, 0)
            force_mag = overlap * inv_dist * 50000
            
            fx = force_mag * dx
            fy = force_mag * dy
            
            forces[mask, 0] += fx[mask] * 0.5
            forces[mask, 1] += fy[mask] * 0.5
        
        # Boundary forces
        for i in range(n):
            for d in [0, 1]:
                coord = centers[i, d]
                r = radii[i]
                if coord - r < 0:
                    forces[i, d] += 200000 * abs(coord - r)
                if coord + r > 1:
                    forces[i, d] -= 200000 * abs(coord + r - 1)
        
        # Adaptive learning rate
        lr = 0.005 / (1 + step * 0.000015)
        
        # Update positions
        centers += lr * forces
        centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
        centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)
        
        # Gradually increase radii (pressure)
        radii *= 1 + 0.000015
        
        # Occasional perturbation to escape local optima
        if step % 50000 == 0 and step > 0:
            centers += np.random.randn(n, 2) * 0.003
            centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
            centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)
    
    # Final cleanup: ensure no overlaps
    centers, radii = final_cleanup(centers, radii, n)
    
    return centers, radii


def final_cleanup(centers: np.ndarray, radii: np.ndarray, n: int) -> tuple[np.ndarray, np.ndarray]:
    # Iteratively resolve any remaining small overlaps
    for _ in range(500):
        max_overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j]:
                    overlap = radii[i] + radii[j] - dist
                    if overlap > max_overlap:
                        max_overlap = overlap
        
        if max_overlap < 1e-10:
            break
    
    return centers, radii
