import numpy as np
import scipy.optimize
import math
import time

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    (Read-only function provided in prompt)
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  
                return False

    return True

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to find optimal radii that maximize sum(r).
    Constraints:
    1. r_i >= 0
    2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i (Boundary)
    3. r_i + r_j <= dist(i, j) (Non-overlap)
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c = -np.ones(n)
    
    # Constraints matrix A_ub @ r <= b_ub
    
    # 1. Boundary constraints
    # r_i <= x_i  => 1*r_i <= x_i
    # r_i <= 1-x_i => 1*r_i <= 1-x_i
    # etc.
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x
        A_ub.append(row)
        b_ub.append(1 - x)
        
        # r_i <= y
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        A_ub.append(row)
        b_ub.append(1 - y)

    # 2. Pairwise constraints
    # r_i + r_j <= dist
    # This creates a sparse structure, but for N=26, dense is fine.
    # There are N*(N-1)/2 pairs.
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Bounds for r: [0, 1] (1 is loose upper bound)
    bounds = [(0, 1) for _ in range(n)]

    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback if LP fails (should not happen with valid inputs)
            # Return small radii
            return np.full(n, 1e-5), 1e-5 * n
    except Exception:
        return np.full(n, 1e-5), 1e-5 * n

def generate_hex_grid_init(n_circles):
    """
    Generate a hexagonal grid initialization for n_circles.
    """
    centers = []
    
    # Try to fit in rows
    # Approximate radius 0.1
    # Vertical spacing sqrt(3)/2 * 2r = r*sqrt(3) ~ 0.1732
    # Horizontal spacing 2r = 0.2
    
    # Heuristic to determine rows
    # Area approx: n * pi * r^2 ~ 1 => r ~ sqrt(1/(26*pi)) ~ 0.11
    # Let's try r=0.11
    r_est = 0.11
    h = r_est * math.sqrt(3)
    w = 2 * r_est
    
    y_curr = r_est
    row_idx = 0
    
    while len(centers) < n_circles:
        # Determine shift for this row (hex packing)
        shift = r_est if row_idx % 2 == 1 else 0
        
        # Determine how many circles fit in this row
        # x starts at r_est + shift? 
        # Actually, for odd rows, we can start at r_est + shift if it fits.
        # But simpler: x = r_est + shift + k*2r_est
        
        # Check bounds
        x_curr = r_est + shift
        count_in_row = 0
        while x_curr + r_est <= 1.0 + 1e-9 and len(centers) < n_circles:
            centers.append([x_curr, y_curr])
            count_in_row += 1
            x_curr += w
        
        y_curr += h
        row_idx += 1
        
    return np.array(centers[:n_circles])

def run_packing():
    n_circles = 26
    best_score = -1.0

    best_radii = None
    
    # Time limit for optimization

    starts.append(generate_hex_grid_init(n_circles))

        radii, score = solve_radii_lp(current_centers)

        # Use a cooling factor for step size

    # Ensure strict validity for output (handle numerical noise)
