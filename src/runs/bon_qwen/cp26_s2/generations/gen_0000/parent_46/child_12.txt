import numpy as np
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')

def compute_overlap_penalty(centers, radii):
    """Compute a smooth penalty for circle overlaps and boundary violations."""
    n = centers.shape[0]
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap**2
        
        r = radii[i]
        cx, cy = centers[i]
        if cx - r < 0:
            penalty += (cx - r)**2
        elif cx + r > 1:
            penalty += (cx + r - 1)**2
        if cy - r < 0:
            penalty += (cy - r)**2
        elif cy + r > 1:
            penalty += (cy + r - 1)**2
    return penalty

def position_objective(x, radii):
    """Objective function for scipy.optimize.minimize."""
    n = len(radii)
    centers = x.reshape(n, 2)
    return compute_overlap_penalty(centers, radii)
