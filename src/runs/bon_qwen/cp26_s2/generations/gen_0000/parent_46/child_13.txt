import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Helper to create initial centers using force-directed layout
    def generate_initial_centers(seed=42):
        rng = np.random.RandomState(seed)
        centers = rng.rand(n, 2) * 0.8 + 0.1 # Start in [0.1, 0.9]
        
        # Repulsion steps to spread points
        for _ in range(100):
            forces = np.zeros_like(centers)
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist_sq = np.sum(diff**2)
                    dist = math.sqrt(dist_sq)
                    if dist < 1e-4:
                        dist = 1e-4
                        diff = rng.rand(2) * 0.01
                    # Repulsive force
                    f = diff / (dist**2)
                    forces[i] += f
                    forces[j] -= f
                
                # Wall repulsion (keep away from 0 and 1)
                for k in range(2):
                    if centers[i, k] < 0.1:
                        forces[i, k] += 5.0 * (0.1 - centers[i, k])
                    elif centers[i, k] > 0.9:
                        forces[i, k] -= 5.0 * (centers[i, k] - 0.9)
            
            centers += forces * 0.005
            centers = np.clip(centers, 0.01, 0.99)
        return centers

    # Generate a few different initial configurations
    candidates = []
    for seed in [42, 123, 456]:
        candidates.append(generate_initial_centers(seed))
    
    # Also add a grid-based initialization
    cols = 6
    rows = 5
    grid_centers = []
    count = 0
    for r in range(rows):
        for c in range(cols):
            if count >= n:
                break
            x = 0.1 + c * (0.8 / (cols - 1))
            y = 0.1 + r * (0.8 / (rows - 1))
            # Offset odd rows for hexagonal feel
            if r % 2 == 1:
                x += (0.8 / (cols - 1)) / 2
            grid_centers.append([x, y])
            count += 1
    grid_centers = np.array(grid_centers[:n])
    candidates.append(grid_centers)

    best_sum_radii = -1.0
    best_result = None

    # Define constraints for SLSQP
    # We need to construct the constraint list once or inside the loop.
    # Given the size, constructing it inside is fine.
    
    for init_centers in candidates:
        # Initial radii estimate
        # A safe small radius
        init_radii = np.full(n, 0.05)
        
        # Flatten variables: [x1, ..., xn, y1, ..., yn, r1, ..., rn]
        x0 = np.hstack([init_centers[:, 0], init_centers[:, 1], init_radii])
        
        # Bounds
        # x, y in [0, 1]
        # r in [0, 0.5] (though effectively smaller)
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        
        # Constraints
        constraints = []
        
        # 1. Boundary constraints: x_i - r_i >= 0, 1 - x_i - r_i >= 0, etc.
        # These are linear.
        for i in range(n):
            idx_x = i
            idx_y = n + i
            idx_r = 2 * n + i
            
            # x_i - r_i >= 0
            # fun: x_i - r_i
            # jac: [0... 1 at idx_x ... -1 at idx_r ...]
            constraints.append({
                'type': 'ineq',
                'fun': lambda z, i=i: z[i] - z[2*n+i],
                'jac': lambda z, i=i: np.array([1.0 if k==i else -1.0 if k==2*n+i else 0.0 for k in range(len(z))])
            })
            # 1 - x_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda z, i=i: 1.0 - z[i] - z[2*n+i],
                'jac': lambda z, i=i: np.array([-1.0 if k==i else -1.0 if k==2*n+i else 0.0 for k in range(len(z))])
            })
            # y_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda z, i=i: z[n+i] - z[2*n+i],
                'jac': lambda z, i=i: np.array([1.0 if k==n+i else -1.0 if k==2*n+i else 0.0 for k in range(len(z))])
            })
            # 1 - y_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda z, i=i: 1.0 - z[n+i] - z[2*n+i],
                'jac': lambda z, i=i: np.array([-1.0 if k==n+i else -1.0 if k==2*n+i else 0.0 for k in range(len(z))])
            })

        # 2. Non-overlap constraints: dist(i,j) - (r_i + r_j) >= 0
        # Non-linear.
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi = i
                idx_yi = n + i
                idx_ri = 2 * n + i
                idx_xj = j
                idx_yj = n + j
                idx_rj = 2 * n + j
                
                def con_overlap(z, _i=i, _j=j, _xi=idx_xi, _yi=idx_yi, _ri=idx_ri, _xj=idx_xj, _yj=idx_yj, _rj=idx_rj):
                    dx = z[_xi] - z[_xj]
                    dy = z[_yi] - z[_yj]
                    dist = math.sqrt(dx*dx + dy*dy)
                    return dist - (z[_ri] + z[_rj])
                
                def jac_overlap(z, _i=i, _j=j, _xi=idx_xi, _yi=idx_yi, _ri=idx_ri, _xj=idx_xj, _yj=idx_yj, _rj=idx_rj):
                    dx = z[_xi] - z[_xj]
                    dy = z[_yi] - z[_yj]
                    dist = math.sqrt(dx*dx + dy*dy)
                    if dist < 1e-9:
                        # Parallel centers, return 0 gradient (should not happen with positive radii)
                        return np.zeros(len(z))
                    
                    jac = np.zeros(len(z))
                    # d(dist)/dx_i = dx/dist
                    jac[_xi] = dx / dist
                    jac[_xj] = -dx / dist
                    jac[_yi] = dy / dist
                    jac[_yj] = -dy / dist
                    jac[_ri] = -1.0
                    jac[_rj] = -1.0
                    return jac

                constraints.append({
                    'type': 'ineq',
                    'fun': con_overlap,
                    'jac': jac_overlap
                })

        # Objective: Maximize sum(r) -> Minimize -sum(r)
        def objective(z):
            return -np.sum(z[2*n:])

        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints, 
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            
            if res.success or res.nit > 0:
                current_sum = -res.fun
                # Check validity roughly (some constraints might be slightly violated due to tolerance)
                # But we trust SLSQP for now.
                if current_sum > best_sum_radii:
                    # Extract solution
                    centers_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
                    radii_opt = res.x[2*n:]
                    
                    # Clip negative radii if any
                    radii_opt = np.maximum(radii_opt, 0.0)
                    
                    best_sum_radii = current_sum
                    best_result = (centers_opt, radii_opt)
                    
        except Exception:
            continue

    if best_result is not None:
        centers, radii = best_result
        return centers, radii, np.sum(radii)
    else:
        # Fallback to a valid simple packing if optimization fails
        # Grid packing with small radius
        centers_fallback = np.zeros((n, 2))
        k = 0
        for r in range(6):
            for c in range(5):
                if k < n:
                    centers_fallback[k, 0] = 0.2 + c * 0.15
                    centers_fallback[k, 1] = 0.2 + r * 0.15
                    k += 1
        radii_fallback = np.full(n, 0.05)
        return centers_fallback, radii_fallback, np.sum(radii_fallback)
