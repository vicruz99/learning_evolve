import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # Function to validate and return the result
    def solve_packing(x0, seed=None):
        if seed is not None:
            np.random.seed(seed)
            
        # Objective function: Minimize negative sum of radii
        def objective(vars):
            radii = vars[2::3]
            return -np.sum(radii)

        # Constraints
        # 1. Boundary constraints: 0 <= x, y <= 1 (handled by bounds)
        #    And r <= x <= 1-r, r <= y <= 1-r
        # 2. Overlap constraints: dist(i, j) >= r_i + r_j
        
        def constraint_boundary(vars):
            # Returns a vector of constraint values >= 0
            c = []
            for i in range(n):
                x = vars[3*i]
                y = vars[3*i+1]
                r = vars[3*i+2]
                # x >= r  => x - r >= 0
                c.append(x - r)
                # 1 - x >= r => 1 - x - r >= 0
                c.append(1.0 - x - r)
                # y >= r
                c.append(y - r)
                # 1 - y >= r
                c.append(1.0 - y - r)
                # r >= 0
                c.append(r)
            return np.array(c)

        def constraint_overlap(vars):
            # Returns a vector of constraint values >= 0
            c = []
            # Extract coordinates and radii for vectorized computation if possible, 
            # but loop is fine for n=26
            centers = vars[:2*n].reshape(n, 2)
            radii = vars[2::3]
            
            for i in range(n):
                xi, yi = centers[i]
                ri = radii[i]
                for j in range(i + 1, n):
                    xj, yj = centers[j]
                    rj = radii[j]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r = ri + rj
                    # dist >= sum_r  => dist^2 >= sum_r^2
                    c.append(dist_sq - sum_r**2)
            return np.array(c)

        # Bounds
        # x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for i in range(n):
            bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

        cons = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_overlap}
        ]

        # Optimization options
        options = {
            'maxiter': 1000,
            'ftol': 1e-12,
            'disp': False
        }

        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        
        if res.success:
            return res.x, -res.fun
        else:
            # Return best found even if not fully converged
            return res.x, -res.fun

    # --- Initialization Phase ---
    # Try to generate a valid, spread-out configuration to start from
    def get_initial_guess():
        # Random positions
        centers = np.random.rand(n, 2)
        # Start with small radii
        radii = np.ones(n) * 0.02
        
        # Force-directed layout to separate circles and increase radii
        # This helps avoid local minima where circles are stacked
        learning_rate = 0.01
        min_dist = 0.05

            forces = np.zeros((n, 2))
            current_radii = radii.copy()

            for i in range(n):
                # Max possible radius based on boundaries
                r_boundary = min(centers[i, 0], 1 - centers[i, 0], 
                                 centers[i, 1], 1 - centers[i, 1])
                
                # Max possible radius based on neighbors
                r_neighbors = np.inf
                for j in range(n):
                    if i == j: continue
                    dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                                   (centers[i, 1] - centers[j, 1])**2)
                    # Required separation
                    req_sep = dist - current_radii[j]
                    if req_sep < r_neighbors:
                        r_neighbors = req_sep
                
                # New radius is min of constraints, slightly relaxed to allow movement
                max_r = min(r_boundary, r_neighbors)
                if max_r < 0: max_r = 0.01 # Keep positive
                
                # Smoothly update radius
                radii[i] = 0.9 * radii[i] + 0.1 * max_r

            # Apply forces to move centers apart
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist_sq = dx**2 + dy**2
                    dist = np.sqrt(dist_sq)
                    
                    target_dist = radii[i] + radii[j]
                    if dist < target_dist and dist > 1e-6:
                        # Repulsion force proportional to overlap
                        overlap = target_dist - dist
                        force_mag = overlap * 0.5
                        fx = (dx / dist) * force_mag
                        fy = (dy / dist) * force_mag
                        
                        forces[i] += [fx, fy]
                        forces[j] -= [fx, fy]

            centers += forces * learning_rate

        vars = np.zeros(3 * n)

            vars[3*i] = centers[i, 0]
            vars[3*i+1] = centers[i, 1]
            vars[3*i+2] = radii[i]
        return vars

    best_vars = None

    # Run optimization with multiple restarts
    for seed in range(10):
        x0 = get_initial_guess()

        x0 += np.random.randn(3*n) * 0.005
        
        final_vars, current_sum = solve_packing(x0, seed=seed)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_vars = final_vars

    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = (best_vars[3*i], best_vars[3*i+1])
        radii[i] = best_vars[3*i+2]

    # Final validation check (internal)
    # Ensure no negative radii due to numerical errors
    radii = np.maximum(radii, 1e-9)
    
    return centers, radii, np.sum(radii)
