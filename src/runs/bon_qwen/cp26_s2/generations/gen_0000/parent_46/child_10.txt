from scipy.optimize import minimize

def initialize_hexagonal(n):

def compute_max_radii(centers):
    """Compute maximum valid radii for given center positions."""
    n = centers.shape[0]
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        
        for j in range(n):
            if i != j:
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                max_r = min(max_r, dist / 2)
        
        radii[i] = max(0, max_r)
    
    return radii


def force_optimize_with_growth(centers, radii, n_iter=30000, growth_factor=1.00005):

    n = centers.shape[0]
    centers = centers.copy().astype(np.float64)

                    overlap = min_dist - dist

                    fx = force_mag * dx / dist
                    fy = force_mag * dy / dist
                    
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy

                forces[i, 1] += 500.0 * (r - y)

def constraints_vectorized(params, n=26):
    """Vectorized constraint evaluation for SLSQP."""
    c = params[:2 * n].reshape(n, 2)
    r = params[2 * n:]
    
    # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    bc = np.concatenate([
        c[:, 0] - r,
        1 - c[:, 0] - r,
        c[:, 1] - r,
        1 - c[:, 1] - r,
    ])
    
    # Pairwise constraints: dist(i,j) >= r_i + r_j
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    pc = dists[i_idx, j_idx] - r[i_idx] - r[j_idx]
    
    return np.concatenate([bc, pc])


def objective_func(params, n=26):
    """Objective: negative sum of radii."""
    return -np.sum(params[2 * n:])


def optimize_slsqp(centers, radii, maxiter=2000):
    """SLSQP optimization with explicit constraints."""
    n = 26
    
    params_init = np.concatenate([centers.flatten(), radii])
    
    # Bounds: positions in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': lambda p: constraints_vectorized(p, n)}
    
    result = minimize(
        objective_func,
        params_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        args=(n,),
        options={
            'maxiter': maxiter,
            'ftol': 1e-12,
            'disp': False
        }
    )
    
    if result.success or result.status == 0:
        return result.x[:2 * n].reshape(n, 2), result.x[2 * n:]
    return centers, radii
