import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    num_vars = n * 3  # x, y, r for each circle
    
    # Constants
    PENALTY_FACTOR = 1000.0
    EPS = 1e-9

    def compute_cost(params):
        """
        Computes the cost function: -sum(radii) + penalty for violations.
        """
        centers = params[:n*2].reshape((n, 2))
        radii = params[n*2:]
        
        cost = -np.sum(radii)
        penalty = 0.0
        
        # 1. Boundary violations
        # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        # Violation is max(0, - (x-r)), etc.
        
        # Left/Bottom boundaries: r > x or r > y
        # Violation: max(0, r - x)
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            
            # Check boundaries
            if x < r: penalty += (r - x)**2
            if x + r > 1: penalty += (x + r - 1)**2
            if y < r: penalty += (r - y)**2
            if y + r > 1: penalty += (y + r - 1)**2
            
        # 2. Overlap violations
        # dist >= r_i + r_j
        # Violation: max(0, r_i + r_j - dist)^2
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i,0] - centers[j,0])**2 + (centers[i,1] - centers[j,1])**2)
                sum_r = radii[i] + radii[j]
                if dist < sum_r:
                    penalty += (sum_r - dist)**2
                    
        return cost + PENALTY_FACTOR * penalty

    # Bounds for optimization: [x, y, r]
    # x, y in [0, 1], r in [0, 0.5] (since max radius is 0.5)
    # Actually r can be at most 0.5, but effectively smaller for packing.
    # To allow optimization to find valid spots, let's allow x,y to go slightly out 
    # or just [0,1]. Since penalty handles it, [0,1] is fine.
    # However, for DE, bounds must be inclusive.
    bounds = [(0.0, 1.0) for _ in range(n*2)] + [(0.0, 0.5) for _ in range(n)]
    
    # Run Differential Evolution
    # popsize=30 gives 30*78 candidates, reasonable trade-off.
    # maxiter=1000 might be slow, but let's try a balanced setting.
    # tol=1e-8 for precision.
    print("Running Differential Evolution optimization...")
    de_result = differential_evolution(
        compute_cost,
        bounds,
        seed=42,
        maxiter=500, 
        popsize=25,
        tol=1e-8,
        mutation=(0.5, 1.5),
        recombination=0.7,
        polish=True
    )
    
    best_params = de_result.x
    
    # Extract centers and radii
    centers = best_params[:n*2].reshape((n, 2))
    radii = best_params[n*2:]
    
    # Refine with L-BFGS-B (local optimizer) to ensure tightness
    # We use a wrapper that passes only the variables to minimize
    def refine_objective(vars):
        return compute_cost(vars)
        
    refine_result = minimize(
        refine_objective,
        best_params,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-12}
    )
    
    final_params = refine_result.x
    centers = final_params[:n*2].reshape((n, 2))
    radii = final_params[n*2:]
    
    # Post-processing: Ensure no negative radii and clamp to valid range if needed
    radii = np.maximum(radii, 0.0)
    centers[:, 0] = np.clip(centers[:, 0], 0.0, 1.0)
    centers[:, 1] = np.clip(centers[:, 1], 0.0, 1.0)
    
    # Final check and correction for numerical stability
    # If a circle is slightly outside or overlapping due to float errors, shrink it
    # This is a safety measure. The optimizer should have found a valid low-cost state.
    
    # Calculate sum of radii
    sum_radii = np.sum(radii)
    
    print(f"Optimization finished. Sum of radii: {sum_radii:.6f}")
    
    return centers, radii, sum_radii
