import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Try many initial configurations with different arrangements and perturbations
    for trial in range(80):
        centers, radii = generate_initial(n, trial)
        x0 = np.concatenate([centers.flatten(), radii])
        
        # First phase: coarse optimization
        result = minimize(neg_sum, x0, args=(n,),
                         method='SLSQP',
                         bounds=get_bounds(n),
                         constraints={'type': 'ineq', 'fun': get_constraints, 'args': (n,)},
                         options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
        
        # Second phase: fine-tuning from best result
        if result.success or (not np.isinf(result.fun) and not np.isnan(result.fun)):
            result2 = minimize(neg_sum, result.x, args=(n,),
                             method='SLSQP',
                             bounds=get_bounds(n),
                             constraints={'type': 'ineq', 'fun': get_constraints, 'args': (n,)},
                             options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            
            if result2.fun < result.fun:
                result = result2
        
        current_sum = -result.fun
        if not np.isnan(current_sum) and current_sum > best_sum:
            best_sum = current_sum
            best_centers = result.x[:2*n].reshape(n, 2).copy()
            best_radii = result.x[2*n:].copy()
    
    return best_centers, best_radii, best_sum


def generate_initial(n, seed):
    """Generate initial hexagonal packing configuration with some variety."""
    rng = np.random.RandomState(seed)
    
    # Try different row arrangements
    arrangements = [
        [4, 5, 4, 5, 4, 4],  # 26 total
        [5, 4, 5, 4, 4, 4],  # 26 total
        [5, 5, 5, 5, 4, 2],  # 26 total
        [5, 5, 6, 5, 5],      # 26 total
        [6, 4, 6, 4, 6],      # 26 total
        [5, 5, 5, 5, 6],      # 26 total
        [4, 6, 4, 6, 6],      # 26 total
    ]
    
    idx = seed % len(arrangements)
    row_counts = arrangements[idx]
    
    centers = []
    
    num_rows = len(row_counts)
    y_spacing = 1.0 / (num_rows + 1)
    max_cols = max(row_counts)
    x_spacing = 1.0 / (max_cols + 1)
    
    for row_idx, count in enumerate(row_counts):
        if len(centers) >= n:
            break
        
        y = (row_idx + 1) * y_spacing
        
        # Alternate offset for hexagonal packing
        offset = 0.5 if row_idx % 2 == 1 else 0.0
        
        for col_idx in range(count):
            if len(centers) >= n:
                break
            
            x = (col_idx + 1 + offset) * x_spacing
            
            centers.append([x, y])
    
    # Ensure we have exactly n circles
    while len(centers) < n:
        x = 0.1 + rng.rand() * 0.8
        y = 0.1 + rng.rand() * 0.8
        centers.append([x, y])
    
    centers = np.array(centers[:n])
    
    # Small perturbation to break symmetry
    centers += rng.randn(*centers.shape) * 0.01
    centers = np.clip(centers, 0.05, 0.95)
    
    # Initial radius based on arrangement
    initial_r = 0.04 + (seed % 5) * 0.005
    radii = np.ones(n) * initial_r
    
    return centers, radii


def neg_sum(x, n):
    """Objective: negative sum of radii (for minimization)."""
    return -np.sum(x[2*n:])


def get_bounds(n):
    """Bounds: centers in [0,1], radii in [0, 0.5]."""
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))
    return bounds


def get_constraints(x, n):
    """
    Compute constraint violations.
    All constraints should be >= 0 for feasibility.
    """
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    out = []
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    for i in range(n):
        # x >= r
        out.append(centers[i, 0] - radii[i])
        # x + r <= 1
        out.append(1.0 - centers[i, 0] - radii[i])
        # y >= r
        out.append(centers[i, 1] - radii[i])
        # y + r <= 1
        out.append(1.0 - centers[i, 1] - radii[i])
    
    # Non-overlap constraints: distance >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            out.append(dist - radii[i] - radii[j])
    
    # Non-negative radius constraints
    for i in range(n):
        out.append(radii[i])
    
    return np.array(out)
