import numpy as np
from scipy.optimize import minimize

def run_packing():

    row_counts = [6, 5, 6, 5, 4]

    dy = np.sqrt(3) * r_init

    y = r_init

        x_start = r_init + (dx / 2.0) * (rc % 2)
        for k in range(rc):
            centers_init.append([x_start + k * dx, y])
        y += dy

    centers_init = np.clip(centers_init, 0.01, 0.99)
    radii_init = np.full(N, r_init)

def penalty_obj(x, N, mu):
    radii = x[2*N:]
    obj = -np.sum(radii)
    
    cx = x[0:2*N:3]
    cy = x[1:2*N:3]
    
    # Boundary violations (quadratic penalty for negative slack)
    obj += mu * np.sum(np.minimum(0, cx - radii)**2)
    obj += mu * np.sum(np.minimum(0, 1.0 - cx - radii)**2)
    obj += mu * np.sum(np.minimum(0, cy - radii)**2)
    obj += mu * np.sum(np.minimum(0, 1.0 - cy - radii)**2)
    
    # Pairwise overlap violations
    C = x[:2*N].reshape((N, 2))
    R = radii
    
    # Vectorized distance matrix computation
    diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    rad_sum = R[:, np.newaxis] + R[np.newaxis, :]
    
    # Extract upper triangle indices to avoid double counting and self-comparison
    i_idx, j_idx = np.triu_indices(N, k=1)
    violations = rad_sum[i_idx, j_idx]**2 - dist_sq[i_idx, j_idx]
    obj += mu * np.sum(np.maximum(0, violations)**2)
    
    return obj
