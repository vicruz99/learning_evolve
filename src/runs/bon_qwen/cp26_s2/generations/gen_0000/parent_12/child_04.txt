def create_initial_packing(n, seed=0):
    np.random.seed(seed)

    centers = []
    radii = []

    # Hexagonal packing arrangement: rows with 6, 5, 6, 5, 4 = 26 circles
    rows = [6, 5, 6, 5, 4]

    r = 0.088

    y = r
    row_idx = 0
    for num_circles in rows:
        # Alternate offset for hexagonal staggering
        x_offset = r + (row_idx % 2) * r * 0.55
        for i in range(num_circles):
            x = x_offset + i * 2.0 * r
            if x + r <= 1.0:
                centers.append([x, y])
                radii.append(r)
        y += r * np.sqrt(3.0) * 0.96
        row_idx += 1

    # Fill remaining if needed with random small circles
    while len(centers) < n:
        cx = float(np.random.uniform(0.15, 0.85))
        cy = float(np.random.uniform(0.15, 0.85))
        centers.append([cx, cy])
        radii.append(0.04)

    centers_arr = np.array(centers[:n])
    radii_arr = np.array(radii[:n])

    return centers_arr, radii_arr


def optimize_packing(centers, radii, n):

    x0 = np.concatenate([centers.flatten(), radii])

    # Bounds: centers in [0, 1], radii in [small positive, 0.5]

def penalized_objective(x, n, penalty):
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    # Objective: minimize negative sum of radii
    obj = -np.sum(radii)

    # Boundary constraint penalties
    for i in range(n):
        r = radii[i]
        cx = centers[i, 0]
        cy = centers[i, 1]

        # Left boundary: cx >= r
        violation = r - cx
        if violation > 0:
            obj += penalty * violation * violation

        # Right boundary: cx + r <= 1
        violation = cx + r - 1.0
        if violation > 0:
            obj += penalty * violation * violation

        # Bottom boundary: cy >= r
        violation = r - cy
        if violation > 0:
            obj += penalty * violation * violation

        # Top boundary: cy + r <= 1
        violation = cy + r - 1.0
        if violation > 0:
            obj += penalty * violation * violation

    # Overlap constraint penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-12)
            min_dist = radii[i] + radii[j]

            violation = min_dist - dist
            if violation > 0:
                obj += penalty * violation * violation

    return obj


def project_to_feasible(centers, radii, n):
    # Multiple passes to ensure strict feasibility
    for _ in range(200):
        changed = False

        # Enforce boundary constraints
        for i in range(n):
            r = radii[i]
            cx = centers[i, 0]
            cy = centers[i, 1]

            max_r = min(cx, 1.0 - cx, cy, 1.0 - cy)
            if r > max_r + 1e-13:
                radii[i] = max(1e-10, max_r)
                changed = True

        # Enforce non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]

                if dist < min_dist - 1e-12:
                    # Reduce the larger radius
                    scale = 0.999
                    if radii[i] >= radii[j]:
                        new_r = max(1e-10, (dist - radii[j]) * scale)
                        if new_r < radii[i]:
                            radii[i] = new_r
                            changed = True
                    else:
                        new_r = max(1e-10, (dist - radii[i]) * scale)
                        if new_r < radii[j]:
                            radii[j] = new_r
                            changed = True

        if not changed:
            break

    return centers, radii
