import numpy as np
from scipy.optimize import minimize
import math

def compute_soft_radii_sum(centers, lam=50.0):
    """
    Compute the sum of soft-minimum radii for a set of centers.
    Uses log-sum-exp to approximate min(x) = -1/lam * log(sum(exp(-lam * x))).
    """
    n = centers.shape[0]
    sum_radii = 0.0
    
    # Precompute pairwise squared distances to avoid redundant calculations?
    # For N=26, O(N^2) is small, but let's be careful with memory/speed.
    # We can compute distances on the fly.
    
    for i in range(n):
        x, y = centers[i]
        
        # Constraints terms for soft min
        # 1. Boundary distances
        terms = [x, 1.0 - x, y, 1.0 - y]
        
        # 2. Inter-circle distances / 2
        # dist(c_i, c_j) / 2
        for j in range(n):
            if i == j:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            terms.append(dist / 2.0)
            
        # Soft min
        # To prevent overflow/underflow in exp, we can subtract max(terms)
        # But terms are positive distances, max is <= 1.
        # With lam=50, exp(-50*1) is very small but representable.
        
        # Using log-sum-exp trick for stability if needed, 
        # but for positive small numbers it's usually fine.
        # Let's just compute directly.
        
        # exp terms
        # To handle numerical stability, find min_val first? 
        # Actually, standard log-sum-exp:
        # min_u approx -1/lam * log( sum exp(-lam * u) )
        # If u are large, -lam*u is very negative, exp is 0.
        
        exp_terms = np.exp(-lam * np.array(terms))
        sum_exp = np.sum(exp_terms)
        
        # If sum_exp is 0 (all terms large), radius is large. 
        # But terms are bounded by 1.
        soft_r = -1.0 / lam * math.log(sum_exp + 1e-300) # epsilon for safety
        sum_radii += soft_r
        
    return sum_radii

def compute_exact_radii(centers):
    """
    Compute exact feasible radii for given centers.
    """
    n = centers.shape[0]
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        # Max radius limited by boundaries
        r = min(x, 1.0 - x, y, 1.0 - y)
        
        # Limited by other circles
        for j in range(n):
            if i == j:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            if dist / 2.0 < r:
                r = dist / 2.0
        
        radii[i] = r
    return radii
