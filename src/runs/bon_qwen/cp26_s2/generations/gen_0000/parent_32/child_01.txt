def obj_penalty(x, n, penalty):
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    result = -np.sum(radii)
    for i in range(n):
        for d in range(2):
            v1 = -centers[i, d] + radii[i]
            if v1 > 0:
                result += penalty * v1 * v1
            v2 = centers[i, d] + radii[i] - 1.0
            if v2 > 0:
                result += penalty * v2 * v2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < 1e-15:
                dist = 1e-15
            v = radii[i] + radii[j] - dist
            if v > 0:
                result += penalty * v * v
    return result
