import numpy as np
from scipy.optimize import minimize


def objective_func(x, n):
    return -np.sum(x[2 * n :])


def make_boundary_constraints(n):
    constraints = []
    for i in range(n):
        idx = i
        constraints.append(
            {
                "type": "ineq",
                "fun": lambda x, i=idx: x[3 * i] - x[3 * i + 2],
            }
        )
        constraints.append(
            {
                "type": "ineq",
                "fun": lambda x, i=idx: 1.0 - x[3 * i] - x[3 * i + 2],
            }
        )
        constraints.append(
            {
                "type": "ineq",
                "fun": lambda x, i=idx: x[3 * i + 1] - x[3 * i + 2],
            }
        )
        constraints.append(
            {
                "type": "ineq",
                "fun": lambda x, i=idx: 1.0 - x[3 * i + 1] - x[3 * i + 2],
            }
        )
    return constraints


def make_overlap_constraints(n):
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            ii, jj = i, j
            constraints.append(
                {
                    "type": "ineq",
                    "fun": lambda x, i=ii, j=jj: (x[3 * i] - x[3 * j]) ** 2
                    + (x[3 * i + 1] - x[3 * j + 1]) ** 2
                    - (x[3 * i + 2] + x[3 * j + 2]) ** 2,
                }
            )
    return constraints


def make_bounds(n):
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
    return bounds


def initialize_corner_heavy(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0

    r_corner = 0.14
    corners = [
        [r_corner, r_corner],
        [1.0 - r_corner, r_corner],
        [r_corner, 1.0 - r_corner],
        [1.0 - r_corner, 1.0 - r_corner],
    ]
    for c in corners:
        centers[idx] = c
        radii[idx] = r_corner
        idx += 1

    r_edge = 0.07
    edge_positions = []
    for i in range(2):
        x = 0.35 + i * 0.30
        edge_positions.append([x, r_edge])
        edge_positions.append([x, 1.0 - r_edge])
    for i in range(2):
        y = 0.35 + i * 0.30
        edge_positions.append([r_edge, y])
        edge_positions.append([1.0 - r_edge, y])

    for ep in edge_positions:
        centers[idx] = ep
        radii[idx] = r_edge
        idx += 1

    r_int = 0.035
    interior_positions = []
    for row in range(4):
        for col in range(4):
            x = 0.28 + col * 0.16
            y = 0.28 + row * 0.16
            interior_positions.append([x, y])

    for ip in interior_positions:
        if idx < n:
            centers[idx] = ip
            radii[idx] = r_int
            idx += 1

    while idx < n:
        centers[idx] = [0.5 + (idx - 16) * 0.05, 0.5]
        radii[idx] = 0.025
        idx += 1

    return centers, radii


def initialize_alternate(n):

        if idx < n:

            radii[idx] = r_edge
            idx += 1
        if idx < n:
            centers[idx] = [x, 1.0 - r_edge]

            idx += 1
    for i in range(2):
        y = 0.30 + i * 0.40
        if idx < n:
            centers[idx] = [r_edge, y]
            radii[idx] = r_edge
            idx += 1
        if idx < n:
            centers[idx] = [1.0 - r_edge, y]

    r_int = 0.03
    hex_spacing = 0.15
    for row in range(5):
        for col in range(4):
            if idx >= n:
                break

            y = 0.2 + row * hex_spacing * 0.866
            if x <= 0.8 and y <= 0.8:
                centers[idx] = [x, y]
                radii[idx] = r_int
                idx += 1

    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.02
        idx += 1

def force_directed_relax(centers, radii, n, iterations=20000):
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(iterations):
        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-12 else 0.0001
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    repulsion = overlap * 40.0
                    fx = repulsion * dx / dist
                    fy = repulsion * dy / dist
                    forces[i][0] += fx
                    forces[i][1] += fy
                    forces[j][0] -= fx
                    forces[j][1] -= fy

        dt = 0.0015 / (1.0 + iteration * 0.00003)
        centers[:, 0] += forces[:, 0] * dt
        centers[:, 1] += forces[:, 1] * dt

        for i in range(n):
            r = radii[i]
            eps = 1e-7
            centers[i][0] = max(r + eps, min(1.0 - r - eps, centers[i][0]))
            centers[i][1] = max(r + eps, min(1.0 - r - eps, centers[i][1]))

        if iteration % 15 == 0:
            inflow = 0.00008 * (1.0 + 0.5 * np.sin(iteration * 0.01))
            radii += inflow

        for i in range(n):
            x, y = centers[i]
            radii[i] = min(x, 1.0 - x, y, 1.0 - y, radii[i])
            radii[i] = max(0.0, radii[i])

    return centers, radii

def scipy_optimize_step(n, centers, radii):
    x0 = np.concatenate([centers.flatten(), radii])
    constraints = make_boundary_constraints(n) + make_overlap_constraints(n)
    bounds = make_bounds(n)

    try:
        result = minimize(
            lambda x: objective_func(x, n),
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 500, "ftol": 1e-12, "disp": False},
        )
        if result.success or result.status >= 0:
            centers_opt = result.x[: 2 * n].reshape((n, 2))
            radii_opt = result.x[2 * n :]
            return centers_opt, radii_opt
    except Exception:
        pass
    return centers, radii


def ensure_validity(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        r = min(x, 1.0 - x, y, 1.0 - y, r)
        r = max(0.0, r)
        radii[i] = r
        centers[i] = [x, y]
    return centers, radii


def run_packing():

            else:

            centers, radii = ensure_validity(centers, radii, n)

            for refine_round in range(3):

                centers, radii = ensure_validity(centers, radii, n)

            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()

    for restart in range(5):
