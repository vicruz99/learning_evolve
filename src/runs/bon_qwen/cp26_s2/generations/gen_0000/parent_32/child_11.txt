import numpy as np
from scipy.optimize import minimize

    best_radii = None

        [5, 5, 6, 5, 5],
        [4, 6, 5, 6, 5],
        [6, 4, 6, 5, 5],

def force_optimize(centers, radii, n, max_outer=4000, max_inner=150):
    centers = centers.copy().astype(np.float64)
    radii = radii.copy().astype(np.float64)
    
    for outer in range(max_outer):
        # Decay growth rate over time
        growth = 1.0 + 0.003 / (1 + outer * 0.0008)
        radii *= growth
        
        # Resolve overlaps with repulsion
        for inner in range(max_inner):
            forces = np.zeros((n, 2))
            max_overlap = 0
            
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist_sq = dx * dx + dy * dy
                    dist = np.sqrt(dist_sq)
                    
                    if dist < 1e-12:
                        dist = 1e-12
                        dx, dy = 1e-12, 0
                    
                    required = radii[i] + radii[j]
                    overlap = required - dist
                    max_overlap = max(max_overlap, overlap)
                    
                    if overlap > 0:
                        # Stronger repulsion for larger overlaps
                        force_mag = overlap * 3.0
                        fx = dx / dist * force_mag
                        fy = dy / dist * force_mag
                        forces[i, 0] -= fx
                        forces[i, 1] -= fy
                        forces[j, 0] += fx
                        forces[j, 1] += fy
            
            # Apply forces with adaptive step size
            step = 0.08
            centers += forces * step
            
            # Clamp to valid region
            for i in range(n):
                max_r = min(centers[i, 0], 1 - centers[i, 0],
                            centers[i, 1], 1 - centers[i, 1])
                radii[i] = min(radii[i], max(0, max_r))
                centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
                centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
            
            if max_overlap < 1e-8:
                break
    
    return centers, radii


def scipy_optimize(centers, radii, n):
    params = np.concatenate([centers.flatten(), radii])
    
    def objective(p):
        c = p[:2 * n].reshape(n, 2)
        r = p[2 * n:]
        
        penalty = 0.0
        
        # Boundary penalties (quadratic for smoothness)
        for i in range(n):
            penalty += max(0.0, r[i] - c[i, 0]) ** 2 * 5000
            penalty += max(0.0, r[i] - (1 - c[i, 0])) ** 2 * 5000
            penalty += max(0.0, r[i] - c[i, 1]) ** 2 * 5000
            penalty += max(0.0, r[i] - (1 - c[i, 1])) ** 2 * 5000
        
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                overlap = r[i] + r[j] - dist
                penalty += max(0.0, overlap) ** 2 * 5000
        
        return -np.sum(r) + penalty
    
    bounds = [(0.005, 0.995)] * (2 * n) + [(0.001, 0.5)] * n
    
    result = minimize(objective, params, bounds=bounds, method='L-BFGS-B',
                      options={'maxiter': 8000, 'ftol': 1e-16, 'gtol': 1e-12})
    
    c_opt = result.x[:2 * n].reshape(n, 2)
    r_opt = result.x[2 * n:]
    
    return c_opt, r_opt


def fix_violations(centers, radii, n):
    centers = centers.copy()
    radii = radii.copy()
    
    # Fix boundary violations
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0],
                    centers[i, 1], 1 - centers[i, 1])
        radii[i] = max(0, min(radii[i], max_r))
        centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
    
    # Fix overlap violations by reducing radii
    for _ in range(500):
        max_overlap = 0
        max_i, max_j = -1, -1
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
                    max_i, max_j = i, j
        
        if max_overlap <= 1e-12:
            break
        
        # Reduce both radii slightly
        reduction = max_overlap * 0.5 + 1e-10
        radii[max_i] = max(0, radii[max_i] - reduction)
        radii[max_j] = max(0, radii[max_j] - reduction)
    
    # Final radius clamping
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0],
                    centers[i, 1], 1 - centers[i, 1])
        radii[i] = max(0, min(radii[i], max_r))
    
    return centers, radii
