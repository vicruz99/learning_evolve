import numpy as np
from scipy.optimize import minimize

def _objective(v, n):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(v[2*n:])

def _constraints(v, n):
    """Constraint function: returns array of all inequality constraints >= 0"""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.linalg.norm(diff, axis=2)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Only upper triangle is needed (i < j)
    triu = np.triu_indices(n, k=1)
    con.append(dist[triu] - r_sum[triu])
    
    return np.concatenate(con)

def run_packing():
    n = 26
    
    # Initialize with a hexagonal lattice pattern for fast convergence
    centers = []
    r_init = 0.095
    spacing = 2.0 * r_init
    h = spacing * np.sqrt(3.0) / 2.0
    
    # 6 rows with alternating shifts: 4,5,4,5,4,4 -> 26 circles
    row_counts = [4, 5, 4, 5, 4, 4]
    y_base = 0.15
    
    for r_idx, count in enumerate(row_counts):
        offset = (spacing / 2.0) if r_idx % 2 == 1 else 0.0
        row_width = (count - 1) * spacing
        start_x = (1.0 - row_width) / 2.0 - offset
        for c in range(count):
            x = start_x + c * spacing
            y = y_base + r_idx * h
            centers.append([x, y])
            
    centers = np.array(centers)
    
    # Flatten to optimization vector: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = r_init
        
    # Bounds for variables
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
    
    cons = {'type': 'ineq', 'fun': _constraints, 'args': (n,)}
    
    # Run SLSQP optimization
    res = minimize(_objective, x0, args=(n,), method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
                   
    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = res.x[2*n:]
    
    # Post-process to guarantee strict feasibility
    final_radii = np.maximum(final_radii, 1e-10)
    final_centers[:, 0] = np.clip(final_centers[:, 0], final_radii, 1.0 - final_radii)
    final_centers[:, 1] = np.clip(final_centers[:, 1], final_radii, 1.0 - final_radii)
    
    return final_centers, final_radii, float(np.sum(final_radii))
