import numpy as np
from scipy.optimize import minimize

def boundary_constraints(p, n):
    r = p[:n]
    centers = p[n:].reshape((n, 2))
    x = centers[:, 0]
    y = centers[:, 1]
    
    # x - r >= 0
    c1 = x - r
    # 1 - (x + r) >= 0
    c2 = 1 - (x + r)
    # y - r >= 0
    c3 = y - r
    # 1 - (y + r) >= 0
    c4 = 1 - (y + r)
    
    return np.concatenate([c1, c2, c3, c4])

def overlap_constraints(p, n):
    r = p[:n]
    centers = p[n:].reshape((n, 2))
    
    # Compute pairwise squared distances
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2) # (n, n)
    
    # Compute (r_i + r_j)^2
    # r shape (n,)
    sum_r = r[:, np.newaxis] + r[np.newaxis, :]
    min_dist_sq = sum_r**2
    
    # We need lower triangle (i < j)
    # dist_sq[i, j] - min_dist_sq[i, j] >= 0
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    constraints = dist_sq[mask] - min_dist_sq[mask]
    
    return constraints

def all_constraints(p, n):
    c_bound = boundary_constraints(p, n)
    c_overlap = overlap_constraints(p, n)
    return np.concatenate([c_bound, c_overlap])
