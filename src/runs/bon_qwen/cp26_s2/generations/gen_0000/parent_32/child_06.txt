import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Heuristic Initialization: 6-row Hexagonal Packing
    # Row counts: 5, 4, 5, 4, 5, 3 (Total 26)
    row_counts = [5, 4, 5, 4, 5, 3]
    centers = []
    
    # Base radius for initialization
    r_init = 0.09
    # Vertical and horizontal spacings for hex lattice
    dy = r_init * np.sqrt(3)
    dx = 2 * r_init
    
    current_y = r_init
    for i, count in enumerate(row_counts):
        # Stagger odd rows by dx/2
        x_start = r_init + (0.5 * dx if i % 2 == 1 else 0)
        
        for _ in range(count):
            centers.append([x_start, current_y])
            x_start += dx
        current_y += dy
        
    centers = np.array(centers)
    
    # 2. Random Perturbation to escape local minima
    rng = np.random.default_rng(42)
    centers += rng.uniform(-0.005, 0.005, size=centers.shape)
    
    # Ensure initial centers are within bounds
    centers = np.clip(centers, 0.01, 0.99)

    # 3. Define the Optimization Problem
    # Variables: x0, y0, r0, x1, y1, r1, ... (3 * n)
    x0 = np.concatenate([centers.flatten(), np.full(n, 0.09)])

    def objective(x):
        # Maximize sum of radii -> Minimize negative sum
        radii = x[2::3]
        return -np.sum(radii)

    def constraints_factory(n):
        cons = []
        # Boundary constraints
        for i in range(n):
            xi = 3 * i
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[xi] - x[xi+2]}) # x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[xi] - x[xi+2]}) # x + r <= 1
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[xi+1] - x[xi+2]}) # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[xi+1] - x[xi+2]}) # y + r <= 1
            
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                xi, xj = 3 * i, 3 * j
                def dist_constraint(x, i=i, j=j):
                    dist_sq = (x[xi] - x[xj])**2 + (x[xi+1] - x[xj+1])**2
                    rad_sum = x[xi+2] + x[xj+2]
                    return dist_sq - rad_sum**2
                cons.append({'type': 'ineq', 'fun': dist_constraint})
        return cons

    constraints = constraints_factory(n)

    # 4. Solve
    res = minimize(objective, x0, method='SLSQP', constraints=constraints, options={'ftol': 1e-12, 'maxiter': 1000})
    
    if not res.success:
        print("Warning: Optimization did not converge perfectly, using best result.")

    # 5. Format Output
    final_centers = np.array(res.x[::3]).reshape(-1, 2)
    final_radii = np.array(res.x[2::3])
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

# Execute to get the result
if __name__ == "__main__":
    centers, radii, s = run_packing()
    print(f"Sum of radii: {s:.5f}")
