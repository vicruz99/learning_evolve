import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initialization: Hexagonal Lattice
    # We estimate a radius slightly less than 0.1 to ensure initial feasibility.
    # A hexagonal packing allows for higher density.
    # We arrange rows with alternating counts to fit in the square.
    # Estimated r ~ 0.1. 
    # Rows: 5, 4, 5, 4, 5, 3 (total 26) might fit.
    # Let's generate a robust initial guess.
    
    # Initial radius estimate
    r_init = 0.095
    
    centers = []
    radii = []
    
    y = r_init
    row_idx = 0
    
    # We need to place 26 circles. 
    # Strategy: Fill rows in a hexagonal pattern.
    # Full rows (aligned with x-axis start) can fit floor((1-2r)/2r) + 1 circles?
    # Width available = 1. Circle diameter = 2r.
    # Number of circles m such that m*2r <= 1 => m <= 1/2r.
    # For r=0.095, 1/0.19 = 5.26 -> 5 circles fit in a row.
    # Shifted rows (indented by r) fit m such that 2r + (m-1)2r + r <= 1? 
    # Actually centers at r, 3r... vs 2r, 4r...
    # Shifted row centers start at 2r. Last at 2r + (m-1)2r.
    # Constraint: 2r + (m-1)2r + r <= 1 => 3r + 2r(m-1) <= 1.
    # For r=0.095: 0.285 + 0.19(m-1) <= 1 => 0.19(m-1) <= 0.715 => m-1 <= 3.7 => m=4.
    
    # Pattern: 5, 4, 5, 4, 5, 3? Sum = 26.
    # Let's try to construct this.
    
    target_counts = [5, 4, 5, 4, 5, 3]
    
    count = 0
    while count < n:
        m = target_counts[row_idx % len(target_counts)]
        
        if row_idx % 2 == 0:
            # Full row
            # Centers at r, 3r, 5r, ...
            # x_k = r + k * 2r for k in 0..m-1
            xs = [r_init + k * 2 * r_init for k in range(m)]
        else:
            # Shifted row
            # Centers at 2r, 4r, ...
            # x_k = 2r + k * 2r for k in 0..m-1
            xs = [2 * r_init + k * 2 * r_init for k in range(m)]
            
        # Check if we have enough circles to take from this row
        take = min(m, n - count)
        for x in xs[:take]:
            centers.append([x, y])
            radii.append(r_init)
            count += 1
        
        # Move to next row
        y += r_init * np.sqrt(3)
        row_idx += 1
        
        if count >= n:
            break
            
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    # Size: 3 * n
    x0 = np.concatenate([centers.T.flatten(), radii])
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
        
    def objective(x_vars):
        # We want to maximize sum of radii, so minimize negative sum
        r_vals = x_vars[2::3]
        return -np.sum(r_vals)
        
    def constraint_wall(x_vars):
        # Returns values that must be >= 0 for feasible region? 
        # SLSQP expects fun(x) >= 0 for 'ineq' type? 
        # Actually scipy.optimize.minimize constraints:
        # type 'eq': fun(x) = 0
        # type 'ineq': fun(x) >= 0
        # So we need to transform our <= 0 constraints to >= 0.
        
        constraints_list = []
        for i in range(n):
            xi = x_vars[3*i]
            yi = x_vars[3*i+1]
            ri = x_vars[3*i+2]
            
            # x - r >= 0
            constraints_list.append(xi - ri)
            # 1 - x - r >= 0  => 1 - xi - ri >= 0
            constraints_list.append(1.0 - xi - ri)
            # y - r >= 0
            constraints_list.append(yi - ri)
            # 1 - y - r >= 0
            constraints_list.append(1.0 - yi - ri)
            
        return np.array(constraints_list)

    def constraint_overlap(x_vars):
        constraints_list = []
        for i in range(n):
            xi = x_vars[3*i]
            yi = x_vars[3*i+1]
            ri = x_vars[3*i+2]
            for j in range(i + 1, n):
                xj = x_vars[3*j]
                yj = x_vars[3*j+1]
                rj = x_vars[3*j+2]
                
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                # dist >= ri + rj  =>  dist - ri - rj >= 0
                constraints_list.append(dist - ri - rj)
        return np.array(constraints_list)

    # Combine constraints
    def all_constraints(x_vars):
        c1 = constraint_wall(x_vars)
        c2 = constraint_overlap(x_vars)
        return np.concatenate([c1, c2])

    cons = {'type': 'ineq', 'fun': all_constraints}

    # 3. Run Optimization
    # Using SLSQP
    try:
        result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'ftol': 1e-12, 'maxiter': 1000, 'disp': False})
        x_opt = result.x
    except Exception:
        # Fallback if optimization fails
        x_opt = x0

    # Extract results
    centers_opt = x_opt[:2*n].reshape((n, 2))
    radii_opt = x_opt[2*n:]
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii

if __name__ == "__main__":
    centers, radii, s_r = run_packing()
    print(f"Sum of radii: {s_r}")
    # Basic validation
    import math
    valid = True
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            valid = False
        for j in range(i + 1, 26):
            dist = math.hypot(centers[i][0] - centers[j][0], centers[i][1] - centers[j][1])
            if dist < radii[i] + radii[j] - 1e-9:
                valid = False
    print(f"Valid packing: {valid}")
