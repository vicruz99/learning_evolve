import numpy as np
from scipy.optimize import minimize

def generate_hexagonal_grid(n, width=1.0, height=1.0):
    """
    Generates n points on a hexagonal grid scaled to fit within width x height.
    """
    # Estimate row and column counts
    # Area of hexagonal cell is sqrt(3)/2 * s^2. 
    # Roughly n * sqrt(3)/2 * s^2 ~= width * height
    # s ~= sqrt( 2 * width * height / (n * sqrt(3)) )
    # But we want to fit exactly, so let's just try a dense grid and pick n points.
    
    # Heuristic for spacing
    est_area = width * height
    cell_area = est_area / n
    # For hex grid, vertical spacing is s*sqrt(3)/2, horizontal is s.
    # Area per point approx s^2 * sqrt(3)/2.
    # s approx sqrt( 2 * cell_area / sqrt(3) )
    s = np.sqrt(2 * cell_area / np.sqrt(3))
    
    points = []
    y = 0
    row = 0
    while len(points) < n:
        x = 0
        if row % 2 == 1:
            x = s / 2
        while x + s/2 < width and len(points) < n:
            if y + s/2 < height:
                points.append([x, y])
            x += s
        y += s * np.sqrt(3) / 2
        row += 1
        
    # Convert to numpy array
    centers = np.array(points[:n])
    
    # Center the grid in the square
    if len(centers) > 0:
        cx, cy = np.mean(centers, axis=0)
        centers -= np.array([cx - 0.5, cy - 0.5])
        
    return centers

def compute_initial_radii(centers):
    """
    Computes initial radii based on minimum distance to boundaries and other circles.
    """
    n = centers.shape[0]
    radii = np.ones(n) * 0.1 # Default small radius
    
    # Boundary distance
    for i in range(n):
        x, y = centers[i]
        r_bound = min(x, 1-x, y, 1-y)
        radii[i] = r_bound
        
    # Inter-circle distance
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            r_pair = dist / 2
            radii[i] = min(radii[i], r_pair)
            radii[j] = min(radii[j], r_pair)
            
    # Scale down slightly to ensure strict feasibility
    return radii * 0.9

def objective_func(vars, n):
    """
    Objective: Maximize sum of radii -> Minimize -sum(radii)
    """
    radii = vars[2*n:]
    return -np.sum(radii)

def constraint_func(vars, n):
    """
    Returns a list of constraint dictionaries for scipy.optimize.minimize
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    constraints = []
    
    # Boundary constraints
    # x_i >= r_i  => x_i - r_i >= 0
    # 1 - x_i >= r_i => 1 - x_i - r_i >= 0
    # Same for y
    
    for i in range(n):
        idx = i
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i] - v[2*n+i]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[2*i] - v[2*n+i]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i+1] - v[2*n+i]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[2*i+1] - v[2*n+i]})
        
    # Non-overlap constraints
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: \
                    (v[2*i] - v[2*j])**2 + (v[2*i+1] - v[2*j+1])**2 - (v[2*n+i] + v[2*n+j])**2
            })
            
    return constraints
