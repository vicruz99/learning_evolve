def compute_objective_and_gradient(params, n_circles):
    """
    Computes the objective value (sum of radii) and the penalty for constraint violations.
    Also computes gradients if needed, but we will use a method that approximates them or use numerical gradients.
    For SLSQP, we can provide the objective and constraints, or just objective with penalty.
    Given the complexity, a penalty method in the objective is often easier to implement for non-convex problems 
    where constraint definitions can be tedious. However, SLSQP expects constraints as separate functions.
    
    Let's use a pure penalty method with a method that handles bounds, like L-BFGS-B or Powell, 
    or simply SLSQP with the penalty inside the objective and no explicit constraints (except bounds).
    Actually, SLSQP with penalty in objective is a common heuristic.
    
    Returns: -sum(radii) + penalty (to minimize)
    """
    centers = params[0:2*n_circles].reshape(n_circles, 2)
    radii = params[2*n_circles:]
    
    sum_radii = np.sum(radii)
    
    # Penalty parameters
    # High weight to enforce constraints
    penalty_weight = 1000.0
    
    violation = 0.0
    
    # Boundary constraints
    # Circle i must be inside [0,1]x[0,1]
    # x_i - r_i >= 0  => r_i <= x_i
    # x_i + r_i <= 1  => r_i <= 1 - x_i
    # same for y
    
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        
        # Check boundaries
        # We want r <= x, r <= 1-x, r <= y, r <= 1-y
        # Violation is max(0, r - x), etc.
        # Using a smooth approximation or just max(0, val)^2
        
        val_x_low = r - x
        if val_x_low > 0:
            violation += val_x_low**2
            
        val_x_high = r - (1 - x)
        if val_x_high > 0:
            violation += val_x_high**2
            
        val_y_low = r - y
        if val_y_low > 0:
            violation += val_y_low**2
            
        val_y_high = r - (1 - y)
        if val_y_high > 0:
            violation += val_y_high**2

    # Overlap constraints
    # dist(i, j) >= r_i + r_j
    # Violation if r_i + r_j - dist > 0
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                violation += overlap**2
                
    return -sum_radii + penalty_weight * violation
