import numpy as np
import scipy.optimize as opt

def calculate_violations(centers, radii):
    """
    Calculates the total violation of constraints.
    Returns 0 if valid, positive value otherwise.
    """
    n = centers.shape[0]
    violation = 0.0
    
    # Boundary violations
    # x - r >= 0, 1 - (x + r) >= 0  => r <= x <= 1-r
    # Same for y
    
    # Left/Right walls
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Distance to left wall: x - r
        if x - r < 0:
            violation += (x - r)**2
        # Distance to right wall: 1 - (x + r)
        if 1 - (x + r) < 0:
            violation += (1 - (x + r))**2
        # Distance to bottom wall: y - r
        if y - r < 0:
            violation += (y - r)**2
        # Distance to top wall: 1 - (y + r)
        if 1 - (y + r) < 0:
            violation += (1 - (y + r))**2
            
    # Pairwise overlaps
    # dist >= r1 + r2  => dist - (r1+r2) >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penetration = min_dist - dist
                violation += penetration**2
                
    return violation

def objective_function(params, n_circles):
    """
    Objective function to minimize.
    We want to maximize sum of radii, so we minimize -sum(radii).
    We add a large penalty for violations.
    """
    centers = params[:2*n_circles].reshape((n_circles, 2))
    radii = params[2*n_circles:]
    
    # Ensure radii are non-negative (though bounds handle this, for safety)
    radii = np.maximum(radii, 0)
    
    sum_radii = np.sum(radii)
    penalty_weight = 10000.0
    violation = calculate_violations(centers, radii)
    
    # If violation is high, it dominates
    return -sum_radii + penalty_weight * violation

def get_bounds(n_circles):
    """
    Returns bounds for the optimization variables.
    x, y in [0, 1], r in [0, 0.5]
    """
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    return bounds

def generate_initial_config(n_circles):

    rows = int(np.ceil(n_circles / cols))
    
    step_x = 1.0 / (cols + 1)
    step_y = 1.0 / (rows + 1)
    
    initial_centers = []
    initial_radii = []
    
    count = 0
    for r in range(rows):

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
