def relax_packing(centers, radii, n):
    """
    Simple force-directed relaxation to expand circles.
    """
    centers = centers.copy()
    radii = radii.copy()
    
    # Increment radii slightly
    radii += 0.0005
    
    # Resolve overlaps and boundary violations
    for _ in range(20): # Iterations per expansion
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                c1, c2 = centers[i], centers[j]
                dist = np.linalg.norm(c1 - c2)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    overlap = min_dist - dist
                    direction = (c1 - c2) / dist
                    # Push apart
                    forces[i] += direction * overlap * 0.5
                    forces[j] -= direction * overlap * 0.5
        
        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0: forces[i, 0] += (r - x) * 0.5
            if x + r > 1: forces[i, 0] -= (x + r - 1) * 0.5
            if y - r < 0: forces[i, 1] += (r - y) * 0.5
            if y + r > 1: forces[i, 1] -= (y + r - 1) * 0.5
            
        centers += forces
        
    return centers, radii

def validate_packing(centers, radii):

    Validate that circles don't overlap and are inside the unit square

    Args:

        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """

    # Check for NaN values

        print("NaN values detected in circle centers")
        return False

        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
