import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    
    def neg_obj(flat_centers):
        """
        Computes the negative sum of radii for a given set of centers.
        Radii are implicitly determined by the tightest constraints (walls or neighbors).
        """
        centers = flat_centers.reshape(N, 2)
        x, y = centers[:, 0], centers[:, 1]
        
        # Distance to the four walls
        r_wall = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
        
        # Pairwise distances between circles
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)  # Ignore self-distance
        
        # Max radius allowed by neighbor collisions
        r_circ = np.min(dists, axis=1) / 2.0
        
        # Feasible radius is the minimum of wall and circle constraints
        r = np.minimum(r_wall, r_circ)
        return -np.sum(r)

    bounds = [(0.0, 1.0)] * (N * 2)
    
    best_result = None
    best_val = np.inf
    
    # Multiple restarts with Differential Evolution followed by Nelder-Mead
    seeds = [42, 123, 456, 789, 0]
    for seed in seeds:
        try:
            # Global search
            res_de = differential_evolution(
                neg_obj, bounds, 
                popsize=15, maxiter=150, 
                seed=seed, tol=1e-5, polish=False
            )
            # Local refinement
            res_nm = minimize(
                neg_obj, res_de.x, 
                method='Nelder-Mead', 
                options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-8}
            )
            if res_nm.fun < best_val:
                best_val = res_nm.fun
                best_result = res_nm
        except Exception:
            continue
            
    # Fallback if all seeds fail
    if best_result is None:
        res_de = differential_evolution(neg_obj, bounds, popsize=20, maxiter=100, seed=0, polish=True)
        best_result = res_de
        
    centers = best_result.x.reshape(N, 2)
    
    # Compute final radii accurately
    x, y = centers[:, 0], centers[:, 1]
    r_wall = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_circ = np.min(dists, axis=1) / 2.0
    radii = np.minimum(r_wall, r_circ)
    radii = np.maximum(radii, 0.0)  # Ensure non-negativity
    
    return centers, radii, float(np.sum(radii))
