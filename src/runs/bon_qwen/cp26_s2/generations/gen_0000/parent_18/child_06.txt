import numpy as np

def run_packing():
    np.random.seed(42)
    n = 26
    
    def is_valid(centers, radii, tol=1e-10):
        """Check if packing is valid"""
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -tol or x + r > 1 + tol or y - r < -tol or y + r > 1 + tol:
                return False
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - tol:
                    return False
        return True
    
    def get_max_radius_at(center, centers, radii, exclude_idx=-1):
        """Get maximum possible radius for a circle at given center"""
        max_r = min(center[0], 1 - center[0], center[1], 1 - center[1])
        for j in range(len(centers)):
            if j == exclude_idx:
                continue
            d = np.linalg.norm(center - centers[j])
            max_r = min(max_r, d - radii[j])
        return max(0.0, max_r)
    
    def force_optimize(centers, radii, n_steps=2000, cooling_power=2.0):
        """Force-directed optimization with cooling"""
        for step in range(n_steps):
            # Cooling schedule
            progress = step / n_steps
            alpha = 0.04 * max(0.0005, (1 - progress) ** cooling_power)
            
            # Compute forces
            forces = np.zeros_like(centers)
            
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):

                    dist = np.linalg.norm(diff)

                        continue

                    if dist < min_dist:
                        force_mag = (min_dist - dist) / (dist + 0.001) * 80.0
                        forces[i] += force_mag * unit_diff
                        forces[j] -= force_mag * unit_diff
                
                # Boundary repulsion

                    if centers[i, dim] - radii[i] < 0.001:

        return centers, radii
