import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Uses a force-based inflation algorithm followed by numerical refinement.
    """
    np.random.seed(42) # For reproducibility
    n = 26
    
    # --- 1. Initialization ---
    # Random centers within safe margins
    centers = np.random.uniform(0.1, 0.9, size=(n, 2))
    radii = np.full(n, 0.01)
    
    # --- 2. Force-Based Inflation Simulation ---
    # Parameters for the simulation
    dt = 0.05       # Time step for position updates
    growth_rate = 0.0005 # Rate at which radii grow per step
    num_inflation_steps = 1000
    num_settle_steps = 50 # Steps to resolve overlaps at current size
    
    # Stiffness constants for repulsive forces
    k_overlap = 5.0
    k_wall = 10.0
    damping = 0.9 # Velocity damping to prevent oscillation
    
    # Initialize velocities
    velocities = np.zeros_like(centers)

    for step in range(num_inflation_steps):
        # A. Grow Radii
        # Grow all circles slightly. 
        # We could grow them differently, but uniform growth is a good baseline.
        radii += growth_rate
        
        # B. Calculate Forces and Update Positions (Settle phase)
        # We run several sub-steps to resolve overlaps caused by growth
        for _ in range(num_settle_steps):
            forces = np.zeros_like(centers)
            
            # 1. Circle-Circle Repulsion
            # Vectorized computation for efficiency
            # Compute all pairwise distance vectors and squared distances
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # Shape (N, N, 2)
            dist_sq = np.sum(diff**2, axis=2) # Shape (N, N)
            dist = np.sqrt(dist_sq + 1e-12) # Avoid division by zero
            
            # Radii sum matrix
            rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
            
            # Overlap amount: positive if overlapping
            # We want to penalize dist < rad_sum
            overlap = rad_sum - dist
            
            # Mask for overlapping pairs (and upper triangle to avoid double counting if needed, 
            # but here we sum forces so full matrix is fine if we handle symmetry)
            # Actually, simpler to just iterate or use masked arrays. 
            # For N=26, explicit loop is fast enough and clearer.
            
            for i in range(n):
                for j in range(i + 1, n):
                    d = dist[i, j]
                    r_sum = radii[i] + radii[j]
                    if d < r_sum:
                        # Overlap exists
                        # Force magnitude proportional to overlap
                        force_mag = k_overlap * (r_sum - d)
                        
                        # Direction from j to i
                        if d > 1e-9:
                            dir_vec = (centers[i] - centers[j]) / d
                        else:
                            dir_vec = np.random.rand(2) * 0.1 # Random push if centers coincide
                        
                        forces[i] += force_mag * dir_vec
                        forces[j] -= force_mag * dir_vec

            # 2. Boundary Repulsion
            # Left wall
            mask_left = centers[:, 0] < radii
            forces[mask_left, 0] += k_wall * (radii[mask_left] - centers[mask_left, 0])
            
            # Right wall
            mask_right = centers[:, 0] > 1 - radii
            forces[mask_right, 0] -= k_wall * (centers[mask_right, 0] - (1 - radii[mask_right]))
            
            # Bottom wall
            mask_bottom = centers[:, 1] < radii
            forces[mask_bottom, 1] += k_wall * (radii[mask_bottom] - centers[mask_bottom, 1])
            
            # Top wall
            mask_top = centers[:, 1] > 1 - radii
            forces[mask_top, 1] -= k_wall * (centers[mask_top, 1] - (1 - radii[mask_top]))
            
            # Update velocities and positions
            velocities += forces * dt
            velocities *= damping
            centers += velocities * dt
            
            # Hard clamp to keep centers roughly inside to prevent explosion
            centers = np.clip(centers, 0, 1)

        # Progress tracking (optional, silent)
        # if step % 200 == 0:
        #     print(f"Step {step}, Sum Radii: {np.sum(radii):.4f}")

    # --- 3. Local Optimization Refinement ---
    # Use scipy to fine-tune the positions and radii
    # We maximize sum(radii) -> minimize -sum(radii)
    # Subject to non-overlap and boundary constraints.
    # We use a penalty method or SLSQP. SLSQP handles constraints well.
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.ravel(), radii])
    
    def objective(vars_flat):
        # Maximize sum of radii -> minimize negative sum
        r = vars_flat[2::3]
        return -np.sum(r)
    
    def constraint_boundary(vars_flat):
        # x >= r, 1-x >= r, y >= r, 1-y >= r
        # => r - x <= 0, r - (1-x) <= 0, etc.
        # We return value >= 0 for scipy constraints.
        # So we return - (violation).
        # x - r >= 0 => x - r
        c = []
        for i in range(n):
            idx = i * 3
            x, y, r = vars_flat[idx], vars_flat[idx+1], vars_flat[idx+2]
            c.append(x - r)
            c.append((1 - x) - r)
            c.append(y - r)
            c.append((1 - y) - r)
        return np.array(c)

    def constraint_overlap(vars_flat):
        # dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        c = []
        coords = vars_flat[:2*n].reshape(n, 2)
        r = vars_flat[2*n:]
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((coords[i] - coords[j])**2)
                r_sum_sq = (r[i] + r[j])**2
                # dist_sq - r_sum_sq >= 0
                c.append(dist_sq - r_sum_sq)
        return np.array(c)

    # Define constraints for SLSQP
    # 'ineq' means function value >= 0
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    # Bounds: x, y in [0, 1], r >= 0 (and maybe <= 0.5)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # Run optimization
    # SLSQP might be slow with many constraints, but N=26 is manageable.
    # Providing gradient might help but finite diff is okay.
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            final_centers = res.x[:2*n].reshape(n, 2)
            final_radii = res.x[2*n:]
        else:
            # Fallback to simulation result if optimization fails
            final_centers = centers
            final_radii = radii
    except Exception:
        final_centers = centers
        final_radii = radii

    # Final validation and cleanup
    # Ensure radii are non-negative
    final_radii = np.maximum(final_radii, 0)
    
    # Recalculate sum
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
