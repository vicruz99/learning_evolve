import numpy as np
from scipy.optimize import minimize

def bound_constraints(x):
    """Enforces that circles stay within the unit square boundaries."""
    n = 26
    res = np.zeros(4 * n)
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        res[4*i] = xi - ri
        res[4*i+1] = 1.0 - xi - ri
        res[4*i+2] = yi - ri
        res[4*i+3] = 1.0 - yi - ri
    return res

def pair_constraints(x):
    """Enforces that no two circles overlap."""
    n = 26
    m = n * (n - 1) // 2
    res = np.zeros(m)
    idx = 0
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            dx = xi - xj
            dy = yi - yj
            res[idx] = np.sqrt(dx*dx + dy*dy) - (ri + rj)
            idx += 1
    return res

def objective(x):
    """Objective function: minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(x[2::3])

                centers[idx] = [x_start, y_curr]
                idx += 1

    # Add small perturbation to break symmetry and improve optimizer convergence
    np.random.seed(42)
    centers += np.random.uniform(-0.005, 0.005, centers.shape)

    return best_centers, best_radii, float(np.sum(best_radii))
