import numpy as np
from scipy.optimize import minimize


def compute_max_radii(centers, max_iter=2000):
    """Compute maximum feasible radii for given centers using Gauss-Seidel iteration."""
    n = centers.shape[0]
    radii = np.ones(n) * 0.001

    for iteration in range(max_iter):
        old_radii = radii.copy()
        for i in range(n):
            # Boundary constraints
            r = centers[i, 0]
            r = min(r, 1.0 - centers[i, 0])
            r = min(r, centers[i, 1])
            r = min(r, 1.0 - centers[i, 1])
            # Neighbor constraints
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = np.sqrt(dx * dx + dy * dy)
                    r = min(r, d - radii[j])
            radii[i] = max(0.0, r)

        if np.max(np.abs(radii - old_radii)) < 1e-13:
            break

    return radii


def objective_penalty(params, n=26, penalty_weight=1e8):
    """Objective with penalty for constraint violations. Smooth enough for gradient methods."""
    centers = params[:2 * n].reshape(-1, 2)
    radii = params[2 * n:]

    obj = -np.sum(radii)
    penalty = 0.0

    for i in range(n):
        # Boundary: x >= r
        if centers[i, 0] < radii[i]:
            penalty += (radii[i] - centers[i, 0]) ** 2
        # Boundary: x + r <= 1
        if centers[i, 0] + radii[i] > 1.0:
            penalty += (centers[i, 0] + radii[i] - 1.0) ** 2
        # Boundary: y >= r
        if centers[i, 1] < radii[i]:
            penalty += (radii[i] - centers[i, 1]) ** 2
        # Boundary: y + r <= 1
        if centers[i, 1] + radii[i] > 1.0:
            penalty += (centers[i, 1] + radii[i] - 1.0) ** 2

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            sum_r = radii[i] + radii[j]
            if d < sum_r:
                penalty += (sum_r - d) ** 2

    return obj + penalty_weight * penalty


def hexagonal_lattice(n, scale=1.0):
    """Generate hexagonal lattice positions for n circles."""
    positions = []
    y = 0.12 * scale
    row = 0
    while len(positions) < n:
        x_offset = 0.12 * scale if row % 2 == 1 else 0
        x = 0.1 * scale + x_offset
        while x < 0.9 * scale and len(positions) < n:
            positions.append([x, y])
            x += 0.22 * scale
        y += 0.19 * scale
        row += 1
    return np.array(positions[:n])


def grid_positions(n, spacing):
    """Generate grid positions for n circles."""
    centers = []
    y = spacing
    while y < 1.0 - spacing and len(centers) < n:
        x = spacing
        while x < 1.0 - spacing and len(centers) < n:
            centers.append([x, y])
            x += spacing
        y += spacing
    return np.array(centers[:n])


def random_positions(n, seed):
    """Generate random positions for n circles."""
    np.random.seed(seed)
    return np.random.uniform(0.15, 0.85, (n, 2))


def refine_with_exact_radii(centers, n=26):
    """Refine solution by computing exact maximum radii for optimized positions."""
    radii = compute_max_radii(centers, max_iter=5000)
    return centers, radii


def run_packing():
    n = 26

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Generate diverse initial configurations
    configs = []

    # Hexagonal lattices with various scales
    for scale in [0.85, 0.90, 0.93, 0.95, 0.97, 1.00, 1.03]:
        centers = hexagonal_lattice(n, scale)
        radii = np.ones(n) * 0.09
        params = np.concatenate([centers.flatten(), radii])
        configs.append(params)

    # Grid configurations
    for spacing in [0.15, 0.17, 0.19, 0.20, 0.21, 0.23]:
        centers = grid_positions(n, spacing)
        radii = np.ones(n) * spacing / 2.5
        params = np.concatenate([centers.flatten(), radii])
        configs.append(params)

    # Random configurations
    for seed in range(40):
        centers = random_positions(n, seed)
        radii = np.ones(n) * 0.05
        params = np.concatenate([centers.flatten(), radii])
        configs.append(params)

    # Corner-biased configurations: larger circles in corners
    for seed in range(20):
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        # Place some near corners
        corner_candidates = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
        for i in range(4):
            centers[i] = corner_candidates[i]
        # Place rest randomly
        remaining = n - 4
        centers[4:] = np.random.uniform(0.15, 0.85, (remaining, 2))
        radii = np.ones(n) * 0.08
        radii[:4] = 0.15
        params = np.concatenate([centers.flatten(), radii])
        configs.append(params)

    # Edge-biased configurations
    for seed in range(15):
        np.random.seed(seed)
        centers = np.zeros((n, 2))

        edge_positions = []

            edge = np.random.randint(4)
            if edge == 0:
                edge_positions.append([np.random.uniform(0.2, 0.8), 0.1])
            elif edge == 1:
                edge_positions.append([np.random.uniform(0.2, 0.8), 0.9])

            else:
                edge_positions.append([0.9, np.random.uniform(0.2, 0.8)])

        params = np.concatenate([centers.flatten(), radii])
        configs.append(params)

                bounds.append((0.0001, 0.499))  # Radius bounds

            centers_opt = result.x[:2 * n].reshape(-1, 2)
            radii_opt = result.x[2 * n:]

            radii_opt = np.maximum(radii_opt, 0.0)

            final_obj = objective_penalty(result.x, n, 1e8)
            penalty_only = final_obj + np.sum(radii_opt)

            if penalty_only < 0.01:  # Acceptable feasibility

            best_centers = centers_refined
            best_radii = radii_refined

        # Further position optimization with exact radii objective
        params = best_centers.flatten()

            c = centers_flat.reshape(-1, 2)
            r = compute_max_radii(c, max_iter=500)
            return -np.sum(r)

        result = minimize(
            neg_sum_exact,
            params,
            method='Nelder-Mead',

        centers_final = result.x.reshape(-1, 2)
        radii_final = compute_max_radii(centers_final, max_iter=3000)

        if final_sum > best_sum:

            best_centers = centers_final
            best_radii = radii_final

        params2 = best_centers.flatten()
        result2 = minimize(
            neg_sum_exact,

            method='Powell',

        centers_final2 = result2.x.reshape(-1, 2)

    # Final validation and cleanup

    if best_centers is not None:
        best_centers = np.clip(best_centers, 0.0, 1.0)
