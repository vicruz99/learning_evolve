import numpy as np
from scipy.optimize import minimize
from functools import partial
import random


def objective(vars_flat):
    """Negated sum of radii (we minimize negative to maximize sum)."""
    n = 26
    radii = vars_flat[2 * n:]
    return -np.sum(radii)


def boundary_constraint_left(vars_flat, i):
    """x_i - r_i >= 0"""
    return vars_flat[2 * i] - vars_flat[52 + i]


def boundary_constraint_right(vars_flat, i):
    """1 - x_i - r_i >= 0"""
    return 1.0 - vars_flat[2 * i] - vars_flat[52 + i]


def boundary_constraint_bottom(vars_flat, i):
    """y_i - r_i >= 0"""
    return vars_flat[2 * i + 1] - vars_flat[52 + i]


def boundary_constraint_top(vars_flat, i):
    """1 - y_i - r_i >= 0"""
    return 1.0 - vars_flat[2 * i + 1] - vars_flat[52 + i]


def overlap_constraint(vars_flat, i, j):
    """||c_i - c_j|| - r_i - r_j >= 0"""
    x1 = vars_flat[2 * i]
    y1 = vars_flat[2 * i + 1]
    x2 = vars_flat[2 * j]
    y2 = vars_flat[2 * j + 1]
    r1 = vars_flat[52 + i]
    r2 = vars_flat[52 + j]
    dx = x1 - x2
    dy = y1 - y2
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - r1 - r2


def make_hexagonal_centers(n):
    """Create a hexagonal grid initial placement."""
    centers = np.zeros((n, 2))
    idx = 0
    
    spacing = 0.20
    offset = 0.08
    
    for row in range(7):
        for col in range(5):
            if idx >= n:
                break
            x = offset + col * spacing + (row % 2) * spacing / 2
            y = offset + row * spacing * 0.866
            if x < 1.0 and y < 1.0:
                centers[idx] = [x, y]
                idx += 1
    
    while idx < n:
        centers[idx] = [0.5 + random.uniform(-0.1, 0.1), 0.5 + random.uniform(-0.1, 0.1)]
        centers[idx] = np.clip(centers[idx], 0.1, 0.9)
        idx += 1
    
    return centers


def make_corner_heavy_centers(n):
    """Create placement emphasizing corners for larger circles."""
    centers = np.zeros((n, 2))
    idx = 0
    
    corner_margin = 0.15
    corners = [
        [corner_margin, corner_margin],
        [1 - corner_margin, corner_margin],
        [corner_margin, 1 - corner_margin],
        [1 - corner_margin, 1 - corner_margin],
    ]
    for c in corners:
        centers[idx] = c
        idx += 1
    
    edge_margin = 0.08
    edge_positions = []
    for i in range(3):
        t = 0.25 + i * 0.25
        edge_positions.append([t, edge_margin])
        edge_positions.append([t, 1 - edge_margin])
        edge_positions.append([edge_margin, t])
        edge_positions.append([1 - edge_margin, t])
    
    for pos in edge_positions:
        if idx < n:
            centers[idx] = pos
            idx += 1
    
    while idx < n:
        row = (idx - 12) // 4
        col = (idx - 12) % 4
        x = 0.25 + col * 0.15
        y = 0.25 + row * 0.15
        centers[idx] = [x, y]
        idx += 1
    
    return centers


def make_random_centers(n):
    """Create random initial placement."""
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i] = [
            random.uniform(0.05, 0.95),
            random.uniform(0.05, 0.95),
        ]
    return centers


def run_optimization(centers_init, radii_init, maxiter=1500):
    """Run SLSQP optimization from given initial point."""
    n = 26
    
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((0.0, 0.5))
    
    constraints = []
    for i in range(n):
        constraints.append({
            'type': 'ineq',
            'fun': partial(boundary_constraint_left, i=i),
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(boundary_constraint_right, i=i),
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(boundary_constraint_bottom, i=i),
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(boundary_constraint_top, i=i),
        })
    
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': partial(overlap_constraint, i=i, j=j),
            })
    
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        constraints=constraints,
        bounds=bounds,
        options={
            'maxiter': maxiter,
            'ftol': 1e-14,
            'disp': False,
        },
    )
    
    centers_opt = result.x[:2 * n].reshape((n, 2))
    radii_opt = result.x[2 * n:]
    total = np.sum(radii_opt)
    
    return centers_opt, radii_opt, total, result.success
