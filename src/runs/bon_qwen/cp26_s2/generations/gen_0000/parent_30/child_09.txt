import numpy as np
from scipy.optimize import minimize


def constraint_fun(x, ctype, i, j=None):
    if ctype == 'ri_ge_0':
        return x[3 * i + 2]
    elif ctype == 'xi_ge_ri':
        return x[3 * i] - x[3 * i + 2]
    elif ctype == 'xi_le_1_minus_ri':
        return 1 - x[3 * i] - x[3 * i + 2]
    elif ctype == 'yi_ge_ri':
        return x[3 * i + 1] - x[3 * i + 2]
    elif ctype == 'yi_le_1_minus_ri':
        return 1 - x[3 * i + 1] - x[3 * i + 2]
    elif ctype == 'non_overlap':
        return (x[3 * i] - x[3 * j]) ** 2 + (x[3 * i + 1] - x[3 * j + 1]) ** 2 - (x[3 * i + 2] + x[3 * j + 2]) ** 2


def objective(x):
    n = len(x) // 3
    return -sum(x[3 * i + 2] for i in range(n))


def hexagonal_init(n, cols_pattern):
    num_rows = len(cols_pattern)
    x = np.zeros(3 * n)
    idx = 0

    max_cols = max(cols_pattern)
    r_max_h = 1.0 / (2 * max_cols)
    if num_rows > 1:
        r_max_v = 1.0 / ((num_rows - 1) * np.sqrt(3) + 2)
    else:
        r_max_v = 0.5
    r_est = min(r_max_h, r_max_v) * 0.8

    for row_idx in range(num_rows):
        ncols = cols_pattern[row_idx]
        y = r_est + row_idx * np.sqrt(3) * r_est
        y = max(r_est, min(1 - r_est, y))

        x_start = r_est + (row_idx % 2) * r_est
        x_step = 2 * r_est

        for col_idx in range(ncols):
            x_pos = x_start + col_idx * x_step
            x_pos = max(r_est, min(1 - r_est, x_pos))

            x[3 * idx] = x_pos
            x[3 * idx + 1] = y
            x[3 * idx + 2] = r_est
            idx += 1

    return x


def random_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.zeros(3 * n)

    for i in range(n):
        r = 0.02 + rng.rand() * 0.04
        x[3 * i + 2] = r
        x[3 * i] = r + rng.rand() * (1 - 2 * r)
        x[3 * i + 1] = r + rng.rand() * (1 - 2 * r)

    return x


def make_bounds(n):
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 0.5))
    return bounds


def slsqp_optimize(x0, n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_fun, 'args': ('ri_ge_0', i, None)})
        cons.append({'type': 'ineq', 'fun': constraint_fun, 'args': ('xi_ge_ri', i, None)})
        cons.append({'type': 'ineq', 'fun': constraint_fun, 'args': ('xi_le_1_minus_ri', i, None)})
        cons.append({'type': 'ineq', 'fun': constraint_fun, 'args': ('yi_ge_ri', i, None)})
        cons.append({'type': 'ineq', 'fun': constraint_fun, 'args': ('yi_le_1_minus_ri', i, None)})

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constraint_fun, 'args': ('non_overlap', i, j)})

    bounds = make_bounds(n)

    result = minimize(objective, x0, method='SLSQP', constraints=cons, bounds=bounds,
                      options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})

    return result


def optimize_radii_given_centers(centers, radii, n, iterations=80):
    radii = radii.copy()

    for _ in range(iterations):
        changed = False
        for i in range(n):
            xi, yi = centers[i]
            r_i = min(xi, 1 - xi, yi, 1 - yi)
            for j in range(n):
                if i != j:
                    dist = np.sqrt((xi - centers[j][0]) ** 2 + (yi - centers[j][1]) ** 2)
                    r_i = min(r_i, dist - radii[j])
            r_i = max(0, r_i)
            if abs(r_i - radii[i]) > 1e-12:
                radii[i] = r_i
                changed = True
        if not changed:
            break

    return radii


def optimize_positions_given_radii(centers, radii, n, lr=0.01, iterations=1000):
    centers = centers.copy()

    for it in range(iterations):
        forces = np.zeros((n, 2))
        max_overlap = 0

        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]

            if xi < ri:
                forces[i, 0] += (ri - xi) * 50
            elif xi > 1 - ri:
                forces[i, 0] -= (xi - (1 - ri)) * 50

            if yi < ri:
                forces[i, 1] += (ri - yi) * 50
            elif yi > 1 - ri:
                forces[i, 1] -= (yi - (1 - ri)) * 50

            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 0 else 1e-10
                min_dist = radii[i] + radii[j]

                if dist < min_dist:
                    overlap = min_dist - dist
                    max_overlap = max(max_overlap, overlap)
                    strength = overlap * 100 / (dist + 0.01)
                    fx = strength * dx / dist
                    fy = strength * dy / dist
                    forces[i] -= np.array([fx, fy])
                    forces[j] += np.array([fx, fy])

        decayed_lr = lr / (1 + it * 0.002)
        centers += decayed_lr * forces

        if max_overlap < 1e-10:
            break

    return centers


def alternating_optimize(centers, radii, n, outer_iters=30):
    centers = centers.copy()
    radii = radii.copy()

    for outer in range(outer_iters):
        radii = optimize_radii_given_centers(centers, radii, n, iterations=50)
        centers = optimize_positions_given_radii(centers, radii, n, lr=0.005, iterations=500)

    return centers, radii


def refine_local(centers, radii, n, steps=200):
    centers = centers.copy()
    radii = radii.copy()

    for _ in range(steps):
        improved = False
        for i in range(n):
            best_r = radii[i]
            r_try = radii[i] * 1.005
            r_try = min(r_try, centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])

            valid = True
            for j in range(n):
                if i != j:
                    dist = np.sqrt((centers[i][0] - centers[j][0]) ** 2 + (centers[i][1] - centers[j][1]) ** 2)
                    if dist < r_try + radii[j] - 1e-10:
                        valid = False
                        break

            if valid and r_try > best_r + 1e-12:
                radii[i] = r_try
                improved = True

        if not improved:
            break

    return centers, radii


def run_packing():
    n = 26

        [4, 7, 5, 7, 3],

    inits = []

            inits.append(hexagonal_init(n, pat))

        current_sum = sum(radii)

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    centers, radii = alternating_optimize(best_centers, best_radii, n, outer_iters=40)
    centers, radii = refine_local(centers, radii, n, steps=300)

    centers = np.array([[max(0, min(1, c[0])), max(0, min(1, c[1]))] for c in centers])
    radii = np.array([max(0, r) for r in radii])

    for i in range(n):
        radii[i] = min(radii[i], centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])

    final_sum = sum(radii)

    return centers, radii, final_sum
