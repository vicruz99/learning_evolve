5. **Multi-start** - Run from multiple initial configurations and keep the best result.

The key insight is that the radius optimization for fixed centers is convex (an LP), while the position optimization is non-convex. By alternating between the two, we can efficiently search for good solutions.

import numpy as np
from scipy.optimize import linprog


def make_row_pattern(counts):
    """Create a hexagonal lattice arrangement with given row counts."""
    n = sum(counts)
    centers = np.zeros((n, 2))
    idx = 0
    max_per_row = max(counts)
    
    # Calculate spacing
    dx = 0.85 / (max_per_row + 1)
    dy = dx * np.sqrt(3) / 2
    
    total_height = (len(counts) - 1) * dy
    y_start = (1 - total_height) / 2
    
    for row, count in enumerate(counts):
        y = y_start + row * dy
        x_start = (1 - count * dx) / 2
        for col in range(count):
            if row % 2 == 1:
                x = x_start + col * dx + dx / 2
            else:
                x = x_start + col * dx
            centers[idx] = [x, y]
            idx += 1
    
    return centers


def optimize_radii_lp(centers, n):
    """Solve LP to find optimal radii for given centers.
    
    Maximize sum(r_i)
    Subject to:
      r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i (boundary)
      r_i + r_j <= ||c_i - c_j|| for all i < j (non-overlap)
      r_i >= 0
    """
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    A_ub_rows = []
    b_ub_vals = []
    
    # Boundary constraints: r_i <= boundary_limit
    for i in range(n):
        b_i = min(
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1]
        )
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_vals.append(b_i)
    
    # Non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.sqrt(
                (centers[i, 0] - centers[j, 0]) ** 2 + 
                (centers[i, 1] - centers[j, 1]) ** 2
            )
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(d_ij)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: compute radii greedily
        radii = np.ones(n) * 0.02
        return radii


def compute_forces(centers, radii, n):
    """Compute repulsive forces for position optimization."""
    forces = np.zeros((n, 2))
    
    for i in range(n):
        # Boundary forces
        for dim in range(2):
            dist_to_left = centers[i, dim] - radii[i]
            dist_to_right = 1.0 - centers[i, dim] - radii[i]
            
            if dist_to_left < 0.08:
                forces[i, dim] += 0.5 / (dist_to_left + 0.005)
            if dist_to_right < 0.08:
                forces[i, dim] -= 0.5 / (dist_to_right + 0.005)
        
        # Inter-circle forces
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff ** 2))
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist + 0.05:
                if dist > 1e-10:
                    force_mag = max(0, (min_dist - dist + 0.03)) / (dist + 0.001)
                    forces[i] += diff * force_mag
                    forces[j] -= diff * force_mag
    
    return forces


def force_optimize(centers, n, n_iter=3000, init_temp=0.03):
    """Iterative force-based optimization with simulated annealing."""
    best_centers = centers.copy()
    best_radii = optimize_radii_lp(centers, n)
    best_sum = np.sum(best_radii)
    
    temperature = init_temp
    cooling = 0.9995
    
    for iteration in range(n_iter):
        temperature *= cooling
        
        # Compute optimal radii
        radii = optimize_radii_lp(best_centers, n)
        current_sum = np.sum(radii)
        
        # Compute forces
        forces = compute_forces(best_centers, radii, n)
        
        # Normalize forces
        force_norm = np.linalg.norm(forces, axis=1, keepdims=True)
        force_norm = np.maximum(force_norm, 1e-10)
        forces_normalized = forces / force_norm * np.minimum(force_norm, 0.1)
        
        # Update centers
        step = 0.008 / (1 + iteration * 0.0005)
        proposed = best_centers + forces_normalized * step
        proposed = np.clip(proposed, 0.002, 0.998)
        
        # Evaluate proposed
        proposed_radii = optimize_radii_lp(proposed, n)
        proposed_sum = np.sum(proposed_radii)
        
        # Accept/reject (simulated annealing)
        delta = proposed_sum - current_sum
        if delta > 0 or np.random.rand() < np.exp(delta / max(temperature, 1e-10)):
            best_centers = proposed
            if proposed_sum > best_sum:
                best_sum = proposed_sum
                best_radii = proposed_radii.copy()
    
    return best_centers, best_radii, best_sum


def try_optimization(counts, n_iter=3000, init_temp=0.03, seed=42):
    """Try optimization from a given row pattern."""
    np.random.seed(seed)
    centers = make_row_pattern(counts)
    return force_optimize(centers, sum(counts), n_iter, init_temp)
