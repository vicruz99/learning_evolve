        # Let's use explicit inequality constraints: x - r >= 0, 1 - r - x >= 0, etc.

            })

                'fun': lambda v, i=i: 1.0 - v[x_idx] - v[r_idx]

                'type': 'ineq',
                'fun': lambda v, i=i: v[y_idx] - v[r_idx]
            })

def is_valid_packing(x, n):
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Check boundaries
    for i in range(n):
        if radii[i] < -1e-9: return False
        if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1 - radii[i] + 1e-9: return False
        if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1 - radii[i] + 1e-9: return False
        
    # Check overlaps
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True
