import numpy as np
from scipy.optimize import linprog, minimize


def create_hexagonal_packing(n, row_counts=None):
    """Create hexagonal packing with given row counts."""
    if row_counts is None:
        row_counts = [5, 6, 5, 5, 5]
    
    centers = []
    r = 0.085
    
    for row_i, count in enumerate(row_counts):
        y = r + row_i * r * np.sqrt(3)
        if row_i % 2 == 0:
            x_start = r
        else:
            x_start = 2 * r
        
        for col_j in range(count):
            x = x_start + col_j * 2 * r
            centers.append([x, y])
    
    return np.array(centers[:n])


def create_grid_packing(n, rows, cols):
    """Create a regular grid packing."""
    centers = []
    x_spacing = 1.0 / (cols + 1)
    y_spacing = 1.0 / (rows + 1)
    
    count = 0
    for i in range(1, rows + 1):
        for j in range(1, cols + 1):
            if count >= n:
                break
            centers.append([j * x_spacing, i * y_spacing])
            count += 1
        if count >= n:
            break
    
    return np.array(centers[:n])


def compute_optimal_radii(centers):
    """Given fixed centers, compute optimal radii using LP.
    
    Maximize sum(r_i) subject to:
    - r_i + r_j <= dist(i,j) for all i < j
    - r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i
    - r_i >= 0
    """
    n = centers.shape[0]
    
    c = -np.ones(n)
    
    num_pair_constraints = n * (n - 1) // 2
    num_boundary_constraints = n
    num_constraints = num_pair_constraints + num_boundary_constraints
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    constraint_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A_ub[constraint_idx, i] = 1.0
            A_ub[constraint_idx, j] = 1.0
            b_ub[constraint_idx] = dist
            constraint_idx += 1
    
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[constraint_idx, i] = 1.0
        b_ub[constraint_idx] = max(bound, 1e-10)
        constraint_idx += 1
    
    bounds = [(0, None) for _ in range(n)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        radii = np.maximum(result.x, 1e-10)
        return radii
    else:
        return np.ones(n) * 0.05


def compute_objective_and_penalty(params, n):
    """Compute objective (negative sum of radii) and penalty for constraint violations."""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-10)
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        if x - r < 0:
            penalty += 500.0 * (x - r) ** 2
        if x + r > 1.0:
            penalty += 500.0 * (x + r - 1.0) ** 2
        if y - r < 0:
            penalty += 500.0 * (y - r) ** 2
        if y + r > 1.0:
            penalty += 500.0 * (y + r - 1.0) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += 500.0 * (min_dist - dist) ** 2
    
    return obj + penalty, obj, penalty


def objective_func(params, n):
    """Wrapper for objective function."""
    result, _, _ = compute_objective_and_penalty(params, n)
    return result


def optimize_positions(centers, radii, n, max_iter=1500):
    """Optimize positions given initial centers and radii using L-BFGS-B."""
    params0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
    for i in range(n):
        bounds.append((1e-8, 0.5))
    
    result = minimize(objective_func, params0, args=(n,),
                     method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(result.x[2 * n:], 1e-10)
    
    return centers_opt, radii_opt


def iterative_refinement(centers_init, radii_init, n, max_rounds=15):
    """Iteratively optimize positions and radii."""
    centers = centers_init.copy()
    radii = radii_init.copy()
    
    for round_idx in range(max_rounds):
        # Step 1: Optimize radii with LP
        radii = compute_optimal_radii(centers)
        
        # Step 2: Optimize positions
        centers, radii = optimize_positions(centers, radii, n, max_iter=800)
        
        # Step 3: Re-optimize radii with LP
        radii = compute_optimal_radii(centers)
        
        current_sum = np.sum(radii)
        if round_idx % 5 == 0:
            print(f"Round {round_idx}: sum_radii = {current_sum:.6f}")
    
    return centers, radii
