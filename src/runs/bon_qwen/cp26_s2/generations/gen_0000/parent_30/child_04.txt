def check_validity(centers, radii):
    """
    Helper to check if a packing is valid.
    Returns True if valid, False otherwise.
    """
    n = centers.shape[0]
    # Check bounds
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0: return False
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    
    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j]) ** 2)
            r_sum = radii[i] + radii[j]
            if dist_sq < (r_sum - 1e-9) ** 2:
                return False
    return True

def expand_packing(centers, radii, steps=2000, growth_rate=0.001):

def optimize_with_slsqp(centers, radii):

    Use scipy.optimize to fine-tune the packing.

    constraints = []

    # y - r >= 0

                return -(v[idx_x] - v[idx_r]) # -x + r <= 0 => r <= x. Wait. x >= r => x - r >= 0 => r - x <= 0.
                # Actually scipy expects function returning value <= 0 (ineq) or == 0 (eq).
                # Let's use inequality type 'ineq' which means fun(v) >= 0.
                # So x - r >= 0.
                pass 
            return c1
        
        # Using 'ineq': fun(v) >= 0
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[3*i] - v[3*i+2]
        })
        # 1 - x - r >= 0 => x + r <= 1
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]
        })

    # 2. Overlap constraints
    # dist_ij >= ri + rj
    # sqrt(...) - (ri + rj) >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = 3*i
            idx_yi = 3*i + 1
            idx_ri = 3*i + 2
            idx_xj = 3*j
            idx_yj = 3*j + 1

                    dy = v[idx_yi] - v[idx_yj]
                    dist = np.sqrt(dx*dx + dy*dy)

            constraints.append({
