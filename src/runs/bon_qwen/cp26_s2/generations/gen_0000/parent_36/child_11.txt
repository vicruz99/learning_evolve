

import numpy as np

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a force-directed simulation with multiplicative radius growth.
    """
    n = 26
    np.random.seed(42)
    
    # Initialize centers on a hexagonal-like grid pattern
    centers = np.zeros((n, 2))
    idx = 0
    # Configuration: 5, 4, 5, 4, 5, 3 circles per row (total 26)
    rows_cfg = [5, 4, 5, 4, 5, 3]
    for i, cnt in enumerate(rows_cfg):
        for j in range(cnt):
            x = (j + 0.5 * (i % 2)) * 2.0 / 10.0 + 0.1
            y = i * 1.0 / 6.0 + 0.1
            centers[idx] = [x, y]
            idx += 1
            
    # Add small random jitter to break symmetry and avoid local minima
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.ones(n) * 0.05
    
    # Simulation parameters
    dt = 0.005  # Movement step size
    max_iter = 50000
    
    for it in range(max_iter):
        # Compute pairwise distances
        # diff shape: (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, np.inf)
        
        # Compute overlaps: r_i + r_j - d_ij
        sum_r = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = np.maximum(sum_r - dist, 0.0)
        
        max_overlap = np.max(overlap)
        
        # Compute boundary violations
        # We want centers to stay within [r, 1-r]
        # Positive violation means push inward required
        viol_x = np.zeros(n)
        viol_x += np.maximum(radii - centers[:, 0], 0.0)      # Left wall
        viol_x -= np.maximum(centers[:, 0] + radii - 1.0, 0.0) # Right wall
        
        viol_y = np.zeros(n)
        viol_y += np.maximum(radii - centers[:, 1], 0.0)      # Bottom wall
        viol_y -= np.maximum(centers[:, 1] + radii - 1.0, 0.0) # Top wall
        
        max_viol = np.max(np.abs(np.concatenate([viol_x, viol_y])))
        
        if max_overlap < 1e-9 and max_viol < 1e-9:
            # No conflicts detected: grow radii multiplicatively for efficiency
            radii *= 1.0005
        else:
            # Resolve conflicts using repulsive forces
            safe_dist = np.where(dist < 1e-9, 1e-9, dist)
            dir_mat = diff / safe_dist[:, :, np.newaxis]
            # Force proportional to overlap along the line connecting centers
            force_mat = dir_mat * overlap[:, :, np.newaxis]
            forces = np.sum(force_mat, axis=1)
            
            # Add strong boundary correction forces
            forces[:, 0] += viol_x * 200.0
            forces[:, 1] += viol_y * 200.0
            
            # Update positions
            centers += forces * dt
            
            # Prevent numerical explosion
            centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
            
            # Gradually reduce step size to settle
            if it % 1000 == 0:
                dt *= 0.97
                
    # Final projection step to guarantee strict constraint satisfaction
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    # Minimum distance to boundaries
    min_d_bound = np.min(np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ]))
    
    # Minimum separation between circles
    sep = dist - radii[:, np.newaxis] - radii[np.newaxis, :]
    min_d_sep = np.min(sep)
    
    # Apply uniform shrinkage if needed
    slack = min(min_d_bound, min_d_sep)
    if slack < 0:
        radii += slack
        radii = np.maximum(radii, 1e-9)
        
    total_sum = np.sum(radii)
    return centers, radii, total_sum
