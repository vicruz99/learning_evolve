def compute_max_sum_radii(centers):

    Given a set of centers, computes the maximum sum of radii for non-overlapping circles
    inside the unit square using Linear Programming.
    """
    n = centers.shape[0]
    
    # Objective: maximize sum of radii -> minimize -sum of radii
    c_obj = np.ones(n) * -1
    
    # Bounds for radii: [0, max possible based on boundaries]
    # Boundary constraint: x - r >= 0 => r <= x
    #                     x + r <= 1 => r <= 1 - x
    #                     same for y
    r_upper = np.full(n, 1.0)
    for i in range(n):
        x, y = centers[i]
        dist_to_border = min(x, 1 - x, y, 1 - y)
        r_upper[i] = max(0.0, dist_to_border)
    
    bounds = [(0, r_upper[i]) for i in range(n)]
    
    # Constraints: r_i + r_j <= dist(i, j)
    # We construct the A_ub and b_ub matrices
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i + 1, n):

            b_ub.append(dist)

    b_ub = np.array(b_ub)
    
    # Solve LP
