

Looking at this circle packing problem, I need to maximize the sum of 26 circle radii in a unit square.

<strategy>
My strategy involves multiple approaches combined:

1. **Hexagonal Grid Initialization**: Start with a staggered hexagonal pattern which is known to be more efficient than square grids for circle packing.

2. **SLSQP Constrained Optimization**: Use scipy's SLSQP solver with proper constraints:
   - Non-overlap: distance(c_i, c_j) >= r_i + r_j for all pairs
   - Boundary: each circle must fit within [0,1]²
   - Objective: maximize sum of radii

3. **Multiple Random Starts**: Try different initial configurations (hexagonal + multiple random) to escape local optima and find a better global solution.

4. **Force-Based Refinement**: Apply a repulsive force method that can escape local optima where gradient-based methods might get stuck.

5. **Validation**: Ensure all constraints are satisfied before returning.

Key implementation details:
- Flatten (x, y, r) for all circles into a single optimization vector
- Define explicit functions for constraints (no lambdas per rules)
- Use default arguments in nested function definitions to properly capture loop indices
- Return the best result across all attempts
</strategy>

import numpy as np
from scipy.optimize import minimize


def hex_init(n):
    """Generate hexagonal grid initialization"""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)
    idx = 0
    row = 0
    while idx < n:
        y = 0.11 + row * 0.14
        if y > 0.92:
            break
        if row % 2 == 0:
            cols_in_row = 5
            x_start = 0.12
        else:
            cols_in_row = 4
            x_start = 0.22
        for col in range(cols_in_row):
            if idx >= n:
                break
            x = x_start + col * 0.17
            if x <= 0.92:
                centers[idx] = [x, y]
                idx += 1
        row += 1
    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.05
        idx += 1
    return centers, radii


def random_init(n):
    """Generate random initialization"""
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.random.uniform(0.05, 0.15, n)
    return centers, radii


def validate_solution(centers, radii, n):
    """Validate a packing solution"""
    if np.any(np.isnan(centers)) or np.any(np.isnan(radii)):
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def optimize_packing_slsqp(centers, radii, n):
    """Optimize using SLSQP constrained optimizer"""
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((1e-10, 0.5))
    
    def objective(x):
        return -np.sum(x[2 * n:])
    
    constraints = []
    
    for i in range(n):
        for j in range(i + 1, n):
            ii, jj = i, j
            def make_overlap(i_idx, j_idx):
                def constraint_func(x):
                    ci = np.array([x[2 * i_idx], x[2 * i_idx + 1]])
                    cj = np.array([x[2 * j_idx], x[2 * j_idx + 1]])
                    ri = x[2 * n + i_idx]
                    rj = x[2 * n + j_idx]
                    dist = np.sqrt(np.sum((ci - cj) ** 2))
                    return dist - ri - rj
                return constraint_func
            constraints.append({'type': 'ineq', 'fun': make_overlap(ii, jj)})
    
    for i in range(n):
        ii = i
        def make_left(i_idx):
            def constraint_func(x):
                return x[2 * i_idx] - x[2 * n + i_idx]
            return constraint_func
        constraints.append({'type': 'ineq', 'fun': make_left(ii)})
        
        def make_right(i_idx):
            def constraint_func(x):
                return 1 - x[2 * i_idx] - x[2 * n + i_idx]
            return constraint_func
        constraints.append({'type': 'ineq', 'fun': make_right(ii)})
        
        def make_bottom(i_idx):
            def constraint_func(x):
                return x[2 * i_idx + 1] - x[2 * n + i_idx]
            return constraint_func
        constraints.append({'type': 'ineq', 'fun': make_bottom(ii)})
        
        def make_top(i_idx):
            def constraint_func(x):
                return 1 - x[2 * i_idx + 1] - x[2 * n + i_idx]
            return constraint_func
        constraints.append({'type': 'ineq', 'fun': make_top(ii)})
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-10})
    
    if result.success:
        opt_centers = result.x[:2 * n].reshape(n, 2)
        opt_radii = result.x[2 * n:]
        total = np.sum(opt_radii)
        if validate_solution(opt_centers, opt_radii, n):
            return opt_centers, opt_radii, total
    
    return None, None, 0.0


def force_based_optimization(centers, radii, n, max_iter=8000):
    """Force-based optimization with repulsion and growth"""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(max_iter):
        step_size = 0.003 * (1.0 / (1.0 + iteration * 0.001))
        
        dx = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                dvec = centers[i] - centers[j]
                dist = np.sqrt(np.sum(dvec ** 2))
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-12:
                    overlap = (min_dist - dist) / dist
                    force = overlap * 0.5
                    direction = dvec / dist
                    dx[i] += force * direction
                    dx[j] -= force * direction
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                dx[i, 0] += (0 - x + r) * 3
            if x + r > 1:
                dx[i, 0] -= (x + r - 1) * 3
            if y - r < 0:
                dx[i, 1] += (0 - y + r) * 3
            if y + r > 1:
                dx[i, 1] -= (y + r - 1) * 3
        
        for i in range(n):
            centers[i] += step_size * dx[i]
            centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
        
        if iteration % 10 == 0 and iteration > 0:
            has_overlap = False
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    if dist < radii[i] + radii[j]:
                        has_overlap = True
                        break
                if has_overlap:
                    break
            
            if not has_overlap:
                growth = 0.0003 * (1.0 - iteration / max_iter)
                for i in range(n):
                    max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
                    radii[i] = min(radii[i] + growth, max_r)
    
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = min(radii[i], max_r - 1e-9)
    
    total = np.sum(radii)
    return centers, radii, total


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_total = 0.0
    
    hex_centers, hex_radii = hex_init(n)
    fc, fr, ft = force_based_optimization(hex_centers, hex_radii, n, max_iter=6000)
    if validate_solution(fc, fr, n) and ft > best_total:
        best_centers = fc.copy()
        best_radii = fr.copy()
        best_total = ft
    
    opt_c, opt_r, opt_t = optimize_packing_slsqp(fc, fr, n)
    if opt_t > best_total:
        best_centers = opt_c.copy()
        best_radii = opt_r.copy()
        best_total = opt_t
    
    for attempt in range(8):
        rc, rr = random_init(n)
        fc2, fr2, ft2 = force_based_optimization(rc, rr, n, max_iter=5000)
        if validate_solution(fc2, fr2, n) and ft2 > best_total:
            best_centers = fc2.copy()
            best_radii = fr2.copy()
            best_total = ft2
        
        opt_c2, opt_r2, opt_t2 = optimize_packing_slsqp(fc2, fr2, n)
        if opt_t2 > best_total:
            best_centers = opt_c2.copy()
            best_radii = opt_r2.copy()
            best_total = opt_t2
    
    oc2, or2, ot2 = optimize_packing_slsqp(best_centers, best_radii, n)
    if ot2 > best_total:
        best_centers = oc2
        best_radii = or2
        best_total = ot2
    
    if best_centers is None:
        best_centers, best_radii = hex_init(n)
        best_total = np.sum(best_radii)
    
    return best_centers, best_radii, best_total
