import numpy as np
from scipy.optimize import minimize


def generate_hexagonal_config(n, num_rows, base_spacing_x, base_spacing_y, offset_pattern='stagger'):
    """Generate a hexagonal-like initial configuration."""
    centers = []
    
    for row in range(num_rows):
        # Determine number of circles in this row
        if offset_pattern == 'stagger':
            is_odd = row % 2 == 1
        else:
            is_odd = False
        
        # Compute horizontal offset for staggered rows
        h_offset = base_spacing_x / 2 if is_odd else 0
        v_offset = row * base_spacing_y
        
        # Determine how many circles fit in this row
        num_in_row = n // num_rows + (1 if row < n % num_rows else 0)
        
        for col in range(num_in_row):
            x = base_spacing_x / 2 + col * base_spacing_x + h_offset
            y = base_spacing_y / 2 + v_offset
            centers.append([x, y])
    
    return np.array(centers[:n])


def compute_forces(centers, radii, repulsion_k=500, boundary_k=100):
    """Compute repulsive forces between circles and boundary forces."""
    n = len(centers)
    forces = np.zeros((n, 2))
    
    # Repulsion between overlapping circles
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[j] - centers[i]
            dist_sq = np.sum(diff ** 2)
            dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
            
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                # Force proportional to overlap, inversely proportional to distance
                force_mag = repulsion_k * overlap
                forces[i] -= force_mag * diff / dist
                forces[j] += force_mag * diff / dist
    
    # Boundary repulsion (push circles away from walls)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        margin = 0.01  # Extra margin
        
        if x - r < margin:
            forces[i][0] += boundary_k * (margin - (x - r))
        if x + r > 1 - margin:
            forces[i][0] -= boundary_k * ((x + r) - (1 - margin))
        if y - r < margin:
            forces[i][1] += boundary_k * (margin - (y - r))
        if y + r > 1 - margin:
            forces[i][1] -= boundary_k * ((y + r) - (1 - margin))
    
    return forces


def force_directed_optimization(centers, radii, n, max_iter=10000):
    """Run force-directed optimization to grow radii and resolve overlaps."""
    centers = centers.copy()
    radii = radii.copy()
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    learning_rate = 0.005
    growth_rate = 0.00008
    
    for step in range(max_iter):
        forces = compute_forces(centers, radii)
        
        # Update positions
        centers += learning_rate * forces
        
        # Clamp to valid region
        centers = np.clip(centers, radii[:, None] + 1e-10, 1 - radii[:, None] - 1e-10)
        
        # Grow radii slowly
        radii += growth_rate
        
        # Track best
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_centers = centers.copy()
            best_radii = radii.copy()
            best_sum = current_sum
        
        # Decay rates
        learning_rate *= 0.9997
        growth_rate *= 0.9999
    
    return best_centers, best_radii, best_sum


def penalty_optimize(centers_init, radii_init, n, penalty_weight=5000):
    """Optimize using penalty method with scipy."""
    
    def objective(params):
        c = params[:2 * n].reshape(n, 2)
        r = params[2 * n:]
        
        # Objective: maximize sum of radii
        obj = -np.sum(r)
        
        # Smooth overlap penalty
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                req = r[i] + r[j]
                if req > dist:
                    overlap = req - dist
                    obj += penalty_weight * overlap ** 2
        
        # Smooth boundary penalty
        for i in range(n):
            x, y = c[i]
            ri = r[i]
            for val in [x - ri, 1 - x - ri, y - ri, 1 - y - ri]:
                if val < 0:
                    obj += penalty_weight * val ** 2
        
        # Penalty for negative radii
        for i in range(n):
            if r[i] < 0:
                obj += penalty_weight * r[i] ** 2
        
        return obj
    
    params0 = np.concatenate([centers_init.flatten(), radii_init])
    
    result = minimize(
        objective, params0, method='Nelder-Mead',
        options={
            'maxiter': 100000,
            'xatol': 1e-12,
            'fatol': 1e-12,
            'adaptive': True
        }
    )
    
    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]
    
    return centers, radii, result.fun


def validate_and_clean(centers, radii, n):
    """Validate and clean up a packing configuration."""
    # Ensure non-negative radii
    radii = np.maximum(radii, 0)
    
    # Ensure centers are within valid range
    centers = np.clip(centers, radii[:, None], 1 - radii[:, None])
    
    # Iteratively resolve any remaining overlaps
    for _ in range(500):
        max_overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                req = radii[i] + radii[j]
                if dist < req and dist > 1e-15:
                    overlap = req - dist
                    max_overlap = max(max_overlap, overlap)
                    # Push apart
                    push = overlap / 2
                    nx, ny = dx / dist, dy / dist
                    centers[i, 0] -= push * nx
                    centers[i, 1] -= push * ny
                    centers[j, 0] += push * nx
                    centers[j, 1] += push * ny
        
        # Clamp to valid range
        centers = np.clip(centers, radii[:, None], 1 - radii[:, None])
        
        if max_overlap < 1e-14:
            break
    
    return centers, radii
