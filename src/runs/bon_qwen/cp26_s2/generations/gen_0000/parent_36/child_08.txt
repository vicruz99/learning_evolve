import numpy as np
from scipy.optimize import minimize

def compute_objective(vars_, N):
    """
    Computes the negative sum of radii plus penalty terms for constraint violations.
    Minimizing this function maximizes the sum of radii while respecting boundaries and overlaps.
    """
    centers = vars_[:2*N].reshape(N, 2)
    radii = vars_[2*N:]
    
    # Wall constraints: circles must be inside [0,1]x[0,1]
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    pen = np.sum(np.maximum(0.0, r - x)**2)
    pen += np.sum(np.maximum(0.0, r + x - 1.0)**2)
    pen += np.sum(np.maximum(0.0, r - y)**2)
    pen += np.sum(np.maximum(0.0, r + y - 1.0)**2)
    
    # Overlap constraints: distance between centers must be >= sum of radii
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    diff = c1 - c2
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Only check each pair once
    i_idx, j_idx = np.triu_indices(N, k=1)
    violations = np.maximum(0.0, r_sum[i_idx, j_idx] - dists[i_idx, j_idx])
    pen += np.sum(violations**2)
    
    mu = 1000.0
    return -np.sum(radii) + mu * pen

def run_packing():
    N = 26
    best_val = np.inf
    best_vars = None
    
    # Run optimization from multiple starting points to avoid poor local minima
    for seed in range(5):
        np.random.seed(seed)
        
        # Initialize with a perturbed hexagonal-like grid
        centers_init = np.zeros((N, 2))
        idx = 0
        for i in range(5):
            for j in range(6):
                if idx < N:
                    centers_init[idx, 0] = 0.1 + i * 0.18 + (0.09 if j % 2 == 1 else 0.0)
                    centers_init[idx, 1] = 0.1 + j * 0.15
                    idx += 1
        
        # Add small random noise to break symmetry and aid exploration
        centers_init += np.random.uniform(-0.02, 0.02, size=centers_init.shape)
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        radii_init = np.full(N, 0.05)
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 1.0)] * N
        
        res = minimize(
            compute_objective, 
            x0, 
            args=(N,), 
            method='L-BFGS-B', 
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
        )
        
        if res.fun < best_val:
            best_val = res.fun
            best_vars = res.x
            
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    
    # Final numerical safety clipping
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.clip(radii, 1e-6, 1.0)
    
    return centers, radii, np.sum(radii)
