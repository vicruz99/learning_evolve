import numpy as np
from scipy.optimize import minimize

def smooth_min_objective(x_flat, k=100.0):
    """Objective function to minimize (negative sum of radii) using smooth approximation."""
    n = 26
    centers = x_flat.reshape(n, 2)
    
    # Distances to boundaries
    d_bound = np.minimum(centers, 1.0 - centers)
    min_bound = np.min(d_bound, axis=1)
    
    # Pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d_pair = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(d_pair, np.inf)
    min_pair = np.min(d_pair, axis=1) / 2.0
    
    # Stack constraints for each circle
    vals = np.stack([min_bound, min_pair], axis=1)
    
    # Smooth min approximation: -1/k * log(sum(exp(-k * v)))
    shifted = -k * vals
    m = np.min(shifted, axis=1, keepdims=True)
    log_sum = np.log(np.sum(np.exp(shifted - m), axis=1, keepdims=True))
    smooth_r = (m + log_sum) / k
    
    return -np.sum(smooth_r)

def compute_exact_radii(centers):
    """Compute exact maximum radii given fixed centers."""
    n = centers.shape[0]
    d_bound = np.minimum(centers, 1.0 - centers)
    min_bound = np.min(d_bound, axis=1)
    
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d_pair = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(d_pair, np.inf)
    min_pair = np.min(d_pair, axis=1) / 2.0
    
    return np.minimum(min_bound, min_pair)
