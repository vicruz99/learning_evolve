import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    Uses a hexagonal lattice initialization followed by SLSQP optimization.
    """
    n = 26
    
    # 1. Initialization: Hexagonal Layout
    # Pattern: 6, 5, 6, 5, 4
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Target radius for initialization (fits 6 circles in width with staggered rows)
    r_init = 0.09 
    
    idx = 0
    row_counts = [6, 5, 6, 5, 4]
    vertical_step = np.sqrt(3) * r_init
    
    for row_idx, count in enumerate(row_counts):
        # Horizontal spacing
        # For the staggered rows (count=5), we shift by r_init to nestle in gaps
        shift = r_init if row_idx % 2 == 1 else 0
        
        # Y-coordinate for the row center
        y = r_init + row_idx * vertical_step
        
        # Distribute 'count' circles evenly in X within [shift, 1-shift]
        # To mimic hex packing, we space them by 2*r_init
        # We place them centered or just spread evenly.
        # For a robust start, linear spacing works well.
        
        x_start = shift
        x_end = 1.0 - shift
        
        if count > 0:
            # Generate x coordinates
            # We use linspace to distribute them evenly in the available space
            # However, strict hex spacing is 2*r. 
            # Let's just spread them to ensure no initial overlap.
            xs = np.linspace(x_start, x_end, count)
            
            # Ensure we don't exceed boundaries slightly due to float errors
            xs = np.clip(xs, r_init, 1 - r_init)
            
            centers[idx : idx + count, 0] = xs
            centers[idx : idx + count, 1] = y
            radii[idx : idx + count] = r_init
            idx += count

    # 2. Optimization Setup
    # Variables: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
        
    # Constraints
    cons = []
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(n):
        i_idx = 3 * i
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[i_idx] - x[i_idx+2]}) # x >= r
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i_idx] - x[i_idx+2]}) # x <= 1-r
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[i_idx+1] - x[i_idx+2]}) # y >= r
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i_idx+1] - x[i_idx+2]}) # y <= 1-r

    # Overlap constraints: dist(ci, cj) >= ri + rj
    # We use Euclidean distance constraint
    for i in range(n):
        for j in range(i + 1, n):
            i_idx = 3 * i
            j_idx = 3 * j
            cons.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: 
                    np.sqrt((x[3*i] - x[3*j])**2 + (x[3*i+1] - x[3*j+1])**2) - x[3*i+2] - x[3*j+2]
            })

    # Objective: Maximize sum of radii -> Minimize -sum(radii)
    def objective(x):
        r = x[2::3]
        return -np.sum(r)

    # 3. Run Optimization
    # SLSQP is suitable for constrained non-linear optimization
    try:
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons, 
            options={
                'maxiter': 2000, 
                'ftol': 1e-9,
                'disp': False
            }
        )
        x_opt = res.x
    except Exception as e:
        # Fallback to initial guess if optimization fails
        x_opt = x0

    # 4. Extract and Validate Results
    centers_opt = x_opt[:3*n].reshape((n, 3))[:, :2]
    radii_opt = x_opt[2::3]
    
    # Ensure non-negative radii (safety clamp)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
