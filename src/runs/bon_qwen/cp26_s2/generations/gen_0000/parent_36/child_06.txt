import numpy as np

def simulate_packing(n, centers, radii, seed):
    """
    Runs a grow-and-push simulation to maximize radii sum.
    """
    np.random.seed(seed)
    growth_step = 1.2e-3
    push_iters = 12
    
    for step in range(3000):
        # Decay growth rate over time
        growth = growth_step / (1.0 + step / 600.0)
        radii += growth
        
        # Push overlapping circles apart and enforce boundaries
        for _ in range(push_iters):
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist = np.hypot(dx, dy)
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist and dist > 1e-9:
                        overlap = min_dist - dist
                        nx = dx / dist
                        ny = dy / dist
                        # Push apart equally
                        centers[i, 0] -= nx * overlap * 0.5
                        centers[i, 1] -= ny * overlap * 0.5
                        centers[j, 0] += nx * overlap * 0.5
                        centers[j, 1] += ny * overlap * 0.5
                        
                # Enforce square boundaries
                r = radii[i]
                centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
                centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
                
        # Add jitter periodically to escape local minima
        if step % 40 == 0 and step > 0:
            jitter = 1e-3 / (1.0 + step / 400.0)
            centers += np.random.uniform(-jitter, jitter, size=centers.shape)
            for i in range(n):
                r = radii[i]
                centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
                centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
                
    return centers, radii
