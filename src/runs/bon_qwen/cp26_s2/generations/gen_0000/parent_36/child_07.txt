import numpy as np
from scipy.optimize import minimize, linprog


def objective(params, n):
    radii = params[2 * n:]
    return -np.sum(radii)


def boundary_x_ge_r(params, i, n):
    return params[2 * i] - params[2 * n + i]


def boundary_x_le_1mr(params, i, n):
    return 1 - params[2 * i] - params[2 * n + i]


def boundary_y_ge_r(params, i, n):
    return params[2 * i + 1] - params[2 * n + i]


def boundary_y_le_1mr(params, i, n):
    return 1 - params[2 * i + 1] - params[2 * n + i]


def non_overlap(params, i, j, n):
    centers_i = params[2 * i : 2 * i + 2]
    centers_j = params[2 * j : 2 * j + 2]
    dist = np.sqrt(np.sum((centers_i - centers_j) ** 2))
    return dist - params[2 * n + i] - params[2 * n + j]


def generate_hexagonal_init(n, spacing=0.12):
    centers = np.zeros((n, 2))
    idx = 0

    y = spacing
    row = 0
    while idx < n and y <= 1 - spacing:
        x_offset = spacing * np.sqrt(3) / 2 if row % 2 == 1 else 0
        x = spacing + x_offset
        while x <= 1 - spacing and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += spacing * np.sqrt(3)
        y += spacing * 1.5
        row += 1

    while idx < n:
        centers[idx] = [0.5, 0.5 + 0.02 * (idx - n + 1)]
        idx += 1

    return centers


def generate_random_init(n, seed=42):
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.15, 0.85, (n, 2))
    return centers


def optimize_radii_lp(centers, n):
    c = -np.ones(n)

    A_ub = []
    b_ub = []

    for i in range(n):
        A_ub.append(np.eye(n)[i])
        b_ub.append(centers[i, 0])

        A_ub.append(np.eye(n)[i])
        b_ub.append(1 - centers[i, 0])

        A_ub.append(np.eye(n)[i])
        b_ub.append(centers[i, 1])

        A_ub.append(np.eye(n)[i])
        b_ub.append(1 - centers[i, 1])

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")

    if result.success:
        return result.x
    else:
        return np.ones(n) * 0.01


def build_constraints(n):
    cons = []
    for i in range(n):
        cons.append({"type": "ineq", "fun": boundary_x_ge_r, "args": (i, n)})
        cons.append({"type": "ineq", "fun": boundary_x_le_1mr, "args": (i, n)})
        cons.append({"type": "ineq", "fun": boundary_y_ge_r, "args": (i, n)})
        cons.append({"type": "ineq", "fun": boundary_y_le_1mr, "args": (i, n)})

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({"type": "ineq", "fun": non_overlap, "args": (i, j, n)})

    return cons


def build_bounds(n):
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    return bounds


def run_optimization(n, params_init, cons, bounds):
    try:
        result = minimize(
            objective,
            params_init,
            args=(n,),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 8000, "ftol": 1e-15, "disp": False},
        )
        return result
    except Exception:
        return None


def run_packing():

    bounds = build_bounds(n)

            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = result.x[: 2 * n].reshape((n, 2))
                best_radii = result.x[2 * n :]

    if best_centers is not None:

        best_sum = np.sum(best_radii)

        for iteration in range(3):

                    best_centers = new_centers

    return best_centers, best_radii, best_sum
