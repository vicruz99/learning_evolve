import numpy as np

def compute_distances(centers):
    """Compute pairwise Euclidean distances between centers."""
    n = centers.shape[0]
    # Use broadcasting for efficiency
    # centers is (n, 2)
    # diff is (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    return dists

def check_boundary_constraints(centers, radii):
    """
    Check if circles are inside the unit square.
    Returns a mask of violated constraints and the amount of violation.
    """
    n = centers.shape[0]
    violations = np.zeros(n)
    
    # Left wall: x - r < 0 => violation r - x
    if np.any(centers[:, 0] - radii < 0):
        violations += np.maximum(0, radii - centers[:, 0])
        
    # Right wall: x + r > 1 => violation x + r - 1
    if np.any(centers[:, 0] + radii > 1):
        violations += np.maximum(0, centers[:, 0] + radii - 1)
        
    # Bottom wall: y - r < 0 => violation r - y
    if np.any(centers[:, 1] - radii < 0):
        violations += np.maximum(0, radii - centers[:, 1])
        
    # Top wall: y + r > 1 => violation y + r - 1
    if np.any(centers[:, 1] + radii > 1):
        violations += np.maximum(0, centers[:, 1] + radii - 1)
        
    return violations

def resolve_overlaps_and_boundaries(centers, radii, steps=50, force_strength=10.0, wall_strength=10.0):
    """
    Iteratively resolve overlaps and boundary violations using forces.
    """
    n = centers.shape[0]
    
    for _ in range(steps):

            forces[too_close_left, 0] += push[too_close_left]

            push = wall_strength * (centers[:, 1] + radii - 1)
            forces[too_close_top, 1] -= push[too_close_top]

        # But we need to push apart even if just touching to allow radius growth?
        # Actually, for resolution, we just need to fix overlaps.
        
        # Vectorized distance computation
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15) # Avoid div by zero
        
        # Mask for lower triangle
        mask = np.tril(np.ones((n, n), dtype=bool), k=-1)
        
        # Find overlapping pairs
        sum_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = sum_radii - dists
        
        # Only consider positive overlaps
        active_overlap = np.maximum(0, overlap)
        active_overlap = active_overlap * mask
        
        # Calculate forces
        # Force is proportional to overlap, directed along the line connecting centers
        # F = k * overlap * (diff / dist)
        # Note: diff vector points from j to i (row i - row j)
        # If i and j overlap, i should move away from j (direction diff)
        
        # Normalize diff
        # Handle zero distance
        inv_dists = np.where(dists > 1e-12, 1.0 / dists, 0.0)
        unit_diff = diff * inv_dists[:, :, np.newaxis]
        
        # Force magnitude
        f_mag = force_strength * active_overlap
        
        # Apply force to centers
        # Sum of forces on i from all j
        # force_on_i = sum_j (f_mag_ij * unit_diff_ij)
        forces += np.sum(f_mag[:, :, np.newaxis] * unit_diff, axis=1)
        
        # Update centers
        # Simple Euler step
        dt = 0.05 / (1 + np.max(np.abs(forces))) # Adaptive step size
        centers += dt * forces
        
        # Clip centers to valid range (loosely, to prevent explosion)

        centers[:, 0] = np.clip(centers[:, 0], 0.0, 1.0)
        centers[:, 1] = np.clip(centers[:, 1], 0.0, 1.0)

    return centers
