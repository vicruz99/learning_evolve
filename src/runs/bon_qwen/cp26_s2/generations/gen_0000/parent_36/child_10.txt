import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def generate_initial_packing():
    """Generates a valid, dense initial packing using a hexagonal grid and expansion."""
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.1)
    idx = 0
    
    # Hexagonal-ish layout: 5 rows with counts summing to 26
    row_counts = [5, 5, 5, 5, 6]
    y = 0.1
    for r_i, count in enumerate(row_counts):
        x = 0.1 + (r_i % 2) * 0.05
        for _ in range(count):
            centers[idx] = [x, y]
            idx += 1
            x += 0.2
        y += 0.2

    # Scale to comfortably fit inside [0,1]
    max_extent = max(np.max(centers[:, 0]), np.max(centers[:, 1]))
    scale = 0.85 / max_extent
    centers *= scale
    radii *= scale

    # Iterative expansion to tighten the packing
    for _ in range(1200):
        radii *= 1.001
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                req = radii[i] + radii[j]
                if dist < req and dist > 1e-9:
                    push = (req - dist) / (2.0 * dist)
                    centers[i, 0] += dx * push
                    centers[i, 1] += dy * push
                    centers[j, 0] -= dx * push
                    centers[j, 1] -= dy * push
            
            # Enforce boundaries strictly
            for k in range(2):
                if centers[i, k] - radii[i] < 0:
                    centers[i, k] = radii[i]
                if centers[i, k] + radii[i] > 1:
                    centers[i, k] = 1.0 - radii[i]
    return centers, radii

def obj_func(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2 * N_CIRCLES:])

def cons_func(x):
    """Returns array of constraint values that must be >= 0."""
    n = N_CIRCLES
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, same for y
    c.append(centers[:, 0] - radii)
    c.append(1.0 - centers[:, 0] - radii)
    c.append(centers[:, 1] - radii)
    c.append(1.0 - centers[:, 1] - radii)
    # Radii non-negativity
    c.append(radii)
    
    # Pairwise non-overlap: dist(i,j) - (r_i + r_j) >= 0
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    req = radii[:, None] + radii[None, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.append(dists[mask] - req[mask])
    
    return np.concatenate(c)

def run_packing():
    centers0, radii0 = generate_initial_packing()
    x0 = np.concatenate([centers0.flatten(), radii0])
    
    cons = {'type': 'ineq', 'fun': cons_func}
    bounds = [(0, 1)] * (2 * N_CIRCLES) + [(0, 1)] * N_CIRCLES
    
    result = minimize(obj_func, x0, method='COBYLA', constraints=cons,
                      bounds=bounds, options={'maxiter': 18000, 'rhobeg': 0.1, 'rhoend': 1e-7})
    
    if result.success:
        cx = result.x[:2 * N_CIRCLES].reshape((N_CIRCLES, 2))
        r = result.x[2 * N_CIRCLES:]
    else:
        cx = centers0
        r = radii0
        
    return cx, r, np.sum(r)
