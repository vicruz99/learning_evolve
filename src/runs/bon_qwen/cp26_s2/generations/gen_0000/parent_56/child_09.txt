import numpy as np
from scipy.optimize import minimize

def obj_func(z, N):
    """Objective function: negative sum of radii (to maximize sum)"""
    return -np.sum(z[2::3])

def constr_func(z, N):
    """Constraint function: returns array of constraint values (must be >= 0)"""
    centers = z[:2*N].reshape(N, 2)
    radii = z[2::3]
    cons = []
    
    # Box constraints: circles must be inside [0,1]x[0,1]
    for i in range(N):
        ri = radii[i]
        cons.append(centers[i, 0] - ri)          # x >= r
        cons.append(1.0 - centers[i, 0] - ri)     # x + r <= 1
        cons.append(centers[i, 1] - ri)          # y >= r
        cons.append(1.0 - centers[i, 1] - ri)     # y + r <= 1
        
    # Pairwise non-overlap constraints: dist^2 >= (ri + rj)^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    rad_sum_sq = (radii[:, np.newaxis] + radii[np.newaxis, :])**2
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    cons.extend(dist_sq[mask] - rad_sum_sq[mask])
    
    return np.array(cons)

def run_packing():
    N = 26
    # Initial configuration: hexagonal grid
    # Using a slightly smaller radius to ensure strict feasibility initially
    r0 = 0.075
    z0 = np.zeros(3*N)
    row_counts = [5, 5, 6, 5, 5]
    y = r0
    idx = 0
    for row_idx, count in enumerate(row_counts):
        for k in range(count):
            if row_idx % 2 == 0:
                x = r0 + k * (2 * r0)
            else:
                x = r0 + r0 + k * (2 * r0)
            z0[idx] = x
            z0[idx+1] = y
            z0[idx+2] = r0
            idx += 1
        y += np.sqrt(3) * r0
        
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (3*N)
    for i in range(N):
        bounds[3*i+2] = (0.0, 0.5)
        
    cons = {'type': 'ineq', 'fun': constr_func, 'args': (N,)}
    
    best_res = None
    best_sum = -1.0
    
    # Multi-start optimization to improve chances of finding global optimum
    for seed in range(5):
        np.random.seed(100 + seed)
        z_init = z0.copy()
        # Add small random perturbation to positions
        z_init[:2*N] += np.random.uniform(-0.003, 0.003, (2*N,))
        
        # Ensure initial point satisfies box constraints strictly
        for i in range(N):
            ri = z_init[3*i+2]
            z_init[3*i] = np.clip(z_init[3*i], ri, 1.0 - ri)
            z_init[3*i+1] = np.clip(z_init[3*i+1], ri, 1.0 - ri)
            
        try:
            res = minimize(obj_func, z_init, args=(N,), method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 3000, 'ftol': 1e-11, 'disp': False})
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_res = res
        except Exception:
            pass
            
    if best_res is not None:
        z_opt = best_res.x
        centers = z_opt[:2*N].reshape((N, 2))
        radii = z_opt[2::3]
        return centers, radii, np.sum(radii)
    else:
        # Fallback to initial guess
        return z0[:2*N].reshape((N, 2)), z0[2::3], np.sum(z0[2::3])
