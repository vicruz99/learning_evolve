import numpy as np
from scipy.optimize import minimize

def get_penalty(centers, radii, n_circles=26):
    """
    Calculates the penalty for constraint violations.
    Returns a positive value if constraints are violated, 0 otherwise.
    """
    penalty = 0.0
    
    # Boundary constraints: x +/- r must be in [0, 1]
    # Equivalent to r <= x <= 1-r and r <= y <= 1-r
    # We penalize if r > x or r > (1-x) etc.
    r = radii.reshape(-1, 1) # (N, 1)
    x = centers[:, 0:1]      # (N, 1)
    y = centers[:, 1:2]      # (N, 1)
    
    # Violations: 
    # left: x - r < 0  => r - x > 0
    # right: x + r > 1 => x + r - 1 > 0
    # bottom: y - r < 0
    # top: y + r > 1
    
    boundary_viol = (r - x) + (x + r - 1) + (r - y) + (y + r - 1)
    # Only keep positive violations
    boundary_viol = np.maximum(boundary_viol, 0)
    penalty += np.sum(boundary_viol ** 2)
    
    # Overlap constraints
    # Distance between centers >= r_i + r_j
    # dist - (r_i + r_j) >= 0
    # Violation: (r_i + r_j) - dist > 0
    
    # Vectorized distance calculation
    # Centers shape (N, 2)
    # c_diff shape (N, N, 2)
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(c_diff ** 2, axis=2))
    
    # Radii sum shape (N, N)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap violation
    overlap_viol = r_sum - dist
    # Diagonal is 0, we can ignore or it will be 0 naturally
    np.fill_diagonal(overlap_viol, 0)
    
    # Only keep positive violations
    overlap_viol = np.maximum(overlap_viol, 0)
    penalty += np.sum(overlap_viol ** 2)
    
    return penalty

def objective(x, n_circles=26):
    """
    Objective function to minimize.
    We want to maximize sum(radii), so we minimize -sum(radii).
    Plus a large penalty for constraint violations.
    """
    # Unpack variables
    # x contains: [x1, x2, ..., x26, y1, ..., y26, r1, ..., r26]
    # Actually, let's keep it simple: 
    # First n*2 elements are centers (x,y) flattened
    # Last n elements are radii
    
    centers = x[:n_circles * 2].reshape(n_circles, 2)
    radii = x[n_circles * 2:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    obj = -np.sum(radii)
    
    # Penalty term
    # We want to find a valid packing, so penalty must be zero in the end.
    # Using a high weight.
    weight = 1000.0
    p = get_penalty(centers, radii, n_circles)
    obj += weight * p
    
    return obj
