import numpy as np
from scipy.optimize import minimize

def compute_overlap_penalty(centers, radii):
    n = len(radii)
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2
    return penalty

def compute_boundary_penalty(centers, radii):
    n = len(radii)
    penalty = 0.0
    for i in range(n):
        for d in range(2):
            if centers[i][d] < radii[i]:
                penalty += (radii[i] - centers[i][d]) ** 2
            if centers[i][d] > 1 - radii[i]:
                penalty += (centers[i][d] - 1 + radii[i]) ** 2
    return penalty

def force_based_optimize(n, centers_init, radii_init, seed_val):
    np.random.seed(seed_val)
    centers = centers_init.copy()
    radii = radii_init.copy()
    
    dt = 0.05
    repulsion_strength = 15.0
    boundary_strength = 20.0
    growth_rate = 5e-6
    
    for iteration in range(30000):
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-12:
                    overlap = min_dist - dist
                    direction = diff / dist
                    f = direction * overlap * repulsion_strength
                    forces[i] += f
                    forces[j] -= f
        
        for i in range(n):
            for d in range(2):
                if centers[i][d] < radii[i]:
                    forces[i][d] += (radii[i] - centers[i][d]) * boundary_strength
                if centers[i][d] > 1 - radii[i]:
                    forces[i][d] -= (centers[i][d] - 1 + radii[i]) * boundary_strength
        
        centers += dt * forces
        
        for i in range(n):
            centers[i][0] = max(radii[i] + 1e-10, min(1 - radii[i] - 1e-10, centers[i][0]))
            centers[i][1] = max(radii[i] + 1e-10, min(1 - radii[i] - 1e-10, centers[i][1]))
        
        radii += growth_rate
        radii = np.maximum(radii, 1e-8)
        
        for i in range(n):
            max_r = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
            radii[i] = min(radii[i], max_r - 1e-10)
        
        if iteration % 2000 == 0 and iteration > 0:
            jitter = 0.003 * (1 - iteration / 30000)
            centers += np.random.randn(n, 2) * jitter

                centers[i][1] = max(radii[i] + 1e-10, min(1 - radii[i] - 1e-10, centers[i][1]))
    
    return centers, radii


def local_optimize_scipy(n, centers_init, radii_init):
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    def objective(x):
        co = x[:2 * n].reshape(n, 2)
        ro = x[2 * n:]
        obj = -np.sum(ro)
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((co[i] - co[j]) ** 2))
                min_dist = ro[i] + ro[j]
                if dist < min_dist:
                    obj += 5000.0 * (min_dist - dist) ** 2
        
        for i in range(n):
            for d in range(2):
                if co[i][d] < ro[i]:
                    obj += 5000.0 * (ro[i] - co[i][d]) ** 2
                if co[i][d] > 1 - ro[i]:
                    obj += 5000.0 * (co[i][d] - 1 + ro[i]) ** 2
        
        for i in range(n):
            if ro[i] < 0:
                obj += 5000.0 * ro[i] ** 2
        
        return obj
    
    result = minimize(objective, x0, method='BFGS', 
                     options={'maxiter': 5000, 'gtol': 1e-10})
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    
    for i in range(n):
        radii_opt[i] = max(1e-10, radii_opt[i])
        radii_opt[i] = min(radii_opt[i], centers_opt[i][0], 1 - centers_opt[i][0],
                          centers_opt[i][1], 1 - centers_opt[i][1])
    
    return centers_opt, radii_opt


def cleanup_and_validate(centers, radii, n):
    for iteration in range(2000):
        max_overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    max_overlap = max(max_overlap, overlap)

                        direction = (centers[i] - centers[j]) / dist
                        scale = 0.5
                        centers[i] += direction * overlap * scale
                        centers[j] -= direction * overlap * scale
        
        for i in range(n):
            for d in range(2):
                if centers[i][d] < radii[i]:
                    centers[i][d] = radii[i]
                if centers[i][d] > 1 - radii[i]:
                    centers[i][d] = 1 - radii[i]

            radii[i] = max(1e-10, radii[i])
        
        if max_overlap < 1e-12:
            break
