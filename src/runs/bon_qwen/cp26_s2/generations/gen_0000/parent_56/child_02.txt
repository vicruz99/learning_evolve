from scipy.optimize import minimize, NonlinearConstraint

def repulsion_force(c1, r1, c2, r2):
    """Calculates the repulsive force vector for circle 1 due to circle 2."""
    diff = c1 - c2
    dist = np.linalg.norm(diff)
    min_dist = r1 + r2
    if dist < 1e-9:
        return np.random.rand(2) * 0.01
    
    if dist < min_dist:
        # Strong repulsion to resolve overlap
        force = (min_dist - dist) * (diff / dist) * 1.0
    else:
        # Soft repulsion to maintain separation
        force = 0
    return force

def boundary_force(center, r, bounds):

    force = np.zeros(2)

def force_directed_relaxation(centers, radii, iterations=2000, growth_rate=0.00005, damping=0.9):

        growth_rate *= 0.999
        
    return centers, radii

def objective(params, n):

def inequality_constraints(params, n):
