import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
_PENALTY_WEIGHT = 1000.0


def extract_centers(params):
    return params[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)


def extract_radii(params):
    return params[2 * N_CIRCLES:]


def compute_boundary_violation(params):
    centers = extract_centers(params)
    radii = extract_radii(params)
    v = 0.0
    for i in range(N_CIRCLES):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            v += (x - r) ** 2
        if x + r > 1:
            v += (x + r - 1) ** 2
        if y - r < 0:
            v += (y - r) ** 2
        if y + r > 1:
            v += (y + r - 1) ** 2
        if r < 0:
            v += r ** 2
    return v


def compute_overlap_violation(params):
    centers = extract_centers(params)
    radii = extract_radii(params)
    v = 0.0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d2 = dx * dx + dy * dy
            d = np.sqrt(d2) if d2 > 1e-20 else 1e-10
            sr = radii[i] + radii[j]
            if d < sr:
                v += (sr - d) ** 2
    return v


def compute_total_violation(params):
    return compute_boundary_violation(params) + compute_overlap_violation(params)


def neg_sum_radii(params):
    return -np.sum(extract_radii(params))


def penalized_objective(params):
    return neg_sum_radii(params) + _PENALTY_WEIGHT * compute_total_violation(params)


def make_hex_row(row_idx, count, h_spacing, v_spacing, y_start):
    global _PENALTY_WEIGHT
    row = np.zeros((count, 2))
    offset = (row_idx % 2) * h_spacing * 0.5
    y = y_start + row_idx * v_spacing
    for col in range(count):
        x = 0.1 + offset + col * h_spacing
        row[col] = [x, y]
    return row


def initialize_hexagonal():
    centers = np.zeros((N_CIRCLES, 2))
    h = 0.185
    v = h * np.sqrt(3) / 2
    y0 = 0.09

    configs = [(6, 0.05), (5, 0.19), (6, 0.05), (5, 0.19), (4, 0.05)]
    idx = 0
    for row_i, (cnt, y_adj) in enumerate(configs):
        offset = (row_i % 2) * h * 0.5
        y = y0 + row_i * v + y_adj
        for col in range(cnt):
            x = 0.1 + offset + col * h
            centers[idx] = [x, y]
            idx += 1

    return centers


def try_optimize(params0, max_iter=3000):
    global _PENALTY_WEIGHT
    bounds = [(0.01, 0.99)] * (2 * N_CIRCLES) + [(0.001, 0.5)] * N_CIRCLES
    result = minimize(
        penalized_objective,
        params0,
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-15, "gtol": 1e-10},
    )
    return result.x


def sequential_penalty_optimize(params0):
    global _PENALTY_WEIGHT
    current = params0.copy()
    best = current.copy()
    best_sum = -np.inf

    for pw in [50, 200, 1000, 5000, 20000, 100000, 500000]:
        _PENALTY_WEIGHT = float(pw)
        current = try_optimize(current, max_iter=2000)
        r = extract_radii(current)
        s = np.sum(r)
        if s > best_sum:
            best = current.copy()
            best_sum = s

    return best, best_sum


def growing_circles_algo():
    n = N_CIRCLES
    centers = initialize_hexagonal().copy()
    radii = np.full(n, 0.015)

    lr = 0.003
    for _ in range(15000):
        for i in range(n):
            radii[i] += lr * 0.1

        for _ in range(50):
            # Compute max feasible radius for each circle
            new_radii = np.zeros(n)
            for i in range(n):
                mr = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
                for j in range(n):
                    if i != j:
                        d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                        mr = min(mr, (d - radii[j]) / 2)
                new_radii[i] = max(mr, 0.0001)
            radii = new_radii

            # Compute forces to increase sum of radii
            forces = np.zeros((n, 2))
            for i in range(n):
                # Gradient of boundary constraint
                if centers[i, 0] <= 0.25:
                    forces[i, 0] += 0.001
                elif centers[i, 0] >= 0.75:
                    forces[i, 0] -= 0.001
                if centers[i, 1] <= 0.25:
                    forces[i, 1] += 0.001
                elif centers[i, 1] >= 0.75:
                    forces[i, 1] -= 0.001

                # Repulsion from neighbors
                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        d = np.sqrt(dx * dx + dy * dy)
                        if d < radii[i] + radii[j] + 0.03 and d > 1e-10:
                            rep = max(0, (radii[i] + radii[j] + 0.02 - d) / d)
                            forces[i, 0] += dx * rep * 0.01
                            forces[i, 1] += dy * rep * 0.01

            centers += forces * lr
            centers[:, 0] = np.clip(centers[:, 0], 0.01, 0.99)
            centers[:, 1] = np.clip(centers[:, 1], 0.01, 0.99)

        lr *= 0.9995

    return centers, radii


def clean_solution(centers, radii):
    n = N_CIRCLES
    # Recompute radii based on actual positions
    for _ in range(10):
        for i in range(n):
            r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                    r = min(r, (d - radii[j]) / 2)
            radii[i] = max(r, 0)

    # Clip centers
    for i in range(n):
        centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))

    # Final radius recompute
    for i in range(n):
        r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                r = min(r, (d - radii[j]) / 2)
        radii[i] = max(r, 0)

    return centers, radii


def run_packing():

    if s > best_sum:
        best_params = p.copy()
        best_sum = s
    print(f"  Hexagonal result: sum = {s:.6f}")

    # Strategy 2: Growing circles
    print("Trying growing circles...")
    gc_centers, gc_radii = growing_circles_algo()
    gc_params = np.concatenate([gc_centers.flatten(), gc_radii])
    gc_sum = np.sum(gc_radii)
    print(f"  Growing circles result: sum = {gc_sum:.6f}")

    # Further optimize growing circles result
    _PENALTY_WEIGHT = 100.0
    p2, s2 = sequential_penalty_optimize(gc_params)
    if s2 > best_sum:
        best_params = p2.copy()
        best_sum = s2

    # Strategy 3: Random restarts
    print("Trying random restarts...")
    for seed in range(80):
        np.random.seed(seed)
        centers_r = np.random.uniform(0.12, 0.88, (N_CIRCLES, 2))
        radii_r = np.full(N_CIRCLES, 0.05)
        params_r = np.concatenate([centers_r.flatten(), radii_r])

        # Quick optimization with moderate penalty
        _PENALTY_WEIGHT = 1000.0
        params_opt = try_optimize(params_r, max_iter=1500)

        # Then refine with higher penalties
        _PENALTY_WEIGHT = 10000.0
        params_opt = try_optimize(params_opt, max_iter=1000)
        _PENALTY_WEIGHT = 100000.0
        params_opt = try_optimize(params_opt, max_iter=1000)

        radii_opt = extract_radii(params_opt)
        violation = compute_total_violation(params_opt)
        s_opt = np.sum(radii_opt)

        if violation < 1e-6 and s_opt > best_sum:

    centers, radii = clean_solution(centers, radii)

    return centers, radii, float(np.sum(radii))
