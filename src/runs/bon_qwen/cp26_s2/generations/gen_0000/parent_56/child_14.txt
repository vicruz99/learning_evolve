import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    num_vars = 3 * n
    
    # Precompute constraints
    # x indices: 0 to n-1
    # y indices: n to 2n-1
    # r indices: 2n to 3n-1
    
    cons_list = []
    
    # Boundary constraints: 
    # 1. x_i - r_i >= 0
    # 2. 1 - x_i - r_i >= 0
    # 3. y_i - r_i >= 0
    # 4. 1 - y_i - r_i >= 0
    for i in range(n):
        def f_bound_x_low(v, i=i): return v[i] - v[2*n+i]
        cons_list.append({'type': 'ineq', 'fun': f_bound_x_low})
        
        def f_bound_x_high(v, i=i): return 1.0 - v[i] - v[2*n+i]
        cons_list.append({'type': 'ineq', 'fun': f_bound_x_high})
        
        def f_bound_y_low(v, i=i): return v[n+i] - v[2*n+i]
        cons_list.append({'type': 'ineq', 'fun': f_bound_y_low})
        
        def f_bound_y_high(v, i=i): return 1.0 - v[n+i] - v[2*n+i]
        cons_list.append({'type': 'ineq', 'fun': f_bound_y_high})
        
    # Distance constraints: 
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i+1, n):
            def f_dist(v, i=i, j=j):
                dx = v[i] - v[j]
                dy = v[n+i] - v[n+j]
                dr = v[2*n+i] + v[2*n+j]
                return dx*dx + dy*dy - dr*dr
            cons_list.append({'type': 'ineq', 'fun': f_dist})

    # Objective: maximize sum of radii -> minimize negative sum
    def objective(v):
        return -np.sum(v[2*n:])

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_val = -np.inf
    best_x = None
    
    # Run multiple restarts to find global optimum
    # Using a fixed seed for reproducibility in some runs, but random for others
    for attempt in range(50):

        x_init = np.random.uniform(0.2, 0.8, n)

        r_init = np.full(n, 0.06) 
        # Add slight variation
        r_init = r_init + np.random.uniform(-0.02, 0.02, n)
        r_init = np.clip(r_init, 0.01, 0.25)
        
        vars_init = np.concatenate([x_init, y_init, r_init])
        
        try:
            res = minimize(objective, vars_init, method='SLSQP', bounds=bounds, 
                           constraints=cons_list, options={'ftol': 1e-9, 'maxiter': 2000})
            
            current_sum_r = -res.fun
            if current_sum_r > best_val:
                # Basic feasibility check before updating best
                # We rely on the optimizer, but let's ensure no gross violations
                # The constraints in minimize should handle this, but numerical issues might occur.
                best_val = current_sum_r
                best_x = res.x
                
        except Exception as e:
            continue
            
    if best_x is None:
        # Fallback if optimization failed completely (should not happen)
        # Return a valid small packing

        return centers, radii, 0.0

    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]
    sum_radii = float(np.sum(radii))
    
    # Final validation check (simulated) to ensure robustness
