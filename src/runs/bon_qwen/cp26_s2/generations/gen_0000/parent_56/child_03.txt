import numpy as np
from scipy.optimize import linprog


def compute_optimal_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n)

    A_ub = []
    b_ub = []

    for i in range(n):
        x, y = centers[i]
        for val in [x, 1 - x, y, 1 - y]:
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(val)

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return np.full(n, 1e-6)


def generate_hexagonal(rows_config):
    centers = []
    n_rows = len(rows_config)

    for row_idx, count in enumerate(rows_config):
        y = (row_idx + 1) / (n_rows + 1)
        if row_idx % 2 == 0:
            x_start = 1.0 / (count + 1)
        else:
            x_start = 0.5 / (count + 1)
        x_spacing = 1.0 / (count + 1)
        for col in range(count):
            x = x_start + col * x_spacing
            centers.append([x, y])

    return np.array(centers)


def local_optimize(centers, n, iterations=80):
    centers = centers.copy()
    for it in range(iterations):
        radii = compute_optimal_radii(centers)
        lr = 0.0003 / (1 + it * 0.015)

        for i in range(n):
            grad = np.zeros(2)
            for j in range(n):
                if i == j:
                    continue
                diff = centers[j] - centers[i]
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-10:
                    continue
                threshold = radii[i] + radii[j] + 0.02
                if dist < threshold:
                    force = (threshold - dist)
                    grad += diff / dist * force * lr
            centers[i] += grad
            centers[i][0] = np.clip(centers[i][0], 0.02, 0.98)
            centers[i][1] = np.clip(centers[i][1], 0.02, 0.98)

    return centers


def run_packing():
    np.random.seed(42)
    n = 26

    configs = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [7, 5, 6, 5, 3],
        [6, 6, 6, 5, 3],
        [5, 5, 6, 6, 4],
        [4, 6, 6, 6, 4],
        [6, 6, 5, 6, 3],
        [5, 7, 5, 6, 3],
        [6, 5, 7, 5, 3],
        [7, 6, 5, 5, 3],
        [5, 6, 6, 5, 4],
        [6, 5, 5, 6, 4],
    ]

    best_sum = -1
    best_centers = None
    best_radii = None

    for config in configs:
        if sum(config) != n:
            continue

        for trial in range(8):
            centers = generate_hexagonal(config)
            if trial > 0:
                centers += np.random.uniform(-0.025, 0.025, centers.shape)
                centers = np.clip(centers, 0.06, 0.94)

            centers = local_optimize(centers, n, iterations=50)
            radii = compute_optimal_radii(centers)
            s = np.sum(radii)

            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()

    return best_centers, best_radii, best_sum
