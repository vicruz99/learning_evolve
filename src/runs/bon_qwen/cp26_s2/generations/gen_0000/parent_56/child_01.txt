from scipy.optimize import minimize

def generate_hex_grid(n):

    x, y, r = [], [], []
    # Approximate diameter for 26 circles is ~0.2
    dy = 0.1732  # sqrt(3)/2 * 0.2

    y_curr = 0.15

        x_curr = row_offset

            x.append(x_curr)
            y.append(y_curr)
            r.append(0.09) # Initial small radius
            x_curr += dx
            count += 1
        y_curr += dy
    return np.array(x), np.array(y), np.array(r)

def objective(vars):
    """Objective function: Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def get_constraints(vars, n):
    """Calculate constraint violations: Boundary and Overlap."""
    c_b = []
    for i in range(n):
        x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
        c_b.extend([x - r, 1 - x - r, y - r, 1 - y - r])
    
    c_o = []
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            c_o.append(dist_sq - (ri + rj)**2)
            
    return np.concatenate([c_b, c_o])

def run_packing():

    best_sum = 0.0
