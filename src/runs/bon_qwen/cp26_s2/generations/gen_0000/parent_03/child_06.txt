import numpy as np
from scipy.optimize import linprog, differential_evolution

def get_optimal_radii(centers):
    """
    Solves the Linear Program to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to packing and boundary constraints.
    """
    n = centers.shape[0]
    
    # Maximize sum(r_i) => Minimize -sum(r_i)
    c = -np.ones(n)
    
    # Bound constraints for r_i: 0 <= r_i <= distance to nearest boundary
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        # Distance to left, right, bottom, top boundaries
        dist_bound = min(x, 1-x, y, 1-y)
        # If center is outside [0,1], valid radius is 0
        dist_bound = max(0.0, dist_bound) 
        bounds_r.append((0, dist_bound))
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(C_i, C_j)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            # Constraint row: [0...1...1...0] <= dist
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Solve LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def objective(centers_flat):
    """
    Objective function for optimization: maximize sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    # Clip centers to ensure they are within the square for valid distance calculations
    # though the LP logic handles it, clipping prevents numerical issues with far-out points
    centers = np.clip(centers, 1e-5, 1-1e-5)
    
    radii, sum_r = get_optimal_radii(centers)
    return -sum_r # Minimize negative sum

def generate_hex_init(n):
    """Generate a hexagonal grid initialization."""
    centers = []
    # Approximate spacing for 26 circles
    # 5x5 grid has 25, we can stagger to fit 26
    row_spacing = 1.0 / 6.0
    col_spacing = 1.0 / 5.0
    
    y = 0.5 * row_spacing
    count = 0
    while count < n:
        x = 0.5 * col_spacing
        shift = (count % 2) * 0.5 * col_spacing
        while x <= 1.0 - 1e-5 and count < n:
            centers.append([x + shift, y])
            x += col_spacing
            count += 1
        y += row_spacing * np.sqrt(3)/2 * 0.5 # Adjust vertical spacing
    return np.array(centers[:n])

def generate_grid_init(n):
    """Generate a square grid initialization."""
    centers = []
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    dx = 1.0 / cols
    dy = 1.0 / rows
    for r in range(rows):
        for c in range(cols):
            if len(centers) >= n: break
            centers.append([c * dx + dx/2, r * dy + dy/2])
    return np.array(centers[:n])

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Initial configurations
    init_configs = [
        generate_grid_init(n),
        generate_hex_init(n),
        np.random.rand(n, 2)
    ]

    for config in init_configs:
        # Phase 1: Local Search (Hill Climbing with Annealing)
        centers = config.copy()
        current_radii, current_sum = get_optimal_radii(centers)
        
        step_size = 0.05
        temperature = 1.0
        
        for i in range(2000):
            idx = np.random.randint(n)
            perturbation = np.random.normal(0, step_size, 2)
            new_centers = centers.copy()
            new_centers[idx] += perturbation
            
            # Keep inside bounds
            new_centers = np.clip(new_centers, 0.01, 0.99)
            
            new_radii, new_sum = get_optimal_radii(new_centers)
            
            # Acceptance criteria
            if new_sum > current_sum or np.random.random() < np.exp((new_sum - current_sum) / max(temperature, 1e-5)):
                centers = new_centers
                current_radii = new_radii
                current_sum = new_sum
                temperature *= 0.995
            else:
                temperature *= 0.9999
                if i % 50 == 0:
                    step_size *= 0.95

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers
            best_radii = current_radii

    # Phase 2: Global Refinement using Differential Evolution
    if best_centers is not None:
        # Define bounds for DE: each coordinate in [0, 1]
        bounds = [(0.05, 0.95)] * (n * 2)
        
        # Use DE to refine the best configuration found
        try:
            result = differential_evolution(
                objective, 
                bounds, 
                x0=best_centers.flatten(), 
                maxiter=100, 
                popsize=20,
                mutation=(0.5, 1),
                recombination=0.9,
                tol=1e-4,
                seed=42,
                disp=False
            )
            if result.success:
                final_centers = result.x.reshape(n, 2)
                final_radii, final_sum = get_optimal_radii(final_centers)
                if final_sum > best_sum:
                    best_sum = final_sum
                    best_centers = final_centers
                    best_radii = final_radii
        except Exception:
            pass

    # Ensure best_centers is not None (fallback to random if everything failed)
    if best_centers is None:
        best_centers = np.random.rand(n, 2)
        best_radii, best_sum = get_optimal_radii(best_centers)

    return best_centers, best_radii, best_sum
