import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initial Configuration: Hexagonal Packing (6, 5, 6, 5, 4)
    # This staggered pattern is denser than a square grid.
    # Initial radius guess r=0.095 is a safe, valid starting point.
    r_init = 0.095
    dy = np.sqrt(3) * r_init
    row_counts = [6, 5, 6, 5, 4]
    
    centers = []
    for i, count in enumerate(row_counts):
        y = r_init + i * dy
        # Offset even rows to create the hexagonal pattern
        dx = r_init if i % 2 == 0 else 2 * r_init
        x_start = dx
        for j in range(count):
            x = x_start + j * 2 * r_init
            centers.append([x, y])
    
    centers = np.array(centers)
    radii = np.full(n, r_init)
    
    # 2. Optimization Setup
    # We optimize a single vector of 52 variables (26 x-coords, 26 y-coords, 26 radii)
    # But to maintain symmetry and speed, we can try optimizing all parameters.
    x0 = np.concatenate([centers.flatten(), radii])

    # Objective: Maximize sum of radii -> Minimize negative sum of radii
    def objective(vars):
        radii = vars[52:]
        return -np.sum(radii)

    # Constraints: Circle i and j must not overlap
    # dist(i, j) >= r_i + r_j
    # We implement this as an inequality constraint: dist^2 - (r_i + r_j)^2 >= 0
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            def cons_func(vars, i=i, j=j):
                c_i = vars[2*i : 2*i+2]
                c_j = vars[2*j : 2*j+2]
                r_i = vars[52 + i]
                r_j = vars[52 + j]
                dist_sq = np.sum((c_i - c_j)**2)
                return dist_sq - (r_i + r_j)**2
            cons.append({'type': 'ineq', 'fun': cons_func})

    # Constraints: Boundary checks (center +/- radius in [0, 1])
    for i in range(n):
        def left(vars, i=i): return vars[2*i] - vars[52+i]
        def right(vars, i=i): return 1 - (vars[2*i] + vars[52+i])
        def bottom(vars, i=i): return vars[2*i+1] - vars[52+i]
        def top(vars, i=i): return 1 - (vars[2*i+1] + vars[52+i])
        
        cons.append({'type': 'ineq', 'fun': left})
        cons.append({'type': 'ineq', 'fun': right})
        cons.append({'type': 'ineq', 'fun': bottom})
        cons.append({'type': 'ineq', 'fun': top})

    # Bounds: Centers in [0, 1], Radii >= 0
    bounds = []
    for _ in range(26):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
    for _ in range(26):
        bounds.append((0, 1)) # r

    # 3. Run Optimization
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                   options={'maxiter': 500, 'ftol': 1e-9})

    # 4. Extract and Return Results
    final_centers = res.x[:52].reshape(26, 2)
    final_radii = res.x[52:]
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, float(sum_radii)
