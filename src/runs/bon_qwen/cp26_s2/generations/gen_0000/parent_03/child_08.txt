import numpy as np
from scipy.optimize import minimize, linprog


def objective(params, n=26):
    radii = params[:n]
    centers = params[n:].reshape((n, 2))
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Overlap penalties (quadratic for smoothness)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += 1e5 * (min_dist - dist) ** 2
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            penalty += 1e5 * r ** 2
        if x - r < 0:
            penalty += 1e5 * (x - r) ** 2
        if x + r > 1:
            penalty += 1e5 * (x + r - 1) ** 2
        if y - r < 0:
            penalty += 1e5 * (y - r) ** 2
        if y + r > 1:
            penalty += 1e5 * (y + r - 1) ** 2
    
    return obj + penalty


def optimize_radii_given_centers(centers, n=26):
    c = np.ones(n) * (-1)
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
    
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1 - x, y, 1 - y]:
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(max(0, val))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.ones(n) * 0.05


def repulsion_step(centers, radii, n=26, lr=0.005, iterations=200):
    centers = centers.copy()
    
    for it in range(iterations):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force_mag = overlap / dist * 10.0
                    fx = dx * force_mag
                    fy = dy * force_mag
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0:
                forces[i, 0] += 20.0 * abs(x - r)
            if x + r > 1:
                forces[i, 0] -= 20.0 * abs(x + r - 1)
            if y - r < 0:
                forces[i, 1] += 20.0 * abs(y - r)
            if y + r > 1:
                forces[i, 1] -= 20.0 * abs(y + r - 1)
        
        centers += lr * forces
        centers = np.clip(centers, 0.001, 0.999)
    
    return centers
