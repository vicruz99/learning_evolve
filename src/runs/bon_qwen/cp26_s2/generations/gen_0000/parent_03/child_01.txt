The problem is to pack 26 circles in a unit square to maximize the sum of radii. This is a non-linear optimization problem. 

My strategy involves:
1.  **Initialization**: Start with a structured grid (hexagonal-like) perturbed by random noise to break symmetry and help escape local minima. This provides a better starting point than purely random placement.
2.  **Penalty Method Optimization**: Use a homotopy (continuation) method with a penalty function. The objective is to maximize the sum of radii, equivalent to minimizing $-\sum r_i$. Constraints (boundary and non-overlap) are converted into penalty terms added to the objective. The penalty weight $\lambda$ is increased in stages to guide the optimizer from a feasible region to a tight packing.
3.  **Solver**: Use `scipy.optimize.minimize` with the `L-BFGS-B` method, which handles bound constraints efficiently. Bounds are set for coordinates $[0, 1]$ and radii $[0, 0.5]$.
4.  **Post-processing Expansion**: After optimization, perform a uniform expansion of radii until the first constraint becomes active. This "tightens" the packing, often squeezing out additional sum of radii that numerical tolerances might have missed.
5.  **Multi-start**: Run the optimization multiple times with different random seeds to find a better global optimum.

This approach balances computational efficiency with the ability to find high-density packings close to the theoretical optimum.

import numpy as np
from scipy.optimize import minimize

# Global constant for number of circles
N_CIRCLES = 26

def compute_violations(centers, radii):
    """
    Compute constraint violations for boundary and overlap constraints.
    Returns an array of violation magnitudes.
    """
    n = centers.shape[0]
    
    # Boundary violations: r <= x <= 1-r and r <= y <= 1-r
    # Violation is max(0, r - x), max(0, r - (1-x)), etc.
    v_boundary = np.zeros(4 * n)
    v_boundary[:n] = np.maximum(0, radii - centers[:, 0])
    v_boundary[n:2*n] = np.maximum(0, radii - (1.0 - centers[:, 0]))
    v_boundary[2*n:3*n] = np.maximum(0, radii - centers[:, 1])
    v_boundary[3*n:4*n] = np.maximum(0, radii - (1.0 - centers[:, 1]))
    
    # Overlap violations: dist >= r_i + r_j
    # Violation is max(0, r_i + r_j - dist)
    # Use broadcasting for efficiency
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Get upper triangle indices to avoid double counting and self-interaction
    iu, ju = np.triu_indices(n, k=1)
    v_overlap = np.maximum(0, r_sums[iu, ju] - dists[iu, ju])
    
    return np.concatenate([v_boundary, v_overlap])

def objective_func(v, lam):
    """
    Objective function: minimize -sum(radii) + penalty.
    """
    n = N_CIRCLES
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    
    # Primary objective: maximize sum of radii
    obj = -np.sum(radii)
    
    # Penalty term
    violations = compute_violations(centers, radii)
    pen = np.sum(violations**2)
    
    return obj + lam * pen

def get_initial_guess(seed):
    """
    Generate initial guess based on a perturbed grid.
    """
    n = N_CIRCLES
    rng = np.random.default_rng(seed)
    
    # Create a 5x5 grid pattern
    pts = []
    for row in range(5):
        y = 0.15 + row * 0.18
        for col in range(5):
            x = 0.15 + col * 0.18
            pts.append([x, y])
    # Add 26th point near the top
    pts.append([0.5, 0.85])
    
    pts = np.array(pts[:n])
    
    # Add random noise to break symmetry
    noise = rng.normal(0, 0.02, size=(n, 2))
    pts = np.clip(pts + noise, 0.05, 0.95)
    
    # Initial small radii
    radii = np.ones(n) * 0.05
    
    return np.concatenate([pts.flatten(), radii])

def expand_radii(centers, radii):
    """
    Uniformly expand radii until a constraint is hit.
    Returns updated centers and radii.
    """
    n = centers.shape[0]
    max_factor = 1.0
    
    # Check boundary constraints
    for i in range(n):
        r = radii[i]
        if r < 1e-12: continue
        x, y = centers[i]
        # Factor to hit boundary: r_new = r * (1 + t) => r(1+t) = x => t = x/r - 1
        max_factor = min(max_factor, (x - r)/r, (1.0 - x - r)/r, (y - r)/r, (1.0 - y - r)/r)
        
    # Check overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            r_sum = radii[i] + radii[j]
            if r_sum < 1e-12: continue
            dist = np.linalg.norm(centers[i] - centers[j])
            max_factor = min(max_factor, (dist - r_sum)/r_sum)
            
    # Only expand if possible and valid
    if max_factor > 1e-9:
        radii = radii * (1.0 + max_factor)
        
    return centers, radii

def run_packing():
    """
    Main function to run the packing optimization.
    """
    n = N_CIRCLES
    
    # Define bounds for variables: [x1, y1, ..., xn, yn, r1, ..., rn]
    # Coordinates in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
    
    best_v = None
    best_sum_radii = -1.0
    
    # Run multiple optimizations with different seeds
    for seed in range(5):
        v0 = get_initial_guess(seed)
        v_curr = v0.copy()
        
        # Homotopy schedule: increase penalty weight gradually
        # This helps the optimizer navigate the complex constraint landscape
        for lam in [50.0, 500.0, 2000.0]:
            res = minimize(
                objective_func,
                v_curr,
                args=(lam,),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1500, 'ftol': 1e-13, 'gtol': 1e-8}
            )
            v_curr = res.x
            
        # Extract results
        centers = v_curr[:2*n].reshape((n, 2))
        radii = v_curr[2*n:]
        
        # Post-processing: Expand radii to tighten packing
        centers, radii = expand_radii(centers, radii)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_v = np.concatenate([centers.flatten(), radii])
            
    best_centers = best_v[:2*n].reshape((n, 2))
    best_radii = best_v[2*n:]
    
    return best_centers, best_radii, best_sum_radii
