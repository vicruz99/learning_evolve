import numpy as np
from scipy.optimize import minimize

def objective(p, n=26):
    """Objective function: minimize -sum(r) + penalty for constraint violations."""
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Primary objective: maximize sum of radii
    val = -np.sum(r)
    
    # Penalty term
    pen = 0.0
    
    # Boundary constraints: circles must be inside [0,1]x[0,1]
    # x >= r  => r - x <= 0
    pen += np.sum(np.maximum(0.0, r - x)**2)
    # x + r <= 1 => x + r - 1 <= 0
    pen += np.sum(np.maximum(0.0, x + r - 1.0)**2)
    # y >= r
    pen += np.sum(np.maximum(0.0, r - y)**2)
    # y + r <= 1
    pen += np.sum(np.maximum(0.0, y + r - 1.0)**2)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            ov = r[i] + r[j] - d
            if ov > 0.0:
                pen += ov**2
                
    return val + 5000.0 * pen

def run_packing():
    n = 26
    best_fun = np.inf
    best_x = None
    
    # Multiple restarts to avoid local minima
    for seed in range(15):
        np.random.seed(seed)
        # Random initial positions, small initial radii
        xs = np.random.rand(n)
        ys = np.random.rand(n)
        rs = np.full(n, 0.04)
        p0 = np.concatenate([xs, ys, rs])
        
        # Bounds: x,y in [0,1], r in [0, 0.5]
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
        
        res = minimize(objective, p0, args=(n,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-10})
        
        if res.fun < best_fun:
            best_fun = res.fun
            best_x = res.x.copy()
            
    # Extract results
    xs = best_x[0::3]
    ys = best_x[1::3]
    rs = best_x[2::3]
    
    # Enforce bounds strictly
    xs = np.clip(xs, 0.0, 1.0)
    ys = np.clip(ys, 0.0, 1.0)
    rs = np.clip(rs, 0.0, 0.5)
    
    # Ensure circles are within boundaries by shrinking radii if necessary
    for i in range(n):
        rs[i] = min(rs[i], xs[i], 1.0 - xs[i], ys[i], 1.0 - ys[i])
        
    centers = np.column_stack([xs, ys])
    return centers, rs, np.sum(rs)
