def compute_potential_and_grad(centers, r):
    """
    Computes the potential (sum of squared violations) and gradient with respect to centers.
    Violations:
    1. Overlaps: max(0, r_i + r_j - dist)^2. Here r_i=r_j=r, so max(0, 2r - dist)^2.
    2. Boundaries: max(0, r - x)^2, max(0, x - (1-r))^2, same for y.
    """
    n = centers.shape[0]
    potential = 0.0
    grad = np.zeros_like(centers)

    # x < r => violation (r - x)
    # x > 1 - r => violation (x - (1 - r))

            err = r - x
            potential += err**2
            grad[i, 0] -= 2 * err # d/dx (r-x)^2 = 2(r-x)(-1)

            err = x - (1 - r)
            potential += err**2
            grad[i, 0] += 2 * err # d/dx (x - (1-r))^2 = 2(...)

            grad[i, 1] -= 2 * err

            err = y - (1 - r)
            potential += err**2
            grad[i, 1] += 2 * err

    # dist < 2r => violation (2r - dist)
    # dist = sqrt((xi-xj)^2 + (yi-yj)^2)
    # violation = (2r - dist)^2
    # d/dxi violation = 2(2r - dist) * (-1/dist) * (xi - xj)

    return potential, grad
