import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    rng = np.random.default_rng(42)

    # --- 1. Initialization (Hexagonal Lattice) ---
    # Estimate radius based on hexagonal packing density
    # Density ~ 0.9069, Area = 1
    # n * pi * r^2 <= 0.9069
    # r_approx = sqrt(0.9069 / (n * pi))
    r_approx = math.sqrt(0.9069 / (n * math.pi))
    
    # Hexagonal packing vertical spacing is sqrt(3)/2 * 2r = r*sqrt(3)
    # But to fit in unit square, we scale coordinates.
    # Let's try to fit a grid that might hold 26 circles.
    # A rough estimate for row count. 
    # If r ~ 0.1, height ~ 1, rows ~ 1 / (0.1*1.732) ~ 5-6.
    # Let's try to pack them in a hex grid within [0,1]x[0,1]
    
    centers = np.zeros((n, 2))
    
    # We'll try to fill rows.
    row_idx = 0
    col_idx = 0
    count = 0
    
    # Parameters for initial placement
    # We place them slightly tighter to allow optimizer to push them apart
    spacing_x = 2.0 * r_approx * 0.95 
    spacing_y = math.sqrt(3)/2 * 2.0 * r_approx * 0.95
    
    y = r_approx * 0.95
    
    while count < n:
        x = r_approx * 0.95 + col_idx * spacing_x
        if row_idx % 2 == 1:
            x += spacing_x / 2 # Shift for hexagonal
        
        if x + r_approx * 0.95 <= 1.0:
            centers[count] = [x, y]
            count += 1
            col_idx += 1
        else:
            col_idx = 0
            y += spacing_y
            row_idx += 1
            if y + r_approx * 0.95 > 1.0:
                # If we run out of vertical space, reset or just break
                # For robustness, just fill remaining randomly if needed
                break

    # Fill any remaining spots randomly if lattice failed to place all
    if count < n:
        for i in range(count, n):
            centers[i] = rng.uniform(r_approx, 1.0 - r_approx, 2)

    # Reshape for optimizer: flatten (x1, y1, x2, y2, ...)
    x0 = centers.flatten()

    # --- 2. Optimization ---
    # We want to maximize the minimum distance between circles and walls.
    # Let d_min be the largest value such that all pairwise distances >= d_min
    # and all wall distances >= d_min/2.
    # Actually, let's optimize for radius r directly.
    # But r is coupled.
    # Alternative: Maximize a function f(centers) = min( pairwise_dists, wall_dists*2 )
    # Then optimal r = f(centers_opt) / 2.
    
    def objective(params):
        # params is flattened centers
        c = params.reshape((n, 2))
        
        # Calculate pairwise distances
        # To avoid O(n^2) inside loop every time, but n=26 is small.
        # dists = np.sqrt( ((c[:, np.newaxis, :] - c[np.newaxis, :, :]) ** 2).sum(axis=2) )
        # np.fill_diagonal(dists, 1.0) # Ignore self
        # min_dist = np.min(dists)
        
        # Faster explicit loop for small n
        min_d = 10.0
        for i in range(n):
            # Wall distances
            x, y = c[i]
            # Distance to walls: min(x, 1-x, y, 1-y)
            # But constraint is circle radius r <= distance to wall.
            # So 2r <= 2*distance.
            # We want to maximize 2r. So let's track 2*wall_dist.
            w_dist = min(x, 1.0 - x, y, 1.0 - y)
            val = 2.0 * w_dist
            if val < min_d:
                min_d = val
            
            # Pairwise distances
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = math.sqrt(dx*dx + dy*dy)
                if d < min_d:
                    min_d = d
                    if min_d < 1e-6: break
            if min_d < 1e-6: break
        
        # We want to maximize min_d, so minimize -min_d
        return -min_d

    # Bounds for centers: [0, 1]
    bounds = [(0.0, 1.0)] * (2 * n)

    # Use L-BFGS-B or SLSQP. L-BFGS-B is good for bound constraints.
    # Since objective is non-smooth (min function), we can use a smooth approximation
    # or just rely on the optimizer finding a good spot. 
    # A simple log-sum-exp smoothing or just direct min might work if we get lucky.
    # Given the noise of 'min', a derivative-free method like Nelder-Mead might be safer,
    # but slower. Let's try SLSQP with multiple restarts or just one good run.
    # Actually, 'min' is not differentiable, so gradient based methods might struggle.
    # However, for packing, the active constraints usually dominate.
    
    # Let's use Nelder-Mead (derivative-free) which handles non-smooth functions better.
    res = opt.minimize(objective, x0, method='Nelder-Mead', options={'maxiter': 50000, 'xatol': 1e-6, 'fatol': 1e-9})
    
    optimal_centers = res.x.reshape((n, 2))
    
    # --- 3. Calculate Radii ---
    # The max feasible radius r is half of the min distance found.
    # Let's recompute the exact min distance from the optimal centers to be precise.
    min_d = 2.0 # Start with max possible (diameter 2)
    
    for i in range(n):
        x, y = optimal_centers[i]
        # Distance to wall
        w = min(x, 1.0 - x, y, 1.0 - y)
        if 2.0 * w < min_d:
            min_d = 2.0 * w
            
        for j in range(i + 1, n):
            dx = optimal_centers[i, 0] - optimal_centers[j, 0]
            dy = optimal_centers[i, 1] - optimal_centers[j, 1]
            d = math.sqrt(dx*dx + dy*dy)
            if d < min_d:
                min_d = d

    r = min_d / 2.0
    
    # Assign radii
    radii = np.full(n, r)
    
    # Sum of radii
    sum_radii = np.sum(radii)
    
    return optimal_centers, radii, sum_radii

# Run the packing
