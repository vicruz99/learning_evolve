import numpy as np
import scipy.optimize as opt

def run_packing():
    np.random.seed(42)
    n = 26
    best_obj = 0
    best_centers = None
    best_radii = None

    # Hexagonal Lattice Initialization
    r_init = 0.095
    s = 2 * r_init
    h = s * np.sqrt(3) / 2
    
    rows = [5, 5, 5, 5, 5, 1]
    centers = []
    for i, count in enumerate(rows):
        y = r_init + i * h
        x_start = r_init if i % 2 == 0 else r_init + s / 2
        for j in range(count):
            centers.append([x_start + j * s, y])
            
    centers = np.array(centers)

    # Optimization setup
    x0 = np.concatenate([centers.flatten(), np.full(n, r_init)])
    bounds = []
    for _ in range(n):
        bounds.extend([
            (0, 1), (0, 1), # x, y bounds
            (0, 1)          # radius bounds
        ])

    for attempt in range(10):
        # Add noise to initialization
        noise = np.random.normal(0, 0.01, len(x0))
        x0_noisy = np.clip(x0 + noise, [0]*len(x0), [1]*len(x0))

        def objective(x):
            return -sum(x[2*i+2] for i in range(n))

        # Constraints
        constraints = []
        
        # Boundary and radius constraints
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*i+2]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i] - x[2*i+2]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i+1] - x[2*i+2]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i+1] - x[2*i+2]})

        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                def dist_constraint(x, i=i, j=j):
                    dx = x[2*i] - x[2*j]
                    dy = x[2*i+1] - x[2*j+1]
                    r_sum = x[2*i+2] + x[2*j+2]
                    return dx**2 + dy**2 - r_sum**2
                constraints.append({'type': 'ineq', 'fun': dist_constraint})

        res = opt.minimize(
            objective, x0_noisy, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )

        current_sum = sum(res.x[2*i+2] for i in range(n))
        if current_sum > best_obj:
            best_obj = current_sum
            best_centers = res.x[:2*n].reshape((n, 2))
            best_radii = res.x[2*n:]

    return best_centers, best_radii, best_obj
