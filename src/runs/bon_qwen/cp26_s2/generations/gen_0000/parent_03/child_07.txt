import numpy as np
from scipy.optimize import linprog
import math

def get_max_sum_radii(centers):
    """
    Solves the Linear Programming problem to find the maximum sum of radii
    for a fixed set of centers.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        sum_radii: float, sum of the optimal radii
        radii: np.array, the optimal radii
        success: bool
    """
    n = centers.shape[0]
    # Variables: r_0 ... r_{n-1}
    # Maximize sum(r) => Minimize -sum(r)
    c_obj = np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. r_i + r_j <= dist(i, j) for all pairs
    # 2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i (Boundary)
    
    # Precompute distances and constraints
    A_ub = []
    b_ub = []
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            
            # Row vector with 1 at i and j
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints and Bounds
    # Instead of adding constraints for every boundary, we can use variable bounds
    # r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if max_r < 0: max_r = 0.0
        bounds.append((0.0, max_r))
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(-c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, -res.x, True
        else:
            return 0.0, np.zeros(n), False
    except Exception:
        return 0.0, np.zeros(n), False

def get_initial_centers(n):
    """
    Initializes centers in a hexagonal lattice pattern.
    """
    centers = []
    # Approximate grid size
    cols = math.ceil(math.sqrt(n))
    rows = math.ceil(n / cols)
    
    # Scale factor to fit in square
    # Hexagonal packing height ratio is sqrt(3)/2 * 2r = sqrt(3)r
    # Width is 2r per circle.
    # We want to fit rows and cols.
    
    # Let's just create a grid and jitter slightly to avoid symmetry issues
    # A 5x6 grid is 30 points. We take first 26.
    # But hexagonal shift is better.
    
    # Let's try a 6x5 rectangular arrangement of points, shifted for hex
    # 5 rows, 6 cols = 30. We need 26.
    
    x_step = 0.20 # Approx 1/5
    y_step = 0.17 # Approx 1/6
    
    # We will refine these later, just need a spread out init
    # Simple uniform grid
    grid_x = np.linspace(0.05, 0.95, 6)
    grid_y = np.linspace(0.05, 0.95, 5)
    
    pts = []
    for y in grid_y:
        row_pts = []
        for x in grid_x:
            pts.append([x, y])
            if len(pts) >= 26:
                break
        if len(pts) >= 26:
            break
        row_pts
    
    # If we have points, add small random noise to break symmetry
    centers = np.array(pts[:26])
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    # Clip to stay in bounds
    centers = np.clip(centers, 0.01, 0.99)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42) # For reproducibility
    
    # 1. Initialize
    centers = get_initial_centers(n)
    
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    
    # 2. Optimization Loop
    num_iterations = 600
    base_step = 0.02
    
    for it in range(num_iterations):
        # Solve LP for current centers
        current_sum, radii, success = get_max_sum_radii(centers)
        
        if success and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        # If we didn't find a better solution, we try to move centers
        # Compute forces based on "tightness" of constraints
        # Tight pairs: r_i + r_j is close to dist(i, j)
        
        forces = np.zeros_like(centers)
        
        # We need radii to compute forces. Use the ones just found or best ones.
        # If LP failed, we might use previous best or zeros.
        if not success:
            radii = np.zeros(n)
            
        # Identify active constraints
        # Pairs
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                
                sum_r = radii[i] + radii[j]
                
                # If touching or close (within 5% of distance or absolute small gap)
                # We want to push apart if they are limiting the radii
                # The condition r_i + r_j <= dist is active if dist - (r_i+r_j) is small
                gap = dist - sum_r
                
                if gap < 1e-3: # Active constraint
                    # Repulsive force
                    if dist > 1e-6:
                        nx = dx / dist
                        ny = dy / dist
                        # Force magnitude: stronger if closer to touching
                        # Just a constant push
                        mag = 1.0 
                        forces[i, 0] += nx * mag
                        forces[i, 1] += ny * mag
                        forces[j, 0] -= nx * mag
                        forces[j, 1] -= ny * mag

        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Check boundaries
            # Left wall: x - r approx 0 => x approx r
            if x - r < 1e-3 and x - r > -1e-3:
                forces[i, 0] += 1.0
            # Right wall
            if (1 - x) - r < 1e-3 and (1 - x) - r > -1e-3:
                forces[i, 0] -= 1.0
            # Bottom
            if y - r < 1e-3 and y - r > -1e-3:
                forces[i, 1] += 1.0
            # Top
            if (1 - y) - r < 1e-3 and (1 - y) - r > -1e-3:
                forces[i, 1] -= 1.0

        # Normalize forces to prevent huge jumps
        # But simple accumulation is okay if step size is small
        
        # Adaptive step size: decrease over time or based on progress
        # Add some noise to escape local optima
        if it % 100 == 0 and it > 0:
            centers += np.random.uniform(-0.02, 0.02, centers.shape)
            
        # Apply forces
        step = base_step * (1.0 - it / num_iterations) # Decay step
        if step < 0.001: step = 0.001
        
        centers += forces * step
        
        # Project back to valid region [0, 1]
        # Actually centers must be in [0,1], but radii must fit.
        # If center is at 0, radius must be 0.
        # Let's just keep centers in [0,1].
        centers = np.clip(centers, 0.0, 1.0)
        
    # Final validation and return
    final_sum, final_radii, _ = get_max_sum_radii(best_centers)
    
    # Ensure valid packing
    # The LP guarantees valid radii for the centers.
    # But we return best_centers and best_radii.
    
    return best_centers, best_radii, best_sum
