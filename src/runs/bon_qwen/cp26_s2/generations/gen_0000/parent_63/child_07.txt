import warnings
warnings.filterwarnings('ignore')


def compute_objective(params, n):

    return -np.sum(params[2 * n:])


def compute_constraints(params, n):
    """Compute all constraint values (must be >= 0)."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    constraints = []
    
    # Boundary constraints for each circle
    for i in range(n):
        x, y = centers[i]
        r = radii[i]

        constraints.append(x - r)
        constraints.append(1.0 - x - r)
        constraints.append(y - r)
        constraints.append(1.0 - y - r)
        # Non-negative radius
        constraints.append(r)
    
    # Non-overlap constraints for each pair
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-30)
            constraints.append(dist - radii[i] - radii[j])
    
    return np.array(constraints)


def create_hexagonal_init(n):

            centers.append([x, y])

def create_grid_init(n):
