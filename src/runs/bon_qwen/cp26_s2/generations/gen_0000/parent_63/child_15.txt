import numpy as np
from scipy.optimize import minimize, linprog

def get_distance_matrix(centers):
    """Computes the pairwise distance matrix for a set of centers."""
    # Calculate pairwise squared differences
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    return dists

def potential_energy(centers):
    """
    Computes the repulsive potential energy.
    Minimizing this pushes points apart and away from boundaries.
    """
    n = centers.shape[0]
    dists = get_distance_matrix(centers)
    
    # Inverse square distance between points (with small epsilon to avoid singularity)
    # np.fill_diagonal sets diagonal to 0, 1/0 is inf, so we mask it
    np.fill_diagonal(dists, np.inf)
    point_energy = np.sum(1.0 / (dists**2 + 1e-8))
    
    # Inverse square distance to boundaries
    boundary_dists = np.minimum(
        np.minimum(centers[:, 0], 1 - centers[:, 0]),
        np.minimum(centers[:, 1], 1 - centers[:, 1])
    )
    boundary_energy = np.sum(1.0 / (boundary_dists**2 + 1e-8))
    
    return point_energy + boundary_energy

def generate_initial_centers(n):
    """Generates an initial configuration using a perturbed hexagonal grid."""
    centers = np.zeros((n, 2))
    
    # Estimate grid dimensions for hexagonal packing
    # Area per circle approx sqrt(3)/2 * d^2
    # Just placing them in a rectangular arrangement that approximates hex packing
    
    # We can simply place them in rows
    rows = int(np.ceil(np.sqrt(n * np.sqrt(3) / 2)))
    # Adjust rows to be reasonable
    rows = max(4, min(7, rows))
    
    idx = 0
    x_step = 1.1 / (n / rows + 1)
    y_step = 1.1 / rows
    
    for r in range(rows):
        y = y_step * (r + 0.5)
        # Offset x for odd rows to create hexagonal pattern
        x_offset = (0.5 if r % 2 == 1 else 0) * (x_step / 2)
        
        cols = int(np.ceil((n - idx) / 1.0)) # Estimate columns needed
        for c in range(int(cols)):
            if idx >= n:
                break
            x = x_step * (c + 0.5) + x_offset
            
            # Add small jitter to break symmetry
            centers[idx, 0] = np.clip(x + np.random.uniform(-0.01, 0.01), 0.05, 0.95)
            centers[idx, 1] = np.clip(y + np.random.uniform(-0.01, 0.01), 0.05, 0.95)
            idx += 1
            
    return centers

def solve_radii_lp(centers):
    """
    Solves the Linear Programming problem to maximize sum of radii 
    given fixed centers.
    
    Maximize sum(r_i)
    Subject to:
        r_i + r_j <= dist(i, j) for all i < j
        r_i <= boundary_dist(i) for all i
        r_i >= 0
    """
    n = centers.shape[0]
    
    # Calculate distances between all pairs
    dists = get_distance_matrix(centers)
    
    # Calculate max radius based on boundaries for each circle
    r_max_boundary = np.minimum(
        np.minimum(centers[:, 0], 1 - centers[:, 0]),
        np.minimum(centers[:, 1], 1 - centers[:, 1])
    )
    
    # Objective: Minimize -sum(r)
    c = np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # We need to construct matrices for r_i + r_j <= d_ij
    # And r_i <= r_max_boundary_i
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints
    # r_i <= r_max_boundary_i
    # This is equivalent to I @ r <= r_max_boundary
    # We can add these to A_ub or handle separately. 
    # Adding to A_ub is straightforward.
    A_ub.append(np.eye(n))
    b_ub.append(r_max_boundary)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # Highs method is generally robust
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: equal radii based on min distance
        min_d = np.min(r_max_boundary)
        min_pair_d = np.min(dists[np.triu_indices(n, k=1)])
        r = min(min_d, min_pair_d / 2)
        return np.full(n, max(r, 0))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42) # Reproducibility
    n = 26
    
    # Generate initial centers
    centers = generate_initial_centers(n)
    
    # Optimize centers to maximize separation (minimize potential energy)
    # Use bounds to keep centers within [0, 1]
    bounds = [(0, 1) for _ in range(2 * n)]
    
    # Reshape centers to 1D for scipy.optimize
    x0 = centers.flatten()
    
    # Optimization
    res_opt = minimize(
        fun=lambda x: potential_energy(x.reshape(n, 2)),
        x0=x0,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 2000, 'ftol': 1e-12}
    )
    
    optimized_centers = res_opt.x.reshape(n, 2)
    
    # Solve for radii using Linear Programming
    radii = solve_radii_lp(optimized_centers)
    
    # Ensure no NaN or negative values (safety)
    radii = np.nan_to_num(radii, nan=0.0)
    radii = np.maximum(radii, 0.0)
    
    # Final sum
    sum_radii = np.sum(radii)
    
    # Return result
    return optimized_centers, radii, sum_radii
