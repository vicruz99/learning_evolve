                     # Shift x_pos by half spacing?

                     # Let's stick to uniform for safety, or careful hex.
                     pass
                
                centers[current_idx] = [x_pos, y_pos]
                radii[current_idx] = r_init
                current_idx += 1

        # But with uniform rows, we can use 2*r.
        # Let's use 2*r to ensure no vertical overlap.
        y_pos += 2 * r_init

        pass

        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
        dist_sq = np.sum(diff**2, axis=2) # (n, n)

    # Objective: Maximize sum of radii.

    try:

    return final_centers, final_radii, sum_radii
