import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Objective function: minimize negative sum of radii."""
    n = 26
    return -np.sum(v[2::3])

def eval_constraints(v):
    """Evaluate all inequality constraints: g(v) >= 0."""
    n = 26
    n_bounds = 4 * n
    n_pairs = n * (n - 1) // 2
    c = np.zeros(n_bounds + n_pairs)
    
    # Boundary constraints: x-r >= 0, y-r >= 0, x+r <= 1, y+r <= 1
    for i in range(n):
        ix = 3 * i
        iy = 3 * i + 1
        ir = 3 * i + 2
        c[4*i] = v[ix] - v[ir]
        c[4*i+1] = v[iy] - v[ir]
        c[4*i+2] = 1.0 - (v[ix] + v[ir])
        c[4*i+3] = 1.0 - (v[iy] + v[ir])
        
    # Pairwise non-overlap constraints: dist^2 >= (ri + rj)^2
    idx = n_bounds
    for i in range(n):
        for j in range(i + 1, n):
            dx = v[3*i] - v[3*j]
            dy = v[3*i+1] - v[3*j+1]
            dr = v[3*i+2] + v[3*j+2]
            c[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
    return c

def run_packing():
    n = 26
    
    # Initial configuration: uniform grid spread in the square
    cols = 5
    rows = 6
    x_vals = np.linspace(0.1, 0.9, cols)
    y_vals = np.linspace(0.1, 0.9, rows)
    init_centers = np.array([[x, y] for x in x_vals for y in y_vals])
    init_centers = init_centers[:n]
    
    init_radii = np.full(n, 0.05)
    
    # Flatten to optimization vector [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    x0[0::3] = init_centers[:, 0]
    x0[1::3] = init_centers[:, 1]
    x0[2::3] = init_radii
    
    # Bounds: x,y in [0,1], r in [1e-4, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-4, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': eval_constraints}
    
    # First optimization pass
    res1 = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                    options={'maxiter': 1500, 'ftol': 1e-9})
                    
    # Second refinement pass
    res2 = minimize(objective, res1.x, method='SLSQP', bounds=bounds, constraints=constraints,
                    options={'maxiter': 1500, 'ftol': 1e-10})
                    
    final_v = res2.x
    centers = final_v[:2*n].reshape(n, 2)
    radii = final_v[2*n:]
    
    # Ensure radii are strictly positive to satisfy validation
    radii = np.maximum(radii, 1e-6)
    
    return centers, radii, float(np.sum(radii))
