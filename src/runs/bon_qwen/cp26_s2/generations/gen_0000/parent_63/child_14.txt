import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def _objective(vars):
    """Objective function: minimize negative sum of radii."""
    radii = vars[2*N_CIRCLES:]
    return -np.sum(radii)

def _constraints(vars):
    """Inequality constraints: boundaries and non-overlap."""
    n = N_CIRCLES
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    num_constr = 4 * n + n * (n - 1) // 2
    cons_vals = np.empty(num_constr)
    idx = 0
    
    # Boundary constraints: circle i must be inside [0,1]x[0,1]
    for i in range(n):
        cons_vals[idx] = x[i] - r[i]           # x >= r
        idx += 1
        cons_vals[idx] = 1.0 - x[i] - r[i]     # 1-x >= r
        idx += 1
        cons_vals[idx] = y[i] - r[i]           # y >= r
        idx += 1
        cons_vals[idx] = 1.0 - y[i] - r[i]     # 1-y >= r
        idx += 1
        
    # Non-overlap constraints: distance^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dr = r[i] + r[j]
            cons_vals[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
            
    return cons_vals

def run_packing():
    n = N_CIRCLES
    
    # Initial configuration: Hexagonal lattice pattern [6, 5, 6, 5, 4]
    centers = []
    r_init = 0.08
    y = r_init
    row_counts = [6, 5, 6, 5, 4]
    
    for k, count in enumerate(row_counts):
        x = r_init + (k % 2) * r_init
        for _ in range(count):
            centers.append([x, y])
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        
    centers = np.array(centers)
    radii = np.full(n, r_init)
    
    # Flatten to optimization vector: [x1, ..., xn, y1, ..., yn, r1, ..., rn]
    x0 = np.concatenate([centers.ravel(), radii])
    
    # Variable bounds
    lb = np.array([0.0] * (2 * n) + [0.0] * n)
    ub = np.array([1.0] * (2 * n) + [0.5] * n)
    bounds = list(zip(lb, ub))
    
    cons = {'type': 'ineq', 'fun': _constraints}
    options = {'maxiter': 5000, 'ftol': 1e-9, 'disp': False}
    
    # Run optimization
    res = minimize(_objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    
    opt_vars = res.x
    opt_centers = opt_vars[:2*n].reshape((n, 2))
    opt_radii = opt_vars[2*n:]
    
    # Post-processing to guarantee validity against numerical drift
    opt_radii = np.maximum(opt_radii, 1e-7)
    opt_centers[:, 0] = np.clip(opt_centers[:, 0], opt_radii, 1.0 - opt_radii)
    opt_centers[:, 1] = np.clip(opt_centers[:, 1], opt_radii, 1.0 - opt_radii)
    
    return opt_centers, opt_radii, float(np.sum(opt_radii))
