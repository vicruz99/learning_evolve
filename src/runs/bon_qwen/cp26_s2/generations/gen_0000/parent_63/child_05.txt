import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    best_sum_r = 0.0
    best_centers = None
    best_radii = None

    # Helper to generate hexagonal grid initialization
    def get_hex_init(radius_factor=0.8):
        # Target radius roughly 0.1, so start smaller
        r_start = 0.06 * radius_factor
        
        # Hexagonal lattice parameters
        # Vertical spacing: sqrt(3) * r
        # Horizontal spacing: 2 * r
        dy = np.sqrt(3) * r_start * 1.2 # Add some padding
        dx = 2 * r_start * 1.2

        centers = []
        # Iterate to fill 26 spots
        y = r_start
        row_idx = 0
        while len(centers) < n:
            x_start = r_start if row_idx % 2 == 0 else r_start + dx / 2
            x = x_start
            while x < 1.0 - r_start and len(centers) < n:
                centers.append([x, y])
                x += dx
            y += dy
            row_idx += 1
        
        centers = np.array(centers[:n])
        radii = np.full(n, r_start)
        return centers, radii

    # Objective function: maximize sum of radii => minimize negative sum
    def objective(vars):
        # vars is flattened [x1, y1, r1, x2, y2, r2, ...]
        # Extract radii
        rs = vars[2::3]
        return -np.sum(rs)

    # Constraint: Circle containment
    # x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    def constraint_containment(vars):
        constraints = []
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i+1]
            r = vars[3*i+2]
            constraints.append(x - r)
            constraints.append(1.0 - x - r)
            constraints.append(y - r)
            constraints.append(1.0 - y - r)
        return np.array(constraints)

    # Constraint: Non-overlap
    # dist_ij >= ri + rj
    # dist_ij^2 >= (ri + rj)^2
    def constraint_overlap(vars):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
                xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx*dx + dy*dy
                r_sum = ri + rj
                
                # Constraint: dist_sq - r_sum^2 >= 0
                constraints.append(dist_sq - r_sum*r_sum)
        return np.array(constraints)

    # Bounds for variables
    # 0 <= x <= 1, 0 <= y <= 1, 0 <= r <= 0.5
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n

    # Constraints definition
    cons = [
        {'type': 'ineq', 'fun': constraint_containment},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Run optimization multiple times with different initializations
    attempts = 5
    for k in range(attempts):
        # Generate initial guess
        # Perturb slightly based on attempt index
        centers_init, radii_init = get_hex_init(radius_factor=0.7 + 0.1 * k)
        
        # Flatten to 1D array for scipy
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        try:
            res = opt.minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
            )
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum_r:
                    # Check validity explicitly just in case
                    final_centers = res.x.reshape(n, 3)[:, :2]
                    final_radii = res.x.reshape(n, 3)[:, 2]
                    
                    # Simple validation check
                    valid = True
                    for i in range(n):
                        if final_radii[i] < 0: valid = False; break
                        if final_centers[i,0] - final_radii[i] < -1e-9 or final_centers[i,0] + final_radii[i] > 1 + 1e-9: valid = False; break
                        if final_centers[i,1] - final_radii[i] < -1e-9 or final_centers[i,1] + final_radii[i] > 1 + 1e-9: valid = False; break
                    
                    if valid:
                         # Check overlaps roughly
                         for i in range(n):
                             for j in range(i+1, n):
                                 dist = np.hypot(final_centers[i,0]-final_centers[j,0], final_centers[i,1]-final_centers[j,1])
                                 if dist < final_radii[i] + final_radii[j] - 1e-9:
                                     valid = False; break
                             if not valid: break
                         
                         if valid:
                            best_sum_r = current_sum
                            best_centers = final_centers
                            best_radii = final_radii

        except Exception as e:
            # If optimization fails, continue to next attempt
            pass

    # If no valid result found from optimization, return a safe fallback (e.g. small circles)
    if best_centers is None:
        # Fallback: 5x5 grid of radius 0.05 plus one, just to be safe
        # Though this is suboptimal, it ensures the code returns something.
        # However, given the logic above, we should have found something.
        # Let's try one last deterministic grid.
        centers = []
        radii = []
        # 5 rows, 5 cols = 25
        for r in range(5):
            for c in range(5):
                centers.append([0.1 + c*0.2, 0.1 + r*0.2])
                radii.append(0.05) # Small enough to fit? 0.1+0.8=0.9. 0.05 fits.
                if len(centers) >= 26: break
            if len(centers) >= 26: break
        
        # Actually, let's just return the best found or a default.
        # If best_centers is still None, construct a trivial valid packing.
        if best_centers is None:
             # 26 circles of radius 0.05 in a grid?
             # 5x5 grid is 25. Add one in corner?
             # Let's just use the init logic with small radius.
             centers_init, radii_init = get_hex_init(0.5)
             best_centers = centers_init
             best_radii = radii_init
             best_sum_r = np.sum(best_radii)

    return best_centers, best_radii, best_sum_r
