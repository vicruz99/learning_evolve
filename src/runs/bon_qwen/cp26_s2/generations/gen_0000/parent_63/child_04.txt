def compute_penalty(centers, r):
    """
    Computes the penalty for a configuration of centers and radius r.
    Penalty is the sum of squared violations of:
    1. Circles inside unit square [0,1]x[0,1]
    2. Circles non-overlapping (distance >= 2r)
    """
    n = centers.shape[0]
    penalty = 0.0
    
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # Violations: max(0, r - x) and max(0, x - (1-r))
    
    # Left wall
    left_viol = np.maximum(0, r - x)
    penalty += np.sum(left_viol**2)
    
    # Right wall
    right_viol = np.maximum(0, x - (1 - r))
    penalty += np.sum(right_viol**2)
    
    # Bottom wall
    bottom_viol = np.maximum(0, r - y)
    penalty += np.sum(bottom_viol**2)
    
    # Top wall
    top_viol = np.maximum(0, y - (1 - r))
    penalty += np.sum(top_viol**2)
    
    # Pairwise distance constraints: dist >= 2r
    # Vectorized distance calculation
    # diff[i, j] = centers[i] - centers[j]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # We only care about pairs i < j to avoid double counting and self-distances
    # Use a mask for lower triangle excluding diagonal
    mask = np.tril(np.ones((n, n), dtype=bool), k=-1)
    pair_dists = dists[mask]
    
    # Violation is max(0, 2r - dist)
    violations = np.maximum(0, 2 * r - pair_dists)
    penalty += np.sum(violations**2)
    
    return penalty
