1.  **Initialization**: A good initial configuration is crucial. We will generate a hexagonal lattice arrangement, as it is the densest circle packing in 2D. We start with a grid spacing that allows a base radius of 0.1 (diameter 0.2) with a 0.1 margin from walls. This fits approximately 27 circles; we will select 26 points from this lattice to serve as initial centers. Initial radii will be set slightly below the theoretical maximum (e.g., 0.09) to ensure strict feasibility.
2.  **Optimization**: We use `scipy.optimize.minimize` with the 'SLSQP' method, which handles inequality constraints efficiently.
    -   **Variables**: The state vector includes $x, y$ coordinates and radius $r$ for each of the 26 circles ($78$ variables).
    -   **Objective**: Minimize the negative sum of radii.
    -   **Constraints**:
        -   Boundary constraints: $0 \le x - r$, $x + r \le 1$, etc.
        -   Overlap constraints: Squared distance between centers must be at least the squared sum of radii, i.e., $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$. Using the squared form avoids non-smooth square roots in gradients.
3.  **Refinement**: The optimizer will iteratively adjust positions and grow radii until constraints become active (circles touch). The hexagonal initialization helps avoid local optima associated with grid alignments.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Grid
    # We generate a hexagonal lattice of points that fits in the unit square.
    # Spacing s = 0.2 allows for radius r = 0.1 with margin 0.1.
    # This grid typically contains ~27 points; we take the first 26.
    
    points = []
    margin = 0.1
    s = 0.2
    dy = s * np.sqrt(3) / 2.0
    
    y = margin
    row = 0
    
    # Generate points until we have at least n
    while len(points) < n:
        if row % 2 == 0:
            # Even row: x starts at margin
            x = margin
            while x <= 1.0 + 1e-9:
                points.append([x, y])
                x += s
        else:
            # Odd row: x starts at margin + s/2
            x = margin + s / 2.0
            while x <= 1.0 + 1e-9:
                points.append([x, y])
                x += s
        
        y += dy
        row += 1
    
    # Take the first n points
    initial_centers = np.array(points[:n])
    
    # Initial radii: set to 0.09 (slightly less than 0.1 to be strictly feasible initially)
    initial_radii = np.full(n, 0.09)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = initial_centers[i, 0]
        x0[3 * i + 1] = initial_centers[i, 1]
        x0[3 * i + 2] = initial_radii[i]
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 0.5))
    
    # Constraints
    constraints = []
    
    # Helper to add boundary constraints
    # x - r >= 0  => x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    def boundary_constraints(v):
        res = []
        for i in range(n):
            idx = 3 * i
            x = v[idx]
            y = v[idx + 1]
            r = v[idx + 2]
            res.append(x - r)
            res.append(1 - x - r)
            res.append(y - r)
            res.append(1 - y - r)
        return np.array(res)
    
    constraints.append({'type': 'ineq', 'fun': boundary_constraints})
    
    # Distance constraints: dist_sq >= (r_i + r_j)^2
    def distance_constraints(v):
        res = []
        for i in range(n):
            idx_i = 3 * i
            xi, yi, ri = v[idx_i], v[idx_i + 1], v[idx_i + 2]
            for j in range(i + 1, n):
                idx_j = 3 * j
                xj, yj, rj = v[idx_j], v[idx_j + 1], v[idx_j + 2]
                
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx * dx + dy * dy
                sum_r = ri + rj
                res.append(dist_sq - sum_r * sum_r)
        return np.array(res)
    
    constraints.append({'type': 'ineq', 'fun': distance_constraints})
    
    # Objective: Minimize negative sum of radii
    def objective(v):
        r_sum = 0.0
        for i in range(n):
            r_sum += v[3 * i + 2]
        return -r_sum
    
    # Run Optimization
    # SLSQP is suitable for constrained non-linear optimization
    try:
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints, 
            options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
        )
        
        # Check if optimization was successful or at least improved
        if res.success or (res.nit > 0):
            x_opt = res.x
        else:
            # Fallback to initial if failed completely
            x_opt = x0
    except Exception as e:
        print(f"Optimization error: {e}")
        x_opt = x0

    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = x_opt[3 * i]
        centers[i, 1] = x_opt[3 * i + 1]
        radii[i] = x_opt[3 * i + 2]
        
    # Ensure radii are non-negative (clipping numerical noise)
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    print(f"Sum of radii: {sum_r}")
