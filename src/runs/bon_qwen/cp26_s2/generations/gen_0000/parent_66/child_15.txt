import math

def run_packing():

    """

    def distance(c1, c2):
        return np.sqrt((c1[0] - c2[0])**2 + (c1[1] - c2[1])**2)

    def apply_forces(centers, radii, k_repulse=50.0, k_boundary=100.0):
        """
        Applies repulsive forces between circles and boundaries to resolve overlaps.
        Updates centers in-place.
        """
        forces = np.zeros_like(centers)
        n = len(centers)

        # Inter-circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                vec = centers[i] - centers[j]
                dist = np.linalg.norm(vec)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    # Normal vector
                    norm_vec = vec / dist
                    # Penetration depth
                    penetration = min_dist - dist
                    # Force proportional to penetration
                    force = norm_vec * penetration * k_repulse
                    forces[i] += force
                    forces[j] -= force
        
        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            
            # Left wall
            if x - r < 0:
                forces[i, 0] += k_boundary * (r - x)
            # Right wall
            if x + r > 1:
                forces[i, 0] -= k_boundary * (x + r - 1)
            # Bottom wall
            if y - r < 0:
                forces[i, 1] += k_boundary * (r - y)
            # Top wall
            if y + r > 1:
                forces[i, 1] -= k_boundary * (y + r - 1)

        centers += forces * 0.01 # Small time step
