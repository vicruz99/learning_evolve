import numpy as np
from scipy.optimize import minimize
import math

def hexagonal_init(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.1
    
    cols = 6
    rows = 5
    
    r = 0.1
    dx = 2 * r * 1.08
    dy = dx * math.sqrt(3) / 2
    
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = (col + 0.5) * dx + 0.06
            y = (row + 0.5) * dy + 0.08
            if row % 2 == 1:
                x += dx / 2
            centers[idx] = [np.clip(x, 0.2, 0.8), np.clip(y, 0.2, 0.8)]
            idx += 1
    
    return centers, radii

def objective(x, n):
    radii = x[2*n:]
    return -np.sum(radii)

def bounds_fn(n):
    bounds = []
    for i in range(n):
        bounds.append((0.05, 0.95))
        bounds.append((0.05, 0.95))
    for i in range(n):
        bounds.append((0.01, 0.5))
    return bounds

def constraints_fn(n):
    cons = []
    
    def overlap_con(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        c = []
        for i in range(n):
            for j in range(i+1, n):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                c.append(dist_sq - (radii[i] + radii[j])**2)
        return np.array(c)
    
    def boundary_con(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        c = []
        for i in range(n):
            c.append(centers[i][0] - radii[i])
            c.append(1 - centers[i][0] - radii[i])
            c.append(centers[i][1] - radii[i])
            c.append(1 - centers[i][1] - radii[i])
        return np.array(c)
    
    def radius_con(x):
        return x[2*n:]
    
    cons.append({'type': 'ineq', 'fun': overlap_con})
    cons.append({'type': 'ineq', 'fun': boundary_con})
    cons.append({'type': 'ineq', 'fun': radius_con})
    
    return cons

def validate_and_fix(centers, radii, n):
    for _ in range(200):
        changed = False
        for i in range(n):
            max_r = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
            radii[i] = min(radii[i], max_r)
            radii[i] = max(0, radii[i])
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-10:
                    total_r = radii[i] + radii[j]
                    if total_r > 1e-10:
                        scale = dist / total_r * 0.99995
                        radii[i] *= scale
                        radii[j] *= scale
                        changed = True
        
        if not changed:
            break
    
    return centers, radii

def run_packing():
    n = 26
    
    centers, radii = hexagonal_init(n)
    x0 = np.concatenate([centers.flatten(), radii])
    
    best_x = x0.copy()
    best_obj = objective(x0, n)
    
    constraints = constraints_fn(n)
    bounds = bounds_fn(n)
    
    for trial in range(80):
        if trial > 0:
            x_trial = best_x + np.random.normal(0, 0.0015, len(best_x))
            x_trial[:2*n] = np.clip(x_trial[:2*n], 0.08, 0.92)
            x_trial[2*n:] = np.clip(x_trial[2*n:], 0.03, 0.4)
            x0 = x_trial
        
        try:
            result = minimize(objective, x0, args=(n,), method='SLSQP',
                            bounds=bounds,
                            constraints=constraints,
                            options={'maxiter': 800, 'ftol': 1e-16})
            
            if result.fun < best_obj - 1e-10:
                best_x = result.x.copy()
                best_obj = result.fun
        except Exception:
            pass
    
    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]
    
    centers, radii = validate_and_fix(centers, radii, n)
    
    return centers, radii, np.sum(radii)
