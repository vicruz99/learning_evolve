import numpy as np
from scipy.optimize import minimize
import math

def compute_interstitial_radius(r):
    """Calculate the radius of a circle that fits in the gap of three mutually tangent circles of radius r."""
    if r <= 0: return 0
    return r * (2 / math.sqrt(3) - 1)

def generate_initial_configuration():
    """Generates a starting hexagonal packing of 25 circles with a 26th candidate."""
    r = 0.1
    centers = []
    radii = []
    
    # 5 rows
    for i in range(5):
        row_radius = r
        y = r + i * r * math.sqrt(3)
        
        # Offset odd rows
        shift = r if i % 2 == 1 else 0
        count = 4 if i % 2 == 1 else 5
        
        for j in range(count):
            x = r + shift + j * 2 * r
            centers.append([x, y])
            radii.append(row_radius)
            
    # 26th circle inserted into a void at the center of the arrangement
    # Targeting a gap between row 1 and row 2
    gap_radius = compute_interstitial_radius(r)
    # Approximate center for the gap in a standard hex grid
    center_x = 0.5 
    center_y = r * math.sqrt(3) 
    centers.append([center_x, center_y])
    radii.append(gap_radius)
    
    return np.array(centers), np.array(radii)

def calculate_loss(optim_params, n):
    """Objective function to minimize (negative sum of radii + penalty for overlaps/boundaries)."""
    penalty_scale = 1000.0
    radius_scale = 1.0
    
    # Unpack parameters
    centers = optim_params[:2*n].reshape((n, 2))
    radii = optim_params[2*n:]
    
    # Clip radii to be non-negative
    radii = np.clip(radii, 1e-9, 0.5)
    
    loss = -np.sum(radii) # We want to maximize sum, so minimize negative sum
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Penalize if circle is outside [0,1]
        if x - r < 0: loss += penalty_scale * (x - r)
        if x + r > 1: loss += penalty_scale * (x + r - 1)
        if y - r < 0: loss += penalty_scale * (y - r)
        if y + r > 1: loss += penalty_scale * (y + r - 1)
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                loss += penalty_scale * (min_dist - dist)
                
    return loss

def run_packing():
    n = 26
    
    # 1. Generate initial configuration
    init_centers, init_radii = generate_initial_configuration()

    # Using Nelder-Mead as it handles non-differentiable penalty landscapes well without gradients
    res = minimize(
        calculate_loss, 
        init_params, 
        args=(n,), 
        method='Nelder-Mead', 
        options={'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 50000, 'adaptive': True}
    )

    optimal_params = res.x

    radii = np.clip(radii, 1e-9, 0.5)

        r = radii[i]
        # Ensure centers allow the circle to be inside
        centers[i, 0] = np.clip(x, r, 1 - r)
        centers[i, 1] = np.clip(y, r, 1 - r)

    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)
