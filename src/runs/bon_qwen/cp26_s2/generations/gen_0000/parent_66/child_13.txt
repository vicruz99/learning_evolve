import numpy as np
from scipy.optimize import minimize


def compute_max_radii(centers, max_iter=300):
    """Compute maximum feasible radii for given centers."""
    n = centers.shape[0]
    # Initialize with boundary-limited radii
    radii = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    radii = np.maximum(radii, 0.0)

    for iteration in range(max_iter):
        max_change = 0.0
        for i in range(n):
            old_r = radii[i]
            # Compute distances to all other centers
            dx = centers[:, 0] - centers[i, 0]
            dy = centers[:, 1] - centers[i, 1]
            dists = np.sqrt(dx * dx + dy * dy)
            dists[i] = np.inf  # Exclude self-distance
            # Constraint: r_i <= dist(i,j) - r_j for all j
            candidates = dists - radii
            new_r = min(old_r, np.min(candidates))
            new_r = max(new_r, 0.0)
            if abs(new_r - old_r) > 1e-12:
                radii[i] = new_r
                max_change = max(max_change, abs(new_r - old_r))
        if max_change < 1e-12:
            break
    return radii


def objective_function(centers_flat, n):
    """Negative sum of radii for given centers."""
    centers = centers_flat.reshape(n, 2)
    radii = compute_max_radii(centers)
    return -np.sum(radii)


def generate_hexagonal_init(n):
    """Generate initial hexagonal grid configuration."""
    row_counts = [6, 5, 6, 5, 4]
    centers = []
    r = 0.10
    dx = 2.05 * r
    dy = dx * np.sqrt(3.0) / 2.0

    idx = 0
    row_idx = 0
    for count in row_counts:
        for col in range(count):
            x = 0.08 + col * dx
            if row_idx % 2 == 1:
                x += dx / 2.0
            y = 0.08 + row_idx * dy
            centers.append([x, y])
            idx += 1
        row_idx += 1

    return np.array(centers[:n])


def generate_random_init(n, seed=None):
    """Generate a random initial configuration."""
    if seed is not None:
        rng = np.random.RandomState(seed)
    else:
        rng = np.random.RandomState()

    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i, 0] = rng.uniform(0.05, 0.95)
        centers[i, 1] = rng.uniform(0.05, 0.95)
    return centers


def run_packing():
    n = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Phase 1: Powell optimization from multiple hexagonal starting points
    for trial in range(8):
        centers0 = generate_hexagonal_init(n).copy()

        if trial > 0:
            perturbation = np.random.randn(n, 2) * 0.015 * (1.0 + trial * 0.5)
            centers0 = centers0 + perturbation
            centers0 = np.clip(centers0, 0.03, 0.97)

        x0 = centers0.flatten()
        bounds = [(0.01, 0.99) for _ in range(2 * n)]

        try:
            result = minimize(
                objective_function,
                x0,
                args=(n,),
                method='Powell',
                bounds=bounds,
                options={
                    'maxiter': 5000,
                    'xtol': 1e-9,
                    'ftol': 1e-9,
                    'disp': False
                }
            )

            centers = result.x.reshape(n, 2)
            radii = compute_max_radii(centers)
            total = np.sum(radii)

            if total > best_sum:
                best_sum = total
                best_centers = centers.copy()
                best_radii = radii.copy()
        except Exception:
            continue

    # Phase 2: Try some random initial configurations
    for trial in range(5):
        centers0 = generate_random_init(n, seed=trial * 42)
        x0 = centers0.flatten()
        bounds = [(0.01, 0.99) for _ in range(2 * n)]

        try:
            result = minimize(
                objective_function,
                x0,
                args=(n,),
                method='Powell',
                bounds=bounds,
                options={
                    'maxiter': 3000,
                    'xtol': 1e-9,
                    'ftol': 1e-9,
                    'disp': False
                }
            )

            centers = result.x.reshape(n, 2)
            radii = compute_max_radii(centers)
            total = np.sum(radii)

            if total > best_sum:
                best_sum = total
                best_centers = centers.copy()
                best_radii = radii.copy()
        except Exception:
            continue

    # Phase 3: Local refinement using coordinate descent with simulated annealing
    centers = best_centers.copy()
    radii = best_radii.copy()
    current_sum = best_sum

    temperature = 0.005
    cooling_rate = 0.9998

    for step in range(50000):
        i = np.random.randint(n)
        step_size = 0.008 * (1.0 + temperature)

        old_x = centers[i, 0]
        old_y = centers[i, 1]

        # Try perturbation
        centers[i, 0] = old_x + np.random.uniform(-step_size, step_size)
        centers[i, 1] = old_y + np.random.uniform(-step_size, step_size)
        centers[i, 0] = np.clip(centers[i, 0], 0.02, 0.98)
        centers[i, 1] = np.clip(centers[i, 1], 0.02, 0.98)

        new_radii = compute_max_radii(centers)
        new_sum = np.sum(new_radii)
        delta = new_sum - current_sum

        # Acceptance criterion (simulated annealing)
        accept = delta > 0 or (temperature > 1e-10 and np.random.rand() < np.exp(delta / max(temperature, 1e-10)))

        if accept:
            current_sum = new_sum
            radii = new_radii
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            centers[i, 0] = old_x
            centers[i, 1] = old_y

        temperature *= cooling_rate

    # Final cleanup: ensure all values are valid
    best_radii = np.maximum(best_radii, 0.0)

    # Verify and adjust if needed
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        best_centers[i, 0] = np.clip(x, r + 1e-12, 1.0 - r - 1e-12)
        best_centers[i, 1] = np.clip(y, r + 1e-12, 1.0 - r - 1e-12)

    # Recompute radii after clipping to ensure validity
    final_radii = compute_max_radii(best_centers)
    final_sum = np.sum(final_radii)

    if final_sum < best_sum - 1e-6:
        best_radii = final_radii
        best_sum = final_sum

    return best_centers, best_radii, best_sum
