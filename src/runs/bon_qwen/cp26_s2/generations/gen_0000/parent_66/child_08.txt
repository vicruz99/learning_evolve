1.  **Boundary Constraints**: $r_i \le x_i \le 1 - r_i$ and $r_i \le y_i \le 1 - r_i$ for all $i$.
2.  **Non-overlap Constraints**: The Euclidean distance between centers of any two circles $i$ and $j$ must be at least the sum of their radii, i.e., $\sqrt{(x_i - x_j)^2 + (y_i - y_j)^2} \ge r_i + r_j$.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26,) with radius of each circle
        sum_radii: float, the sum of radii
    """
    n = 26
    
    # Set seed for reproducibility
    np.random.seed(42)
    
    # 1. Initialization
    # Start with a valid configuration to help the optimizer.
    # Place circles randomly in the square with small radii.
    # We leave a margin of 0.1 from boundaries to ensure initial validity.
    centers_init = np.random.uniform(0.1, 0.9, size=(n, 2))
    radii_init = np.full(n, 0.02)  # Small initial radius
    
    # Flatten variables into a single vector: [x1, y1, r1, x2, y2, r2, ...]
    # Total size: 3 * n
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers_init[i, 0]
        x0[3 * i + 1] = centers_init[i, 1]
        x0[3 * i + 2] = radii_init[i]
        
    # 2. Bounds
    # x, y must be in [0, 1]
    # r must be >= 0. Upper bound for r is 0.5 (diameter 1).
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x_i
        bounds.append((0.0, 1.0))  # y_i
        bounds.append((0.0, 0.5))  # r_i
        
    # 3. Constraints
    # We define two types of constraints:
    # 1. Boundary constraints: circles must be inside [0,1]x[0,1]
    #    x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    # 2. Overlap constraints: distance(i,j) >= r_i + r_j
    #    sqrt((x_i-x_j)^2 + (y_i-y_j)^2) - r_i - r_j >= 0
    
    constraints = []
    
    # Boundary constraints function
    def boundary_constraints(vars):
        vals = np.zeros(4 * n)
        idx = 0
        for i in range(n):
            x = vars[3 * i]
            y = vars[3 * i + 1]
            r = vars[3 * i + 2]
            vals[idx] = x - r
            vals[idx + 1] = 1.0 - x - r
            vals[idx + 2] = y - r
            vals[idx + 3] = 1.0 - y - r
            idx += 4
        return vals

    constraints.append({'type': 'ineq', 'fun': boundary_constraints})
    
    # Overlap constraints function
    # There are n*(n-1)/2 pairs
    num_pairs = n * (n - 1) // 2
    
    def overlap_constraints(vars):
        vals = np.zeros(num_pairs)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                xi = vars[3 * i]
                yi = vars[3 * i + 1]
                ri = vars[3 * i + 2]
                
                xj = vars[3 * j]
                yj = vars[3 * j + 1]
                rj = vars[3 * j + 2]
                
                dx = xi - xj
                dy = yi - yj
                dist = np.sqrt(dx*dx + dy*dy)
                
                # Constraint: dist >= ri + rj  =>  dist - ri - rj >= 0
                vals[idx] = dist - ri - rj
                idx += 1
        return vals

    constraints.append({'type': 'ineq', 'fun': overlap_constraints})
    
    # 4. Objective Function
    # We want to maximize sum(r_i), so we minimize -sum(r_i)
    def objective(vars):
        r_sum = 0.0
        for i in range(n):
            r_sum += vars[3 * i + 2]
        return -r_sum

    # 5. Optimization
    # Use SLSQP which supports non-linear constraints and bounds.
    # maxiter is set high to allow convergence.
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    
    # 6. Extract Results
    # Check if optimization was successful
    if not res.success:
        # If failed, we still return the best found solution
        pass
    
    best_x = res.x
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = best_x[3 * i]
        centers[i, 1] = best_x[3 * i + 1]
        radii[i] = best_x[3 * i + 2]
        
    sum_radii = np.sum(radii)
    
    # Optional: Post-processing to ensure strict validity (though solver should handle it)
    # Sometimes numerical errors might cause tiny violations.
    # We can clip radii to be non-negative and centers to be within bounds relative to radii.
    # But the constraints enforced this.
    
    return centers, radii, float(sum_radii)

# Function to validate the packing (provided in prompt, for reference)
def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    print(f"Sum of radii: {sum_r}")
    is_valid = validate_packing(centers, radii)
