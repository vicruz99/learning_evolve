import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


class ObjectiveFunction:
    def __call__(self, x):
        radii = x[2 * N_CIRCLES:]
        return -np.sum(radii)


class PairwiseConstraint:
    def __init__(self, i, j):
        self.i = i
        self.j = j

    def __call__(self, x):
        cx = x[0:N_CIRCLES]
        cy = x[N_CIRCLES:2 * N_CIRCLES]
        radii = x[2 * N_CIRCLES:]
        dx = cx[self.i] - cx[self.j]
        dy = cy[self.i] - cy[self.j]
        dist = np.sqrt(dx ** 2 + dy ** 2)
        return dist - radii[self.i] - radii[self.j]


class LeftConstraint:
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        cx = x[0:N_CIRCLES]
        radii = x[2 * N_CIRCLES:]
        return cx[self.i] - radii[self.i]


class RightConstraint:
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        cx = x[0:N_CIRCLES]
        radii = x[2 * N_CIRCLES:]
        return 1.0 - cx[self.i] - radii[self.i]


class BottomConstraint:
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        cy = x[N_CIRCLES:2 * N_CIRCLES]
        radii = x[2 * N_CIRCLES:]
        return cy[self.i] - radii[self.i]


class TopConstraint:
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        cy = x[N_CIRCLES:2 * N_CIRCLES]
        radii = x[2 * N_CIRCLES:]
        return 1.0 - cy[self.i] - radii[self.i]


class RadiusConstraint:
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        radii = x[2 * N_CIRCLES:]
        return radii[self.i]


def build_constraints():
    constraints = []
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            constraints.append({'type': 'ineq', 'fun': PairwiseConstraint(i, j)})
    for i in range(N_CIRCLES):
        constraints.append({'type': 'ineq', 'fun': LeftConstraint(i)})
        constraints.append({'type': 'ineq', 'fun': RightConstraint(i)})
        constraints.append({'type': 'ineq', 'fun': BottomConstraint(i)})
        constraints.append({'type': 'ineq', 'fun': TopConstraint(i)})
        constraints.append({'type': 'ineq', 'fun': RadiusConstraint(i)})
    return constraints


def generate_hexagonal_initial():
    rows = [6, 5, 6, 5, 4]
    r_init = 0.075

    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, r_init)

    idx = 0
    for row_idx, count in enumerate(rows):
        y = r_init + row_idx * np.sqrt(3.0) * r_init
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = 2.0 * r_init
        for col in range(count):
            x = x_start + col * 2.0 * r_init
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def generate_grid_initial():
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.07)

    spacing = 0.18
    positions = []
    for i in range(5):
        for j in range(5):
            positions.append([spacing + j * spacing * 0.9, spacing + i * spacing * 0.9])
    positions.append([0.5, 0.5])

    for idx, pos in enumerate(positions):
        centers[idx] = pos

    return centers, radii


def run_packing():
    constraints = build_constraints()
    bounds = [(0, 1)] * (2 * N_CIRCLES) + [(0, 0.5)] * N_CIRCLES
    obj_func = ObjectiveFunction()

    best_centers = None
    best_radii = None
    best_sum = -1.0

    initial_configs = []

    c1, r1 = generate_hexagonal_initial()
    initial_configs.append((c1, r1))

    c2, r2 = generate_grid_initial()
    initial_configs.append((c2, r2))

    c3, r3 = generate_hexagonal_initial()
    c3 = c3 + np.random.RandomState(42).randn(*c3.shape) * 0.02
    c3 = np.clip(c3, 0, 1)
    initial_configs.append((c3, r3))

    for config_idx, (centers, radii) in enumerate(initial_configs):
        x0 = np.concatenate([centers.ravel(), radii])

        for restart in range(8):
            if restart == 0:
                x_try = x0.copy()
            else:
                x_try = x0.copy()
                noise = np.random.randn(len(x_try)) * 0.015
                x_try[:2 * N_CIRCLES] += noise[:2 * N_CIRCLES]
                x_try[:2 * N_CIRCLES] = np.clip(x_try[:2 * N_CIRCLES], 0.02, 0.98)
                x_try[2 * N_CIRCLES:] = np.clip(x_try[2 * N_CIRCLES:], 0.01, 0.3)

            try:
                result = minimize(
                    obj_func,
                    x_try,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
                )

                current_sum = -result.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = result.x[:2 * N_CIRCLES].reshape((N_CIRCLES, 2))
                    best_radii = result.x[2 * N_CIRCLES:].copy()
            except Exception:
                continue

    if best_centers is None:
        centers, radii = generate_hexagonal_initial()
        return centers, radii, np.sum(radii)

    return best_centers, best_radii, np.sum(best_radii)
