import numpy as np
from scipy.optimize import minimize

def compute_penalty(pos, n, r, upper_indices):
    """
    Computes the squared penalty for constraint violations.
    Returns 0 if all circles are within bounds and non-overlapping.
    """
    c = pos.reshape(n, 2)
    p = 0.0
    
    # Boundary penalties: push centers away from edges if too close
    p += np.sum(np.maximum(0, r - c[:, 0])**2)
    p += np.sum(np.maximum(0, c[:, 0] - (1 - r))**2)
    p += np.sum(np.maximum(0, r - c[:, 1])**2)
    p += np.sum(np.maximum(0, c[:, 1] - (1 - r))**2)
    
    # Pairwise distance penalties: push centers apart if overlapping
    diff = c[np.newaxis, :, :] - c[:, np.newaxis, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    violations = np.maximum(0, 2 * r - dist)
    p += np.sum(violations[upper_indices]**2)
    
    return p

def run_packing():
    n = 26
    # Precompute indices for upper triangle to avoid double-counting pairs
    upper_indices = np.triu_indices(n, k=1)
    
    # 1. Initialize centers on a hexagonal grid
    centers = []
    cols = 6
    rows = 5
    x_step = 1.0 / (cols + 1)
    y_step = 1.0 / (rows + 1)
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            # Stagger every other row for hexagonal packing
            x = x_step * (j + 1 + (i % 2) * 0.5)
            y = y_step * (i + 1)
            centers.append([x, y])
        if len(centers) >= n:
            break
            
    centers = np.array(centers[:n])
    
    # Small random perturbation to break symmetry and help escape local minima
    np.random.seed(42)
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    
    # 2. Iteratively increase radius while optimizing positions
    r = 0.05
    delta = 0.0003
    
    for _ in range(1500):
        bnds = [(r, 1 - r) for _ in range(2 * n)]
        res = minimize(compute_penalty, centers.flatten(), 
                       args=(n, r, upper_indices),
                       method='L-BFGS-B', bounds=bnds,
                       options={'ftol': 1e-15, 'maxiter': 300})
        centers = res.x.reshape(n, 2)
        
        # If feasible (penalty ~0), try a larger radius
        if res.fun < 1e-12:
            r += delta
        else:
            # Otherwise, shrink step size to refine current radius
            delta *= 0.9
            if delta < 1e-7:
                break
                
    # 3. Final high-precision refinement at the largest feasible radius
    bnds = [(r, 1 - r) for _ in range(2 * n)]
    res = minimize(compute_penalty, centers.flatten(),

    r = r - 1e-7
    radii = np.full(n, r)
    
    return centers, radii, np.sum(radii)
