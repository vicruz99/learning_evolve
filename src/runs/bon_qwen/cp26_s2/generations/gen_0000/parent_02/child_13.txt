import numpy as np
from scipy.optimize import minimize, linprog


def compute_optimal_radii(centers, n):
    """Given center positions, compute optimal radii via LP to maximize sum of radii.
    
    Maximizes sum(r_i) subject to:
    - r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i (boundary)
    - r_i + r_j <= ||c_i - c_j|| for all i < j (non-overlap)
    - r_i >= 0
    """
    c_obj = -np.ones(n)  # maximize sum(r_i) = minimize -sum(r_i)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        upper = min(x, 1 - x, y, 1 - y)
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(max(1e-14, upper))
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(c_i, c_j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(max(1e-14, dist))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        radii = np.maximum(result.x, 1e-14)
        return radii
    else:
        return np.full(n, 0.01)


def neg_sum_radii(centers_flat, n):
    """Objective function for center optimization: negative sum of optimal radii."""
    centers = centers_flat.reshape(n, 2)
    radii = compute_optimal_radii(centers, n)
    return -np.sum(radii)


def initialize_hexagonal(n):
    """Initialize centers in a hexagonal offset pattern.
    
    For 26 circles: 5 rows with distribution [5, 5, 5, 5, 6]
    Odd rows are horizontally offset for hexagonal packing.
    """
    rows = [5, 5, 5, 5, 6]
    
    centers = []
    y_positions = np.linspace(1 / 6, 5 / 6, 5)
    
    for row_idx, count in enumerate(rows):
        y = y_positions[row_idx]
        x_spacing = 1.0 / (count + 1)
        x_positions = [x_spacing * (i + 1) for i in range(count)]
        
        # Offset odd rows for hexagonal pattern
        if row_idx % 2 == 1:
            x_positions = [x + x_spacing / 2 for x in x_positions]
        
        for x in x_positions:
            centers.append([x, y])
    
    return np.array(centers)


def initialize_dense_grid(n):
    """Alternative initialization: dense grid pattern."""
    # Try to place in a 5x5 grid with one extra
    rows = [5, 5, 6, 5, 5]  # 26 total
    
    centers = []
    y_positions = np.linspace(0.125, 0.875, 5)
    
    for row_idx, count in enumerate(rows):
        y = y_positions[row_idx]
        x_spacing = 1.0 / (count + 1)
        x_positions = [x_spacing * (i + 1) for i in range(count)]
        
        for x in x_positions:
            centers.append([x, y])
    
    return np.array(centers)


def run_packing() -> tuple:
    """Pack 26 circles in unit square to maximize sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds for center coordinates (keep circles strictly inside)
    bounds = [(0.001, 0.999)] * (2 * n)
    
    # Try multiple initialization strategies and restarts
    initializations = [
        initialize_hexagonal(n),
        initialize_dense_grid(n),
    ]
    
    for init_idx, centers0 in enumerate(initializations):
        for restart in range(35):
            if restart == 0:
                x0 = centers0.flatten().copy()
            else:
                # Random perturbation of the base initialization
                perturbation = np.random.uniform(-0.06, 0.06, (n, 2))
                x0 = np.clip(centers0 + perturbation, 0.02, 0.98).flatten()
            
            try:
                # Optimize centers using SLSQP
                result = minimize(
                    neg_sum_radii,
                    x0,
                    args=(n,),
                    method='SLSQP',
                    bounds=bounds,
                    options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
                )
                
                centers_opt = result.x.reshape(n, 2)
                radii_opt = compute_optimal_radii(centers_opt, n)
                s = np.sum(radii_opt)
                
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
                    
            except Exception:
                continue
    
    # Final cleanup: ensure all values are valid
    best_radii = np.maximum(best_radii, 1e-12)
    
    # Verify and adjust if needed
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        # Ensure circle is inside unit square
        best_radii[i] = min(r, x, 1 - x, y, 1 - y, best_radii[i])
        best_radii[i] = max(best_radii[i], 1e-12)
    
    # Adjust radii to ensure no overlaps (conservative adjustment)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((best_centers[i] - best_centers[j]) ** 2))
            if dist < best_radii[i] + best_radii[j]:
                # Reduce radii proportionally
                excess = best_radii[i] + best_radii[j] - dist
                reduction = excess / 2 + 1e-12
                best_radii[i] = max(best_radii[i] - reduction, 1e-12)
                best_radii[j] = max(best_radii[j] - reduction, 1e-12)
    
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
