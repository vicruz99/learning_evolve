import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Negative sum of radii to be minimized."""
    return -np.sum(vars[2::3])

def boundary_con(vars):
    """Boundary constraints: circles inside [0,1]^2."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    # Returns array where each element must be >= 0
    return np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])

def overlap_con(vars):
    """Non-overlap constraints: distance between centers >= sum of radii."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    num_constraints = N * (N - 1) // 2
    cons = np.empty(num_constraints)
    idx = 0
    for i in range(N):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, N):
            dist = np.hypot(xi - x[j], yi - y[j])
            cons[idx] = dist - ri - r[j]
            idx += 1
    return cons

def run_packing():
    constraints = [
        {'type': 'ineq', 'fun': boundary_con},
        {'type': 'ineq', 'fun': overlap_con}
    ]
    
    # Bounds: x,y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(1e-6, 0.5)] * N
    
    best_vars = None
    best_sum_r = -np.inf
    rng = np.random.RandomState(2024)
    
    # Run multiple trials from perturbed hexagonal lattices
    for trial in range(8):
        # Generate hex grid layout
        centers = []
        s = 0.24 + rng.uniform(-0.02, 0.02)
        for i in range(8):
            y = i * s * np.sqrt(3) / 2.0
            if y > 1.0:
                break
            x_start = 0.0 if i % 2 == 0 else s / 2.0
            for j in range(7):
                x = x_start + j * s
                if x > 1.0:
                    break
                centers.append([x, y])
                
        if len(centers) < N:
            centers = rng.rand(N, 2) * 0.8 + 0.1
        else:
            centers = np.array(centers[:N])
            
        # Add small noise to break symmetry and avoid flat valleys
        centers += rng.randn(N, 2) * 0.015
        centers = np.clip(centers, 0.15, 0.85)
        
        # Initial radii: small enough to guarantee feasibility
        r_init = np.full(N, 0.045)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = r_init
        
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints, 
            options={'maxiter': 1500, 'ftol': 1e-10, 'disp': False}
        )
        
        if res.success:
            current_sum_r = -res.fun
            if current_sum_r > best_sum_r:
                best_sum_r = current_sum_r
                best_vars = res.x
                
    if best_vars is None:
        # Fallback to last attempted configuration if optimization failed entirely
        centers_fallback = np.clip(centers, 0.1, 0.9)
        radii_fallback = np.full(N, 0.04)
        return centers_fallback, radii_fallback, np.sum(radii_fallback)
        
    centers_opt = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii_opt = best_vars[2::3]
    
    # Ensure strict non-negativity and clip tiny numerical errors
    radii_opt = np.maximum(radii_opt, 1e-7)
    
    return centers_opt, radii_opt, best_sum_r
