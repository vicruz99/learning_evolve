def compute_overlap_violations(centers, radii):
    """Compute constraint violations for non-overlap and boundary."""
    n = len(radii)
    violations = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        violations.append(r - x)  # x - r >= 0  => r - x <= 0
        violations.append(r - (1 - x))  # 1 - x - r >= 0 => r - (1-x) <= 0
        violations.append(r - y)
        violations.append(r - (1 - y))
    
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            violations.append(radii[i] + radii[j] - dist)  # dist - (r_i + r_j) >= 0 => r_i + r_j - dist <= 0
    
    return np.array(violations)

def compute_pairwise_dist_constraint(params, i, j):
    """Constraint: distance(i,j) >= r_i + r_j, i.e., dist - r_i - r_j >= 0"""
    ci = params[2 * i:2 * i + 2]
    cj = params[2 * j:2 * j + 2]
    ri = params[2 * 26 + i]
    rj = params[2 * 26 + j]
    dist = np.sqrt(np.sum((ci - cj) ** 2))
    return dist - ri - rj


def compute_pairwise_jacobian(params, i, j, n=26):
    """Jacobian of pairwise constraint."""
    jac = np.zeros(3 * n)
    ci = params[2 * i:2 * i + 2]
    cj = params[2 * j:2 * j + 2]
    ri = params[2 * n + i]
    rj = params[2 * n + j]
    diff = ci - cj
    dist = np.sqrt(np.sum(diff ** 2))
    
    if dist > 1e-10:
        # d(dist)/d(ci) = diff / dist
        jac[2 * i:2 * i + 2] = diff / dist
        jac[2 * j:2 * j + 2] = -diff / dist
    
    jac[2 * n + i] = -1
    jac[2 * n + j] = -1
    
    return jac

        row = idx // 5
        col = idx % 5
        centers[idx] = [0.1 + col * 0.2, 0.1 + row * 0.2]
        idx += 1

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
