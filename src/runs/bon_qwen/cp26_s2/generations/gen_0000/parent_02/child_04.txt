

To maximize the sum of radii for 26 circles in a unit square, we must solve a non-convex optimization problem. The strategy involves a two-level approach: an outer optimization of circle centers using `scipy.optimize.minimize` and an inner Linear Programming (LP) problem to calculate the maximum valid radii for a given set of centers.

### Strategy
1.  **Initialization**: Start with a dense, regular arrangement (a perturbed 6x5 grid) to ensure circles are initially well-separated and have non-zero radii.
2.  **Inner Loop (LP)**: For any fixed set of centers, the maximum sum of radii is determined by a Linear Program. The constraints ensure that $r_i + r_j \leq \|c_i - c_j\|$ (no overlap) and $r_i \leq \text{distance to boundary}$ (inside square).
3.  **Outer Loop (Center Optimization)**: We maximize the sum of radii from the LP. We use the Nelder-Mead simplex algorithm, which is effective for non-smooth functions. Multiple random restarts are used to avoid local minima.
4.  **Validation**: The final configuration is adjusted slightly to ensure it is strictly valid (removing tiny overlaps) before returning the results.

import numpy as np
import scipy.optimize
import scipy.spatial
import warnings

# Suppress warnings for cleaner output during optimization
warnings.filterwarnings('ignore')

def run_packing():
    """
    Returns (centers, radii, sum_radii) for 26 circles in a unit square.
    """
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    def solve_radii(centers):
        """Solves the LP to find the maximum sum of radii for fixed centers."""
        n = centers.shape[0]
        dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
        
        # Objective: minimize -sum(r)
        c_obj = np.ones(n) * -1.0
        
        # Constraints: A_ub * r <= b_ub
        # Pairwise constraints
        pairs = np.array(np.triu_indices(n, k=1))
        n_pairs = pairs.shape[1]
        A_ub = np.zeros((n_pairs + 4 * n, n))
        b_ub = np.zeros(n_pairs + 4 * n)
        
        # Pairwise distances
        A_ub[:n_pairs, pairs[0]] += 1.0
        A_ub[:n_pairs, pairs[1]] += 1.0
        b_ub[:n_pairs] = dists[pairs[0], pairs[1]]
        
        # Boundary constraints
        row_idx = n_pairs
        for i in range(n):
            A_ub[row_idx + 0, i] = 1.0; b_ub[row_idx + 0] = centers[i, 0]
            A_ub[row_idx + 1, i] = 1.0; b_ub[row_idx + 1] = 1.0 - centers[i, 0]
            A_ub[row_idx + 2, i] = 1.0; b_ub[row_idx + 2] = centers[i, 1]
            A_ub[row_idx + 3, i] = 1.0; b_ub[row_idx + 3] = 1.0 - centers[i, 1]
            row_idx += 4

        # Bounds: 0 <= r_i
        bounds = [(0.0, None)] * n

        try:
            res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                return -res.fun, res.x
            return 0.0, np.zeros(n)
        except Exception:
            return 0.0, np.zeros(n)

    def objective(centers_flat):
        centers = centers_flat.reshape(N, 2)
        # Penalize centers outside square (though bounds should handle this)
        if np.any(centers < 0) or np.any(centers > 1):
            return 1000.0
        # Solve LP
        s, r = solve_radii(centers)
        # We maximize sum of radii, so minimize negative sum
        return -s

    # Optimization loop with multiple restarts
    bounds_52 = [(0.0, 1.0)] * (N * 2)
    
    for seed in range(20):
        np.random.seed(seed)
        # Initialize with a hexagonal-like grid
        x_vals = np.linspace(0.05, 0.95, 6)
        y_vals = np.linspace(0.05, 0.95, 5)
        grid = np.array([[x, y] for x in x_vals for y in y_vals])
        
        # Take the first N points and add a small jitter
        init_centers = grid[:N].copy()
        init_centers += np.random.uniform(-0.02, 0.02, size=init_centers.shape)
        init_centers = np.clip(init_centers, 0.0, 1.0)
        
        # Use Nelder-Mead for center optimization
        try:
            res = scipy.optimize.minimize(objective, init_centers.flatten(), method='Nelder-Mead', 
                                         options={'xatol': 1e-5, 'fatol': 1e-6, 'maxiter': 2000})
            
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape(N, 2).copy()
                _, best_radii = solve_radii(best_centers)
        except Exception:
            continue

    # Final strict validation adjustment (shrink radii slightly to remove numerical overlap)
    if best_radii is not None:
        n = len(best_radii)
        dists = np.sqrt(np.sum((best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]) ** 2, axis=2))
        overlap = np.maximum(0, (best_radii[:, None] + best_radii[None, :] - dists) + 1e-9)
        max_overlap = np.max(overlap)
        
        # If there is any significant overlap, shrink radii
        if max_overlap > 1e-8:
            scale = (1.0 - 1.5e-9) # Just a tiny shrink to be safe if numerical
            best_radii = best_radii * (1.0 - max_overlap / (np.sum(best_radii) + 1e-10) * 0.001)
            
    # Recalculate final sum
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, final_sum

centers, radii, sum_r = run_packing()

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False
    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False
    return True

print(f"Sum of radii: {sum_r:.6f}")
