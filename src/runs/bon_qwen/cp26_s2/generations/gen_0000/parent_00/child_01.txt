import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_max_radius(centers, n):
    """
    Computes the maximum equal radius that fits for given centers.
    Constraints: distance to boundaries, distance between circles.
    """
    # Distance to boundaries
    dist_bound = np.min([
        np.min(centers[:, 0]),
        np.min(1.0 - centers[:, 0]),
        np.min(centers[:, 1]),
        np.min(1.0 - centers[:, 1])
    ])
    
    # Pairwise distances
    # Shape: (n, n)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Extract lower triangle distances (i < j)
    lower_tri = np.tril(np.ones((n, n), dtype=bool), k=-1)
    dists_pairs = dists[lower_tri]
    dist_min_pair = np.min(dists_pairs)
    
    # Radius is limited by the tightest constraint
    return min(dist_bound, dist_min_pair / 2.0)

def objective_func(params, n):
    """
    Objective function for optimization: negative of max feasible radius.
    """
    centers = params.reshape(n, 2)
    return -compute_max_radius(centers, n)

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # 1. Global search using Differential Evolution
    res_de = differential_evolution(
        objective_func,
        bounds,
        args=(n,),
        popsize=15,
        maxiter=500,
        seed=42,
        workers=1,
        mutation=(0.5, 1.5),
        recombination=0.7,
        tol=1e-7
    )
    
    # 2. Local refinement using Nelder-Mead
    res_local = minimize(
        objective_func,
        res_de.x,
        args=(n,),
        method='Nelder-Mead',
        options={'maxiter': 3000, 'xatol': 1e-6, 'fatol': 1e-6}
    )
    
    best_centers = res_local.x.reshape(n, 2)
    best_r = -res_local.fun
    
    # Ensure strict validity by clamping centers to feasible region
    # This handles any minor numerical drift from the optimizer
    best_centers = np.clip(best_centers, best_r, 1.0 - best_r)
    
    radii = np.full(n, best_r)
    return best_centers, radii, np.sum(radii)
