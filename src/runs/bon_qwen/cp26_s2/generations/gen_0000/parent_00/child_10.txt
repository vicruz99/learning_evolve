def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Returns (sum_radii, radii).
    """
    n = centers.shape[0]
    
    # Clip centers to [0, 1] to handle any out-of-bounds from optimizer
    centers = np.clip(centers, 0.0, 1.0)
    
    # Objective: max sum(r) <=> min -sum(r)
    c_obj = -np.ones(n)

    # We need to build the sparse matrix A_ub for these constraints
    # A_ub @ r <= b_ub
    
    num_constraints = n * (n - 1) // 2
    rows = np.zeros(num_constraints * 2, dtype=int)
    cols = np.zeros(num_constraints * 2, dtype=int)
    data = np.ones(num_constraints * 2)
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))

            cols[idx] = i
            idx += 1
            rows[idx] = k
            cols[idx] = j
            idx += 1
            b_ub[k] = dist
            k += 1
