import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Finds a packing of 26 circles in a unit square [0,1]x[0,1] that maximizes 
    the sum of radii.
    """
    n_circles = 26
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None

    # Generate multiple initial configurations to escape local minima
    initial_configs = []

    # 1. Hexagonal-ish grid
    centers_hex = np.zeros((n_circles, 2))
    # Try to fit in a pattern. 5x5 grid is 25. Add 1 at center.
    # Let's create a dense packing
    row_y = 0.1
    col_x = 0.1
    idx = 0
    for r in range(5):
        for c in range(5):
            if idx < n_circles:
                # Stagger rows
                shift = 0.1 if r % 2 == 1 else 0
                centers_hex[idx] = [col_x + c * 0.2 + shift, row_y + r * 0.2 * 0.866]
                idx += 1
    if idx < n_circles:
        centers_hex[idx] = [0.5, 0.5]
    initial_configs.append(centers_hex)

    # 2. Random initialization
    np.random.seed(42)
    centers_rand = np.random.rand(n_circles, 2)
    initial_configs.append(centers_rand)

    # 3. Another grid, maybe shifted
    centers_grid = np.zeros((n_circles, 2))
    idx = 0
    for r in range(5):
        for c in range(6):
            if idx < n_circles:
                centers_grid[idx] = [0.1 + c * 0.16, 0.1 + r * 0.22]
                idx += 1
    initial_configs.append(centers_grid)

    # Optimization function
    def optimize_from_initial(centers_init):
        # Variables: [x1, y1, r1, x2, y2, r2, ...]
        # Size: 3 * 26 = 78
        x0 = np.zeros(3 * n_circles)
        for i in range(n_circles):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = 0.05  # Start with small radii to ensure feasibility
        
        bounds = []
        for i in range(n_circles):
            # x in [0, 1], y in [0, 1], r in [0, 0.5]
            bounds.extend([
                (0, 1), (0, 1), (0, 0.5)
            ])

        # Constraints
        cons = []

        # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        # => r <= x, r <= 1-x, r <= y, r <= 1-y
        for i in range(n_circles):
            # r <= x
            cons.append({
                'type': 'ineq', 
                'fun': (lambda v, idx=i: v[3*idx] - v[3*idx+2])
            })
            # r <= 1-x
            cons.append({
                'type': 'ineq', 
                'fun': (lambda v, idx=i: (1 - v[3*idx]) - v[3*idx+2])
            })
            # r <= y
            cons.append({
                'type': 'ineq', 
                'fun': (lambda v, idx=i: v[3*idx+1] - v[3*idx+2])
            })
            # r <= 1-y
            cons.append({
                'type': 'ineq', 
                'fun': (lambda v, idx=i: (1 - v[3*idx+1]) - v[3*idx+2])
            })

        # Overlap constraints: dist >= r1 + r2
        # (x1-x2)^2 + (y1-y2)^2 >= (r1+r2)^2
        # We enforce dist - (r1+r2) >= 0
        # To avoid sqrt, we can use squared, but standard is dist >= sum_radii
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                cons.append({
                    'type': 'ineq',
                    'fun': (lambda v, i=i, j=j: 
                            np.sqrt((v[3*i]-v[3*j])**2 + (v[3*i+1]-v[3*j+1])**2) - (v[3*i+2] + v[3*j+2]))
                })

        # Objective: Maximize sum of radii -> Minimize negative sum
        def objective(v):
            return -np.sum(v[2::3])

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success or (res.fun < -best_sum_radii):
                centers_out = np.zeros((n_circles, 2))
                radii_out = np.zeros(n_circles)
                for i in range(n_circles):
                    centers_out[i] = [res.x[3*i], res.x[3*i+1]]
                    radii_out[i] = res.x[3*i+2]
                
                current_sum = np.sum(radii_out)
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_centers = centers_out
                    best_radii = radii_out
        except Exception as e:
            # Silently fail and try next seed if optimization crashes
            pass

    # Run optimization for each initial configuration
    for config in initial_configs:
        optimize_from_initial(config)
        
        # Try a perturbed version of the current best if found
        if best_centers is not None:
            perturbed = best_centers + np.random.normal(0, 0.02, best_centers.shape)
            perturbed = np.clip(perturbed, 0.01, 0.99)
            optimize_from_initial(perturbed)

    # Fallback if optimization failed completely (should not happen)
    if best_centers is None:
        best_centers = np.random.rand(n_circles, 2)
        best_radii = np.full(n_circles, 0.01)
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
