import numpy as np
from scipy.optimize import minimize

def compute_energy(centers, R):
    """Compute overlap and boundary violation energy for a given configuration."""
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Upper triangle indices for unique pairs
    i, j = np.triu_indices(centers.shape[0], k=1)
    pairwise_energy = np.sum(np.maximum(0.0, 2*R - dists[i, j])**2)
    
    # Boundary violations
    x = centers[:, 0]
    y = centers[:, 1]
    b_energy = np.sum(np.maximum(0.0, R - x)**2 + 
                      np.maximum(0.0, R - y)**2 + 
                      np.maximum(0.0, R - (1.0 - x))**2 + 
                      np.maximum(0.0, R - (1.0 - y))**2)
                      
    return pairwise_energy + b_energy

def obj_func(x, R, n):
    """Objective function for scipy optimizer."""
    centers = x.reshape(n, 2)
    return compute_energy(centers, R)

def run_packing():

            idx += 1

    bounds = [(0.0, 1.0)] * (2 * n)

    low, high = 0.08, 0.115

        mid = (low + high) / 2.0
