

To maximize the sum of radii for 26 circles in a unit square, we treat this as a constrained optimization problem. We aim to maximize $\sum r_i$ subject to non-overlap constraints ($(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$) and boundary constraints. 

Using the SLSQP algorithm from `scipy.optimize`, we start from a dense hexagonal lattice initialization and perform multiple random restarts to escape local optima. This approach typically converges to a valid packing with a sum of radii exceeding the 2.636 target.

```python
import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    def hex_init(offset_x=0, offset_y=0):
        # Hexagonal grid initialization
        pts = []
        y = 0.15
        while y <= 0.85:
            x = 0.15 if int(y / 0.26) % 2 == 0 else 0.15 + 0.26 * 0.5
            while x <= 0.85:
                pts.append((x + offset_x, y + offset_y))
                x += 0.26
            y += 0.26
        pts = np.array(pts[:n])
        # Ensure within [0,1]
        pts = np.clip(pts, 1e-5, 1 - 1e-5)
        return pts, np.ones(n) * 0.05

    def constraints(vars):
        centers = vars[:2*n].reshape((n, 2))
        radii = vars[2*n:]
        violations = []
        
        # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
        violations.append(centers[:, 0] - radii)
        violations.append(1 - centers[:, 0] - radii)
        violations.append(centers[:, 1] - radii)
        violations.append(1 - centers[:, 1] - radii)
        
        # Overlap constraints: dist^2 >= (r_i + r_j)^2
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        # Upper triangle indices
        idx = np.triu_indices(n, k=1)
        violations.append(dist_sq[idx] - r_sum[idx]**2)
        
        return np.concatenate(violations)

    def objective(vars):
        return -np.sum(vars[2*n:])

    bounds = [(0, 1)] * (2*n) + [(1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}

    for attempt in range(30):
        rng = np.random.default_rng(attempt)
        centers0, radii0 = hex_init(rng.uniform(-0.02, 0.02), rng.uniform(-0.02, 0.02))
        x0 = np.concatenate([centers0.flatten(), radii0])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
            if res.success:
                centers_opt = res.x[:2*n].reshape((n, 2))
                radii_opt = res.x[2*n:]
                s = np.sum(radii_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
        except Exception:
            pass

    # Final validation check
    centers = np.clip(best_centers, 1e-6, 1 - 1e-6)
    radii = best_radii
    return centers, radii, float(np.sum(radii))
```
