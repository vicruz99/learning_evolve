import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Helper to compute cost with penalty
    def compute_cost(params):
        # params shape (n_circles, 3) -> x, y, r
        centers = params[:, :2]
        radii = params[:, 2]
        
        # Objective: Maximize sum of radii -> Minimize negative sum
        cost = -np.sum(radii)
        
        # Penalty parameters
        penalty_weight = 1000.0
        
        # Boundary constraints: x in [r, 1-r], y in [r, 1-r]
        # Equivalent to: r <= x, r <= 1-x, r <= y, r <= 1-y
        # Violation: max(0, r - x)^2 + ...
        
        # Check x bounds
        viol_left = np.maximum(0, radii - centers[:, 0])
        viol_right = np.maximum(0, radii - (1.0 - centers[:, 0]))
        
        # Check y bounds
        viol_down = np.maximum(0, radii - centers[:, 1])
        viol_up = np.maximum(0, radii - (1.0 - centers[:, 1]))
        
        boundary_penalty = np.sum(viol_left**2 + viol_right**2 + viol_down**2 + viol_up**2)
        
        # Overlap constraints: dist(i, j) >= r_i + r_j
        # Violation: max(0, r_i + r_j - dist)^2
        # Vectorized computation for efficiency
        # Centers shape (n, 2), Radii shape (n,)
        
        # Compute pairwise distances
        # diff shape (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Pairwise sum of radii
        rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        # Violations where r_i + r_j > dist
        # We only care about i < j to avoid double counting and self-checks
        # Create a mask for upper triangle
        mask = np.triu(np.ones((n_circles, n_circles), dtype=bool), k=1)
        
        violations = np.maximum(0, rad_sum - dists)
        overlap_penalty = np.sum(violations[mask]**2)
        
        cost += penalty_weight * (boundary_penalty + overlap_penalty)
        
        return cost

    def run_optimization(initial_centers, initial_radii, seed=None):

            np.random.seed(seed)
            
        # Combine into single vector for optimizer
        # Shape: (n_circles, 3)
        initial_params = np.hstack([initial_centers, initial_radii.reshape(-1, 1)])

        bounds = []

            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r

        # Since we use penalty method, we don't need hard constraints in solver, 
        # but bounds help keep parameters sane.
        # Nelder-Mead is robust for non-smooth but slow. 
        # L-BFGS-B is good for smooth penalties.
        
        try:
            result = minimize(
                compute_cost,
                initial_params,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-9}
            )
            
            if result.success or result.nit > 0:
                final_params = result.x.reshape(n_circles, 3)
                final_centers = final_params[:, :2]
                final_radii = final_params[:, 2]
                
                # Calculate actual sum
                current_sum = np.sum(final_radii)
                return final_centers, final_radii, current_sum
                
        except Exception:
            pass
