import numpy as np
import scipy.optimize as opt
import itertools

def get_distance_matrix(centers):
    """Compute pairwise distance squared matrix."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sum(diff**2, axis=2)

def objective_func(params, n, lambda_penalty):
    """
    Objective function to minimize: -sum(radii) + penalty for constraint violations.
    params: array of shape (n*3) containing [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: minimize negative sum of radii (maximize sum)
    obj = -np.sum(radii)
    
    # Penalty term initialization
    penalty = 0.0
    
    # 1. Boundary Penalties
    # Constraints: r <= x <= 1-r  =>  x-r >= 0, 1-x-r >= 0
    # Violation if r > x or r > 1-x
    # We use squared max(0, violation)
    
    # Left boundary: x - r >= 0 => violation if r - x > 0
    left_violations = np.maximum(0, radii - centers[:, 0])
    penalty += np.sum(left_violations**2)
    
    # Right boundary: 1 - x - r >= 0 => violation if r - (1-x) > 0 => r + x - 1 > 0
    right_violations = np.maximum(0, radii + centers[:, 0] - 1)
    penalty += np.sum(right_violations**2)
    
    # Bottom boundary: y - r >= 0
    bottom_violations = np.maximum(0, radii - centers[:, 1])
    penalty += np.sum(bottom_violations**2)
    
    # Top boundary: 1 - y - r >= 0
    top_violations = np.maximum(0, radii + centers[:, 1] - 1)
    penalty += np.sum(top_violations**2)
    
    # 2. Overlap Penalties
    # Constraint: dist^2 >= (r_i + r_j)^2
    # Violation if (r_i + r_j)^2 - dist^2 > 0
    
    # Compute squared distances
    dist_sq = get_distance_matrix(centers)
    
    # Compute sum of radii matrix (symmetric)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Violation matrix: max(0, r_sum_sq - dist_sq)
    # We only care about i < j, but computing all and dividing by 2 or masking is fine.
    # Using upper triangle to avoid double counting
    indices = np.triu_indices(n, k=1)
    violations = np.maximum(0, r_sum_sq[indices] - dist_sq[indices])
    
    penalty += np.sum(violations**2)
    
    return obj + lambda_penalty * penalty

def get_bounds(n):
    """Return bounds for variables [x, y, r] for n circles."""
    bounds = []
    for _ in range(n):
        # x in [0, 1]
        bounds.append((0.0, 1.0))
        # y in [0, 1]
        bounds.append((0.0, 1.0))
        # r in [0, 0.5] (max possible radius is 0.5)
        bounds.append((0.0, 0.5))
    return bounds

def generate_hexagonal_init(n):
    """Generate a hexagonal grid initial configuration."""
    centers = []
    radii = []
    # Approximate radius for 26 circles in hex packing
    # Area density approx 0.9. 26 * pi * r^2 approx 0.9. r approx 0.105
    r_est = 0.10
    row_height = np.sqrt(3) * r_est
    
    y = r_est
    row_idx = 0
    while len(centers) < n:
        row_circles = 0
        x = r_est + (0.5 if row_idx % 2 == 1 else 0) # Shift odd rows
        while x < 1.0 and len(centers) < n:
            centers.append([x, y])
            radii.append(r_est)
            x += 2 * r_est
            row_circles += 1
        y += row_height
        row_idx += 1
        
    return np.array(centers), np.array(radii)

def generate_random_init(n):
    """Generate random initial configuration."""
    centers = np.random.rand(n, 2) * 0.8 + 0.1 # Slightly inward
    radii = np.full(n, 0.05) # Small initial radius
    return centers, radii

def generate_grid_init(n):
    """Generate a grid initial configuration."""
    k = int(np.ceil(np.sqrt(n)))
    # Create a k x k grid
    x = np.linspace(1.0/(2*k), 1.0 - 1.0/(2*k), k)
    y = np.linspace(1.0/(2*k), 1.0 - 1.0/(2*k), k)
    xx, yy = np.meshgrid(x, y)
    pts = np.vstack([xx.ravel(), yy.ravel()]).T
    r = 1.0 / (2*k)
    
    centers = pts[:n]
    radii = np.full(n, r)
    return centers, radii
