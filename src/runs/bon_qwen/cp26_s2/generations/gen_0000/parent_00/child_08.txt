```python
from scipy.optimize import minimize
import math

def boundary_constraints(z):
    # Returns array of values >= 0
    # x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    n = len(z) // 3
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    return np.concatenate([c1, c2, c3, c4])

def distance_constraints(z):
    # Returns array of values >= 0
    # For all i < j: dist^2 - (ri + rj)^2 >= 0
    n = len(z) // 3
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    
    # We need to compute all pairwise distances
    # This can be done vectorized but might be memory heavy?
    # N=26 -> 26x26 matrix is small.
    
    # x diff matrix
    # x[i] - x[j]
    # Use broadcasting
    # x_col shape (n, 1)
    x_col = x.reshape(-1, 1)

    dy = y_col - y
```

```python
    # x <= 1-r => 1 - x - r >= 0
    # y >= r  => y - r >= 0
    # y <= 1-r => 1 - y - r >= 0
    
    c = np.zeros(4 * n)
    c[0::4] = x - r
    c[1::4] = 1.0 - x - r
    c[2::4] = y - r
    c[3::4] = 1.0 - y - r
    return c

def distance_constraints(z):
    n = len(z) // 3
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    
    # Vectorized calculation of (dist^2 - (r_i+r_j)^2) for i < j
    x_col = x.reshape(-1, 1)
    y_col = y.reshape(-1, 1)
    r_col = r.reshape(-1, 1)
    
    dx = x_col - x
    dy = y_col - y
    dr = r_col + r
    
    d2 = dx*dx + dy*dy
    r2 = dr*dr
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    vals = d2[mask] - r2[mask]
    
    return vals.flatten()

CONSTRAINTS = [
    {'type': 'ineq', 'fun': boundary_constraints},
    {'type': 'ineq', 'fun': distance_constraints}
]

    n = N_CIRCLES
    
    # Generate initial centers in a hexagonal grid pattern
    initial_centers = []
    row_y = 0.1
    row_idx = 0

    best_sum = -1.0
    
    # Try multiple restarts
    for _ in range(15):
        z_curr = z0.copy()

        z_curr[1::3] += np.random.uniform(-0.02, 0.02, n)

        z_curr[2::3] = 0.05 # Reset radii
        
        try:

                           options={'maxiter': 500, 'ftol': 1e-12})

                if current_sum > best_sum:
                    best_sum = current_sum
                    best_z = res.x

            continue

        radii = best_z[2::3]
        # Ensure radii are non-negative (should be, but just in case)
        radii = np.maximum(radii, 0.0)
        return centers, radii, np.sum(radii)
    
    # Fallback
    return initial_centers, initial_radii, np.sum(initial_radii)
```

One optimization for speed:

```python
import numpy as np
from scipy.optimize import minimize

# Global constant for number of circles
N_CIRCLES = 26

def objective_func(z):
    """
    Objective function to minimize.
    We want to maximize sum of radii, so we minimize negative sum of radii.
    z is a 1D array: [x1, y1, r1, x2, y2, r2, ...]
    """
    # Radii are at indices 2, 5, 8, ...
    return -np.sum(z[2::3])

def boundary_constraints(z):
    """
    Constraints ensuring circles are within the unit square [0,1]x[0,1].
    Returns a vector of constraint values >= 0.
    """
    n = len(z) // 3
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    
    # Constraints:
    # x >= r       => x - r >= 0
    # x <= 1 - r   => 1 - x - r >= 0
    # y >= r       => y - r >= 0
    # y <= 1 - r   => 1 - y - r >= 0
    
    c = np.zeros(4 * n)
    c[0::4] = x - r
    c[1::4] = 1.0 - x - r
    c[2::4] = y - r
    c[3::4] = 1.0 - y - r
    return c

def distance_constraints(z):
    """
    Constraints ensuring circles do not overlap.
    For every pair (i, j), dist(ci, cj) >= ri + rj.
    Equivalent to: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0.
    Returns a vector of constraint values >= 0.
    """
    n = len(z) // 3
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    
    # Vectorized computation for all pairs
    # Create column vectors for broadcasting
    x_col = x.reshape(-1, 1)
    y_col = y.reshape(-1, 1)
    r_col = r.reshape(-1, 1)
    
    dx = x_col - x
    dy = y_col - y
    dr = r_col + r
    
    d2 = dx*dx + dy*dy
    r2 = dr*dr
    
    # We only need constraints for i < j (upper triangle)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    vals = d2[mask] - r2[mask]
    
    return vals.flatten()

# Define the constraints list for the optimizer
CONSTRAINTS = [
    {'type': 'ineq', 'fun': boundary_constraints},
    {'type': 'ineq', 'fun': distance_constraints}
]

def run_packing():
    """
    Solves the circle packing problem to maximize sum of radii.
    """
    n = N_CIRCLES
    
    # 1. Generate a good initial configuration
    # We use a hexagonal grid pattern which is denser than a square grid.
    initial_centers = []
    row_y = 0.1
    row_idx = 0
    
    # Fill rows until we have n points
    while len(initial_centers) < n:
        # Stagger rows for hexagonal packing
        # Even rows start at 0.1, Odd rows start at 0.2
        if row_idx % 2 == 0:
            col_x = 0.1
        else:
            col_x = 0.2
        
        # Add points in the current row
        while col_x <= 0.9 and len(initial_centers) < n:
            initial_centers.append([col_x, row_y])
            col_x += 0.2
        
        # Move to next row
        row_y += 0.173205  # sqrt(3)/2 * 0.2
        row_idx += 1
    
    initial_centers = np.array(initial_centers[:n])
    # Start with small radii to ensure the initial point is feasible
    initial_radii = np.full(n, 0.05)
    
    # Flatten the state vector: [x1, y1, r1, x2, y2, r2, ...]
    z0 = np.zeros(3 * n)
    for i in range(n):
        z0[3*i] = initial_centers[i, 0]
        z0[3*i+1] = initial_centers[i, 1]
        z0[3*i+2] = initial_radii[i]
        
    best_z = None
    best_sum = -1.0
    
    # 2. Optimization Loop with multiple restarts to avoid local minima
    # We perform several runs with slight perturbations to the initial positions.
    for attempt in range(10):
        z_curr = z0.copy()
        
        # Add small random noise to centers to explore different configurations
        noise_x = np.random.uniform(-0.02, 0.02, n)
        noise_y = np.random.uniform(-0.02, 0.02, n)
        
        z_curr[0::3] = z_curr[0::3] + noise_x
        z_curr[1::3] = z_curr[1::3] + noise_y
        
        # Ensure centers remain within a safe margin [0.05, 0.95]
        # This helps maintain feasibility for the small initial radii
        z_curr[0::3] = np.clip(z_curr[0::3], 0.05, 0.95)
        z_curr[1::3] = np.clip(z_curr[1::3], 0.05, 0.95)
        
        # Reset radii to small value
        z_curr[2::3] = 0.05
        
        try:
            # Run SLSQP optimizer
            res = minimize(objective_func, z_curr, method='SLSQP', constraints=CONSTRAINTS, 
                           options={'maxiter': 500, 'ftol': 1e-12})
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_z = res.x
        except Exception:
            continue

    # 3. Extract and return the best solution found
    if best_z is not None:
        centers = best_z[0::3].reshape(n, 2)
        radii = best_z[2::3]
        # Ensure radii are non-negative (safety check)
        radii = np.maximum(radii, 0.0)
        return centers, radii, np.sum(radii)
    
    # Fallback to initial configuration if optimization fails
    return initial_centers, initial_radii, np.sum(initial_radii)
```
