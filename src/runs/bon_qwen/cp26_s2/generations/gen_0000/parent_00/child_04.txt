import numpy as np
from scipy.optimize import minimize
import warnings

# Suppress optimization warnings for cleaner output
warnings.filterwarnings('ignore')

N_CIRCLES = 26
PAIR_INDICES = [(i, j) for i in range(N_CIRCLES) for j in range(i+1, N_CIRCLES)]

def objective_func(x):
    """Minimize negative sum of radii => Maximize sum of radii"""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Compute all inequality constraints: boundary and non-overlap"""
    n_cons = 4 * N_CIRCLES + len(PAIR_INDICES)
    cons = np.empty(n_cons)
    idx = 0
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    for i in range(N_CIRCLES):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        cons[idx] = xi - ri
        cons[idx+1] = 1.0 - xi - ri
        cons[idx+2] = yi - ri
        cons[idx+3] = 1.0 - yi - ri
        idx += 4
        
    # Non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    for i, j in PAIR_INDICES:
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
        dx, dy = xi - xj, yi - yj
        cons[idx] = dx*dx + dy*dy - (ri + rj)**2
        idx += 1
        
    return cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Bounds for x, y in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0) for _ in range(3 * N_CIRCLES)]
    for i in range(N_CIRCLES):
        bounds[3*i+2] = (0.0, 0.5)

    best_val = 0.0
    best_x = None
    
    # Multiple restarts to escape local optima
    for seed in range(15):
        rng = np.random.RandomState(seed + 42)
        x0 = np.zeros(3 * N_CIRCLES)
        
        if seed < 8:
            # Staggered hexagonal grid initialization
            cols = 5
            rows = 6
            sx = 1.0 / (cols + 1)
            sy = 1.0 / (rows + 1)
            idx = 0
            for r in range(rows):
                for c in range(cols):
                    if idx >= N_CIRCLES: break
                    x0[3*idx] = sx * (c + 1 + (0.5 if r % 2 == 1 else 0))
                    x0[3*idx+1] = sy * (r + 1)
                    idx += 1
            x0[2::3] = 0.075  # Initial radius
        else:
            # Random uniform initialization
            x0[:2*N_CIRCLES] = rng.uniform(0.15, 0.85, 2*N_CIRCLES)
            x0[2::3] = 0.04   # Small initial radius to ensure feasibility
            
        # Add tiny noise to break symmetry and avoid flat gradients
        x0[:2*N_CIRCLES] += rng.normal(0, 0.005, 2*N_CIRCLES)
        x0[:2*N_CIRCLES] = np.clip(x0[:2*N_CIRCLES], 0.01, 0.99)
        x0[2::3] = np.clip(x0[2::3], 0.01, 0.4)
        
        try:
            res = minimize(objective_func, x0, method='SLSQP',
                           bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraint_func},
                           options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False})
            
            current_sum = -res.fun
            if current_sum > best_val:
                best_val = current_sum
                best_x = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails unexpectedly
    if best_x is None:
        best_x = np.zeros(3 * N_CIRCLES)
        best_x[:2*N_CIRCLES] = np.random.uniform(0.2, 0.8, 2*N_CIRCLES)
        best_x[2::3] = 0.05

    centers = best_x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_x[2::3]
    
    # Slight safety shrink to guarantee validation passes comfortably
    # The solver satisfies constraints >= 0, but numerical tolerance helps
    radii = np.maximum(radii - 1e-9, 0.0)
    
    return centers, radii, float(np.sum(radii))
