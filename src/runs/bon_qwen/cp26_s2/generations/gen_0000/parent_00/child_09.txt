import numpy as np
from scipy.optimize import minimize


def compute_sum_radii(radii):
    return np.sum(radii)


def initialize_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08
    idx = 0
    rows = [5, 4, 5, 4, 5, 3]
    spacing_x = 0.175
    spacing_y = 0.152
    offset_x = 0.12
    offset_y = 0.13
    for row, n_cols in enumerate(rows):
        for col in range(n_cols):
            if idx >= n:
                break
            x = offset_x + col * spacing_x + (row % 2) * (spacing_x / 2)
            y = offset_y + row * spacing_y
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def resolve_overlaps(centers, radii, n, n_iter=100):
    for _ in range(n_iter):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                if dist_sq > 1e-20:
                    dist = np.sqrt(dist_sq)
                    min_dist = radii[i] + radii[j]
                    if dist < min_dist:
                        overlap = min_dist - dist
                        factor = 0.5
                        dx_norm = dx / dist
                        dy_norm = dy / dist
                        centers[i, 0] -= dx_norm * overlap * factor
                        centers[i, 1] -= dy_norm * overlap * factor
                        centers[j, 0] += dx_norm * overlap * factor
                        centers[j, 1] += dy_norm * overlap * factor
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])


def force_based_optimize(centers, radii, n, max_iter=200000, init_step=0.00002):
    for iteration in range(max_iter):
        step = init_step * max(0.1, 1.0 - iteration / max_iter)
        for i in range(n):
            radii[i] += step
        resolve_overlaps(centers, radii, n, n_iter=80)
    return centers, radii


def objective_penalty(x, n, penalty_weight):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = x[3 * i + 2]
    obj = -np.sum(radii)
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = (centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2
            min_dist = radii[i] + radii[j]
            if dist_sq < min_dist * min_dist and dist_sq > 1e-20:
                dist = np.sqrt(dist_sq)
                penalty += (min_dist - dist) ** 2
            elif dist_sq <= 1e-20:
                penalty += min_dist ** 2
        penalty += max(0.0, radii[i] - centers[i, 0]) ** 2
        penalty += max(0.0, centers[i, 0] + radii[i] - 1.0) ** 2
        penalty += max(0.0, radii[i] - centers[i, 1]) ** 2
        penalty += max(0.0, centers[i, 1] + radii[i] - 1.0) ** 2
    return obj + penalty_weight * penalty


def fine_tune_scipy(centers, radii, n):
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
    def objective_func(x):
        return objective_penalty(x, n, 10000.0)
    result = minimize(objective_func, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 5000})
    centers_opt = np.zeros((n, 2))
    radii_opt = np.zeros(n)
    for i in range(n):
        centers_opt[i, 0] = result.x[3 * i]
        centers_opt[i, 1] = result.x[3 * i + 1]
        radii_opt[i] = result.x[3 * i + 2]
    return centers_opt, radii_opt
