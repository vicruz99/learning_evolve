import numpy as np
from scipy.optimize import minimize
from functools import partial


def compute_constraints(params, n):
    """Compute all constraint values for the optimization."""
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]
    cons = []

    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(n):
        cons.append(centers[i, 0] - radii[i])
        cons.append(1.0 - centers[i, 0] - radii[i])
        cons.append(centers[i, 1] - radii[i])
        cons.append(1.0 - centers[i, 1] - radii[i])

    # Non-overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-10)
            r_sum = radii[i] + radii[j]
            cons.append(dist - r_sum)

    return np.array(cons)


def compute_objective(params, n):
    """Objective: negative sum of radii (to be minimized)."""
    radii = params[2 * n:]
    return -np.sum(radii)


def generate_hexagonal_initial(n, base_radius):
    """Generate initial hexagonal packing configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * base_radius

    # 5 rows: 5, 6, 5, 6, 4 = 26 circles
    row_configs = [(5, 0.0), (6, 0.5), (5, 0.0), (6, 0.5), (4, 0.0)]

    idx = 0
    for row_idx, (n_cols, offset_factor) in enumerate(row_configs):
        for col in range(n_cols):
            if idx >= n:
                break
            x = 0.05 + col * (base_radius * 2.0) + offset_factor * (base_radius * 2.0) * np.cos(np.pi / 6)
            y = 0.05 + row_idx * (base_radius * 2.0) * np.sin(np.pi / 6)
            centers[idx] = [x, y]
            radii[idx] = base_radius
            idx += 1

    return centers, radii


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26

    best_centers = None
    best_radii = None
    best_sum = 0.0

        rng = np.random.RandomState(42 + trial * 7)
        perturbation = rng.uniform(-0.02, 0.02, (n, 2))

        centers = np.clip(centers, 0.05, 0.95)

        x0 = np.concatenate([centers.flatten(), radii])

        cons = {'type': 'ineq', 'fun': constraint_func}

        # Bounds: coordinates in [0, 1], radii in [0, 0.5]
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

        # Run optimization
        result = minimize(
            compute_objective,
            x0,
            args=(n,),
            bounds=bounds,
            constraints=cons,
            method='SLSQP',

        current_sum = -result.fun

            best_centers = result.x[:2 * n].reshape((n, 2))
