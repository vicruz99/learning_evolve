import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a force-directed expansion simulation followed by local optimization.
    """
    np.random.seed(42)  # For reproducibility
    n_circles = 26
    
    # --- Phase 1: Initialization ---
    # Start with a slightly perturbed grid or random positions
    # A grid helps avoid initial heavy overlaps
    centers = np.zeros((n_circles, 2))
    # Try to fit in a 6x5 grid pattern roughly
    cols = 6
    rows = 5
    # Generate grid points
    x_spacing = 1.0 / (cols + 1)
    y_spacing = 1.0 / (rows + 1)
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n_circles:
                break
            # Add some jitter
            centers[idx, 0] = (c + 1) * x_spacing + np.random.uniform(-0.05, 0.05)
            centers[idx, 1] = (r + 1) * y_spacing + np.random.uniform(-0.05, 0.05)
            idx += 1
    
    # Ensure all are within bounds initially
    centers = np.clip(centers, 0.05, 0.95)
    
    # Initialize velocities
    velocities = np.zeros_like(centers)
    
    # --- Phase 2: Force-Directed Expansion (Molecular Dynamics) ---
    radius = 0.01
    max_radius = 0.2 # Upper bound estimate
    dr = 0.0005 # Radius growth step
    
    # Simulation parameters
    steps_per_growth = 100
    total_growths = 3000 # Allow radius to grow up to ~1.5 (clamped later)
    
    damping = 0.9
    repulsion_strength = 5.0
    temperature = 0.02 # Initial thermal noise
    cooling_rate = 0.999
    
    best_centers = centers.copy()
    best_radius = radius
    
    # Precompute indices for distance calculation
    # We need to check all pairs
    
    for growth_step in range(total_growths):
        # Increase radius
        radius += dr
        
        # Cooling
        temperature *= cooling_rate
        
        for _ in range(steps_per_growth):
            forces = np.zeros_like(centers)
            
            # Compute pairwise repulsive forces
            # Vectorized distance computation
            # centers shape (N, 2)
            # diff shape (N, N, 2)
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            # dists shape (N, N)
            dists = np.linalg.norm(diff, axis=2)

            # Add small epsilon to avoid division by zero
            # We only care about i < j, but vectorized calc is easy
            # Create a mask for pairs with distance < 2*radius
            # Diagonal is 0, ignore it
            np.fill_diagonal(dists, np.inf)
            
            overlap = 2 * radius - dists
            is_overlapping = overlap > 1e-6
            
            # Force direction: diff / dist
            # Normalize diff
            safe_dists = np.where(dists > 1e-6, dists, 1.0)
            directions = diff / safe_dists[:, :, np.newaxis]
            
            # Repulsive force proportional to overlap
            # F = strength * overlap * direction
            # Only apply if overlapping
            force_magnitude = np.zeros_like(dists)
            force_magnitude[is_overlapping] = repulsion_strength * overlap[is_overlapping]
            
            # Apply forces to centers
            # Force on i from j is in direction (c_i - c_j)
            # Sum forces over j
            forces += np.sum(directions * force_magnitude[:, :, np.newaxis], axis=1)
            
            # Boundary repulsion
            # Walls at 0 and 1
            # If center[radius] is too close to 0 or 1, push back
            # Distance to left wall: x
            # Distance to right wall: 1-x
            # Required distance: radius
            
            # Left wall (x < radius)
            dist_left = centers[:, 0]
            penetration_left = radius - dist_left
            mask_left = penetration_left > 1e-6
            forces[mask_left, 0] += repulsion_strength * penetration_left[mask_left]
            
            # Right wall (1 - x < radius) => x > 1 - radius
            dist_right = 1.0 - centers[:, 0]
            penetration_right = radius - dist_right
            mask_right = penetration_right > 1e-6
            forces[mask_right, 0] -= repulsion_strength * penetration_right[mask_right]

            dist_bottom = centers[:, 1]
            penetration_bottom = radius - dist_bottom
            mask_bottom = penetration_bottom > 1e-6
            forces[mask_bottom, 1] += repulsion_strength * penetration_bottom[mask_bottom]
            
            # Top wall (1 - y < radius)
            dist_top = 1.0 - centers[:, 1]
            penetration_top = radius - dist_top
            mask_top = penetration_top > 1e-6
            forces[mask_top, 1] -= repulsion_strength * penetration_top[mask_top]

            velocities += forces * (0.01) # scaling factor for force
            velocities *= damping
            # Add thermal noise
            velocities += np.random.normal(0, temperature, size=velocities.shape)
            
            # Update positions
            centers += velocities
            
            # Clamp to [0, 1] strictly to prevent escaping
            centers = np.clip(centers, 0.0, 1.0)

            # (Simple damping handles this mostly)

                radius = current_r + 0.001

    # This is equivalent to maximizing min( pairwise_distances, 2*boundary_distances )

        # coords is 1D array of size 2*n
        c = coords.reshape(-1, 2)

        min_dist = np.min(dists)

        # Dist to boundary is min(x, 1-x, y, 1-y)

        # But pairwise gives 2*r <= dist_pair => r <= dist_pair / 2
        # So we want to maximize min( dist_pair/2, dist_boundary )
        # Which is 1/2 * max( min( dist_pair, 2*dist_boundary ) )
        
        # Let's compute 2 * dist_boundary
        x = c[:, 0]
        y = c[:, 1]
        d_bound_x = np.minimum(x, 1 - x)
        d_bound_y = np.minimum(y, 1 - y)
        d_bound = np.minimum(d_bound_x, d_bound_y)
        scaled_d_bound = 2 * d_bound
        
        # Global min

        val = np.minimum(min_dist, np.min(scaled_d_bound))
        
        return -val # Minimize negative

    # Flatten best_centers
    x0 = best_centers.flatten()
    
    # Use a robust optimizer. SLSQP or Nelder-Mead.
    # Nelder-Mead doesn't require gradients and handles non-smooth objectives okay.
    # But for high dimensions (52), it might be slow.
    # Let's try 'L-BFGS-B' if we provide gradient? No, non-smooth.
    # 'Powell' is good.
    # Or just multiple random restarts of a local search.
    
    # Since we are close to optimum, a few iterations of a gradient-free method might suffice.
    # But let's use scipy.optimize.minimize with Nelder-Mead.
    
    try:
        res = opt.minimize(objective, x0, method='Nelder-Mead', 
                           options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6})
        opt_centers = res.x.reshape(-1, 2)
        opt_val = -res.fun
        
        # Verify and compute radius
        # The objective returned -min(dists, 2*bounds). 
        # So optimal radius is opt_val / 2.
        # But wait, objective was -min(dist, 2*bound).
        # So max radius = opt_val / 2.
        
        # Let's re-calculate radius from centers to be sure
        final_r = calculate_max_radius(opt_centers, n_circles)
        
        if final_r > best_radius:
            best_centers = opt_centers
            best_radius = final_r
            
    except Exception:
        pass # Fallback to best_centers from simulation

    # --- Phase 4: Final Adjustment & Validation ---
    # The optimization might have pushed centers to boundary or created overlaps due to non-smoothness.
    # We need to compute the exact valid radius for the final centers.
    
    # Actually, the 'calculate_max_radius' function computes the largest r such that circles don't overlap.
    # We can set radii to that value.
    
    final_r = calculate_max_radius(best_centers, n_circles)
    
    # If the calculated radius is slightly less than best_radius due to numerical issues in optimization,
    # that's fine. We use the valid radius.
    
    radii = np.full(n_circles, final_r)
    sum_radii = np.sum(radii)
    
    # Double check validity
    if not validate_packing(best_centers, radii):
        # Fallback: reduce radius slightly to ensure validity
        # This shouldn't happen if calculate_max_radius is correct, 
        # but floating point errors exist.
        final_r = final_r * 0.999
        radii = np.full(n_circles, final_r)
        sum_radii = np.sum(radii)

    return best_centers, radii, sum_radii

def check_feasibility(centers, n):
    """
    Checks if a configuration is valid and returns the max feasible radius.
    Returns (True, radius) or (False, radius).
    """
    # Compute min pairwise distance
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, np.inf)
    min_pair_dist = np.min(dists)
    
    # Compute min boundary distance
    x = centers[:, 0]
    y = centers[:, 1]
    d_bound = np.minimum(np.minimum(x, 1-x), np.minimum(y, 1-y))
    min_bound_dist = np.min(d_bound)
    
    # Radius is limited by pair dist / 2 and bound dist
    # r <= min_pair_dist / 2
    # r <= min_bound_dist
    r_limit = min(min_pair_dist / 2.0, min_bound_dist)
    
    # Validity check: overlaps?
    # Overlap if dist < 2*r. But r is variable.
    # If we assume r = r_limit, then dist >= 2*r_limit should hold for pairs?
    # min_pair_dist >= 2 * (min_pair_dist / 2) is equality.
    # So yes, if we set r = r_limit, it is valid.
    
    return True, r_limit

def calculate_max_radius(centers, n):
    """
    Calculates the maximum radius r such that circles with given centers
    fit in the square without overlap.
    """
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, np.inf)
    min_pair_dist = np.min(dists)
    
    # Boundary distances
    x = centers[:, 0]
    y = centers[:, 1]
    # Distance to nearest boundary
    d_bound = np.minimum(np.minimum(x, 1-x), np.minimum(y, 1-y))
    min_bound_dist = np.min(d_bound)
    
    # Constraints:
    # 1. 2r <= min_pair_dist  => r <= min_pair_dist / 2
    # 2. r <= min_bound_dist
    r = min(min_pair_dist / 2.0, min_bound_dist)
    
    return max(0.0, r)

def validate_packing(centers, radii):
    """
    Validation function provided in the prompt.
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
