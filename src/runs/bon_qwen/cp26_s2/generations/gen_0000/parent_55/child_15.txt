import numpy as np
from scipy.optimize import minimize


def objective_func(x):
    """Minimize negative sum of radii (equivalent to maximizing sum)."""
    n = 26
    radii = x[2*n:]
    return -np.sum(radii)


def constraint_func(x):
    """Compute all inequality constraints. All must be >= 0."""
    n = 26
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]

    con = []

    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(n):
        con.append(centers[i, 0] - radii[i])
        con.append(1 - centers[i, 0] - radii[i])
        con.append(centers[i, 1] - radii[i])
        con.append(1 - centers[i, 1] - radii[i])

    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            con.append(dist - radii[i] - radii[j])

    # Non-negative radii
    for i in range(n):
        con.append(radii[i])

    return np.array(con)


def hexagonal_init(n, seed=0):
    """Generate initial hexagonal packing with 6 rows of alternating 5/4 circles."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    r = 0.085
    row = 0
    idx = 0
    while idx < n:
        num_circles = 5 if row % 2 == 0 else 4
        for col in range(num_circles):
            if idx >= n:
                break
            x = r + col * 2.0 * r
            if row % 2 == 1:
                x += r
            y = r + row * r * np.sqrt(3.0)
            centers[idx] = [x, y]
            radii[idx] = r
            idx += 1
        row += 1

    return centers, radii


def grid_init(n, seed=0):
    """Generate a near-uniform grid initial configuration."""
    np.random.seed(seed)
    # Arrange roughly as a 6x5 grid with some perturbation
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.085

    idx = 0
    for row in range(6):
        num_in_row = 5 if row < 5 else 1
        for col in range(num_in_row):
            if idx >= n:
                break
            x = r + col * 2.0 * r + (np.random.random() - 0.5) * 0.02
            y = r + row * r * np.sqrt(3.0) * 0.95 + (np.random.random() - 0.5) * 0.02
            centers[idx] = [np.clip(x, r + 0.001, 1 - r - 0.001),
                            np.clip(y, r + 0.001, 1 - r - 0.001)]
            radii[idx] = r
            idx += 1

    return centers, radii


def random_init(n, seed=0):
    """Generate random valid initial configuration."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.05

    for i in range(n):
        placed = False
        attempts = 0
        while not placed and attempts < 1000:
            cx = np.random.uniform(r + 0.01, 1 - r - 0.01)
            cy = np.random.uniform(r + 0.01, 1 - r - 0.01)
            ok = True
            for j in range(i):
                d = np.hypot(cx - centers[j, 0], cy - centers[j, 1])
                if d < r + radii[j] + 0.001:
                    ok = False
                    break
            if ok:
                centers[i] = [cx, cy]
                radii[i] = r
                placed = True
            attempts += 1

    return centers, radii


def optimize_from_init(n, x0, max_iter=5000):
    """Run SLSQP optimization from given initial point."""
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = [{'type': 'ineq', 'fun': constraint_func}]

    result = minimize(objective_func, x0, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})

    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    return centers, radii, np.sum(radii)


def validate_and_fix(centers, radii, n):
    """Check validity and fix minor numerical issues."""
    fixed_centers = centers.copy()
    fixed_radii = radii.copy()

    # Clip centers to valid range
    for i in range(n):
        r = fixed_radii[i]
        fixed_centers[i, 0] = np.clip(fixed_centers[i, 0], r, 1 - r)
        fixed_centers[i, 1] = np.clip(fixed_centers[i, 1], r, 1 - r)

    # Fix small overlaps by slightly reducing radii if needed
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(fixed_centers[i, 0] - fixed_centers[j, 0],
                         fixed_centers[i, 1] - fixed_centers[j, 1])
            if d < fixed_radii[i] + fixed_radii[j] - 1e-12:
                # Reduce radii slightly to resolve overlap
                excess = (fixed_radii[i] + fixed_radii[j] - d) / 2.0 + 1e-8
                fixed_radii[i] -= excess
                fixed_radii[j] -= excess
                fixed_radii[i] = max(0, fixed_radii[i])
                fixed_radii[j] = max(0, fixed_radii[j])
                # Re-clip centers
                fixed_centers[i, 0] = np.clip(fixed_centers[i, 0], fixed_radii[i], 1 - fixed_radii[i])
                fixed_centers[i, 1] = np.clip(fixed_centers[i, 1], fixed_radii[i], 1 - fixed_radii[i])
                fixed_centers[j, 0] = np.clip(fixed_centers[j, 0], fixed_radii[j], 1 - fixed_radii[j])
                fixed_centers[j, 1] = np.clip(fixed_centers[j, 1], fixed_radii[j], 1 - fixed_radii[j])

    return fixed_centers, fixed_radii


def run_packing():

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Try multiple initial configurations

    if np.isnan(best_centers).any() or np.isnan(best_radii).any():

        best_centers, best_radii = hexagonal_init(n, 0)
