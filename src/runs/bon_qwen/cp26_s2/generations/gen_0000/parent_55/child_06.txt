import numpy as np
import scipy.optimize as opt
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing():
    N = 26
    
    def get_objective(vars):
        # vars = [x1, y1, r1, ..., x26, y26, r26]
        radii = vars[2::3]
        return -np.sum(radii)

    def get_constraint_distance(vars, i, j):
        x1, y1, r1 = vars[3*i:3*i+3]
        x2, y2, r2 = vars[3*j:3*j+3]
        dist_sq = (x1 - x2)**2 + (y1 - y2)**2
        min_dist = r1 + r2
        return dist_sq - min_dist**2

    def get_constraint_boundary(vars, i, dim):
        # dim 0 for x, 1 for y
        coord = vars[3*i + dim]
        r = vars[3*i + 2]
        return coord - r - 0.0 # must be >= 0 -> - (coord - r) <= 0? 
                               # scipy constraints: fun(x) >= 0
        # We need coord - r >= 0
        # and 1 - coord - r >= 0

    constraints = []
    
    # Pairwise distance constraints
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append({
                'type': 'ineq',
                'fun': get_constraint_distance,
                'args': (i, j)
            })

    # Boundary constraints
    for i in range(N):
        # x - r >= 0
        def bound_low_x(vars, idx=i): return vars[3*idx] - vars[3*idx+2]
        constraints.append({'type': 'ineq', 'fun': bound_low_x})
        
        # 1 - x - r >= 0
        def bound_high_x(vars, idx=i): return 1.0 - vars[3*idx] - vars[3*idx+2]
        constraints.append({'type': 'ineq', 'fun': bound_high_x})
        
        # y - r >= 0
        def bound_low_y(vars, idx=i): return vars[3*idx+1] - vars[3*idx+2]
        constraints.append({'type': 'ineq', 'fun': bound_low_y})
        
        # 1 - y - r >= 0
        def bound_high_y(vars, idx=i): return 1.0 - vars[3*idx+1] - vars[3*idx+2]
        constraints.append({'type': 'ineq', 'fun': bound_high_y})
        
        # r >= 0
        def bound_r(vars, idx=i): return vars[3*idx+2]
        constraints.append({'type': 'ineq', 'fun': bound_r})

    def hex_init(seed):
        np.random.seed(seed)
        # Hexagonal packing initialization
        r_init = 0.095
        dx = 2 * r_init
        dy = math.sqrt(3) * r_init
        xs = []
        ys = []
        # Fill rows
        rows = 0
        count = 0
        while count < N:
            x_offset = 0 if rows % 2 == 0 else r_init
            # Max circles in row
            max_cols = int((1 - 2*r_init - x_offset) / dx) + 1
            for c in range(max_cols):
                if count < N:
                    x = r_init + c * dx + x_offset
                    y = r_init + rows * dy
                    if x + r_init <= 1.0 and y + r_init <= 1.0:
                        xs.append(x)
                        ys.append(y)
                        count += 1
            rows += 1
            if rows > 10: break # Safety break
        
        # Add remaining circles if any (random jitter)
        while count < N:
            x = np.random.uniform(r_init, 1-r_init)
            y = np.random.uniform(r_init, 1-r_init)
            xs.append(x)
            ys.append(y)
            count += 1
            
        vars = np.zeros(N * 3)
        for i in range(N):
            vars[3*i] = xs[i]
            vars[3*i+1] = ys[i]
            vars[3*i+2] = r_init
            
        # Add small random noise to break symmetry
        noise = np.random.normal(0, 0.001, size=vars.shape)
        return vars + noise

    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Try multiple random seeds to find global optimum
    for seed in range(10):
        x0 = hex_init(seed)
        try:
            res = opt.minimize(get_objective, x0, method='SLSQP', constraints=constraints, 
                               options={'maxiter': 500, 'ftol': 1e-8})
            
            if res.success:
                curr_sum = -res.fun
                centers = res.x.reshape(N, 3)[:, :2]
                radii = res.x.reshape(N, 3)[:, 2]
                
                # Clamp tiny radii or positions to ensure validity
                radii = np.maximum(radii, 1e-6)
                centers = np.clip(centers, 1e-6, 1-1e-6)

                # Validate
                if validate_packing(centers, radii):
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = centers
                        best_radii = radii
        except Exception as e:
            continue

    if best_centers is None:
        # Fallback to a simple valid grid if optimization failed
        centers = np.zeros((N, 2))
        radii = np.ones(N) * 0.05
        # Place in 6x5 grid pattern roughly
        idx = 0
        for r in range(6):
            for c in range(5):
                if idx < N:
                    centers[idx] = [0.1 + c * 0.2, 0.1 + r * 0.15]
                    idx += 1
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum

# Execute and display result
centers, radii, total_sum = run_packing()
print(f"Total Sum of Radii: {total_sum:.4f}")
