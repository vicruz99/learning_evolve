import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # --- 1. Initialization (Hexagonal Lattice) ---
    # Approximate radius for 26 circles in a hexagonal packing
    # Target sum 2.636 -> avg r ~ 0.1. 
    # Hexagonal packing of equal circles r ~ 0.097 is a good start.
    r_init = 0.09
    
    # Generate hexagonal grid points
    # Vertical spacing sqrt(3)*r, Horizontal spacing 2*r
    dy = np.sqrt(3) * r_init
    dx = 2 * r_init
    
    centers = []
    y = r_init
    row_idx = 0
    while len(centers) < n:
        x_start = r_init + (0.5 * dx if row_idx % 2 == 1 else 0)
        x = x_start
        while x <= 1 - r_init:
            centers.append([x, y])
            x += dx
            if len(centers) >= n:
                break
        y += dy
        row_idx += 1
        
    centers = np.array(centers[:n])
    
    # --- 2. Setup Optimization Variables ---
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = r_init
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    
    # --- 3. Define Constraints ---
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        idx = 3 * i
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]})
        
    # Overlap constraints
    # dist_sq >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            idx_i, idx_j = 3 * i, 3 * j
            constraints.append({
                'type': 'ineq', 
                'fun': lambda v, i=i, j=j: 
                    (v[idx_i] - v[idx_j])**2 + (v[idx_i+1] - v[idx_j+1])**2 - (v[idx_i+2] + v[idx_j+2])**2
            })
            
    # --- 4. Objective Function ---
    def objective(v):
        # Maximize sum of radii -> Minimize negative sum
        return -np.sum(v[2::3])

    # --- 5. Run Optimization ---
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 2000, 'ftol': 1e-9}
    )
    
    # --- 6. Extract Results ---
    best_centers = result.x[:2*n].reshape(n, 2)
    best_radii = result.x[2*n:]
    
    return best_centers, best_radii, np.sum(best_radii)
