import numpy as np
from scipy.optimize import linprog, differential_evolution

def calculate_max_sum_radii(centers):
    """
    Solves the LP to find the maximum radii for fixed centers.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints Matrix A_ub and vector b_ub
    A_ub_rows = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= dist_to_boundary
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1
        
        # r_i <= x
        A_ub_rows.append(row)
        b_ub_list.append(x)
        # r_i <= 1 - x
        A_ub_rows.append(row)
        b_ub_list.append(1 - x)
        # r_i <= y
        A_ub_rows.append(row)
        b_ub_list.append(y)
        # r_i <= 1 - y
        A_ub_rows.append(row)
        b_ub_list.append(1 - y)
        
    # Inter-circle constraints: r_i + r_j <= dist(centers[i], centers[j])
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub_rows.append(row)
            b_ub_list.append(dist)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_list)
    
    # Bounds for r_i: [0, infinity)
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun # Return sum of radii
    return 0.0

def objective(x, n):
    """Objective function for the optimizer. Minimizes negative sum of radii."""
    centers = np.array(x).reshape(-1, 2)
    # Penalty for NaNs
    if np.isnan(centers).any():
        return 1e12
    return -calculate_max_sum_radii(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    dim = 2 * n
    bounds_coords = [(0, 1)] * dim
    
    # 1. Generate Diverse Initial Populations
    initial_populations = []
    
    # Strategy A: 5x5 Grid + 1 in center (Slightly perturbed)
    centers_a = []
    for r in range(5):
        for c in range(5):
            centers_a.append([0.1 + 0.2 * c, 0.1 + 0.2 * r])
    centers_a.append([0.5, 0.5]) # 26th circle
    initial_populations.append(np.array(centers_a).flatten())
    
    # Strategy B: 5x5 Grid + 1 in corner
    centers_b = list(centers_a[:-1])
    centers_b.append([0.1, 0.1])
    initial_populations.append(np.array(centers_b).flatten())
    
    # Strategy C: Staggered Hexagonal-like Pattern
    centers_c = []
    row_y = [0.1, 0.25, 0.5, 0.75, 0.9]
    for i, y in enumerate(row_y):
        # Alternate row lengths: 5, 4, 5, 4, 4 -> sum=22. Need 26.
        # Let's try lengths: 6, 5, 6, 5, 4
        if i == 0: count = 6; shift = 0.0
        elif i == 1: count = 5; shift = 0.1
        elif i == 2: count = 6; shift = 0.0
        elif i == 3: count = 5; shift = 0.1
        else: count = 4; shift = 0.0
        
        step_x = 1.0 / (count + 1)
        for k in range(count):
            x = (k + 1) * step_x + shift
            if 0 <= x <= 1:
                centers_c.append([x, y])
    initial_populations.append(np.array(centers_c).flatten())
    
    # Strategy D: Random Uniform
    np.random.seed(123)
    centers_d = np.random.uniform(0, 1, dim)
    initial_populations.append(centers_d)

    # 2. Run Differential Evolution
    # popsize=15 is a balance between diversity and speed for 52 variables
    res = differential_evolution(
        objective, 
        bounds_coords, 
        args=(n,),
        popsize=15, 
        maxiter=200, 
        mutation=(0.5, 1.0), 
        recombination=0.7, 
        seed=42, 
        workers=1, 
        tol=1e-7,
        initial_populations=initial_populations
    )
    
    best_centers = res.x.reshape(-1, 2)
    sum_radii = -res.fun
    
    # 3. Extract final radii via LP
    radii = np.zeros(n)
    # Re-solve LP to get the vector of radii
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = best_centers[i]
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row); b_ub.append(x)
        A_ub.append(row); b_ub.append(1 - x)
        A_ub.append(row); b_ub.append(y)
        A_ub.append(row); b_ub.append(1 - y)
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((best_centers[i] - best_centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1; row[j] = 1
            A_ub.append(row); b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds_r = [(0, None) for _ in range(n)]
    
    lp_res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if lp_res.success:
        radii = lp_res.x
    
    return best_centers, radii, sum_radii
