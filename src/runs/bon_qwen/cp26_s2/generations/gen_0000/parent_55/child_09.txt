import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False

    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True

def compute_loss(vars, weight):
    """
    Compute the objective (negative sum of radii) plus penalty terms for constraints.
    vars: [x1, y1, r1, x2, y2, r2, ...]
    """
    N = 26
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    loss = -np.sum(r)
    
    # Penalty weight scaling
    w = weight
    
    # 1. Boundary Penalties
    # x - r >= 0  =>  r - x <= 0
    # x + r <= 1  =>  x + r - 1 <= 0
    # y - r >= 0  =>  r - y <= 0
    # y + r <= 1  =>  y + r - 1 <= 0
    
    # We penalize violations.
    # Use smooth penalty or simple squared max.
    # max(0, u)^2
    
    # x - r < 0
    pen_x_low = np.maximum(0, r - x)
    # x + r > 1
    pen_x_high = np.maximum(0, x + r - 1.0)
    # y - r < 0
    pen_y_low = np.maximum(0, r - y)
    # y + r > 1
    pen_y_high = np.maximum(0, y + r - 1.0)
    
    boundary_penalty = np.sum(pen_x_low**2) + np.sum(pen_x_high**2) + \
                       np.sum(pen_y_low**2) + np.sum(pen_y_high**2)
    
    # 2. Overlap Penalties
    # dist(i, j) >= r_i + r_j
    # Violation: r_i + r_j - dist > 0
    
    overlap_penalty = 0.0
    
    # Vectorized distance calculation for speed
    # N is small (26), so loop is fine, but vectorized is faster.
    # x shape (N,), y shape (N,)
    # Compute pairwise squared distances
    # dist_sq[i,j] = (x_i - x_j)^2 + (y_i - y_j)^2
    
    # Using broadcasting
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    # Distances on diagonal are 0, we ignore them.
    # We only care about i < j, but summing all and dividing by 2 works too 
    # if we handle diagonal.
    
    # Safe sqrt
    dist = np.sqrt(np.maximum(dist_sq, 0.0))
    
    # r_i + r_j matrix
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Violation
    violation = r_sum - dist
    # Set diagonal to 0
    np.fill_diagonal(violation, 0)
    
    # We only sum positive violations
    # Since matrix is symmetric, we can sum all and divide by 2, or use triu
    # triu(1) excludes diagonal
    pos_violation = np.maximum(0, violation)
    overlap_penalty = np.sum(pos_violation**2) / 2.0 # Divide by 2 because symmetric sum
    
    return loss + w * (boundary_penalty + overlap_penalty)
