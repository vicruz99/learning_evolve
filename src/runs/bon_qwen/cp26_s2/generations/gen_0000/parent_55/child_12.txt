import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty-based optimization approach with L-BFGS-B.
    Assumes equal radii for all circles to maximize density and sum.
    """
    N = 26
    
    # Objective function to minimize
    # We want to maximize r, so we minimize -r.
    # We add penalties for boundary and overlap violations.
    def objective(vars):
        # vars layout: [x1, y1, ..., xN, yN, r]
        # Total size: 2*N + 1
        centers = vars[:2*N].reshape(N, 2)
        r = vars[2*N]
        
        # 1. Boundary Penalty
        # Circles must be inside [0, 1] x [0, 1]
        # Violation depth: max(0, r - x), max(0, x - (1-r)), etc.
        pen_boundary = 0.0
        for i in range(N):
            x, y = centers[i]
            
            # Left boundary
            if x < r:
                pen_boundary += (x - r)**2
            # Right boundary
            if x > 1 - r:
                pen_boundary += (x - (1 - r))**2
            # Bottom boundary
            if y < r:
                pen_boundary += (y - r)**2
            # Top boundary
            if y > 1 - r:
                pen_boundary += (y - (1 - r))**2
        
        # 2. Overlap Penalty
        # Circles must not overlap: dist >= 2*r
        # Violation depth: max(0, 2*r - dist)
        pen_overlap = 0.0
        # Optimization: Vectorized distance calculation is faster but loop is acceptable for N=26
        # Using a loop for clarity and sufficient speed
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                overlap = 2 * r - dist
                if overlap > 0:
                    pen_overlap += overlap**2
        
        # Weight penalties heavily to enforce constraints
        # The term -r encourages r to grow. 
        # If constraints are satisfied, penalty is 0, and we minimize -r (maximize r).
        penalty_weight = 5000.0
        return -r + penalty_weight * (pen_boundary + pen_overlap)

    # Helper to create initial hexagonal grid configuration
    def get_initial_guess(seed=0):
        np.random.seed(seed)
        centers = np.zeros((N, 2))
        idx = 0

        # We want to fit 26 circles. A 5-row structure works well.
        # Row counts: 5, 6, 5, 6, 4 sums to 26?

        # Let's use a dense grid initialization.

        # y coordinates roughly spaced by 0.2
        # x coordinates roughly spaced by 0.2
        
        # Generate points in a pattern
        # 5 rows
        for row in range(5):
            y = 0.15 + row * 0.18  # Approximate spacing
            
            # Shift odd rows for hexagonal pattern
            shift = 0.09 if row % 2 == 1 else 0.0
            
            # Generate x coordinates
            # We need enough x points. 
            # Spacing 0.18 implies roughly 5-6 points per row.
            x_start = 0.1 + shift
            x_step = 0.18
            
            # Number of points in this row to fill up to N
            # Heuristic distribution
            if row == 0: count = 5
            elif row == 1: count = 6
            elif row == 2: count = 5
            elif row == 3: count = 6
            else: count = 4 # 5+6+5+6+4 = 26
            
            for k in range(count):
                if idx < N:
                    x = x_start + k * x_step
                    # Add small random noise to break symmetry and help optimizer
                    centers[idx, 0] = x + np.random.uniform(-0.01, 0.01)
                    centers[idx, 1] = y + np.random.uniform(-0.01, 0.01)
                    idx += 1
        
        # Fill any remaining slots if heuristic failed (shouldn't happen with above logic)
        while idx < N:
            centers[idx, 0] = np.random.uniform(0.1, 0.9)
            centers[idx, 1] = np.random.uniform(0.1, 0.9)
            idx += 1
            
        # Initial radius (feasible small radius)
        r_start = 0.05
        
        return np.concatenate([centers.flatten(), [r_start]])

    # Bounds for optimization
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)]
    
    best_r = 0.0
    best_vars = None
    
    # Run multiple trials with different random seeds to find global optimum
    num_trials = 10
    for trial in range(num_trials):
        x0 = get_initial_guess(seed=trial)
        
        try:
            # Use L-BFGS-B for bounded optimization
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
            
            if res.success:
                current_r = res.x[2*N]
                # Check if this solution is actually valid (penalty should be near 0)
                # The objective value is -r + penalty.

                if res.fun < -current_r + 1e-6: # Small tolerance for numerical errors

                        best_r = current_r
                        best_vars = res.x
        except Exception:
            continue

        # Actually, the optimizer should find something better than grid.
        pass

        final_centers = best_vars[:2*N].reshape(N, 2)
        final_r = best_vars[2*N]
    else:
        # Default fallback
        final_centers = np.random.uniform(0.1, 0.9, (N, 2))
        final_r = 0.05

    final_radii = np.full(N, final_r)

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

# Verification / Validation function (Read-only, do not modify)
def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    # Check shapes
    if len(centers) != len(radii):
        return False
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False

    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True
