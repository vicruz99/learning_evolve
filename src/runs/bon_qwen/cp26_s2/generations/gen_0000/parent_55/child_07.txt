import numpy as np
from scipy.optimize import linprog, minimize

def compute_optimal_radii(centers):
    """
    Computes the optimal radii for a given set of centers to maximize sum of radii.
    Solves: Maximize sum(r_i) subject to:
    1. r_i >= 0
    2. r_i <= distance to boundaries
    3. r_i + r_j <= distance between centers i and j
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r_i)
    c = -np.ones(n)
    
    # Bounds for r_i: [0, min_dist_to_boundary]
    bounds = []
    for i in range(n):
        x, y = centers[i]
        dist_to_boundary = min(x, 1.0 - x, y, 1.0 - y)
        # Ensure non-negative upper bound
        ub = max(0.0, dist_to_boundary)
        bounds.append((0.0, ub))
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    # These are inequality constraints A_ub @ r <= b_ub
    # We construct the matrix rows for each pair
    
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    row_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            # Calculate Euclidean distance
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            
            # Set coefficients for r_i and r_j
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dist
            row_idx += 1
            
    # Solve LP
    # method='highs' is efficient for dense problems of this size
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x # Return sum of radii and radii vector
    else:
        # Fallback if LP fails (e.g. infeasible), though unlikely with valid centers
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for the center optimizer.
    Minimizes the negative sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    
    # Clip centers to [0, 1] to ensure valid bounds for LP
    centers = np.clip(centers, 0.0, 1.0)
    
    total_sum, radii = compute_optimal_radii(centers)
    
    # Return negative sum for minimization
    return -total_sum

def run_packing() -> tuple:
    """
    Main function to find the optimal packing of 26 circles.
    Returns (centers, radii, sum_radii).

    n_circles = 26
    best_sum = 0.0

    def generate_hex_grid():
        points = []
        # Spacing estimation for ~26 circles
        # Area ~ 1/26, hex area ~ 0.866 * s^2 -> s ~ 0.22
        spacing = 0.22
        sqrt3_half = np.sqrt(3) / 2
        
        y = 0.1 # Start slightly inside
        while y <= 0.9:
            x = 0.1
            # Alternate row offset
            offset = (int(round((y - 0.1) / spacing)) % 2) * (spacing / 2)
            
            current_x = x + offset
            while current_x <= 0.9:
                points.append([current_x, y])
                current_x += spacing
            
            y += spacing * sqrt3_half
            
        # If we have more points than needed, shuffle and pick
        points = np.array(points)
        if len(points) > n_circles:
            np.random.shuffle(points)
            points = points[:n_circles]
        elif len(points) < n_circles:
            # Add random points if grid is too sparse
            while len(points) < n_circles:
                points = np.vstack([points, [np.random.rand(), np.random.rand()]])
                
        return points

    # We will run the optimizer multiple times with different seeds to find global optimum
    # Using Powell method as it is robust for non-smooth objectives
    
    for attempt in range(5): # 5 restarts
        np.random.seed(attempt * 42 + 1)

        # Mix of hexagonal grid and random perturbation
        init_centers = generate_hex_grid()

            )
