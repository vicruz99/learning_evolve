import numpy as np
from scipy.optimize import linprog
import math


def solve_radii_lp(centers, n):
    """
    Given fixed centers, solve LP to find optimal radii maximizing sum.
    Constraints:
      - r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i (boundary)
      - r_i + r_j <= ||c_i - c_j|| (non-overlap)
      - r_i >= 0
    """
    c_obj = -np.ones(n)
    
    constraints = []
    rhs = []
    
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            constraints.append(row)
            rhs.append(val)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints.append(row)
            rhs.append(dist)
    
    A_ub = np.array(constraints)
    b_ub = np.array(rhs)
    bounds = [(0.0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x, -result.fun
    return np.full(n, 0.001), 0.026


def hexagonal_grid(n):
    """Generate hexagonal grid positions for n circles."""
    positions = []
    spacing = 1.0 / math.sqrt(n * 0.75)
    
    y = spacing * 0.5
    row = 0
    while len(positions) < n and y <= 1.0:
        x_offset = 0.0 if row % 2 == 0 else spacing * 0.5
        x = x_offset
        while x <= 1.0 and len(positions) < n:
            if x >= 0:
                positions.append([x, y])
            x += spacing
        y += spacing * math.sqrt(3) / 2
        row += 1
    
    while len(positions) < n:
        positions.append([0.5, 0.5])
    
    return np.array(positions[:n])


def regular_grid(n):
    """Generate regular grid positions for n circles."""
    cols = math.ceil(math.sqrt(n))
    rows = math.ceil(n / cols)
    positions = []
    
    for r in range(rows):
        for c in range(cols):
            if len(positions) >= n:
                break
            x = (c + 0.5) / cols
            y = (r + 0.5) / rows
            positions.append([x, y])
        if len(positions) >= n:
            break
    
    return np.array(positions[:n])


def random_init(n, seed=None):
    """Generate random initial positions."""
    rng = np.random.RandomState(seed)
    return rng.uniform(0.1, 0.9, (n, 2))


def local_search(centers, n, iterations=8000):
    """
    Monte Carlo local search to optimize positions.
    For each trial, perturb one circle and accept if sum of radii improves.
    """
    centers = centers.copy()
    radii, current_sum = solve_radii_lp(centers, n)
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = current_sum
    
    step = 0.12
    
    for it in range(iterations):
        idx = np.random.randint(n)
        perturbation = np.random.uniform(-step, step, 2)
        new_pos = np.clip(centers[idx] + perturbation, 0.015, 0.985)
        
        new_centers = centers.copy()
        new_centers[idx] = new_pos
        
        radii, new_sum = solve_radii_lp(new_centers, n)
        
        if new_sum > best_sum + 1e-14:
            best_sum = new_sum
            best_centers = new_centers.copy()
            best_radii = radii.copy()
            centers = new_centers
            step = min(0.15, step * 1.015)
        else:
            step *= 0.997
    
    return best_centers, best_radii, best_sum


def clustered_init(n):
    """Generate clustered initialization with circles grouped in regions."""
    positions = []
    # Create a perturbed hexagonal grid
    pts = hexagonal_grid(n)
    rng = np.random.RandomState(99)
    for i in range(n):
        pts[i] += rng.uniform(-0.02, 0.02, 2)
        pts[i] = np.clip(pts[i], 0.05, 0.95)
    return pts


def boundary_init(n):
    """Initialize with emphasis on boundary and corner regions."""
    positions = []
    rng = np.random.RandomState(77)
    
    # Some circles near corners
    corners = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    positions.extend(corners)
    
    # Some circles along edges
    for _ in range(8):
        side = rng.randint(4)
        if side == 0:  # top
            positions.append([rng.uniform(0.1, 0.9), 0.12])
        elif side == 1:  # bottom
            positions.append([rng.uniform(0.1, 0.9), 0.88])
        elif side == 2:  # left
            positions.append([0.12, rng.uniform(0.1, 0.9)])
        else:  # right
            positions.append([0.88, rng.uniform(0.1, 0.9)])
    
    # Rest in interior
    while len(positions) < n:
        positions.append(rng.uniform(0.2, 0.8, 2).tolist())
    
    return np.array(positions[:n])
