import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Generate a robust hexagonal initial guess
    # We aim for a hexagonal packing which is denser than a square grid.
    # With 26 circles, we can arrange them in rows like 5, 5, 5, 5, 6 (or similar)
    # but shifted.
    
    # Estimate a starting radius. For N=26 in a square, r is approx 0.101
    r_est = 0.101
    
    # Generate points in a hexagonal lattice pattern
    # Pitch x = 2r, Pitch y = r * sqrt(3)
    pts = []
    rows = [6, 5, 5, 5, 5] # Total 26
    y = r_est
    for i, count in enumerate(rows):
        # Shift alternate rows by r_est to fit in hexagonal packing
        x_start = r_est + (i % 2) * r_est
        for j in range(count):
            x = x_start + j * (2 * r_est)
            pts.append([x, y])
        y += r_est * np.sqrt(3)

    # Normalize to fit in unit square if necessary, though r_est should be close
    centers = np.array(pts[:n])
    
    # If points are out of bounds, we just clip/scale for the initial guess
    # to ensure the optimizer starts with a valid or near-valid configuration.
    centers = np.clip(centers, 0.01, 0.99)
    
    # Initialize radii
    radii = np.full(n, r_est)

    # --- Optimization Setup ---

    def objective(x_flat):
        """Objective to minimize: Negative sum of radii + penalty for violations."""
        centers_opt = x_flat[:2*n].reshape((n, 2))
        radii_opt = x_flat[2*n:]
        
        # Ensure radii are positive for calculations
        r = np.maximum(radii_opt, 1e-6)
        
        cost = -np.sum(r)
        penalty = 0.0
        
        # Boundary constraint penalty
        # Center - r >= 0 => r - x <= 0
        # Center + r <= 1 => x + r - 1 <= 0
        for i in range(n):
            x, y = centers_opt[i]
            ri = r[i]
            
            # Left/Right
            if x - ri < 0: penalty += 1000 * (x - ri)**2
            if x + ri > 1: penalty += 1000 * (x + ri - 1)**2
            # Top/Bottom
            if y - ri < 0: penalty += 1000 * (y - ri)**2
            if y + ri > 1: penalty += 1000 * (y + ri - 1)**2
            
        # Overlap constraint penalty
        # dist >= r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers_opt[i, 0] - centers_opt[j, 0])**2 + 
                               (centers_opt[i, 1] - centers_opt[j, 1])**2)
                min_dist = r[i] + r[j]
                if dist < min_dist:
                    penalty += 1000 * (min_dist - dist)**2
                    
        return cost + penalty

    # Differential Evolution for global search
    # Bounds: centers in [0, 1], radii in [0, 0.5]
    bounds_opt = []
    for _ in range(2*n):
        bounds_opt.append((0.0, 1.0))
    for _ in range(n):
        bounds_opt.append((1e-4, 0.5))
        
    # Run DE with a reasonable number of iterations
    res_de = minimize(objective, centers.ravel().tolist() + radii.tolist(), 
                      method='Nelder-Mead', 
                      options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-8})
                      
    # Refine with L-BFGS-B if possible, but Nelder-Mead is often better for non-smooth
    # Let's just run a second pass of Nelder-Mead from the best result
    
    best_x = res_de.x
    best_val = res_de.fun
    
    # Multiple restarts to ensure we find the global max
    for _ in range(5):
        # Add small noise
        noisy_x = best_x + np.random.normal(0, 0.005, len(best_x))
        noisy_x = np.clip(noisy_x, [b[0] for b in bounds_opt], [b[1] for b in bounds_opt])
        
        res_restart = minimize(objective, noisy_x, method='Nelder-Mead', 
                               options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-8})
        
        if res_restart.fun < best_val:
            best_x = res_restart.x
            best_val = res_restart.fun

    # Extract final results
    final_centers = best_x[:2*n].reshape((n, 2))
    final_radii = best_x[2*n:]
    
    # Clean up radii (ensure non-negative)
    final_radii = np.maximum(final_radii, 0.0)
    
    sum_r = np.sum(final_radii)
    
    return final_centers, final_radii, sum_r
