from scipy.optimize import minimize


def objective(x, n):
    """Negative sum of radii (minimization)."""
    return -np.sum(x[2 * n :])


def dist_constraints(x, n):
    """Squared distance constraints: dist^2 >= (r_i + r_j)^2."""
    n_constraints = n * (n - 1) // 2
    vals = np.empty(n_constraints)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi = x[2 * i], x[2 * i + 1]
            xj, yj = x[2 * j], x[2 * j + 1]
            ri, rj = x[2 * n + i], x[2 * n + j]
            dist_sq = (xi - xj) ** 2 + (yi - yj) ** 2
            min_dist_sq = (ri + rj) ** 2
            vals[idx] = dist_sq - min_dist_sq
            idx += 1
    return vals


def boundary_constraints(x, n):
    """Boundary and non-negativity constraints."""
    vals = np.empty(5 * n)
    for i in range(n):
        xi, yi = x[2 * i], x[2 * i + 1]
        ri = x[2 * n + i]

        vals[5 * i + 2] = yi - ri
        vals[5 * i + 3] = 1.0 - yi - ri
        vals[5 * i + 4] = ri
    return vals


def all_constraints(x, n):
    """Combine all constraints."""
    return np.concatenate([dist_constraints(x, n), boundary_constraints(x, n)])


def constraint_fun(x, n):
    """Wrapper for constraints."""
    return all_constraints(x, n)

        x_start = initial_r + offset
        for col_idx in range(size):
            x = x_start + col_idx * 2 * initial_r
            centers.append([x, y])
            radii.append(initial_r)

    return np.array(centers), np.array(radii)


def clip_valid(centers, radii):
    """Clip centers to valid range given radii."""
    for i in range(len(radii)):

        centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1 - r, centers[i, 1]))
    return centers

    cons = {"type": "ineq", "fun": constraint_fun, "args": (n,)}
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    result = minimize(
        objective,
        x0,
        args=(n,),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": tol},
    )
    return result
