To maximize the sum of radii for 26 circles in a unit square, we must achieve a high packing density. A standard $5 \times 5$ grid allows for circles of radius 0.1 (sum = 2.5), but fitting 26 circles requires a denser arrangement or non-equal radii. A hexagonal lattice is denser than a square grid.

My strategy is:
1.  **Initialization**: Create a structured initial configuration of 26 circles based on a hexagonal-like pattern. Specifically, I will use 5 rows with circle counts 6, 5, 5, 5, 5. This distributes the circles well across the square and approximates a dense packing.
2.  **Optimization**: Use `scipy.optimize.minimize` with the L-BFGS-B method to optimize the positions (centers) and sizes (radii) of the circles.
    *   **Objective**: Maximize the sum of radii (implemented as minimizing negative sum).
    *   **Constraints**: Enforce non-overlap and boundary constraints using a penalty function. The penalty adds a large cost to the objective if circles overlap or go outside the unit square.
    *   **Variables**: The optimizer adjusts all 52 coordinates (26 centers) and 26 radii simultaneously.
3.  **Refinement**: The penalty-based approach allows the optimizer to "push" circles apart and expand their radii until they reach the physical limits imposed by the square and each other.
4.  **Compliance**: The code adheres to all rules: top-level helper functions, no closures, no lambdas, and returns the required format.

```python
import numpy as np
import scipy.optimize

def _packing_objective(x, i_idx, j_idx):
    """
    Objective function for circle packing optimization.
    Minimizes negative sum of radii plus penalties for constraint violations.
    
    Args:
        x: Flattened array of [x1, y1, ..., x26, y26, r1, ..., r26]
        i_idx: Indices for pairwise overlap checks (i)
        j_idx: Indices for pairwise overlap checks (j)
    """
    n = 26
    # Reshape centers and radii
    c = x[:52].reshape(n, 2)
    r = x[52:]
    
    # Objective: Maximize sum of radii => Minimize -sum(r)
    val = -np.sum(r)
    penalty = 0.0
    
    # --- Boundary Constraints ---
    # Circle i must be inside [0, 1] x [0, 1]
    # x_i - r_i >= 0  =>  r_i - x_i <= 0
    # x_i + r_i <= 1  =>  x_i + r_i - 1 <= 0
    # Same for y
    
    # Left boundary violation
    viol = r - c[:, 0]
    pos = viol[viol > 0]
    penalty += 1000.0 * np.sum(pos**2)
    
    # Right boundary violation
    viol = c[:, 0] + r - 1.0
    pos = viol[viol > 0]
    penalty += 1000.0 * np.sum(pos**2)
    
    # Bottom boundary violation
    viol = r - c[:, 1]
    pos = viol[viol > 0]
    penalty += 1000.0 * np.sum(pos**2)
    
    # Top boundary violation
    viol = c[:, 1] + r - 1.0
    pos = viol[viol > 0]
    penalty += 1000.0 * np.sum(pos**2)
    
    # --- Overlap Constraints ---
    # Distance between centers >= sum of radii
    # dist(i, j) >= r_i + r_j
    
    c_i = c[i_idx, :]
    c_j = c[j_idx, :]
    r_i = r[i_idx]
    r_j = r[j_idx]
    
    dx = c_i[:, 0] - c_j[:, 0]
    dy = c_i[:, 1] - c_j[:, 1]
    dists = np.sqrt(dx*dx + dy*dy)
    
    # Violation: (r_i + r_j) - dist > 0
    violations = (r_i + r_j) - dists
    pos_viol = violations[violations > 0]
    penalty += 1000.0 * np.sum(pos_viol**2)
    
    return val + penalty

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns centers, radii, and sum_radii.
    """
    n = 26
    
    # --- Initialization ---
    # We initialize with a pattern resembling a hexagonal packing.
    # Pattern: 5 rows with 6, 5, 5, 5, 5 circles respectively (Total 26).
    # This provides a dense and feasible starting configuration.
    
    centers = []
    # Y coordinates for 5 rows, spaced to fill the vertical space
    ys = [0.12, 0.33, 0.50, 0.67, 0.88]
    
    # Row 0: 6 circles
    # Spread horizontally to cover the width
    xs = np.linspace(0.05, 0.95, 6)
    for x in xs:
        centers.append([x, ys[0]])
        
    # Row 1: 5 circles (shifted relative to Row 0 to mimic hexagonal packing)
    xs = np.linspace(0.12, 0.88, 5)
    for x in xs:
        centers.append([x, ys[1]])
        
    # Row 2: 5 circles
    xs = np.linspace(0.05, 0.95, 5)
    for x in xs:
        centers.append([x, ys[2]])
        
    # Row 3: 5 circles (shifted)
    xs = np.linspace(0.12, 0.88, 5)
    for x in xs:
        centers.append([x, ys[3]])
        
    # Row 4: 5 circles
    xs = np.linspace(0.05, 0.95, 5)
    for x in xs:
        centers.append([x, ys[4]])
        
    centers = np.array(centers)
    radii = np.full(n, 0.04) # Initial radius small enough to avoid immediate overlap
    
    # Precompute indices for pairwise constraints to pass to objective
    i_idx, j_idx = np.triu_indices(n, k=1)
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    # x and y coordinates: 52 variables
    # Radii: 26 variables
    bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)] * n
    
    # Flatten initial guess: [x1, y1, ..., x26, y26, r1, ..., r26]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Optimization
    # Use L-BFGS-B which supports bounds
    try:
        res = scipy.optimize.minimize(_packing_objective, x0, method='L-BFGS-B', 
                                      args=(i_idx, j_idx), bounds=bounds, 
                                      options={'maxiter': 5000, 'ftol': 1e-12})
        x_opt = res.x
    except Exception:
        # Fallback to initial configuration if optimization fails
        x_opt = x0
        
    # Extract solution
    centers_opt = x_opt[:52].reshape(n, 2)
    radii_opt = x_opt[52:]
    
    # Ensure physical validity (non-negative radii, within bounds)
    radii_opt = np.maximum(radii_opt, 0.0)
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
```
