import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    """
    n = 26
    # Initialize centers and radii
    # Hexagonal packing heuristic initialization
    rows = 5
    cols = 5
    
    # Start with a dense hexagonal-ish arrangement
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09  # Initial guess
    
    idx = 0
    # Generate a grid that is roughly hexagonal
    # We will fill 5 rows. 
    # Row 0: 5 circles
    # Row 1: 5 circles (shifted)
    # ...
    # To get 26, we might need to adjust counts or just start with a good scatter.
    
    # Let's try to generate points in a grid and then let the optimizer find the best radii.
    # A simple 5x5 grid with one extra point in a gap is a good start.
    
    # 5x5 grid
    grid_size = 5
    base_spacing = 0.2
    offset = 0.1
    
    # Add 25 circles in a grid
    for r in range(grid_size):
        for c in range(grid_size):
            centers[idx] = [offset + c * base_spacing, offset + r * base_spacing]
            idx += 1
            
    # Add 26th circle in a gap (center of the square, or a gap)
    # In a 0.1 radius grid, gaps are at (0.2, 0.2) etc? No, centers are 0.1, 0.3...
    # Gap centers are at (0.2, 0.2). Distance to (0.1, 0.1) is sqrt(0.02) ~ 0.141.
    # Radius 0.1 fits if gap radius is 0.141 - 0.1 = 0.041.
    centers[25] = [0.2, 0.2] # Just a starting position
    
    # Variable vector: [x0, y0, r0, x1, y1, r1, ...]
    # Total 3 * 26 = 78 variables
    x0 = np.hstack([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r >= 0. (We can bound r <= 0.5)
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)]) # x, y, r
        
    def objective(vars_vec):
        # Maximize sum of radii => Minimize negative sum
        r = vars_vec[2::3]
        return -np.sum(r)
    
    def constraint_overlap(vars_vec):
        # dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # vars_vec: [x0, y0, r0, x1, y1, r1, ...]
        # i: 3i, 3i+1, 3i+2
        
        constraints = []
        for i in range(n):
            xi = vars_vec[3*i]
            yi = vars_vec[3*i+1]
            ri = vars_vec[3*i+2]
            
            for j in range(i + 1, n):
                xj = vars_vec[3*j]
                yj = vars_vec[3*j+1]
                rj = vars_vec[3*j+2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                # dist_sq >= sum_r^2
                # return value >= 0
                constraints.append(dist_sq - sum_r**2)
        
        return np.array(constraints)

    def constraint_boundary(vars_vec):
        # x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
        cons = []
        for i in range(n):
            xi = vars_vec[3*i]
            yi = vars_vec[3*i+1]
            ri = vars_vec[3*i+2]
            
            cons.append(xi - ri)
            cons.append(1 - (xi + ri))
            cons.append(yi - ri)
            cons.append(1 - (yi + ri))
            
        return np.array(cons)

    # Combine constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Run optimization
    # Using SLSQP
    res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
    
    if res.success:
        final_vars = res.x
        final_centers = np.zeros((n, 2))
        final_radii = np.zeros(n)
        
        for i in range(n):
            final_centers[i] = [final_vars[3*i], final_vars[3*i+1]]
            final_radii[i] = final_vars[3*i+2]
            
        sum_radii = np.sum(final_radii)
        return final_centers, final_radii, sum_radii
    else:
        # Fallback to initial or simple grid if optimization fails
        # This should not happen with a valid start
        return centers, radii, np.sum(radii)
