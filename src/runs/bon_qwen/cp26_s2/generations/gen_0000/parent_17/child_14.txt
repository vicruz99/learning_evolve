import numpy as np
from scipy.optimize import minimize, dual_annealing
import math

def run_packing():
    """
    Finds a packing of 26 circles in a unit square maximizing the sum of radii.
    Uses a hybrid approach:
    1. Initialize with a dense hexagonal-like grid.
    2. Use force-directed simulation to resolve overlaps and expand radii.
    3. Refine using scipy.optimize with soft constraints.
    """
    np.random.seed(42)
    N = 26

    # 1. Initialization
    # Start with a grid configuration that is likely to be close to optimal.
    # A 5x5 grid has 25 circles of radius 0.1. We add a 26th in the center gap.
    # Then we optimize.
    
    # Grid parameters
    # 5x5 grid centers
    grid_coords = []
    for i in range(5):
        for j in range(5):
            x = 0.1 + i * 0.2
            y = 0.1 + j * 0.2
            grid_coords.append([x, y])
    
    # Add 26th circle in the center gap (0.2, 0.2) relative to 0.1,0.1? 
    # Center of the square formed by (0.1,0.1), (0.3,0.1), (0.1,0.3), (0.3,0.3) is (0.2, 0.2).
    grid_coords.append([0.2, 0.2])
    
    # Initial radii: Large for grid, small for the extra one
    initial_radii = [0.1] * 25 + [0.01] * 1
    initial_centers = np.array(grid_coords)
    
    # Better initialization: Randomize slightly to break symmetry
    centers = initial_centers.copy()
    radii = np.array(initial_radii)
    centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
    
    # Ensure inside bounds
    radii = np.clip(radii, 0, 0.5)
    
    # 2. Force-Directed Optimization (Expansion + Repulsion)
    # This step tries to grow radii while pushing overlapping circles apart.
    
    num_steps = 2000
    current_sum_radii = np.sum(radii)
    
    # Parameters
    repulsion_strength = 50.0
    expansion_rate = 1.0005 # Slow growth
    damping = 0.5
    
    velocities = np.zeros_like(centers)
    
    for step in range(num_steps):
        # Grow radii slightly
        radii *= expansion_rate
        
        # Calculate forces
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        for i in range(N):
            for j in range(i + 1, N):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                
                if dist < 1e-9:
                    dist = 1e-9
                    diff = np.array([1e-9, 0.0]) # Avoid div by zero
                
                overlap = min_dist - dist
                if overlap > 0:
                    # Force proportional to overlap, direction away from each other
                    force_vec = (diff / dist) * overlap * repulsion_strength
                    forces[i] += force_vec
                    forces[j] -= force_vec
        
        # Boundary repulsion
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            
            # Left
            if x - r < 0:
                forces[i, 0] += (0 - (x - r)) * repulsion_strength
            # Right
            if x + r > 1:
                forces[i, 0] -= ((x + r) - 1) * repulsion_strength
            # Bottom
            if y - r < 0:
                forces[i, 1] += (0 - (y - r)) * repulsion_strength
            # Top
            if y + r > 1:
                forces[i, 1] -= ((y + r) - 1) * repulsion_strength
                
        # Update velocities and positions
        velocities = velocities * damping + forces * 0.01 # Small step size for forces
        centers += velocities
        
        # Project velocities if we want to stabilize, but simple integration works
        
        # Periodic check: if configuration is valid, record it
        # But for optimization, we just keep running.
        
    # After force simulation, radii might be too large (overlaps exist).
    # We need to find the maximum uniform scaling factor or adjust radii.
    # Actually, the force simulation maintained constraints roughly, but with growth,
    # it might have pushed circles out or allowed slight overlaps.
    # Let's do a final clean-up using scipy to maximize sum of radii
    # subject to hard constraints, starting from current state.
    
    # 3. Refinement using scipy.optimize
    # We optimize variables: [x1, y1, r1, ..., x26, y26, r26]
    
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(N):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r
    
    # Define constraints
    # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # => x - r >= 0, 1 - x - r >= 0, etc.
    # 2. Non-overlap: dist >= r_i + r_j => dist^2 >= (r_i + r_j)^2
    
    constraints = []
    
    # Boundary constraints
    for i in range(N):
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[3*i] - v[3*i+2]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]
        })
        
    # Pairwise constraints
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: 
                    (v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2 - (v[3*i+2] + v[3*j+2])**2
            })
            
    # Objective: Maximize sum of radii => Minimize negative sum
    def objective(v):
        s = 0.0
        for i in range(N):
            s += v[3*i+2]
        return -s

    # Run optimizer
    # SLSQP is good for constraints
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        if res.success or res.fun < -2.0: # Check if improvement
            x0 = res.x
    except Exception:
        pass # Fallback to previous state
    
    # Extract results
    final_centers = np.zeros((N, 2))
    final_radii = np.zeros(N)
    
    for i in range(N):
        final_centers[i, 0] = x0[3*i]
        final_centers[i, 1] = x0[3*i+1]
        final_radii[i] = x0[3*i+2]
        
    # Validate and clamp just in case
    # Although optimizer should handle it, numerical errors might occur.
    # If a circle is slightly out, shrink radius.
    for i in range(N):
        x, y = final_centers[i]
        r = final_radii[i]
        # Boundary clamping
        min_dist_bound = min(x, 1-x, y, 1-y)
        if min_dist_bound < r:
            r = max(0, min_dist_bound)
            final_radii[i] = r
            
    # Final validation check (optional, but good for sanity)
    # We assume the optimizer did a good job.
    
    total_sum = np.sum(final_radii)
    
    return final_centers, final_radii, total_sum
