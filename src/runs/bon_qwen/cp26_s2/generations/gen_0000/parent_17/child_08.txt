    *   **Optimization**: Use the L-BFGS-B solver to adjust the $(x, y)$ centers and radii of all 26 circles. We add a small gradient descent step to the penalty function to actively maximize the sum of radii.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Initialization Strategy: 5-6-5-6-4 Hexagonal Pattern ---
    centers = []
    radii = []
    
    # Target radius roughly 0.1014 to reach sum 2.636
    r_init = 0.101
    row_height = r_init * np.sqrt(3)
    
    y_pos = r_init
    pattern = [5, 6, 5, 6, 4]
    
    for row_idx, count in enumerate(pattern):
        # In hexagonal packing, rows with different counts usually shift.
        # Even rows (indices 1, 3) with count 6 are shifted right by r_init
        # to fit between the circles of the 5-count rows.
        if row_idx % 2 == 1:
            x_start = r_init + r_init 
        else:
            x_start = r_init
            
        for i in range(count):
            x = x_start + i * (2 * r_init)
            centers.append([x, y_pos])
            radii.append(r_init)
            
        y_pos += row_height

    # Convert to arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # --- 2. Optimization Function ---
    
    def objective(x_vars):
        # x_vars contains: [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
        # Total length = 26 * 2 + 26 = 78
        
        centers_opt = x_vars[:52].reshape(26, 2)
        radii_opt = x_vars[52:]
        
        # 1. Sum of radii (we minimize negative sum to maximize)
        # We weight it heavily to drive the optimizer to increase radii
        cost = -np.sum(radii_opt) * 1000.0 
        
        # 2. Penalty for overlaps
        # dist < r1 + r2
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j]) ** 2))
                r_sum = radii_opt[i] + radii_opt[j]
                if dist < r_sum:
                    overlap = r_sum - dist
                    cost += overlap ** 2 * 10000.0

        # 3. Penalty for boundaries
        # x - r < 0 or x + r > 1
        for i in range(n):
            r = radii_opt[i]
            x, y = centers_opt[i]
            
            # Left
            if x < r: cost += (r - x)**2 * 10000.0
            # Right
            if x > 1 - r: cost += (x - (1 - r))**2 * 10000.0
            # Bottom
            if y < r: cost += (r - y)**2 * 10000.0
            # Top
            if y > 1 - r: cost += (y - (1 - r))**2 * 10000.0
            
            # Radii must be positive
            if r < 0.01: cost += (0.01 - r)**2 * 10000.0

        return cost

    # --- 3. Setup Variables and Bounds ---
    # Flattened initial state
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: 
    # x, y in [0, 1]
    # r in [0.05, 0.2] (Reasonable limits for 26 circles)
    bounds = []
    for _ in range(26):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
    for _ in range(26):
        bounds.append((0.05, 0.2)) # r

    # --- 4. Run Optimization ---
    # Use L-BFGS-B for bounded optimization
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 2000, 'ftol': 1e-12})
    
    # --- 5. Post-Processing and Validation ---
    final_centers = res.x[:52].reshape(26, 2)
    final_radii = res.x[52:]
    
    # Clip radii to ensure no negative/nan
    final_radii = np.clip(final_radii, 1e-6, None)
    
    # Safety check: if validation fails, fallback to safe grid
    # We define a simple validator inside to avoid external dependency issues
    # though the prompt says we don't modify the validator function.
    
    # Simple overlap check logic
    valid = True
    for i in range(n):

        r = final_radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            valid = False; break
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((final_centers[i] - final_centers[j]) ** 2))
            if dist < final_radii[i] + final_radii[j] - 1e-9:
                valid = False; break
        if not valid: break

    if not valid:
