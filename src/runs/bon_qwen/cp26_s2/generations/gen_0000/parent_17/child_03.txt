import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns centers (26, 2), radii (26,), and sum_radii.
    """
    n = 26
    n_vars = 3 * n  # x, y, r for each circle
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # Objective function: Minimize -sum(r)
    def objective(v):
        r = v[2::3]
        return -np.sum(r)
        
    # Constraint function: Returns vector of constraint values (must be >= 0)
    def constraints(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        
        # 1. Boundary constraints
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        c_bound = np.concatenate([
            x - r,
            1.0 - x - r,
            y - r,
            1.0 - y - r
        ])
        
        # 2. Overlap constraints
        # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
        x_col = x.reshape(-1, 1)
        y_col = y.reshape(-1, 1)
        r_col = r.reshape(-1, 1)
        
        # Distance squared matrix
        diff_x = x_col - x_col.T
        diff_y = y_col - y_col.T
        D2 = diff_x**2 + diff_y**2
        
        # Radius sum squared matrix
        r_sum = r_col + r_col.T
        R2 = r_sum**2
        
        # Constraint values
        C = D2 - R2
        
        # Extract upper triangular part (i < j) to avoid duplicates and self-constraints
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        c_overlap = C[mask]
        
        return np.concatenate([c_bound, c_overlap])
        
    # Number of constraints
    n_bound = 4 * n
    n_overlap = n * (n - 1) // 2
    n_total_cons = n_bound + n_overlap
    
    # Jacobian of constraints
    def constraints_jac(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        
        jac = np.zeros((n_total_cons, n_vars))
        
        # 1. Boundary constraints Jacobian
        # Rows 0 to n-1: x - r >= 0 -> d/dx=1, d/dr=-1
        for i in range(n):
            jac[i, 3*i] = 1.0
            jac[i, 3*i+2] = -1.0
            
        # Rows n to 2n-1: 1 - x - r >= 0 -> d/dx=-1, d/dr=-1
        for i in range(n):
            jac[n+i, 3*i] = -1.0
            jac[n+i, 3*i+2] = -1.0
            
        # Rows 2n to 3n-1: y - r >= 0 -> d/dy=1, d/dr=-1
        for i in range(n):
            jac[2*n+i, 3*i+1] = 1.0
            jac[2*n+i, 3*i+2] = -1.0
            
        # Rows 3n to 4n-1: 1 - y - r >= 0 -> d/dy=-1, d/dr=-1
        for i in range(n):
            jac[3*n+i, 3*i+1] = -1.0
            jac[3*n+i, 3*i+2] = -1.0
            
        # 2. Overlap constraints Jacobian
        # Indices start at 4n
        base_idx = 4 * n
        row_idx = 0
        
        for i in range(n):
            for j in range(i + 1, n):
                c_idx = base_idx + row_idx
                
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dr = r[i] + r[j]
                
                # Constraint: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
                
                # w.r.t x[i]: 2(xi-xj)
                jac[c_idx, 3*i] = 2.0 * dx
                # w.r.t x[j]: -2(xi-xj)
                jac[c_idx, 3*j] = -2.0 * dx
                
                # w.r.t y[i]: 2(yi-yj)
                jac[c_idx, 3*i+1] = 2.0 * dy
                # w.r.t y[j]: -2(yi-yj)
                jac[c_idx, 3*j+1] = -2.0 * dy
                
                # w.r.t r[i]: -2(ri+rj)
                jac[c_idx, 3*i+2] = -2.0 * dr
                # w.r.t r[j]: -2(ri+rj)
                jac[c_idx, 3*j+2] = -2.0 * dr
                
                row_idx += 1
                
        return jac

    # Define NonlinearConstraint
    nl_constraints = NonlinearConstraint(constraints, np.zeros(n_total_cons), np.inf, jac=constraints_jac)

    best_centers = None
    best_radii = None
    
    # Helper to create initial guess
    def make_v0(method):
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        
        if method == 'random':
            centers[:, 0] = np.random.uniform(0.1, 0.9, n)
            centers[:, 1] = np.random.uniform(0.1, 0.9, n)
            radii[:] = 0.02
        elif method == 'grid':
            cols = 6
            rows = 5
            idx = 0
            for r in range(rows):
                for c in range(cols):
                    if idx < n:
                        centers[idx, 0] = (c + 1) * (1.0 / (cols + 1))
                        centers[idx, 1] = (r + 1) * (1.0 / (rows + 1))
                        radii[idx] = 0.02
                        idx += 1
        elif method == 'hex':
            # Hexagonal packing heuristic
            r_init = 0.08
            radii[:] = r_init
            
            row_y = r_init
            row_idx = 0
            count = 0
            while count < n:
                offset = r_init if row_idx % 2 == 1 else 0
                row_x = r_init + offset
                while row_x + r_init <= 1.0 and count < n:
                    centers[count, 0] = row_x
                    centers[count, 1] = row_y
                    count += 1
                    row_x += 2 * r_init
                row_y += r_init * np.sqrt(3)
                row_idx += 1
            
            for k in range(count, n):
                centers[k, 0] = np.random.rand() * 0.8 + 0.1
                centers[k, 1] = np.random.rand() * 0.8 + 0.1
                radii[k] = 0.02

        v0 = np.zeros(3 * n)
        for i in range(n):
            v0[3*i] = centers[i, 0]
            v0[3*i+1] = centers[i, 1]
            v0[3*i+2] = radii[i]
        return v0

    # Run optimization with different seeds
    methods = ['hex', 'grid', 'random', 'random', 'random']
    np.random.seed(42) 

    for m in methods:
        v0 = make_v0(m)
        try:
            # SLSQP is generally robust for this problem size
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=nl_constraints, 
                           options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
            
            # Verify feasibility
            c_val = constraints(res.x)
            if np.min(c_val) >= -1e-7: # Tolerance for numerical errors
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = np.zeros((n, 2))
                    best_radii = np.zeros(n)
                    for i in range(n):
                        best_centers[i] = [res.x[3*i], res.x[3*i+1]]
                        best_radii[i] = res.x[3*i+2]
        except Exception:
            continue

    if best_centers is None:
        # Fallback valid configuration
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.01)
        for i in range(n):
            centers[i] = [0.1 + i * 0.03, 0.5]
        best_sum = np.sum(radii)
        best_centers = centers
        best_radii = radii
        
    return best_centers, best_radii, best_sum
