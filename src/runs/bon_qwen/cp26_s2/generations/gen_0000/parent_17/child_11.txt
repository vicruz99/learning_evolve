   - Non-negative radii

3. **Multiple Restarts**: Run optimization from several different starting configurations to avoid local optima.

4. **Constraint Implementation**: Define all constraint functions at the top level without closures by using a global constant N=26 and a single constraint function that returns all constraint values as an array.

5. **Fallback**: Include an iterative force-directed approach as fallback if scipy optimization fails.

import numpy as np
from scipy.optimize import minimize
import random

N = 26

def objective(vars):
    """Maximize sum of radii (negate for minimization)."""
    radii = vars[2::3]
    return -np.sum(radii)

def constraints_func(vars):
    """Return all inequality constraints as a single array.
    
    Constraints:
    - x_i - r_i >= 0 (left boundary)
    - 1 - x_i - r_i >= 0 (right boundary)  
    - y_i - r_i >= 0 (bottom boundary)
    - 1 - y_i - r_i >= 0 (top boundary)
    - r_i >= 0 (non-negative radius)
    - dist(i,j) - r_i - r_j >= 0 (non-overlap)
    """
    n = N
    result = []
    
    # Boundary and non-negativity constraints (5 per circle)
    for i in range(n):
        result.append(vars[3*i] - vars[3*i + 2])
        result.append(1.0 - vars[3*i] - vars[3*i + 2])
        result.append(vars[3*i + 1] - vars[3*i + 2])
        result.append(1.0 - vars[3*i + 1] - vars[3*i + 2])
        result.append(vars[3*i + 2])
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = vars[3*i] - vars[3*j]
            dy = vars[3*i + 1] - vars[3*j + 1]
            dist = np.sqrt(dx * dx + dy * dy)
            result.append(dist - vars[3*i + 2] - vars[3*j + 2])
    
    return np.array(result)

def create_hexagonal_initial(n, perturbation_scale=0.0):
    """Create initial hexagonal grid placement for n circles."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.04)
    
    idx = 0
    spacing_x = 0.19
    spacing_y = spacing_x * np.sqrt(3) / 2
    
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            x = 0.1 + col * spacing_x + (spacing_x / 2 if row % 2 == 1 else 0)
            y = 0.1 + row * spacing_y
            
            if perturbation_scale > 0:
                x += np.random.normal(0, perturbation_scale)
                y += np.random.normal(0, perturbation_scale)
            
            centers[idx] = [np.clip(x, 0.05, 0.95), np.clip(y, 0.05, 0.95)]
            idx += 1
    
    return centers, radii

def create_random_initial(n):
    """Create random initial placement for n circles."""
    centers = np.random.uniform(0.15, 0.85, size=(n, 2))
    radii = np.full(n, 0.03)
    return centers, radii

def iterative_force_optimization(centers, radii, n, iterations=50000, growth_rate=1.00005):
    """Iteratively grow circles and resolve conflicts using force-directed method."""
    for iteration in range(iterations):
        # Grow radii slightly
        radii *= growth_rate
        
        # Enforce boundary constraints by moving centers
        for i in range(n):
            for dim in range(2):
                if centers[i, dim] - radii[i] < 0:
                    centers[i, dim] = radii[i]
                if centers[i, dim] + radii[i] > 1:
                    centers[i, dim] = 1.0 - radii[i]
        
        # Resolve overlaps with repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-12:
                    # Push circles apart proportionally
                    push_strength = (min_dist - dist) * 0.5
                    direction = diff / dist
                    centers[i] += direction * push_strength
                    centers[j] -= direction * push_strength
        
        # Re-enforce boundaries after push
        for i in range(n):
            for dim in range(2):
                if centers[i, dim] - radii[i] < 0:
                    centers[i, dim] = radii[i]
                if centers[i, dim] + radii[i] > 1:
                    centers[i, dim] = 1.0 - radii[i]
    
    return centers, radii

def optimize_with_scipy(centers, radii, n):
    """Optimize using scipy SLSQP."""
    vars0 = np.zeros(3 * n)
    for i in range(n):
        vars0[3 * i] = centers[i, 0]
        vars0[3 * i + 1] = centers[i, 1]
        vars0[3 * i + 2] = radii[i]
    
    constraint = {'type': 'ineq', 'fun': constraints_func}
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    result = minimize(
        objective, 
        vars0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=[constraint], 
        options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
    )
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(result.x[2 * n:], 0)
    
    return centers_opt, radii_opt, -result.fun

def run_packing():
    n = N
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Strategy 1: Multiple scipy optimizations from different starts
    initial_configs = []
    
    # Config 1: Hexagonal with small perturbation
    c, r = create_hexagonal_initial(n, perturbation_scale=0.0)
    initial_configs.append((c, r))
    
    # Config 2: Hexagonal with larger perturbation
    c, r = create_hexagonal_initial(n, perturbation_scale=0.03)
    initial_configs.append((c, r))
    
    # Config 3-6: Random perturbations of hexagonal
    for _ in range(4):
        c, r = create_hexagonal_initial(n, perturbation_scale=0.05)
        initial_configs.append((c, r))
    
    # Config 7-8: Random initial
    for _ in range(2):
        c, r = create_random_initial(n)
        initial_configs.append((c, r))
    
    for centers, radii in initial_configs:
        try:
            opt_centers, opt_radii, opt_sum = optimize_with_scipy(centers, radii, n)
            if opt_sum > best_sum:
                best_sum = opt_sum
                best_centers = opt_centers.copy()
                best_radii = opt_radii.copy()
        except Exception:
            pass
    
    # Strategy 2: Iterative force optimization (often finds better solutions)
    centers_force, radii_force = create_hexagonal_initial(n, perturbation_scale=0.01)
    centers_force, radii_force = iterative_force_optimization(
        centers_force, radii_force, n, iterations=80000, growth_rate=1.00003
    )
    force_sum = np.sum(radii_force)
    
    if force_sum > best_sum:
        best_sum = force_sum
        best_centers = centers_force.copy()
        best_radii = radii_force.copy()
    
    # Strategy 3: Hybrid - use scipy result, then refine with force method
    if best_centers is not None:
        hybrid_centers, hybrid_radii = best_centers.copy(), best_radii.copy()
        hybrid_centers, hybrid_radii = iterative_force_optimization(
            hybrid_centers, hybrid_radii, n, iterations=30000, growth_rate=1.00002
        )
        hybrid_sum = np.sum(hybrid_radii)
        
        if hybrid_sum > best_sum:
            best_sum = hybrid_sum
            best_centers = hybrid_centers.copy()
            best_radii = hybrid_radii.copy()
    
    # Final validation and cleanup
    if best_centers is None:
        # Absolute fallback
        best_centers, best_radii = create_hexagonal_initial(n)
        best_sum = np.sum(best_radii)
    
    best_radii = np.maximum(best_radii, 0)
    
    # Ensure centers are within bounds
    best_centers = np.clip(best_centers, best_radii.reshape(-1, 1), 1 - best_radii.reshape(-1, 1))
    
    return best_centers, best_radii, best_sum
