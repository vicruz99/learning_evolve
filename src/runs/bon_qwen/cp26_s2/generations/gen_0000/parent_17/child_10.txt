from scipy.optimize import differential_evolution, minimize, linprog
from scipy.spatial.distance import cdist

def get_optimal_radii_for_centers(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Variables: r_0, ..., r_25.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= dist(centers[i], centers[j]) for all i < j
      r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Coefficients for objective function: max sum(r_i) -> min -sum(r_i)
    c = -np.ones(n)
    
    # Bounds for r_i: [0, infinity] (upper bound handled by constraints)
    # Actually, r_i cannot exceed 0.5.
    bounds = [(0, 0.5) for _ in range(n)]

    # 1. Boundary constraints
    # r_i <= x_i  => r_i - x_i <= 0 ? No, x_i is constant.
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    
    # Construct A_ub and b_ub
    # We can add constraints row by row or build matrix.
    # Let's build lists of rows.
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)

def objective_function_for_de(x_flat):
    """
    Objective function for Differential Evolution.
    Input: flat array of 52 elements (26 centers * 2)
    Output: negative sum of radii (to minimize)
    """
    n = 26
    centers = x_flat.reshape(n, 2)
    
    # Clip centers to [0, 1] just in case, though DE handles bounds
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_radii, _ = get_optimal_radii_for_centers(centers)
    
    # We want to maximize sum_radii, so return negative
    return -sum_radii

def validate_and_refine(centers, radii):

    Takes an initial guess of centers and radii and refines it using SLSQP.
    This ensures strict constraint satisfaction.
    """

    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total 78 variables.
    # But optimizing all at once with quadratic constraints is hard.
    # Instead, let's fix the radii logic or just optimize centers again?
    # Actually, SLSQP can handle non-linear constraints.
    
    # Let's try to optimize centers and radii jointly.
    # Initial guess
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds_list = []
    for _ in range(n):
        bounds_list.append((0.0, 1.0)) # x
        bounds_list.append((0.0, 1.0)) # y
        bounds_list.append((0.0, 0.5)) # r
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    def obj(vars):
        r_vars = vars[2::3]
        return -np.sum(r_vars)
    
    # Constraints
    # 1. Boundary: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    #    Equivalent to: r <= x, r <= 1-x, r <= y, r <= 1-y
    #    r - x <= 0
    #    r - (1-x) <= 0 => r + x - 1 <= 0
    #    r - y <= 0
    #    r - (1-y) <= 0 => r + y - 1 <= 0
    
    # 2. Non-overlap: dist(i, j) >= r_i + r_j
    #    dist(i, j) - (r_i + r_j) >= 0
    
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        idx_x = i * 3
        idx_y = i * 3 + 1
        idx_r = i * 3 + 2
        
        # r <= x  => r - x <= 0
        constraints.append({
            'type': 'ineq', # - (r - x) >= 0 => x - r >= 0
            'fun': lambda vars, k=idx_x, kr=idx_r: vars[k] - vars[kr]
        })
        # r <= 1 - x => r + x - 1 <= 0 => 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda vars, k=idx_x, kr=idx_r: 1.0 - vars[k] - vars[kr]
        })
        # r <= y
        constraints.append({
            'type': 'ineq',
            'fun': lambda vars, k=idx_y, kr=idx_r: vars[k] - vars[kr]
        })
        # r <= 1 - y
        constraints.append({
            'type': 'ineq',
            'fun': lambda vars, k=idx_y, kr=idx_r: 1.0 - vars[k] - vars[kr]
        })

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = i * 3
            idx_yi = i * 3 + 1
            idx_ri = i * 3 + 2
            
            idx_xj = j * 3
            idx_yj = j * 3 + 1
            idx_rj = j * 3 + 2
            
            # dist >= ri + rj
            # sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
            
            def dist_constraint(vars, i=i, j=j):
                xi = vars[i*3]
                yi = vars[i*3+1]
                ri = vars[i*3+2]
                xj = vars[j*3]
                yj = vars[j*3+1]
                rj = vars[j*3+2]
                
                d = math.sqrt((xi-xj)**2 + (yi-yj)**2)
                return d - (ri + rj)
            
            constraints.append({
                'type': 'ineq',
                'fun': dist_constraint
            })

    # Run optimization
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds_list, constraints=constraints, 
                       options={'ftol': 1e-12, 'maxiter': 500})
        
        if res.success:
            final_vars = res.x
            final_centers = np.zeros((n, 2))
            final_radii = np.zeros(n)
            for i in range(n):
                final_centers[i] = [final_vars[i*3], final_vars[i*3+1]]
                final_radii[i] = final_vars[i*3+2]
            return final_centers, final_radii, -res.fun
        else:
            # If optimization fails, return original
            return centers, radii, np.sum(radii)
    except Exception:
        return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    import numpy as np # Import here just in case, though top level has it
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
