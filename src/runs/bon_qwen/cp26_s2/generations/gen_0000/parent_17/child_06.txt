import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, lam):
    """
    Compute the penalized objective function for circle packing.
    Minimizes -sum(radii) + lam * (boundary_penalties + overlap_penalties)
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    pen = 0.0
    
    # Boundary penalties
    pen += np.sum(np.maximum(0, radii - centers[:,0])**2)
    pen += np.sum(np.maximum(0, centers[:,0] + radii - 1)**2)
    pen += np.sum(np.maximum(0, radii - centers[:,1])**2)
    pen += np.sum(np.maximum(0, centers[:,1] + radii - 1)**2)
    
    # Overlap penalties (vectorized)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    i_idx, j_idx = np.tril_indices(n, -1)
    overlaps = radii[i_idx] + radii[j_idx] - dists[i_idx, j_idx]
    pen += np.sum(np.maximum(0, overlaps)**2)
    
    return -np.sum(radii) + lam * pen

def run_packing():
    n = 26
    lam = 5000.0
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Try multiple random seeds to avoid poor local minima
    for seed in range(15):
        rng = np.random.RandomState(seed)
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.08)
        
        # Hexagonal grid initialization
        idx = 0
        row = 0
        while idx < n:
            cols = 6 if row % 2 == 0 else 5
            if row == 4: cols = 4
            if row > 4: cols = 0
            
            y = 0.1 + row * 0.18
            for c in range(cols):
                if idx >= n: break
                x = 0.1 + c * 0.16
                if row % 2 == 1: x += 0.08
                centers[idx] = [x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)]
                idx += 1
            row += 1
            
        centers = np.clip(centers, 0.05, 0.95)
        x0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
        
        res = minimize(compute_objective, x0, args=(n, lam), bounds=bounds, method='L-BFGS-B', 
                       options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 5000})
        
        c_opt = res.x[:2*n].reshape(n, 2)
        r_opt = res.x[2*n:]
        
        # Strict validity check
        valid = True
        for i in range(n):
            if r_opt[i] < 0: valid = False; break
            if c_opt[i,0] - r_opt[i] < -1e-9 or c_opt[i,0] + r_opt[i] > 1 + 1e-9: valid = False; break
            if c_opt[i,1] - r_opt[i] < -1e-9 or c_opt[i,1] + r_opt[i] > 1 + 1e-9: valid = False; break
            
        if valid:
            for i in range(n):
                for j in range(i+1, n):
                    d = np.sqrt(np.sum((c_opt[i]-c_opt[j])**2))
                    if d < r_opt[i] + r_opt[j] - 1e-9:
                        valid = False; break
                if not valid: break
                
        if valid:
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
                
    # Safety projection to guarantee 1e-12 tolerance compliance
    if best_centers is not None:
        for i in range(n):
            best_radii[i] = min(best_radii[i], 
                                best_centers[i,0], 1-best_centers[i,0],
                                best_centers[i,1], 1-best_centers[i,1])
            
        for _ in range(5):
            changed = False
            for i in range(n):
                for j in range(i+1, n):
                    d = np.sqrt(np.sum((best_centers[i]-best_centers[j])**2))
                    if d < best_radii[i] + best_radii[j]:
                        scale = d / (best_radii[i] + best_radii[j])
                        best_radii[i] *= scale
                        best_radii[j] *= scale
                        changed = True
            if not changed: break
            
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
