import numpy as np
from scipy.optimize import minimize

def generate_initial_config(n):
    """
    Generates a random initial configuration for n circles within [0.1, 0.9].
    """
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    return centers

def overlap_penalty(centers, radii):
    """
    Calculates a penalty value for overlaps and boundary violations.
    Returns sum of squared violations.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Pairwise overlaps
    # Vectorized distance calculation
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    # dist shape: (n, n)
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # Sum of radii shape: (n, n)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violations: max(0, r_i + r_j - dist)
    violations = r_sum - dist
    violations = np.maximum(violations, 0)
    
    # Sum of squared violations for upper triangle (i < j)
    triu_indices = np.triu_indices(n, k=1)
    pair_violations = violations[triu_indices]
    penalty += np.sum(pair_violations**2)
    
    # Boundary violations
    # x >= r  =>  r - x <= 0. Violation if r - x > 0
    left_viol = radii - centers[:, 0]
    penalty += np.sum(np.maximum(left_viol, 0)**2)
    
    # x <= 1 - r  =>  x + r - 1 <= 0. Violation if x + r - 1 > 0
    right_viol = centers[:, 0] + radii - 1.0
    penalty += np.sum(np.maximum(right_viol, 0)**2)
    
    # y >= r
    bottom_viol = radii - centers[:, 1]
    penalty += np.sum(np.maximum(bottom_viol, 0)**2)
    
    # y <= 1 - r
    top_viol = centers[:, 1] + radii - 1.0
    penalty += np.sum(np.maximum(top_viol, 0)**2)
    
    return penalty

def check_feasibility(n, r, num_trials=3):
    """
    Attempts to find a valid packing of n circles with radius r.
    Returns (is_feasible, centers).
    """
    # Bounds for centers: x, y in [r, 1-r]
    bounds = [(r, 1 - r) for _ in range(n)] * 2
    
    def obj(x):
        c = x.reshape(n, 2)
        rad = np.full(n, r)
        return overlap_penalty(c, rad)

    best_val = float('inf')
    best_x = None
    
    for _ in range(num_trials):
        init_c = generate_initial_config(n)
        try:
            # Use L-BFGS-B with bounds
            res = minimize(obj, init_c.flatten(), method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000})
            if res.fun < best_val:
                best_val = res.fun
                best_x = res.x
        except Exception:
            pass
            
    # Tolerance for valid packing
    if best_val < 1e-6 and best_x is not None:
        return True, best_x.reshape(n, 2)
    else:
        return False, None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # We aim to maximize sum of radii.
    # A strong heuristic is to find the maximum radius r for equal circles.
    # Target sum 2.636 implies r approx 0.1014.

    high = 0.11 # Upper bound estimate
    best_r = 0.05
    best_centers = None

    feas, centers = check_feasibility(n, 0.10, num_trials=5)
    if feas:
        low = 0.10
        best_r = 0.10
        best_centers = centers
    else:
        # If 0.10 is not feasible (unlikely), search lower
        high = 0.10
        low = 0.05
        
    # Binary search for maximum feasible r
    for _ in range(50):
        if high - low < 1e-5:
            break
        mid = (low + high) / 2.0
        feas, centers = check_feasibility(n, mid, num_trials=3)
        if feas:
            low = mid
            best_r = mid
            best_centers = centers
        else:
            high = mid
            
    # Construct result
    radii = np.full(n, best_r)
    if best_centers is None:
        # Fallback if no valid packing found (should not happen for r=0.05)
        best_centers = generate_initial_config(n)
        # Ensure radii are compatible with centers
        # Just clip radii to fit centers?

        radii = np.full(n, 0.05)

        1 - best_centers[:, 1]

    sum_radii = np.sum(radii)
