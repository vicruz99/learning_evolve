import numpy as np
from scipy.optimize import minimize
import math


def compute_violation(centers, radii, n):
    """Compute total constraint violation for validation."""
    v = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            v += (x - r) ** 2
        if x + r > 1:
            v += (x + r - 1) ** 2
        if y - r < 0:
            v += (y - r) ** 2
        if y + r > 1:
            v += (y + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                v += overlap ** 2

    return v


def objective(params, n, penalty):
    """Objective function: negative sum of radii plus penalty for violations."""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-12)

    obj = -np.sum(radii)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            obj += penalty * (x - r) ** 2
        if x + r > 1:
            obj += penalty * (x + r - 1) ** 2
        if y - r < 0:
            obj += penalty * (y - r) ** 2
        if y + r > 1:
            obj += penalty * (y + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                obj += penalty * overlap ** 2

    return obj


def make_hexagonal_packing(n):
    """Create a hexagonal grid initial packing."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.03

    idx = 0
    row = 0

    while idx < n:
        # Vertical spacing for hexagonal packing
        y = 0.08 + row * 0.16

        remaining = n - idx
        if row % 2 == 0:
            ncols = min(6, remaining)
        else:
            ncols = min(5, remaining)

        offset = 0.08 if row % 2 == 1 else 0

        for col in range(ncols):
            x = 0.08 + col * 0.16 + offset
            centers[idx] = [x, y]
            idx += 1

        row += 1

    return centers, radii


def optimize_from_start(n, centers_init, radii_init, num_restarts=30, num_phases=10):
    """Run optimization with multiple restarts and penalty annealing."""
    best_centers = None
    best_radii = None
    best_sum = 0.0
    best_violation = float("inf")

    for restart in range(num_restarts):
        centers = centers_init.copy()
        radii = radii_init.copy()

        # Add random perturbation
        centers = centers + np.random.randn(n, 2) * 0.015
        radii = radii + np.random.randn(n) * 0.008
        radii = np.maximum(radii, 0.01)
        centers = np.clip(centers, 0.01, 0.99)

        params = np.concatenate([centers.flatten(), radii])

        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

        # Multi-phase optimization with increasing penalty
        for phase in range(num_phases):
            penalty = 200 * (phase + 1) ** 2

            try:
                result = minimize(
                    objective,
                    params,
                    args=(n, penalty),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": 2000, "ftol": 1e-15},
                )
                params = result.x.copy()
            except Exception:
                pass

        centers_opt = params[:2 * n].reshape(n, 2)
        radii_opt = np.maximum(params[2 * n:], 0)

        violation = compute_violation(centers_opt, radii_opt, n)
        current_sum = np.sum(radii_opt)

        if violation < best_violation:
            best_violation = violation
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
            best_sum = current_sum

    return best_centers, best_radii, best_sum
