

Looking at this circle packing problem, I need to strategically place 26 circles in a unit square to maximize the sum of radii.

<strategy>
My strategy involves:

1. **Hexagonal Packing Base**: Start with a hexagonal arrangement (optimal for equal circles) with rows like 5-6-5-6-4 or similar configurations. For equal circles in hexagonal packing with 5 rows, the optimal radius is r = 1/(2 + 4√3) ≈ 0.0907, giving sum ≈ 2.358.

2. **Unequal Radii Optimization**: To exceed the equal-circle limit, allow unequal radii. Circles near boundaries can be larger, and interior circles can be smaller, potentially achieving higher total sum.

3. **Multi-strategy Approach**:
   - Hexagonal grid initialization
   - Random initialization with local optimization
   - Use differential evolution for global search
   - Local perturbation refinement
   - Multiple random seeds for robustness

4. **Post-processing**: Normalize configurations to fully utilize the unit square space, maximizing radii.

5. **Validation**: Ensure all circles satisfy non-overlap and boundary constraints.

import numpy as np
from scipy.optimize import differential_evolution, minimize

def validate_packing(centers, radii):
    """Validate that circles don't overlap and are inside the unit square"""
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True

def compute_overlap_penalty(centers, radii):
    """Compute penalty for overlaps and boundary violations"""
    n = len(radii)
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            if d < r_sum:
                penalty += (r_sum - d) ** 2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        for bound in [r - x, y - r, x + r - 1, y + r - 1]:
            if bound > 0:
                penalty += bound ** 2
    return penalty

def make_initial_hexagonal(n):
    """Create initial hexagonal packing"""
    centers = np.zeros((n, 2))
    idx = 0
    
    # Try different row configurations for n=26
    configs = [
        [5, 6, 5, 6, 4],
        [6, 5, 6, 5, 4],
        [5, 5, 5, 5, 6],
        [6, 6, 6, 4, 4],
        [5, 5, 6, 5, 5],
    ]
    
    best_sum = 0
    best_c = None
    best_r = None
    
    for row_counts in configs:
        if sum(row_counts) != n:
            continue
        
        c = np.zeros((n, 2))
        idx = 0
        for ri, count in enumerate(row_counts):
            for ci in range(count):
                c[idx, 0] = ci * 2 + (ri % 2)
                c[idx, 1] = ri * np.sqrt(3)
                idx += 1
        
        # Normalize to unit square
        cx, cy = c.min(axis=0)
        cx2, cy2 = c.max(axis=0)
        if cx2 - cx > 0:
            c[:, 0] = (c[:, 0] - cx) / (cx2 - cx)
        if cy2 - cy > 0:
            c[:, 1] = (c[:, 1] - cy) / (cy2 - cy)
        
        # Compute max equal radius
        r = 0.5
        for i in range(n):
            r = min(r, c[i, 0], 1 - c[i, 0], c[i, 1], 1 - c[i, 1])
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                r = min(r, d / 2)
        
        r = max(r, 0.01)
        radii_arr = np.ones(n) * r
        
        if np.sum(radii_arr) > best_sum:
            best_sum = np.sum(radii_arr)
            best_c = c.copy()
            best_r = radii_arr.copy()
    
    return best_c, best_r

def normalize_and_compute_radii(centers):
    """Normalize centers to unit square and compute max radii"""
    n = len(centers)
    c = centers.copy()
    
    # Shift and scale to fit in [0,1]^2 with margin
    cx, cy = c.min(axis=0)
    cx2, cy2 = c.max(axis=0)
    
    margin = 0.001
    if cx2 - cx > 2 * margin:
        c[:, 0] = margin + (c[:, 0] - cx) / (cx2 - cx) * (1 - 2 * margin)
    if cy2 - cy > 2 * margin:
        c[:, 1] = margin + (c[:, 1] - cy) / (cy2 - cy) * (1 - 2 * margin)
    
    # Compute radii based on distances to boundaries and other circles
    radii = np.ones(n) * 0.5
    for i in range(n):
        radii[i] = min(c[i, 0], 1 - c[i, 0], c[i, 1], 1 - c[i, 1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                radii[i] = min(radii[i], d / 2)
    
    return c, np.maximum(radii, 0.001)

def local_optimize(centers, radii, n=26, iterations=100000):
    """Local optimization via perturbation"""
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(best_radii)
    
    for it in range(iterations):
        i = np.random.randint(n)
        step = 0.02 * np.exp(-it / 20000)
        
        # Perturb position
        dc = np.random.uniform(-step, step, 2)
        # Perturb radius
        dr = np.random.uniform(-step, step)
        
        centers_new = best_centers.copy()
        radii_new = best_radii.copy()
        
        centers_new[i, 0] = np.clip(best_centers[i, 0] + dc[0], radii_new[i], 1 - radii_new[i])
        centers_new[i, 1] = np.clip(best_centers[i, 1] + dc[1], radii_new[i], 1 - radii_new[i])
        radii_new[i] = np.clip(best_radii[i] + dr, 0.001, 0.3)
        
        # Quick validation
        valid = True
        x, y = centers_new[i]
        r = radii_new[i]
        if x < r or x > 1 - r or y < r or y > 1 - r:
            valid = False
        
        if valid:
            for j in range(n):
                if i == j:
                    continue
                d = np.sqrt(np.sum((centers_new[i] - centers_new[j]) ** 2))
                if d < radii_new[i] + radii_new[j] - 1e-10:
                    valid = False
                    break
        
        if valid:
            new_sum = np.sum(radii_new)
            if new_sum > best_sum:
                best_centers = centers_new
                best_radii = radii_new
                best_sum = new_sum
            elif np.random.random() < 0.01 * np.exp(-it / 50000):
                best_centers = centers_new
                best_radii = radii_new
                best_sum = new_sum
    
    return best_centers, best_radii, best_sum

def run_packing():
