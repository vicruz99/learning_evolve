def initialize_packing(n):
    """Initialize circles in a hexagonal packing pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.095

    idx = 0
    y = 0.095
    row = 0
    hex_spacing = np.sqrt(3) * 0.095

    while idx < n:
        x = 0.095
        offset = 0.095 * (row % 2)
        x = 0.095 + offset
        while x + 0.095 <= 1.0 + 1e-10 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * 0.095
        y += hex_spacing * 0.98
        row += 1

    return centers[:n], radii[:n]


def is_feasible(centers, radii, n):
    """Check if packing is feasible."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def get_violations(centers, radii, n):
    """Get total constraint violation magnitude."""
    violation = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            violation += (x - r) ** 2
        if x + r > 1:
            violation += (x + r - 1) ** 2
        if y - r < 0:
            violation += (y - r) ** 2
        if y + r > 1:
            violation += (y + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violation += (min_dist - dist) ** 2
    return violation

    best_radii = radii.copy()

def ensure_feasibility(centers, radii, n):
    """Ensure all constraints are satisfied by shrinking radii if needed."""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(1000):
        changed = False
        
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            max_r = min(x, 1-x, y, 1-y)
            if r > max_r + 1e-10:
                radii[i] = max(0, max_r - 1e-10)
                changed = True
        
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist - 1e-10:
                    # Shrink both circles proportionally
                    scale = (dist + 1e-10) / (min_dist + 1e-12)
                    radii[i] *= scale
                    radii[j] *= scale
                    changed = True
        
        if not changed:
            break
    
    return centers, radii
