import numpy as np
from scipy.optimize import minimize


def constr_x_min(x, i):
    return x[3 * i] - x[3 * i + 2]


def constr_x_max(x, i):
    return 1.0 - x[3 * i] - x[3 * i + 2]


def constr_y_min(x, i):
    return x[3 * i + 1] - x[3 * i + 2]


def constr_y_max(x, i):
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def constr_r_pos(x, i):
    return x[3 * i + 2]


def constr_nonoverlap(x, i, j):
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    dist_sq = dx * dx + dy * dy
    if dist_sq < 1e-30:
        return 0.0
    dist = np.sqrt(dist_sq)
    return dist - x[3 * i + 2] - x[3 * j + 2]


def objective(x, n):
    s = 0.0
    for i in range(n):
        s += x[3 * i + 2]
    return -s


def create_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constr_x_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_x_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_y_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_y_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_r_pos, 'args': (i,)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constr_nonoverlap, 'args': (i, j)})
    return constraints


def init_hex_grid(n, row_counts, spacing_x=0.18, spacing_y=0.155, offset_x=0.09, offset_y=0.07):
    centers = np.zeros((n, 2))
    idx = 0
    for row, count in enumerate(row_counts):
        y = offset_y + row * spacing_y
        for col in range(count):
            x = offset_x + col * spacing_x
            if row % 2 == 1:
                x += spacing_x * 0.5
            centers[idx] = [x, y]
            idx += 1
    return centers


def force_directed_cleanup(centers, radii, n, iterations=1500):
    for it in range(iterations):
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                target = radii[i] + radii[j] + 0.0005
                if dist < target and dist > 1e-10:
                    push = (target - dist) * 0.3
                    fx = dx / dist * push
                    fy = dy / dist * push
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
        centers += 0.5 * forces
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r + 0.001, min(1.0 - r - 0.001, centers[i, 0]))
            centers[i, 1] = max(r + 0.001, min(1.0 - r - 0.001, centers[i, 1]))
    return centers


def run_packing():
    n = 26
    constraints = create_constraints(n)

    best_sum = -1.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    configs = [
        [5, 4, 5, 4, 5, 3],
        [5, 5, 5, 5, 4, 2],
        [4, 5, 4, 5, 4, 4],
        [5, 4, 5, 5, 4, 3],
        [4, 4, 5, 5, 5, 3],
        [5, 5, 4, 5, 4, 3],
        [3, 5, 4, 5, 5, 4],
    ]

    for cfg_idx, row_counts in enumerate(configs):
        centers = init_hex_grid(n, row_counts)
        radii = np.ones(n) * 0.07
        centers = force_directed_cleanup(centers, radii, n, iterations=500)

        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3 * i] = centers[i, 0]
            x0[3 * i + 1] = centers[i, 1]
            x0[3 * i + 2] = radii[i]

        try:
            result = minimize(objective, x0, args=(n,),
                             constraints=constraints,
                             method='SLSQP',
                             options={'maxiter': 3000, 'ftol': 1e-12})

            fc = np.zeros((n, 2))
            fr = np.zeros(n)
            for i in range(n):
                fc[i, 0] = result.x[3 * i]
                fc[i, 1] = result.x[3 * i + 1]
                fr[i] = max(0.0, result.x[3 * i + 2])

            current_sum = sum(fr)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = fc.copy()
                best_radii = fr.copy()
        except Exception:
            pass

    for cfg_idx, row_counts in enumerate(configs):
        centers = init_hex_grid(n, row_counts, spacing_x=0.17, spacing_y=0.15, offset_x=0.065, offset_y=0.065)
        radii = np.ones(n) * 0.08
        centers = force_directed_cleanup(centers, radii, n, iterations=500)

        x0 = np.zeros(3 * n)

            x0[3 * i + 1] = centers[i, 1]

            pass

    # Final validity fix: shrink any circles that slightly violate constraints
    for i in range(n):
        max_r = min(best_centers[i, 0], 1.0 - best_centers[i, 0],

        for j in range(n):
            if i != j:
                dx = best_centers[i, 0] - best_centers[j, 0]
                dy = best_centers[i, 1] - best_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                max_r = min(max_r, dist - best_radii[j])
        best_radii[i] = max(0.0, max_r)

    best_sum = sum(best_radii)

    return best_centers, best_radii, best_sum
