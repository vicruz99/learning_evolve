from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Uses a numerical optimization approach starting from a hexagonal lattice configuration.

    # Helper function to compute the maximum feasible radius for a given set of centers
    def get_radius(centers):
        # Distance to boundaries
        r_bound = np.min([
            np.min(centers[:, 0]),
            np.min(1.0 - centers[:, 0]),
            np.min(centers[:, 1]),
            np.min(1.0 - centers[:, 1])
        ])
        
        # Distance between circles (divided by 2)
        # Compute pairwise distances
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        
        # Ignore diagonal (distance 0) and keep upper triangle
        np.fill_diagonal(dists, np.inf)
        r_inter = np.min(dists) / 2.0
        
        return min(r_bound, r_inter)

    def objective(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        # Penalty if centers are out of bounds to guide the optimizer
        if np.any(centers < 0) or np.any(centers > 1):
            return 10.0 # Large positive value (bad)

        # Nelder-Mead doesn't use gradients, but min is non-smooth.
        # We just return -radius.
        r = get_radius(centers)
        return -r

    best_r = 0.0
    n_circles = 26

    n_restarts = 10

    return best_centers, radii, sum_radii
