import numpy as np
from scipy.optimize import minimize
import math

def get_radii(centers):
    """
    Calculates the maximum valid radius for each circle given their centers.
    For each circle, the radius is limited by the unit square boundaries
    and the distance to other circles.
    """
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Initialize radii based on distance to boundaries
    for i in range(n):
        x, y = centers[i]
        r_x = min(x, 1 - x)
        r_y = min(y, 1 - y)
        radii[i] = min(r_x, r_y)
        
    # Iterate to constrain radii based on distances to other circles
    # Since radii depend on each other (r_i + r_j <= dist), we can solve this 
    # by noting that for a fixed configuration of centers, the optimal radii
    # are determined by the Voronoi-like constraints.
    # A simple valid assignment that maximizes sum is often tricky, but 
    # a conservative valid assignment is r_i = min(dist_to_boundary, min_{j!=i}(dist_ij/2)).
    # However, to strictly satisfy r_i + r_j <= dist_ij, the formula r_i = min(...) works 
    # if we consider the "bottleneck".
    # Actually, if we define r_i = min(d_i_to_boundary, min_{j!=i} (dist_ij / 2)),
    # then for any pair i, j:
    # r_i <= dist_ij / 2 and r_j <= dist_ij / 2
    # => r_i + r_j <= dist_ij. This satisfies the non-overlap constraint.
    # Is this optimal for sum? It's a valid lower bound.
    # Often, one circle might be constrained by boundary, another by neighbor.
    # This heuristic is robust.
    
    for i in range(n):
        min_dist_to_other = np.inf
        for j in range(n):
            if i == j:
                continue
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < min_dist_to_other:
                min_dist_to_other = dist
        
        r_boundary = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = min(r_boundary, min_dist_to_other / 2)
        
    return radii

    n = 26
    centers = centers_flat.reshape((n, 2))
    radii = get_radii(centers)
    return -np.sum(radii)
