from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """

    # A hexagonal lattice allows denser packing than a square grid.
    # We start with a safe radius to ensure validity.
    r_init = 0.04
    centers = np.zeros((n_circles, 2))
    radii = np.full(n_circles, r_init)

    # Vertical distance between rows: sqrt(3)/2 * diameter = sqrt(3) * r

    row_idx = 0

        y = r_init + row_idx * np.sqrt(3) * r_init
        
        # Alternate between standard and shifted rows
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = r_init + r_init # Shift by diameter/2 (which is r)
            
        x = x_start
        while x <= 1 - r_init and idx < n_circles:
            if idx < n_circles:
                centers[idx] = [x, y]
                idx += 1
            x += 2 * r_init
        
        row_idx += 1

    # We iteratively grow the radius and re-optimize positions to find the
