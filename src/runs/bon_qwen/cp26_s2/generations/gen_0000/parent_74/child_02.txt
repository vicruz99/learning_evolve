import numpy as np
from scipy.optimize import minimize

def generate_initial_centers(N):
    """Generates a valid staggered hexagonal grid layout."""
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    while idx < N:
        y = row * 0.20
        x_start = 0.10 if row % 2 == 1 else 0.0
        col = 0
        while idx < N:
            x = x_start + col * 0.20
            if x < 1.0:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1
    return centers

def packing_objective(vars, N):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars[2*N:])

def packing_constraints(vars, N):
    """
    Returns an array of constraint values. All must be >= 0.
    Includes pairwise non-overlap and boundary containment.
    """
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    
    # Vectorized pairwise distance constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    c_diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    d2 = np.sum(c_diff**2, axis=2)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    diff = d2 - r_sum**2
    
    # Extract upper triangle to avoid duplicates and self-constraints
    i, j = np.triu_indices(N, k=1)
    cons = diff[i, j]
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    cons = np.concatenate([
        cons, 
        c[:, 0] - r, 
        1.0 - c[:, 0] - r, 
        c[:, 1] - r, 
        1.0 - c[:, 1] - r
    ])
    return cons

def run_packing():
    np.random.seed(42)
    N = 26
    
    # Start with a valid configuration to ensure solver stability
    centers = generate_initial_centers(N)
    radii = np.full(N, 0.09)
    x0 = np.hstack([centers.ravel(), radii])
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    def objective_wrapper(v):
        return packing_objective(v, N)
        
    def constraint_wrapper(v):
        return packing_constraints(v, N)
        
    cons = {'type': 'ineq', 'fun': constraint_wrapper}
    
    best_vars = x0.copy()
    best_sum = np.sum(radii)
    
    # Iterative optimization with perturbations to escape local minima
    for _ in range(3):
        res = minimize(
            objective_wrapper, 
            x0, 
            method='SLSQP', 
            bounds=bounds,
            constraints=cons, 
            options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
        )
        
        current_sum = np.sum(res.x[2*N:])
        if current_sum > best_sum:
            best_sum = current_sum
            best_vars = res.x

        x0 = best_vars + np.random.normal(0, 0.002, size=best_vars.shape)
        x0 = np.clip(x0, 0, 1)
        x0[-N:] = np.clip(x0[-N:], 0, 0.5)
        
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    
    return centers, radii, np.sum(radii)
