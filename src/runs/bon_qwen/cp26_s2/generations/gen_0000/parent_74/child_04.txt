import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def objective_func(x):
    """Objective: maximize sum of radii (minimize negative sum)"""
    radii = x[2*N:3*N]
    return -np.sum(radii)

def constraint_func(x):
    """All constraints as a vector (each must be >= 0)"""
    centers = x[:2*N].reshape(N, 2)
    radii = x[2*N:3*N]
    
    c = []
    
    # Boundary constraints: circle must be inside unit square
    for i in range(N):
        c.append(centers[i, 0] - radii[i])        # x - r >= 0
        c.append(1 - centers[i, 0] - radii[i])    # x + r <= 1
        c.append(centers[i, 1] - radii[i])        # y - r >= 0
        c.append(1 - centers[i, 1] - radii[i])    # y + r <= 1
        c.append(radii[i])                        # r >= 0
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    for i in range(N):
        for j in range(i+1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            c.append(dist - radii[i] - radii[j])
    
    return np.array(c)

def make_hexagonal_packing(offset, spacing):
    """Create hexagonal grid packing"""
    centers = np.zeros((N, 2))
    idx = 0
    rows = [5, 6, 5, 6, 4]  # 26 circles total
    
    for row_idx, row_count in enumerate(rows):
        y = offset + row_idx * spacing
        for col in range(row_count):
            if idx >= N:
                break
            x = offset + col * spacing
            if row_idx % 2 == 1:
                x += spacing / 2
            centers[idx] = [x, y]
            idx += 1
    
    return centers

def make_grid_packing(offset, spacing_x, spacing_y):
    """Create grid packing"""
    centers = np.zeros((N, 2))
    idx = 0
    
    for row in range(6):
        for col in range(5):
            if idx >= N:
                break
            x = offset + col * spacing_x
            y = offset + row * spacing_y
            centers[idx] = [x, y]
            idx += 1
    
    return centers

def make_staggered_packing(offset, spacing):
    """Create staggered packing"""
    centers = np.zeros((N, 2))
    idx = 0
    rows = [5, 5, 5, 5, 4, 2]
    
    for row_idx, row_count in enumerate(rows):
        y = offset + row_idx * spacing

            if idx >= N:
                break
            x = offset + col * spacing
            if row_idx % 2 == 1:
                x += spacing / 2
            centers[idx] = [x, y]
            idx += 1
    
    return centers

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii"""
    # Variables: r_1, ..., r_N
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = -np.ones(N)
    
    # Constraints: A_ub @ r <= b_ub
    # 1) r_i + r_j <= dist(i,j) for all pairs
    # 2) r_i <= boundary_i for all i
    # 3) r_i >= 0 (handled by bounds)
    
    A_rows = []
    b_rows = []
    
    # Pairwise constraints
    for i in range(N):
        for j in range(i+1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            
            row = np.zeros(N)
            row[i] = 1
            row[j] = 1
            A_rows.append(row)
            b_rows.append(dist)
    
    # Boundary constraints
    for i in range(N):
        row = np.zeros(N)
        row[i] = 1
        A_rows.append(row)
        b_rows.append(min(centers[i, 0], 1-centers[i, 0],
                         centers[i, 1], 1-centers[i, 1]))
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bounds: r_i >= 0
    bounds = [(0, None)] * N
    
    # Solve LP
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.ones(N) * 0.01

def compute_min_slack(centers, radii):
    """Compute minimum slack over all constraints"""
    min_s = float('inf')
    
    for i in range(N):
        s = min(centers[i, 0] - radii[i],
                1 - centers[i, 0] - radii[i],
                centers[i, 1] - radii[i],
                1 - centers[i, 1] - radii[i])
        min_s = min(min_s, s)
    
    for i in range(N):
        for j in range(i+1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            s = dist - radii[i] - radii[j]
            min_s = min(min_s, s)
    
    return min_s

def optimize_centers_given_radii(centers_init, radii):
    """Optimize centers given fixed radii to maximize minimum slack"""
    
    def min_slack_neg(x):
        c = x.reshape(N, 2)
        ms = compute_min_slack(c, radii)
        return -ms
    
    x0 = centers_init.flatten()
    bounds = [(0, 1)] * (2*N)
    
    # Use Nelder-Mead for robustness
    result = minimize(min_slack_neg, x0, method='Nelder-Mead',
                     options={'maxiter': 50000, 'xatol': 1e-12, 'fatol': 1e-12})
    
    return result.x.reshape(N, 2)
