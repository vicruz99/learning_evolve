import numpy as np

def compute_repulsion_forces(centers, radii, stiffness):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            dist = np.sqrt(dist_sq)
            if dist < 1e-10:
                dist = 1e-10
                dx = 1e-10
                dy = 0.0
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                force_mag = stiffness * overlap / dist
                forces[i, 0] += force_mag * dx
                forces[i, 1] += force_mag * dy
                forces[j, 0] -= force_mag * dx
                forces[j, 1] -= force_mag * dy
    return forces

def compute_boundary_forces(centers, radii, stiffness):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    for i in range(n):
        r = radii[i]
        cx, cy = centers[i]
        if cx < r:
            forces[i, 0] += stiffness * (r - cx)
        if cx > 1.0 - r:
            forces[i, 0] -= stiffness * (cx - (1.0 - r))
        if cy < r:
            forces[i, 1] += stiffness * (r - cy)
        if cy > 1.0 - r:
            forces[i, 1] -= stiffness * (cy - (1.0 - r))
    return forces

    np.random.seed(42)

            if row % 2 == 1:
                x += spacing / 2
            # Add small random perturbation to break symmetry
            centers[idx] = [x + np.random.uniform(-0.02, 0.02), 
                            y + np.random.uniform(-0.02, 0.02)]
            idx += 1
    while idx < n:
        centers[idx] = [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]
        idx += 1
        
    radii = np.full(n, 0.02)
    
    learning_rate = 0.06
    radii_growth = 1.5e-5
    stiffness = 120.0
    max_steps = 80000
    
    for step in range(max_steps):
        forces = compute_repulsion_forces(centers, radii, stiffness)
        forces += compute_boundary_forces(centers, radii, stiffness)
        
        centers += learning_rate * forces
        # Enforce boundary constraints strictly
        centers = np.clip(centers, radii[:, None], 1.0 - radii[:, None])
        
        # Grow radii
        radii += radii_growth
        
        # Cooling schedule
        if step % 3000 == 0:
            learning_rate *= 0.97
            radii_growth *= 0.98
            
    # Final adjustment to guarantee strict validity
    for _ in range(500):
        max_overlap = 0.0
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                dist = np.sqrt(dx*dx + dy*dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
            r = radii[i]
            cx, cy = centers[i]
            b_overlap = max(max(0, r - cx), max(0, cx + r - 1), max(0, r - cy), max(0, cy + r - 1))
            if b_overlap > max_overlap:
                max_overlap = b_overlap
                
        if max_overlap <= 1e-12:
            break
        shrink_amount = max_overlap / 2.0 + 1e-7
        radii -= shrink_amount
        radii = np.maximum(radii, 1e-6)
        
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
