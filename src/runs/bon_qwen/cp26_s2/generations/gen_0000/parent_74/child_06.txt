import numpy as np
from scipy.optimize import minimize


def compute_violation(params, n):
    """Compute constraint violation for a given parameter vector."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    violation = 0.0

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            violation += (x - r) ** 2
        if x + r > 1:
            violation += (x + r - 1) ** 2
        if y - r < 0:
            violation += (y - r) ** 2
        if y + r > 1:
            violation += (y + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violation += (dist - min_dist) ** 2

    for i in range(n):
        if radii[i] < 0:
            violation += radii[i] ** 2

    return violation


def objective_penalty(params, n, weight):
    """Objective: maximize sum of radii with penalty for violations."""
    radii = params[2 * n:]
    violation = compute_violation(params, n)
    return -np.sum(radii) + weight * violation


def physics_optimize(centers, radii, n, steps=30000):
    """Physics-based optimization using repulsive forces."""
    centers = centers.copy()
    radii = radii.copy()
    lr = 0.005
    temperature = 0.02

    for step in range(steps):
        # Gradually increase radii
        growth_rate = 5e-7 * (1 + step / 1000)
        for i in range(n):
            radii[i] += growth_rate

        for i in range(n):
            fx, fy = 0.0, 0.0
            x, y = centers[i]
            r = radii[i]

            # Boundary forces
            margin = r + 1e-4
            if x < margin:
                fx += 50 * (margin - x)
            if x > 1 - margin:
                fx -= 50 * (x - (1 - margin))
            if y < margin:
                fy += 50 * (margin - y)
            if y > 1 - margin:
                fy -= 50 * (y - (1 - margin))

            # Repulsive forces from other circles
            for j in range(n):
                if i == j:
                    continue
                dx = x - centers[j, 0]
                dy = y - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq + 1e-12)
                desired = radii[i] + radii[j]

                if dist < desired:
                    strength = 500 * (desired - dist) / (dist + 1e-10)
                    fx += strength * dx
                    fy += strength * dy
                elif dist < desired * 1.3:
                    strength = 20 * (desired - dist) / (dist + 1e-10)
                    fx += strength * dx
                    fy += strength * dy

            # Temperature-based noise
            fx += np.random.randn() * temperature
            fy += np.random.randn() * temperature

            # Update position
            new_x = np.clip(x + fx * lr, 1e-4, 1 - 1e-4)
            new_y = np.clip(y + fy * lr, 1e-4, 1 - 1e-4)
            centers[i, 0] = new_x
            centers[i, 1] = new_y

        temperature *= 0.9998
        lr *= 0.9997

    return centers, radii


def hexagonal_initial(n):
    """Create hexagonal lattice initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.095

    idx = 0
    for row in range(7):
        for col in range(5):
            if idx >= n:
                break
            x = 0.1 + col * 0.19 + (row % 2) * 0.095
            y = 0.1 + row * 0.16
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def corner_initial(n):
    """Create corner-dominant initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05

    # 4 corner circles
    corners = [(0.14, 0.14), (0.86, 0.14), (0.14, 0.86), (0.86, 0.86)]
    for i, (cx, cy) in enumerate(corners):
        centers[i] = [cx, cy]
        radii[i] = 0.14

    # Remaining circles in grid
    idx = 4
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            x = 0.22 + col * 0.13
            y = 0.22 + row * 0.12
            centers[idx] = [x, y]
            radii[idx] = 0.06
            idx += 1

    return centers, radii


def random_initial(n):
    """Create random initial configuration."""
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.random.uniform(0.03, 0.1, n)
    return centers, radii


def gradient_refine(params0, n, weight=1e10, max_iter=8000):
    """Refine using L-BFGS-B gradient optimization."""
    bounds = [(0.005, 0.995)] * (2 * n) + [(0.002, 0.3)] * n

    def obj(p):
        return objective_penalty(p, n, weight)

    try:
        result = minimize(
            obj, params0, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        return result.x
    except Exception:
        return params0


def validate_fast(centers, radii, n):
    """Fast validation of packing."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
        if r < 0:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if d < radii[i] + radii[j] - 1e-10:
                return False
    return True


def uniform_grow(centers, radii, n):
    """Try to uniformly grow all circles."""
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(5000):
        # Find minimum slack
        min_slack = 1.0

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            min_slack = min(min_slack, x - r, 1 - x - r, y - r, 1 - y - r)

        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                slack = d - radii[i] - radii[j]
                min_slack = min(min_slack, slack)

        if min_slack > 1e-8:
            for i in range(n):
                radii[i] += min_slack * 0.5

    return centers, radii


def local_optimize(centers, radii, n):
    """Try local optimizations by moving individual circles."""
    centers = centers.copy()
    radii = radii.copy()
    improved = True

    while improved:
        improved = False
        for i in range(n):
            best_pos = centers[i].copy()
            best_r = radii[i]

            # Try moving in 8 directions + stay
            for dx in [-0.005, 0, 0.005]:
                for dy in [-0.005, 0, 0.005]:
                    test_centers = centers.copy()
                    test_radii = radii.copy()
                    test_centers[i] = np.clip(
                        centers[i] + np.array([dx, dy]), 0.01, 0.99
                    )

                    if validate_fast(test_centers, test_radii, n):
                        # Try growing this circle
                        min_sep = 1.0
                        for j in range(n):
                            if i == j:
                                continue
                            d = np.sqrt(np.sum((test_centers[i] - test_centers[j]) ** 2))
                            min_sep = min(min_sep, d - test_radii[j])
                        min_sep = min(min_sep, test_centers[i, 0], 1 - test_centers[i, 0],
                                     test_centers[i, 1], 1 - test_centers[i, 1])
                        new_r = max(test_radii[i], min_sep)

                        if new_r > best_r + 1e-8:
                            best_r = new_r
                            best_pos = test_centers[i].copy()

            if best_r > radii[i] + 1e-8:
                centers[i] = best_pos
                radii[i] = best_r
                improved = True

    return centers, radii
