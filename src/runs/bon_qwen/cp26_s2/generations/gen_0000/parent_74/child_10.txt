import numpy as np
from scipy.optimize import minimize

N = 26
DIM = 3 * N

def objective(v):
    return -np.sum(v[2::3])

def objective_jac(v):
    jac = np.zeros(DIM)
    jac[2::3] = -1.0
    return jac

def constraints(v):
    m = 4*N + N*(N-1)//2
    res = np.empty(m)
    idx = 0
    for i in range(N):
        x, y, r = v[3*i], v[3*i+1], v[3*i+2]
        res[idx] = x - r
        res[idx+1] = 1.0 - x - r
        res[idx+2] = y - r
        res[idx+3] = 1.0 - y - r
        idx += 4
    for i in range(N):
        for j in range(i+1, N):
            x1, y1, r1 = v[3*i], v[3*i+1], v[3*i+2]
            x2, y2, r2 = v[3*j], v[3*j+1], v[3*j+2]
            dx, dy = x1 - x2, y1 - y2
            res[idx] = dx*dx + dy*dy - (r1+r2)**2
            idx += 1
    return res

def constraints_jac(v):
    m = 4*N + N*(N-1)//2
    jac = np.zeros((m, DIM))
    idx = 0
    for i in range(N):
        b = 3*i
        jac[idx, b] = 1.0; jac[idx+1, b] = -1.0
        jac[idx, b+2] = -1.0; jac[idx+1, b+2] = -1.0
        jac[idx+2, b+1] = 1.0; jac[idx+3, b+1] = -1.0
        jac[idx+2, b+2] = -1.0; jac[idx+3, b+2] = -1.0
        idx += 4
    for i in range(N):
        for j in range(i+1, N):
            bi, bj = 3*i, 3*j
            x1, y1, r1 = v[bi], v[bi+1], v[bi+2]
            x2, y2, r2 = v[bj], v[bj+1], v[bj+2]
            dx, dy = x1 - x2, y1 - y2
            sr = r1 + r2
            jac[idx, bi] = 2.0*dx; jac[idx, bi+1] = 2.0*dy; jac[idx, bi+2] = -2.0*sr
            jac[idx, bj] = -2.0*dx; jac[idx, bj+1] = -2.0*dy; jac[idx, bj+2] = -2.0*sr
            idx += 1
    return jac

def run_packing():
    best_v = None
    best_sum = -1.0
    
    inits = []
    
    # 1. Hexagonal lattice initialization
    pts = []
    r0 = 0.08
    for row in range(7):
        for col in range(7):
            if len(pts) >= N: break
            x = (col + (row % 2) * 0.5) * 2 * r0 * 1.05 + r0
            y = row * np.sqrt(3) * r0 * 1.05 + r0
            if x <= 1.0 - r0 and y <= 1.0 - r0:
                pts.append([x, y])
        if len(pts) >= N: break
    while len(pts) < N: pts.append([0.5, 0.5])
    inits.append(np.array(pts[:N]))
    
    # 2. Grid initialization
    pts = []
    for i in range(6):
        for j in range(6):
            if len(pts) >= N: break
            pts.append([0.1 + i*0.18, 0.1 + j*0.18])
    inits.append(np.array(pts[:N]))
    
    # 3. Random feasible initializations
    rng = np.random.RandomState(42)
    for _ in range(3):
        pts = []
        for _ in range(N):
            while True:
                p = rng.uniform(0.1, 0.9, 2)
                if all(np.hypot(p[0]-pp[0], p[1]-pp[1]) > 0.25 for pp in pts):
                    pts.append(p)
                    break
        inits.append(np.array(pts[:N]))
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints, 'jac': constraints_jac}
    
    for centers_init in inits:
        v0 = np.zeros(DIM)
        for i in range(N):
            v0[3*i] = centers_init[i, 0]
            v0[3*i+1] = centers_init[i, 1]
            v0[3*i+2] = 0.08
            
        res = minimize(objective, v0, method='SLSQP', jac=objective_jac, 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
        
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_v = res.x
            
    # Refinement step to escape local optima
    if best_v is not None:
        perturbed = best_v.copy()
        rng_pert = np.random.RandomState(999)
        perturbed += rng_pert.uniform(-0.005, 0.005, DIM)
        for i in range(0, DIM, 3):
            perturbed[i] = np.clip(perturbed[i], 0.0, 1.0)
            perturbed[i+1] = np.clip(perturbed[i+1], 0.0, 1.0)
            perturbed[i+2] = np.clip(perturbed[i+2], 0.0, 0.5)
            
        res2 = minimize(objective, perturbed, method='SLSQP', jac=objective_jac, 
                        bounds=bounds, constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
        if -res2.fun > best_sum:
            best_v = res2.x
            best_sum = -res2.fun

    final_centers = np.zeros((N, 2))
    final_radii = np.zeros(N)
    for i in range(N):
        final_centers[i, 0] = best_v[3*i]
        final_centers[i, 1] = best_v[3*i+1]
        final_radii[i] = best_v[3*i+2]
        
    final_radii = np.maximum(final_radii, 0.0)
        
    return final_centers, final_radii, np.sum(final_radii)
