import numpy as np
from scipy.optimize import minimize
import math

def compute_penalty_and_grad(centers, radii):
    """Compute squared overlap penalty and its gradient w.r.t centers."""
    n = len(radii)
    penalty = 0.0
    grad = np.zeros_like(centers)
    
    # Boundary penalties
    for i in range(n):
        r = radii[i]
        for axis in range(2):
            # Lower bound
            ov = r - centers[i, axis]
            if ov > 0:
                penalty += ov * ov
                grad[i, axis] -= 2.0 * ov
            # Upper bound
            ov = r - (1.0 - centers[i, axis])
            if ov > 0:
                penalty += ov * ov
                grad[i, axis] += 2.0 * ov
                
    # Pairwise penalties
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = diff[0]*diff[0] + diff[1]*diff[1]
            dist = math.sqrt(dist_sq)
            if dist < 1e-12:
                dist = 1e-12
                diff = np.array([0.001, 0.001])
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            if overlap > 0:
                penalty += overlap * overlap
                factor = -2.0 * overlap / dist
                grad[i] += factor * diff
                grad[j] -= factor * diff
    return penalty, grad

def relax_positions(centers, radii, steps=80, lr=0.015):
    """Force-directed relaxation to resolve overlaps quickly."""
    n = len(radii)
    for _ in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            r = radii[i]
            for axis in range(2):
                ov = r - centers[i, axis]
                if ov > 0:
                    forces[i, axis] -= ov * 15.0
                ov = r - (1.0 - centers[i, axis])
                if ov > 0:
                    forces[i, axis] += ov * 15.0
                    
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = math.sqrt(diff[0]*diff[0] + diff[1]*diff[1])
                if dist < 1e-12:
                    dist = 1e-12
                    diff = np.array([0.001, 0.001])
                min_d = radii[i] + radii[j]
                if dist < min_d:
                    ov = min_d - dist
                    f = ov * 15.0 / dist
                    forces[i] += f * diff
                    forces[j] -= f * diff
                    
        centers += forces * lr
        centers = np.clip(centers, 0.0, 1.0)
    return centers

def _obj_fun(x, radii):
    c = x.reshape(-1, 2)
    p, _ = compute_penalty_and_grad(c, radii)
    return p

def _jac_fun(x, radii):
    c = x.reshape(-1, 2)
    _, g = compute_penalty_and_grad(c, radii)
    return g.flatten()

def optimize_lbfgs(centers, radii):
    """High-precision optimization using L-BFGS-B."""
    x0 = centers.flatten()
    bounds = [(0.0, 1.0)] * (2 * len(radii))
    res = minimize(_obj_fun, x0, args=(radii,), jac=_jac_fun, method='L-BFGS-B', 
                   bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-16})
    return res.x.reshape(-1, 2), res.fun

def run_packing():
    np.random.seed(2024)
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.ones(n) * 0.01
    
    for restart in range(10):

        if restart < 5:

            y = 0.07

                row += 1

        best_radii *= 0.99995
