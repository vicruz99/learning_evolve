import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution

def get_optimal_radii_and_sum(centers_flat):
    """
    For a fixed set of centers, solve the LP to maximize sum of radii.
    centers_flat: np.array of shape (52,) containing x1, y1, x2, y2, ...
    Returns: (radii_array, sum_radii)
    """
    n = 26
    centers = centers_flat.reshape(n, 2)
    
    # Variables: r_1, ..., r_26
    c_obj = -np.ones(n) # We maximize sum(r) by minimizing -sum(r)
    
    # Constraints: A_ub @ r <= b_ub
    
    # 1. Boundary Constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # This creates 4*n constraints.
    # r_i - x_i <= 0
    # r_i + x_i <= 1
    # r_i - y_i <= 0
    # r_i + y_i <= 1
    
    A_ub_rows = []
    b_ub_vals = []
    
    for i in range(n):
        xi, yi = centers[i]
        
        # r_i <= xi  =>  r_i <= xi
        row1 = np.zeros(n)
        row1[i] = 1.0
        A_ub_rows.append(row1)
        b_ub_vals.append(xi)
        
        # r_i <= 1 - xi  =>  r_i + xi <= 1  =>  r_i <= 1 - xi
        row2 = np.zeros(n)
        row2[i] = 1.0
        A_ub_rows.append(row2)
        b_ub_vals.append(1.0 - xi)
        
        # r_i <= yi
        row3 = np.zeros(n)
        row3[i] = 1.0
        A_ub_rows.append(row3)
        b_ub_vals.append(yi)
        
        # r_i <= 1 - yi
        row4 = np.zeros(n)
        row4[i] = 1.0
        A_ub_rows.append(row4)
        b_ub_vals.append(1.0 - yi)
        
    # 2. Overlap Constraints: r_i + r_j <= dist_ij
    # Number of pairs: n*(n-1)/2 = 325
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(dist)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            return radii, -res.fun
        else:
            # If LP fails (should not happen with valid centers), return zeros
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def objective_function(centers_flat):
    """
    Objective function for optimization: returns negative sum of radii.
    """
    _, sum_radii = get_optimal_radii_and_sum(centers_flat)
    return -sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Runs the packing optimization for 26 circles.
    Returns: (centers, radii, sum_radii)
    """
    n_circles = 26
    
    # Strategy: Use Differential Evolution for global search on centers,
    # then refine with local optimization.
    
    # 1. Global Optimization with Differential Evolution
    # Bounds for centers: [0, 1] for each coordinate
    bounds_centers = [(0, 1) for _ in range(n_circles * 2)]
    
    # Seed for reproducibility and robustness
    seed_val = 42 
    
    print("Starting global optimization (Differential Evolution)...")
    # popsize and maxiter tuned for performance vs quality
    # 100 function evaluations might be low, but DE is fast. 
    # We allow more iterations.
    global_res = differential_evolution(
        objective_function, 
        bounds_centers, 
        seed=seed_val,
        popsize=15,
        maxiter=50,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        polish=False, # We will polish manually
        disp=True
    )
    
    best_centers_flat = global_res.x
    
    print("Global optimization finished.")
    
    # 2. Local Refinement with SLSQP (if needed, though DE usually gets close)
    # DE is derivative-free and might have some residual error.
    # However, objective_function calls linprog which is non-smooth (kinks).
    # A standard gradient method might struggle, but L-BFGS-B or SLSQP with finite diff works.
    # Given the non-smoothness, we might just rely on DE result or run a few more steps.
    # Let's try a quick local optimization using Nelder-Mead or Powell which handle non-smoothness better.
    
    print("Refining centers...")
    try:
        local_res = minimize(
            objective_function,
            best_centers_flat,
            method='Nelder-Mead',
            options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 200}
        )
        if local_res.success:
            best_centers_flat = local_res.x
    except Exception:
        pass

    # 3. Final computation of radii
    centers_final_flat = best_centers_flat
    radii_final, sum_radii_final = get_optimal_radii_and_sum(centers_final_flat)
    
    # Reshape centers
    centers_final = centers_final_flat.reshape(n_circles, 2)
    
    # Ensure constraints are met (numerical safety)
    # Although LP guarantees r_i <= bounds, floating point might cause tiny violations.
    # We clamp radii to be strictly within bounds relative to centers.
    # But since we used LP, r_i <= x_i is satisfied.
    # We just need to ensure r_i >= 0.
    radii_final = np.maximum(radii_final, 0.0)
    
    # Re-calculate sum just in case
    sum_radii_final = np.sum(radii_final)
    
    # Validation check (optional but good practice)
    # We can trust the LP, but let's verify structure.
    
    return centers_final, radii_final, sum_radii_final
