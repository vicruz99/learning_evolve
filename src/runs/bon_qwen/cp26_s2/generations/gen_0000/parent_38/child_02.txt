def calculate_overlap_penalty(centers, radii):
    """
    Calculates a penalty if circles overlap or are out of bounds.
    Returns 0 if valid, positive value otherwise.
    """
    n = len(centers)
    penalty = 0.0
    
    # Boundary constraints
    # Circle i must be inside [0,1]x[0,1]
    # x - r >= 0, x + r <= 1  =>  r <= x <= 1-r
    # Equivalent to: r - x <= 0 and x + r - 1 <= 0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            penalty += (r - x)**2
        if x > 1 - r:
            penalty += (x - (1 - r))**2
        if y < r:
            penalty += (r - y)**2
        if y > 1 - r:
            penalty += (y - (1 - r))**2
            
    # Non-overlap constraints
    # dist(i, j) >= r_i + r_j
    for i in range(n):

            dist_sq = np.sum((centers[i] - centers[j])**2)
            dist = np.sqrt(dist_sq)
            sum_r = radii[i] + radii[j]
            if dist < sum_r:
                penalty += (sum_r - dist)**2
                
    return penalty
