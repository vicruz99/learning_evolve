def solve_radii_lp(centers, n):

    Solves the Linear Program to maximize sum of radii for fixed centers.
    Variables: r_0, ..., r_{n-1}

    2. 2*r_i <= x_i          (left wall)
    3. 2*r_i <= 1 - x_i      (right wall)
    4. 2*r_i <= y_i          (bottom wall)
    5. 2*r_i <= 1 - y_i      (top wall)
    6. r_i >= 0
    """
    # Precompute distances and boundary distances
    dist_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):

            dist_matrix[i, j] = d
            dist_matrix[j, i] = d # Symmetric, though we only use i<j
