from scipy.optimize import minimize

def run_packing():
    n = 26

    def objective(x):
        # x contains [x1, y1, r1, x2, y2, r2, ...]
        radii = x[2::3]
        return -np.sum(radii)

    def boundary_constraints(x):
        cons = []
        for i in range(n):
            idx = i * 3
            xi, yi, ri = x[idx], x[idx+1], x[idx+2]
            # x - r >= 0
            cons.append(xi - ri)
            # 1 - x - r >= 0
            cons.append(1.0 - xi - ri)
            # y - r >= 0
            cons.append(yi - ri)
            # 1 - y - r >= 0
            cons.append(1.0 - yi - ri)
            # r >= 0
            cons.append(ri)
        return cons

    def overlap_constraints(x):
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                idx_i, idx_j = i * 3, j * 3
                xi, yi, ri = x[idx_i], x[idx_i+1], x[idx_i+2]
                xj, yj, rj = x[idx_j], x[idx_j+1], x[idx_j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx**2 + dy**2
                sum_r = ri + rj
                
                # dist^2 >= (r_i + r_j)^2
                cons.append(dist_sq - sum_r**2)
        return cons

    # Combine constraints
    def all_constraints(x):
        return boundary_constraints(x) + overlap_constraints(x)

    def generate_grid_init():
        x_init = np.zeros(n * 3)
        # 5x5 grid plus 1 in center
        # Actually 5x5 is 25. Let's do a dense packing.
        # Try a slightly compressed grid
        count = 0
        # 5 rows, 5 or 6 cols
        # Let's try to fit 26 in a rectangular-ish shape
        # 6x5 grid would be 30, too many.
        # 5 rows of 5 = 25. Add one in center? Center is occupied.
        # Let's try rows: 6, 5, 5, 5, 5 -> 26
        # Width for 6: 1/6 ~ 0.166 spacing.
        # Width for 5: 1/5 = 0.2 spacing.
        
        # Better: Hexagonal-like

        # Row 1: 5 circles (offset)

        # Row 3: 5 circles

        rows = [6, 5, 6, 5, 4]

        y_step = 0.8 / 4 # 5 rows, spacing

        r_est = 0.08
        
        count = 0
        for r_idx, num_circles in enumerate(rows):
            y = y_start + r_idx * y_step
            # x spacing
            x_start = 0.1
            x_end = 0.9
            x_step = (x_end - x_start) / (num_circles - 1) if num_circles > 1 else 0
            
            for c in range(num_circles):
                x = x_start + c * x_step
                # Offset odd rows
                if r_idx % 2 == 1:
                    x += x_step / 2
                
                if count < n:
                    idx = count * 3
                    x_init[idx] = x
                    x_init[idx+1] = y
                    x_init[idx+2] = r_est

            x_init[idx] = np.random.rand()
            x_init[idx+1] = np.random.rand()

            count += 1
        return x_init

    def generate_random_init():
        x_init = np.random.rand(n * 3)
        # Set radii small to start
        x_init[2::3] = np.random.rand(n) * 0.05
        # Ensure inside bounds
        for i in range(n):

            x, y, r = x_init[idx], x_init[idx+1], x_init[idx+2]

            x_init[idx+2] = max(r, 0.001)
            x_init[idx] = max(r, min(1-r, x))
            x_init[idx+1] = max(r, min(1-r, y))
        return x_init

    def optimize(start_x):
        # Bounds: x, y in [0,1], r in [0, 0.5]
        bounds = [(0, 1)] * (n * 3)

        cons = []

    return centers, radii, best_sum
