import numpy as np
from scipy.optimize import minimize

def objective(x):
    # x is [x0, y0, r0, x1, y1, r1, ...]
    # We want to maximize sum(r), so minimize -sum(r)
    radii = x[2::3]
    return -np.sum(radii)

def constraint_overlap(x):
    n = 26
    # Reshape to (n, 3)
    points = x.reshape(-1, 3)
    centers = points[:, :2]
    radii = points[:, 2]
    
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            # dist >= r_i + r_j  =>  dist - r_i - r_j >= 0
            constraints.append(dist - radii[i] - radii[j])
    return np.array(constraints)

def constraint_boundary(x):
    n = 26
    points = x.reshape(-1, 3)
    centers = points[:, :2]
    radii = points[:, 2]
    
    constraints = []
    for i in range(n):
        # x - r >= 0

        constraints.append(centers[i, 1] - radii[i])
        # 1 - r - y >= 0
        constraints.append(1.0 - radii[i] - centers[i, 1])
    return np.array(constraints)

    # Compute pairwise distances
    # centers shape (n, 2)
    # diff = centers[:, None, :] - centers[None, :, :] # (n, n, 2)
    # dists = np.sqrt(np.sum(diff**2, axis=2)) # (n, n)
    
    # This creates full matrix, n=26 is small (26x26=676).
    # But we only need upper triangle.
    
    # Using broadcasting
    c1 = centers[:, np.newaxis, :] # (26, 1, 2)
    c2 = centers[np.newaxis, :, :] # (1, 26, 2)
    dists = np.linalg.norm(c1 - c2, axis=2) # (26, 26)
    
    r1 = radii[:, np.newaxis] # (26, 1)
    r2 = radii[np.newaxis, :] # (1, 26)
    sums_r = r1 + r2
    
    # Constraint: dists - sums_r >= 0
    # We only need upper triangle (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    valid_constraints = (dists - sums_r)[mask]
    
    return valid_constraints

import numpy as np
from scipy.optimize import minimize

def objective(x):
    # x is a flat array of shape (n*3,)
    # structure: [x0, y0, r0, x1, y1, r1, ...]
    # We want to maximize sum of radii, so we minimize negative sum.
    radii = x[2::3]
    return -np.sum(radii)

def constraint_overlap(x):
    n = 26
    # Reshape x to (n, 3)
    points = x.reshape(-1, 3)
    centers = points[:, :2]
    radii = points[:, 2]
    
    # Calculate pairwise distances
    # centers shape: (n, 2)
    # Using broadcasting to compute distance matrix
    # c1 shape: (n, 1, 2)
    # c2 shape: (1, n, 2)
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    
    # Squared Euclidean distance
    diff = c1 - c2
    dist_sq = np.sum(diff**2, axis=2)
    dists = np.sqrt(np.maximum(dist_sq, 0.0)) # Avoid negative due to float errors
    
    # Radii sums
    r1 = radii[:, np.newaxis]
    r2 = radii[np.newaxis, :]
    sums_r = r1 + r2
    
    # Constraint: dist >= r_i + r_j  <=>  dist - (r_i + r_j) >= 0
    # We only need constraints for i < j (upper triangle)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    valid_constraints = (dists - sums_r)[mask]
    
    return valid_constraints

def constraint_boundary(x):
    n = 26
    points = x.reshape(-1, 3)
    centers = points[:, :2]
    radii = points[:, 2]
    
    # Constraints:
    # x >= r  => x - r >= 0
    # x <= 1 - r => 1 - r - x >= 0
    # y >= r  => y - r >= 0
    # y <= 1 - r => 1 - r - y >= 0
    
    c1 = centers[:, 0] - radii
    c2 = 1.0 - radii - centers[:, 0]
    c3 = centers[:, 1] - radii
    c4 = 1.0 - radii - centers[:, 1]
    
    return np.concatenate([c1, c2, c3, c4])

def run_packing():
    n = 26
    
    # Define bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n * 3):
        if i % 3 == 2: # radius index
            bounds.append((0.0, 0.5))
        else:
            bounds.append((0.0, 1.0))
    
    # Define constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]
    
    best_x = None
    best_fun = np.inf # We are minimizing -sum(r), so lower is better
    
    # Helper to run optimization
    def try_optimize(x0):
        nonlocal best_x, best_fun
        # Ensure x0 is within bounds (clipping)
        # Although SLSQP should handle it, clipping helps feasibility
        x0_clipped = x0.copy()
        for i in range(len(x0_clipped)):
            low, high = bounds[i]
            if x0_clipped[i] < low:
                x0_clipped[i] = low
            elif x0_clipped[i] > high:
                x0_clipped[i] = high
        
        try:
            res = minimize(objective, x0_clipped, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 5000, 'ftol': 1e-12})

                    best_x = res.x.copy()
        except Exception:
