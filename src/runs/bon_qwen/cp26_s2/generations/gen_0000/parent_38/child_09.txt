import numpy as np
from scipy.optimize import minimize

def center_loss(c_flat, n, radii, penalty):
    c = c_flat.reshape(n, 2)
    loss = 0.0
    for i in range(n):
        x, y = c[i]
        r = radii[i]
        loss += penalty * (np.maximum(0, r - x)**2 + np.maximum(0, r - (1.0 - x))**2 + 
                           np.maximum(0, r - y)**2 + np.maximum(0, r - (1.0 - y))**2)
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            req = radii[i] + radii[j]
            if req > dist:
                loss += penalty * (req - dist) ** 2
    return loss

def joint_loss(v, n, penalty):
    centers = v[:2 * n].reshape(n, 2)
    radii = v[2 * n:]
    loss = -np.sum(radii)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        loss += penalty * (np.maximum(0, r - x)**2 + np.maximum(0, r - (1.0 - x))**2 + 
                           np.maximum(0, r - y)**2 + np.maximum(0, r - (1.0 - y))**2)
