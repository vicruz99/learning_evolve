   These are encoded as inequality constraints $\ge 0$.
4. **Initialization**: Use multiple random starts and one structured grid start to mitigate local minima traps. Initial radii are set conservatively (0.08) to satisfy constraints initially.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_constraints(v):
    n = N_CIRCLES
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    cons = []
    # Boundary constraints: r <= x <= 1-r  and  r <= y <= 1-r
    # Formulated as g(v) >= 0
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Non-overlap constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = (x[i] - x[j])**2 + (y[i] - y[j])**2
            rad_sum = r[i] + r[j]
            cons.append(dist_sq - rad_sum**2)
            
    return np.array(cons)

def constraint_fun(v):
    return compute_constraints(v)

def objective_fun(v):
    # Maximize sum of radii => minimize negative sum
    return -np.sum(v[2::3])

def run_packing():
    n = N_CIRCLES
    best_sum_r = -1.0
    best_x = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    starts = []
    
    # 1. Random starts
    for seed in range(25):
        np.random.seed(seed)
        centers = np.random.uniform(0.2, 0.8, size=(n, 2))
        radii = np.full(n, 0.08)
        starts.append(np.concatenate([centers.flatten(), radii]))
        
    # 2. Structured start (5x5 grid + 1 extra)
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append((0.1 + i * 0.2, 0.1 + j * 0.2))
    pts.append((0.05, 0.95))
    arr = np.zeros(3 * n)
    for k in range(n):
        arr[3*k] = pts[k][0]
        arr[3*k+1] = pts[k][1]
        arr[3*k+2] = 0.08
    starts.append(arr)
    
    # Optimization loop
    for x0 in starts:
        try:
            res = minimize(objective_fun, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            curr_sum = -res.fun
            if curr_sum > best_sum_r:
                best_sum_r = curr_sum
                best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        # Fallback
        np.random.seed(42)
        centers = np.random.uniform(0.2, 0.8, size=(n, 2))
        radii = np.full(n, 0.05)
        return centers, radii, float(np.sum(radii))
        
    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]
    
    # Post-processing to ensure strict validity against tolerance
    # Enforce boundary constraints strictly
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        radii[i] = min(r, x, 1-x, y, 1-y)
        
    # Iteratively resolve any remaining numerical overlaps
    for _ in range(10):
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                if dist < radii[i] + radii[j] - 1e-14:
                    overlap = radii[i] + radii[j] - dist
                    delta = overlap / 2.0 + 1e-14
                    radii[i] -= delta
                    radii[j] -= delta
                    if radii[i] < 0: radii[i] = 0
                    if radii[j] < 0: radii[j] = 0
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
