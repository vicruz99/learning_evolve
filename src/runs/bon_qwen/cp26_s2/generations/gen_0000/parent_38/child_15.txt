import numpy as np
from scipy.optimize import minimize


def smooth_max(t, eps=1e-6):
    """Smooth approximation of max(0, t) that is differentiable everywhere."""
    return 0.5 * (t + np.sqrt(t**2 + eps**2))


def compute_objective(x, n, penalty_weight):
    """Compute objective function with smooth penalties for constraints."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    penalty = 0.0
    
    # Boundary constraint penalties
    for i in range(n):
        penalty += smooth_max(radii[i] - centers[i, 0])**2
        penalty += smooth_max(radii[i] - (1.0 - centers[i, 0]))**2
        penalty += smooth_max(radii[i] - centers[i, 1])**2
        penalty += smooth_max(radii[i] - (1.0 - centers[i, 1]))**2
    
    # Overlap constraint penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            penalty += smooth_max(radii[i] + radii[j] - dist)**2
    
    return -np.sum(radii) + penalty_weight * penalty


def is_valid(centers, radii, n):
    """Check if the packing is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    
    return True


def hexagonal_init(n):
    """Initialize circles in a hexagonal packing pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09
    
    rows = [6, 5, 6, 5, 4]
    idx = 0
    for row_idx, count in enumerate(rows):
        y = 0.1 + row_idx * 0.18
        offset = 0.09 if row_idx % 2 == 1 else 0
        for col_idx in range(count):
            x = 0.1 + col_idx * 0.18 + offset
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def run_packing():
    n = 26
    
    best_sum = -1.0
    best_centers = None
    best_radii = None

    for trial in range(300):
        np.random.seed(trial * 137 + 42)
        
        # Use hexagonal init with perturbation
        centers, radii = hexagonal_init(n)
        
        # Add random perturbation
        centers = centers + np.random.randn(n, 2) * 0.03
        radii = radii + np.random.randn(n) * 0.02
        radii = np.maximum(radii, 0.01)

        x0 = np.concatenate([centers.flatten(), radii])
        
        # Define bounds
        bounds = []
        for i in range(2 * n):
            bounds.append((0.005, 0.995))
        for i in range(n):
            bounds.append((0.005, 0.495))
        
        # Optimize with high penalty weight
        result = minimize(
            compute_objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            args=(1e10,),
            options={
                'maxiter': 80000,
                'ftol': 1e-15,
                'gtol': 1e-10,
                'eps': 1e-8
            }
        )
        
        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
        
        # Validate the solution
        if is_valid(centers_opt, radii_opt, n):
            s = np.sum(radii_opt)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
