import numpy as np
from scipy.optimize import differential_evolution

def compute_max_radii(centers):
    """
    Computes the maximum possible radius for each circle given their centers.
    Radii are constrained by boundaries and nearest neighbors.
    """
    n = centers.shape[0]
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    d_bound = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    # Distance to other circles
    # Compute pairwise Euclidean distances efficiently using broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_neighbors = np.min(dists, axis=1) / 2.0
    
    # Radius is limited by both boundaries and neighbors
    return np.minimum(d_bound, d_neighbors)

def objective(x):
    """
    Objective function for optimization: negative sum of radii.
    """
    centers = x.reshape(-1, 2)
    radii = compute_max_radii(centers)
    # Minimize negative sum to maximize sum of radii
    return -np.sum(radii)

def run_packing():
    """
    Finds the optimal packing of 26 circles in a unit square to maximize sum of radii.
    Returns centers, radii, and sum_radii.
    """
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Use differential evolution for robust global optimization.
    # It effectively navigates the non-smooth landscape defined by min() operations.
    # polish=True applies a local BFGS optimization at the end for precision.
    res = differential_evolution(
        objective, 
        bounds, 
        maxiter=400, 
        popsize=20, 
        tol=1e-8, 
        mutation=(0.5, 1.5), 
        recombination=0.7, 
        seed=42,
        polish=True
    )
    
    centers = res.x.reshape(-1, 2)
    radii = compute_max_radii(centers)
    
    # Ensure non-negative radii to handle any floating point inaccuracies
    radii = np.maximum(radii, 0.0)
    
    total_sum = np.sum(radii)
    return centers, radii, total_sum
