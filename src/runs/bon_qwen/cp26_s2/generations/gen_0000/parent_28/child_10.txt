def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square to maximize sum of radii.

    num_trials = 5
    max_steps = 1500
    lr = 0.05 # Learning rate for position update
    dr = 0.0004 # Radius increment per step
    relax_iters = 100 # Relaxation iterations per expansion step
    boundary_k = 20.0 # Boundary force constant

        current_radii = radii.copy()

        for step in range(max_steps):
            # Increase radii
            current_radii += dr

            for _ in range(relax_iters):

                # Compute pairwise differences: shape (N, N, 2)
                diff = current_centers[:, np.newaxis, :] - current_centers[np.newaxis, :, :]
