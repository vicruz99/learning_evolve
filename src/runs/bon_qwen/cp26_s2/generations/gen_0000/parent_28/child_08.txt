import numpy as np
from scipy.optimize import minimize


def objective(v, n):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(v[2::3])


def build_constraints(n):
    """Build list of constraint dictionaries for scipy.optimize."""
    constraints = []

    # Pairwise non-overlap: ||c_i - c_j||^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            def pair_con(v, i=i, j=j):
                dx = v[3 * i] - v[3 * j]
                dy = v[3 * i + 1] - v[3 * j + 1]
                dist_sq = dx * dx + dy * dy
                r_i = v[3 * i + 2]
                r_j = v[3 * j + 2]
                return dist_sq - (r_i + r_j) * (r_i + r_j)
            constraints.append({'type': 'ineq', 'fun': pair_con})

    # Boundary constraints for each circle
    for i in range(n):
        # x_i >= r_i
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3 * i] - v[3 * i + 2]})
        # x_i <= 1 - r_i
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3 * i] - v[3 * i + 2]})
        # y_i >= r_i
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3 * i + 1] - v[3 * i + 2]})
        # y_i <= 1 - r_i
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3 * i + 1] - v[3 * i + 2]})
        # r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3 * i + 2]})

    return constraints


def ensure_feasibility(centers, radii, n):
    """Post-process to ensure all constraints are satisfied."""
    # Clip radii to boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1.0 - x, y, 1.0 - y)
        r = max(r, 0.0)
        radii[i] = r

    # Iteratively resolve overlaps by shrinking the smaller circle more
    for _ in range(500):
        feasible = True
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                sum_r = radii[i] + radii[j]
                if dist < sum_r - 1e-14:
                    overlap = sum_r - dist
                    # Shrink proportionally to radii (shrink smaller more to preserve larger)
                    factor_i = radii[i] / (sum_r + 1e-15)
                    factor_j = radii[j] / (sum_r + 1e-15)
                    radii[i] -= overlap * factor_i
                    radii[j] -= overlap * factor_j
                    radii[i] = max(radii[i], 0.0)
                    radii[j] = max(radii[j], 0.0)
                    feasible = False
        if feasible:
            break

    # Final boundary check
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(radii[i], x, 1.0 - x, y, 1.0 - y)
        radii[i] = max(radii[i], 0.0)

    return centers, radii


def generate_hexagonal_init(n, r_init):
    """Generate hexagonal lattice initial configuration."""
    centers = []
    y = r_init
    row = 0
    while len(centers) < n:
        x_start = r_init if row % 2 == 0 else 2.0 * r_init
        x = x_start
        while x + r_init <= 1.0 and len(centers) < n:
            centers.append([x, y])
            x += 2.0 * r_init
        y += r_init * np.sqrt(3.0)
        row += 1
    return np.array(centers[:n])


def generate_grid_init(n, r_init):
    """Generate grid-based initial configuration."""
    centers = []
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    step_x = (1.0 - 2 * r_init) / max(cols - 1, 1)
    step_y = (1.0 - 2 * r_init) / max(rows - 1, 1)
    for r in range(rows):
        for c in range(cols):
            if len(centers) < n:
                x = r_init + c * step_x
                y = r_init + r * step_y
                centers.append([x, y])
    return np.array(centers[:n])


def run_packing():
    n = 26
    constraints = build_constraints(n)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Try multiple initial configurations
    init_configs = []

    # Hexagonal packings with different radii
    for r in [0.085, 0.09, 0.095]:
        centers = generate_hexagonal_init(n, r)
        radii = np.full(n, r)
        vars_init = np.zeros(3 * n)
        vars_init[0::3] = centers[:, 0]
        vars_init[1::3] = centers[:, 1]
        vars_init[2::3] = radii
        init_configs.append(vars_init)

    # Grid packings with different radii
    for r in [0.08, 0.085, 0.09]:
        centers = generate_grid_init(n, r)
        radii = np.full(n, r)
        vars_init = np.zeros(3 * n)
        vars_init[0::3] = centers[:, 0]
        vars_init[1::3] = centers[:, 1]
        vars_init[2::3] = radii
        init_configs.append(vars_init)

    # Random configurations
    for seed in [42, 123, 456]:
        np.random.seed(seed)
        centers = np.random.rand(n, 2) * 0.6 + 0.2
        radii = np.full(n, 0.05)
        vars_init = np.zeros(3 * n)
        vars_init[0::3] = centers[:, 0]
        vars_init[1::3] = centers[:, 1]
        vars_init[2::3] = radii
        init_configs.append(vars_init)

    # Run optimization for each initial config
    for vars_init in init_configs:
        result = minimize(
            objective,
            vars_init.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
        )

        centers_opt = np.column_stack((result.x[0::3], result.x[1::3]))
        radii_opt = result.x[2::3]

        centers_opt, radii_opt = ensure_feasibility(centers_opt, radii_opt, n)

        current_sum = np.sum(radii_opt)

        if current_sum > best_sum:
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
            best_sum = current_sum

    # Additional refinement: try to increase radii locally
    for _ in range(50):
        for i in range(n):
            # Try increasing radius of circle i
            old_r = best_radii[i]
            best_radii[i] = min(old_r * 1.005, 0.5)
            # Check feasibility
            feasible = True
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((best_centers[i] - best_centers[j]) ** 2))
                if dist < best_radii[i] + best_radii[j] - 1e-12:
                    feasible = False
                    break
            # Check boundary
            x, y = best_centers[i]
            if best_radii[i] > x - 1e-12 or best_radii[i] > 1.0 - x - 1e-12 or best_radii[i] > y - 1e-12 or best_radii[i] > 1.0 - y - 1e-12:
                feasible = False

            if not feasible:
                # Find max feasible radius for this circle
                max_r = best_radii[i]
                for j in range(n):
                    if i == j:
                        continue
                    dist = np.sqrt(np.sum((best_centers[i] - best_centers[j]) ** 2))
                    max_r = min(max_r, dist - best_radii[j])
                max_r = min(max_r, x, 1.0 - x, y, 1.0 - y)
                best_radii[i] = max(max_r, 0.0)

    # Final feasibility enforcement
    best_centers, best_radii = ensure_feasibility(best_centers, best_radii, n)
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
