The problem of packing circles to maximize the sum of radii is non-convex due to the distance constraints. A robust approach combines:
1. **Initialization**: Place circles on a perturbed hexagonal lattice, which approximates optimal local packing density.
2. **Penalty-based Local Optimization**: Use L-BFGS-B to maximize the sum of radii while penalizing boundary violations and overlaps. A smooth approximation of the max function ensures stable gradients. The penalty weight is gradually increased (annealing) to drive the solution towards feasibility.
3. **Multiple Restarts**: Run the optimizer from several perturbed initial configurations to escape local minima.
4. **LP Projection**: After finding optimal centers via penalty methods, fix the centers and solve a Linear Program (LP) to find the exact maximum radii that satisfy all non-overlap and boundary constraints. This guarantees a valid packing and often improves the sum of radii by optimally utilizing the available space around the converged centers.
This hybrid strategy leverages the speed of gradient-based methods for geometry adjustment and the precision of LP for radius assignment.

import numpy as np
from scipy.optimize import minimize, linprog

def objective_func(v, N, pen):
    """
    Computes the objective function for circle packing optimization.
    Minimizes negative sum of radii + penalty for overlaps and boundary violations.
    Uses a smooth approximation for max(0, x)^2 to ensure differentiability.
    """
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    # Distance to nearest boundary for each circle
    dist_bound = np.minimum(np.minimum(c, 1 - c), 0.5).min(axis=1)
    pb = r - dist_bound
    # Smooth penalty: approx max(0, pb)^2
    pen_bound = np.sum(0.5 * (pb + np.sqrt(pb**2 + 1e-8))**2)
    
    # Pairwise distances for upper triangle
    i, j = np.triu_indices(N, k=1)
    diffs = c[i] - c[j]
    dists = np.sqrt(np.sum(diffs**2, axis=1) + 1e-12)
    po = r[i] + r[j] - dists
    # Smooth penalty for overlaps
    pen_overlap = np.sum(0.5 * (po + np.sqrt(po**2 + 1e-8))**2)
    
    return -np.sum(r) + pen * (pen_bound + pen_overlap)

def run_packing():
    np.random.seed(42)
    N = 26
    
    # 1. Initialize centers on a hexagonal lattice pattern
    centers = np.zeros((N, 2))
    idx = 0
    y = 0.12
    while idx < N and y < 0.95:
        x = 0.12 if int((y - 0.12) / 0.15) % 2 == 0 else 0.20
        while idx < N and x < 0.95:
            centers[idx] = [x, y]
            idx += 1
            x += 0.18
        y += 0.154
        
    # Fill any remaining circles if the grid didn't cover exactly 26
    if idx < N:
        for k in range(idx, N):
            centers[k] = [0.2 + (k-idx)*0.12, 0.2 + (k-idx)*0.12]
            
    radii_init = np.full(N, 0.06)
    x0 = np.concatenate([centers.ravel(), radii_init])
    bounds = [(0.0, 1.0)]*(2*N) + [(0.0, 0.5)]*N
    
    best_x = x0
    best_obj = np.inf
    
    # 2. Optimization with multiple restarts and increasing penalty
    for seed in range(10):
        x_curr = x0.copy()
        # Perturb initial guess
        x_curr[:2*N] += np.random.randn(2*N) * 0.03
        x_curr[:2*N] = np.clip(x_curr[:2*N], 0.05, 0.95)
        x_curr[2*N:] += np.random.randn(N) * 0.02
        x_curr[2*N:] = np.clip(x_curr[2*N:], 0.01, 0.25)
        
        # Annealing penalty schedule
        for pen in [100, 500, 2000, 10000, 50000]:
            res = minimize(objective_func, x_curr, args=(N, pen), method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8})
            x_curr = res.x
            if res.fun < best_obj:
                best_obj = res.fun
                best_x = x_curr
                
    c_best = best_x[:2*N].reshape(N, 2)
    r_best = best_x[2*N:]
    
    # 3. LP Projection: Fix centers, find optimal radii subject to constraints
    # This guarantees a valid packing and often improves the objective
    A_ub = np.zeros((N*(N-1)//2 + 4*N, N))
    b_ub = np.zeros(N*(N-1)//2 + 4*N)
    idx_c = 0
    
    i, j = np.triu_indices(N, k=1)
    diffs = c_best[i] - c_best[j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    
    for k in range(len(i)):
        A_ub[idx_c, i[k]] = 1.0
        A_ub[idx_c, j[k]] = 1.0
        b_ub[idx_c] = dists[k] - 1e-9  # Slight margin for numerical safety
        idx_c += 1
        
    dist_b = np.minimum(np.minimum(c_best, 1 - c_best), 0.5).min(axis=1)
    for k in range(N):
        A_ub[idx_c, k] = 1.0
        b_ub[idx_c] = dist_b[k] - 1e-9
        idx_c += 1
        
    try:
        # Maximize sum(r) <=> Minimize -sum(r)
        res_lp = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
        if res_lp.success:
            r_best = res_lp.x
    except Exception:
        # Fallback to optimized radii if LP fails (rare)
        pass
        
    sum_radii = np.sum(r_best)
    return c_best, r_best, sum_radii
