I'll use a multi-stage optimization approach:

1. **Initial Configuration**: Generate a hexagonal packing of 26 circles with radius ~0.11, which fits approximately 26-27 circles in the unit square.

2. **Full Nonlinear Optimization with SLSQP**: Encode the problem as:
   - Variables: [x₁,y₁,r₁, x₂,y₂,r₂, ..., x₂₆,y₂₆,r₂₆] (78 variables)
   - Objective: maximize sum of radii (minimize negative sum)
   - Constraints: boundary (rᵢ ≤ min(xᵢ,1-xᵢ,yᵢ,1-yᵢ)) and non-overlap (‖cᵢ-cⱼ‖ ≥ rᵢ+rⱼ)
   - Use class-based constraint objects to avoid lambdas and closures

3. **Alternating Optimization Fallback**: If SLSQP doesn't converge well:
   - Fix centers, solve LP for optimal radii (fast, exact)
   - Fix radii, optimize centers to reduce overlaps (nonlinear)
   - Iterate

4. **Multiple Restarts**: Try several initial configurations (different hexagonal offsets, slightly different radii) to avoid local optima.

5. **Constraint Repair**: After optimization, reduce radii slightly to ensure numerical validity.

Key insight: For maximizing sum of radii (not areas), boundary circles can be larger than interior ones. The optimization should naturally discover this.

The target sum is 2.636, meaning average radius ≈ 0.1014. Hexagonal packing of equal circles with r≈0.105 gives sum≈2.73, so we should easily exceed the target with unequal circles that exploit boundaries.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist, squareform

N_CIRCLES = 26
STATE_DIM = N_CIRCLES * 3


class BoundaryConstraintX:
    """Constraint: x_i >= r_i"""
    def __init__(self, idx_x, idx_r):
        self.idx_x = idx_x
        self.idx_r = idx_r

    def __call__(self, x):
        return x[self.idx_x] - x[self.idx_r]


class BoundaryConstraint1MX:
    """Constraint: 1 - x_i >= r_i"""
    def __init__(self, idx_x, idx_r):
        self.idx_x = idx_x
        self.idx_r = idx_r

    def __call__(self, x):
        return 1.0 - x[self.idx_x] - x[self.idx_r]


class BoundaryConstraintY:
    """Constraint: y_i >= r_i"""
    def __init__(self, idx_y, idx_r):
        self.idx_y = idx_y
        self.idx_r = idx_r

    def __call__(self, x):
        return x[self.idx_y] - x[self.idx_r]


class BoundaryConstraint1MY:
    """Constraint: 1 - y_i >= r_i"""
    def __init__(self, idx_y, idx_r):
        self.idx_y = idx_y
        self.idx_r = idx_r

    def __call__(self, x):
        return 1.0 - x[self.idx_y] - x[self.idx_r]


class NonoverlapConstraint:
    """Constraint: ||c_i - c_j|| >= r_i + r_j"""
    def __init__(self, idx_xi, idx_yi, idx_ri, idx_xj, idx_yj, idx_rj):
        self.idx_xi = idx_xi
        self.idx_yi = idx_yi
        self.idx_ri = idx_ri
        self.idx_xj = idx_xj
        self.idx_yj = idx_yj
        self.idx_rj = idx_rj

    def __call__(self, x):
        dx = x[self.idx_xi] - x[self.idx_xj]
        dy = x[self.idx_yi] - x[self.idx_yj]
        dist = np.sqrt(dx * dx + dy * dy)
        return dist - x[self.idx_ri] - x[self.idx_rj]


def objective_function(x):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)"""
    radii = x[2::3]
    return -np.sum(radii)


def get_bounds():
    """Return bounds for all variables: x,y in [0,1], r in [0,1]"""
    bounds = []
    for i in range(N_CIRCLES):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    return bounds


def get_constraints():
    """Return all constraint dictionaries for SLSQP"""
    constraints = []

    # Boundary constraints (4 per circle)
    for i in range(N_CIRCLES):
        constraints.append({'type': 'ineq', 'fun': BoundaryConstraintX(3 * i, 3 * i + 2)})
        constraints.append({'type': 'ineq', 'fun': BoundaryConstraint1MX(3 * i, 3 * i + 2)})
        constraints.append({'type': 'ineq', 'fun': BoundaryConstraintY(3 * i + 1, 3 * i + 2)})
        constraints.append({'type': 'ineq', 'fun': BoundaryConstraint1MY(3 * i + 1, 3 * i + 2)})

    # Non-overlap constraints (C(26,2) = 325)
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            constraints.append({
                'type': 'ineq',
                'fun': NonoverlapConstraint(3 * i, 3 * i + 1, 3 * i + 2, 3 * j, 3 * j + 1, 3 * j + 2)
            })

    return constraints


def generate_hexagonal_packing(offset_x=0.0, offset_y=0.0, radius=0.11):
    """Generate initial hexagonal packing configuration"""
    centers = []
    r = radius

    y = r + offset_y
    row = 0
    max_rows = 20
    while len(centers) < N_CIRCLES and row < max_rows and y + r <= 1.0:
        x = r + offset_x
        if row % 2 == 1:
            x = r * 2.0 + offset_x

        while x + r <= 1.0 and len(centers) < N_CIRCLES:
            centers.append([x, y])
            x += 2 * r

        y += r * np.sqrt(3)
        row += 1

    # Fill remaining with grid if needed
    while len(centers) < N_CIRCLES:
        for x in np.linspace(r, 1 - r, 10):
            if len(centers) >= N_CIRCLES:
                break
            for y in np.linspace(r, 1 - r, 10):
                if len(centers) >= N_CIRCLES:
                    break
                centers.append([x + offset_x * 0.1, y + offset_y * 0.1])

    centers = np.array(centers[:N_CIRCLES])
    radii = np.ones(N_CIRCLES) * r
    return centers, radii


def unflatten_state(x):
    """Convert flattened optimization variable to centers and radii arrays"""
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.zeros(N_CIRCLES)
    for i in range(N_CIRCLES):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = x[3 * i + 2]
    return centers, radii


def flatten_state(centers, radii):
    """Convert centers and radii to flattened optimization variable"""
    x = np.zeros(STATE_DIM)
    for i in range(N_CIRCLES):
        x[3 * i] = centers[i, 0]
        x[3 * i + 1] = centers[i, 1]
        x[3 * i + 2] = radii[i]
    return x


def repair_constraints(centers, radii):
    """Repair constraint violations by reducing radii"""
    radii = radii.copy()

    # Fix boundary violations
    for i in range(N_CIRCLES):
        x, y = centers[i]
        ri_max = min(x, 1.0 - x, y, 1.0 - y)
        radii[i] = min(radii[i], ri_max - 1e-10)

    # Fix overlap violations iteratively
    for _ in range(200):
        max_overlap = 0.0
        worst_i, worst_j = -1, -1
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
                    worst_i, worst_j = i, j

        if max_overlap <= 1e-10:
            break

        # Reduce both radii proportionally to their sizes
        total_r = radii[worst_i] + radii[worst_j]
        if total_r > 0:
            radii[worst_i] -= max_overlap * 0.5 * radii[worst_i] / total_r
            radii[worst_j] -= max_overlap * 0.5 * radii[worst_j] / total_r

    radii = np.maximum(radii, 1e-12)
    return centers, radii


def optimize_radii_lp(centers):
    """Solve LP to maximize sum of radii given fixed centers"""
    dists = squareform(pdist(centers))

    c = -np.ones(N_CIRCLES)

    A_ub = []
    b_ub = []

    # Boundary constraints
    for i in range(N_CIRCLES):
        xi, yi = centers[i]
        ri_max = min(xi, 1.0 - xi, yi, 1.0 - yi)
        row = np.zeros(N_CIRCLES)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(ri_max - 1e-12)

    # Non-overlap constraints
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            row = np.zeros(N_CIRCLES)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j] - 1e-12)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None) for _ in range(N_CIRCLES)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return np.maximum(result.x, 1e-12)
    else:
        return np.ones(N_CIRCLES) * 0.05


def centers_penalty(x_reshaped, radii):
    """Penalty function for center optimization"""
    centers = x_reshaped.reshape(N_CIRCLES, 2)
    penalty = 0.0

    for i in range(N_CIRCLES):
        xi, yi = centers[i]
        ri = radii[i]
        penalty += max(0.0, ri - xi) ** 2
        penalty += max(0.0, ri - (1.0 - xi)) ** 2
        penalty += max(0.0, ri - yi) ** 2
        penalty += max(0.0, ri - (1.0 - yi)) ** 2

    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2

    return penalty


def optimize_centers_nl(centers, radii):
    """Optimize centers given fixed radii using penalty method"""
    x0 = centers.flatten()

    def obj(x):
        return centers_penalty(x, radii)

    bounds = [(0.0, 1.0) for _ in range(2 * N_CIRCLES)]

    result = minimize(obj, x0, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': 1000, 'ftol': 1e-15})
    return result.x.reshape(N_CIRCLES, 2)


def alternating_optimization():
    """Alternating LP for radii and nonlinear optimization for centers"""
    centers, radii = generate_hexagonal_packing(radius=0.105)

    for iteration in range(40):
        # Step 1: Optimize radii given centers (LP)
        radii = optimize_radii_lp(centers)

        # Step 2: Optimize centers given radii (nonlinear)
        centers = optimize_centers_nl(centers, radii)

    return centers, radii


def run_packing():
    """Main function to pack 26 circles in unit square maximizing sum of radii"""

    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Strategy 1: Full SLSQP optimization with multiple restarts
    initial_configs = [
        generate_hexagonal_packing(radius=0.11, offset_x=0.0, offset_y=0.0),
        generate_hexagonal_packing(radius=0.105, offset_x=0.02, offset_y=0.02),
        generate_hexagonal_packing(radius=0.12, offset_x=0.0, offset_y=0.0),
        generate_hexagonal_packing(radius=0.10, offset_x=0.05, offset_y=0.05),
    ]

    constraints = get_constraints()
    bounds = get_bounds()

    for trial, (centers, radii) in enumerate(initial_configs):
        x0 = flatten_state(centers, radii)

        result = minimize(objective_function, x0, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False})

        centers_opt, radii_opt = unflatten_state(result.x)
        centers_opt, radii_opt = repair_constraints(centers_opt, radii_opt)

        if validate_packing(centers_opt, radii_opt):
            current_sum = np.sum(radii_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    # Strategy 2: Alternating LP/nonlinear optimization
    centers_alt, radii_alt = alternating_optimization()
    centers_alt, radii_alt = repair_constraints(centers_alt, radii_alt)

    if validate_packing(centers_alt, radii_alt):
        alt_sum = np.sum(radii_alt)
        if alt_sum > best_sum:
            best_sum = alt_sum
            best_centers = centers_alt.copy()
            best_radii = radii_alt.copy()

    # Strategy 3: Fine-tune best solution with SLSQP
    if best_centers is not None:
        x0 = flatten_state(best_centers, best_radii)
        result = minimize(objective_function, x0, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': 3000, 'ftol': 1e-15, 'disp': False})

        centers_opt, radii_opt = unflatten_state(result.x)
        centers_opt, radii_opt = repair_constraints(centers_opt, radii_opt)

        if validate_packing(centers_opt, radii_opt):

        best_centers, best_radii = repair_constraints(best_centers, best_radii)
