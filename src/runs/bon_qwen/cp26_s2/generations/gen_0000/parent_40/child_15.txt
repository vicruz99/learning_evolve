import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_constraints(x):
    """
    Compute all inequality constraints for the packing problem.
    x is a flat array of length 3*N: [x0, y0, r0, x1, y1, r1, ...]
    Returns an array where all elements must be >= 0.
    """
    centers = x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = x[2 * N_CIRCLES:]
    
    # Boundary constraints: 
    # x_i - r_i >= 0
    # 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0
    c_bound = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])
    
    # Distance constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2 for all i < j
    # Vectorized pairwise computation
    diff = centers[:, None, :] - centers[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, None] + radii[None, :]
    
    # Extract upper triangular indices to avoid duplicates and self-comparison
    triu_idx = np.triu_indices(N_CIRCLES, k=1)
    c_dist = dist_sq[triu_idx] - r_sum[triu_idx]**2
    
    return np.concatenate([c_bound, c_dist])

def compute_objective(x):
    """
    Objective function to minimize (negative sum of radii).
    """
    radii = x[2 * N_CIRCLES:]
    return -np.sum(radii)

def run_packing():
    best_val = -np.inf
    best_x = None
    
    # Bounds for variables: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    constraint = {'type': 'ineq', 'fun': compute_constraints}
    
    # Run multiple trials with different initial configurations
    for trial in range(20):
        centers0 = np.zeros((N_CIRCLES, 2))
        radii0 = np.full(N_CIRCLES, 0.08)
        
        if trial % 2 == 0:
            # Square grid layout initialization
            idx = 0
            for r in range(5):
                for c in range(6):
                    if idx < N_CIRCLES:
                        centers0[idx] = [(c + 0.5) / 6.0, (r + 0.5) / 5.0]
                        idx += 1
        else:
            # Hexagonal layout initialization
            idx = 0
            row_counts = [6, 5, 6, 5, 4]
            for i, count in enumerate(row_counts):
                y = (i + 0.5) / 5.0
                start_x = 0.0 if i % 2 == 0 else 1.0 / 12.0
                for j in range(count):
                    if idx < N_CIRCLES:
                        centers0[idx] = [start_x + j * (1.0 / 6.0), y]
                        idx += 1
                        
        # Add random perturbation to explore configuration space
        centers0 += np.random.uniform(-0.02, 0.02, centers0.shape)
        radii0 += np.random.uniform(-0.01, 0.01, N_CIRCLES)
        
        # Clip to ensure initial point respects bounds
        centers0 = np.clip(centers0, 0.1, 0.9)
        radii0 = np.clip(radii0, 0.05, 0.2)
        
        x0 = np.concatenate([centers0.flatten(), radii0])
        
        try:
            res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                          constraints=constraint, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_val:
                best_val = -res.fun
                best_x = res.x
        except Exception:
            continue
            
    best_centers = best_x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    best_radii = best_x[2 * N_CIRCLES:]
    
    return best_centers, best_radii, np.sum(best_radii)
