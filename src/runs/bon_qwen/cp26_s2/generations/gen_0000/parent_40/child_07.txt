import numpy as np
from scipy.optimize import minimize


def hex_init(n, spacing_factor=1.0):
    """Initialize centers in hexagonal lattice pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        cols = 5
        shift = 0.095 if row % 2 == 1 else 0
        for col in range(min(cols, n - idx)):
            centers[idx, 0] = 0.1 + col * 0.185 * spacing_factor + shift
            centers[idx, 1] = 0.1 + row * 0.162 * spacing_factor
            idx += 1
            if idx >= n:
                break
        row += 1
    return centers


def obj_overlap_penalty(x, n, radii):
    """Compute overlap penalty for given centers and fixed radii."""
    centers = x.reshape((n, 2))
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            rs = radii[i] + radii[j]
            if d < rs:
                penalty += (rs - d) ** 2
    return penalty


def obj_boundary_penalty(x, n):
    """Compute boundary violation penalty for given centers with radius constraint."""
    centers = x.reshape((n, 2))
    penalty = 0.0
    for i in range(n):
        x_i, y_i = centers[i]
        penalty += max(0, -(x_i - 0.05)) ** 2
        penalty += max(0, -(1 - x_i - 0.05)) ** 2
        penalty += max(0, -(y_i - 0.05)) ** 2
        penalty += max(0, -(1 - y_i - 0.05)) ** 2
    return penalty


def obj_combined(x, n, radii):

    centers = x.reshape((n, 2))

def optimize_centers(centers, radii, n, max_iter=1000):
    """Optimize circle centers to minimize overlaps for fixed radii."""
    x0 = centers.flatten()
    bounds = [(0.0, 1.0)] * (2 * n)

    def objective(x):
        return obj_combined(x, n, radii)

    result = minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12}
    )
    return result.x.reshape((n, 2))


def check_feasibility(centers, radii, n):
    """Check if packing is feasible."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r - 1e-10 or x > 1 - r + 1e-10 or y < r - 1e-10 or y > 1 - r + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            if d < radii[i] + radii[j] - 1e-10:
                return False
    return True

def gradual_expansion(centers, radii, n, steps=400, step_size=0.0004):
    """Gradually expand radii while optimizing centers."""

        # Increase radii

            # Back off if not feasible

            # Re-optimize
            centers = optimize_centers(centers, radii, n, max_iter=500)

    return centers, radii
