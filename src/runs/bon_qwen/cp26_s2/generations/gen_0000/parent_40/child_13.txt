import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    # Minimize negative sum of radii
    return -np.sum(v[2::3])

def constraint_fun(v):
    # Extract variables
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b_cons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    dist_sq = dx**2 + dy**2
    
    # Only need upper triangle indices to avoid duplicates
    idx = np.triu_indices(N, k=1)
    p_cons = dist_sq[idx] - dr[idx]**2
    
    return np.concatenate([b_cons, p_cons])

def get_initial_guess():
    np.random.seed(42)
    pts = []
    # Hexagonal lattice parameters
    dy = 0.165
    dx = dy * np.sqrt(3) / 2
    
    for row in range(7):
        for col in range(5):
            x = 0.13 + col * dx
            y = 0.13 + row * dy
            if row % 2 == 1:
                x += dx / 2
            if x < 0.88 and y < 0.88:
                pts.append([x, y])
                
    pts = np.array(pts[:N])
    # Small random perturbation to break symmetry and aid optimization
    pts += np.random.uniform(-0.012, 0.012, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    x0 = np.zeros(3 * N)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    x0[2::3] = 0.065  # Initial feasible radius
    return x0

def run_packing():
    x0 = get_initial_guess()
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    constraints = {'type': 'ineq', 'fun': constraint_fun}
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
    
    v = res.x
    centers = np.column_stack((v[0::3], v[1::3]))
    radii = v[2::3]
    
    # Apply slight shrinkage to guarantee strict validity within 1e-12 tolerance
    radii *= 0.998
    
    return centers, radii, float(np.sum(radii))
