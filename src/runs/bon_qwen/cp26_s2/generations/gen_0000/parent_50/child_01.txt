import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    
    # 1. Initialize centers in a hexagonal pattern
    # Arrangement: 5, 6, 5, 6, 4 = 26 circles
    row_counts = [5, 6, 5, 6, 4]
    centers = []
    
    idx = 0
    y = 0.0 # Start y offset
    for r_idx, count in enumerate(row_counts):
        x = 0.0
        # Stagger every other row (rows 1 and 3 in 0-indexed list)
        if r_idx % 2 == 1:
            x = 0.5 # Shift half a diameter relative to previous rows (normalized)
        
        for c in range(count):
            centers.append([x, y])
            x += 1.0 # Move horizontally
            idx += 1
        y += 1.0 # Move vertically
        
    centers = np.array(centers, dtype=float)
    
    # 2. Scale and Center the pattern in the [0,1]x[0,1] square
    min_x, min_y = centers.min(axis=0)
    max_x, max_y = centers.max(axis=0)
    
    width = max_x - min_x
    height = max_y - min_y
    
    # Calculate the largest uniform radius that fits the current pattern aspect ratio
    # We want to expand the pattern so that it touches boundaries
    # Initial radius is 1.0 (arbitrary unit for the lattice)
    current_r = 1.0
    
    # Distance to boundaries:
    # Left: (width) - 1 (diameter of circle) ... actually simpler to scale the grid
    # Scale factor s. New width = s * width. New height = s * height.
    # Constraints: s * width + 2r_fit <= 1? 
    # Easier: Just scale coordinates to fit and solve for r.
    
    # Normalize pattern to fit [0,1]x[0,1] with some initial padding
    centers -= [min_x, min_y]
    scale_x = 1.0 / width
    scale_y = 1.0 / height
    scale = min(scale_x, scale_y)
    
    centers *= scale
    
    # Center the pattern
    centers_x_mid = centers[:, 0].mean()
    centers_y_mid = centers[:, 1].mean()
    centers[:, 0] -= centers_x_mid - 0.5
    centers[:, 1] -= centers_y_mid - 0.5
    
    # Estimate initial radius based on minimum distance between points
    # Or simply a small value to start optimization
    # The gap between centers in our scaled grid
    min_dist = np.inf
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if d < min_dist:
                min_dist = d
    
    initial_r = min_dist / 2.0 * 0.95 # Start slightly smaller to avoid immediate overlap
    radii = np.full(n, initial_r)
    
    # 3. Define the optimization problem
    # We want to maximize sum(radii) <=> minimize -sum(radii)
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Constraints
    constraints = []
    
    # Boundary constraints: 0 <= x <= 1, 0 <= y <= 1, r >= 0
    # x_i - r_i >= 0
    # x_i + r_i <= 1
    # y_i - r_i >= 0
    # y_i + r_i <= 1
    # r_i >= 0
    
    # Non-overlap: dist(i,j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    
    # We will pass these as 'ineq' constraints: fun(x) >= 0
    
    def add_bound_constraints(constraints_list, centers_idx, radii_idx):
        # x - r >= 0
        def c1(x): return x[centers_idx] - x[radii_idx]
        # x + r <= 1 => 1 - x - r >= 0
        def c2(x): return 1.0 - x[centers_idx] - x[radii_idx]
        constraints_list.append({'type': 'ineq', 'fun': c1})
        constraints_list.append({'type': 'ineq', 'fun': c2})

    def add_y_bound_constraints(constraints_list, centers_idx, radii_idx):
        # y - r >= 0
        def c1(x): return x[centers_idx] - x[radii_idx]
        # y + r <= 1 => 1 - y - r >= 0
        def c2(x): return 1.0 - x[centers_idx] - x[radii_idx]
        constraints_list.append({'type': 'ineq', 'fun': c1})
        constraints_list.append({'type': 'ineq', 'fun': c2})

    def add_overlap_constraint(constraints_list, i, j):
        def overlap_check(x):
            x1, y1 = x[3*i], x[3*i+1]
            r1 = x[3*i+2]
            x2, y2 = x[3*j], x[3*j+1]
            r2 = x[3*j+2]
            dist_sq = (x1-x2)**2 + (y1-y2)**2
            sum_r = r1 + r2
            # We want dist >= sum_r. 
            # Using squared version to avoid sqrt, but careful with signs.
            # dist_sq >= sum_r^2. 
            # Since radii are non-negative, this is valid.
            return dist_sq - sum_r**2
        constraints_list.append({'type': 'ineq', 'fun': overlap_check})

    # Build constraint list
    for i in range(n):
        # x constraint
        add_bound_constraints(constraints, 3*i, 3*i+2)
        # y constraint
        add_y_bound_constraints(constraints, 3*i+1, 3*i+2)
        # r constraint (r >= 0) - handled by bounds usually, but for completeness
        # r_i >= 0
        def make_r_ge_0(idx):
            def c(x): return x[3*idx+2]
            return c
        constraints.append({'type': 'ineq', 'fun': make_r_ge_0(i)})

    for i in range(n):
        for j in range(i+1, n):
            add_overlap_constraint(constraints, i, j)
            
    # Bounds for variables: x, y in [0,1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    def objective(x):
        r = x[2::3] # Extract all radii
        return -np.sum(r)

    # 4. Run Optimization
    # L-BFGS-B is good for bound constraints
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-9})
    
    if res.success:
        optimal_centers = np.array(res.x).reshape(n, 3)[:, :2]
        optimal_radii = np.array(res.x).reshape(n, 3)[:, 2]
        total_sum = np.sum(optimal_radii)
        
        # Final cleanup: ensure non-negative and valid (though optimizer should do it)
        optimal_radii = np.maximum(optimal_radii, 0.0)
        
        return optimal_centers, optimal_radii, total_sum
    else:
        # Fallback to initial guess if optimization fails (unlikely)
        return centers, radii, np.sum(radii)
