To pack 26 circles in a unit square maximizing the sum of radii, we employ a hybrid optimization strategy combining **Linear Programming (LP)** for optimal radius sizing and **Force-Directed Layout** for center placement. 

**Algorithm Overview:**
1.  **Initialization:** We begin by placing 26 circle centers in a staggered grid (hexagonal-like) pattern to provide a dense initial configuration.
2.  **Iterative Optimization Loop:** For a fixed number of iterations:
    *   **Radius Optimization (LP):** For fixed center positions, we solve a linear programming problem to maximize $\sum r_i$ subject to non-overlap ($r_i + r_j \le d_{ij}$) and boundary constraints. This provides the theoretical maximum radii for the current layout.
    *   **Force Calculation:** We compute repulsive forces between centers. To guide the layout toward a more optimal configuration, we apply stronger repulsion to pairs of circles where the radius sum is nearly equal to their distance (tight constraints), effectively "pushing" them apart to create more space.
    *   **Position Update:** Centers are moved based on these forces, with damping applied to ensure stability and prevent oscillation.
3.  **Finalization:** A final LP run is performed on the optimized center positions to return the precise, valid radii and the maximum sum of radii.

This approach ensures that at every step, the radii are as large as possible given the current geometry, while the positions are iteratively adjusted to maximize that potential.

import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # 1. Initialize centers in a staggered grid pattern
    # We attempt a 5x6 grid with a shift for odd rows
    rows, cols = 6, 5
    points = []
    dx = 1.0 / (cols + 1)
    dy = 1.0 / (rows + 1)

    for r_idx in range(rows):
        y = (r_idx + 1) * dy
        # Shift x for odd rows to create hexagonal packing structure
        shift = 0.5 * dx if r_idx % 2 == 1 else 0
        for c_idx in range(cols):
            x = (c_idx + 1) * dx + shift
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])

    # Ensure we have exactly n points
    if len(points) > n:
        points = points[:n]
    elif len(points) < n:
        # Fill remaining with random points if grid was too sparse
        while len(points) < n:
            points.append([np.random.rand(), np.random.rand()])
            
    centers = np.array(points)

    # 2. Optimization Loop
    iterations = 300
    damping = 0.85
    force_factor = 50.0

    for step in range(iterations):
        # --- Step 2a: Solve LP for optimal radii given current centers ---
        # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
        c_obj = -np.ones(n)

        # Constraints matrix A_ub @ r <= b_ub
        # We collect all inequality constraints here
        constraints_A = []
        constraints_b = []

        # Pairwise distance constraints: r_i + r_j <= dist_ij
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                constraints_A.append(row)
                constraints_b.append(dist)

        # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        for i in range(n):
            x, y = centers[i]
            row_x = np.zeros(n)
            row_x[i] = 1.0
            constraints_A.append(row_x)
            constraints_b.append(x)
            
            row_x_inv = np.zeros(n)
            row_x_inv[i] = 1.0
            constraints_A.append(row_x_inv)
            constraints_b.append(1.0 - x)

            row_y = np.zeros(n)
            row_y[i] = 1.0
            constraints_A.append(row_y)
            constraints_b.append(y)

            row_y_inv = np.zeros(n)
            row_y_inv[i] = 1.0
            constraints_A.append(row_y_inv)
            constraints_b.append(1.0 - y)

        A_ub = np.array(constraints_A)
        b_ub = np.array(constraints_b)

        # Bounds for radii: r_i >= 0
        bounds = [(0, None) for _ in range(n)]

        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if res.success:
            radii = res.x
            
            # --- Step 2b: Calculate Forces to move centers ---
            # We apply repulsion, especially where constraints are tight.
            forces = np.zeros((n, 2))
            
            # Identify tight pairwise constraints to determine repulsion strength
            tight_pairs = np.zeros(n) # Count of tight constraints per circle
            
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.linalg.norm(centers[i] - centers[j])
                    sum_r = radii[i] + radii[j]
                    gap = dist - sum_r
                    
                    # If circles are touching or very close, apply repulsion
                    # The tighter the gap, the stronger the push
                    if gap < 1e-4:
                        tight_pairs[i] += 1
                        tight_pairs[j] += 1
                        if dist > 1e-9:
                            direction = (centers[i] - centers[j]) / dist
                            # Force magnitude inversely proportional to gap to avoid singularity
                            # and proportional to sum of radii (mass heuristic)
                            f_mag = force_factor * (radii[i] + radii[j]) / (gap + 1e-6)
                            forces[i] += direction * f_mag
                            forces[j] -= direction * f_mag
            
            # Update velocities and positions
            # Use simple integration: new_pos = old_pos + force * step_size * damping
            # We use a small step size for stability
            step_size = 0.001 * (1.0 + tight_pairs / 10.0) # Slightly larger steps for tighter circles
            
            # Add a small noise to escape local minima
            noise = 0.0001 * np.random.randn(n, 2)
            
            centers += (forces * step_size + noise) * damping
            
            # Clamp centers to [0, 1] to prevent floating point drift out of bounds
            # However, we must be careful not to violate boundary constraints instantly.
            # The LP in next iteration will correct radii if centers are too close to walls.
            # Clamping helps keep the simulation stable.
            centers = np.clip(centers, 1e-6, 1.0 - 1e-6)

    # 3. Final LP Run to get exact radii for the final configuration
    # Re-build constraints for final centers
    constraints_A = []
    constraints_b = []

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints_A.append(row)
            constraints_b.append(dist)

        x, y = centers[i]
        row_x = np.zeros(n); row_x[i] = 1.0
        constraints_A.append(row_x); constraints_b.append(x)
        row_x_inv = np.zeros(n); row_x_inv[i] = 1.0
        constraints_A.append(row_x_inv); constraints_b.append(1.0 - x)
        row_y = np.zeros(n); row_y[i] = 1.0
        constraints_A.append(row_y); constraints_b.append(y)
        row_y_inv = np.zeros(n); row_y_inv[i] = 1.0
        constraints_A.append(row_y_inv); constraints_b.append(1.0 - y)

    A_ub = np.array(constraints_A)
    b_ub = np.array(constraints_b)
    bounds = [(0, None) for _ in range(n)]
    c_obj = -np.ones(n)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
    else:
        # Fallback if LP fails (unlikely with valid inputs)
        radii = np.full(n, 0.01)

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
