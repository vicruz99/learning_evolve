import numpy as np
from scipy.optimize import minimize

def get_initial_guess(n=26):

    Generates a hexagonal lattice initial guess for n circles in a unit square.
    """

    # Approximate radius for 26 circles
    r_init = 0.09 
    centers = []

    # Rows and columns based on hexagonal grid
    # Spacing x: 2r, Spacing y: sqrt(3)r
    dy = np.sqrt(3) * r_init
    dx = 2 * r_init
    
    y = r_init
    row_idx = 0

        x = x_start

            centers.append([x, y])
            x += dx
            
            if len(centers) >= n:
                return np.array(centers[:n])
        
        y += dy
        row_idx += 1
        if y > 1 - r_init:
            break
            
    # If not enough centers (unlikely for n=26 with this logic), fill with random
    while len(centers) < n:
        centers.append([np.random.rand(), np.random.rand()])
        
    return np.array(centers[:n])

                        current_centers[j] -= direction * shift
            
            # Resolve boundary collisions

                x, y = current_centers[i]
                r = current_radii[i]

                    current_centers[i, 0] = r

                    current_centers[i, 0] = 1 - r
