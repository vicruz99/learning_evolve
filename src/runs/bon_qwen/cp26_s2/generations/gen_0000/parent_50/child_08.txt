import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses multi-start local optimization.
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Define objective function (negative sum of radii)
    def objective(vars):
        # vars structure: [x0, y0, r0, x1, y1, r1, ...]
        # Flattened: x, y, r for each circle
        radii = vars[2::3]
        return -np.sum(radii)

    # Define constraints
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # r >= 0
    # Distance constraints: dist(i,j) >= r_i + r_j

    def get_constraints(vars):
        constraints = []
        n_vars = len(vars)
        n_circles = n_vars // 3
        
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]

        # Boundary constraints
        for i in range(n_circles):
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[3*i] - v[3*i + 2]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i + 2]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[3*i + 1] - v[3*i + 2]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[3*i + 1] - v[3*i + 2]
            })
            # r >= 0 (handled by bounds usually, but for safety)
            # SLSQP handles bounds, so we might skip this or add it
            
        # Distance constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                def dist_con(v, i=i, j=j):
                    xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                    xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    # We want dist >= ri + rj
                    # dist^2 >= (ri + rj)^2
                    return dist_sq - (ri + rj)**2
                
                constraints.append({
                    'type': 'ineq',
                    'fun': dist_con
                })
        
        return constraints

                vars[3*k] = np.random.rand()
                vars[3*k+1] = np.random.rand()
                vars[3*k+2] = 0.02
        return vars
