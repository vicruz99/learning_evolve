def constraint_func_vec(vars_flat):
    n = 26
    centers = vars_flat[:2*n].reshape(n, 2)
    radii = vars_flat[2*n:]
    
    # Boundary constraints
    # x - r >= 0
    c1 = centers[:, 0] - radii
    # 1 - x - r >= 0
    c2 = 1.0 - centers[:, 0] - radii
    # y - r >= 0
    c3 = centers[:, 1] - radii
    # 1 - y - r >= 0
    c4 = 1.0 - centers[:, 1] - radii
    
    boundary_constr = np.concatenate([c1, c2, c3, c4])
    
    # Distance constraints
    # Compute squared distance matrix
    # centers shape (n, 2)
    # diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    # dist_sq = np.sum(diff**2, axis=2) # (n, n)
    
    # More memory efficient? n=26 is small.
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Radii sum matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraints: dist_sq - r_sum_sq >= 0
    # We only need upper triangle i < j
    # Mask
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    dist_constr = (dist_sq - r_sum_sq)[mask]
    
    return np.concatenate([boundary_constr, dist_constr])

    centers = vars_flat[:2*n].reshape(n, 2)
    radii = vars_flat[2*n:]
    
    # Boundary constraints
    # x >= r  => x - r >= 0
    c1 = centers[:, 0] - radii
    # x <= 1-r => 1 - x - r >= 0
    c2 = 1.0 - centers[:, 0] - radii
    # y >= r => y - r >= 0
    c3 = centers[:, 1] - radii
    # y <= 1-r => 1 - y - r >= 0
    c4 = 1.0 - centers[:, 1] - radii
    
    boundary = np.concatenate([c1, c2, c3, c4])
    
    # Distance constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # dist_sq >= r_sum^2
    
    # Vectorized distance calculation
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # r_sum shape (n, n)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint: dist_sq - r_sum_sq >= 0

    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    dist_constraints = (dist_sq - r_sum_sq)[mask]
