import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP optimization on a hexagonal initial guess.
    """
    n = 26
    
    # --- Helper Functions (Top Level) ---
    
    def dist_matrix(centers):
        """Compute pairwise Euclidean distances."""
        # centers shape (n, 2)
        # Result shape (n, n)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        return np.sqrt(np.sum(diff**2, axis=2))

    def objective(x):
        """
        Objective function to minimize.
        x contains [x_0, y_0, ..., x_n-1, y_n-1, r]
        We want to maximize r, so we minimize -r.
        """
        r = x[-1]
        return -r

    def boundary_constraints(x):
        """
        Constraints: centers must be in [r, 1-r] x [r, 1-r].
        x_i - r >= 0
        1 - x_i - r >= 0
        y_i - r >= 0
        1 - y_i - r >= 0
        """
        centers = x[:-1].reshape(n, 2)
        r = x[-1]
        cons = []
        for i in range(n):
            # Left
            cons.append(centers[i, 0] - r)
            # Right
            cons.append(1.0 - centers[i, 0] - r)
            # Bottom
            cons.append(centers[i, 1] - r)
            # Top
            cons.append(1.0 - centers[i, 1] - r)
        return np.array(cons)

    def overlap_constraints(x):
        """
        Constraints: distance between centers >= 2*r.
        dist(i, j) - 2*r >= 0
        """
        centers = x[:-1].reshape(n, 2)
        r = x[-1]
        dists = dist_matrix(centers)
        # We only need upper triangle constraints
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                cons.append(dists[i, j] - 2 * r)
        return np.array(cons)

    # --- Initialization ---
    
    # We try to find a good initial configuration.
    # A hexagonal packing is usually dense.
    # Let's create a hexagonal grid of points and pick 26 closest to center.

    r_init = 0.1

    points = []
    # Rough bounds for grid generation

    step_y = r_init * np.sqrt(3)

        offset = r_init if j % 2 == 1 else 0
        i = 0
        while offset + i * step_x < max_x:
            points.append([offset + i * step_x, j * step_y])
            i += 1
        j += 1
    
    points = np.array(points)
    
    # Shift points to center of square [0,1]x[0,1]
    min_pt = points.min(axis=0)
    max_pt = points.max(axis=0)
    center_grid = (min_pt + max_pt) / 2
    shift = 0.5 - center_grid
    points = points + shift
    
    # We need exactly 26 points. Select the ones closest to the center of the square (0.5, 0.5)
    dists_to_center = np.linalg.norm(points - 0.5, axis=1)
    indices = np.argsort(dists_to_center)[:n]
    initial_centers = points[indices]
    
    # Initial variables: centers flattened + radius
    x0 = np.hstack((initial_centers.flatten(), r_init))
    
    # --- Optimization ---
    
    # Define constraints for SLSQP
    # SLSQP expects constraints in form {'type': 'ineq', 'fun': f} where f(x) >= 0

        c_bound = boundary_constraints(x)
        c_overlap = overlap_constraints(x)
        return np.concatenate((c_bound, c_overlap))

    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)]

            x0,
            method='SLSQP',
            bounds=bounds,
