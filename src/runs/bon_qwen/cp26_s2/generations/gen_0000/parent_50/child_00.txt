import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective: maximize sum of radii (minimize negative sum)"""
    return -np.sum(vars[2*N_CIRCLES:])

def constraint_func(vars):
    """Compute all inequality constraints: >= 0 means satisfied"""
    n = N_CIRCLES
    c = vars[:2*n].reshape((n, 2))
    r = vars[2*n:]
    
    # Pairwise distances (upper triangle only to avoid duplicates)
    dx = c[:, 0, None] - c[None, 0, :]
    dy = c[:, 1, None] - c[None, 1, :]
    dist = np.sqrt(dx**2 + dy**2)
    idx = np.triu_indices(n, k=1)
    overlap_con = dist[idx] - (r[idx[0]] + r[idx[1]])
    
    # Boundary constraints: center - radius >= 0  and  1 - center - radius >= 0
    b1 = c[:, 0] - r
    b2 = 1.0 - c[:, 0] - r
    b3 = c[:, 1] - r
    b4 = 1.0 - c[:, 1] - r
    
    return np.concatenate([overlap_con, b1, b2, b3, b4])

def run_packing():
    n = N_CIRCLES
    
    # 1. Initialize with a hexagonal lattice for high packing density
    r_init = 0.08
    dx = 2.0 * r_init
    dy = np.sqrt(3.0) * r_init
    
    centers = []
    y = 0.0
    row_idx = 0
    while len(centers) < n:
        x_offset = 0.5 * dx if row_idx % 2 == 1 else 0.0
        x = x_offset
        while x + 2.0 * r_init <= 1.0 + 1e-6 and len(centers) < n:
            centers.append([x + r_init, y + r_init])
            x += dx
        y += dy
        row_idx += 1
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    
    # 2. Flatten variables for optimizer: [x1, y1, ..., xn, yn, r1, ..., rn]
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)] * 2*n + [(1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # 3. Run SLSQP to maximize sum of radii subject to constraints
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 5000, 'ftol': 1e-12})
    
    c_opt = res.x[:2*n].reshape((n, 2))
    r_opt = res.x[2*n:]
    
    return c_opt, r_opt, float(np.sum(r_opt))
