import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars[2*n:])

def compute_constraints(vars, n):
    """Compute inequality constraints: overlaps and boundaries."""
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    constraints = []
    
    # Non-overlap: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j])**2)
            rad_sum = radii[i] + radii[j]
            constraints.append(dist_sq - rad_sum**2)
            
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r)
        constraints.append(1.0 - x - r)
        constraints.append(y - r)
        constraints.append(1.0 - y - r)
        
    return np.array(constraints)

def get_initial_guess(n):
    """Generate a hexagonal lattice initialization scaled to the unit square."""
    centers = []
    idx = 0
    row = 0
    while idx < n:
        shift = 0.0 if row % 2 == 0 else 0.5
        num_in_row = int(np.ceil((n - idx) / 2.0))
        if idx + num_in_row > n:
            num_in_row = n - idx
        
        for col in range(num_in_row):
            centers.append([col + shift, row * np.sqrt(3) / 2.0])
            idx += 1
        row += 1
        
    centers = np.array(centers)
    # Normalize to fit safely inside [0.05, 0.95]
    c_min, c_max = centers.min(axis=0), centers.max(axis=0)
    centers = 0.05 + (centers - c_min) / (c_max - c_min) * 0.9
    
    radii = np.full(n, 0.04)
    return np.hstack([centers.flatten(), radii])

def run_packing():
    n = 26
    best_sum_r = -np.inf
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n

    for seed in range(5):
        np.random.seed(seed)
        x0 = get_initial_guess(n)

        try:
            res = minimize(compute_objective, x0, args=(n,), method='SLSQP',

                    best_radii = res.x[2*n:]
        except Exception:
            continue
            
    if best_centers is None:
        # Fallback safe packing
        grid = np.linspace(0.1, 0.9, 6)
        centers = np.array([[grid[i%6], grid[i//6]] for i in range(n)])
        radii = np.full(n, 0.02)
        return centers, radii, np.sum(radii)

    radii = best_radii.copy()
    centers = best_centers.copy()

    for _ in range(3):
        has_overlap = False
        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(centers[i] - centers[j])

                    has_overlap = True
                    break

    return centers, radii, float(np.sum(radii))
