import numpy as np
from scipy.optimize import minimize

def run_packing():

    Pack 26 circles in a unit square to maximize the sum of radii.
    """

    best_r = 0.0
    
    # Function to calculate overlap energy for a fixed radius r
    # This is used as the objective function for the optimizer
    def overlap_energy(centers_flat, r):
        centers = centers_flat.reshape(n, 2)
        energy = 0.0
        
        # Overlap penalty between circles
        # We only check upper triangle to avoid double counting
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < 2.0 * r:
                    overlap = 2.0 * r - dist
                    energy += overlap * overlap
                    
        return energy

    # Helper to run optimization for a specific radius
    def solve_for_radius(r, init_centers):
        # Bounds for centers: [r, 1-r]
        lower = r
        upper = 1.0 - r
        
        # Clip initial centers to be within bounds
        current_centers = np.clip(init_centers, lower, upper)
        
        # Define bounds for L-BFGS-B
        bounds = [(lower, upper)] * (2 * n)
        
        # Optimize
        # maxiter set moderately high to allow convergence
        res = minimize(
            overlap_energy, 
            current_centers.flatten(), 
            args=(r,), 
            method='L-BFGS-B', 
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-12}
        )
        
        min_energy = res.fun
        opt_centers = res.x.reshape(n, 2)
