import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26

    best_centers = None
    best_radii = None
    best_sum = 0.0

    for trial in range(80):
        centers, radii = try_config(n, trial)

        if validate_simple(centers, radii, n):
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()

    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        for i in range(n):
            best_centers[i] = [0.1 + (i % 5) * 0.2, 0.1 + (i // 5) * 0.2]
            best_radii[i] = 0.0
        best_sum = 0.0

    return best_centers, best_radii, best_sum


def try_config(n, trial):
    centers = initialize(n, trial)
    radii = np.ones(n) * 0.008

    centers, radii = expand(centers, radii, n)
    centers, radii = refine(centers, radii, n)

    return centers, radii


def initialize(n, trial):
    centers = np.zeros((n, 2))

    if trial < 40:
        row_configs = [
            [5, 6, 5, 6, 4],
            [6, 5, 6, 5, 4],
            [7, 6, 7, 6],
            [4, 6, 6, 6, 4],
            [5, 5, 5, 5, 6],
            [6, 6, 6, 4, 4],
            [4, 5, 6, 5, 6],
            [5, 6, 4, 6, 5],
        ]
        config = row_configs[trial % len(row_configs)]
        n_rows = len(config)

        if n_rows > 1:
            y_positions = np.linspace(0.08, 0.92, n_rows)
        else:
            y_positions = [0.5]

        idx = 0
        for row_idx, count in enumerate(config):
            y = y_positions[row_idx]
            if count <= 1:
                for col_idx in range(count):
                    x = 0.5
                    centers[idx] = [np.clip(x, 0.02, 0.98), np.clip(y, 0.02, 0.98)]
                    idx += 1
            else:
                x_positions = np.linspace(0.03, 0.97, count)
                for col_idx in range(count):
                    x = x_positions[col_idx]
                    if row_idx % 2 == 1 and count > 1:
                        offset = (0.97 - 0.03) / (2 * (count - 1))
                        x = x + offset
                    centers[idx] = [np.clip(x, 0.02, 0.98), np.clip(y, 0.02, 0.98)]
                    idx += 1
    else:
        centers = np.random.uniform(0.12, 0.88, (n, 2))

    return centers


def expand(centers, radii, n, max_iter=8000):
    for it in range(max_iter):
        cooling = max(0.03, 1.0 - it / max_iter)

        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist = np.sqrt(dx * dx + dy * dy) + 1e-14
                min_dist = radii[i] + radii[j]

                if dist < min_dist:
                    overlap = min_dist - dist
                    strength = 18.0 * cooling
                    f = strength * overlap / dist
                    fx = f * dx / dist
                    fy = f * dy / dist

                    forces[i][0] -= fx
                    forces[i][1] -= fy
                    forces[j][0] += fx
                    forces[j][1] += fy

        for i in range(n):
            x = centers[i][0]
            y = centers[i][1]
            r = radii[i]
            strength = 40.0 * cooling

            if x - r < 0:
                forces[i][0] += strength * (r - x)
            if x + r > 1:
                forces[i][0] -= strength * (x + r - 1)
            if y - r < 0:
                forces[i][1] += strength * (r - y)
            if y + r > 1:
                forces[i][1] -= strength * (y + r - 1)

        step = 0.0025 * cooling
        centers += step * forces

        for i in range(n):
            centers[i][0] = np.clip(centers[i][0], radii[i], 1 - radii[i])
            centers[i][1] = np.clip(centers[i][1], radii[i], 1 - radii[i])

        if it < max_iter * 0.92:
            growth = 1.0 + 0.00025 * cooling
            radii = radii * growth

    return centers, radii


def refine(centers, radii, n):
    vars_init = np.zeros(3 * n)
    for i in range(n):
        vars_init[3 * i] = centers[i][0]
        vars_init[3 * i + 1] = centers[i][1]
        vars_init[3 * i + 2] = radii[i]

    obj_func = make_objective(n, penalty=80)

    result = minimize(
        obj_func,
        vars_init,
        method='Nelder-Mead',
        options={'maxiter': 8000, 'xatol': 1e-10, 'fatol': 1e-12}
    )

    centers = result.x[:2 * n].reshape((n, 2))
    radii = result.x[2 * n:]

    radii = np.maximum(radii, 0)

    for _ in range(300):
        for i in range(n):
            max_r = min(
                centers[i][0],
                1 - centers[i][0],
                centers[i][1],
                1 - centers[i][1]
            )
            for j in range(n):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            radii[i] = max(0, max_r)

    for i in range(n):
        centers[i][0] = np.clip(centers[i][0], radii[i], 1 - radii[i])
        centers[i][1] = np.clip(centers[i][1], radii[i], 1 - radii[i])

    return centers, radii


def make_objective(n, penalty):
    def obj(vars):
        return penalty_obj_impl(vars, n, penalty)
    return obj


def penalty_obj_impl(vars, n, penalty):
    centers = vars[:2 * n].reshape((n, 2))
    radii = vars[2 * n:]

    obj = -np.sum(radii)

    for i in range(n):
        x = centers[i][0]
        y = centers[i][1]
        r = radii[i]

        gap = x - r
        if gap < 0:
            obj += penalty * gap * gap

        gap = x + r - 1
        if gap > 0:
            obj += penalty * gap * gap

        gap = y - r
        if gap < 0:
            obj += penalty * gap * gap

        gap = y + r - 1
        if gap > 0:
            obj += penalty * gap * gap

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            gap = dist - min_dist
            if gap < 0:
                obj += penalty * gap * gap

    return obj


def validate_simple(centers, radii, n):
    for i in range(n):
        x = centers[i][0]
        y = centers[i][1]
        r = radii[i]

        if r < 0:
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True
