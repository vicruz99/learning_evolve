def init_hexagonal(n, seed=0):

    centers = []
    count = 0

    spacing = 0.115
    row = 0
    while count < n:
        y = 0.06 + row * spacing * np.sqrt(3) / 2
        if y > 0.94:
            break

        cols_in_row = n - count
        offset = spacing / 2 if row % 2 == 1 else 0
        x_start = 0.06 + offset

            if count >= n:

                count += 1
        row += 1

def force_optimize(centers, radii, n, max_iter):
    """Force-directed optimization to resolve overlaps and increase radii."""
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(max_iter):
        forces = np.zeros_like(centers)

        # Gradually increase radii
        if iteration < 20000:
            radii += 0.000003
        else:
            radii += 0.000001

        # Pairwise repulsion forces
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist_sq = dx * dx + dy * dy

                if dist_sq < 1e-24:
                    dist_sq = 1e-24
                    dx = 0.001
                    dy = 0.001

                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]

                if dist < min_dist:
                    overlap = min_dist - dist
                    nx = dx / dist
                    ny = dy / dist
                    f = overlap * 0.5

                    forces[i][0] -= nx * f
                    forces[i][1] -= ny * f
                    forces[j][0] += nx * f
                    forces[j][1] += ny * f

        # Boundary repulsion forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]

            if x - r < 0:
                forces[i][0] += max(0, r - x) * 15
            if x + r > 1:
                forces[i][0] -= max(0, x + r - 1) * 15
            if y - r < 0:
                forces[i][1] += max(0, r - y) * 15
            if y + r > 1:
                forces[i][1] -= max(0, y + r - 1) * 15

        # Apply forces with adaptive step size
        step = 0.15 / (1 + iteration * 0.00001)
        centers += forces * step

        # Clip to valid range
        centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
        centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)

        # Ensure non-negative radii
        radii = np.maximum(radii, 0.001)

    return centers, radii


def bfgs_finetune(centers, radii, n):
    """Fine-tune using BFGS optimization with penalty method."""
    x0 = np.concatenate([centers.flatten(), radii])

    def objective(x):
        r = x[2 * n:]
        return -np.sum(r)

    def penalty(x, w=5000.0):
        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        p = 0.0

        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i][0] - c[j][0]
                dy = c[i][1] - c[j][1]
                d = np.sqrt(dx * dx + dy * dy)
                m = r[i] + r[j]
                if d < m:
                    p += w * (m - d) ** 2

            x_c, y_c = c[i]
            ri = r[i]
            if x_c - ri < 0:
                p += w * (ri - x_c) ** 2
            if x_c + ri > 1:
                p += w * (x_c + ri - 1) ** 2
            if y_c - ri < 0:
                p += w * (ri - y_c) ** 2
            if y_c + ri > 1:
                p += w * (y_c + ri - 1) ** 2
            if ri < 0:
                p += w * ri ** 2

        return p

    def augmented(x):
        return objective(x) + penalty(x)

    result = minimize(augmented, x0, method='BFGS',
                      options={'maxiter': 50000, 'gtol': 1e-14, 'eps': 1e-10})

    x_opt = result.x
    c_opt = x_opt[:2 * n].reshape(n, 2)
    r_opt = x_opt[2 * n:]
    r_opt = np.maximum(r_opt, 0.001)

    return c_opt, r_opt
