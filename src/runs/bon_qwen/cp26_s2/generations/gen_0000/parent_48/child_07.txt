import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Uses Simulated Annealing to optimize centers and Linear Programming to optimize radii.
    """
    n = 26
    np.random.seed(42)

    # --- Step 1: Optimize Centers for Equal Radii ---
    # We treat all circles as having the same radius r and try to maximize r.
    # This is equivalent to maximizing the minimum distance between centers (and to boundaries).
    
    def objective_equal_r(points):
        # points shape: (n, 2)
        # Objective: Maximize r
        # Constraints: r <= dist(center_i, center_j) / 2, r <= dist(center_i, boundary)
        # We minimize -r.
        
        # Calculate distances between all pairs
        dists = np.linalg.norm(points[:, np.newaxis, :] - points[np.newaxis, :, :], axis=2)
        np.fill_diagonal(dists, np.inf) # Ignore self distance
        min_pair_dist = np.min(dists)
        
        # Distance to boundaries
        x = points[:, 0]
        y = points[:, 1]
        dist_to_bound_x = np.minimum(x, 1.0 - x)
        dist_to_bound_y = np.minimum(y, 1.0 - y)
        min_bound_dist = np.minimum(dist_to_bound_x, dist_to_bound_y)
        
        r = 0.5 * min(min_pair_dist, min_bound_dist)
        return -r # Minimize negative r

    # Initial guess: 5x5 grid plus one circle
    # 5x5 grid would be at 0.1, 0.3, 0.5, 0.7, 0.9
    centers_init = []
    for i in range(5):
        for j in range(5):
            centers_init.append([0.1 + i*0.2, 0.1 + j*0.2])
    # Add 26th circle in a corner
    centers_init.append([0.05, 0.05])
    
    # Reshape to (n, 2)
    centers_init = np.array(centers_init)
    # Flatten for optimizer
    x0 = centers_init.flatten()

    # Simulated Annealing to find good centers
    # Bounds: all coordinates in [0, 1]
    bounds = [(0, 1)] * (2 * n)
    
    # Use dual_annealing or basin_hopping
    # basin_hopping is often good for packing problems
    try:
        result = opt.basin_hopping(objective_equal_r, x0, minimizer_kwargs={"method": "L-BFGS-B", "bounds": bounds}, niter=500, stepsize=0.05)
        best_centers = result.x.reshape(n, 2)
    except Exception:
        # Fallback to simple gradient descent if basin_hopping fails
        result = opt.minimize(objective_equal_r, x0, method='L-BFGS-B', bounds=bounds)
        best_centers = result.x.reshape(n, 2)

    # --- Step 2: Optimize Radii using Linear Programming ---
    # Given fixed centers, maximize sum(r_i)
    # Constraints:
    # 1. r_i >= 0
    # 2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # 3. r_i + r_j <= dist(c_i, c_j) for all i < j
    
    # Variables: r_0, ..., r_25 (size 26)
    # Objective: max sum(r) => min -sum(r)
    c = -np.ones(n)
    
    # Inequality constraints: A_ub @ r <= b_ub
    
    # Constraints from boundaries
    # r_i <= x_i  =>  r_i <= x_i
    # r_i <= 1-x_i => r_i <= 1-x_i
    # Same for y
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        xi, yi = best_centers[i]
        
        # r_i <= xi
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(xi)
        
        # r_i <= 1 - xi
        A_ub.append(row.copy())
        b_ub.append(1.0 - xi)
        
        # r_i <= yi
        A_ub.append(row.copy())
        b_ub.append(yi)
        
        # r_i <= 1 - yi
        A_ub.append(row.copy())
        b_ub.append(1.0 - yi)
        
    # Constraints from pairwise distances
    # r_i + r_j <= dist_ij
    # This is r_i + r_j - dist_ij <= 0 ? No, we need A_ub @ r <= b
    # 1*r_i + 1*r_j <= dist_ij
    
    dist_matrix = np.linalg.norm(best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :], axis=2)
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = dist_matrix[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: r_i >= 0
    bounds_lp = [(0, None) for _ in range(n)]
    
    # Solve LP
    try:
        res_lp = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res_lp.success:
            radii = res_lp.x
        else:
            # Fallback: equal radii from Step 1
            # Estimate max equal radius
            dists = np.linalg.norm(best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :], axis=2)
            np.fill_diagonal(dists, np.inf)
            min_pair_dist = np.min(dists)
            min_bound_dist = np.min(np.minimum(best_centers, 1.0 - best_centers))
            r_equal = 0.5 * min(min_pair_dist, min_bound_dist)
            radii = np.full(n, r_equal)
    except Exception:
        # Fallback
        dists = np.linalg.norm(best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :], axis=2)
        np.fill_diagonal(dists, np.inf)
        min_pair_dist = np.min(dists)
        min_bound_dist = np.min(np.minimum(best_centers, 1.0 - best_centers))
        r_equal = 0.5 * min(min_pair_dist, min_bound_dist)
        radii = np.full(n, r_equal)

    # --- Final Validation and Adjustment ---
    # The LP ensures constraints are satisfied mathematically, but floating point errors might occur.
    # We clamp radii to be strictly within bounds if necessary, though LP should handle it.
    # Just ensure no negative radii.
    radii = np.maximum(radii, 1e-9)
    
    # Ensure centers are within bounds (they should be from Step 1)
    best_centers = np.clip(best_centers, 1e-9, 1.0 - 1e-9)

    sum_radii = np.sum(radii)
    
    return best_centers, radii, sum_radii
