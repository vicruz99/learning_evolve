

Looking at this circle packing problem, I need to maximize the sum of radii of 26 circles in a unit square. Let me think strategically about the approach.

<strategy>
1. **Initialization Strategy**: Start with circles arranged in a hexagonal pattern, which is known to be optimal for dense packing. With 26 circles, I'll use 5 rows with counts [5, 6, 5, 6, 4].

2. **Optimization Approach**: Use scipy.optimize.minimize with SLSQP method. The key insight is that this is a constrained optimization problem:
   - Maximize: sum of radii
   - Constraints: circles inside square, no overlaps, non-negative radii

3. **Variable Encoding**: Each circle has 3 variables (x, y, r), giving 78 total variables.

4. **Constraint Handling**: 
   - Boundary: x ± r ∈ [0,1], y ± r ∈ [0,1] for each circle
   - Overlap: distance(i,j) ≥ r_i + r_j for each pair

5. **Multiple Restarts**: Run from different initializations (hexagonal + random) to escape local optima.

6. **Post-processing**: After optimization, do a final cleanup to ensure validity and potentially grow radii further by computing maximum possible radius for each circle given positions.

7. **Key Improvement**: After getting positions from optimization, recompute optimal radii as min(min_distance_to_other_circle - r_j, min_distance_to_boundary), iterating until convergence. This often yields better results than optimizing positions and radii simultaneously.
</strategy>

import numpy as np
from scipy.optimize import minimize
import math

def initialize_hexagonal(n):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.015

    row_counts = [5, 6, 5, 6, 4]

    spacing_x = 1.0 / 7.5
    spacing_y = spacing_x * math.sqrt(3) / 2.0

    idx = 0
    for row in range(5):
        count = row_counts[row]
        y = row * spacing_y + 0.08
        for col in range(count):
            x = col * spacing_x + 0.05 + (0.05 if row % 2 == 1 else 0)
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def flatten(centers, radii):
    """Flatten centers and radii into a single array."""
    n = centers.shape[0]
    x = np.zeros(3 * n)
    for i in range(n):
        x[3 * i] = centers[i, 0]
        x[3 * i + 1] = centers[i, 1]
        x[3 * i + 2] = radii[i]
    return x


def unflatten(x, n):
    """Unflatten array into centers and radii."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = x[3 * i + 2]
    return centers, radii


def objective(x):
    """Objective: maximize sum of radii (minimize negative sum)."""
    radii = x[2::3]
    return -np.sum(radii)


def boundary_constraint(x, args):
    """Boundary constraints for circle i."""
    n, i, ct = args
    xi = x[3 * i]
    yi = x[3 * i + 1]
    ri = x[3 * i + 2]

    if ct == 0:
        return xi - ri
    elif ct == 1:
        return 1.0 - xi - ri
    elif ct == 2:
        return yi - ri
    elif ct == 3:
        return 1.0 - yi - ri


def overlap_constraint(x, args):
    """Overlap constraint for circles i and j."""
    n, i, j = args
    xi, yi = x[3 * i], x[3 * i + 1]
    xj, yj = x[3 * j], x[3 * j + 1]
    ri, rj = x[3 * i + 2], x[3 * j + 2]

    dx = xi - xj
    dy = yi - yj
    dist = math.sqrt(dx * dx + dy * dy)

    if dist < 1e-14:
        return 1e-14
    return dist - ri - rj


def get_constraints(n):
    """Get all constraints."""
    cons = []

    for i in range(n):
        for ct in range(4):
            cons.append({
                'type': 'ineq',
                'fun': boundary_constraint,
                'args': (n, i, ct)
            })

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',
                'fun': overlap_constraint,
                'args': (n, i, j)
            })

    return cons


def get_bounds(n):
    """Get bounds for variables."""
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
    return bounds


def compute_optimal_radii(centers, radii, n, iterations=200):
    """Given fixed positions, compute optimal radii iteratively."""
    for _ in range(iterations):
        for i in range(n):
            min_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    min_r = min(min_r, dist - radii[j])
            radii[i] = max(0.0, min_r)
    return radii


def grow_radii_after_optimization(centers, radii, n):
    """Try to grow radii after position optimization."""
    radii_new = radii.copy()
    for _ in range(500):
        improved = False
        for i in range(n):
            min_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    min_r = min(min_r, dist - radii_new[j])
            new_r = max(0.0, min_r)
            if new_r > radii_new[i] + 1e-12:
                radii_new[i] = new_r
                improved = True
        if not improved:
            break
    return radii_new


def run_packing():
    n = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    np.random.seed(42)

    # Try multiple initializations
    num_trials = 8

    for trial in range(num_trials):
        if trial == 0:
            centers, radii = initialize_hexagonal(n)
        elif trial == 1:
            # Another hexagonal with different spacing
            centers = np.zeros((n, 2))
            radii = np.ones(n) * 0.02
            row_counts = [6, 5, 6, 5, 4]
            spacing_x = 1.0 / 7.0
            spacing_y = spacing_x * math.sqrt(3) / 2.0
            idx = 0
            for row in range(5):
                count = row_counts[row]
                y = row * spacing_y + 0.1
                for col in range(count):
                    x = col * spacing_x + 0.06 + (0.06 if row % 2 == 1 else 0)
                    centers[idx] = [x, y]
                    idx += 1
        else:
            centers = np.random.rand(n, 2)
            radii = np.ones(n) * 0.012

        x0 = flatten(centers, radii)

        result = minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=get_bounds(n),
            constraints=get_constraints(n),
            options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
        )

        x_opt = result.x
        centers_opt, radii_opt = unflatten(x_opt, n)
        radii_opt = np.maximum(radii_opt, 0.0)

        # Post-optimization: grow radii
        radii_opt = grow_radii_after_optimization(centers_opt, radii_opt, n)

        # Clip to ensure feasibility
        for i in range(n):
            max_r = min(
                centers_opt[i, 0],
                1.0 - centers_opt[i, 0],
                centers_opt[i, 1],
                1.0 - centers_opt[i, 1]
            )
            radii_opt[i] = min(radii_opt[i], max_r)

        current_sum = np.sum(radii_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Final verification and cleanup
    # Ensure no overlaps by shrinking if needed
    for i in range(n):
        min_r = min(
            best_centers[i, 0],
            1.0 - best_centers[i, 0],
            best_centers[i, 1],
            1.0 - best_centers[i, 1]
        )
        for j in range(n):
            if i != j:
                dx = best_centers[i, 0] - best_centers[j, 0]
                dy = best_centers[i, 1] - best_centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                required = best_radii[i] + best_radii[j]
                if required > dist + 1e-12:
                    shrink = (required - dist) / 2.0 + 1e-10
                    best_radii[i] = max(0.0, best_radii[i] - shrink)

    # Recompute sum after cleanup
    final_sum = np.sum(best_radii)

    return best_centers, best_radii, final_sum
