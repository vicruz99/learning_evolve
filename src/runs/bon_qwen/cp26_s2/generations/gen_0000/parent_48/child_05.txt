def get_constraints(n, r_idx):
    """
    Helper to generate constraint dictionaries for scipy.optimize.minimize.
    This avoids defining lambdas inside loops with closure issues.
    """
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    # x_i - r >= 0
    for i in range(n):
        idx_x = 2 * i
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[r_idx]
        })
        # 1 - x_i - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[idx_x] - v[r_idx]
        })
        # y_i - r >= 0
        idx_y = 2 * i + 1
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[r_idx]
        })
        # 1 - y_i - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[idx_y] - v[r_idx]
        })

    # Pairwise separation constraints: dist_ij >= 2r
    # dist^2 >= (2r)^2  =>  (xi-xj)^2 + (yi-yj)^2 - 4r^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = 2 * i
            idx_yi = 2 * i + 1
            idx_xj = 2 * j
            idx_yj = 2 * j + 1
            
            # Capture i and j in lambda
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: \
                    (v[idx_xi] - v[idx_xj])**2 + (v[idx_yi] - v[idx_yj])**2 - 4 * (v[r_idx]**2)
            })
            
    return constraints
