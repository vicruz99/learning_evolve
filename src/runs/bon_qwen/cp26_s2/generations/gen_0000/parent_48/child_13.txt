import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def compute_constraints(v):
    """
    Compute boundary and non-overlap constraints.
    Returns an array of values that must be >= 0.
    """
    n = 26
    c = v.reshape(n, 3)
    x = c[:, 0]
    y = c[:, 1]
    r = c[:, 2]

    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    bound_cons = np.concatenate([
        x - r, 
        1.0 - x - r, 
        y - r, 
        1.0 - y - r
    ])

    # Overlap constraints: ||c_i - c_j|| >= r_i + r_j for i < j
    # Vectorized pairwise distance calculation
    diff = c[:, np.newaxis, :2] - c[np.newaxis, :, :2]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-14)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]

    # Extract upper triangle indices (i < j)
    i_idx, j_idx = np.triu_indices(n, k=1)
    overlap_cons = dist[i_idx, j_idx] - r_sum[i_idx, j_idx]

    return np.concatenate([bound_cons, overlap_cons])

def run_packing():
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_v = None
    best_val = -np.inf
    np.random.seed(42)

    # Bounds for variables: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': compute_constraints}

    for trial in range(8):
        # 1. Initialize with hexagonal lattice + jitter
        centers = []
        step = 0.135
        for row in range(10):
            for col in range(10):
                cx = col * step + (row % 2) * step / 2.0
                cy = row * step * np.sqrt(3) / 2.0
                if 0 <= cx <= 1 and 0 <= cy <= 1:
                    centers.append([cx, cy])
                    if len(centers) == n:
                        break
            if len(centers) == n:
                break
        
        centers = np.array(centers)
        # Add controlled random perturbation
        centers += np.random.uniform(-0.015, 0.015, centers.shape)
        # Clip to keep strictly inside initially
        centers = np.clip(centers, 0.12, 0.88)

        # Flatten to optimization vector [x0, y0, r0, x1, y1, r1, ...]
        v0 = np.zeros(3 * n)
        for i in range(n):
            v0[3*i] = centers[i, 0]
            v0[3*i+1] = centers[i, 1]
            v0[3*i+2] = 0.035  # Small initial radius ensuring feasibility

        # 2. Run SLSQP optimization
        try:
            res = minimize(
                objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
            )
            if res.success:
                # Verify constraint satisfaction with tolerance
                cons_val = compute_constraints(res.x)
                if np.all(cons_val >= -1e-7):
                    current_val = -res.fun
                    if current_val > best_val:
                        best_val = current_val
                        best_v = res.x.copy()
        except Exception:
            continue

    # 3. Fallback to valid grid packing if optimization fails
    if best_v is None:
        fallback_centers = []
        for j in range(5):
            for i in range(5):
                fallback_centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
        fallback_centers.append([0.5, 0.05])
        fallback_r = np.full(26, 0.09)
        fallback_v = np.zeros(78)
        for i in range(26):
            fallback_v[3*i] = fallback_centers[i][0]
            fallback_v[3*i+1] = fallback_centers[i][1]
            fallback_v[3*i+2] = fallback_r[i]
        best_v = fallback_v

    # 4. Post-processing to guarantee strict validity
    centers_out = np.column_stack((best_v[0::3], best_v[1::3]))
    radii_out = best_v[2::3].copy()
    
    cons_val = compute_constraints(best_v)
    min_violation = np.min(cons_val)
    if min_violation < -1e-9:

        radii_out = np.maximum(radii_out - shrink_amount, 0.0)
        best_v[2::3] = radii_out

    sum_radii = np.sum(radii_out)
    return centers_out, radii_out, sum_radii
