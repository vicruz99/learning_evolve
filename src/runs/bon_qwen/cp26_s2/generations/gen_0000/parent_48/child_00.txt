import numpy as np
from scipy.optimize import minimize


def compute_objective(vars):
    """Compute objective: negative sum of radii plus penalty for constraint violations."""
    n = len(vars) // 3
    c = vars[:2 * n].reshape(n, 2)
    r = vars[2 * n:]

    obj = -np.sum(r)
    penalty = 0.0

    # Boundary constraint penalties
    for i in range(n):
        penalty += 500.0 * max(0.0, r[i] - c[i, 0]) ** 2
        penalty += 500.0 * max(0.0, r[i] - (1.0 - c[i, 0])) ** 2
        penalty += 500.0 * max(0.0, r[i] - c[i, 1]) ** 2
        penalty += 500.0 * max(0.0, r[i] - (1.0 - c[i, 1])) ** 2
        penalty += 500.0 * max(0.0, -r[i]) ** 2

    # Non-overlap constraint penalties
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((c[i] - c[j]) ** 2) + 1e-12)
            penalty += 500.0 * max(0.0, r[i] + r[j] - d) ** 2

    return obj + penalty


def fix_violations(centers, radii, n):
    """Fix boundary and overlap violations."""
    # Fix boundary violations
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], max(max_r - 1e-10, 1e-10))

    # Fix overlap violations iteratively
    for _ in range(500):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if d < radii[i] + radii[j]:
                    overlap = (radii[i] + radii[j] - d) / 2.0
                    radii[i] = max(radii[i] - overlap, 1e-10)
                    radii[j] = max(radii[j] - overlap, 1e-10)
                    changed = True
        if not changed:
            break

    return centers, radii


def grow_radii(centers, radii, n):
    """Grow each circle to fill available space."""
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                max_r = min(max_r, d - radii[j])
        radii[i] = max(max_r - 1e-10, 1e-10)
    return radii


def repel_overlaps(centers, radii, n, lr):
    """Repel overlapping circles."""
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[j] - centers[i]
            d = np.sqrt(np.sum(diff ** 2))
            if d < radii[i] + radii[j] and d > 1e-10:
                overlap = radii[i] + radii[j] - d
                force = overlap * lr / d
                centers[i] -= force * diff
                centers[j] += force * diff
    return centers


def clamp_to_boundary(centers, radii, n):
    """Clamp centers to stay within boundary given radii."""
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-10, 1.0 - radii[i] - 1e-10)
        centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-10, 1.0 - radii[i] - 1e-10)
    return centers


def initial_placement(n):
    """Create initial hexagonal-inspired placement."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)

    idx = 0
    pattern = [6, 5, 6, 5, 4]
    for row in range(5):
        cols = pattern[row]
        for col in range(cols):
            if idx < n:
                x = 0.07 + col * 0.148 + (row % 2) * 0.074
                y = 0.07 + row * 0.185
                centers[idx] = [x, y]
                idx += 1

    return centers, radii


def run_packing():
    n = 26

    # Phase 1: Initial placement and force-directed pre-optimization
    centers, radii = initial_placement(n)

    # Force-directed relaxation
    for iteration in range(3000):
        lr = 0.05 * (1.0 - iteration / 3000.0)

        # Grow radii
        radii = grow_radii(centers, radii, n)

        # Repel overlaps
        centers = repel_overlaps(centers, radii, n, lr)

        # Clamp to boundary
        centers = clamp_to_boundary(centers, radii, n)

    # Phase 2: scipy optimization
    x0 = np.concatenate([centers.flatten(), radii])

    best_obj = np.inf
    best_x = x0.copy()

    for restart in range(5):
        x0_try = x0.copy()
        if restart > 0:
            x0_try[:2 * n] += np.random.randn(2 * n) * 0.008
            x0_try[2 * n:] += np.random.randn(n) * 0.003

        result = minimize(
            compute_objective,
            x0_try,
            method="L-BFGS-B",
            bounds=[(0.0, 1.0)] * (2 * n) + [(0.01, 0.3)] * n,
            options={"maxiter": 20000, "ftol": 1e-15}
        )

        if result.fun < best_obj:
            best_obj = result.fun
            best_x = result.x.copy()

    c_opt = best_x[:2 * n].reshape(n, 2)
    r_opt = best_x[2 * n:]

    # Phase 3: Fix violations and final refinement
    c_opt, r_opt = fix_violations(c_opt, r_opt, n)

    # Phase 4: Another round of force-directed refinement
    for iteration in range(2000):
        lr = 0.03 * (1.0 - iteration / 2000.0)

        radii = grow_radii(c_opt, r_opt, n)
        c_opt = repel_overlaps(c_opt, r_opt, n, lr)
        c_opt = clamp_to_boundary(c_opt, r_opt, n)

    c_opt, r_opt = fix_violations(c_opt, r_opt, n)

    # Phase 5: Final scipy optimization on refined solution
    x0_final = np.concatenate([c_opt.flatten(), r_opt])

    best_obj2 = np.inf
    best_x2 = x0_final.copy()

    for restart in range(3):
        x0_try = x0_final.copy()
        if restart > 0:
            x0_try[:2 * n] += np.random.randn(2 * n) * 0.005
            x0_try[2 * n:] += np.random.randn(n) * 0.002

        result = minimize(
            compute_objective,
            x0_try,
            method="L-BFGS-B",
            bounds=[(0.0, 1.0)] * (2 * n) + [(0.01, 0.3)] * n,
            options={"maxiter": 25000, "ftol": 1e-15}
        )

        if result.fun < best_obj2:
            best_obj2 = result.fun
            best_x2 = result.x.copy()

    c_opt = best_x2[:2 * n].reshape(n, 2)
    r_opt = best_x2[2 * n:]

    c_opt, r_opt = fix_violations(c_opt, r_opt, n)

    # Final radius growth pass
    r_opt = grow_radii(c_opt, r_opt, n)
    c_opt, r_opt = fix_violations(c_opt, r_opt, n)

    total = np.sum(r_opt)

    return c_opt, r_opt, total
