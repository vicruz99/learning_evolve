from scipy.optimize import minimize

def calculate_objective_and_constraints(x, n=26):
    """
    Calculates the objective (sum of radii) and constraint violations.
    x is a vector of shape (n * 3) containing [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]

    # Objective: Maximize sum of radii (minimize negative sum)
    obj = -np.sum(radii)

    # Constraints are handled as penalty terms in the objective for simplicity 
    # or via explicit constraints. Here we use penalty method for stability 
    # in the optimization function structure, but strictly we should use bounds.
    # However, scipy minimize with constraints is better.
    return obj, centers, radii

def objective(params, n=26):
    """Objective function to minimize: -sum(radii) + penalties"""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Penalty for negative radii (should be covered by bounds, but safe to add)
    penalty = np.sum(np.maximum(0, -radii)**2)
    
    # Penalty for boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0  => r - x <= 0
        penalty += np.maximum(0, r - x)**2
        # x + r <= 1  => x + r - 1 <= 0
        penalty += np.maximum(0, x + r - 1)**2
        # y - r >= 0
        penalty += np.maximum(0, r - y)**2
        # y + r <= 1
        penalty += np.maximum(0, y + r - 1)**2

    # Penalty for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist)**2 * 10.0 # Strong penalty for overlap

    return -np.sum(radii) + penalty

def get_constraints(n=26):
    """Define explicit constraints for scipy optimizer"""
    cons = []
    
    # Boundary constraints: 0 <= x - r, 1 >= x + r, etc.
    # x - r >= 0  -> g(x) >= 0
    # We can define these as constraints, but with 26 circles it's 104 constraints.
    # SLSQP handles this well.
    for i in range(n):
        # x_i - r_i >= 0
        def c_xr(i=i):
            # params shape is 3n
            # x_i is at index 3*i, r_i is at 3*i + 2
            return lambda p: p[3*i] - p[3*i+2]
        cons.append({'type': 'ineq', 'fun': c_xr()})
        
        # 1 - x_i - r_i >= 0
        def c_1xr(i=i):
            return lambda p: 1.0 - p[3*i] - p[3*i+2]
        cons.append({'type': 'ineq', 'fun': c_1xr()})
        
        # y_i - r_i >= 0
        def c_yr(i=i):
            return lambda p: p[3*i+1] - p[3*i+2]
        cons.append({'type': 'ineq', 'fun': c_yr()})
        
        # 1 - y_i - r_i >= 0
        def c_1yr(i=i):
            return lambda p: 1.0 - p[3*i+1] - p[3*i+2]
        cons.append({'type': 'ineq', 'fun': c_1yr()})

    # Overlap constraints: dist_ij >= r_i + r_j
    # dist_ij^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            def c_dist(i=i, j=j):
                def fun(p):
                    xi, yi = p[3*i], p[3*i+1]
                    ri = p[3*i+2]
                    xj, yj = p[3*j], p[3*j+1]
                    rj = p[3*j+2]
                    
                    dx = xi - xj
                    dy = yi - yj
                    dist_sq = dx*dx + dy*dy
                    r_sum = ri + rj
                    return dist_sq - r_sum*r_sum
                return fun
            cons.append({'type': 'ineq', 'fun': c_dist()})
            
    return cons
