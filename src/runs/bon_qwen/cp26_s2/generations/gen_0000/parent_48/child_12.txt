import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def objective(x):
    """Negative sum of radii (we minimize this to maximize sum)."""
    return -np.sum(x[2 * N_CIRCLES:])


def constraint_func(x):
    """Returns all constraints as a vector that must be >= 0."""
    centers = x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = x[2 * N_CIRCLES:]

    constraints = []

    # Boundary constraints: center ± radius in [0, 1]
    constraints.append(centers[:, 0] - radii)
    constraints.append(1 - centers[:, 0] - radii)
    constraints.append(centers[:, 1] - radii)
    constraints.append(1 - centers[:, 1] - radii)

    # Non-overlap constraints using vectorized pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs ** 2, axis=2))
    ri_plus_rj = radii[:, np.newaxis] + radii[np.newaxis, :]

    i, j = np.triu_indices(N_CIRCLES, k=1)
    constraints.append(dists[i, j] - ri_plus_rj[i, j])

    # Non-negative radii
    constraints.append(radii)

    return np.concatenate(constraints)


def local_refine(centers, radii, n):
    """Force-based local refinement to escape local minima."""
    for iteration in range(2000):
        forces = np.zeros((n, 2))

        # Repulsion between overlapping/nearby circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-10:
                    dist = 1e-10
                    diff = np.random.randn(2) * 1e-5
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    force = (min_dist - dist) / dist * diff
                    forces[i] += force
                    forces[j] -= force

        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                forces[i, 0] += max(0, r - x) * 10
            if x + r > 1:
                forces[i, 0] -= max(0, x + r - 1) * 10
            if y - r < 0:
                forces[i, 1] += max(0, r - y) * 10
            if y + r > 1:
                forces[i, 1] -= max(0, y + r - 1) * 10

        # Update centers with decaying step size
        step_size = 0.005 / (1 + iteration * 0.0005)
        centers += forces * step_size

        # Clip to valid range
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

        # Try to grow radii slightly
        if iteration % 5 == 0:
            for i in range(n):
                max_r = min(
                    centers[i, 0],
                    1 - centers[i, 0],
                    centers[i, 1],
                    1 - centers[i, 1],
                )
                for j in range(n):
                    if i != j:
                        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                        max_r = min(max_r, dist - radii[j])
                radii[i] = max(radii[i] * 1.0005, max_r - 1e-10)

    return centers, radii


def run_packing():
    n = N_CIRCLES

    best_sum = 0
    best_centers = None
    best_radii = None

    num_trials = 150

    for trial in range(num_trials):
        np.random.seed(trial * 17 + 3)

        # Different initialization patterns for diversity
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.04

        pattern = trial % 5

        if pattern == 0:
            # Hexagonal grid with offset rows
            idx = 0
            for row in range(5):
                offset = (row % 2) * 0.5
                for col in range(6):
                    if idx >= n:
                        break
                    centers[idx, 0] = (col + 0.5 + offset) / 6 + np.random.randn() * 0.01
                    centers[idx, 1] = (row + 0.5) / 5 + np.random.randn() * 0.01
                    centers[idx] = np.clip(centers[idx], 0.05, 0.95)
                    idx += 1
            radii[:] = 0.04

        elif pattern == 1:
            # Random initialization
            centers = np.random.rand(n, 2) * 0.7 + 0.15
            radii[:] = 0.03 + np.random.rand(n) * 0.02

        elif pattern == 2:
            # Corner-heavy initialization
            idx = 0
            corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
            for corner in corners:
                if idx < n:
                    centers[idx] = corner + np.random.randn(2) * 0.02
                    radii[idx] = 0.06
                    idx += 1
            for i in range(idx, n):
                centers[i] = np.random.rand(2) * 0.5 + 0.25
                radii[i] = 0.04

        elif pattern == 3:
            # Perturbed rectangular grid
            for i in range(n):
                centers[i, 0] = (i % 6 + 0.5) / 6 + np.random.randn() * 0.02
                centers[i, 1] = (i // 6 + 0.5) / 5 + np.random.randn() * 0.02
                centers[i] = np.clip(centers[i], 0.05, 0.95)
            radii[:] = 0.04

        else:
            # Dense center cluster with edge circles
            for i in range(n):
                if i < 8:
                    # Edge circles
                    angle = i * 2 * np.pi / 8
                    r = 0.45
                    centers[i, 0] = 0.5 + r * np.cos(angle)
                    centers[i, 1] = 0.5 + r * np.sin(angle)
                    centers[i] = np.clip(centers[i], 0.1, 0.9)
                    radii[i] = 0.04

                    centers[i] = np.random.rand(2) * 0.4 + 0.3
                    radii[i] = 0.03

        x0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0, 1) for _ in range(2 * n)] + [(0, 0.5) for _ in range(n)]

        cons = {"type": "ineq", "fun": constraint_func}

        result = minimize(
            objective,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 5000, "ftol": 1e-15, "disp": False},

        radii_opt = result.x[2 * n:]

        centers_opt, radii_opt = local_refine(centers_opt.copy(), radii_opt.copy(), n)

        # Second SLSQP pass after refinement
        x0_refined = np.concatenate([centers_opt.flatten(), radii_opt])
        result2 = minimize(
            objective,
            x0_refined,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 3000, "ftol": 1e-15, "disp": False},
        )

        centers_opt = result2.x[:2 * n].reshape(n, 2)
        radii_opt = result2.x[2 * n:]

        # Validate and update best
        if np.all(radii_opt >= -1e-10):
            # Check boundary constraints
            valid = True
            for i in range(n):
                x, y = centers_opt[i]
                r = radii_opt[i]
                if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                    valid = False
                    break

            if valid:
                # Check overlaps
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j]) ** 2))
                        if dist < radii_opt[i] + radii_opt[j] - 1e-8:
                            valid = False
                            break
                    if not valid:
                        break

            if valid:
                s = radii_opt.sum()
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()

    # Fallback if nothing worked
    if best_centers is None:
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.048
        for i in range(n):
            centers[i, 0] = (i % 6 + 0.5) / 6
            centers[i, 1] = (i // 6 + 0.5) / 5
        best_sum = radii.sum()
        best_centers = centers
        best_radii = radii

    return best_centers, best_radii, best_sum
