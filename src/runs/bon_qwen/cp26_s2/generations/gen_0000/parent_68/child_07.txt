    *   **Non-overlap:** The distance between the centers of any two circles must be at least the sum of their radii.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Helper to create a grid-based initial guess
    def create_initial_guess(perturbation=0.0):
        coords = []
        radii = []
        
        # 5x5 grid for 25 circles
        step = 0.2
        start = 0.1
        for i in range(5):
            for j in range(5):
                x = start + i * step
                y = start + j * step
                # Add perturbation
                if perturbation > 0:
                    x += np.random.uniform(-perturbation, perturbation)
                    y += np.random.uniform(-perturbation, perturbation)
                coords.append([x, y])
                radii.append(0.1)
        
        # 26th circle in a gap, e.g., at (0.2, 0.2)
        # Distance to nearest grid points (0.1,0.1), etc is sqrt(0.02) ~ 0.141
        # With r=0.1, max gap radius is ~0.041. Start with 0.04.
        x_26, y_26 = 0.2, 0.2
        if perturbation > 0:
            x_26 += np.random.uniform(-perturbation, perturbation)
            y_26 += np.random.uniform(-perturbation, perturbation)
        coords.append([x_26, y_26])
        radii.append(0.04)
        
        # Flatten to optimization vector: [x0, y0, r0, x1, y1, r1, ...]
        x0 = []
        for i in range(n):
            x0.extend([coords[i][0], coords[i][1], radii[i]])
        
        return np.array(x0)

    # Define objective function: minimize negative sum of radii
    def objective(v):
        r = v[2::3]
        return -np.sum(r)

    # Define constraints
    constraints = []
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, same for y
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[idx_x] - v[idx_r]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[idx_y] - v[idx_r]
        })

    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2
            
            def dist_constraint(v, i=i, j=j):
                dx = v[idx_xi] - v[idx_xj]
                dy = v[idx_yi] - v[idx_yj]
                dr = v[idx_ri] + v[idx_rj]
                return (dx**2 + dy**2) - (dr**2)
            
            constraints.append({
                'type': 'ineq',
                'fun': dist_constraint
            })

    # Bounds for variables
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])

    best_sum = -1.0
    best_result = None

    # Run optimization multiple times with different seeds/perturbations
    # to increase chance of finding a better local optimum
    num_runs = 5
    
    for run in range(num_runs):
        # Use a small perturbation for subsequent runs to escape symmetry
        perturbation = 0.01 if run > 0 else 0.0
        x0 = create_initial_guess(perturbation=perturbation)
        
        try:
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
            )
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_result = res
        except Exception:
            continue

    if best_result is None:
        # Fallback to initial guess if optimization fails
        x0 = create_initial_guess()
        centers = np.array([[x0[3*i], x0[3*i+1]] for i in range(n)])
        radii = np.array([x0[3*i+2] for i in range(n)])
        return centers, radii, np.sum(radii)

    final_v = best_result.x
    centers = np.array([[final_v[3*i], final_v[3*i+1]] for i in range(n)])
    radii = np.array([final_v[3*i+2] for i in range(n)])
    final_sum = np.sum(radii)

    return centers, radii, final_sum
