import numpy as np
from scipy.optimize import minimize

# Problem constants
N_CIRCLES = 26

def objective(vars):
    """
    Objective function to minimize.
    We want to maximize sum of radii, so we minimize -sum(radii).
    vars format: [x0, y0, r0, x1, y1, r1, ...]
    """
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_func(vars):
    """
    Computes all inequality constraints g(vars) >= 0.
    Returns a flattened numpy array of constraint values.
    """
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    cons = []
    
    # 1. Boundary Constraints
    # x >= r  => x - r >= 0
    cons.append(x - r)
    # x + r <= 1 => 1 - x - r >= 0
    cons.append(1.0 - x - r)
    # y >= r  => y - r >= 0
    cons.append(y - r)
    # y + r <= 1 => 1 - y - r >= 0
    cons.append(1.0 - y - r)
    
    # 2. Non-overlap Constraints
    # dist(i, j)^2 >= (r_i + r_j)^2
    # dist_sq - r_sum_sq >= 0
    
    # Vectorized computation of pairwise differences
    # Shape (n, n)
    x_diff = x[:, np.newaxis] - x[np.newaxis, :]
    y_diff = y[:, np.newaxis] - y[np.newaxis, :]
    
    dist_sq = x_diff**2 + y_diff**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint value for each pair
    overlap_constr = dist_sq - r_sum_sq
    
    # We only need to enforce constraints for i < j (upper triangle, excluding diagonal)
    # Use triu_indices to extract strictly upper triangular elements
    i_idx, j_idx = np.triu_indices(n, k=1)
    cons.append(overlap_constr[i_idx, j_idx])
    
    # Concatenate all constraint vectors into a single array
    return np.concatenate(cons)
