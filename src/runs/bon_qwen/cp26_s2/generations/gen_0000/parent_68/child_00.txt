import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    """
    Solves the Linear Programming problem to find the maximum radii for fixed centers.
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum of radii => minimize -1 * sum(r)
    c_obj = np.ones(n) * -1.0

    # Inequality constraints matrix A_ub @ r <= b_ub
    # 1. Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    #    We handle this via 'bounds' parameter, but also strictly enforce in A_ub if needed.
    #    Actually, bounds are easier.
    
    bounds_list = []
    for i in range(n):
        x, y = centers[i]
        # Radius cannot exceed distance to any of the 4 walls
        max_r = min(x, 1 - x, y, 1 - y)
        # Ensure non-negative
        max_r = max(0.0, max_r)
        bounds_list.append((0.0, max_r))

    # 2. Pairwise distance constraints: r_i + r_j <= dist(c_i, c_j)
    #    Matrix form: A_ub @ r <= b_ub
    rows = []
    dists = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            dists.append(dist)
    
    if rows:
        A_ub = np.array(rows)
        b_ub = np.array(dists)
    else:
        A_ub = np.zeros((0, n))
        b_ub = np.zeros(0)

    # Solve LP
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_list, method='highs')
    
    if res.success:
        radii = res.x
        return radii, -res.fun
    else:
        # Fallback to zero radii if infeasible (should not happen with r=0)
        return np.zeros(n), 0.0

def get_sum_radii(centers_flat):
    """Objective function for optimization."""
    centers = centers_flat.reshape(-1, 2)
    _, sum_r = solve_radii(centers)
    return -sum_r

def generate_hex_grid(n):
    """Generate a hexagonal grid of n points."""
    points = []
    row = 0
    while len(points) < n:
        y = 0.1 + row * 0.2
        if y > 0.9:
            break
        offset = 0.1 if row % 2 == 1 else 0.0
        col = 0
        while True:
            x = 0.1 + col * 0.2 + offset
            if x > 0.9:
                break
            points.append([x, y])
            if len(points) >= n:
                break
            col += 1
        row += 1
    return np.array(points[:n])

def run_packing():
    np.random.seed(42)
    best_centers = None
    best_radii = None
    max_sum = 0.0

    # Strategy 1: Hexagonal Grid
    for _ in range(2):
        centers = generate_hex_grid(26)
        # Add small random noise
        centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
        # Clip to safe bounds
        centers = np.clip(centers, 0.05, 0.95)
        
        res = minimize(get_sum_radii, centers.flatten(), method='Nelder-Mead', 
                       options={'maxiter': 2000, 'xatol': 1e-5, 'fatol': 1e-5})
        if -res.fun > max_sum:
            max_sum = -res.fun
            best_centers = res.x.reshape(26, 2)

    # Strategy 2: Random Initializations
    for _ in range(5):
        centers = np.random.uniform(0.1, 0.9, size=(26, 2))
        res = minimize(get_sum_radii, centers.flatten(), method='Nelder-Mead', 
                       options={'maxiter': 2000, 'xatol': 1e-5, 'fatol': 1e-5})
        if -res.fun > max_sum:
            max_sum = -res.fun
            best_centers = res.x.reshape(26, 2)

    # Final radius calculation for best centers
    final_radii, final_sum = solve_radii(best_centers)
    
    return best_centers, final_radii, final_sum
