import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: 5x5 grid for first 25 circles, 1 in center
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.1) # Initial guess for grid circles
    
    # Place 25 circles in a 5x5 grid
    idx = 0
    for r in range(5):
        for c in range(5):
            centers[idx, 0] = 0.1 + c * 0.2
            centers[idx, 1] = 0.1 + r * 0.2
            idx += 1
            
    # Place 26th circle in the center with a small radius to start
    centers[25, 0] = 0.5
    centers[25, 1] = 0.5
    radii[25] = 0.001
    
    # 2. Pre-optimization: Force-directed relaxation
    # We try to expand radii and resolve overlaps using simple forces
    dt = 0.01
    expansion_rate = 0.0005
    
    for step in range(500):
        forces = np.zeros_like(centers)
        
        # Increase radii slightly
        radii = np.maximum(radii * (1 + expansion_rate), 0)
        
        for i in range(n):
            # Boundary repulsion
            # Left
            if centers[i, 0] < radii[i]:
                forces[i, 0] += (radii[i] - centers[i, 0]) * 10
            # Right
            if centers[i, 0] > 1 - radii[i]:
                forces[i, 0] -= (centers[i, 0] - (1 - radii[i])) * 10
            # Bottom
            if centers[i, 1] < radii[i]:
                forces[i, 1] += (radii[i] - centers[i, 1]) * 10
            # Top
            if centers[i, 1] > 1 - radii[i]:
                forces[i, 1] -= (centers[i, 1] - (1 - radii[i])) * 10
                
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-6:
                    # Repulsive force proportional to overlap
                    overlap = min_dist - dist
                    fx = (dx / dist) * overlap * 5.0
                    fy = (dy / dist) * overlap * 5.0
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    # Shrink radii slightly if overlap is severe to help convergence
                    if overlap > 0.01:
                        radii[i] *= 0.99
                        radii[j] *= 0.99
                    else:
                        radii[i] += 0.0001
                        radii[j] += 0.0001
        
        centers += forces * dt

    # Combine centers and radii into a single vector
    x0 = np.hstack((centers.flatten(), radii))

        # Return negative sum of radii for minimization
        r_vec = x[2*n:]
        return -np.sum(r_vec)

    constraints = []
    
    # Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    for i in range(n):
        x_idx = 2 * i
        y_idx = 2 * i + 1
        r_idx = 2 * n + i
        
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[x_idx] - x[r_idx]})
        # 1 - (x + r) >= 0 => x + r <= 1
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[x_idx] - x[r_idx]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[y_idx] - x[r_idx]})
        # 1 - (y + r) >= 0 => y + r <= 1
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[y_idx] - x[r_idx]})
        
    # Non-negative radii
    for i in range(n):
        r_idx = 2 * n + i
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[r_idx]})

    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # We use squared distance to avoid sqrt in constraint function for stability
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi = 2 * i, 2 * i + 1
            xj, yj = 2 * j, 2 * j + 1
            ri, rj = 2 * n + i, 2 * n + j
            
            def dist_sq_ge_sum_r_sq(x, i=i, j=j):
                dx = x[xi] - x[xj]
                dy = x[yi] - x[yj]
                dist_sq = dx*dx + dy*dy
                sum_r = x[ri] + x[rj]
                return dist_sq - sum_r*sum_r
            
            constraints.append({'type': 'ineq', 'fun': dist_sq_ge_sum_r_sq})

    # Bounds
    bounds = []
    for i in range(2 * n): # x and y coordinates
        bounds.append((0, 1))
    for i in range(n): # radii
        bounds.append((0, None))

    # Run optimization
    res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-9})

    final_centers = final_x[:2*n].reshape((n, 2))
    final_radii = final_x[2*n:]

    final_radii = np.maximum(final_radii, 1e-6)

    # If any circle is out of bounds, shrink radius
    for i in range(n):

        r = final_radii[i]
        max_r = min(x, 1-x, y, 1-y)
        if max_r < 1e-6:
            max_r = 1e-6
        if r > max_r:
            final_radii[i] = max_r

    return final_centers, final_radii, np.sum(final_radii)
