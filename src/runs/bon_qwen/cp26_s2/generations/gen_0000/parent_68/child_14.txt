import numpy as np
from scipy.optimize import minimize


def run_packing():

def generate_hex_init(n, seed):
    np.random.seed(seed * 7919 + 13)

    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)

    idx = 0
    row = 0
    dx = 0.175
    dy = dx * np.sqrt(3) / 2

    while idx < n:
        offset = dx / 2 if row % 2 == 1 else 0
        n_cols = 6 if row % 2 == 0 else 5
        for col in range(n_cols):
            if idx >= n:
                break
            centers[idx, 0] = 0.12 + col * dx + offset + np.random.uniform(-0.015, 0.015)
            centers[idx, 1] = 0.10 + row * dy + np.random.uniform(-0.015, 0.015)
            idx += 1
        row += 1

    centers[:, 0] -= np.mean(centers[:, 0]) - 0.5
    centers[:, 1] -= np.mean(centers[:, 1]) - 0.5
    centers = np.clip(centers, 0.08, 0.92)

    return centers, radii


def local_optimize(n, centers_init, radii_init):
    x0 = np.concatenate([centers_init.flatten(), radii_init])

    result_bfgs = minimize(penalty_objective, x0, args=(n,), method='BFGS',
                           options={'maxiter': 80000, 'gtol': 1e-12})

    centers = result_bfgs.x[:2 * n].reshape((n, 2))
    radii = result_bfgs.x[2 * n:]

    x0_nm = result_bfgs.x.copy()
    result_nm = minimize(penalty_objective, x0_nm, args=(n,), method='Nelder-Mead',
                         options={'maxiter': 200000, 'xatol': 1e-10, 'fatol': 1e-12})

    centers_nm = result_nm.x[:2 * n].reshape((n, 2))
    radii_nm = result_nm.x[2 * n:]

    if -result_nm.fun < -result_bfgs.fun:
        return centers_nm, radii_nm
    return centers, radii


def penalty_objective(x, n):
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]

    obj = -np.sum(radii)
    penalty = 0.0
    w = 5000.0

    for i in range(n):
        x_i = centers[i, 0]
        y_i = centers[i, 1]
        r_i = max(0.0, radii[i])

        v = r_i - x_i
        if v > 0:
            penalty += w * v * v
        v = x_i + r_i - 1.0
        if v > 0:
            penalty += w * v * v
        v = r_i - y_i
        if v > 0:
            penalty += w * v * v
        v = y_i + r_i - 1.0
        if v > 0:
            penalty += w * v * v

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = max(0.0, radii[i]) + max(0.0, radii[j])
            v = min_dist - dist
            if v > 0:
                penalty += w * v * v

    return obj + penalty


def is_packing_valid(centers, radii, n):
    for i in range(n):
        if radii[i] < -1e-10:
            return False
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True


def make_simple_valid(n):
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.04)
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            centers[idx, 0] = 0.12 + col * 0.16
            centers[idx, 1] = 0.12 + row * 0.16
            idx += 1
    return centers, radii
