import numpy as np

def solve_radii_lp(centers):
    """
    Given fixed centers, find the maximum sum of radii using Linear Programming.
    Variables: r_1, ..., r_n
    Maximize: sum(r_i)
    Constraints:
      1. r_i >= 0
      2. r_i <= dist_to_boundary(center_i)
      3. r_i + r_j <= dist(center_i, center_j) for all i < j
    """
    n = centers.shape[0]
    
    # Coefficients for objective function: maximize sum(r) -> minimize -sum(r)
    c = -np.ones(n)
    
    # Inequality constraints matrix A_ub * x <= b_ub
    # We collect all constraints here
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    # Also r_i >= 0 handled by bounds
    
    # Distance constraints: r_i + r_j <= d_ij
    # This is r_i + r_j <= d_ij  =>  r_i + r_j - d_ij <= 0 ? 
    # Actually linprog solves min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds.
    # Constraint: r_i + r_j <= d_ij
    # So row in A_ub has 1 at i, 1 at j. b_ub has d_ij.
    
    A_ub = []
    b_ub = []
    
    # Precompute distances
    # dist_matrix[i, j]
    dist_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist_matrix[i, j] = np.linalg.norm(centers[i] - centers[j])
            dist_matrix[j, i] = dist_matrix[i, j]
            
    # Boundary constraints
    # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # This can be handled by bounds directly: (0, limit_i)
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        limit = min(x, 1.0 - x, y, 1.0 - y)
        if limit < 0:
            limit = 0
        bounds.append((0.0, limit))
        
    # Pairwise constraints
    # r_i + r_j <= dist_matrix[i, j]
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist_matrix[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Solve LP
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback if LP fails (unlikely with valid centers)
            return np.zeros(n)
    except Exception:
        return np.zeros(n)
