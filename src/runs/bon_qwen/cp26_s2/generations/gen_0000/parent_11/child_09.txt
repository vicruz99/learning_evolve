import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # 1. Initialization: Staggered Hexagonal Grid
    # ---------------------------------------------------------
    # Approximate hex packing: area density ~ 0.9069
    # N * pi * r^2 approx 0.9 -> r approx sqrt(0.9 / (26 * pi)) ~ 0.105
    # We use a slightly smaller radius to ensure fit, then optimize.
    
    centers = []
    r_init = 0.1
    # Row spacing for hex is sqrt(3)*r, but we will scale later.
    # Using relative coordinates first.
    row_spacing = np.sqrt(3) * r_init
    col_spacing = 2 * r_init
    
    rows = 6
    for r in range(rows):
        # Odd rows shifted by r_init
        offset = r_init if r % 2 == 1 else 0
        y = r * row_spacing + r_init
        
        # Calculate max x for this row
        # We want to fit as many as possible, but exactly 26.
        # Let's just generate points in a grid and pick the best 26 
        # or just a standard pattern.

        if r == 0: count = 5
        elif r == 1: count = 6
        elif r == 2: count = 5
        elif r == 3: count = 6
        elif r == 4: count = 4
        else: count = 0
        
        for c in range(count):
            x = c * col_spacing + offset + r_init
            centers.append([x, y])
            
    centers = np.array(centers)

    # Calculate bounding box of current centers
    min_x, min_y = centers.min(axis=0)
    max_x, max_y = centers.max(axis=0)

    # But we will let the optimizer handle boundaries.
    # Just ensure they are in [0,1]
    if max_x > 0: centers[:, 0] -= min_x
    if max_y > 0: centers[:, 1] -= min_y
    
    # Scale to fit in unit square
    if max_x > 1: centers[:, 0] /= (max_x * 1.1) # slight buffer
    if max_y > 1: centers[:, 1] /= (max_y * 1.1)
    
    # Randomize slightly to break symmetry
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)

    # 2. Optimization Loop
    # ---------------------------------------------------------
    radii = np.ones(n) * 0.01

    pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]

    best_centers = centers.copy()
    best_radii = radii.copy()
    
    # LP Iterations
    for iteration in range(150):
        
        # --- Phase A: Solve LP for Radii ---
        # Maximize sum(r)
        # Constraints:
        # r_i <= x_i (Left)
        # r_i <= 1-x_i (Right)
        # r_i <= y_i (Bottom)
        # r_i <= 1-y_i (Top)
        # r_i + r_j <= dist(c_i, c_j)

        # Inequality constraints: A_ub @ r <= b_ub
        # 4 wall constraints per circle + pairwise constraints
        A_ub = []
        b_ub = []
        
        for i in range(n):
            # Left: 1*r_i <= x_i
            row_left = np.zeros(n)
            row_left[i] = 1.0
            A_ub.append(row_left)
            b_ub.append(centers[i, 0])
            
            # Right: 1*r_i <= 1 - x_i

            row_right[i] = 1.0
            A_ub.append(row_right)
            b_ub.append(1.0 - centers[i, 0])
            
            # Bottom: 1*r_i <= y_i
            row_bot = np.zeros(n)
            row_bot[i] = 1.0
            A_ub.append(row_bot)
            b_ub.append(centers[i, 1])

            A_ub.append(row_top)

                    row_pair[i] = 1.0

                # Apply forces
                centers += forces
