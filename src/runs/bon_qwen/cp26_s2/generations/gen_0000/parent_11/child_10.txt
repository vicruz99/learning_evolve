import numpy as np
from scipy.optimize import minimize


def compute_penalty(centers, radii, n, weight=1000.0):
    """Compute penalty for constraint violations."""
    penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        penalty += weight * max(0, r - x) ** 2
        penalty += weight * max(0, r - (1 - x)) ** 2
        penalty += weight * max(0, r - y) ** 2
        penalty += weight * max(0, r - (1 - y)) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            penalty += weight * max(0, radii[i] + radii[j] - dist) ** 2
    
    return penalty


def objective(params, n, weight=1000.0):
    """Objective: maximize sum of radii (minimize negative sum) plus penalties."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    return -np.sum(radii) + compute_penalty(centers, radii, n, weight)


def make_valid(centers, radii, n):

def create_hexagonal_layout(n, rows_config, r0):

    centers = np.zeros((n, 2))
    radii = np.ones(n) * r0
    dx = 2.0 * r0
    dy = np.sqrt(3.0) * r0
    
    idx = 0
    for row in range(len(rows_config)):
        nc = rows_config[row]
        y = r0 + row * dy
        off = r0 if row % 2 == 1 else 0.0
        
        if nc > 1:
            total_width = (nc - 1) * dx
            x_start = (1.0 - total_width) / 2.0 + r0 + off
        else:
            x_start = 0.5
        
        for col in range(nc):
            centers[idx, 0] = x_start + col * dx
            centers[idx, 1] = y
            idx += 1
    
    return centers, radii


def optimize_from_config(n, centers0, radii0, weight=2000.0, maxiter=30000):
    """Optimize packing from an initial configuration."""
    params = np.concatenate([centers0.ravel(), radii0])
    bounds = [(0.01, 0.99) for _ in range(2 * n)] + [(0.001, 0.5) for _ in range(n)]
    
    result = minimize(
        objective,
        params,
        args=(n, weight),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16}
    )
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(result.x[2 * n:], 1e-10)
    
    return centers_opt, radii_opt
