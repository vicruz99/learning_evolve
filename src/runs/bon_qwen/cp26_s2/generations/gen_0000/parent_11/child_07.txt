import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def solve_radii(centers):
    """
    Solve the linear program to find maximum radii for given centers.
    Maximize sum(r_i) subject to r_i + r_j <= dist(i,j) and boundary constraints.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Minimize negative sum = Maximize sum
    
    n_bound = 4 * n
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_bound + n_pairs, n))
    b_ub = np.zeros(n_bound + n_pairs)
    
    idx = 0
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        b_vals = [x, 1-x, y, 1-y]
        for b in b_vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = b
            idx += 1
            
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x if res.success else np.zeros(n)

def objective(x):
    """Objective function for optimization: negative sum of optimal radii."""
    centers = x.reshape(-1, 2)
    radii = solve_radii(centers)
    return -np.sum(radii)

def run_packing():
    # 26 circles => 52 coordinates
    bounds = [(0.0, 1.0)] * 52
    
    # Global optimization with Differential Evolution
    de_result = differential_evolution(
        objective, 
        bounds, 
        popsize=20, 
        maxiter=300, 
        seed=42, 
        polish=False, 
        tol=1e-7
    )
    
    # Local refinement with Nelder-Mead
    nm_result = minimize(
        objective, 
        de_result.x, 
        method='Nelder-Mead', 
        options={'xatol': 1e-4, 'fatol': 1e-6}
    )
    
    best_centers = nm_result.x.reshape(-1, 2)
    # Slight inward clipping to avoid numerical boundary issues during validation
    best_centers = np.clip(best_centers, 1e-5, 1 - 1e-5)
    
    best_radii = solve_radii(best_centers)
    return best_centers, best_radii, float(np.sum(best_radii))
