import numpy as np
from scipy.optimize import minimize

def get_valid_radii(centers, n):
    """
    Computes the maximal valid radii for fixed centers by iteratively clipping.
    Ensures circles stay within boundaries and do not overlap.
    """
    # Initialize radii based on distance to boundaries
    r = np.array([min(c[0], 1-c[0], c[1], 1-c[1]) for c in centers])
    
    # Iteratively enforce non-overlap constraints (Gauss-Seidel style)
    for _ in range(200):
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], 
                        centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                max_r = min(max_r, dist - r[j])
            if max_r < r[i]:
                r[i] = max_r
    return np.maximum(r, 0.0)

def packing_objective(vars, n, penalty_weight):
    """
    Objective function for optimization: maximizes sum of radii with penalty for violations.
    """
    cs = vars[:2*n].reshape(n, 2)
    rs = vars[2*n:]
    pen = 0.0
    
    # Boundary constraint penalties
    for i in range(n):
        x, y = cs[i]
        r = rs[i]
        dist_b = min(x, 1-x, y, 1-y)
        if r > dist_b:
            pen += (r - dist_b)**2
            
    # Overlap constraint penalties
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt(np.sum((cs[i]-cs[j])**2))
            if d < rs[i] + rs[j]:
                pen += (rs[i] + rs[j] - d)**2
                
    return -np.sum(rs) + penalty_weight * pen

def run_packing():
    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    # Try multiple restarts to avoid local minima
    for seed in range(5):
        np.random.seed(seed)
        
        # Hexagonal-like initialization
        counts = [4, 5, 4, 5, 4, 4]
        centers = []
        y = 0.12 + np.random.uniform(-0.02, 0.02)
        dy = 0.145 + np.random.uniform(-0.01, 0.01)
        
        for c in counts:
            x_start = (1.0 - (c-1)*0.18) / 2.0 + np.random.uniform(-0.02, 0.02)
            dx = 0.18 + np.random.uniform(-0.01, 0.01)
            for k in range(c):
                cx = x_start + k * dx
                cy = y
                centers.append([np.clip(cx, 0.05, 0.95), np.clip(cy, 0.05, 0.95)])
            y += dy
            
        centers = np.array(centers[:n])
        radii = np.full(n, 0.01)
        x0 = np.hstack([centers.flatten(), radii])
        bounds = [(0.0, 1.0)]*(2*n) + [(0.001, 0.5)]*n
        
        try:
            res = minimize(packing_objective, x0, args=(n, 1000.0), 
                           method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 10000, 'ftol': 1e-12})
            
            c_temp = res.x[:2*n].reshape(n, 2)
            r_temp = get_valid_radii(c_temp, n)
            s_temp = np.sum(r_temp)
            
            if s_temp > best_sum_r:
                best_sum_r = s_temp
                best_centers = c_temp
                best_radii = r_temp
        except Exception:
            continue
            
    # Fallback in case of complete failure
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        best_sum_r = 0.0
        
    return best_centers, best_radii, best_sum_r
