### Strategy
1.  **Hexagonal Initialization:** We generate centers for 26 circles using a hexagonal lattice. Rows are staggered by half a horizontal spacing to maximize density.
2.  **Scaling and Optimization:** We start with a conservative radius and use a numerical optimizer (`scipy.optimize.minimize`) to maximize the sum of radii. The solver adjusts the centers to maintain the required separation distance while pushing the circles as large as possible against the boundaries and each other.
3.  **Constraint Enforcement:** We enforce non-overlap ($d_{ij} \ge r_i + r_j$) and boundary conditions ($r_i \le x_i \le 1-r_i$) using inequality constraints.

The code below implements this strategy using a multi-start optimization to avoid local minima, ensuring a robust and high-quality packing.

```python
import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    n_circles = 26
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None

    def objective(params):
        # Maximize sum of radii by minimizing negative sum
        return -np.sum(params[2 * n_circles : 3 * n_circles])

    def constraints_factory(centers, radii):
        constr = []
        # Boundary constraints: 0 <= x - r and r <= x <= 1 - r etc.
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            constr.append({'type': 'ineq', 'fun': lambda p, idx=i: p[idx] - p[2*n_circles + idx]})
            constr.append({'type': 'ineq', 'fun': lambda p, idx=i: (1.0 - p[idx]) - p[2*n_circles + idx]})
            constr.append({'type': 'ineq', 'fun': lambda p, idx=i: p[n_circles + idx] - p[2*n_circles + idx]})
            constr.append({'type': 'ineq', 'fun': lambda p, idx=i: (1.0 - p[n_circles + idx]) - p[2*n_circles + idx]})
            
        # Non-overlap constraints: distance >= r_i + r_j
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                xi, yi = centers[i]
                xj, yj = centers[j]
                ri, rj = radii[i], radii[j]
                
                def dist_constraint(p, i=i, j=j, xi=xi, yi=yi, xj=xj, yj=yj):
                    pi_x, pi_y = p[i], p[n_circles + i]
                    pj_x, pj_y = p[j], p[n_circles + j]
                    ri, rj = p[2*n_circles + i], p[2*n_circles + j]
                    dx, dy = pi_x - pj_x, pi_y - pj_y
                    dist = math.sqrt(dx**2 + dy**2)
                    return dist - (ri + rj)
                
                constr.append({'type': 'ineq', 'fun': dist_constraint})
        return constr

    # Generate initial hexagonal grid
    base_centers = []
    r_init = 0.095
    h_sp = 2 * r_init
    v_sp = math.sqrt(3) * r_init
    count = 0
    
    row = 0
    while count < n_circles and row < 6:
        cols = 6 if row % 2 == 0 else 5
        if row > 3:
            cols = 5
        if row > 4:
            cols = 4
            
        for c in range(cols):
            if count >= n_circles:
                break
            x = r_init + c * h_sp
            y = r_init + row * v_sp + (0.5 * h_sp if row % 2 != 0 else 0)
            if x + r_init <= 1.0 and y + r_init <= 1.0:
                base_centers.append([x, y])
                count += 1
        row += 1
    
    init_r = np.full(n_circles, r_init)
    
    # Run optimization from multiple starts to ensure robustness
    for trial in range(5):
        # Perturb initial conditions
        centers = np.array(base_centers) + np.random.uniform(-0.01, 0.01, (n_circles, 2))
        radii = np.full(n_circles, 0.090)
        
        # Clip to valid range
        centers = np.clip(centers, 0.05, 0.95)
        
        x0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
        
        # Bounds
        bnds = [(0, 1)] * (2 * n_circles) + [(0, 0.2)] * n_circles
        
        res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=constraints_factory(centers, radii), 
                       options={'maxiter': 500, 'disp': False})
        
        if res.success or res.fun < best_sum_radii:
            c_opt = res.x[:2 * n_circles].reshape(n_circles, 2)
            r_opt = res.x[2 * n_circles:]
            
            # Final validation check
            valid = True
            for i in range(n_circles):
                x, y = c_opt[i]
                r = r_opt[i]
                if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                    valid = False
                    break
            
            if valid and -res.fun > best_sum_radii:
                best_sum_radii = -res.fun
                best_centers = c_opt
                best_radii = r_opt

    # Sort radii descending to ensure consistent output
    if best_radii is not None:
        order = np.argsort(-best_radii)
        best_centers = best_centers[order]
        best_radii = best_radii[order]

    return best_centers, best_radii, float(np.sum(best_radii))
```
