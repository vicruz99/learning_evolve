import numpy as np
import math


def compute_distance(x1, y1, x2, y2):
    dx = x2 - x1
    dy = y2 - y1
    return math.sqrt(dx * dx + dy * dy)


def create_hexagonal_lattice(n_circles, spacing=0.18):
    centers = []
    y = spacing + 0.05
    row = 0
    while len(centers) < n_circles:
        x = spacing + 0.05
        if row % 2 == 1:
            x += spacing * math.sqrt(3) / 2
        while x <= 1 - spacing - 0.05 and len(centers) < n_circles:
            centers.append([x, y])
            x += spacing
        y += spacing * math.sqrt(3) / 2
        row += 1
    return np.array(centers[:n_circles])


def force_based_optimize(centers, radii, n_circles, n_iterations=40000):
    for iteration in range(n_iterations):
        forces_x = np.zeros(n_circles)
        forces_y = np.zeros(n_circles)

        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist_sq = dx * dx + dy * dy
                if dist_sq < 1e-12:
                    dist = 1e-6
                    dx = 1e-6
                    dy = 0
                else:
                    dist = math.sqrt(dist_sq)

                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    force = overlap * 150.0
                    fx = dx / dist * force
                    fy = dy / dist * force
                    forces_x[i] -= fx
                    forces_y[i] -= fy
                    forces_x[j] += fx
                    forces_y[j] += fy

        for i in range(n_circles):
            if centers[i][0] - radii[i] < 0:
                forces_x[i] += (radii[i] - centers[i][0]) * 300.0
            if centers[i][0] + radii[i] > 1:
                forces_x[i] -= (centers[i][0] + radii[i] - 1) * 300.0
            if centers[i][1] - radii[i] < 0:
                forces_y[i] += (radii[i] - centers[i][1]) * 300.0
            if centers[i][1] + radii[i] > 1:
                forces_y[i] -= (centers[i][1] + radii[i] - 1) * 300.0

        lr = 0.008 / (1 + iteration * 0.000015)
        centers[:, 0] += forces_x * lr
        centers[:, 1] += forces_y * lr

        expansion = 0.00004 / (1 + iteration * 0.000008)
        radii += expansion

    return centers, radii


def post_process(centers, radii, n_circles):
    for i in range(n_circles):
        max_r = min(
            centers[i][0], 1.0 - centers[i][0],
            centers[i][1], 1.0 - centers[i][1]
        )
        radii[i] = min(radii[i], max(0.0, max_r))
        radii[i] = max(radii[i], 0.0)
        centers[i][0] = np.clip(centers[i][0], radii[i] + 1e-9, 1.0 - radii[i] - 1e-9)
        centers[i][1] = np.clip(centers[i][1], radii[i] + 1e-9, 1.0 - radii[i] - 1e-9)

    for iteration in range(3000):
        any_overlap = False
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist = compute_distance(
                    centers[i][0], centers[i][1],
                    centers[j][0], centers[j][1]
                )
                if dist < radii[i] + radii[j] - 1e-12:
                    any_overlap = True
                    reduction = (radii[i] + radii[j] - dist) / 2.0 + 1e-6
                    radii[i] = max(0.0, radii[i] - reduction)
                    radii[j] = max(0.0, radii[j] - reduction)
        if not any_overlap:
            break

    for iteration in range(500):
        improved = False
        for i in range(n_circles):
            max_r = min(
                centers[i][0], 1.0 - centers[i][0],
                centers[i][1], 1.0 - centers[i][1]
            )
            for j in range(n_circles):
                if i != j:
                    dx = centers[j][0] - centers[i][0]
                    dy = centers[j][1] - centers[i][1]
                    dist = compute_distance(
                        centers[i][0], centers[i][1],
                        centers[j][0], centers[j][1]
                    )
                    max_r = min(max_r, dist - radii[j])
            max_r = max(0.0, max_r)
            if max_r > radii[i] + 1e-8:
                radii[i] = max_r
                improved = True
        if not improved:
            break

    return centers, radii


def run_packing():
    n_circles = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    for restart in range(15):
        np.random.seed(restart * 7 + 3)

        if restart < 8:
            spacing = 0.16 + restart * 0.005
            centers = create_hexagonal_lattice(n_circles, spacing)
            initial_radius = 0.07 + restart * 0.003
        else:
            centers = np.random.uniform(0.1, 0.9, size=(n_circles, 2))
            initial_radius = 0.06

        radii = np.full(n_circles, initial_radius)

        centers, radii = force_based_optimize(
            centers.copy(), radii.copy(), n_circles, n_iterations=50000
        )

        centers, radii = post_process(centers, radii, n_circles)

        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    return best_centers, best_radii, best_sum
