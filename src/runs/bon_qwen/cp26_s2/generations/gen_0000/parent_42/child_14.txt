import numpy as np
from scipy.optimize import minimize

def compute_penalty_and_objective(x, n):
    """Compute objective (negative sum of radii) and penalty for constraint violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    obj = -np.sum(radii)
    penalty = 0.0
    penalty_coeff = 5000.0
    
    # Boundary constraints: each circle must be inside [0,1]x[0,1]
    for i in range(n):
        cx = centers[i, 0]
        cy = centers[i, 1]
        r = radii[i]
        
        # x - r >= 0
        viol = r - cx
        if viol > 0:
            penalty += viol ** 2
        
        # x + r <= 1
        viol = cx + r - 1
        if viol > 0:
            penalty += viol ** 2
        
        # y - r >= 0
        viol = r - cy
        if viol > 0:
            penalty += viol ** 2
        
        # y + r <= 1
        viol = cy + r - 1
        if viol > 0:
            penalty += viol ** 2
    
    # Non-overlap constraints: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            
            viol = radii[i] + radii[j] - dist
            if viol > 0:
                penalty += viol ** 2
    
    return obj + penalty_coeff * penalty

def initialize_hexagonal_grid(n):
    """Initialize circle centers on a hexagonal grid pattern."""
    centers = np.zeros((n, 2))
    
    idx = 0
    row = 0
    
    while idx < n:
        # Determine number of circles in this row
        if row % 2 == 0:
            cols_in_row = 5
        else:
            cols_in_row = 5
        
        for col in range(cols_in_row):
            if idx >= n:
                break
            
            # Base grid positions
            x = (col + 0.5) / 5.0
            y = (row + 0.5) / 5.0
            
            # Shift odd rows for hexagonal packing
            if row % 2 == 1:
                x += 1.0 / 10.0
            
            centers[idx] = [x, y]
            idx += 1
        
        row += 1
    
    return centers

def initialize_random_perturbed(n, seed):
    """Initialize with a hexagonal grid plus random perturbation."""
    np.random.seed(seed)
    centers = initialize_hexagonal_grid(n)
    
    # Add small random perturbations
    noise = np.random.randn(n, 2) * 0.03
    centers = centers + noise
    centers = np.clip(centers, 0.15, 0.85)
    
    return centers

def create_bounds(n):
    """Create bounds for the optimization variables."""
    bounds = []
    for i in range(2 * n):
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((0.0, 0.5))
    return bounds

def create_initial_radii(n, seed):
    """Create initial radii values."""
    np.random.seed(seed + 1000)
    # Start with a reasonable radius that allows packing
    base_radius = 0.06
    variation = 0.02
    radii = np.full(n, base_radius) + np.random.rand(n) * variation
    return radii

def refine_radii(centers, radii, n):
    """After optimization, refine radii to be maximal given positions."""
    refined_radii = radii.copy()
    
    for iteration in range(50):
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            )
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    max_r = min(max_r, dist - refined_radii[j])
            refined_radii[i] = max(0, max_r)
    
    return refined_radii

def validate_internal(centers, radii, n):
    """Internal validation to check if solution is feasible."""
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        
        if r < 0:
            return False
        
        if cx - r < -1e-10 or cx + r > 1 + 1e-10:
            return False
        if cy - r < -1e-10 or cy + r > 1 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True

def run_single_optimization(n, seed):
    """Run a single optimization from a given seed."""
    centers = initialize_random_perturbed(n, seed)
    radii = create_initial_radii(n, seed)
    
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = create_bounds(n)
    
    def objective_wrapper(x):
        return compute_penalty_and_objective(x, n)
    
    result = minimize(
        objective_wrapper,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 10000, 'ftol': 1e-15, 'gtol': 1e-10}
    )
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    
    # Ensure non-negative radii
    radii_opt = np.maximum(radii_opt, 0)
    centers_opt = np.clip(centers_opt, 0, 1)
    
    # Refine radii
    radii_opt = refine_radii(centers_opt, radii_opt, n)
    
    # Clip centers again after refinement
    for i in range(n):
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], radii_opt[i], 1 - radii_opt[i])
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], radii_opt[i], 1 - radii_opt[i])
    
    return centers_opt, radii_opt

def run_packing():
    """Main function to pack 26 circles in a unit square."""
    n = 26
    
    best_sum = -1
    best_centers = None
    best_radii = None
    
    # Try multiple seeds for diversity
    seeds = list(range(20))
    
    for seed in seeds:
        centers, radii = run_single_optimization(n, seed)
        
        if validate_internal(centers, radii, n):
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
    
    # If no valid solution found, create a safe fallback
    if best_centers is None:
        centers = initialize_hexagonal_grid(n)
        radii = np.full(n, 0.03)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)
    
    return best_centers, best_radii, best_sum
