import numpy as np
from scipy.optimize import minimize, linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Function to solve for optimal radii given centers
    # Returns the sum of radii. 
    # If infeasible (should not happen with r=0), returns a very small number.
    def get_max_radius_sum(centers):
        # centers shape (n, 2)
        
        # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
        c_obj = -np.ones(n)
        
        # Constraints: A_ub @ r <= b_ub
        # 1. Pairwise non-overlap: r_i + r_j <= dist(c_i, c_j)
        # 2. Boundary constraints are handled by variable bounds
        
        A_ub = []
        b_ub = []
        
        # Precompute distances to walls for bounds
        bounds = []
        for i in range(n):
            x, y = centers[i]
            # Distance to nearest wall
            dist_wall = min(x, 1.0 - x, y, 1.0 - y)
            # Radius must be >= 0 and <= dist_wall
            # Clip dist_wall to be non-negative just in case
            dist_wall = max(0.0, dist_wall)
            bounds.append((0.0, dist_wall))
            
        # Pairwise constraints
        # For each pair (i, j), r_i + r_j <= dist_ij
        # This corresponds to a row in A_ub with 1 at i, 1 at j
        for i in range(n):
            for j in range(i + 1, n):
                dist_ij = np.sqrt(np.sum((centers[i] - centers[j])**2))
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist_ij)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Solve LP
        # Using 'highs' method which is efficient
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                # The value of objective function is -sum(radii)
                # So sum(radii) = -res.fun
                return -res.fun, res.x

                # If LP fails, return 0 and zeros
                return 0.0, np.zeros(n)

            return 0.0, np.zeros(n)

        s, radii = get_max_radius_sum(centers)
