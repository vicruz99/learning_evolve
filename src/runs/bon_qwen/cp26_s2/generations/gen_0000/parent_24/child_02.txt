import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n, pen_weight):
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Objective: maximize sum of radii => minimize negative sum
    loss = -np.sum(radii)
    
    # Boundary penalties: x >= r, x <= 1-r, y >= r, y <= 1-r
    # Violations are penalized quadratically
    loss += pen_weight * np.sum(np.maximum(0, radii - centers[:, 0])**2)
    loss += pen_weight * np.sum(np.maximum(0, centers[:, 0] + radii - 1.0)**2)
    loss += pen_weight * np.sum(np.maximum(0, radii - centers[:, 1])**2)
    loss += pen_weight * np.sum(np.maximum(0, centers[:, 1] + radii - 1.0)**2)
    
    # Pairwise non-overlap penalties: dist(i,j) >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # Ignore self-distances
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = r_sum - dists
    loss += pen_weight * np.sum(np.maximum(overlap, 0.0)**2)
    
    return loss
