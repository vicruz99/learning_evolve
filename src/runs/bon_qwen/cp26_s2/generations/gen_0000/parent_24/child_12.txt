import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def obj_func(p):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(p[2*N_CIRCLES:])

def obj_jac(p):
    """Jacobian of the objective function."""
    jac = np.zeros(3*N_CIRCLES)
    jac[2*N_CIRCLES:] = -1.0
    return jac

def con_x_min(p, n):
    """x_i - r_i >= 0"""
    return p[0:2*n:2] - p[2*n:]

def con_x_max(p, n):
    """1 - x_i - r_i >= 0"""
    return 1.0 - p[0:2*n:2] - p[2*n:]

def con_y_min(p, n):
    """y_i - r_i >= 0"""
    return p[1:2*n:2] - p[2*n:]

def con_y_max(p, n):
    """1 - y_i - r_i >= 0"""
    return 1.0 - p[1:2*n:2] - p[2*n:]

def con_r_min(p, n):
    """r_i >= 0.005 (prevents degenerate circles)"""
    return p[2*n:] - 0.005

def con_sep(p, n):
    """Non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0"""
    centers = p[:2*n].reshape(-1, 2)
    radii = p[2*n:]
    diff = centers[:, None, :] - centers[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    rad_sum = radii[:, None] + radii[None, :]
    mask = np.tril(np.ones((n, n), dtype=bool), -1)
    return dist_sq[mask] - rad_sum[mask]**2

def jac_sep(p, n):
    """Jacobian of non-overlap constraints."""
    centers = p[:2*n].reshape(-1, 2)
    radii = p[2*n:]
    n_con = n * (n - 1) // 2
    jac = np.zeros((n_con, 3*n))
    idx = 0
    for i in range(n):
        xi, yi = 2*i, 2*i+1
        ri = 2*n + i
        for j in range(i+1, n):
            xj, yj = 2*j, 2*j+1
            rj = 2*n + j
            
            di_x = centers[i, 0] - centers[j, 0]
            di_y = centers[i, 1] - centers[j, 1]
            ri_rj = radii[i] + radii[j]
            
            jac[idx, xi] = 2.0 * di_x
            jac[idx, yi] = 2.0 * di_y
            jac[idx, xj] = -2.0 * di_x
            jac[idx, yj] = -2.0 * di_y
            jac[idx, ri] = -2.0 * ri_rj
            jac[idx, rj] = -2.0 * ri_rj
            idx += 1
    return jac

def run_packing():
    n = N_CIRCLES
    
    constraints = [
        {'type': 'ineq', 'fun': con_x_min, 'args': (n,)},
        {'type': 'ineq', 'fun': con_x_max, 'args': (n,)},
        {'type': 'ineq', 'fun': con_y_min, 'args': (n,)},
        {'type': 'ineq', 'fun': con_y_max, 'args': (n,)},
        {'type': 'ineq', 'fun': con_r_min, 'args': (n,)},
        {'type': 'ineq', 'fun': con_sep, 'jac': jac_sep, 'args': (n,)}
    ]
    
    best_sum = -1.0
    best_p = None
    
    # Run multiple restarts to escape local optima
    for seed in range(30):
        rng = np.random.default_rng(seed)
        p = np.zeros(3*n)
        
        # Structured initial guess (perturbed grid)
        gx = np.linspace(0.15, 0.85, 6)
        gy = np.linspace(0.15, 0.85, 5)
        grid = np.array([[x, y] for y in gy for x in gx])
        p[:2*n] = grid[:n, :].ravel()
        
        # Add controlled random perturbation
        p[:2*n] += rng.uniform(-0.03, 0.03, size=2*n)
        p[:2*n] = np.clip(p[:2*n], 0.08, 0.92)
        
        # Small initial radii to ensure feasibility
        p[2*n:] = 0.025
        
        res = minimize(
            obj_func, 
            p, 
            method='SLSQP', 
            constraints=constraints,
            jac=obj_jac,
            options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False}
        )
        
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_p = res.x.copy()
            
    centers = best_p[:2*n].reshape(-1, 2)
    radii = best_p[2*n:]
    return centers, radii, np.sum(radii)
