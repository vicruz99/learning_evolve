def get_coords(r, u, v):
    """
    Computes x, y coordinates from radius r and auxiliary variables u, v.
    x = r + (1 - 2*r)*u
    y = r + (1 - 2*r)*v
    This maps u, v in [0,1] to the valid range [r, 1-r] for x, y.
    """
    # Handle edge cases where 1-2r might be slightly negative due to numerical issues
    # though bounds should prevent r > 0.5
    scale = 1.0 - 2.0 * r
    x = r + scale * u
    y = r + scale * v
    return x, y

    xs = np.zeros(n)

    rs = np.zeros(n)

        u = z[3*i]

        ys[i] = y
        rs[i] = r
