import numpy as np
from scipy.optimize import minimize

def generate_hex_init(n, seed):
    np.random.seed(seed)
    r_est = 0.07
    pts = []
    y = r_est
    row = 0
    while y <= 1.0 - r_est:
        x = r_est if row % 2 == 0 else 2 * r_est
        while x <= 1.0 - r_est:
            pts.append([x, y])
            x += 2 * r_est
        y += np.sqrt(3) * r_est
        row += 1
        
    pts = np.array(pts[:n])
    if pts.shape[0] < n:
        extra = np.random.uniform(0.1, 0.9, (n - pts.shape[0], 2))
        pts = np.vstack([pts, extra])
        
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    pts = np.clip(pts, 0.01, 0.99)
    r_init = np.full(n, 0.05)
    return np.concatenate([pts.flatten(), r_init])

def compute_penalty(var, P):
    n = 26
    centers = var[:2*n].reshape(n, 2)
    radii = var[2*n:]
    
    obj = -np.sum(radii)
    pen = 0.0
    
    v_x1 = np.maximum(0, radii - centers[:, 0])
    v_x2 = np.maximum(0, radii - (1 - centers[:, 0]))
    v_y1 = np.maximum(0, radii - centers[:, 1])
    v_y2 = np.maximum(0, radii - (1 - centers[:, 1]))
    pen += np.sum((v_x1 + v_x2 + v_y1 + v_y2)**2)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d2 = np.sum(diff**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    viol = r_sum**2 - d2
    
    iu, ju = np.triu_indices(n, k=1)
    pen += np.sum(np.maximum(0, viol[iu, ju])**2)
    
    return obj + P * pen

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for s in range(25):
        x0 = generate_hex_init(n, s)
        
        P = 10.0
        curr = x0
        for _ in range(8):
            curr += np.random.uniform(-1e-5, 1e-5, curr.shape)
            res = minimize(compute_penalty, curr, args=(P,), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-13})
            curr = res.x
            P *= 10.0
            
        c = curr[:2*n].reshape(n, 2)
        r = curr[2*n:]
        
        valid = True
        if np.any(r - c[:,0] > 1e-8) or np.any(r - c[:,1] > 1e-8) or \
           np.any(r - (1-c[:,0]) > 1e-8) or np.any(r - (1-c[:,1]) > 1e-8):
            valid = False
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt(np.sum((c[i]-c[j])**2))
                if d < r[i] + r[j] - 1e-8:
                    valid = False
                    break
            if not valid: break
            
        if valid:
            s_r = np.sum(r)
            if s_r > best_sum:
                best_sum = s_r
                best_centers = c.copy()
                best_radii = r.copy()
                
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        best_sum = 0.0
        
    return best_centers, best_radii, best_sum
