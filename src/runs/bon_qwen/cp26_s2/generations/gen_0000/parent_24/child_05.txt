from scipy.optimize import minimize

def run_packing():

    # Or simpler: generate a dense hexagonal lattice and select 26 points.

    s = 0.2 
    dy = s * np.sqrt(3) / 2

    # We'll try to fill the square
    y = 0.0
    row_idx = 0
    while True:

        if row_idx % 2 == 1:
            x = s / 2
        while x + 0.1 < 1.1: # +0.1 is radius margin roughly
            points.append([x + 0.1, y + 0.1]) # Add radius offset to be inside
            if len(points) >= n:
                break
            x += s
        if len(points) >= n:
            break
        y += dy
        row_idx += 1

            dists.append(y)       # dist to bottom wall
            dists.append(1 - y)   # dist to top wall
