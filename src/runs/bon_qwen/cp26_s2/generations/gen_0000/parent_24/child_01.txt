import numpy as np
from scipy.optimize import minimize

N = 26

def obj(x):
    return -np.sum(x[2*N:])

def cons(x):
    pts = x[:2*N].reshape(N, 2)
    r = x[2*N:]
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c = np.concatenate([
        pts[:, 0] - r,
        1 - pts[:, 0] - r,
        pts[:, 1] - r,
        1 - pts[:, 1] - r
    ])
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    diff = pts[:, None, :] - pts[None, :, :]
    dist2 = np.sum(diff**2, axis=2)
    rsum = r[:, None] + r[None, :]
    idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dist2[idx] - rsum[idx]**2])
    return c

def run_packing():
    bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N
    con = {'type': 'ineq', 'fun': cons}
    
    best_val = 0.0
    best_x = None
    
    for seed in range(30):
        np.random.seed(seed)
        if seed < 15:
            ys = np.linspace(0.1, 0.9, 6)
            xs = np.linspace(0.1, 0.9, 5)
            pts0 = np.array([(x, y) for y in ys for x in xs if len(pts0) < N])
            pts0 += np.random.uniform(-0.03, 0.03, pts0.shape)
            pts0 = np.clip(pts0, 0.05, 0.95)
        else:
            pts0 = np.random.rand(N, 2) * 0.8 + 0.1
            
        r0 = np.full(N, 0.04)
        x0 = np.concatenate([pts0.flatten(), r0])
        
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds, 
                          constraints=con, options={'maxiter': 1500, 'ftol': 1e-10})
            val = -obj(res.x)
            if val > best_val:
                best_val = val
                best_x = res.x.copy()
        except Exception:
            pass
            
    if best_x is None:
        best_x = np.zeros(3*N)
        for i in range(N):
            best_x[3*i] = (i % 5 + 1) * 0.2
            best_x[3*i+1] = (i // 5 + 1) * 0.2
            best_x[3*i+2] = 0.01
            
    centers = best_x[:2*N].reshape(N, 2)
    radii = best_x[2*N:]
    return centers, radii, float(np.sum(radii))
