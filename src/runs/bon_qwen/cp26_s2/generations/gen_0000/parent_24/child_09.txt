def run_packing():

                idx += 1

    centers += np.random.uniform(-0.001, 0.001, size=(n, 2))

            r = r_target
            
        forces = np.zeros((n, 2))
        
        for i in range(n):
            # Wall repulsion forces
            for d in range(2):
                if centers[i, d] < r:
                    forces[i, d] += (r - centers[i, d]) / (r + 1e-9)
                elif centers[i, d] > 1 - r:
                    forces[i, d] -= (centers[i, d] - (1 - r)) / (r + 1e-9)

                dist = np.sqrt(dist_sq)

                    else:
                        fx = overlap
                        fy = 0.0
                    forces[i] += [fx, fy]
                    forces[j] -= [fx, fy]

        centers += forces * lr

            lr *= 0.9995

    centers[:, 0] = np.clip(centers[:, 0], r, 1 - r)

    return centers, radii, np.sum(radii)
