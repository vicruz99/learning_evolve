import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_fun(vars):
    """
    Returns array of constraint values. All must be >= 0.
    Includes boundary and non-overlap constraints.
    """
    C = vars.reshape(N_CIRCLES, 3)
    x, y, r = C[:,0], C[:,1], C[:,2]
    cons = []
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)
    
    # Non-overlap constraints: distance^2 >= (r_i + r_j)^2
    xx = x[:, None] - x[None, :]
    yy = y[:, None] - y[None, :]
    rr = r[:, None] + r[None, :]
    dist2 = xx**2 + yy**2
    idx = np.triu_indices(N_CIRCLES, k=1)
    cons.append(dist2[idx] - rr[idx]**2)
    
    return np.concatenate(cons)

def run_packing():
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Variable bounds: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    for seed in range(15):
        # 1. Initialize with hexagonal lattice
        r_guess = 0.095
        dy = r_guess * np.sqrt(3)
        dx = 2 * r_guess
        centers_init = []
        y = r_guess
        row_parity = 0
        while len(centers_init) < n:
            x = r_guess if row_parity == 0 else r_guess + dx/2
            while x <= 1 - r_guess and len(centers_init) < n:
                centers_init.append([x, y])
                x += dx
            y += dy
            row_parity = 1 - row_parity
            
        centers_init = np.array(centers_init)
        
        # 2. Perturb initial configuration
        np.random.seed(seed)
        centers_init += np.random.uniform(-0.01, 0.01, centers_init.shape)
        centers_init = np.clip(centers_init, r_guess, 1 - r_guess)
        
        r_init = r_guess + np.random.uniform(-0.005, 0.005, n)
        r_init = np.clip(r_init, 0.01, 0.2)
        
        x0 = np.hstack([centers_init.ravel(), r_init])
        
        # 3. Optimize
        try:
            res = minimize(objective, x0, method='SLSQP', 
                           bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraint_fun}, 
                           options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x[:2*n].reshape(n, 2)
                best_radii = res.x[2*n:]
        except Exception:
            continue
            
    # 4. Post-processing to guarantee strict feasibility
    if best_centers is not None:
        # Clamp to boundaries
        for i in range(n):
            x, y = best_centers[i]
            r_max = min(x, 1-x, y, 1-y)
            if best_radii[i] > r_max:
                best_radii[i] = r_max
                
        # Iteratively shrink overlapping circles
        for _ in range(30):
            changed = False
            for i in range(n):
                for j in range(i+1, n):
                    dist = np.hypot(best_centers[i,0]-best_centers[j,0], 
                                    best_centers[i,1]-best_centers[j,1])
                    req = best_radii[i] + best_radii[j]
                    if dist < req - 1e-10:
                        shrink = (req - dist) / 2.0 + 1e-10
                        best_radii[i] -= shrink
                        best_radii[j] -= shrink
                        changed = True
            if not changed:
                break
                
        best_radii = np.maximum(best_radii, 0.0)
        
        # Final boundary clamp
        for i in range(n):
            x, y = best_centers[i]
            r_max = min(x, 1-x, y, 1-y)
            best_radii[i] = min(best_radii[i], r_max)
            
    return best_centers, best_radii, np.sum(best_radii)
