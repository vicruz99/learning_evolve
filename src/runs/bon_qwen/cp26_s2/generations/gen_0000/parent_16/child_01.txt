import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26

    # We create a dense hexagonal pattern to ensure a high-quality starting point.
    initial_radii = 0.09
    centers = []
    row_count = 0

    while len(centers) < n:
        y = initial_radii + row_count * initial_radii * np.sqrt(3)
        
        # Shift x-offset for alternating rows
        x_start = initial_radii * 2.0 if row_count % 2 == 1 else initial_radii
        x = x_start
        
        while x <= 1.0 - initial_radii:
            if len(centers) < n:
                centers.append([x, y])
            x += 2.0 * initial_radii
        row_count += 1

    centers = np.array(centers[:n])
    
    # --- Step 2: Optimization Setup ---
    def objective_function(v):
        # Reshape the flat vector into centers and radii
        centers = v[:2 * n].reshape((n, 2))
        radii = v[2 * n:]
        
        total_sum = -np.sum(radii) # Negative because we minimize
        
        penalty = 0.0
        
        # Boundary constraints penalty
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            
            # Distance to boundaries
            d_left = x - r
            d_right = (1.0 - x) - r
            d_down = y - r
            d_up = (1.0 - y) - r
            
            boundaries = np.array([d_left, d_right, d_down, d_up])
            violations = boundaries[boundaries < 0]
            penalty += np.sum(violations ** 2)

            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    penetration = min_dist - dist
                    penalty += penetration ** 2
                    
        # Weights to ensure constraints are respected
        penalty_weight = 100.0
        return total_sum + penalty_weight * penalty

    # Define bounds for all variables
    bounds = []

        bounds.append((0.0, 1.0)) # x, y bounds
    for _ in range(n):
        bounds.append((0.0, 0.5)) # radius bounds

    # --- Step 3: Global Optimization (Differential Evolution) ---
    # Using a robust global optimizer to find a high-quality configuration
    result_global = differential_evolution(
        objective_function,
        bounds,
        seed=42,
        maxiter=1000,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.9,
        popsize=20,
        polish=False # We polish manually later
    )

    best_v = result_global.x
    best_centers = best_v[:2 * n].reshape((n, 2))
    best_radii = best_v[2 * n:]

    result_local = minimize(
        objective_function,
        best_v,

    )

    final_radii = final_v[2 * n:]

    # Clip radii to non-negative and ensure valid bounds
    final_radii = np.maximum(final_radii, 0.0)
    final_centers = np.clip(final_centers, 0.0, 1.0)
    
    # Final sum calculation
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
