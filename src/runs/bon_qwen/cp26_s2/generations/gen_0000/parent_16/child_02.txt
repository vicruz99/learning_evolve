import numpy as np
from scipy.optimize import minimize

def my_objective(x, n):
    """
    Objective function to maximize sum of radii.
    We minimize the negative sum.
    x is a flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """
    # Radii are at indices 2, 5, 8, ... (every 3rd element starting at 2)
    r = x[2::3]
    return -np.sum(r)

def my_constraints(x, n):
    """
    Constraint function. Returns an array of values that must be >= 0.
    Includes boundary constraints and pairwise distance constraints.
    """
    # Reshape to (n, 3) where columns are x, y, r
    points = x.reshape(-1, 3)
    cx = points[:, 0]
    cy = points[:, 1]
    cr = points[:, 2]
    
    cons = []
    
    # Boundary constraints:
    # 1. x - r >= 0
    cons.append(cx - cr)
    # 2. 1 - x - r >= 0
    cons.append(1 - cx - cr)
    # 3. y - r >= 0
    cons.append(cy - cr)
    # 4. 1 - y - r >= 0
    cons.append(1 - cy - cr)
    
    # Distance constraints:
    # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
    # Equivalent to: dist_sq - (ri + rj)^2 >= 0
    
    # Vectorized pairwise differences
    # cx shape (n,), cx[:, None] shape (n, 1)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    dist_sq = dx**2 + dy**2
    r_sum_sq = dr**2
    
    # Constraint matrix: dist_sq - r_sum_sq
    # We need this to be >= 0 for all i != j
    # Since the matrix is symmetric, we only need to check i < j (upper triangle)
    mat = dist_sq - r_sum_sq
    
    # Extract upper triangle elements
    # np.triu_indices(n, k=1) gives indices for i < j
    i, j = np.triu_indices(n, k=1)
    dist_cons = mat[i, j]
    
    cons.append(dist_cons)
    
    return np.concatenate(cons)
