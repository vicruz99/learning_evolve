import numpy as np
from scipy.optimize import minimize

def run_packing():
    # 1. Setup initial configuration (5x5 grid + 1 extra)
    n = 26
    initial_r = 0.08  # Start slightly smaller than 0.1 to allow space
    
    # Generate grid points
    x_coords = np.linspace(0.1, 0.9, 5)
    y_coords = np.linspace(0.1, 0.9, 5)
    
    centers = []
    radii = []
    
    # Add 25 circles in a grid
    for i in x_coords:
        for j in y_coords:
            centers.append([i, j])
            radii.append(initial_r)
            
    # Add 26th circle in a gap (e.g., between the first four)
    centers.append([0.2, 0.2])
    radii.append(initial_r)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # 2. Prepare optimization variables
    # Variables vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]

    # Define bounds for [0, 1] square and positive radii
    bounds = []
    for i in range(n):
        bounds.extend([
            (0.0, 1.0),  # x
            (0.0, 1.0),  # y
            (0.0, 0.5)   # r
        ])

    # 3. Objective function (Negative sum of radii + Penalties)
    def objective(vars):
        sum_r = 0.0
        penalty = 0.0
        penalty_weight = 100.0 # Weight for constraint penalties
        
        # Extract variables
        xs = vars[0::3]
        ys = vars[1::3]
        rs = vars[2::3]
        
        sum_r = np.sum(rs)
        
        # Boundary penalties: max(0, r - x)^2 + max(0, r - (1-x))^2 ...
        for i in range(n):
            xi, yi, ri = xs[i], ys[i], rs[i]
            # Left
            val = ri - xi
            if val > 0: penalty += val**2
            # Right
            val = ri - (1.0 - xi)
            if val > 0: penalty += val**2
            # Bottom
            val = ri - yi
            if val > 0: penalty += val**2
            # Top
            val = ri - (1.0 - yi)
            if val > 0: penalty += val**2

        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((xs[i] - xs[j])**2 + (ys[i] - ys[j])**2)
                sum_radii = rs[i] + rs[j]
                overlap = sum_radii - dist
                if overlap > 0:
                    penalty += overlap**2
                    
        return -sum_r + penalty_weight * penalty

    # 4. Run Optimization
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 2000, 'ftol': 1e-15})
    
    final_vars = result.x
    
    # 5. Extract results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_vars[3 * i]
        final_centers[i, 1] = final_vars[3 * i + 1]
        final_radii[i] = final_vars[3 * i + 2]
        
    total_sum = np.sum(final_radii)
    
    return final_centers, final_radii, total_sum
