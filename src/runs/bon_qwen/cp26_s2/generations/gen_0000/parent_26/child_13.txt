

<strategy>
To pack 26 circles in a unit square to maximize the sum of radii, we employ a two-phase strategy.

**Phase 1: High-Quality Initialization**
Equal-radius circles arranged in a hexagonal lattice provide the densest packing density ($\approx 90.69\%$). Since a perfect $5 \times 5$ square grid leaves significant empty space, we initialize our centers using a staggered hexagonal pattern. This layout minimizes initial gaps and gives the optimizer a strong starting point for expansion.

**Phase 2: Numerical Optimization**
We define a penalty-based objective function to maximize the sum of radii.
1.  **Objective**: Maximize $\sum r_i$ by minimizing $-\sum r_i$.
2.  **Penalty**: We enforce constraints by adding a large weighted penalty for any overlaps between circles or between circles and the square boundaries. The penalty term $\sum \max(0, r_i + r_j - d_{ij})^2$ ensures circles are pushed apart until they touch but do not overlap.
3.  **Solver**: We use `scipy.optimize.minimize` with the `L-BFGS-B` algorithm. We run this optimization with multiple random restarts to ensure we escape local minima and converge to the global maximum (or a very tight local approximation).
</strategy>

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Penalty weight for constraints. Higher means stricter enforcement.
    penalty_weight = 1e5
    
    def objective(vars):
        # vars shape: (n, 3) -> [x, y, r]
        centers = vars[:, :2]
        radii = vars[:, 2]
        
        # Objective: Maximize sum of radii (minimize negative sum)
        score = -np.sum(radii)
        
        # Constraint 1: Boundary constraints
        # x - r >= 0  => r - x <= 0
        # 1 - x - r >= 0 => x + r - 1 <= 0
        # Same for y
        violations = np.maximum(0, radii - centers[:, 0])
        violations += np.maximum(0, centers[:, 0] + radii - 1)
        violations += np.maximum(0, radii - centers[:, 1])
        violations += np.maximum(0, centers[:, 1] + radii - 1)
        penalty += np.sum(violations**2)
        
        # Constraint 2: Non-overlap
        # dist(i, j) >= r_i + r_j  => r_i + r_j - dist <= 0
        # Vectorized pairwise distance
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        # Upper triangle mask to avoid double counting and self
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        sum_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlaps = np.maximum(0, sum_radii - dists)
        penalty += np.sum(np.sum(overlaps**2) * mask)
        
        return score + penalty_weight * penalty

    # We need to define penalty inside the closure or pass it. 
    # Re-writing objective to be self-contained or using a wrapper.
    
    def objective_function(vars):
        centers = vars[:, :2]
        radii = vars[:, 2]
        
        score = -np.sum(radii)
        
        # Boundary
        v1 = np.maximum(0, radii - centers[:, 0])
        v2 = np.maximum(0, centers[:, 0] + radii - 1)
        v3 = np.maximum(0, radii - centers[:, 1])
        v4 = np.maximum(0, centers[:, 1] + radii - 1)
        boundary_penalty = np.sum(v1**2 + v2**2 + v3**2 + v4**2)
        
        # Overlaps
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        sum_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap_matrix = np.maximum(0, sum_radii - dists)
        # Sum over upper triangle
        overlap_penalty = np.sum(overlap_matrix[np.triu_indices(n, k=1)]) ** 2

        return score + penalty_weight * (boundary_penalty + overlap_penalty)

    def generate_initial_guess():
        # Hexagonal packing initialization
        # We want to fit 26 circles. 
        # Rows with approx spacing sqrt(3)/2 * diameter.
        # Let's try to fit roughly 6 rows.
        centers = []
        y = 0.15 # Initial margin
        row_idx = 0
        while len(centers) < 26:
            x = 0.15
            shift = 0.2 if row_idx % 2 == 1 else 0.0
            while x < 1.0 - 0.15 and len(centers) < 26:
                centers.append([x + shift, y, 0.08]) # Initial radius estimate
                x += 0.35 # Approx horizontal spacing
            y += 0.3 # Approx vertical spacing
            row_idx += 1
        return np.array(centers[:26])

    # Run multiple restarts to find the best global optimum
    for _ in range(10):
        # Random initialization around the hexagonal grid or fully random
        # Using the grid but with some noise usually works best
        initial_guess = generate_initial_guess()
        # Add small random noise to escape exact grid symmetries if needed
        initial_guess[:, :2] += np.random.uniform(-0.02, 0.02, size=(n, 2))
        
        # Bounds for [x, y, r]
        # x, y in [0, 1], r in [0, 0.5]
        bounds = [(0, 1), (0, 1), (1e-4, 0.5)] * n
        
        # Flatten initial guess for scipy
        x0 = initial_guess.flatten()
        
        res = minimize(objective_function, x0, method='L-BFGS-B', bounds=bounds, 
                       options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 2000})
        
        if res.success or res.fun < 0: # Negative fun means we minimized -sum, so sum is positive
            final_centers = res.x[:n*2].reshape(n, 2)
            final_radii = res.x[n*2:]
            
            # Validation check
            valid = True
            # Check boundaries
            if np.any(final_radii - final_centers[:, 0] > 1e-6) or \
               np.any(final_centers[:, 0] + final_radii - 1 > 1e-6) or \
               np.any(final_radii - final_centers[:, 1] > 1e-6) or \
               np.any(final_centers[:, 1] + final_radii - 1 > 1e-6):
                valid = False
            
            # Check overlaps (approximate check)
            diffs = final_centers[:, np.newaxis, :] - final_centers[np.newaxis, :, :]
            dists = np.linalg.norm(diffs, axis=2)
            sum_r = final_radii[:, np.newaxis] + final_radii[np.newaxis, :]
            if np.any(dists[np.triu_indices(n, k=1)] < sum_r[np.triu_indices(n, k=1)] - 1e-6):
                valid = False
                
            if valid:
                current_sum = np.sum(final_radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = final_centers.copy()
                    best_radii = final_radii.copy()

    # If optimization failed to produce a valid result (rare), fallback to simple grid
    if best_centers is None:
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.1
        # Just place them in a grid to be safe
        idx = 0
        for r in range(6):
            for c in range(5):
                if idx < n:
                    centers[idx] = [0.1 + c * 0.2, 0.1 + r * 0.2]
                    idx += 1
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
