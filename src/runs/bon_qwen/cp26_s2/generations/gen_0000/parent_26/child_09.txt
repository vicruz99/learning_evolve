

To maximize the sum of radii for 26 circles in a unit square, we use a hybrid optimization strategy. We initialize the circles in a hexagonal pattern—known for its high density—and then employ the SLSQP (Sequential Least Squares Programming) algorithm from `scipy.optimize`. 

This solver respects the boundary and non-overlap constraints while maximizing the linear sum of radii. To avoid local optima and numerical instability (singularities), we use squared penalties for overlaps and bounds to ensure the radii remain positive and the centers stay within the square.

```python
import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization
    # We use a hexagonal packing start which is more efficient than a square grid.
    np.random.seed(42)
    xs = []
    ys = []
    
    # Hexagonal rows: 5-4-5-4-5 pattern to fit 23 circles, plus extra in rows.
    r_init = 0.1
    y_row = r_init
    row_type = 0 # 0 for offset, 1 for aligned
    
    while len(xs) < n:
        if row_type % 2 == 0:
            # Aligned rows (5 circles)
            x_start = r_init
            count = 5
        else:
            # Offset rows (4 circles)
            x_start = r_init + 0.1
            count = 4
            
        for i in range(count):
            if len(xs) >= n:
                break
            xs.append(x_start + i * 0.2)
            ys.append(y_row)
            
        row_type += 1
        y_row += r_init * np.sqrt(3)
        
    xs = np.array(xs)
    ys = np.array(ys)
    
    # Add slight perturbation to break symmetry
    xs += np.random.uniform(-0.005, 0.005, n)
    ys += np.random.uniform(-0.005, 0.005, n)
    # Ensure initial positions are safely inside [0.1, 0.9]
    xs = np.clip(xs, 0.1, 0.9)
    ys = np.clip(ys, 0.1, 0.9)

    # Initial state: (x1, y1, r1, x2, y2, r2, ...)
    # Start with small radii to let the solver expand them from a valid state
    radii_start = np.full(n, 0.08)
    x0 = np.concatenate([xs, ys, radii_start])

    # 2. Objective Function (Penalty Method)
    penalty_factor = 5000.0

    def objective(params):
        # Unpack
        centers = params[:2 * n].reshape(n, 2)
        radii = params[2 * n:]
        
        # Negative sum of radii (to maximize)
        val = -np.sum(radii)
        
        # Overlap Penalties (Squared to ensure smoothness)
        penalty = 0.0
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            
            # Boundary violations
            if ri - xi > 0: penalty += (ri - xi)**2
            if xi + ri - 1 > 0: penalty += (xi + ri - 1)**2
            if ri - yi > 0: penalty += (ri - yi)**2
            if yi + ri - 1 > 0: penalty += (yi + ri - 1)**2
            
            # Inter-circle overlaps
            for j in range(i + 1, n):
                dist = np.sqrt((xi - centers[j, 0])**2 + (yi - centers[j, 1])**2)
                overlap = ri + radii[j] - dist
                if overlap > 0:
                    penalty += overlap**2
                    
        return val + penalty_factor * penalty

    # 3. Bounds and Constraints
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)]) # x, y bounds
        bounds.append((0.0, 0.5))                # r bounds

    # 4. Optimization
    # SLSQP is robust for non-linear constraints with bounds
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   options={'maxiter': 2000, 'ftol': 1e-12})
    
    # 5. Post-process results
    best_params = res.x
    centers = best_params[:2 * n].reshape(n, 2)
    radii = best_params[2 * n:]
    
    # Ensure radii are strictly non-negative
    radii = np.maximum(radii, 1e-12)
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
```
