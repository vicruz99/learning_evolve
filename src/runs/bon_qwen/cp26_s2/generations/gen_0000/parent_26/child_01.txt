import numpy as np
from scipy.optimize import linprog

def run_packing():
    n = 26
    
    # Initialization: Grid of valid centers for r=0.05
    # This ensures no overlaps and valid boundaries initially
    xs = np.arange(0.05, 1.0, 0.1)
    ys = xs
    grid_x, grid_y = np.meshgrid(xs, ys)
    all_points = np.vstack([grid_x.ravel(), grid_y.ravel()]).T
    
    # Shuffle to pick random subset, or just pick first 26
    # Picking first 26 from row-major might cluster.
    # Let's shuffle with a fixed seed for reproducibility
    rng = np.random.default_rng(42)
    indices = rng.permutation(len(all_points))[:n]
    centers = all_points[indices]
    
    # Precompute pairs
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    num_pairs = len(pairs)
    
    # Build A matrix for LP
    # Rows: num_pairs + n
    # Cols: n
    # We can use a list of lists or dense array. Dense is fine.
    A_ub = np.zeros((num_pairs + n, n))
    for k, (i, j) in enumerate(pairs):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
    for k, i in enumerate(range(n)):
        A_ub[num_pairs + k, i] = 1.0
        
    c_obj = -np.ones(n) # Minimize -sum(r)
    
    # Optimization loop
    step_size = 0.015
    # Decay factor
    decay = 0.995
    
    for iter_num in range(150):
        # Compute distances
        # Vectorized
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Compute L_i
        x = centers[:, 0]
        y = centers[:, 1]
        L = np.minimum(np.minimum(x, 1-x), np.minimum(y, 1-y))
        
        # b_ub vector
        b_ub = np.zeros(num_pairs + n)
        
        # Pair constraints RHS

        # Let's just loop, 325 iters is nothing
        for k, (i, j) in enumerate(pairs):
            b_ub[k] = dists[i, j]
        
        b_ub[num_pairs:] = L
        
        # Solve LP
        bounds = [(0, None) for _ in range(n)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            # Fallback or break
            break
            
        if res.success:
            # duals
            # Check if marginals attribute exists
            if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                duals = res.ineqlin.marginals
                # Marginals for <= constraints are usually negative (shadow prices)
                # We want positive weights for forces
                lambdas = -duals
                # Ensure non-negative
                lambdas = np.maximum(lambdas, 0.0)

                # Pair forces
                # Iterate pairs
                for k, (i, j) in enumerate(pairs):
                    lam = lambdas[k]
                    if lam > 1e-7:
                        dist = dists[i, j]
                        if dist > 1e-9:
                            dir_vec = (centers[i] - centers[j]) / dist
                            forces[i] += lam * dir_vec
                            forces[j] -= lam * dir_vec

                            forces[j] -= (np.random.rand(2) - 0.5) * 0.1

                mu = lambdas[num_pairs:]
                for i in range(n):
                    if mu[i] > 1e-7:
                        xi, yi = centers[i]
                        d_left = xi
                        d_right = 1 - xi
                        d_bottom = yi
                        d_top = 1 - yi

                        # Just check if distance to wall is small?

                        # So r_i = L_i.
                        # And L_i is determined by min distance to wall.
                        # So we check which distance equals L_i.

                        if abs(d_left - L[i]) < tol:

                        if abs(d_bottom - L[i]) < tol:

                        if abs(d_top - L[i]) < tol:
                            push_y -= 1.0
                        
                        mag = np.sqrt(push_x**2 + push_y**2)
                        if mag > 1e-9:
                            push_x /= mag
                            push_y /= mag
                        
                        forces[i, 0] += mu[i] * push_x
                        forces[i, 1] += mu[i] * push_y

                # Update centers
                centers += step_size * forces
                centers = np.clip(centers, 0.0, 1.0)
                
                # Decay step size
                step_size *= decay
            else:
                break
        else:
            break

    # Final radii calculation
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    x = centers[:, 0]
    y = centers[:, 1]
    L = np.minimum(np.minimum(x, 1-x), np.minimum(y, 1-y))
    
    b_ub_final = np.zeros(num_pairs + n)
    for k, (i, j) in enumerate(pairs):
        b_ub_final[k] = dists[i, j]
    b_ub_final[num_pairs:] = L
    
    res_final = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_final, bounds=bounds, method='highs')
    
    if res_final.success:
        radii = res_final.x
    else:
        radii = np.full(n, 0.0) # Should not happen
        
    sum_r = np.sum(radii)
    
    return centers, radii, sum_r
