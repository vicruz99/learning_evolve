```python
import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function to minimize (negative sum of radii)."""
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_boundary(vars):
    """Boundary constraints: circles inside [0,1]x[0,1]."""
    c = np.zeros(4 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x = vars[3 * i]
        y = vars[3 * i + 1]
        r = vars[3 * i + 2]
        # x - r >= 0
        c[4 * i] = x - r
        # 1 - (x + r) >= 0  => x + r <= 1
        c[4 * i + 1] = 1 - x - r
        # y - r >= 0
        c[4 * i + 2] = y - r
        # 1 - (y + r) >= 0  => y + r <= 1
        c[4 * i + 3] = 1 - y - r
    return c

def constraint_overlap(vars):
    """Overlap constraints: distance between centers >= sum of radii."""
    n_constraints = N_CIRCLES * (N_CIRCLES - 1) // 2
    c = np.zeros(n_constraints)
    idx = 0
    for i in range(N_CIRCLES):
        xi = vars[3 * i]
        yi = vars[3 * i + 1]
        ri = vars[3 * i + 2]
        for j in range(i + 1, N_CIRCLES):
            xj = vars[3 * j]
            yj = vars[3 * j + 1]
            rj = vars[3 * j + 2]
            
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx * dx + dy * dy
            sum_r = ri + rj
            c[idx] = dist_sq - sum_r * sum_r
            idx += 1
    return c
```

```python
def constraint_overlap(vars):
    centers = vars[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars[2::3]
    
    # Compute pairwise squared distances
    # dist_sq[i, j] = (xi-xj)^2 + (yi-yj)^2
    # We only need upper triangle
    
    # Using broadcasting
    # centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (N, N, 2)
    dist_sq = np.sum(diff**2, axis=2) # (N, N)
    
    # Sum of radii matrix
    sum_r = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Constraints: dist_sq >= sum_r^2
    # We need a flat array of constraints for i < j
    triu_indices = np.triu_indices(N_CIRCLES, k=1)
    c = dist_sq[triu_indices] - sum_r[triu_indices]**2
    return c
```

```python
import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

# Precompute indices for overlap constraints to avoid recreation
TRIU_INDICES = np.triu_indices(N_CIRCLES, k=1)

def objective(vars):
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_boundary(vars):
    centers = vars[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars[2::3]
    
    c1 = centers[:, 0] - radii
    c2 = 1 - centers[:, 0] - radii
    c3 = centers[:, 1] - radii
    c4 = 1 - centers[:, 1] - radii
    
    return np.concatenate([c1, c2, c3, c4])

def constraint_overlap(vars):
    centers = vars[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars[2::3]
    
    # Pairwise differences
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Pairwise sum of radii
    sum_r = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Constraints for i < j
    return dist_sq[TRIU_INDICES] - sum_r[TRIU_INDICES]**2

# Setup constraints and bounds
cons_boundary = {'type': 'ineq', 'fun': constraint_boundary}
cons_overlap = {'type': 'ineq', 'fun': constraint_overlap}
constraints = [cons_boundary, cons_overlap]
bounds = [(0, 1), (0, 1), (0, None)] * N_CIRCLES

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -np.inf
    best_vars = None

    x0_list = []

    # 1. Grid initialization
    x0 = np.zeros(3 * N_CIRCLES)
    k = 0
    for r in range(5):
        for c in range(5):
            x0[3 * k] = 0.1 + c * 0.2
            x0[3 * k + 1] = 0.1 + r * 0.2
            x0[3 * k + 2] = 0.1
            k += 1
    x0[3 * k] = 0.2
    x0[3 * k + 1] = 0.2
    x0[3 * k + 2] = 0.0414
    x0_list.append(x0)

    # 2. Hexagonal initialization
    x0 = np.zeros(3 * N_CIRCLES)
    k = 0
    rows_counts = [5, 4, 5, 4, 5, 3]
    r_est = 0.09
    y_pos = r_est
    h = r_est * np.sqrt(3)
    for i, count in enumerate(rows_counts):
        shift = (i % 2) * r_est
        for j in range(count):
            if k >= N_CIRCLES: break
            x = r_est + shift + j * (2 * r_est)
            y = y_pos
            x0[3 * k] = x
            x0[3 * k + 1] = y
            x0[3 * k + 2] = r_est
            k += 1
        y_pos += h
    x0_list.append(x0)

    # 3. Random restarts
    for _ in range(15):
        x0 = np.random.rand(3 * N_CIRCLES)
        x0[2::3] = np.random.rand(N_CIRCLES) * 0.05 + 0.05
        x0_list.append(x0)

    for x0 in x0_list:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x
        except Exception:
            pass

    # Refinement
    if best_vars is not None:
        for _ in range(5):
            pert = np.random.normal(0, 0.002, size=best_vars.shape)
            x0 = np.clip(best_vars + pert, 0, 1)
            x0[2::3] = np.maximum(x0[2::3], 1e-5)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                               constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-12})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_vars = res.x
            except Exception:
                pass

    if best_vars is None:
        centers = np.zeros((N_CIRCLES, 2))
        radii = np.zeros(N_CIRCLES)
        return centers, radii, 0.0

    centers = best_vars[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_vars[2::3]
    radii = np.maximum(radii, 0)
    
    return centers, radii, np.sum(radii)
```

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we treat this as a non-linear optimization problem. We maximize $\sum r_i$ subject to:
1.  Boundary constraints: $0 \le x_i - r_i$ and $x_i + r_i \le 1$ (similarly for $y$).
2.  Non-overlap constraints: $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$.

We use `scipy.optimize.minimize` with the SLSQP method. Since the problem is non-convex, we employ multiple random restarts and structured initializations (a $5 \times 5$ grid and a hexagonal lattice) to find a high-quality solution. The constraints are vectorized for efficiency.
</strategy>
