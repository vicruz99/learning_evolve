import numpy as np
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')

def obj_func(v):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)"""
    return -np.sum(v[2::3])

def constr_r_i(v, i):
    return v[i*3+2]

def constr_xi_min(v, i):
    return v[i*3] - v[i*3+2]

def constr_xi_max(v, i):
    return 1.0 - v[i*3] - v[i*3+2]

def constr_yi_min(v, i):
    return v[i*3+1] - v[i*3+2]

def constr_yi_max(v, i):
    return 1.0 - v[i*3+1] - v[i*3+2]

def constr_dist(v, i, j):
    dx = v[i*3] - v[j*3]
    dy = v[i*3+1] - v[j*3+1]
    return np.sqrt(dx*dx + dy*dy + 1e-12) - v[i*3+2] - v[j*3+2]

def run_packing():
    n = 26
    best_sum = -1.0
    best_x = None
    
    # Variable bounds: x in [0,1], y in [0,1], r in [1e-6, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
        
    # Predefine constraint structures to avoid lambda/closures
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constr_r_i, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_xi_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_xi_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_yi_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_yi_max, 'args': (i,)})
        
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constr_dist, 'args': (i, j)})
            
    # Multiple restarts from perturbed hexagonal grids
    for seed in range(20):
        np.random.seed(seed)
        x0 = np.zeros(3 * n)
        cols = 6
        for i in range(n):
            r_idx = i // cols
            c_idx = i % cols
            # Hexagonal lattice coordinates scaled to unit square
            y_coord = (r_idx + 0.5) / ((n + cols - 1) // cols + 0.5)
            x_coord = (c_idx + 0.5 + (r_idx % 2) * 0.5) / (cols + 0.5)
            
            x0[i*3]   = x_coord + np.random.uniform(-0.05, 0.05)
            x0[i*3+1] = y_coord + np.random.uniform(-0.05, 0.05)
            x0[i*3+2] = 0.05  # Start with small feasible radii
            
        # Ensure initial points are strictly inside bounds
        x0[:2*n:3] = np.clip(x0[:2*n:3], 0.02, 0.98)
        x0[1:2*n:3] = np.clip(x0[1:2*n:3], 0.02, 0.98)
        
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            continue
            
    centers = np.column_stack((best_x[:2*n:3], best_x[1:2*n:3]))
    radii = best_x[2::3]
    return centers, radii, best_sum
