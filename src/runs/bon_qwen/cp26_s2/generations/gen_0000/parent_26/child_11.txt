from scipy.optimize import minimize

def compute_objective(params, n, lam):

    Objective function to minimize.
    Minimizes: -sum(radii) + penalty for overlaps and boundary violations.
    """
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]

    dx = np.maximum(0.0, radii - centers[:, 0])
    dx += np.maximum(0.0, radii - (1.0 - centers[:, 0]))
    dy = np.maximum(0.0, radii - centers[:, 1])
    dy += np.maximum(0.0, radii - (1.0 - centers[:, 1]))
    boundary_pen = lam * (np.sum(dx**2) + np.sum(dy**2))

    diff = centers[:, None, :] - centers[None, :, :]

    overlap_pen = lam * np.sum(violations**2)

    return -np.sum(radii) + boundary_pen + overlap_pen
