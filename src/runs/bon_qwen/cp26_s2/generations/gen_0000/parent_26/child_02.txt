import numpy as np
from scipy.optimize import minimize
from functools import partial


def objective(params, n):
    radii = params[2::3]
    return -np.sum(radii)


def boundary_x_min(params, n, i):
    return params[3 * i] - params[3 * i + 2]


def boundary_x_max(params, n, i):
    return 1.0 - params[3 * i] - params[3 * i + 2]


def boundary_y_min(params, n, i):
    return params[3 * i + 1] - params[3 * i + 2]


def boundary_y_max(params, n, i):
    return 1.0 - params[3 * i + 1] - params[3 * i + 2]


def nonneg_radius(params, n, i):
    return params[3 * i + 2]


def nonoverlap(params, n, i, j):
    xi = params[3 * i]
    yi = params[3 * i + 1]
    ri = params[3 * i + 2]
    xj = params[3 * j]
    yj = params[3 * j + 1]
    rj = params[3 * j + 2]
    dx = xi - xj
    dy = yi - yj
    return dx * dx + dy * dy - (ri + rj) * (ri + rj)


def build_constraints(n):
    constraints = []

    for i in range(n):
        constraints.append({
            'type': 'ineq',
            'fun': partial(boundary_x_min, n=n, i=i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(boundary_x_max, n=n, i=i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(boundary_y_min, n=n, i=i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(boundary_y_max, n=n, i=i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(nonneg_radius, n=n, i=i)
        })

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': partial(nonoverlap, n=n, i=i, j=j)
            })

    return constraints


def create_hexagonal_init(n, seed=0):
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)

    idx = 0
    row = 0
    max_rows = 10
    while idx < n and row < max_rows:
        n_in_row = 5
        y = (row + 0.5) / 6.0
        y = np.clip(y, 0.1, 0.9)
        for col in range(n_in_row):
            if idx >= n:
                break
            x = (col + 0.5 + (row % 2) * 0.5) / 5.0
            x = np.clip(x, 0.05, 0.95)
            centers[idx] = [x, y]
            idx += 1
        row += 1

    while idx < n:
        centers[idx] = rng.rand(2) * 0.8 + 0.1
        idx += 1

    params = np.zeros(3 * n)
    for i in range(n):
        params[3 * i] = centers[i, 0]
        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = radii[i]

    return params


def create_grid_init(n, seed=0):
    rng = np.random.RandomState(seed)
    side = int(np.ceil(np.sqrt(n)))
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.07)

    idx = 0
    for i in range(side):
        for j in range(side):
            if idx >= n:
                break
            x = (i + 0.5) / side
            y = (j + 0.5) / side
            centers[idx] = [x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)]
            centers[idx, 0] = np.clip(centers[idx, 0], 0.05, 0.95)
            centers[idx, 1] = np.clip(centers[idx, 1], 0.05, 0.95)
            idx += 1

    params = np.zeros(3 * n)
    for i in range(n):
        params[3 * i] = centers[i, 0]
        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = radii[i]

    return params


def create_random_init(n, seed=0):
    rng = np.random.RandomState(seed)
    centers = rng.rand(n, 2) * 0.7 + 0.15
    radii = np.full(n, 0.05)

    params = np.zeros(3 * n)
    for i in range(n):
        params[3 * i] = centers[i, 0]
        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = radii[i]

    return params


def optimize_once(params, n, constraints, bounds, max_iter=1500):
    try:
        result = minimize(
            objective,
            params,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False}
        )
        return result
    except Exception:
        return None


def run_packing():
    n = 26
    constraints = build_constraints(n)

    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n

    best_sum = -np.inf
    best_params = None

    # Strategy 1: Hexagonal with various seeds
    for seed in range(25):
        params = create_hexagonal_init(n, seed=seed)
        rng = np.random.RandomState(seed * 100 + 1)
        params += rng.randn(3 * n) * 0.015
        params[0::3] = np.clip(params[0::3], 0.01, 0.99)
        params[1::3] = np.clip(params[1::3], 0.01, 0.99)
        params[2::3] = np.maximum(params[2::3], 0.005)

        result = optimize_once(params, n, constraints, bounds, max_iter=2000)
        if result is not None:
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_params = result.x.copy()

    # Strategy 2: Grid-based with various seeds
    for seed in range(15):
        params = create_grid_init(n, seed=seed)
        rng = np.random.RandomState(seed * 200 + 2)
        params += rng.randn(3 * n) * 0.02
        params[0::3] = np.clip(params[0::3], 0.01, 0.99)
        params[1::3] = np.clip(params[1::3], 0.01, 0.99)
        params[2::3] = np.maximum(params[2::3], 0.005)

        result = optimize_once(params, n, constraints, bounds, max_iter=2000)
        if result is not None:
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_params = result.x.copy()

    # Strategy 3: Random initialization with various seeds
    for seed in range(20):
        params = create_random_init(n, seed=seed)
        result = optimize_once(params, n, constraints, bounds, max_iter=2000)
        if result is not None:
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_params = result.x.copy()

    # Strategy 4: Refinement from best found so far with perturbations
    if best_params is not None:
        for seed in range(15):
            rng = np.random.RandomState(seed * 300 + 3)
            params = best_params.copy() + rng.randn(3 * n) * 0.008
            params[0::3] = np.clip(params[0::3], 0.01, 0.99)
            params[1::3] = np.clip(params[1::3], 0.01, 0.99)
            params[2::3] = np.maximum(params[2::3], 0.005)

            result = optimize_once(params, n, constraints, bounds, max_iter=2000)
            if result is not None:
                current_sum = -result.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_params = result.x.copy()

    # Extract results
    centers = np.column_stack([best_params[0::3], best_params[1::3]])
    radii = best_params[2::3]

    return centers, radii, best_sum
