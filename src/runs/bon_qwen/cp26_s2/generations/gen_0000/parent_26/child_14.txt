    y = vars[1::3]
    r = vars[2::3]
    
    # Maximize sum(r) -> Minimize -sum(r)
    score = -np.sum(r)
    
    # Boundary penalties
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    # Violations: r - x > 0, x + r - 1 > 0, etc.
    v1 = r - x
    v2 = x + r - 1
    v3 = r - y
    v4 = y + r - 1
    
    pen_bound = 0.0
    for v in [v1, v2, v3, v4]:
        pos_v = np.maximum(v, 0.0)

            best_radii[i] = min(best_radii[i], best_centers[i,0], 1-best_centers[i,0], best_centers[i,1], 1-best_centers[i,1])
        best_radii[:] = 0.001
        best_sum = 0.026

    return best_centers, best_radii, float(best_sum)
