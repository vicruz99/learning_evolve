import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    """
    Computes a differentiable penalty for overlaps and boundary violations.
    Returns 0.0 if the configuration is valid.
    """
    n = len(radii)
    p = 0.0
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        
        # Boundary penalties
        if x < r: p += (x - r)**2
        elif x > 1-r: p += (x - (1-r))**2
        if y < r: p += (y - r)**2
        elif y > 1-r: p += (y - (1-r))**2
        
        # Pairwise overlap penalties
        for j in range(i+1, n):
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                p += overlap**2
    return p

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26

                idx += 1

    res = minimize(obj_init, centers.flatten(), method='L-BFGS-B', bounds=bounds_init, options={'maxiter': 1000, 'ftol': 1e-14})
    centers = res.x.reshape(n, 2)
    
    # 3. Iterative radius expansion with center adjustment
    step = 0.0006
    for _ in range(2000):

            return compute_penalty(vars.reshape(n, 2), radii)
            
        bounds_cur = [(r, 1-r) for r in radii for _ in range(2)]
        res = minimize(obj, centers.flatten(), method='L-BFGS-B', bounds=bounds_cur, options={'maxiter': 400, 'ftol': 1e-14})
        centers = res.x.reshape(n, 2)
        
        pen = compute_penalty(centers, radii)
        
        # Adaptive step control
        if pen > 1e-5:
            radii -= step
            step *= 0.96

                break

            centers += np.random.normal(0, 1e-5, centers.shape)

        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1-r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1-r)
        radii[i] = np.clip(radii[i], 1e-6, 0.49)
        
    return centers, radii, float(np.sum(radii))
