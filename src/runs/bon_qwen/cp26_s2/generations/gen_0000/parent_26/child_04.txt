import numpy as np
from scipy.optimize import minimize

def calculate_sum(centers, radii):
    """Calculate the sum of radii."""
    return np.sum(radii)

def validate_and_penalty(centers, radii):
    """
    Validates the packing and returns a penalty score.
    Returns 0 if valid, positive value if constraints are violated.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary constraints
    # x - r >= 0  => x >= r
    # x + r <= 1  => 1 - x >= r
    # Same for y
    r = radii
    x, y = centers[:, 0], centers[:, 1]
    
    # Check boundaries
    boundary_violations = np.maximum(0, r - x) + np.maximum(0, r - (1 - x)) + \
                          np.maximum(0, r - y) + np.maximum(0, r - (1 - y))
    penalty += 1000.0 * np.sum(boundary_violations)
    
    # Overlap constraints
    # dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    
    # Vectorized overlap check for efficiency
    # Compute pairwise distances squared
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    dist_sq = np.sum(diff**2, axis=2) # (n, n)
    
    # Sum of radii matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :] # (n, n)
    
    # Constraint: dist_sq >= r_sum^2
    violation = np.maximum(0, r_sum**2 - dist_sq)
    
    # We only care about i < j, but summing all and dividing by 2 or just summing all is fine for penalty
    # Diagonal is 0, so it's fine.
    penalty += 1000.0 * np.sum(violation)
    
    return penalty

def objective_function(vars_flat, n):
    """
    Objective function for optimization.
    Variables: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    """
    centers = vars_flat[0:2*n].reshape(n, 2)
    radii = vars_flat[2*n:]
    
    # Ensure non-negative radii (soft constraint handled by bounds usually, but here in obj)
    if np.any(radii < 0):
        return 1e6 # Huge penalty
    
    # We want to maximize sum of radii, so minimize negative sum
    # Add penalty for violations
    penalty = validate_and_penalty(centers, radii)
    
    obj_val = -np.sum(radii) + penalty
    
    return obj_val

def get_gradient(vars_flat, n):
    """Numerical gradient approximation."""
    eps = 1e-7
    grad = np.zeros_like(vars_flat)
    base_obj = objective_function(vars_flat, n)
    
    for i in range(len(vars_flat)):
        vars_plus = vars_flat.copy()
        vars_plus[i] += eps
        grad[i] = (objective_function(vars_plus, n) - base_obj) / eps
    return grad
