import numpy as np
from scipy.optimize import minimize
from functools import partial


def objective_fn(x, n):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraint_x_min(x, i, n):
    return x[3 * i] - x[3 * i + 2]


def constraint_x_max(x, i, n):
    return 1.0 - x[3 * i] - x[3 * i + 2]


def constraint_y_min(x, i, n):
    return x[3 * i + 1] - x[3 * i + 2]


def constraint_y_max(x, i, n):
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def constraint_radius(x, i, n):
    return x[3 * i + 2]


def constraint_non_overlap(x, i, j, n):
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    r_sum = x[3 * i + 2] + x[3 * j + 2]
    return dx * dx + dy * dy - r_sum * r_sum


def create_hexagonal_initial(n):
    """Create hexagonal grid initial packing."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    n_rows = 7
    n_cols = 5
    s_x = 0.18
    s_y = s_x * np.sqrt(3) / 2

    idx = 0
    for row in range(n_rows):
        for col in range(n_cols):
            if idx >= n:
                break
            x = (col + 1) * s_x + 0.05
            y = row * s_y + 0.1
            if row % 2 == 1:
                x += s_x / 2
            x = max(0.05, min(0.95, x))
            y = max(0.05, min(0.95, y))
            centers[idx] = [x, y]
            radii[idx] = 0.04
            idx += 1
        if idx >= n:
            break

    min_dist = np.inf
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            min_dist = min(min_dist, dist)
    initial_r = min_dist / 2 * 0.85
    radii[:] = initial_r
    for i in range(n):
        r = min(radii[i], centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = r * 0.95
    return centers, radii


def create_boundary_biased_initial(n):
    """Create initial packing biased toward boundaries for larger circles."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0

    corner_positions = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    for pos in corner_positions:
        if idx < n:
            centers[idx] = pos
            radii[idx] = 0.1
            idx += 1

    edge_positions = [
        [0.3, 0.06], [0.5, 0.06], [0.7, 0.06],
        [0.06, 0.3], [0.06, 0.5], [0.06, 0.7],
        [0.94, 0.3], [0.94, 0.5], [0.94, 0.7],
        [0.3, 0.94], [0.5, 0.94], [0.7, 0.94],
    ]
    for pos in edge_positions:
        if idx < n:
            centers[idx] = pos
            radii[idx] = 0.06
            idx += 1

    interior_positions = [
        [0.3, 0.3], [0.5, 0.3], [0.7, 0.3],
        [0.3, 0.5], [0.5, 0.5], [0.7, 0.5],
        [0.3, 0.7], [0.5, 0.7], [0.7, 0.7],
        [0.4, 0.4], [0.6, 0.4], [0.4, 0.6], [0.6, 0.6],
        [0.5, 0.1], [0.1, 0.5], [0.9, 0.5], [0.5, 0.9],
    ]
    for pos in interior_positions:
        if idx < n:
            centers[idx] = pos
            radii[idx] = 0.04
            idx += 1

    for i in range(n):
        r = min(radii[i], centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = r * 0.9
    return centers, radii


def create_dense_grid_initial(n):
    """Create dense grid initial packing."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx >= n:
                break
            x = (j + 0.5) / 5 + 0.05
            y = (i + 0.5) / 6 + 0.03
            x = max(0.05, min(0.95, x))
            y = max(0.05, min(0.95, y))
            centers[idx] = [x, y]
            radii[idx] = 0.05
            idx += 1
        if idx >= n:
            break
    for i in range(n):
        r = min(radii[i], centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = r * 0.9
    return centers, radii


def optimize_from_initial(n, centers, radii, seed=None):
    """Run optimization from a given initial configuration."""
    if seed is not None:
        np.random.seed(seed)
        perturbation = np.random.randn(n, 2) * 0.015
        centers = centers + perturbation
        centers = np.clip(centers, 0.02, 0.98)

    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]

    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': partial(constraint_x_min, i=i, n=n)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_x_max, i=i, n=n)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_y_min, i=i, n=n)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_y_max, i=i, n=n)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_radius, i=i, n=n)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': partial(constraint_non_overlap, i=i, j=j, n=n)})

    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    result = minimize(objective_fn, x0, method='SLSQP',
                      constraints=constraints, bounds=bounds,
                      options={'maxiter': 20000, 'ftol': 1e-15})

    return result


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26

    best_sum = 0
    best_x = None

    # Try multiple initial configurations
    initial_configs = [
        create_hexagonal_initial(n),
        create_boundary_biased_initial(n),
        create_dense_grid_initial(n),
    ]

    for trial, (centers, radii) in enumerate(initial_configs):
        for seed in [None, 42 + trial]:
            result = optimize_from_initial(n, centers, radii, seed=seed)
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_x = result.x.copy()

    if best_x is None:
        centers, radii = create_hexagonal_initial(n)
        return centers, radii, np.sum(radii)

    centers_opt = np.zeros((n, 2))
    radii_opt = np.zeros(n)
    for i in range(n):
        centers_opt[i, 0] = best_x[3 * i]
        centers_opt[i, 1] = best_x[3 * i + 1]
        radii_opt[i] = best_x[3 * i + 2]

    # Post-processing: fix any tiny violations
    radii_opt = np.maximum(radii_opt, 0)
    for i in range(n):
        r = radii_opt[i]
        centers_opt[i, 0] = max(r, min(1 - r, centers_opt[i, 0]))
        centers_opt[i, 1] = max(r, min(1 - r, centers_opt[i, 1]))

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
