import numpy as np
from scipy.optimize import minimize
import math

def objective_func(v):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(v[2::3])

def constraint_func(v):
    """Constraint function: returns array of values that must be >= 0"""
    n = 26
    X = v[0::3]
    Y = v[1::3]
    R = v[2::3]
    
    c_list = []
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_list.extend(X - R)
    c_list.extend(1.0 - X - R)
    c_list.extend(Y - R)
    c_list.extend(1.0 - Y - R)
    
    # Pairwise distance constraints: ||c_i - c_j||^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        xi, yi, ri = X[i], Y[i], R[i]
        for j in range(i + 1, n):
            dx = xi - X[j]
            dy = yi - Y[j]
            c_list.append(dx*dx + dy*dy - (ri + R[j])**2)
            
    return np.array(c_list)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii)
    """
    np.random.seed(42)
    n = 26
    best_v = None
    best_obj = -1e9
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Generate initial guesses
    inits = []
    
    # 1. Hexagonal lattice configuration
    centers_hex = []
    r_init = 0.09
    rows = [6, 5, 6, 5, 4]  # 26 circles total
    y_pos = r_init
    for i, cnt in enumerate(rows):
        x_start = r_init + (r_init if i % 2 == 1 else 0)
        for _ in range(cnt):
            centers_hex.append([x_start, y_pos])
            x_start += 2 * r_init
        y_pos += math.sqrt(3) * r_init
        
    v0 = np.zeros(3 * n)
    for i, c in enumerate(centers_hex):
        v0[3 * i] = c[0]
        v0[3 * i + 1] = c[1]
        v0[3 * i + 2] = r_init
    inits.append(v0)
    
    # 2. Perturbed hexagonal configuration
    v0_pert = v0.copy() + np.random.normal(0, 0.015, size=3 * n)
    v0_pert[0::3] = np.clip(v0_pert[0::3], 0.0, 1.0)
    v0_pert[1::3] = np.clip(v0_pert[1::3], 0.0, 1.0)
    v0_pert[2::3] = np.clip(v0_pert[2::3], 0.02, 0.5)
    inits.append(v0_pert)
    
    # 3. Random configuration
    v0_rand = np.random.uniform(0.2, 0.8, size=2 * n)
    v0_rand_full = np.zeros(3 * n)
    v0_rand_full[0::3] = v0_rand[0::2]
    v0_rand_full[1::3] = v0_rand[1::2]
    v0_rand_full[2::3] = 0.05
    inits.append(v0_rand_full)
    
    # Optimization loop over initial guesses
    for v_init in inits:
        res = minimize(
            objective_func, 
            v_init, 
            method='SLSQP', 
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraint_func},
            options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
        )
        if res.fun < best_obj:
            best_obj = res.fun
            best_v = res.x
            
    # Extract results
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    
    # Ensure strict feasibility against numerical tolerances
    radii = np.maximum(radii, 1e-7)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
