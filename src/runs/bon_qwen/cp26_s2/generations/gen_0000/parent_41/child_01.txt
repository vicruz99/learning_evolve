import numpy as np
from scipy.optimize import linprog, differential_evolution

def solve_radii_lp(centers):
    """
    Given fixed centers, solve for the maximum radii such that circles 
    do not overlap and stay within the unit square.
    Returns (sum_radii, radii_array).
    """
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]

    # Variables: r_1, ..., r_n
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c_obj = np.ones(n) * -1.0

    # Constraints: A_ub * r <= b_ub
    # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # 2. Non-overlap constraints: r_i + r_j <= dist(i, j)

    # Pre-allocate matrices for efficiency
    # 4 constraints per circle for boundaries
    # n*(n-1)/2 constraints for pairwise distances
    num_constraints = 4 * n + (n * (n - 1)) // 2
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    row = 0
    
    # Boundary constraints
    for i in range(n):
        # r_i <= x_i
        A_ub[row, i] = 1.0
        b_ub[row] = x[i]
        row += 1
        
        # r_i <= 1 - x_i
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - x[i]
        row += 1
        
        # r_i <= y_i
        A_ub[row, i] = 1.0
        b_ub[row] = y[i]
        row += 1
        
        # r_i <= 1 - y_i
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - y[i]
        row += 1

    # Non-overlap constraints
    # r_i + r_j <= ||c_i - c_j||
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dist
            row += 1

    # Bounds for radii: 0 <= r_i
    # Upper bound can be 0.5 (max possible radius in unit square) or infinity.
    # 0.5 is safe.
    bounds = [(0.0, 0.5) for _ in range(n)]

    # Solve LP
    # Using 'highs' method for speed and reliability
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            sum_radii = -res.fun
            return sum_radii, radii
        else:
            # If LP fails, return a valid but small solution (e.g., all 0)
            # This shouldn't happen often with valid centers
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for optimization.
    Input: flattened array of centers (x1, y1, x2, y2, ...)
    Output: Negative sum of radii (to minimize)
    """
    n = 26
    centers = centers_flat.reshape((n, 2))
    sum_radii, _ = solve_radii_lp(centers)
    return -sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Runs the optimization to pack 26 circles in a unit square maximizing sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # We perform multiple runs of Differential Evolution to increase chances of finding global optimum
    # Differential Evolution is good for non-smooth, multi-modal landscapes.
    
    # Bounds for variables: x in [0, 1], y in [0, 1]
    # Variable vector length is 2 * n = 52
    bounds_opt = [(0.0, 1.0)] * (2 * n)

    # Configuration parameters
    popsize = 20
    maxiter = 500 # Adjust based on time limit, 500 is reasonable for this size
    atol = 1e-6
    tol = 1e-6

    # Try a few random seeds
    for seed in range(3):
        try:
            res = differential_evolution(
                objective_function,
                bounds_opt,
                popsize=popsize,
                maxiter=maxiter,
                tol=tol,
                atol=atol,
                seed=seed,
                workers=1, # Use single worker to avoid multiprocessing issues in some environments
                updating='deferred',
                mutation=(0.5, 1.5),
                recombination=0.7
            )
            
            current_sum = -res.fun

                best_sum = current_sum
                best_centers = res.x.reshape((n, 2))
                # Re-solve LP to get precise radii for the best centers found

        except Exception as e:
            print(f"Optimization run {seed} failed: {e}")
