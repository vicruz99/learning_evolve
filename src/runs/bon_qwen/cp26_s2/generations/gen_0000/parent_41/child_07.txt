import numpy as np
from scipy.optimize import minimize


def objective(params, n):
    """Objective: maximize sum of radii (minimize negative sum)."""
    radii = params[2 * n:]
    return -np.sum(radii)


def all_constraints(params, n):
    """All inequality constraints: >= 0 for feasibility."""
    result = []
    
    # Boundary constraints: each circle inside [0,1]^2
    for i in range(n):
        x = params[2 * i]
        y = params[2 * i + 1]
        r = params[2 * n + i]
        result.append(x - r)           # r <= x
        result.append(1.0 - x - r)     # r <= 1 - x
        result.append(y - r)           # r <= y
        result.append(1.0 - y - r)     # r <= 1 - y
    
    # Overlap constraints: circles don't overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = params[2 * i] - params[2 * j]
            dy = params[2 * i + 1] - params[2 * j + 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-20)
            result.append(dist - params[2 * n + i] - params[2 * n + j])
    
    return np.array(result)


def init_hexagonal(n, row_sizes):
    """Initialize circles in a hexagonal pattern."""
    r = 0.075
    centers = []
    y = r
    for row_idx, ncols in enumerate(row_sizes):
        if row_idx % 2 == 0:
            # Even row: start at r
            x_start = r
        else:
            # Odd row: offset by r for hexagonal packing
            x_start = r + r
        
        for k in range(ncols):
            x = x_start + k * 2.0 * r
            centers.append([x, y])
        
        y += r * np.sqrt(3)
    
    centers = np.array(centers[:n])
    radii = np.ones(n) * r
    
    # Clamp centers to valid range
    centers[:, 0] = np.clip(centers[:, 0], r, 1.0 - r)
    centers[:, 1] = np.clip(centers[:, 1], r, 1.0 - r)
    
    return centers, radii


def init_spread(n):
    """Initialize circles spread across the square."""
    # Use a structured layout: try to cover the square evenly
    # 6 columns, 5 rows pattern adjusted for 26 circles
    
    row_configs = [
        (5, 0),   # 5 circles, no offset
        (6, 0.5), # 6 circles, half offset
        (5, 0),   # 5 circles, no offset

        (4, 0),   # 4 circles, no offset
    ]
    
    r = 0.08
    centers = []
    y = r
    for row_idx, (ncols, offset) in enumerate(row_configs):
        x_start = r + offset * r
        spacing = (1.0 - 2 * r - offset * r) / (ncols - 1) if ncols > 1 else 0
        for k in range(ncols):
            x = x_start + k * spacing
            centers.append([x, y])
        y += r * np.sqrt(3)
    
    centers = np.array(centers[:n])
    radii = np.ones(n) * r
    
    centers[:, 0] = np.clip(centers[:, 0], r, 1.0 - r)
    centers[:, 1] = np.clip(centers[:, 1], r, 1.0 - r)
    
    return centers, radii


def try_optimize(centers, radii, n, maxiter=5000):
    """Try to optimize packing from given initial configuration."""
    params = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
        bounds.append((0.0, 0.5))
    
    constraint_def = {
        'type': 'ineq',
        'fun': all_constraints,
        'args': (n,)
    }
    
    try:
        result = minimize(
            objective,
            params,
            args=(n,),
            bounds=bounds,
            constraints=[constraint_def],
            method='SLSQP',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False}
        )
        
        if result.success or result.fun < -0.1:
            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            
            # Ensure radii are non-negative
            radii_opt = np.maximum(radii_opt, 1e-10)
            
            # Ensure centers are valid
            for i in range(n):
                radii_opt[i] = min(radii_opt[i], 
                                   centers_opt[i, 0], 
                                   1.0 - centers_opt[i, 0],
                                   centers_opt[i, 1],
                                   1.0 - centers_opt[i, 1])
                centers_opt[i, 0] = np.clip(centers_opt[i, 0], radii_opt[i], 1.0 - radii_opt[i])
                centers_opt[i, 1] = np.clip(centers_opt[i, 1], radii_opt[i], 1.0 - radii_opt[i])
            
            return centers_opt, radii_opt, float(np.sum(radii_opt))
    except Exception:
        pass
    
    return None

def fix_overlaps(centers, radii, n):
    """Fix any remaining numerical overlaps."""
    for _ in range(100):
        max_violation = 0.0
        max_pair = (-1, -1)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > max_violation:
                    max_violation = overlap
                    max_pair = (i, j)
        
        if max_violation < 1e-10:
            break
        
        i, j = max_pair
        # Reduce radii slightly to fix overlap
        reduction = max_violation / 2.0 + 1e-8
        radii[i] = max(radii[i] - reduction, 1e-10)
        radii[j] = max(radii[j] - reduction, 1e-10)
        
        # Also ensure boundary constraints
        for k in range(n):
            radii[k] = min(radii[k],
                           centers[k, 0],
                           1.0 - centers[k, 0],
                           centers[k, 1],
                           1.0 - centers[k, 1])
    
    return centers, radii
