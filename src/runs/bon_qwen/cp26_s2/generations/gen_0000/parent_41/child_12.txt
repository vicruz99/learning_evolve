import numpy as np
from scipy.optimize import minimize
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Penalty weight for constraint violations
    # High weight ensures constraints are respected over objective improvement
    PENALTY_WEIGHT = 5000.0 

    def objective_and_penalty(vars_flat):
        """
        Calculates the negative sum of radii plus penalty for constraint violations.
        vars_flat: array of shape (3*n) containing [x0, y0, r0, x1, y1, r1, ...]
        """
        centers = vars_flat.reshape(-1, 3)[:, :2]
        radii = vars_flat.reshape(-1, 3)[:, 2]
        
        # Ensure radii are non-negative for calculation stability (though bounds handle it)
        radii = np.maximum(radii, 0.0)

        obj_val = -np.sum(radii)
        penalty = 0.0

        # 1. Boundary Constraints
        # x >= r  => x - r >= 0
        # x <= 1-r => x + r <= 1
        # y >= r  => y - r >= 0
        # y <= 1-r => y + r <= 1
        
        x = centers[:, 0]
        y = centers[:, 1]
        r = radii

        # Boundary violations
        viol_left = np.maximum(0.0, r - x)
        viol_right = np.maximum(0.0, x + r - 1.0)
        viol_bottom = np.maximum(0.0, r - y)
        viol_top = np.maximum(0.0, y + r - 1.0)

        penalty += PENALTY_WEIGHT * np.sum(viol_left**2 + viol_right**2 + viol_bottom**2 + viol_top**2)

        # 2. Overlap Constraints
        # dist(i, j) >= r_i + r_j
        # We check all pairs
        # Vectorized pairwise distance calculation might be heavy for n=26 (325 pairs), 
        # but manageable.
        
        # Centers shape (n, 2)
        # Pairwise diffs
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
        dists = np.sqrt(np.sum(diff**2, axis=2)) # (n, n)
        
        # Radii sums matrix
        radii_sums = r[:, np.newaxis] + r[np.newaxis, :] # (n, n)
        
        # Overlap violation: if dist < r1 + r2, violation is (r1+r2) - dist
        # We only care about upper triangle (i < j)
        overlap_violations = np.maximum(0.0, radii_sums - dists)
        
        # Sum of squared violations for unique pairs
        # Mask upper triangle
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        penalty += PENALTY_WEIGHT * np.sum(overlap_violations[mask]**2)

        return obj_val + penalty

    def get_bounds(n):
        # x in [0, 1], y in [0, 1], r in [0, 0.5]
        # 0.5 is max possible radius
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
        return bounds

    bounds = get_bounds(n)

    def validate_and_store(centers, radii):
        nonlocal best_sum, best_centers, best_radii
        
        # Check for NaN
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False

        # Check basic bounds
        if np.any(radii < 0):
            return False
            
        # Check boundary strictly
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-7 or x + r > 1.0 + 1e-7 or \
               y - r < -1e-7 or y + r > 1.0 + 1e-7:
                return False
        
        # Check overlaps strictly
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                if dist < radii[i] + radii[j] - 1e-7:
                    return False
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            return True
        return False

    def optimize_from_init(centers_init, radii_init):
        # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
        vars_init = np.zeros(3 * n)
        for i in range(n):
            vars_init[3*i] = centers_init[i, 0]
            vars_init[3*i+1] = centers_init[i, 1]
            vars_init[3*i+2] = radii_init[i]
        
        # Run optimization
        # L-BFGS-B is good for bounded problems
        res = minimize(objective_and_penalty, vars_init, method='L-BFGS-B', bounds=bounds, 
                       options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 5000})
        
        if res.success or (res.fun < 0): # res.fun is negative sum + penalty
            # Extract result
            res_centers = res.x.reshape(-1, 3)[:, :2]
            res_radii = res.x.reshape(-1, 3)[:, 2]
            
            # If penalty is low, check if valid
            # Calculate penalty separately to be sure
            # (Simplified check: if sum of radii is reasonable and no huge penalty)
            # Actually, just validate directly
            if validate_and_store(res_centers, res_radii):
                pass # Stored

    # --- Initialization Strategies ---

    # 1. Hexagonal-like Packing
    # Try to fit 26 circles in a staggered grid
    # Rows of 5 and 6 might work. 5, 6, 5, 6, 4 sum to 26? No.
    # 5, 5, 5, 5, 6 is 26.
    # Let's try a generic hex grid generation.
    def init_hexagonal():
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.08 # Start with reasonable radius
        idx = 0
        row = 0
        while idx < n:
            y = row * 0.1732 + 0.1 # approximate hex spacing
            # Shift x for odd rows
            x_start = 0.1
            if row % 2 == 1:
                x_start = 0.2
            
            col = 0
            while idx < n:
                x = x_start + col * 0.2
                if x <= 0.9: # Keep within bounds roughly
                    centers[idx] = [x, y]
                    idx += 1
                    col += 1
                else:
                    break
            row += 1
        # Perturb slightly to avoid symmetry lock
        centers += np.random.uniform(-0.005, 0.005, centers.shape)
        # Clamp
        centers = np.clip(centers, 0.05, 0.95)
        return centers, radii

    # 2. Grid 5x5 + Perturbation
    def init_grid():
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.09
        idx = 0
        for r in range(5):
            for c in range(5):
                if idx < n:
                    centers[idx] = [0.1 + c*0.2, 0.1 + r*0.2]
                    idx += 1
        if idx < n:
            # Add 26th circle in a gap, e.g., center of a void
            # Void at (0.2, 0.2) relative to grid?
            # Grid points: (0.1, 0.1), (0.3, 0.1)...
            # Gap center at (0.2, 0.2)
            centers[idx] = [0.2, 0.2]
            idx += 1
        # Add random noise
        centers += np.random.uniform(-0.02, 0.02, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        return centers, radii

    # 3. Random Low Density
    def init_random():
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        radii = np.ones(n) * 0.05
        return centers, radii

    # 4. Halton Sequence (Quasi-random)
    def init_halton():
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.08
        
        def halton_sequence(number, base):
            result = 0.0
            f = 1.0
        i = number
            while i > 0:
                f /= base
                result += f * (i % base)
                i = int(i / base)
            return result

        for i in range(n):
            centers[i, 0] = halton_sequence(i + 2, 2)
            centers[i, 1] = halton_sequence(i + 3, 3)
            # Scale to [0.1, 0.9]
            centers[i, 0] = 0.1 + centers[i, 0] * 0.8
            centers[i, 1] = 0.1 + centers[i, 1] * 0.8
            
        return centers, radii

    # Run optimizations from different starts
    # Hexagonal
    c, r = init_hexagonal()
    optimize_from_init(c, r)
    
    # Grid
    c, r = init_grid()
    optimize_from_init(c, r)
    
    # Halton
    c, r = init_halton()
    optimize_from_init(c, r)
    
    # Random restarts
    for _ in range(10):
        c, r = init_random()
        optimize_from_init(c, r)
        
    # If no valid packing found (unlikely with penalty method but possible if stuck),
    # return a safe fallback.
    if best_centers is None:
        # Fallback: small circles in a grid
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.04
        idx = 0
        for y in np.linspace(0.1, 0.9, 6):
            for x in np.linspace(0.1, 0.9, 5):
                if idx < n:
                    centers[idx] = [x, y]
                    idx += 1
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, float(best_sum)
