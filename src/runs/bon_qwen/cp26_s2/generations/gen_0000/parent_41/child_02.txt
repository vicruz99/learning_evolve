To maximize the sum of radii for 26 circles in a unit square, we employ a computational optimization approach. We use a penalty-based method with L-BFGS-B to balance the objective of increasing radii against the physical constraints of non-overlapping and boundary limits.

1.  **Initial Placement (Hexagonal Lattice):** We initialize the circles in a hexagonal pattern. This arrangement is known to be highly efficient for packing density, providing a superior starting point compared to a simple grid.
2.  **Penalty Function:** We define a loss function that includes the negative sum of radii (to maximize it) plus a weighted sum of squared penalties for any constraint violations (overlap or boundary protrusion).
3.  **Continuation Method:** We iteratively increase the penalty weight. This "anneals" the optimization, first allowing the circles to find optimal positions and then strictly enforcing the constraints as the radii expand.
4.  **Multiple Restarts:** To avoid getting stuck in local optima, we run the optimization multiple times with different random perturbations and initial configurations, selecting the highest valid sum of radii.

import numpy as np
from scipy.optimize import minimize
import math

# Set a fixed seed for reproducibility in local testing, but allow variety in restarts
RANDOM_SEED = 42

def generate_hex_grid(n, scale=0.9, offset=(0.05, 0.05)):
    """Generates a hexagonal grid of n points scaled to fit in a unit square."""
    rows = math.ceil(math.sqrt(n * math.sqrt(3) / 2))
    points = []
    count = 0
    y = 0
    while count < n:
        row_points = math.ceil(math.sqrt(n))
        for x in range(row_points):
            if count >= n:
                break
            shift = (y / (math.sqrt(3) / 2)) % 2
            # Normalized coordinates before scaling
            px = x + shift * 0.5
            py = y * math.sqrt(3) / 2
            points.append([px, py])
            count += 1
        y += 1

    # Normalize to fit in [0, 1] x [0, 1] with some margin
    if not points:
        return np.zeros((0, 2))
    
    pts = np.array(points)
    max_dim = max(np.max(pts[:, 0]), np.max(pts[:, 1]))
    if max_dim == 0: max_dim = 1
    
    # Center and scale
    pts = (pts / max_dim) * scale + offset
    pts = np.clip(pts, 0.05, 0.95) # Ensure margins
    return pts[:n]

def compute_loss_and_grad(params, n, penalty_factor):
    """
    Computes the objective function (negative sum of radii) + penalty for constraints.
    Also returns the gradient.
    """
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    # 1. Objective: Maximize sum of radii -> Minimize -sum
    loss = -np.sum(radii)
    grad_loss = np.zeros_like(params)
    grad_loss[2 * n:] = -1.0

    # 2. Penalties and Gradients
    penalty = 0.0
    grad_penalty = np.zeros_like(params)

    # A. Overlap Constraints
    for i in range(n):
        for j in range(i + 1, n):
            # Vector from j to i
            v = centers[i] - centers[j]
            dist = np.linalg.norm(v)
            sum_r = radii[i] + radii[j]
            
            if dist < sum_r and dist > 1e-9:
                # Penalty: (sum_r - dist)^2
                overlap = sum_r - dist
                penalty += overlap * overlap
                
                # Gradient for centers (w.r.t x_i, y_i)
                # d/d(centers[i]) (-dist) = - (v / dist) * overlap * 2 * overlap ? 
                # Penalty P = (S - D)^2. dP/dCi = 2(S-D) * (-dD/dCi)
                # dD/dCi = v / D
                term = 2 * overlap * (v / dist)
                grad_penalty[2 * i] += term[0]
                grad_penalty[2 * i + 1] += term[1]
                grad_penalty[2 * j] -= term[0]
                grad_penalty[2 * j + 1] -= term[1]
                
                # Gradient for radii
                # dP/dr_i = 2(S-D) * 1
                grad_penalty[2 * n + i] += 2 * overlap
                grad_penalty[2 * n + j] += 2 * overlap

    # B. Boundary Constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # x - r >= 0  => penalty if x < r
        if x < r:
            diff = r - x
            penalty += diff * diff
            grad_penalty[2 * i] -= 2 * diff # d/dx (r-x)^2 = -2(r-x)
            grad_penalty[2 * n + i] += 2 * diff # d/dr (r-x)^2 = 2(r-x)
            
        # x + r <= 1 => penalty if x + r > 1
        if x + r > 1:
            diff = x + r - 1
            penalty += diff * diff
            grad_penalty[2 * i] += 2 * diff
            grad_penalty[2 * n + i] += 2 * diff
            
        # y - r >= 0
        if y < r:
            diff = r - y
            penalty += diff * diff
            grad_penalty[2 * i + 1] -= 2 * diff
            grad_penalty[2 * n + i] += 2 * diff
            
        # y + r <= 1
        if y + r > 1:
            diff = y + r - 1
            penalty += diff * diff
            grad_penalty[2 * i + 1] += 2 * diff
            grad_penalty[2 * n + i] += 2 * diff

    total_loss = loss + penalty_factor * penalty
    total_grad = grad_loss + penalty_factor * grad_penalty
    
    return total_loss, total_grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_loss = float('inf')
    best_centers = None
    best_radii = None
    
    # Number of restarts
    n_restarts = 15
    
    for k in range(n_restarts):
        # 1. Generate Initial Configuration
        # Use hex grid with slight random perturbation
        rng = np.random.RandomState(RANDOM_SEED + k)
        centers_init = generate_hex_grid(n, scale=0.8, offset=(0.1, 0.1))
        
        # Perturb centers
        centers_init += rng.normal(0, 0.02, centers_init.shape)
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        # Initial radii (small to ensure valid start)
        radii_init = np.full(n, 0.01)
        
        params_init = np.concatenate([centers_init.flatten(), radii_init])
        
        # 2. Continuation Optimization
        # Start with low penalty to find position, increase to enforce constraints
        penalty_values = [10.0, 100.0, 1000.0, 10000.0]
        
        current_params = params_init.copy()
        
        for penalty in penalty_values:
            # Define function for scipy
            def obj_and_grad(x):
                return compute_loss_and_grad(x, n, penalty)
            
            # Bounds: x, y in [0, 1], r in [0, 0.5]
            bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
            
            res = minimize(
                obj_and_grad,
                current_params,
                method='L-BFGS-B',
                jac=True,
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-6}
            )
            current_params = res.x
            
            # If optimization failed or didn't converge well, we might stop, 
            # but usually it finds a local minimum.
            if res.success and res.fun < best_loss:
                # Validate validity before updating best
                centers_cand = current_params[:2 * n].reshape(n, 2)
                radii_cand = current_params[2 * n:]
                
                # Simple validation check
                valid = True
                for i in range(n):
                    if radii_cand[i] < 0: valid = False; break
                    x, y = centers_cand[i]
                    if x - radii_cand[i] < -1e-5 or x + radii_cand[i] > 1 + 1e-5:
                        valid = False; break
                    if y - radii_cand[i] < -1e-5 or y + radii_cand[i] > 1 + 1e-5:
                        valid = False; break
                
                if valid:
                    # Check overlaps
                    for i in range(n):
                        for j in range(i + 1, n):
                            d = np.linalg.norm(centers_cand[i] - centers_cand[j])
                            if d < radii_cand[i] + radii_cand[j] - 1e-5:
                                valid = False; break
                        if not valid: break
                
                if valid:
                    sum_r = np.sum(radii_cand)
                    # We want to maximize sum_r, so minimize -sum_r
                    if -sum_r < best_loss:
                        best_loss = -sum_r
                        best_centers = centers_cand
                        best_radii = radii_cand

    if best_centers is None:
        # Fallback to a basic grid if optimization failed completely
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.0)
        # Try to place in grid
        count = 0
        for i in range(6):
            for j in range(6):
                if count >= n: break
                centers[count] = [0.1 + i * 0.18, 0.1 + j * 0.18]
                radii[count] = 0.09
                count += 1
    else:
        centers = best_centers
        radii = best_radii

    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
