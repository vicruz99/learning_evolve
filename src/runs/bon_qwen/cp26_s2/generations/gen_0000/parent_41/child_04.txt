import numpy as np
from scipy.optimize import differential_evolution, linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    """
    n = 26
    
    # 1. Define the objective function for center placement.
    # We want to maximize the minimum distance (to boundaries and other centers).
    def objective(center_vector):
        centers = center_vector.reshape(n, 2)
        
        # Distance to boundaries (0 and 1)
        # A circle at (x,y) with radius r must satisfy: r <= x, r <= 1-x, r <= y, r <= 1-y
        # So the max possible radius limited by walls is min(x, 1-x, y, 1-y)
        wall_dists = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                                np.minimum(centers[:, 1], 1 - centers[:, 1]))
        min_wall_dist = np.min(wall_dists)
        
        # Distance between centers. For equal circles of radius r, distance must be >= 2r.
        # So r <= dist / 2.
        # We compute pairwise distances using broadcasting
        # centers: (n, 2)
        # diffs: (n, n, 2)
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        
        # Mask out diagonal (distance to self)
        dists = dists + np.eye(n) * 1e12
        min_pair_dist = np.min(dists)
        
        # The radius is limited by walls (r) and neighbors (r <= dist/2)
        # We return the maximum possible radius for this configuration
        return min(min_wall_dist, min_pair_dist / 2.0)

    # 2. Run Differential Evolution to find optimal centers
    # Bounds for x and y are [0, 1]
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Optimization settings
    # Population size and iterations are tuned for 52 variables.
    # 'seed' ensures reproducibility.
    result = differential_evolution(
        objective,
        bounds,
        maxiter=2000,
        popsize=20,
        mutation=(0.5, 1.5),
        recombination=0.9,
        seed=42,
        polish=True,
        disp=True
    )

    optimal_centers = result.x.reshape(n, 2)
    
    # 3. Refine radii using Linear Programming
    # Given fixed centers, we maximize sum(r_i) subject to non-overlap and boundary constraints.
    
    # Compute distance matrix
    c_x = optimal_centers[:, 0]
    c_y = optimal_centers[:, 1]
    
    # Pairwise distances
    dist_matrix = np.sqrt(np.sum((optimal_centers[:, np.newaxis, :] - optimal_centers[np.newaxis, :, :])**2, axis=2))
    
    # Construct LP: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints: r_i + r_j <= dist_ij
    # We only need upper triangular pairs to avoid redundancy
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist_matrix[i, j]
            idx += 1
            
    # Add boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = optimal_centers[i]
        dist_wall = min(x, 1-x, y, 1-y)
        A_ub = np.vstack([A_ub, np.zeros((1, n))])
        A_ub[-1, i] = 1.0
        b_ub = np.append(b_ub, dist_wall)

    # Bounds for r_i: [0, 1] (radius can't exceed 1)
    bounds_r = [(0, 1)] * n
    
    # Solve LP
    lp_res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if lp_res.success:
        radii = -lp_res.fun # Sum is negative of min value
        # Extract individual radii
        final_radii = -lp_res.x
        # Note: linprog minimizes c^T x. c was -1s. So it minimizes -sum(r).
        # x contains the radii.
    else:
        # Fallback: if LP fails, use equal radii based on center optimization
        r_val = result.fun
        final_radii = np.full(n, r_val)
        radii = np.sum(final_radii)

    # Re-validate and clean up potential tiny numerical overlaps
    # Ensure radii are non-negative
    final_radii = np.maximum(final_radii, 0.0)
    
    # Final sanity check and return
    # The sum is the target value
    return optimal_centers, final_radii, float(np.sum(final_radii))
