import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initial Placement: Hexagonal lattice with 5 rows
    # Row counts: 5, 6, 5, 6, 4 (Sum = 26)
    row_counts = [5, 6, 5, 6, 4]
    
    centers = []
    y_curr = 0.0
    h = 0.0  # Temporary vertical spacing
    
    # We will refine the radius r during optimization, so we start with a safe estimate.
    # Let's assume r approx 0.1. Vertical spacing in hex packing is r*sqrt(3).
    # We will initialize centers and let the optimizer adjust them.
    
    # To ensure a valid starting point, let's pick a small r_init
    r_init = 0.08
    v_space = r_init * np.sqrt(3)
    
    current_y = r_init
    
    for idx, count in enumerate(row_counts):
        # Calculate x positions for this row
        # Width of row = 2*r + (count-1)*2*r = 2*r*count
        # We center the row horizontally
        row_width = 2 * r_init * count
        x_start = (1.0 - row_width) / 2 + r_init
        
        for j in range(count):
            x = x_start + j * (2 * r_init)
            centers.append([x, current_y])
        
        current_y += v_space

    centers = np.array(centers)
    radii = np.full(n, r_init)

    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Shape: (n * 3,)
    
    def objective(x_flat):
        # Maximize sum of radii => Minimize negative sum of radii
        r = x_flat[2::3]
        return -np.sum(r)

    def penalty(x_flat):
        penalty_val = 0.0
        r = x_flat[2::3]
        c = np.column_stack((x_flat[0::3], x_flat[1::3]))
        
        # Boundary constraints: 0 <= x-r, x+r <= 1, 0 <= y-r, y+r <= 1
        # If violated, add large penalty
        # Using a squared hinge loss for smoothness
        
        # Left wall
        d_left = r - c[:, 0]
        penalty_val += np.sum(np.maximum(0, d_left)**2)
        
        # Right wall
        d_right = c[:, 0] + r - 1.0
        penalty_val += np.sum(np.maximum(0, d_right)**2)
        
        # Bottom wall
        d_bottom = r - c[:, 1]
        penalty_val += np.sum(np.maximum(0, d_bottom)**2)
        
        # Top wall
        d_top = c[:, 1] + r - 1.0
        penalty_val += np.sum(np.maximum(0, d_top)**2)
        
        # Overlap constraints: dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # Penalty if dist < r_i + r_j
        
        # Vectorized distance calculation
        # c: (n, 2), r: (n,)
        # diffs: (n, n, 2)
        diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists_sq = np.sum(diffs**2, axis=2)
        
        # radii sum matrix (n, n)
        rad_sums = r[:, np.newaxis] + r[np.newaxis, :]
        rad_sums_sq = rad_sums**2
        
        # We only care about i < j to avoid double counting and self
        # But simple penalty over all pairs (excluding diagonal) is fine
        # dist_sq - rad_sum_sq < 0 implies overlap
        
        violation = rad_sums_sq - dists_sq
        np.fill_diagonal(violation, 0) # No self overlap
        # Only upper triangle
        violation = np.triu(violation, k=1)
        
        penalty_val += np.sum(np.maximum(0, violation)**2)
        
        return penalty_val

    def cost_func(x_flat):
        return objective(x_flat) + 1000.0 * penalty(x_flat)

    # Initial state vector
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    # Bounds for optimizer
    # x, y in [0, 1]
    # r in [0, 0.5] (max possible radius)
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # 3. Run Optimizer
    # L-BFGS-B handles bounds and is generally efficient for medium-sized problems
    res = minimize(cost_func, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-6})

    # Extract results
    final_centers = np.column_stack((res.x[0::3], res.x[1::3]))
    final_radii = res.x[2::3]
    
    # Ensure non-negative radii (should be handled by bounds, but just in case)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Adjust centers to be strictly within bounds if needed (clamping)
    # Though penalty should have handled it, clamping prevents numerical edge cases
    final_centers[:, 0] = np.clip(final_centers[:, 0], final_radii, 1.0 - final_radii)
    final_centers[:, 1] = np.clip(final_centers[:, 1], final_radii, 1.0 - final_radii)

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
