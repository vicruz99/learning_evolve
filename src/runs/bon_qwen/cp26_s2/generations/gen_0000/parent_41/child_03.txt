import numpy as np
from scipy.optimize import minimize, linprog

def generate_initial_centers(n):
    """
    Generates n points in a hexagonal lattice pattern within [0,1]x[0,1].
    """
    points = []
    # Hexagonal lattice parameters
    spacing_x = 1.0 / np.sqrt(n * np.sqrt(3) / 2)
    spacing_y = spacing_x * np.sqrt(3) / 2
    
    # Generate a grid of points larger than needed
    grid_points = []
    for i in range(int(1.5 * np.sqrt(n)) + 2):
        for j in range(int(1.5 * np.sqrt(n)) + 2):
            x = i * spacing_x + (j % 2) * spacing_x / 2
            y = j * spacing_y
            if 0 <= x <= 1 and 0 <= y <= 1:
                grid_points.append([x, y])
    
    # If we have enough points, select the first n
    if len(grid_points) >= n:
        centers = np.array(grid_points[:n])
    else:
        # Fallback to random placement if grid generation fails
        centers = np.random.rand(n, 2)

    # Center and scale the points to fit tightly in [0,1]
    min_x, min_y = np.min(centers, axis=0)
    max_x, max_y = np.max(centers, axis=0)
    
    # Scale factor to fit in [0.1, 0.9] initially to avoid boundaries
    scale_x = 0.8 / (max_x - min_x) if (max_x - min_x) > 1e-6 else 1.0
    scale_y = 0.8 / (max_y - min_y) if (max_y - min_y) > 1e-6 else 1.0
    scale = min(scale_x, scale_y)
    
    centers = (centers - min_x, min_y) * scale + np.array([0.1, 0.1])
    return centers

def get_max_sum_radii(centers):
    """
    Given fixed centers, solve an LP to maximize the sum of radii.
    """
    n = centers.shape[0]
    m = 4 * n + n * (n - 1) // 2
    
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
    
    # Overlap constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    c_obj = -np.ones(n) # Maximize sum -> Minimize negative sum
    bounds = [(0, None) for _ in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        return -1.0, None

def run_packing():
    n = 26
    centers = generate_initial_centers(n)
    
    def objective(x_flat):
        c = np.clip(x_flat.reshape(-1, 2), 0, 1) # Ensure centers stay in [0,1]
        s, _ = get_max_sum_radii(c)
        return -s # Minimize negative sum
    
    x0 = centers.reshape(-1)
    
    # Use Nelder-Mead for non-smooth objective landscape
    res = minimize(objective, x0, method='Nelder-Mead', 
                   options={'maxiter': 3000, 'xatol': 1e-6, 'fatol': 1e-9})
    
    final_centers = np.clip(res.x.reshape(-1, 2), 0, 1)
    sum_r, final_radii = get_max_sum_radii(final_centers)
    
    return final_centers, final_radii, sum_r
