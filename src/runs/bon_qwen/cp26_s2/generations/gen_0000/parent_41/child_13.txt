import numpy as np
from scipy.optimize import minimize


def compute_penalized_objective(params, n, weight):
    """Compute negative sum of radii plus penalty for constraint violations."""
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]
    
    obj = -np.sum(radii)
    
    # Boundary violation penalties
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        v_left = max(0.0, r - x)
        v_right = max(0.0, r - (1.0 - x))
        v_bottom = max(0.0, r - y)
        v_top = max(0.0, r - (1.0 - y))
        
        obj += weight * (v_left * v_left + v_right * v_right + v_bottom * v_bottom + v_top * v_top)
    
    # Overlap violation penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                obj += weight * overlap * overlap
    
    return obj


def initialize_hexagonal(n):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    
    rows = 6
    cols = 5
    
    row_spacing = 0.14
    col_spacing = 0.17
    
    for row in range(rows):
        n_in_row = cols if row % 2 == 0 else cols - 1
        for col in range(n_in_row):
            if idx >= n:
                break
            offset = 0.5 * col_spacing if row % 2 == 1 else 0.0
            x = 0.08 + col * col_spacing + offset
            y = 0.08 + row * row_spacing
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    
    return centers


def initialize_grid(n):
    """Initialize circles in a regular grid pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    
    rows = 6
    cols = 5
    
    x_spacing = 1.0 / (cols + 1)
    y_spacing = 1.0 / (rows + 1)
    
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = (col + 1) * x_spacing
            y = (row + 1) * y_spacing
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    
    return centers


def initialize_corner_focused(n):

    centers = np.zeros((n, 2))
    idx = 0
    
    # Place 4 circles near corners
    corner_pos = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]

    ]

        if idx < n:
            centers[idx] = pos
            idx += 1
    
    # Fill remaining in center
    remaining = n - idx
    if remaining > 0:
        sub_rows = int(np.ceil(np.sqrt(remaining)))
        sub_cols = int(np.ceil(remaining / sub_rows))

                centers[idx] = [x, y]

                si += 1
            if si >= remaining:
                break
    
    return centers


def force_directed_refine(centers, radii, n, max_iter=3000):
    """Refine configuration using force-directed method."""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(max_iter):
        forces = np.zeros((n, 2))
        lr = 0.005 / (1.0 + iteration / 500.0)
        
        # Repulsion between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                
                target = radii[i] + radii[j] + 0.02
                if dist < target and dist > 1e-10:
                    force_mag = (target - dist) / dist
                    forces[i, 0] += dx * force_mag
                    forces[i, 1] += dy * force_mag
                    forces[j, 0] -= dx * force_mag
                    forces[j, 1] -= dy * force_mag
        
        # Boundary repulsion
        for i in range(n):
            x = centers[i, 0]
            y = centers[i, 1]
            r = radii[i]
            
            margin = 0.02
            if x < r + margin:
                forces[i, 0] += (r + margin - x) * 5.0
            if x > 1.0 - r - margin:
                forces[i, 0] -= (x - (1.0 - r - margin)) * 5.0
            if y < r + margin:
                forces[i, 1] += (r + margin - y) * 5.0
            if y > 1.0 - r - margin:
                forces[i, 1] -= (y - (1.0 - r - margin)) * 5.0
        
        centers = centers + forces * lr
        centers = np.clip(centers, 0.01, 0.99)
        
        # Gradually increase radii
        if iteration % 200 == 0:
            increase = 0.002
            radii = radii + increase
            radii = np.clip(radii, 0.01, 0.5)
    
    return centers, radii


def check_validity(centers, radii, n):
    """Check if configuration is valid."""
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        if r < 0 or np.isnan(r):
            return False
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True


def optimize_from_params(params, n, initial_radii_val):
    """Run full optimization pipeline from initial parameters."""
    centers = params[:2 * n].reshape((n, 2))
    radii = np.full(n, initial_radii_val)
    
    # Force-directed refinement
    centers, radii = force_directed_refine(centers, radii, n, max_iter=2000)
    
    # Prepare for gradient optimization
    opt_params = np.concatenate([centers.flatten(), radii])
    bounds = [(0.01, 0.99)] * (2 * n) + [(0.005, 0.5)] * n
    
    # Progressive penalty optimization
    for weight in [5000, 50000, 200000]:
        result = minimize(
            compute_penalized_objective,
            opt_params,
            args=(n, weight),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        opt_params = result.x
    
    opt_centers = opt_params[:2 * n].reshape((n, 2))
    opt_radii = opt_params[2 * n:]
    
    # Final small radii adjustment: if any radius is too small, clip it
    opt_radii = np.maximum(opt_radii, 0.001)
    
    return opt_centers, opt_radii
