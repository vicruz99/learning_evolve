import numpy as np
from scipy.optimize import minimize


def compute_penalized_objective(vars, n, penalty):
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]

    obj = -np.sum(radii)

    # Boundary violation penalties
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]

        # x - r >= 0
        viol = r - x
        if viol > 0:
            obj += penalty * viol * viol

        # x + r <= 1
        viol = x + r - 1
        if viol > 0:
            obj += penalty * viol * viol

        # y - r >= 0
        viol = r - y
        if viol > 0:
            obj += penalty * viol * viol

        # y + r <= 1
        viol = y + r - 1
        if viol > 0:
            obj += penalty * viol * viol

        # r >= 0
        viol = -r
        if viol > 0:
            obj += penalty * viol * viol

    # Overlap violation penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            if dist_sq < 1e-20:
                dist_sq = 1e-20
            dist = np.sqrt(dist_sq)
            min_dist = radii[i] + radii[j]
            viol = min_dist - dist
            if viol > 0:
                obj += penalty * viol * viol

    return obj


def init_hexagonal(n):
    vars = np.zeros(3 * n)
    
    # Strategy: 6+5+6+5+4 = 26 rows in hexagonal pattern
    rows = [6, 5, 6, 5, 4]
    
    h = 1.0 / 7.0
    v = h * np.sqrt(3.0) / 2.0
    
    idx = 0
    for row_idx, count in enumerate(rows):
        y = v * (row_idx + 1)
        
        if row_idx % 2 == 0:
            x_start = h
        else:
            x_start = 1.5 * h
        
        row_width = (count - 1) * h
        total_row_width = row_width + 2 * x_start
        offset = (1.0 - total_row_width) / 2.0
        
        for c in range(count):
            if idx >= n:
                break
            x = x_start + c * h + offset
            r = h * 0.42
            
            vars[idx] = x
            vars[idx + n] = y
            vars[idx + 2 * n] = r
            idx += 1
        if idx >= n:
            break
    
    # Fill remaining if any
    while idx < n:
        vars[idx] = 0.5
        vars[idx + n] = 0.5
        vars[idx + 2 * n] = 0.05
        idx += 1
    
    return vars


def init_grid(n):
    vars = np.zeros(3 * n)
    
    cols = 6
    rows = 5
    spacing_x = 1.0 / (cols + 1)
    spacing_y = 1.0 / (rows + 1)
    
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n:
                break
            x = spacing_x * (j + 1)
            y = spacing_y * (i + 1)
            
            if i % 2 == 1:
                x += spacing_x * 0.5
            
            x = np.clip(x, 0.05, 0.95)
            r = min(spacing_x, spacing_y) * 0.42
            
            vars[idx] = x
            vars[idx + n] = y
            vars[idx + 2 * n] = r
            idx += 1
        if idx >= n:
            break
    
    while idx < n:
        vars[idx] = 0.5
        vars[idx + n] = 0.5
        vars[idx + 2 * n] = 0.05
        idx += 1
    
    return vars


def init_random(n, seed):
    vars = np.zeros(3 * n)
    rng = np.random.RandomState(seed)
    
    for i in range(n):
        vars[i] = rng.uniform(0.1, 0.9)
        vars[i + n] = rng.uniform(0.1, 0.9)
        vars[i + 2 * n] = 0.05
    
    return vars


def is_valid_packing(centers, radii, n):
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        if np.isnan(r) or r < 0:
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True
