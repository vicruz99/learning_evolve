import numpy as np
from scipy.optimize import minimize
import math

def objective(params, n):
    radii = params[n * 2:]
    return -np.sum(radii)

def constraints(params, n):
    centers = params[:n * 2].reshape(n, 2)
    radii = params[n * 2:]
    
    result = []
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        result.append(x - r)
        result.append(1 - x - r)
        result.append(y - r)
        result.append(1 - y - r)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            min_dist = radii[i] + radii[j]
            result.append(dist_sq - min_dist * min_dist)
    
    return np.array(result)

def make_initial_config_1(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0
    
    corner_coords = [[0.22, 0.22], [0.78, 0.22], [0.22, 0.78], [0.78, 0.78]]
    for coord in corner_coords:
        centers[idx] = coord
        radii[idx] = 0.22
        idx += 1
    
    edge_coords = [
        [0.5, 0.08], [0.5, 0.92],

        [0.92, 0.32], [0.92, 0.68],

def make_initial_config_2(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0
    
    hex_spacing = 0.18
    for row in range(6):
        for col in range(5):
            if idx < n:
                x = 0.1 + col * hex_spacing + (row % 2) * (hex_spacing / 2)
                y = 0.1 + row * hex_spacing * 0.866
                centers[idx] = [x, y]
                
                if row in [0, 5] and col in [0, 4]:
                    radii[idx] = 0.12
                elif row in [0, 5] or col in [0, 4]:
                    radii[idx] = 0.07
                else:
                    radii[idx] = 0.04
                idx += 1
    
    return centers, radii

def force_relaxation(centers, radii, n, iterations=600):
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(iterations):
        grow_factor = 1.005
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            max_r = min(x, 1 - x, y, 1 - y)
            for j in range(n):
                if i != j:
                    dx = x - centers[j, 0]
                    dy = y - centers[j, 1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            
            radii[i] = min(max_r, r * grow_factor)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                if dist < 1e-10:
                    dist = 1e-10
                    dx = 1e-10
                
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    nx = dx / dist
                    ny = dy / dist
                    push = overlap * 0.51
                    centers[i, 0] -= nx * push
                    centers[i, 1] -= ny * push
                    centers[j, 0] += nx * push
                    centers[j, 1] += ny * push
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            move = 0.003
            if x - r > 0.002:
                centers[i, 0] -= move
            if x + r < 0.998:
                centers[i, 0] += move
            if y - r > 0.002:
                centers[i, 1] -= move
            if y + r < 0.998:
                centers[i, 1] += move
    
    return centers, radii

def scipy_optimize(centers, radii, n):
    params = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0, 1)] * (n * 2) + [(0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': lambda p: constraints(p, n)}
    obj = lambda p: objective(p, n)
    
    try:
        result = minimize(obj, params, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
        if result.success and not np.isnan(result.x).any():
            return result.x[:n * 2].reshape(n, 2), result.x[n * 2:]
    except:
        pass
    
    return centers, radii

def ensure_validity(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        r = max(0, r)
        x = max(r, min(1 - r, x))
        y = max(r, min(1 - r, y))
        r = min(r, x, 1 - x, y, 1 - y)
        centers[i] = [x, y]
        radii[i] = max(0, r)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j]:
                overlap = (radii[i] + radii[j]) - dist
                if dist < 1e-10:
                    dist = 1e-10
                    dx = 1e-10
                nx = dx / dist
                ny = dy / dist
                scale = overlap * 0.51
                centers[i, 0] -= nx * scale
                centers[i, 1] -= ny * scale
                centers[j, 0] += nx * scale
                centers[j, 1] += ny * scale
    
    return centers, radii

def run_packing():
    n = 26
    
    best_sum = 0
    best_centers = None
    best_radii = None
    
    configs = [
        make_initial_config_1(n),

        centers_final, radii_final = ensure_validity(centers_scipy, radii_scipy, n)
        
        current_sum = np.sum(radii_final)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_final.copy()
            best_radii = radii_final.copy()
    
    if best_centers is None:
        best_centers, best_radii = force_relaxation(make_initial_config_1(n), n, iterations=1000)
        best_centers, best_radii = ensure_validity(best_centers, best_radii, n)
        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
