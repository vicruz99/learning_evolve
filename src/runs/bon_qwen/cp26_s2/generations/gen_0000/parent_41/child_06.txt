import numpy as np
from scipy.optimize import minimize, linprog


def run_packing():
    n = 26
    
    centers = init_hexagonal(n)
    radii = np.full(n, 0.08)
    
    # Alternating optimization
    for iteration in range(80):
        # Step 1: Optimize radii given positions (LP)
        radii = optimize_radii_lp(centers, n)
        
        # Step 2: Optimize positions given radii (NLP - maximize separation)
        centers = optimize_positions_slsqp(centers, radii, n)
    
    total = np.sum(radii)
    
    # Final safety clip
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        centers[i, 0] = np.clip(x, r, 1 - r)
        centers[i, 1] = np.clip(y, r, 1 - r)
    
    return centers, radii, total


def init_hexagonal(n):
    centers = np.zeros((n, 2))
    idx = 0
    
    # Arrange in roughly hexagonal pattern: 5 rows with 5,5,5,5,6 = 26 circles
    row_counts = [5, 5, 5, 5, 6]
    
    for row in range(5):
        ncols = row_counts[row]
        y = (row + 0.5) / 5.0
        for col in range(ncols):
            if idx >= n:
                break
            x = (col + 0.5) / ncols
            # Offset odd rows for hexagonal packing
            if row % 2 == 1:
                x += 0.05
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    
    return centers


def optimize_radii_lp(centers, n):
    """
    Given fixed positions, solve LP to maximize sum of radii.
    
    Maximize sum(r_i)
    s.t. r_i + r_j <= dist(i,j) for all i < j
         r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
         r_i >= 0
    """
    c_obj = -np.ones(n)  # Minimize negative sum = maximize sum
    
    A_ub = []
    b_ub = []
    
    # Overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A_ub.append(row)
            b_ub.append(dist)
    
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x_i = centers[i, 0]
        y_i = centers[i, 1]
        bounds_list = [x_i, 1.0 - x_i, y_i, 1.0 - y_i]
        for bound in bounds_list:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(max(0.0, bound))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0.0, None) for _ in range(n)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: equal small radii
        return np.full(n, 0.05)


def optimize_positions_slsqp(centers, radii, n):
    """
    Given fixed radii, optimize positions to maximize total pairwise separation.
    This creates space for potentially larger radii in the next LP step.
    """
    x0 = centers.flatten()
    
    def objective(x):
        c = x.reshape(n, 2)
        obj = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                obj += dist
        return -obj  # Minimize negative = maximize total separation
    
    def constraints(x):
        c = x.reshape(n, 2)
        res = []
        
        # Overlap constraints: dist(i,j) >= r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                res.append(dist - radii[i] - radii[j])
        
        # Boundary constraints
        for i in range(n):
            r = radii[i]
            res.append(c[i, 0] - r)        # x >= r
            res.append(1.0 - c[i, 0] - r)  # x <= 1-r
            res.append(c[i, 1] - r)        # y >= r
            res.append(1.0 - c[i, 1] - r)  # y <= 1-r
        
        return np.array(res)
    
    cons = {'type': 'ineq', 'fun': constraints}
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    
    result = minimize(objective, x0, method='SLSQP', constraints=cons,
                      bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-14})
    
    return result.x.reshape(n, 2)
