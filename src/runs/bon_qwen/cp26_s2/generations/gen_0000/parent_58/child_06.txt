import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
            centers: np.array of shape (26, 2)
            radii: np.array of shape (26,)
            sum_radii: float
    """
    n = 26
    
    # Function to generate initial positions based on hexagonal lattice
    def get_initial_config(seed=0):
        rng = np.random.RandomState(seed)
        
        # Hexagonal lattice parameters
        # We want to fit roughly 26 circles. A 6x5 grid or similar is good.
        # Let's generate a dense hex grid and select 26 points.
        
        # Hex grid generation
        # Basis vectors
        v1 = np.array([1.0, 0.0])
        v2 = np.array([0.5, np.sqrt(3)/2])
        
        # Generate points in a bounding box slightly larger than [0,1] to have choices
        # We scale the lattice to fit roughly 5-6 circles per side
        scale = 0.2 # approximate spacing
        
        points = []
        # Try a range of lattice indices
        for i in range(-2, 10):
            for j in range(-2, 10):
                p = i * v1 * scale + j * v2 * scale
                # Check if inside [0, 1] x [0, 1] with some margin for radius
                if 0.1 <= p[0] <= 0.9 and 0.1 <= p[1] <= 0.9:
                    points.append(p)
        
        # If we don't have enough points, fallback to grid
        if len(points) < n:
            grid_x = np.linspace(0.1, 0.9, 6)
            grid_y = np.linspace(0.1, 0.9, 6)
            points = []
            for x in grid_x:
                for y in grid_y:
                    points.append(np.array([x, y]))
        
        # Shuffle and pick n points
        points = np.array(points)
        indices = rng.choice(len(points), size=n, replace=False)
        centers = points[indices]
        
        # Add some random perturbation to break symmetry and help optimization
        # Small perturbation
        perturbation = rng.uniform(-0.02, 0.02, size=(n, 2))
        centers += perturbation
        
        # Ensure centers are within bounds [0.05, 0.95] roughly
        centers = np.clip(centers, 0.05, 0.95)
        
        # Initial radii: small enough to be valid
        radii = np.full(n, 0.05)
        
        return centers, radii

    def objective(vars):
        # vars: [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(r), so minimize -sum(r)
        radii = vars[2*n:]
        return -np.sum(radii)

    def constraints(vars):
        centers = vars[:2*n].reshape(n, 2)
        radii = vars[2*n:]
        cons = []
        
        # Boundary constraints: x-r >= 0, 1-(x+r) >= 0, etc.
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            cons.append(x - r)          # Left
            cons.append(1 - (x + r))    # Right
            cons.append(y - r)          # Bottom
            cons.append(1 - (y + r))    # Top
        
        # Overlap constraints: dist^2 >= (r1 + r2)^2
        # dist^2 - (r1 + r2)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                d_sq = np.sum((centers[i] - centers[j])**2)
                r_sum = radii[i] + radii[j]
                cons.append(d_sq - r_sum**2)
                
        return np.array(cons)

    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
        
    # Run optimization multiple times with different seeds
    num_trials = 5
    
    for trial in range(num_trials):
        try:
            centers_init, radii_init = get_initial_config(seed=trial)
            x0 = np.concatenate([centers_init.ravel(), radii_init])
            
            # Use SLSQP which supports constraints
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': constraints},
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_centers = res.x[:2*n].reshape(n, 2)
                    best_radii = res.x[2*n:]
        except Exception as e:
            continue
            
    # If no solution found or sum is low, fallback to a valid grid solution
    if best_centers is None or best_sum < 2.0: # Fallback threshold
        # Generate a valid 5x5 grid + 1 small circle
        # This is a safe fallback, though suboptimal
        centers_fb = np.array([
            [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
            [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
            [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
            [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
            [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9],
            [0.5, 0.5] # Duplicate center, need to move it
        ])
        # Move last one to a gap
        centers_fb[25] = [0.2, 0.2]
        radii_fb = np.full(26, 0.1)
        radii_fb[25] = 0.04 # Small circle in gap
        best_centers = centers_fb
        best_radii = radii_fb
        best_sum = np.sum(radii_fb)

    return best_centers, best_radii, best_sum
