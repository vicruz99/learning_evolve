def compute_penalty_and_objective(params, n):

    Computes the objective (negative sum of radii) and penalty for constraint violations.
    
    Args:
        params: 1D array of shape (3*n), [x1, y1, r1, x2, y2, r2, ...]
        n: number of circles
        
    Returns:
        loss: float, negative sum of radii + penalty
    """
    # Reshape params
    centers = params.reshape(-1, 3) # (n, 3) where col 0=x, 1=y, 2=r
    x = centers[:, 0]
    y = centers[:, 1]
    r = centers[:, 2]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    obj = -np.sum(r)

    penalty_weight = 1000.0
    
    penalty = 0.0

    # Circles must be inside [0,1]x[0,1]
    # Constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    # Violations: r - x, x + r - 1, r - y, y + r - 1
    
    # Vectorized calculation
    v_x_left = np.maximum(0, r - x)
    v_x_right = np.maximum(0, x + r - 1.0)
    v_y_bottom = np.maximum(0, r - y)
    v_y_top = np.maximum(0, y + r - 1.0)
    
    boundary_penalty = (v_x_left**2 + v_x_right**2 + v_y_bottom**2 + v_y_top**2)
    penalty += np.sum(boundary_penalty) * penalty_weight
    
    # 2. Overlap Constraints
    # Distance between centers >= sum of radii
    # dist^2 >= (r_i + r_j)^2
    # Violation: (r_i + r_j)^2 - dist^2 (if positive)
    # To keep it smooth, we can penalize (r_i + r_j - dist)^2 if dist < r_i + r_j
    
    # Efficient pairwise distance calculation
    # x.shape (n,), compute (x_i - x_j)^2
    # Using broadcasting
    dx = x[:, None] - x[None, :] # (n, n)
    dy = y[:, None] - y[None, :] # (n, n)
    dist_sq = dx**2 + dy**2
    
    # Sum of radii
    r_sum = r[:, None] + r[None, :] # (n, n)
    
    # We only care about i < j, but summing all and dividing by 2 is easier
    # Violation depth: r_sum - sqrt(dist_sq)

    # Better: check if dist_sq < r_sum^2.
    # Penalty term: max(0, r_sum - sqrt(dist_sq))^2

    # Clip dist_sq to avoid negative sqrt (numerical issues)
    dist = np.sqrt(np.maximum(dist_sq, 0))
    
    # Overlap depth
    overlap = r_sum - dist
    
    # Only penalize if overlap > 0
    # np.maximum(0, overlap) gives 0 if dist >= r_sum
    valid_overlap = np.maximum(0, overlap)
    
    # Square the overlap depth
    overlap_penalty = valid_overlap**2

    # Diagonal elements represent self-overlap? No, dist=0, r+r=2r. 
    # But we should exclude i=j.
    
    # Set diagonal to 0 before summing
    np.fill_diagonal(overlap, 0)
    
    penalty += np.sum(overlap**2) * penalty_weight
    
    loss = obj + penalty
    return loss
