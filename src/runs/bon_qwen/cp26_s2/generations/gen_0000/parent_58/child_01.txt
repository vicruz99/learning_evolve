import numpy as np
from scipy.optimize import linprog

def compute_forces(centers, radii, N):
    forces = np.zeros((N, 2))
    k_repulse = 60.0
    k_wall = 100.0
    dt = 0.005
    
    for i in range(N):
        xi, yi = centers[i]
        ri = radii[i]
        
        # Wall repulsion
        if xi - ri < 0.0: forces[i, 0] += k_wall * (ri - xi)
        if xi + ri > 1.0: forces[i, 0] -= k_wall * (xi + ri - 1.0)
        if yi - ri < 0.0: forces[i, 1] += k_wall * (ri - yi)
        if yi + ri > 1.0: forces[i, 1] -= k_wall * (yi + ri - 1.0)
        
        # Circle repulsion
        for j in range(i+1, N):
            dx = xi - centers[j, 0]
            dy = yi - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < 1e-8: dist = 1e-8
            overlap = (ri + radii[j]) - dist
            if overlap > 0:
                fx = k_repulse * overlap * dx / dist
                fy = k_repulse * overlap * dy / dist
                forces[i, 0] += fx
                forces[i, 1] += fy
                forces[j, 0] -= fx
                forces[j, 1] -= fy
    return forces * dt

def simulate_packing(centers, radii, N, steps=1200):
    for step in range(steps):
        # Gradual inflation to push radii towards their limits
        inflation = 1.0 + 0.0002 * np.exp(-step/300.0)
        radii *= inflation
        
        forces = compute_forces(centers, radii, N)
        centers += forces
        
        # Enforce physical bounds
        radii = np.clip(radii, 1e-5, 0.5)
        centers = np.clip(centers, radii, 1.0 - radii)
    return centers, radii

def solve_radii_lp(centers, N):
    # Solve LP: max sum(r_i) s.t. r_i + r_j <= dist(i,j), r_i <= boundary_dist, r_i >= 0
    M = N*(N-1)//2 + 4*N
    A_ub = np.zeros((M, N))
    b_ub = np.zeros(M)
    k = 0
    
    # Pairwise non-overlap constraints
    for i in range(N):
        for j in range(i+1, N):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = dist
            k += 1
            
    # Boundary constraints
    for i in range(N):
        xi, yi = centers[i]
        lim = min(xi, 1.0-xi, yi, 1.0-yi)
        A_ub[k, i] = 1.0
        b_ub[k] = lim
        k += 1
        
    c_obj = -np.ones(N)  # Minimize negative sum => Maximize sum
    bounds = [(0.0, None)] * N
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0.0)
    return np.ones(N) * 0.01

            init_centers[idx] = [x, y]
            idx += 1
        row += 1
        
    # Normalize and center
    init_centers -= init_centers.min(axis=0)
    init_centers /= init_centers.max(axis=0)
    init_centers *= 0.8
    init_centers += 0.1
    
    init_radii = np.full(N, 0.04)
