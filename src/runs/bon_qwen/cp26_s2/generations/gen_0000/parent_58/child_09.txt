import numpy as np
from scipy.optimize import minimize
from functools import partial

# Top-level constraint functions to avoid nested closures or lambdas
def constraint_x_min(x, i):
    return x[3*i] - x[3*i+2]

def constraint_x_max(x, i):
    return 1.0 - x[3*i] - x[3*i+2]

def constraint_y_min(x, i):
    return x[3*i+1] - x[3*i+2]

def constraint_y_max(x, i):
    return 1.0 - x[3*i+1] - x[3*i+2]

def constraint_non_overlap(x, i, j):
    xi, yi, ri = 3*i, 3*i+1, 3*i+2
    xj, yj, rj = 3*j, 3*j+1, 3*j+2
    return (x[xi]-x[xj])**2 + (x[yi]-x[yj])**2 - (x[ri]+x[rj])**2

def objective(x):
    return -np.sum(x[2::3])

def run_packing():
    N = 26
    best_x = None
    best_val = np.inf
    
    # Prepare multiple high-quality starting configurations
    starts = []
    
    # 1. Square grid subset
    gs = np.linspace(0.15, 0.85, 6)
    grid_cfg = []
    for i in range(N):
        grid_cfg.append([gs[i % 6], gs[i // 6]])
    starts.append(np.array(grid_cfg))
    
    # 2. Hexagonal lattice approximation
    hex_cfg = []
    for i in range(N):
        x = 0.15 + (i % 5) * 0.16
        y = 0.15 + (i // 5) * (0.15 * 1.732)
        if (i // 5) % 2 == 1:
            x += 0.08
        hex_cfg.append([x, y])
    starts.append(np.array(hex_cfg))
    
    # 3. Pseudo-random valid configuration
    np.random.seed(42)
    rand_cfg = np.random.rand(N, 2) * 0.5 + 0.25
    starts.append(rand_cfg)
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    for cfg in starts:
        # Start with small valid radii to satisfy constraints initially
        r_init = np.full(N, 0.04)
        x0 = np.hstack([cfg.flatten(), r_init])
        
        # Build constraint list
        cons = []
        for i in range(N):
            cons.append({'type': 'ineq', 'fun': partial(constraint_x_min, i=i)})
            cons.append({'type': 'ineq', 'fun': partial(constraint_x_max, i=i)})
            cons.append({'type': 'ineq', 'fun': partial(constraint_y_min, i=i)})
            cons.append({'type': 'ineq', 'fun': partial(constraint_y_max, i=i)})
        for i in range(N):
            for j in range(i + 1, N):
                cons.append({'type': 'ineq', 'fun': partial(constraint_non_overlap, i=i, j=j)})
                
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'disp': False})
            if res.success and res.fun < best_val:
                best_val = res.fun
                best_x = res.x
        except Exception:
            continue
            
    # Fallback if optimization fails (should not happen with valid starts)
    if best_x is None:
        centers = np.zeros((N, 2))
        for i in range(N):
            centers[i] = [(i % 5) * 0.2 + 0.1, (i // 5) * 0.2 + 0.1]
        radii = np.ones(N) * 0.05
        return centers, radii, np.sum(radii)
        
    centers = best_x[:2 * N].reshape(N, 2)
    radii = best_x[2 * N:]
    return centers, radii, -best_val
