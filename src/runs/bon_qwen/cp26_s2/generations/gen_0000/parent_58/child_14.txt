import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # Helper function to get index in flat vector
    # vector layout: [x1, y1, r1, x2, y2, r2, ...]
    def get_idx(i, dim):
        return i * 3 + dim

    # Objective function: minimize negative sum of radii
    def objective(vars):
        r_vars = vars[2::3]
        return -np.sum(r_vars)

    # Define constraints
    constraints = []

    # Boundary constraints: 
    # x_i >= r_i  => x_i - r_i >= 0
    # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i  => y_i - r_i >= 0
    # y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    
    for i in range(n):
        idx_x = get_idx(i, 0)
        idx_y = get_idx(i, 1)
        idx_r = get_idx(i, 2)
        
        # x >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[get_idx(i, 0)] - v[get_idx(i, 2)]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[get_idx(i, 0)] - v[get_idx(i, 2)]})
        # y >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[get_idx(i, 1)] - v[get_idx(i, 2)]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[get_idx(i, 1)] - v[get_idx(i, 2)]})
        # r >= 0 (handled by bounds usually, but adding constraint for safety if bounds ignored)
        # Actually bounds are better for r >= 0.

    # Non-overlap constraints: distance >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) >= ri + rj
    # Squared form to avoid sqrt in constraint evaluation might be faster, but 
    # scipy handles non-linear fun well. Let's use squared form to be safe with derivatives?
    # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = get_idx(i, 0)
            idx_yi = get_idx(i, 1)
            idx_ri = get_idx(i, 2)
            idx_xj = get_idx(j, 0)
            idx_yj = get_idx(j, 1)
            idx_rj = get_idx(j, 2)
            
            def dist_constraint(v, i=i, j=j):
                dx = v[idx_xi] - v[idx_xj]
                dy = v[idx_yi] - v[idx_yj]
                dr = v[idx_ri] + v[idx_rj]
                return (dx**2 + dy**2) - (dr**2)
            
            constraints.append({'type': 'ineq', 'fun': dist_constraint})

    # Bounds for variables
    # x, y in [0, 1], r in [0, 1] (r can't be > 0.5 practically)
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 1.0)) # r

    # Function to generate initial configuration
    def generate_init(seed=0):
        rng = np.random.RandomState(seed)
        
        # Base layout: 5 rows. One row has 6 circles, others 5. Total 26.
        # Rows configuration
        rows = [5, 5, 6, 5, 5]
        
        centers = []
        y_coords = [0.1, 0.3, 0.5, 0.7, 0.9]
        
        for r_idx, y in enumerate(y_coords):
            count = rows[r_idx]
            if count == 5:
                # 5 circles in row, spaced by 0.2, starting at 0.1
                xs = [0.1 + k * 0.2 for k in range(5)]
            else:
                # 6 circles in row
                # Space evenly. Width 1. Margins small.
                # spacing approx 1/7
                xs = np.linspace(0.05, 0.95, count).tolist()
            
            for x in xs:
                centers.append([x, y])
        
        centers = np.array(centers)
        
        # Add small random noise to centers to break symmetry
        noise_scale = 0.02
        centers += rng.uniform(-noise_scale, noise_scale, size=centers.shape)
        
        # Clip to [0, 1]
        centers = np.clip(centers, 0.0, 1.0)
        
        # Initial radii
        radii = np.full(n, 0.05)
        # Add small noise to radii
        radii += rng.uniform(0, 0.01, size=n)
        
        # Flatten
        x0 = np.hstack([centers.flatten(), radii])
        return x0

    best_result = None
    best_sum = -np.inf

    # Run optimization multiple times with different seeds
    for seed in range(20):
        x0 = generate_init(seed)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success or (res.fun < 0 and res.nit > 10): # Check if we got a valid minimization
                # Extract radii
                r_opt = res.x[2::3]
                sum_r = np.sum(r_opt)
                
                # Validate constraints roughly (allow small tolerance for numerical errors)
                centers_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                
                valid = True
                # Check bounds
                for i in range(n):
                    x, y, r = centers_opt[i, 0], centers_opt[i, 1], r_opt[i]
                    if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
                        valid = False
                        break
                # Check overlaps
                if valid:
                    for i in range(n):
                        for j in range(i+1, n):
                            dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                            if dist < r_opt[i] + r_opt[j] - 1e-6:
                                valid = False
                                break
                        if not valid: break
                
                if valid and sum_r > best_sum:
                    best_sum = sum_r
                    best_result = res
        except Exception as e:
            continue

    if best_result is not None:
        centers = np.column_stack((best_result.x[0::3], best_result.x[1::3]))
        radii = best_result.x[2::3]
        sum_radii = np.sum(radii)
    else:
        # Fallback to a simple grid if optimization fails completely
        # 5x5 grid + 1
        centers = []
        for r in range(5):
            for c in range(5):
                centers.append([0.1 + c*0.2, 0.1 + r*0.2])
        centers.append([0.2, 0.2]) # Just a guess, might overlap
        centers = np.array(centers)
        radii = np.full(26, 0.09) # Conservative
        sum_radii = np.sum(radii)
        
        # Try to fix fallback validity
        # Just return valid simple packing
        # 5 rows of 5 circles is 25. 
        # Let's just return a valid 26 circle packing with small radius
        centers = []
        y_coords = [0.1, 0.3, 0.5, 0.7, 0.9]
        for i, y in enumerate(y_coords):
            if i == 2: # Middle row has 6
                xs = np.linspace(0.08, 0.92, 6)
            else:
                xs = [0.1, 0.3, 0.5, 0.7, 0.9]
            for x in xs:
                centers.append([x, y])
        centers = np.array(centers)
        radii = np.full(26, 0.04)
        sum_radii = 26 * 0.04

    return centers, radii, sum_radii

# Run the packing
centers, radii, sum_radii = run_packing()

# Validation check (local simulation)
def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any(): return False
    if np.isnan(radii).any(): return False
    for i in range(n):
        if radii[i] < 0: return False
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

# Run validation
is_valid = validate_packing(centers, radii)
print(f"Sum of radii: {sum_radii}")
print(f"Is valid: {is_valid}")
