import numpy as np
import math
import scipy.optimize as opt

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # Objective function: Minimize negative sum of radii
    def objective(vars):
        # vars layout: [x0, y0, ..., x25, y25, r0, ..., r25]
        # Actually, let's keep radii separate or interleaved?
        # Let's use: 0..51 -> x, y; 52..77 -> r
        # Or simpler: 3 variables per circle
        # x0, y0, r0, x1, y1, r1 ...
        # This is easier to index.
        # vars shape: (3 * n,)
        
        radii = vars[2::3] # r0, r1, ...
        return -np.sum(radii)

    # Constraints
    def get_constraints():
        constraints = []
        
        # 1. Boundary constraints: 0 <= x-r, x+r <= 1, etc.
        # For circle i:
        # x_i - r_i >= 0  => x_i - r_i >= 0
        # 1 - (x_i + r_i) >= 0 => 1 - x_i - r_i >= 0
        # y_i - r_i >= 0
        # 1 - (y_i + r_i) >= 0
        
        # These are linear constraints.
        # We can define them as functions.
        
        for i in range(n):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_x] - v[idx_r]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_y] - v[idx_r]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
            })
            
        # 2. Non-overlap constraints: dist(i, j) >= r_i + r_j
        # dist(i, j) - r_i - r_j >= 0
        
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = 3 * i, 3 * i + 1, 3 * i + 2
                idx_xj, idx_yj, idx_rj = 3 * j, 3 * j + 1, 3 * j + 2
                
                # Using squared distance might be safer for derivatives?
                # sqrt((xi-xj)^2 + (yi-yj)^2) >= ri + rj
                # This is non-smooth at 0, but we won't be there.
                # However, scipy handles it fine.
                
                def constraint_func(v, i=i, j=j):
                    xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                    xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                    dist = math.hypot(xi - xj, yi - yj)
                    return dist - (ri + rj)
                
                constraints.append({
                    'type': 'ineq',
                    'fun': constraint_func
                })
        
        return constraints

    # Initialization
    # Hexagonal packing initialization
    # We want to fit 26 circles.
    # Let's try to arrange them in rows.
    # Approximate radius 0.1.
    # Hexagonal rows spacing: r * sqrt(3)
    
    r_init = 0.08 # Safe initial radius
    init_vars = np.zeros(3 * n)
    
    count = 0
    row = 0
    while count < n:
        # y coordinate for row
        y = r_init + row * r_init * math.sqrt(3)
        
        # If y + r_init > 1, we might need to shift or stop? 
        # But for initialization, we just place them. 
        # If y is too high, clamp or adjust.
        # Actually, let's just generate a grid that fits roughly.
        
        # Number of circles in this row
        # Width 1. Spacing 2*r.
        # x starts at r_init.
        # If row is even, start at r_init. If odd, shift by r_init?
        # Standard hex: even rows aligned, odd rows shifted.
        
        if row % 2 == 0:
            start_x = r_init
        else:
            start_x = 2 * r_init # Shifted by one radius (diameter/2? no, diameter is 2r, shift is r)
            # Actually shift is r. 
            # Centers at r, 3r, 5r...
            # Shifted centers at 2r, 4r... (shift r)
            # Wait, if first is at r, next is at 3r.
            # Shifted: 2r, 4r. Distance to r is r. Distance to 3r is r.
            # Distance between (r, y) and (2r, y') is sqrt(r^2 + 3r^2) = 2r. Correct.
        
        x = start_x
        while x + r_init <= 1.0 and count < n:
            idx = 3 * count
            init_vars[idx] = x
            init_vars[idx+1] = y
            init_vars[idx+2] = r_init
            
            count += 1
            x += 2 * r_init
        
        row += 1
        
        # Safety break if y gets too high
        if y + r_init > 1.2: # Should not happen for 26 circles
            break

    # Bounds
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Run optimization
    # SLSQP is good for this
    constraints = get_constraints()
    
    # We might run a few times to find a better local optimum
    best_result = None
    best_loss = float('inf')
    
    # Try random perturbations of the initial guess
    np.random.seed(42)
    
    for _ in range(3): # 3 attempts
        current_vars = init_vars.copy()
        # Add small noise
        noise = np.random.normal(0, 0.005, size=current_vars.shape)
        # Ensure radii stay positive and coords in bounds
        current_vars[::3] = np.clip(current_vars[::3] + noise[::3], 0, 1)
        current_vars[1::3] = np.clip(current_vars[1::3] + noise[1::3], 0, 1)
        current_vars[2::3] = np.clip(current_vars[2::3] + noise[2::3], 0.01, 0.4)
        
        try:
            res = opt.minimize(objective, current_vars, method='SLSQP', 
                               bounds=bounds, constraints=constraints, 
                               options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success or res.fun < best_loss:
                best_loss = res.fun
                best_result = res.x
        except Exception:
            continue

    if best_result is None:
        # Fallback to original init if optimization failed
        best_result = init_vars

    # Extract results
    centers = np.array([[best_result[3*i], best_result[3*i+1]] for i in range(n)])
    radii = np.array([best_result[3*i+2] for i in range(n)])
    sum_radii = np.sum(radii)
    
    # Final check to ensure validity (numerical stability)
    # Sometimes constraints might be slightly violated due to tolerance
    # We can clip radii if they go negative (shouldn't happen)
    radii = np.maximum(radii, 0)
    
    return centers, radii, sum_radii
