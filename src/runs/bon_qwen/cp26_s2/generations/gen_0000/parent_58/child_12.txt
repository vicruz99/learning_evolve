To maximize the sum of radii for 26 circles in a unit square, we formulate the problem as a nonlinear optimization task. The objective is to maximize $\sum r_i$ subject to non-overlap and boundary constraints. 

Key steps in the strategy:
1. **Parameterization**: Flatten centers $(x_i, y_i)$ and radii $r_i$ into a single 52-dimensional vector.
2. **Penalty Method**: Since constraints are non-convex and numerous, we use a smooth penalty function added to the negative sum of radii. Overlaps and boundary violations are penalized quadratically. A high penalty weight ensures the optimizer stays within feasible regions.
3. **Multi-Start Optimization**: Circle packing has many local minima. We run `scipy.optimize.minimize` (L-BFGS-B) from 15 different random initial configurations to increase the probability of finding a near-global optimum.
4. **Post-Processing Expansion**: After finding a valid local optimum, we attempt to uniformly scale up all radii by a tiny factor iteratively. This exploits any slack in the constraints and pushes the packing to its tightest configuration.
5. **Validation**: Each candidate solution is checked against the strict validation criteria before updating the best result.

import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def compute_loss(params):
    """Objective function: negative sum of radii + penalty for violations."""
    centers = params[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = params[2*N_CIRCLES:]
    
    # We want to maximize sum of radii -> minimize negative sum
    loss = -np.sum(radii)
    penalty = 0.0
    
    # Penalty for circle overlaps
    for i in range(N_CIRCLES):
        xi, yi = centers[i]
        ri = radii[i]
        for j in range(i + 1, N_CIRCLES):
            dx = xi - centers[j, 0]
            dy = yi - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            required = ri + radii[j]
            if required > dist:
                penalty += (required - dist)**2
                
    # Penalty for boundary violations
    for i in range(N_CIRCLES):
        x, y = centers[i]
        r = radii[i]
        if x < r: penalty += (r - x)**2
        if x > 1.0 - r: penalty += (x - (1.0 - r))**2
        if y < r: penalty += (r - y)**2
        if y > 1.0 - r: penalty += (y - (1.0 - r))**2
        
    # High weight ensures constraints are treated as hard barriers
    return loss + 5000.0 * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Bounds for optimization: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    # Run optimization from multiple random starts to avoid local minima
    for seed in range(20):
        np.random.seed(seed)
        # Initialize centers randomly in the interior, radii moderately sized
        centers = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
        radii = np.ones(N_CIRCLES) * 0.09
        x0 = np.concatenate([centers.flatten(), radii])
        
        # L-BFGS-B is efficient for bound-constrained problems
        res = minimize(
            compute_loss, x0, method='L-BFGS-B', bounds=bounds, 
            options={'maxiter': 10000, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        centers_opt = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
        radii_opt = res.x[2*N_CIRCLES:]
        
        # Quick validation check
        valid = True
        for i in range(N_CIRCLES):
            if radii_opt[i] <= 0:
                valid = False; break
            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < -1e-7 or x + r > 1.0 + 1e-7 or y - r < -1e-7 or y + r > 1.0 + 1e-7:
                valid = False; break
                
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    dx = centers_opt[i, 0] - centers_opt[j, 0]
                    dy = centers_opt[i, 1] - centers_opt[j, 1]
                    if math.sqrt(dx*dx + dy*dy) < radii_opt[i] + radii_opt[j] - 1e-7:
                        valid = False; break
                if not valid: break
                
        if valid:
            s = np.sum(radii_opt)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
                
    # Post-processing: attempt to uniformly scale up radii if slack exists
    if best_centers is not None:
        scale = 1.0001
        for _ in range(100):
            new_radii = best_radii * (1.0 + scale)
            valid = True
            
            # Check boundaries
            for i in range(N_CIRCLES):
                x, y = best_centers[i]
                r = new_radii[i]
                if x - r < 0 or x + r > 1.0 or y - r < 0 or y + r > 1.0:
                    valid = False; break
                    
            if valid:
                # Check overlaps
                for i in range(N_CIRCLES):
                    for j in range(i + 1, N_CIRCLES):
                        dx = best_centers[i, 0] - best_centers[j, 0]
                        dy = best_centers[i, 1] - best_centers[j, 1]
                        if math.sqrt(dx*dx + dy*dy) < new_radii[i] + new_radii[j]:
                            valid = False; break
                    if not valid: break
                    
            if valid:
                best_radii = new_radii
                best_sum = np.sum(best_radii)
            else:
                scale /= 2.0
                if scale < 1e-7:
                    break
                    
    return best_centers, best_radii, best_sum
