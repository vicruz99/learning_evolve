from scipy.optimize import linprog

def run_packing():

    Uses a combination of Linear Programming for radius maximization

        c = -np.ones(n)
        
        A_ub = []
        b_ub = []

                row[j] = 1
                A_ub.append(row)

            row = np.zeros(n)
            row[i] = 1

        # r_i >= 0 (handled by bounds)
        bounds = [(0, None) for _ in range(n)]
        
        A_ub = np.array(A_ub)

        for i in range(n):

            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

    return centers, radii, float(np.sum(radii))
