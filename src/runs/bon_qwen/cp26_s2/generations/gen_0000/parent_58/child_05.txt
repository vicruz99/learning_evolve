import numpy as np
from scipy.optimize import minimize
import warnings

warnings.filterwarnings('ignore')

def compute_penalty_vec(centers, r):
    """Computes the squared overlap penalty for circles of radius r."""
    n = centers.shape[0]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # Overlaps between circles
    overlap = np.maximum(0, 2 * r - dist)
    np.fill_diagonal(overlap, 0)  # Remove self-overlap
    penalty = np.sum(overlap**2)
    
    # Overlaps with boundaries
    # Left & Right walls
    penalty += np.sum(np.maximum(0, r - centers[:, 0])**2)
    penalty += np.sum(np.maximum(0, centers[:, 0] - (1 - r))**2)
    # Bottom & Top walls
    penalty += np.sum(np.maximum(0, r - centers[:, 1])**2)
    penalty += np.sum(np.maximum(0, centers[:, 1] - (1 - r))**2)
    
    return penalty

def objective_func(x, r):

    centers = x.reshape(-1, 2)
    return compute_penalty_vec(centers, r)
