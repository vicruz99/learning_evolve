            if x < r:
                F[i, 0] += k_wall * (r - x)
            # Right wall
            if x > 1 - r:
                F[i, 0] -= k_wall * (x - (1 - r))
            # Bottom wall
            if y < r:
                F[i, 1] += k_wall * (r - y)
            # Top wall
            if y > 1 - r:
                F[i, 1] -= k_wall * (y - (1 - r))
        
        # 2. Inter-circle Repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d2 = dx*dx + dy*dy
                dist = math.sqrt(d2)
                r_sum = radii[i] + radii[j]
                
                if dist < r_sum:
                    # Hard overlap: Strong repulsion
                    overlap = r_sum - dist
                    if dist > 1e-9:
                        fx = (dx / dist) * k_rep * overlap
                        fy = (dy / dist) * k_rep * overlap
                        F[i, 0] += fx
                        F[i, 1] += fy
                        F[j, 0] -= fx
                        F[j, 1] -= fy
                    else:
                        # Avoid division by zero, push apart randomly
                        F[i, 0] += np.random.uniform(-1, 1)
                        F[i, 1] += np.random.uniform(-1, 1)
                        F[j, 0] -= F[i, 0]
                        F[j, 1] -= F[i, 1]
                elif dist < r_sum + 0.15:
                    # Soft repulsion zone: Helps circles settle and avoid jamming
                    # Linear repulsion fading out over distance
                    force_mag = k_rep * (r_sum + 0.15 - dist) / 0.15 * 0.5
                    if dist > 1e-9:
                        fx = (dx / dist) * force_mag
                        fy = (dy / dist) * force_mag
                        F[i, 0] += fx
                        F[i, 1] += fy
                        F[j, 0] -= fx
                        F[j, 1] -= fy
        
        # Integrate Velocity and Position
        v = v * damping + F * dt
        centers += v * dt
        
    # --- Post-processing & Validation ---
    
    # 1. Enforce boundary constraints by shrinking radii if necessary
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        # The maximum radius allowed at (x, y) without crossing boundaries
        max_r_boundary = min(x, 1.0 - x, y, 1.0 - y)
        if r > max_r_boundary:
            radii[i] = max_r_boundary
            
    # 2. Enforce non-overlap constraints by shrinking radii if necessary
    # We iterate to resolve overlaps
    for _ in range(200):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                req_dist = radii[i] + radii[j]
                
                if dist < req_dist - 1e-12:
                    # Overlap detected
                    overlap = req_dist - dist
                    # Shrink both circles proportionally to resolve overlap
                    shrink = overlap / 2.0
                    radii[i] -= shrink
                    radii[j] -= shrink
                    # Ensure non-negative radii
                    if radii[i] < 0: radii[i] = 0.0
                    if radii[j] < 0: radii[j] = 0.0
                    overlap_found = True
        if not overlap_found:
            break
            
    # Calculate final sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
