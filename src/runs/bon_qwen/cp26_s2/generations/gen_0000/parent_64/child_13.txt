import numpy as np
from scipy.optimize import minimize
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Returns centers, radii, and sum_radii.
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # We will try multiple initializations to find the best local optimum
    seeds = [42, 123, 456, 789, 1000, 2023, 3141, 5926, 7890]
    
    for seed in seeds:
        np.random.seed(seed)
        random.seed(seed)
        
        # --- Initialization Strategy: Hexagonal Lattice ---
        # Try to arrange circles in a dense hexagonal pattern
        centers_init = np.zeros((n, 2))
        radii_init = np.full(n, 0.08) # Start with small radii
        
        row_counts = [5, 4, 5, 4, 5, 3] # Sum = 26
        current_idx = 0
        
        # Hexagonal packing geometry
        # Row height is sqrt(3)/2 * diameter
        # We scale to fit in 1x1
        
        y_pos = 0.1
        row_height = 0.1732 # Approx sqrt(3)/2 * 0.2
        
        for row, count in enumerate(row_counts):
            # Offset for odd rows (0-indexed) to create hexagonal nesting
            # If row is odd, shift x by half spacing
            if row % 2 == 1:
                x_start = 0.15 # Shifted
            else:
                x_start = 0.1
            
            spacing = 0.2
            for col in range(count):
                if current_idx < n:
                    x = x_start + col * spacing
                    # Clamp to ensure inside [0,1] roughly
                    x = np.clip(x, 0.0, 1.0)
                    y = np.clip(y_pos, 0.0, 1.0)
                    centers_init[current_idx] = [x, y]
                    current_idx += 1
            
            y_pos += row_height
        
        # If we didn't fill all, fill remaining randomly
        while current_idx < n:
            centers_init[current_idx] = [np.random.rand(), np.random.rand()]
            current_idx += 1

        # Shuffle to avoid symmetry locking
        idx = np.random.permutation(n)
        centers_init = centers_init[idx]

        # --- Optimization ---
        # Variables: [x1, y1, r1, x2, y2, r2, ...]
        # We flatten centers and radii into one vector
        x0 = np.concatenate([centers_init.flatten(), radii_init.flatten()])
        
        # Bounds for variables
        # x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for i in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((1e-5, 0.5)) # r (small positive lower bound)
            
        def objective(vars_flat):
            # Maximize sum of radii -> Minimize negative sum
            r = vars_flat[2::3] # Extract radii: indices 2, 5, 8...
            return -np.sum(r)
            
        def get_centers_radii(vars_flat):
            c = vars_flat[:2*n].reshape(n, 2)
            r = vars_flat[2*n:]
            return c, r

        def constraints(vars_flat):
            c, r = get_centers_radii(vars_flat)
            cons = []
            
            # Boundary constraints: 
            # x >= r, 1-x >= r  =>  x - r >= 0, 1 - x - r >= 0
            # y >= r, 1-y >= r  =>  y - r >= 0, 1 - y - r >= 0
            for i in range(n):
                xi, yi = c[i]
                ri = r[i]
                cons.append(xi - ri)
                cons.append(1.0 - xi - ri)
                cons.append(yi - ri)
                cons.append(1.0 - yi - ri)
                
            # Pairwise non-overlap constraints: dist >= r_i + r_j
            # dist^2 >= (r_i + r_j)^2
            # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
            for i in range(n):
                for j in range(i + 1, n):
                    dx = c[i, 0] - c[j, 0]
                    dy = c[i, 1] - c[j, 1]
                    dist_sq = dx*dx + dy*dy
                    sum_r = r[i] + r[j]
                    cons.append(dist_sq - sum_r*sum_r)
            
            return np.array(cons)

        # SLSQP requires constraints in a specific dict format
        # Since constraints are large array, we can pass a single non-linear constraint
        # But SLSQP handles list of constraints better? 
        # Actually, scipy expects 'type': 'ineq' and 'fun' returning array
        cons_dict = {
            'type': 'ineq',
            'fun': constraints
        }

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons_dict, 
                           options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
            
            if res.success:
                c_opt, r_opt = get_centers_radii(res.x)
                s_opt = np.sum(r_opt)
                
                # Post-processing: Shrink radii slightly to avoid precision issues
                # dist >= r1 + r2 - 1e-12 is allowed.
                # We ensure dist >= r1 + r2 + epsilon
                # A safe scaling factor
                scale = 0.9999
                r_opt = r_opt * scale
                s_opt = np.sum(r_opt)
                
                if s_opt > best_sum:
                    best_sum = s_opt
                    best_centers = c_opt
                    best_radii = r_opt
        except Exception:
            continue

    # Final validation check and adjustment
    if best_centers is not None:
        # Ensure strict validity for the validator
        # The validator allows 1e-12 error, so we are likely safe, 
        # but let's just output the optimized result.
        return best_centers, best_radii, float(np.sum(best_radii))
    else:
        # Fallback to a simple grid if optimization fails
        # 5x5 grid + 1 circle in center? No, 26 circles.
        # Just return a valid small packing
        centers = np.random.rand(26, 2)
        radii = np.full(26, 0.01)
        return centers, radii, 0.26
