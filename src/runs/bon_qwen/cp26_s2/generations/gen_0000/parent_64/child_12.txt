Key insights:
1. A 5×5 grid of equal circles gives radius 0.1, sum = 2.5. Adding one small circle in a gap gives ~2.54.
2. We need to allow unequal radii and optimize jointly - some circles can be larger if others are smaller.
3. The target is 2.636, which requires sum of radii > 2.5, meaning average radius > 0.0977.

Strategy:
1. Start with a 5×5 grid perturbation (25 circles ~ radius 0.1) plus one extra circle
2. Use numerical optimization (SLSQP) to maximize sum of radii with non-overlap and boundary constraints
3. Try multiple random restarts to escape local optima
4. The optimizer will naturally find that some circles can be larger than 0.1 if others shrink, potentially improving the sum

import numpy as np
from scipy.optimize import minimize


def compute_objective_and_constraints(x, n):
    """Compute objective and constraints for circle packing."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    # Objective: negative sum of radii (we minimize)
    obj = -np.sum(radii)
    
    # Constraints
    constraints = []
    
    # Boundary constraints: r_i <= c_i[k] <= 1 - r_i
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]
        # x - r >= 0 => x - r
        constraints.append(x_i - r_i)
        # 1 - x - r >= 0 => 1 - x - r
        constraints.append(1.0 - x_i - r_i)
        # y - r >= 0 => y - r
        constraints.append(y_i - r_i)
        # 1 - y - r >= 0 => 1 - y - r
        constraints.append(1.0 - y_i - r_i)
    
    # Non-negativity of radii
    for i in range(n):
        constraints.append(radii[i])
    
    # Non-overlap constraints: ||c_i - c_j|| - r_i - r_j >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist - radii[i] - radii[j])
    
    return obj, np.array(constraints)


def validate_and_adjust(centers, radii):
    """Validate and adjust radii to ensure feasibility."""
    n = len(radii)
    adjusted_radii = radii.copy()
    
    # Ensure non-negative radii
    adjusted_radii = np.maximum(adjusted_radii, 1e-10)
    
    # Check boundary constraints and adjust
    for i in range(n):
        x, y = centers[i]
        r = adjusted_radii[i]
        max_r = min(x, 1 - x, y, 1 - y, r)
        if max_r < 1e-10:
            max_r = 1e-10
        adjusted_radii[i] = max_r
    
    # Check overlap constraints and adjust
    for iteration in range(50):
        max_violation = 0
        max_pair = (0, 1)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                violation = adjusted_radii[i] + adjusted_radii[j] - dist
                if violation > max_violation:
                    max_violation = violation
                    max_pair = (i, j)
        
        if max_violation < 1e-10:
            break
        
        i, j = max_pair
        reduction = max_violation / 2 + 1e-8
        adjusted_radii[i] = max(adjusted_radii[i] - reduction, 1e-10)
        adjusted_radii[j] = max(adjusted_radii[j] - reduction, 1e-10)
    
    return adjusted_radii


def run_packing():
    """Pack 26 circles in a unit square to maximize sum of radii."""
    n = 26
    best_obj = -np.inf
    best_x = None
    
    # Strategy: Use multiple starting points and optimize
    np.random.seed(42)
    
    # Initial configuration 1: 5x5 grid + 1 circle in gap
    centers_5x5 = []
    radii_5x5 = []
    for i in range(5):
        for j in range(5):
            centers_5x5.append([0.1 + i * 0.2, 0.1 + j * 0.2])
            radii_5x5.append(0.1)
    centers_5x5.append([0.2, 0.2])
    radii_5x5.append(0.035)
    centers_5x5 = np.array(centers_5x5)
    radii_5x5 = np.array(radii_5x5)
    
    x0_5x5 = np.concatenate([centers_5x5.flatten(), radii_5x5])
    
    # Initial configuration 2: Perturbed 5x5 grid
    centers_pert = centers_5x5.copy()
    radii_pert = radii_5x5.copy()
    centers_pert += np.random.normal(0, 0.01, size=centers_pert.shape)
    centers_pert = np.clip(centers_pert, 0.05, 0.95)
    x0_pert = np.concatenate([centers_pert.flatten(), radii_pert])
    
    # Initial configuration 3: Hexagonal-like arrangement
    centers_hex = []
    radii_hex = []
    r_init = 0.095
    d = 2 * r_init
    h = d * np.sqrt(3) / 2
    
    # 5 rows with varying number of circles
    rows = [5, 6, 5, 6, 4]
    y_offset = 0.05
    for row_idx, num_circles in enumerate(rows):
        y = y_offset + row_idx * h
        x_start = 0.05 + (6 - num_circles) * d / 2 if num_circles < 6 else 0.05
        for col_idx in range(num_circles):
            x = x_start + col_idx * d
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers_hex.append([x, y])
                radii_hex.append(r_init)
    
    while len(centers_hex) < n:
        centers_hex.append([0.5, 0.5])
        radii_hex.append(0.05)
    
    centers_hex = np.array(centers_hex[:n])
    radii_hex = np.array(radii_hex[:n])
    x0_hex = np.concatenate([centers_hex.flatten(), radii_hex])
    
    # Initial configuration 4: More spread out
    centers_spread = []
    radii_spread = []
    for i in range(n):
        # Place circles in a quasi-random pattern
        x = (i % 6) * 0.18 + 0.09 + np.random.uniform(-0.02, 0.02)
        y = (i // 6) * 0.22 + 0.11 + np.random.uniform(-0.02, 0.02)
        x = np.clip(x, 0.05, 0.95)
        y = np.clip(y, 0.05, 0.95)
        centers_spread.append([x, y])
        radii_spread.append(0.09)
    
    centers_spread = np.array(centers_spread)
    radii_spread = np.array(radii_spread)
    x0_spread = np.concatenate([centers_spread.flatten(), radii_spread])
    
    initial_configs = [x0_5x5, x0_pert, x0_hex, x0_spread]
    
    # Additional random restarts
    for _ in range(8):
        centers_rand = np.random.uniform(0.15, 0.85, size=(n, 2))
        radii_rand = np.random.uniform(0.04, 0.12, size=n)
        x0_rand = np.concatenate([centers_rand.flatten(), radii_rand])
        initial_configs.append(x0_rand)
    
    # Bounds for variables
    bounds = []
    for i in range(2 * n):
        bounds.append((0.0, 1.0))  # centers
    for i in range(n):
        bounds.append((0.0, 0.5))  # radii
    
    def objective(x):
        obj, _ = compute_objective_and_constraints(x, n)
        return obj
    
    def constraint_func(x):
        _, cons = compute_objective_and_constraints(x, n)
        return cons
    
    # Try each initial configuration
    for x0 in initial_configs:
        try:
            result = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': constraint_func},
                options={'maxiter': 1000, 'ftol': 1e-12}
            )
            
            if result.success or result.fun < -best_obj:
                if -result.fun > best_obj:
                    best_obj = -result.fun
                    best_x = result.x.copy()
        except Exception:
            continue
    
    # Extract best solution
    if best_x is None:
        best_x = x0_5x5
    
    centers = best_x[:2 * n].reshape(n, 2)
    radii = best_x[2 * n:]
    
    # Adjust to ensure feasibility
    radii = validate_and_adjust(centers, radii)
    
    # Re-center if needed
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        centers[i, 0] = np.clip(x, r, 1 - r)
        centers[i, 1] = np.clip(y, r, 1 - r)
    
    # Final validation and adjustment
    radii = validate_and_adjust(centers, radii)
    
    sum_radii = np.sum(radii)
    
    # Ensure no NaN
    centers = np.nan_to_num(centers, nan=0.5)
    radii = np.nan_to_num(radii, nan=0.0)
    radii = np.maximum(radii, 1e-10)
    
    return centers, radii, sum_radii


if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    print(f"Sum of radii: {sum_r:.6f}")
    print(f"Min radius: {np.min(radii):.6f}")
    print(f"Max radius: {np.max(radii):.6f}")
    print(f"Mean radius: {np.mean(radii):.6f}")
