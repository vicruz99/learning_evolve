import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple:
    n = 26

    def objective(params):
        centers = params[:2 * n].reshape(n, 2)
        radii = np.maximum(params[2 * n:], 1e-10)

        obj = -np.sum(radii)
        pw = 500.0

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            v1 = max(0, r - x)
            v2 = max(0, r - (1 - x))
            v3 = max(0, r - y)
            v4 = max(0, r - (1 - y))
            obj += pw * (v1 ** 2 + v2 ** 2 + v3 ** 2 + v4 ** 2)

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    obj += pw * overlap ** 2

        return obj

    def objective_grad(params):
        centers = params[:2 * n].reshape(n, 2)
        radii = np.maximum(params[2 * n:], 1e-10)

        grad = np.zeros(len(params))
        grad[2 * n:] = -1.0
        pw = 500.0

        for i in range(n):
            x, y = centers[i]
            r = radii[i]

            if r > x + 1e-15:
                v = r - x
                grad[2 * n + i] += 2 * pw * v
                grad[2 * i] -= 2 * pw * v

            if r > (1 - x) + 1e-15:
                v = r - (1 - x)
                grad[2 * n + i] += 2 * pw * v
                grad[2 * i] += 2 * pw * v

            if r > y + 1e-15:
                v = r - y
                grad[2 * n + i] += 2 * pw * v
                grad[2 * i + 1] -= 2 * pw * v

            if r > (1 - y) + 1e-15:
                v = r - (1 - y)
                grad[2 * n + i] += 2 * pw * v
                grad[2 * i + 1] += 2 * pw * v

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    grad[2 * n + i] += 2 * pw * overlap
                    grad[2 * n + j] += 2 * pw * overlap
                    if dist > 1e-12:
                        g = 2 * pw * overlap / dist
                        grad[2 * i] -= g * dx
                        grad[2 * i + 1] -= g * dy
                        grad[2 * j] += g * dx
                        grad[2 * j + 1] += g * dy

        return grad

    def grow_radii(centers, radii, n, max_iter=200):
        for iteration in range(max_iter):
            improved = False
            for i in range(n):
                max_r = min(centers[i, 0], 1 - centers[i, 0],
                           centers[i, 1], 1 - centers[i, 1])
                for j in range(n):
                    if i != j:
                        d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 +
                                   (centers[i, 1] - centers[j, 1]) ** 2)
                        max_r = min(max_r, d - radii[j])
                max_r = max(max_r, 0)
                if max_r > radii[i] + 1e-12:
                    radii[i] = max_r
                    improved = True
            if not improved:
                break
        return radii

    def resolve_overlaps(centers, radii, n, max_iter=500):
        for _ in range(max_iter):
            max_overlap = 0
            max_pair = (0, 1)
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 +
                               (centers[i, 1] - centers[j, 1]) ** 2)
                    overlap = radii[i] + radii[j] - d
                    if overlap > max_overlap:
                        max_overlap = overlap
                        max_pair = (i, j)

            if max_overlap < 1e-12:
                break

            i, j = max_pair
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            if d < 1e-15:
                dx, dy, d = 1.0, 0.0, 1.0

            push = max_overlap / 2 + 1e-8
            centers[i, 0] += push * dx / d
            centers[i, 1] += push * dy / d
            centers[j, 0] -= push * dx / d
            centers[j, 1] -= push * dy / d

            centers = np.clip(centers, 0.001, 0.999)

        return centers

    def optimize_from_init(centers_init, radii_init, n, max_iter=30000):
        params = np.concatenate([centers_init.flatten(), radii_init])
        bounds = [(0.005, 0.995)] * (2 * n) + [(0.001, 0.5)] * n

        result = minimize(objective, params, jac=objective_grad,
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})

        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = np.maximum(result.x[2 * n:], 1e-10)
        return centers_opt, radii_opt

    best_sum = -1.0
    best_centers = None
    best_radii = None

    def make_grid_init(seed, perturbation=0.015):
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.09)

        idx = 0
        for i in range(5):
            for j in range(5):
                centers[idx] = [0.1 + i * 0.2 + np.random.randn() * perturbation,
                               0.1 + j * 0.2 + np.random.randn() * perturbation]
                idx += 1
        centers[25] = [0.2 + np.random.randn() * 0.15, 0.2 + np.random.randn() * 0.15]
        radii[25] = 0.03

        centers = np.clip(centers, 0.02, 0.98)
        return centers, radii

    def make_hex_init(seed, perturbation=0.01):
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.07)

        rows = [5, 6, 5, 6, 4]
        idx = 0
        r_est = 0.08
        for row_i, count in enumerate(rows):
            y = 0.05 + row_i * np.sqrt(3) * r_est
            x_offset = r_est * 0.5 if row_i % 2 == 1 else 0
            for col in range(count):
                if idx < n:
                    centers[idx] = [0.05 + x_offset + col * 2 * r_est + np.random.randn() * perturbation,
                                   y + np.random.randn() * perturbation]
                    idx += 1

        centers = np.clip(centers, 0.02, 0.98)
        return centers, radii

    def make_random_init(seed):
        np.random.seed(seed)
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.full(n, 0.04)
        return centers, radii

    # Try many initializations
    init_strategies = [make_grid_init, make_hex_init, make_random_init]

    for strat in init_strategies:
        for p in [0.005, 0.01, 0.02, 0.03, 0.05]:
            for s in range(8):
                centers_init, radii_init = strat(s, p if strat != make_random_init else 0)

                centers_opt, radii_opt = optimize_from_init(centers_init, radii_init, n, max_iter=50000)
                radii_opt = grow_radii(centers_opt, radii_opt, n)
                centers_opt = resolve_overlaps(centers_opt, radii_opt, n)
                radii_opt = grow_radii(centers_opt, radii_opt, n)

                current_sum = np.sum(radii_opt)

                valid = True
                for i in range(n):
                    if radii_opt[i] < 0:
                        valid = False
                        break
                    if (centers_opt[i, 0] - radii_opt[i] < -1e-10 or
                        centers_opt[i, 0] + radii_opt[i] > 1 + 1e-10 or
                        centers_opt[i, 1] - radii_opt[i] < -1e-10 or
                        centers_opt[i, 1] + radii_opt[i] > 1 + 1e-10):
                        valid = False
                        break

                if valid:
                    for i in range(n):
                        for j in range(i + 1, n):
                            d = np.sqrt((centers_opt[i, 0] - centers_opt[j, 0]) ** 2 +
                                       (centers_opt[i, 1] - centers_opt[j, 1]) ** 2)
                            if d < radii_opt[i] + radii_opt[j] - 1e-10:
                                valid = False
                                break
                        if not valid:
                            break

                if valid and current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()

    if best_centers is None:
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        idx = 0
        for i in range(5):
            for j in range(5):
                centers[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
                radii[idx] = 0.1
                idx += 1
        centers[25] = [0.3, 0.3]
        radii[25] = 0.04
        return centers, radii, np.sum(radii)

    centers = best_centers
    radii = best_radii

    # Final aggressive grow
    for _ in range(500):
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0],
                       centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 +
                               (centers[i, 1] - centers[j, 1]) ** 2)
                    max_r = min(max_r, d - radii[j])
            max_r = max(max_r, 0)
            radii[i] = max(radii[i], max_r)

    return centers, radii, float(np.sum(radii))
