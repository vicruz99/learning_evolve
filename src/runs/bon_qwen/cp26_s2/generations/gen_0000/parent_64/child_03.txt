import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # Precompute indices for upper triangle to efficiently handle pairwise constraints
    # We only need to check pairs (i, j) where i < j
    triu_indices = np.triu_indices(n, k=1)
    
    # Objective function: minimize -sum(radii)
    def objective(v):
        # v layout: [x_1...x_n, y_1...y_n, r_1...r_n]
        r = v[2 * n:]
        return -np.sum(r)
    
    # Boundary constraints:
    # x_i >= r_i  => x_i - r_i >= 0
    # 1 - x_i >= r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i => y_i - r_i >= 0
    # 1 - y_i >= r_i => 1 - y_i - r_i >= 0
    def bound_constraints(v):
        x = v[:n]
        y = v[n:2 * n]
        r = v[2 * n:]
        
        c1 = x - r
        c2 = 1.0 - x - r
        c3 = y - r
        c4 = 1.0 - y - r
        
        return np.concatenate([c1, c2, c3, c4])
    
    # Pairwise non-overlap constraints:
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    def pairwise_constraints(v):
        x = v[:n]
        y = v[n:2 * n]
        r = v[2 * n:]
        
        # Extract coordinates for pairs (i, j) with i < j
        xi = x[triu_indices[0]]
        yi = y[triu_indices[0]]
        ri = r[triu_indices[0]]
        
        xj = x[triu_indices[1]]
        yj = y[triu_indices[1]]
        rj = r[triu_indices[1]]
        
        dx = xi - xj
        dy = yi - yj
        dr = ri + rj
        
        # Squared distance minus squared sum of radii
        return dx * dx + dy * dy - dr * dr
    
    constraints = [
        {'type': 'ineq', 'fun': bound_constraints},
        {'type': 'ineq', 'fun': pairwise_constraints}
    ]
    
    # Bounds for variables:
    # x_i, y_i in [0, 1]
    # r_i in [0, 0.5] (radius cannot exceed half the side length)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    best_val = -np.inf
    best_v = None
    
    # Helper to run optimization from a specific initial guess
    def run_opt(v_init):
        nonlocal best_val, best_v
        try:
            res = minimize(objective, v_init, method='SLSQP', 
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
            
            # Check if optimization was successful or found a good value
            # Note: res.success might be False but result might still be valid and good
            # We rely on the objective value and feasibility (implicitly checked by constraints)
            # However, to be safe, we can check if constraints are roughly satisfied
            # But SLSQP tries hard.
            
            val = -res.fun
            if val > best_val:
                best_val = val
                best_v = res.x
        except Exception:
            pass

    # 1. Grid Initialization
    # Create a grid of points inside the square, slightly inset
    # 6 columns, 5 rows = 30 points. We need 26.
    # We can select the first 26 points from a raster scan.
    grid_xs = np.linspace(0.15, 0.85, 6) 
    grid_ys = np.linspace(0.15, 0.85, 5)
    
    grid_points = []
    for y in grid_ys:
        for x in grid_xs:
            grid_points.append((x, y))
            
    # Select 26 points
    selected_points = grid_points[:26]
    
    x_grid = [p[0] for p in selected_points]
    y_grid = [p[1] for p in selected_points]
    r_grid = [0.02] * n # Small initial radius to satisfy constraints
    
    v_init_grid = np.array(x_grid + y_grid + r_grid)
    run_opt(v_init_grid)
    
    # 2. Random Initializations
    # Try a few random starts to escape local minima
    np.random.seed(42)
    for _ in range(10):
        x_rand = np.random.rand(n) * 0.8 + 0.1 # Keep away from very edge initially
        y_rand = np.random.rand(n) * 0.8 + 0.1
        r_rand = np.ones(n) * 0.01
        
        v_init_rand = np.concatenate([x_rand, y_rand, r_rand])
        run_opt(v_init_rand)
        
    # 3. Hexagonal-like perturbation of grid
    # Shift every other row to create hexagonal packing structure
    v_init_hex = v_init_grid.copy()
    # Shift y-coordinates of even rows? Or x-coordinates.
    # Let's just perturb x slightly for alternate rows
    for i in range(n):
        # Determine row index roughly (6 cols)
        row = i // 6
        if row % 2 == 1:
            # Shift x by half spacing approx 0.11
            v_init_hex[i] += 0.05 
            
    run_opt(v_init_hex)
    
    if best_v is None:
        # Fallback to grid init if all failed
        best_v = v_init_grid
    
    # Extract results
    x_out = best_v[:n]
    y_out = best_v[n:2 * n]
    r_out = best_v[2 * n:]
    
    centers = np.column_stack((x_out, y_out))
    radii = r_out
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
