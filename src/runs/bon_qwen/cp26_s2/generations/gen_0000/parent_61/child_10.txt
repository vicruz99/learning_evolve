import numpy as np
from scipy.optimize import linprog


def compute_pairwise_distances(centers):
    """Compute pairwise Euclidean distances between centers."""
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            dists[i, j] = d
            dists[j, i] = d
    return dists


def solve_radii_lp(centers, min_radius=0.001):
    """Solve LP to find optimal radii given fixed centers."""
    n = centers.shape[0]
    dists = compute_pairwise_distances(centers)

    c = -np.ones(n)

    A_ub = []
    b_ub = []

    for i in range(n):
        # ri <= xi
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(centers[i, 0])

        # ri <= 1 - xi
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - centers[i, 0])

        # ri <= yi
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(centers[i, 1])

        # ri <= 1 - yi
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - centers[i, 1])

    for i in range(n):
        for j in range(i + 1, n):
            # ri + rj <= dij
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dists[i, j])

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    bounds = [(min_radius, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x
    return None


def optimize_centers_force(centers, radii, max_iter=3000):

                        forces[j] -= diff * force_mag

                idx += 1

    return centers
