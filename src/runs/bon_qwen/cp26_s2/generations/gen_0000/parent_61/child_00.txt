import numpy as np
from scipy.optimize import minimize

def packing_objective(vars, n, penalty_weight):
    """
    Computes the negative sum of radii plus penalty terms for overlaps and boundary violations.
    """
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    
    obj_val = -np.sum(radii)
    
    # Boundary penalties: r <= x, r <= 1-x, r <= y, r <= 1-y
    ov_x1 = radii - centers[:, 0]
    ov_x2 = radii - (1.0 - centers[:, 0])
    ov_y1 = radii - centers[:, 1]
    ov_y2 = radii - (1.0 - centers[:, 1])
    
    all_ov = np.concatenate([ov_x1, ov_x2, ov_y1, ov_y2])
    pos_ov = np.maximum(all_ov, 0.0)
    obj_val += penalty_weight * np.sum(pos_ov ** 2)
    
    # Overlap penalties: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            req = radii[i] + radii[j]
            if req > dist:
                obj_val += penalty_weight * (req - dist) ** 2
                
    return obj_val

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Generate diverse starting configurations
    starts = []
    
    # 1. Structured Grid
    c_grid = np.zeros((n, 2))
    idx = 0
    for r in range(6):
        for col in range(5):
            if idx >= n: break
            c_grid[idx] = [0.15 + col * 0.18, 0.1 + r * 0.18]
            idx += 1
    starts.append(np.hstack([c_grid.flatten(), np.ones(n) * 0.08]))
    
    # 2. Hexagonal Offset
    c_hex = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        cols = 5 if row % 2 == 0 else 4
        for col in range(cols):
            if idx >= n: break
            c_hex[idx] = [0.15 + col * 0.2, 0.1 + row * 0.17]
            idx += 1
        row += 1
    starts.append(np.hstack([c_hex.flatten(), np.ones(n) * 0.08]))
    
    # 3. Random Clusters
    for _ in range(4):
        c_rand = np.random.rand(n, 2)
        starts.append(np.hstack([c_rand.flatten(), np.ones(n) * 0.05]))
        
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    for x0 in starts:
        # Stage 1: Coarse positioning (lower penalty allows movement)
        res1 = minimize(packing_objective, x0, args=(n, 50.0), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-8, 'gtol': 1e-6})
                       
        # Stage 2: Tight packing & radius maximization (high penalty enforces constraints)
        res = minimize(packing_objective, res1.x, args=(n, 400.0), method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-10, 'gtol': 1e-8})
        
        c_opt = res.x[:2 * n].reshape(n, 2)
        r_opt = res.x[2 * n:]
        
        # Post-processing: ensure strict validity within 1e-12 tolerance
        max_viol = 0.0
        for i in range(n):
            for v in [c_opt[i, 0], 1.0 - c_opt[i, 0], c_opt[i, 1], 1.0 - c_opt[i, 1]]:
                if r_opt[i] > v:
                    viol = r_opt[i] - v
                    if viol > max_viol: max_viol = viol
            for j in range(i + 1, n):
                dist = np.hypot(c_opt[i, 0] - c_opt[j, 0], c_opt[i, 1] - c_opt[j, 1])
                if r_opt[i] + r_opt[j] > dist:
                    viol = r_opt[i] + r_opt[j] - dist
                    if viol > max_viol: max_viol = viol
                    
        if max_viol > 1e-10:
            # Uniformly shrink to clear all violations with safety margin
            r_opt = np.maximum(r_opt - (max_viol + 1e-9), 0.0)
            
        curr_sum = np.sum(r_opt)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    return best_centers, best_radii, best_sum
