import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Assumes equal radii for the optimization and returns the resulting configuration.
    """
    n = 26
    
    # 1. Initialization: Hexagonal Lattice
    # We arrange circles in rows with counts [5, 6, 5, 6, 4] to total 26.
    # This provides a dense starting configuration better than a square grid.
    row_counts = [5, 6, , 5, 6, 4]
    r_init = 0.100  # Initial radius guess, slightly below the theoretical bound
    centers = []
    
    y = r_init
    for count in row_counts:
        # Calculate x coordinates for the row
        # Width occupied by 'count' circles is count * 2*r. 
        # We center them in the unit square [0, 1].
        row_width = count * 2 * r_init
        x_start = (1.0 - row_width) / 2.0 + r_init
        
        for i in range(count):
            x = x_start + i * (2 * r_init)
            centers.append([x, y])
        
        # Move to next row with vertical spacing of sqrt(3)*r for hexagonal packing
        y += np.sqrt(3) * r_init

    centers = np.array(centers)
    # Ensure we have exactly 26 circles
    assert len(centers) == n, f"Expected {n} centers, got {len(centers)}"

    # 2. Optimization: Maximize radius r
    # Variables: [x1, y1, ..., x26, y26, r]
    # We flatten centers to a 1D array and append r.
    x0 = np.hstack([centers.flatten(), [r_init]])
    
    # Define bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    bounds.append((0.0, 0.5)) # Radius bound

    def objective(params):
        # We want to maximize r, so we minimize -r
        return -params[-1]

    def constraint_no_overlap(params):
        c = params[:2*n].reshape(-1, 2)
        r = params[-1]
        dists = np.linalg.norm(c[:, np.newaxis] - c[np.newaxis, :], axis=2)
        # Triangular indices to check unique pairs
        i, j = np.triu_indices(n, k=1)
        return dists[i, j] - 2 * r

    def constraint_boundary_x(params):
        c = params[:2*n].reshape(-1, 2)
        r = params[-1]
        # x - r >= 0 and x + r <= 1
        return np.minimum(c[:, 0] - r, 1.0 - c[:, 0])

    def constraint_boundary_y(params):
        c = params[:2*n].reshape(-1, 2)
        r = params[-1]
        return np.minimum(c[:, 1] - r, 1.0 - c[:, 1])

    # Define constraints for the solver
    cons = [
        {'type': 'ineq', 'fun': constraint_no_overlap},
        {'type': 'ineq', 'fun': constraint_boundary_x},
        {'type': 'ineq', 'fun': constraint_boundary_y}
    ]

    # Run optimization
    # SLSQP handles bounds and constraints well
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-12}
    )

    # Extract results
    final_centers = result.x[:2*n].reshape(-1, 2)
    final_r = result.x[-1]
    final_radii = np.full(n, final_r)
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii
