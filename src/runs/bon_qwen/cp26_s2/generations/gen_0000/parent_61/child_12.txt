import numpy as np
from scipy.optimize import minimize

def compute_forces_and_update(centers, radii, dt, k_repulse, k_boundary, k_pressure, alpha):
    """
    Simulates physics to push circles apart, away from boundaries, and grow radii.
    """
    n = centers.shape[0]
    centers = centers.copy()
    radii = radii.copy()
    forces = np.zeros_like(centers)

    # 1. Repulsion forces between circles
    # Vectorized calculation for efficiency
    # d_vec[i, j] = centers[i] - centers[j]
    d_vec = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(d_vec**2, axis=2)
    dist = np.sqrt(np.maximum(dist_sq, 1e-10))
    
    # Minimum required distance
    min_dist = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Calculate overlap and force direction
    # Overlap = (r1 + r2) - dist. If positive, they are overlapping.
    overlap = min_dist - dist
    
    # Apply repulsion force proportional to overlap
    # We use a softened version to prevent singularities
    force_magnitude = k_repulse * np.maximum(overlap, 0) / (dist + 1e-6)
    
    # Force is applied in direction of d_vec
    # d_vec[i,j] points from j to i. So i is pushed away from j.
    forces += np.sum(force_magnitude[:, :, np.newaxis] * d_vec, axis=1)

    # 2. Boundary repulsion forces
    # Left/Right walls
    x = centers[:, 0]
    r = radii
    # Force if x - r < 0 (penetrating left wall)
    left_pen = r - x
    left_f = k_boundary * np.maximum(left_pen, 0)
    forces[:, 0] += left_f
    
    # Force if x + r > 1 (penetrating right wall)
    right_pen = x + r - 1
    right_f = k_boundary * np.maximum(right_pen, 0)
    forces[:, 0] -= right_f

    # Top/Bottom walls
    y = centers[:, 1]
    bottom_pen = r - y
    bottom_f = k_boundary * np.maximum(bottom_pen, 0)
    forces[:, 1] += bottom_f
    
    top_pen = y + r - 1
    top_f = k_boundary * np.maximum(top_pen, 0)
    forces[:, 1] -= top_f

    # 3. Update centers
    centers += forces * dt
    
    # 4. Grow radii (Pressure to maximize sum)
    # If a circle is not significantly constrained, grow it.
    # We can check if forces are low, or just grow by a small amount and correct with forces next step.
    # A simple strategy: grow radii by a small fraction of current size or fixed amount,
    # then let repulsion forces push centers to make room.
    radii += k_pressure * dt 
    
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- Stage 1: Physics Simulation ---
    
    # Initialize centers on a grid to avoid initial chaos
    # We try to fit 26 circles. A 6x5 grid has 30 spots.
    # We can pick a subset or just fill 6x5 and let optimizer handle 4 extra? 
    # No, we need exactly 26.
    # Let's place them on a 6x5 grid and then remove 4 randomly, or just place 26 on a 6x4.5 grid?
    # Simpler: Place on 6x5 grid, take first 26.
    
    rows = 5
    cols = 6
    # Spacing for initialization
    dx = 1.0 / (cols + 1)
    dy = 1.0 / (rows + 1)
    
    initial_centers = []
    count = 0
    for r in range(1, rows + 1):
        for c in range(1, cols + 1):
            if count < n:
                initial_centers.append([c * dx, r * dy])
                count += 1
    
    centers = np.array(initial_centers)
    # Start with small radii
    radii = np.full(n, 0.02)
    
    # Simulation parameters
    dt = 0.05
    k_repulse = 100.0
    k_boundary = 100.0
    k_pressure = 0.5 # Rate of radius growth
    iterations = 500
    
    # Cooling schedule for pressure to stabilize
    for i in range(iterations):
        # Reduce pressure as we get denser to avoid oscillations
        current_k_pressure = k_pressure * (1.0 - i / (iterations * 1.5))
        if current_k_pressure < 0: current_k_pressure = 0
        
        centers, radii = compute_forces_and_update(
            centers, radii, dt, k_repulse, k_boundary, current_k_pressure, 1.0
        )
        
        # Clamp centers to stay roughly inside during simulation to prevent explosion
        centers[:, 0] = np.clip(centers[:, 0], 1e-6, 1 - 1e-6)
        centers[:, 1] = np.clip(centers[:, 1], 1e-6, 1 - 1e-6)

    # --- Stage 2: Numerical Optimization (SLSQP) ---
    
    # Define objective function: maximize sum of radii => minimize -sum(radii)
    def objective(x_vars):
        r = x_vars[2::3]
        return -np.sum(r)

    # Define constraints
    # 1. Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    def boundary_constraints(x_vars):
        x = x_vars[0::3]
        y = x_vars[1::3]
        r = x_vars[2::3]
        
        c1 = x - r          # >= 0
        c2 = 1 - x - r      # >= 0
        c3 = y - r          # >= 0
        c4 = 1 - y - r      # >= 0
        
        return np.concatenate([c1, c2, c3, c4])

    # 2. Non-overlap constraints: dist >= r_i + r_j
    # We will pass a function that returns the vector of (dist - r_i - r_j) for all pairs
    # SLSQP expects constraints to return positive values for 'ineq' type
    def non_overlap_constraints(x_vars):
        x = x_vars[0::3]
        y = x_vars[1::3]
        r = x_vars[2::3]
        
        n = len(r)
        # Compute all pairwise distances
        # Using broadcasting for vectorization
        # x_diff[i, j] = x[i] - x[j]
        x_diff = x[:, np.newaxis] - x[np.newaxis, :]
        y_diff = y[:, np.newaxis] - y[np.newaxis, :]
        
        dist_sq = x_diff**2 + y_diff**2
        dist = np.sqrt(np.maximum(dist_sq, 1e-12))
        
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        # We only need upper triangle (i < j)
        # Create mask for upper triangle
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        # Constraint: dist - r_sum >= 0
        diffs = dist[mask] - r_sum[mask]
        
        return diffs

    # Initial variable vector
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    # Bounds
    # x, y in [0, 1]
    # r in [0, 0.5] (diameter <= 1)
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x_i
        bounds.append((0, 1)) # y_i
        bounds.append((0, 0.5)) # r_i
        
    # Setup constraints for SLSQP
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': non_overlap_constraints}
    ]
    
    # Run optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        # Extract result
        final_x = res.x
        final_centers = np.zeros((n, 2))
        final_radii = np.zeros(n)
        
        final_centers[:, 0] = final_x[0::3]
        final_centers[:, 1] = final_x[1::3]
        final_radii = final_x[2::3]
        
    except Exception as e:
        # Fallback to physics simulation result if optimizer fails
        final_centers = centers
        final_radii = radii

    # Final cleanup/clamping to ensure strict validity
    # Although SLSQP should handle it, floating point errors can occur.
    # We can slightly shrink radii if needed, but usually it's fine.
    # The validator allows 1e-12 tolerance.
    
    # Ensure non-negative radii
    final_radii = np.maximum(final_radii, 0.0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, float(sum_radii)
