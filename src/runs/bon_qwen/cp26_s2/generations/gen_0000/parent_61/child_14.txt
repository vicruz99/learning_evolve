def compute_cost(x, n, lam):
    """
    Compute the cost function for the optimization.
    Cost = -sum(radii) + lambda * (overlap_penalty + boundary_penalty)
    
    Args:
        x: np.array of shape (3*n,) containing [x1, y1, r1, x2, y2, r2, ...]
        n: number of circles
        lam: penalty multiplier
    
    Returns:
        float: cost value
    """
    cost = 0.0
    
    # Extract centers and radii for easier access if needed, 
    # but direct indexing is faster for vectorized operations or loops
    # To ensure efficiency, we can compute penalties in a loop or vectorized.
    # Given N=26, a loop is perfectly fine and readable.
    
    sum_radii = 0.0
    
    # Boundary Penalties
    # Constraints: r <= x <= 1-r  =>  r - x <= 0, x + r - 1 <= 0
    # Same for y.
    for i in range(n):
        xi = x[3*i]
        yi = x[3*i+1]
        ri = x[3*i+2]
        
        sum_radii += ri
        
        # Boundary violations
        # Left: ri - xi <= 0  => violation max(0, ri - xi)
        # Right: xi + ri - 1 <= 0 => violation max(0, xi + ri - 1)
        # Bottom: ri - yi <= 0
        # Top: yi + ri - 1 <= 0
        
        v_left = max(0.0, ri - xi)
        v_right = max(0.0, xi + ri - 1.0)
        v_bottom = max(0.0, ri - yi)
        v_top = max(0.0, yi + ri - 1.0)
        
        cost += (v_left**2 + v_right**2 + v_bottom**2 + v_top**2)

    # Overlap Penalties
    # Constraint: dist(i, j) >= ri + rj
    # Violation: max(0, ri + rj - dist)
    
    # We iterate over all pairs (i, j) with i < j
    for i in range(n):
        xi = x[3*i]
        yi = x[3*i+1]
        ri = x[3*i+2]
        
        for j in range(i + 1, n):
            xj = x[3*j]
            yj = x[3*j+1]
            rj = x[3*j+2]
            
            dx = xi - xj
            dy = yi - yj
            dist = math.sqrt(dx*dx + dy*dy)
            
            overlap = ri + rj - dist
            if overlap > 0:
                cost += overlap**2

    # Objective: Minimize -sum(radii) + lam * penalty
    return -sum_radii + lam * cost
