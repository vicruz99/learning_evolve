import numpy as np
from scipy.optimize import minimize


def compute_penalty(x, n):
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    penalty = 0.0

    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]

        if r < 0:
            penalty += 1e6 * r ** 2

        if cx - r < 0:
            penalty += (cx - r) ** 2
        if cx + r > 1:
            penalty += (cx + r - 1) ** 2
        if cy - r < 0:
            penalty += (cy - r) ** 2
        if cy + r > 1:
            penalty += (cy + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                diff = min_dist - dist
                penalty += diff ** 2

    return penalty


def objective(x, n, penalty_weight):
    radii = x[2 * n:]
    return -np.sum(radii) + penalty_weight * compute_penalty(x, n)


def init_square_grid(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))

    margin_x = 0.01
    margin_y = 0.01
    spacing_x = (1.0 - 2 * margin_x) / max(cols - 1, 1)
    spacing_y = (1.0 - 2 * margin_y) / max(rows - 1, 1)

    r = min(spacing_x, spacing_y) / 2.0

    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx < n:
                centers[idx, 0] = margin_x + j * spacing_x
                centers[idx, 1] = margin_y + i * spacing_y
                radii[idx] = r
                idx += 1

    return centers, radii


def init_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    r = 0.08
    row_height = np.sqrt(3) * r
    col_width = 2 * r

    idx = 0
    row = 0
    while idx < n:
        y = r + row * row_height
        if y + r > 1.0:
            break

        if row % 2 == 0:
            n_cols = int((1.0 - 2 * r) / col_width) + 1
            x_offset = r
        else:
            n_cols = int((1.0 - 2 * r - col_width / 2) / col_width) + 1
            x_offset = r + col_width / 2

        for col in range(n_cols):
            if idx < n:
                x = x_offset + col * col_width
                if x + r <= 1.0 + 1e-10:
                    centers[idx, 0] = x
                    centers[idx, 1] = y
                    radii[idx] = r
                    idx += 1

        row += 1

    for i in range(idx, n):
        centers[i] = [0.5, 0.5]
        radii[i] = r * 0.5

    return centers, radii


def init_staggered_5x5_plus_1(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    r_base = 0.1
    positions = []

    # 5x5 grid positions
    for i in range(5):
        for j in range(5):
            positions.append((0.1 + j * 0.2, 0.1 + i * 0.2))

    # Add 1st extra circle in a gap (center-ish)
    positions.append((0.2, 0.2))
    positions.append((0.4, 0.2))
    positions.append((0.6, 0.2))
    positions.append((0.8, 0.2))
    positions.append((0.2, 0.4))
    positions.append((0.4, 0.4))
    positions.append((0.6, 0.4))
    positions.append((0.8, 0.4))

    for i in range(n):
        if i < len(positions):
            centers[i] = positions[i]
            if i < 25:
                radii[i] = r_base * 0.95
            else:
                radii[i] = r_base * 0.3
        else:
            centers[i] = [0.5, 0.5]
            radii[i] = 0.05

    return centers, radii


def init_hexagonal_dense(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    r = 0.095
    row_height = np.sqrt(3) * r
    col_width = 2 * r

    idx = 0
    row = 0
    max_rows = 6

    while idx < n and row < max_rows:
        y = r + row * row_height
        if y + r > 1.001:
            row += 1
            continue

        if row % 2 == 0:
            x_start = r
            n_cols = int((1.0 - 2 * r) / col_width) + 1
        else:
            x_start = r + col_width / 2
            n_cols = int((1.0 - 2 * r - col_width / 2) / col_width) + 1

        actual_cols = min(n_cols, n - idx)
        for col in range(actual_cols):
            x = x_start + col * col_width
            if x + r <= 1.001:
                centers[idx, 0] = x
                centers[idx, 1] = y
                radii[idx] = r
                idx += 1

        row += 1

    for i in range(idx, n):
        centers[i] = [0.5, 0.5]
        radii[i] = r * 0.3

    return centers, radii


def init_corner_heavy(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    idx = 0

    corners = [(0.15, 0.15), (0.85, 0.15), (0.15, 0.85), (0.85, 0.85)]
    for c in corners:
        centers[idx] = c
        radii[idx] = 0.15
        idx += 1

    edges = [
        (0.5, 0.1), (0.1, 0.5), (0.9, 0.5), (0.5, 0.9),
        (0.5, 0.3), (0.3, 0.5), (0.7, 0.5), (0.5, 0.7)
    ]
    for c in edges:
        if idx < n:
            centers[idx] = c
            radii[idx] = 0.08
            idx += 1

    interior = [
        (0.3, 0.3), (0.7, 0.3), (0.3, 0.7), (0.7, 0.7),
        (0.5, 0.5), (0.25, 0.5), (0.75, 0.5), (0.5, 0.25), (0.5, 0.75)
    ]
    for c in interior:
        if idx < n:
            centers[idx] = c
            radii[idx] = 0.05
            idx += 1

    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.03
        idx += 1

    return centers, radii


def generate_random_init(n, seed):
    np.random.seed(seed)
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.04)
    return centers, radii


def optimize_packing(centers, radii, n, num_iterations=80, base_penalty=50.0):
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = []
    for i in range(2 * n):
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((1e-8, 0.49))

    penalty = base_penalty
    x = x0.copy()

    for iteration in range(num_iterations):
        result = minimize(
            objective, x, args=(n, penalty),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12}
        )
        x = result.x.copy()
        penalty *= 1.4

    centers_opt = x[:2 * n].reshape(n, 2)
    radii_opt = x[2 * n:]

    return centers_opt, radii_opt


def compute_violation(centers, radii, n):
    centers_arr = centers.reshape(n, 2) if centers.ndim == 1 else centers
    max_violation = 0.0

    for i in range(n):
        cx, cy = centers_arr[i]
        r = radii[i]

        if cx - r < 0:
            max_violation = max(max_violation, -(cx - r))
        if cx + r > 1:
            max_violation = max(max_violation, cx + r - 1)
        if cy - r < 0:
            max_violation = max(max_violation, -(cy - r))
        if cy + r > 1:
            max_violation = max(max_violation, cy + r - 1)

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers_arr[i, 0] - centers_arr[j, 0]
            dy = centers_arr[i, 1] - centers_arr[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                max_violation = max(max_violation, min_dist - dist)

    return max_violation


def shrink_radii_to_fix(centers, radii, n, max_iter=100):
    centers = centers.copy()
    radii = radii.copy()

    for _ in range(max_iter):
        max_viol = 0.0
        viol_idx = -1

        for i in range(n):
            cx, cy = centers[i]
            r = radii[i]

            if cx - r < 0:
                max_viol = max(max_viol, -(cx - r))
                viol_idx = i
            if cx + r > 1:
                max_viol = max(max_viol, cx + r - 1)
                viol_idx = i
            if cy - r < 0:
                max_viol = max(max_viol, -(cy - r))
                viol_idx = i
            if cy + r > 1:
                max_viol = max(max_viol, cy + r - 1)
                viol_idx = i

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    v = min_dist - dist
                    if v > max_viol:
                        max_viol = v
                        viol_idx = -1

        if max_viol < 1e-12:
            break

        if viol_idx >= 0:
            radii[viol_idx] *= 0.999
        else:
            radii *= 0.9999

    return centers, radii


def is_valid_packing(centers, radii, tol=1e-10):
    n = len(radii)

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        cx, cy = centers[i]
        r = radii[i]
        if cx - r < -tol or cx + r > 1 + tol:
            return False
        if cy - r < -tol or cy + r > 1 + tol:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            if dist < radii[i] + radii[j] - tol:
                return False

    return True


def run_packing():
    n = 26

    best_centers = None
    best_radii = None
    best_sum = 0.0

    init_configs = []

    init_configs.append(init_square_grid(n))
    init_configs.append(init_hexagonal(n))
    init_configs.append(init_staggered_5x5_plus_1(n))
    init_configs.append(init_hexagonal_dense(n))
    init_configs.append(init_corner_heavy(n))

    for seed in range(10):
        init_configs.append(generate_random_init(n, seed))

    for cfg_idx, (c_init, r_init) in enumerate(init_configs):
        centers_opt, radii_opt = optimize_packing(
            c_init.copy(), r_init.copy(), n,
            num_iterations=100, base_penalty=100.0
        )

        current_sum = np.sum(radii_opt)

        if current_sum > best_sum:
            if is_valid_packing(centers_opt, radii_opt, tol=1e-8):
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
                best_sum = current_sum
            else:
                # Try to fix minor violations
                c_fixed, r_fixed = shrink_radii_to_fix(centers_opt, radii_opt, n)
                if is_valid_packing(c_fixed, r_fixed, tol=1e-8):
                    if np.sum(r_fixed) > best_sum:
                        best_centers = c_fixed.copy()
                        best_radii = r_fixed.copy()
                        best_sum = np.sum(r_fixed)

    if best_centers is None:
        best_centers, best_radii = init_square_grid(n)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
