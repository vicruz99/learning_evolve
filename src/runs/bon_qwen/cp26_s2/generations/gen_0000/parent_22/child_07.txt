

To maximize the sum of radii for 26 circles in a unit square, we model this as a non-linear optimization problem. We use `scipy.optimize.minimize` with the SLSQP algorithm, which is effective for constrained problems of this size. We employ a penalty-free constraint formulation where the sum of radii is maximized subject to non-overlap and boundary conditions.

### Strategy
1. **Objective**: Maximize the sum of all radii, $\sum r_i$.
2. **Constraints**:
   - **Boundary**: Each circle must be inside $[0, 1] \times [0, 1]$, so $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
   - **Non-overlap**: The distance between centers of any two circles must be at least the sum of their radii.
3. **Initialization**: We use multiple deterministic starting points, including a 5x5 grid with an additional point and a 6x5 grid subset. We also add small random perturbations to break symmetries that can trap local optimizers.
4. **Optimization**: We run SLSQP for each initialization, treating the flattened array of $(x_i, y_i, r_i)$ as variables and the geometric conditions as vectorized inequality constraints.

```python
import numpy as np
from scipy.optimize import minimize

def get_constraints(centers, radii):
    """
    Returns an array of constraint values.
    Constraints are satisfied if the returned array is >= 0.
    """
    n = centers.shape[0]
    constraints = []
    
    # Boundary constraints
    constraints.append(centers[:, 0] - radii)       # x - r >= 0
    constraints.append(1 - centers[:, 0] - radii)   # 1 - x - r >= 0
    constraints.append(centers[:, 1] - radii)       # y - r >= 0
    constraints.append(1 - centers[:, 1] - radii)   # 1 - y - r >= 0
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            constraints.append(dist - radii[i] - radii[j])
            
    return np.concatenate(constraints)

def run_packing():
    n_circles = 26
    num_starts = 5
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds for [x, y, r]
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n_circles

    for start_id in range(num_starts):
        # Initialize centers
        centers_init = np.zeros((n_circles, 2))
        
        if start_id == 0:
            # 5x5 grid + 1
            grid = np.linspace(0.15, 0.85, 5)
            pts = [(x, y) for x in grid for y in grid]
            pts.append((0.5, 0.2))
            centers_init = np.array(pts)
            r_init = 0.09
        elif start_id == 1:
            # 6x5 grid subset
            gx = np.linspace(0.1, 0.9, 6)
            gy = np.linspace(0.1, 0.9, 5)
            pts = [(x, y) for x in gx for y in gy]
            centers_init = np.array(pts[:26])
            r_init = 0.08
        elif start_id == 2:
            # Hexagonal-like
            centers_init = np.random.rand(n_circles, 2)
            r_init = 0.05
        else:
            # Random grid points
            grid = np.linspace(0.1, 0.9, 6)
            pts = [(x, y) for x in grid for y in grid]
            np.random.shuffle(pts)
            centers_init = np.array(pts[:26])
            r_init = 0.08
            
        # Add small random noise to centers to break symmetry
        centers_init += np.random.uniform(-0.01, 0.01, centers_init.shape)
        
        # Initialize variables: [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.zeros(3 * n_circles)
        for i in range(n_circles):
            x0[3*i] = centers_init[i, 0]
            x0[3*i + 1] = centers_init[i, 1]
            x0[3*i + 2] = r_init
        
        def objective(vars):
            radii_arr = vars[2::3]
            return -np.sum(radii_arr)
            
        def constraint_fun(vars):
            centers = np.zeros((n_circles, 2))
            radii = np.zeros(n_circles)
            for i in range(n_circles):
                centers[i] = vars[3*i], vars[3*i+1]
                radii[i] = vars[3*i+2]
            return get_constraints(centers, radii)
            
        cons = {'type': 'ineq', 'fun': constraint_fun}
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-8})
        
        if res.success:
            res_centers = np.zeros((n_circles, 2))
            res_radii = np.zeros(n_circles)
            for i in range(n_circles):
                res_centers[i] = res.x[3*i], res.x[3*i+1]
                res_radii[i] = res.x[3*i+2]
            
            current_sum = np.sum(res_radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res_centers
                best_radii = res_radii
                
    return best_centers, best_radii, best_sum
```
