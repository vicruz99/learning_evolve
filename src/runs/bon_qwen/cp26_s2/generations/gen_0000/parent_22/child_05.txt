    # r_i = min(dist_to_boundary_i, min_j(dist_ij - r_j))
    # Since r_j depends on others, we can use an iterative update.

        new_radii = np.copy(radii)
        for i in range(n):
            # Boundary constraint
            max_r = min(centers[i, 0], 1-centers[i, 0], centers[i, 1], 1-centers[i, 1])
