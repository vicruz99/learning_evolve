import numpy as np
from scipy.optimize import minimize

def generate_initial_centers(n):
    """Generates an initial hexagonal-like placement of n circles."""
    centers = []
    rows_counts = []
    # Distribute 26 circles into 5 rows (6, 5, 6, 5, 4)
    for r in range(5):
        count = 6 if r % 2 == 0 else 5
        if r == 4 and len(centers) + count > 26:
            count = 26 - len(centers)
        if count <= 0: break
        rows_counts.append(count)
    
    current_n = 0
    for r, count in enumerate(rows_counts):
        dy = 1.0 / 5.0 + 0.05 # Approx spacing
        y = 0.1 + r * (dy + 0.02)
        
        # Stagger x for odd rows
        offset_x = 0.05 if r % 2 != 0 else 0.0
        
        for c in range(count):
            if current_n >= n: break
            x = 0.1 + c * (0.9 / (max(count, 1))) + offset_x
            centers.append([x, y])
            current_n += 1
            
    return np.array(centers[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initial Placement
    centers = generate_initial_centers(n)
    radii = np.full(n, 0.08) # Start with a safe small radius
    
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    # 2. Constraints Definition
    cons = []
    
    # Boundary constraints and non-overlap
    for i in range(n):
        # Circle inside square: r <= x <= 1-r, r <= y <= 1-r
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+2] - x[3*i]})           # x - r >= 0 (Wait, constraint is x >= r -> x - r >= 0)
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] + x[3*i+2] - 1.0})      # 1 - x - r >= 0 -> x + r <= 1
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+2] - x[3*i+1]})          # y >= r
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] + x[3*i+2] - 1.0})    # y <= 1-r
        
        for j in range(i + 1, n):
            # Non-overlap: dist >= r_i + r_j
            # (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
            # Inequality function: dist^2 - (r_i+r_j)^2 >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: 
                    (x[3*i] - x[3*j])**2 + (x[3*i+1] - x[3*j+1])**2 - (x[3*i+2] + x[3*j+2])**2
            })

    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # 3. Optimization
    # Objective: Maximize sum(r) -> Minimize -sum(r)
    def objective(x):
        radii_sum = 0.0
        for i in range(n):
            radii_sum += x[3*i+2]
        return -radii_sum

    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 200, 'disp': False})

    # 4. Extract Results
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    for i in range(n):
        best_centers[i] = [res.x[3*i], res.x[3*i+1]]
        best_radii[i] = res.x[3*i+2]
        
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(final_sum)
