def compute_penalty(centers, radii, weight):
    """Compute penalty for constraint violations."""
    n = len(radii)
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        penalty += weight * max(0.0, r - x) ** 2
        penalty += weight * max(0.0, x + r - 1.0) ** 2
        penalty += weight * max(0.0, r - y) ** 2
        penalty += weight * max(0.0, y + r - 1.0) ** 2
    
    # Overlap penalties (vectorized for speed)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            if overlap > 0:
                penalty += weight * overlap * overlap
    
    return penalty


def objective_func(params, n, weight):
    """Objective function: negative sum of radii + penalty."""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.exp(params[2 * n:])
    
    # Objective: maximize sum of radii (minimize negative)
    obj = -np.sum(radii)
    
    # Add penalty
    penalty = compute_penalty(centers, radii, weight)
    
    return obj + penalty

        y += dy
