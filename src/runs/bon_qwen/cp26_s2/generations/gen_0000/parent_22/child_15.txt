3. **Multi-phase optimization**:

import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        return False

    if np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True


def hexagonal_init():
    """Create initial hexagonal lattice for 26 circles"""
    n = 26
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09
    
    row_sizes = [6, 5, 6, 5, 4]
    idx = 0
    
    for row, size in enumerate(row_sizes):
        y = 0.095 + row * 0.19
        if row % 2 == 0:
            x_start = 0.085
            x_step = 0.175
        else:
            x_start = 0.085 + 0.175 / 2
            x_step = 0.175
        
        for col in range(size):
            x = x_start + col * x_step
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def random_init():

def compute_cost(x, n, penalty_weight):
    """Compute objective: maximize sum of radii minus constraint violations"""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    cost = -np.sum(radii)
    
    # Boundary penalties
    for i in range(n):
        x_val, y_val = centers[i]
        r = radii[i]
        if x_val < r:
            cost += penalty_weight * (r - x_val) ** 2
        if x_val > 1 - r:
            cost += penalty_weight * (x_val - (1 - r)) ** 2
        if y_val < r:
            cost += penalty_weight * (r - y_val) ** 2
        if y_val > 1 - r:
            cost += penalty_weight * (y_val - (1 - r)) ** 2
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                cost += penalty_weight * (min_dist - dist) ** 2
    
    return cost


def get_bounds(n):
    """Get optimization bounds for variables"""
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.001, 0.5))
    return bounds


def optimize_once(n, x0, penalty_weight, max_iter=5000):
    """Run single optimization pass"""
    bounds = get_bounds(n)
    result = minimize(
        compute_cost,
        x0,
        args=(n, penalty_weight),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-15}
    )
    return result
