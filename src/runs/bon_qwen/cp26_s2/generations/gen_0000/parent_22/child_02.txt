import numpy as np
from scipy.optimize import minimize

def generate_initial_positions():
    """Generates a hexagonal lattice of 26 points, scaled and centered in [0,1]^2."""
    n = 26
    d = 0.21  # Lattice spacing parameter
    pts = []
    row = 0
    while len(pts) < n:
        col = 0
        x_offset = (row % 2) * d / 2
        y = row * d * np.sqrt(3) / 2
        while len(pts) < n:
            x = x_offset + col * d
            if 0 <= x <= 1 and 0 <= y <= 1:
                pts.append((x, y))
            col += 1
        row += 1
        if row * d * np.sqrt(3) / 2 > 1.1:
            break
            
    pts = np.array(pts[:n])
    min_c = pts.min(axis=0)
    max_c = pts.max(axis=0)
    width = max_c - min_c
    # Scale to 90% of the bounding box to leave room for radii expansion
    scale = 0.90 / max(width) if max(width) > 0 else 1.0
    pts = (pts - min_c) * scale + (1.0 - scale) / 2.0
    return pts

def obj_fun(v):
    """Objective: maximize sum of radii (minimize negative sum)."""
    n = len(v) // 3
    return -np.sum(v[2*n:])

def bound_con(v):
    """Boundary constraints: circles must stay inside [0,1]^2."""
    n = len(v) // 3
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    c1 = centers[:, 0] - radii
    c2 = 1.0 - centers[:, 0] - radii
    c3 = centers[:, 1] - radii
    c4 = 1.0 - centers[:, 1] - radii
    return np.concatenate([c1, c2, c3, c4])

def overlap_con(v):
    """Overlap constraints: squared distance >= sum of radii squared."""
    n = len(v) // 3
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    
    # Vectorized pairwise differences and distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Extract upper triangle (i < j) to avoid duplicates and self-pairs
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return (dist_sq - r_sum**2)[mask]

def run_packing():
    n = 26
    centers_init = generate_initial_positions()
    r_init = np.full(n, 0.08)  # Safe initial radius
    x0 = np.concatenate([centers_init.flatten(), r_init])
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    cons = [
        {'type': 'ineq', 'fun': bound_con},
        {'type': 'ineq', 'fun': overlap_con}
    ]
    
    # SLSQP handles nonlinear constraints efficiently
    res = minimize(obj_fun, x0, bounds=bounds, constraints=cons, method='SLSQP',
                   options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                   
    centers_opt = res.x[:2*n].reshape((n, 2))
    radii_opt = res.x[2*n:]
    
    # Post-processing: enforce strict feasibility within numerical tolerance
    radii_opt = np.maximum(radii_opt, 0.0)
    centers_opt[:, 0] = np.clip(centers_opt[:, 0], radii_opt, 1.0 - radii_opt)
    centers_opt[:, 1] = np.clip(centers_opt[:, 1], radii_opt, 1.0 - radii_opt)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
