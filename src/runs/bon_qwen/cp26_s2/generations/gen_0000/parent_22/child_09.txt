import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective(params, n):

    return -np.sum(radii)

def constraints_factory(n):
    constraints = []
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # This implies r <= x <= 1-r and r <= y <= 1-r
    # So x - r >= 0, 1 - x - r >= 0, etc.
    
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0  =>  x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[idx_x] - p[idx_r]})
        # 1 - x - r >= 0 => 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[idx_x] - p[idx_r]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[idx_y] - p[idx_r]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[idx_y] - p[idx_r]})
        
        # Radius non-negativity (redundant with boundary if x,y in [0,1], but good for safety)
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[idx_r]})

    # Non-overlap constraints: dist_ij >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2
            
            def dist_constraint(p, i=i, j=j):
                xi, yi, ri = p[idx_xi], p[idx_yi], p[idx_ri]
                xj, yj, rj = p[idx_xj], p[idx_yj], p[idx_rj]
                dx = xi - xj
                dy = yi - yj
                return (dx*dx + dy*dy) - (ri + rj)**2
            
            constraints.append({'type': 'ineq', 'fun': dist_constraint})
            
    return constraints
