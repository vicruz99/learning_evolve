I generate a grid of 26 points with sufficient spacing to ensure the initial configuration is strictly valid (non-overlapping and within bounds) with a small radius.

2.  **Numerical Optimization (SLSQP):** I use the Sequential Least Squares Programming (SLSQP) algorithm from `scipy.optimize`. 
    -   **Objective:** Maximize the sum of radii (minimize negative sum).
    -   **Constraints:** The solver respects the boundary constraints ($x \pm r \in [0,1]$) and the non-overlap constraints ($dist \ge r_i + r_j$).
    -   **Variable Vectorization:** The state of the system is represented by a single vector of size 78 ($26 \times 3$ variables for x, y, r), which allows the solver to navigate the search space efficiently.

3.  **Multi-Start Strategy:** Because circle packing is a non-convex problem, the solver can get stuck in local optima. To mitigate this, I run the optimization from multiple random initial configurations within `run_packing` and return the result with the highest sum of radii found.

import numpy as np
from scipy.optimize import minimize

def get_bounds(n):
    """
    Returns the bounds for the optimization variables.
    Variables are arranged as [x1, y1, r1, x2, y2, r2, ...]
    x, y are in [0, 1]. Radius is in [0, 0.5].
    """
    bounds = []
    for _ in range(n):
        bounds.append((0, 1))   # x
        bounds.append((0, 1))   # y
        bounds.append((0, 0.5)) # r
    return bounds

def constraint_function(vars, n):
    """
    Computes the constraint violations.
    Returns a vector where all elements must be >= 0.
    """
    vars = vars.reshape(n, 3)
    centers = vars[:, :2]
    radii = vars[:, 2]
    
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r  =>  r - x <= 0 AND x + r - 1 <= 0
    # But SLSQP expects ineq constraints of form g(x) >= 0.
    # x - r >= 0, 1 - (x + r) >= 0, etc.
    for i in range(n):
        x, y, r = vars[i]
        constraints.append(x - r)
        constraints.append(1.0 - (x + r))
        constraints.append(y - r)
        constraints.append(1.0 - (y + r))
        
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j])**2)
            r_sum_sq = (radii[i] + radii[j])**2
            constraints.append(dist_sq - r_sum_sq)
            
    return np.array(constraints)

def objective_function(vars, n):
    """
    Objective: Maximize sum of radii.
    Returns negative sum of radii for minimization.
    """
    vars = vars.reshape(n, 3)
    radii = vars[:, 2]
    return -np.sum(radii)

def generate_hexagonal_init(n, size=1.0, padding=0.05):
    """
    Generates an initial configuration of n circles using a hexagonal lattice.
    Ensures they are within bounds and non-overlapping by scaling.
    """
    centers = []
    # Approximate circles per row for hex packing
    # We just fill rows until we have n circles
    row_counts = []
    temp_n = n
    # Try to keep row counts balanced
    while temp_n > 0:
        if temp_n > 0:
            row_counts.append(min(5, temp_n)) # Max 5 per row to fit roughly
            temp_n -= row_counts[-1]
    
    # Actually, simpler to just grid them in a triangular pattern
    y = padding
    row_idx = 0
    while len(centers) < n:
        # Determine number of circles in this row to balance
        # A simple heuristic: alternate 5, 4, 5, 4... or similar
        # But let's just place them and shift odd rows
        
        # Calculate x positions
        # We need to fit 'k' circles in width 'size'
        # But since we don't know radius yet, we just place them evenly
        
        # Let's place up to 5 or 6 per row
        remaining = n - len(centers)
        k = min(remaining, 6) # max 6 to be safe
        
        # If we are in an odd row (0-indexed), shift x by half spacing
        # But since we don't know spacing, let's just place in a grid first
        # and the optimizer will fix it.
        
        # Grid placement
        x_positions = np.linspace(padding, size - padding, k)
        
        for x in x_positions:
            if len(centers) >= n:
                break
            centers.append([x, y])
            
        y += 0.35 # vertical spacing
        row_idx += 1
        
    centers = np.array(centers[:n])
    
    # Initialize radii very small to ensure validity
    radii = np.full(n, 0.01)
    
    vars = np.zeros(n * 3)
    vars[0::3] = centers[:, 0]
    vars[1::3] = centers[:, 1]
    vars[2::3] = radii
    
    return vars

def run_packing():
    n = 26
    bounds = get_bounds(n)
    
    best_result = None
    best_sum = -np.inf
    
    # Try multiple restarts to avoid local optima
    for _ in range(5):
        # Random initial positions
        x = np.random.uniform(0.1, 0.9, n)
        y = np.random.uniform(0.1, 0.9, n)
        # Small initial radii
        r = np.full(n, 0.01)
        
        vars_init = np.zeros(n * 3)
        vars_init[0::3] = x
        vars_init[1::3] = y
        vars_init[2::3] = r
        
        # Use hexagonal init for one of the restarts
        if _ == 0:
            vars_init = generate_hexagonal_init(n)
            
        # Define constraints structure for SLSQP
        cons = {
            'type': 'ineq',
            'fun': lambda v: constraint_function(v, n)
        }
        
        try:
            res = minimize(
                objective_function,
                vars_init,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if res.success or res.fun < best_sum * -1 - 0.001: # Heuristic check
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_result = res
                    
        except Exception:
            continue

    if best_result is None:
        # Fallback to a valid simple grid if optimization failed completely
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.01)
        # 5x5 grid + 1
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n:
                    centers[idx] = [0.1 + i*0.2, 0.1 + j*0.2]
                    idx += 1
        if idx < n:
            centers[idx] = [0.5, 0.5]
        sum_r = np.sum(radii)
        return centers, radii, sum_r

    # Extract results
    vars_opt = best_result.x
    centers = vars_opt.reshape(n, 3)[:, :2]
    radii = vars_opt.reshape(n, 3)[:, 2]
    
    # Ensure radii are non-negative (solver might produce tiny negatives due to float precision)
    radii = np.maximum(radii, 0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
