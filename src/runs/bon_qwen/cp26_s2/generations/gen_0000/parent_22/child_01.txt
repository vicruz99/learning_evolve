import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(v.reshape(-1, 3)[:, 2])

def eval_boundary(v):
    """Inequality constraints for circles staying inside the unit square"""
    pts = v.reshape(-1, 3)
    x, y, r = pts[:, 0], pts[:, 1], pts[:, 2]
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    return np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])

def eval_overlap(v):
    """Inequality constraints for non-overlapping circles"""
    pts = v.reshape(-1, 3)
    n = len(pts)
    cons = np.empty(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = pts[i, 0] - pts[j, 0]
            dy = pts[i, 1] - pts[j, 1]
            cons[idx] = dx * dx + dy * dy - (pts[i, 2] + pts[j, 2]) ** 2
            idx += 1
    return cons

def get_initial_guess():
    """Generate a hexagonal-like initial configuration for 26 circles"""
    rows_config = [5, 4, 5, 4, 5, 3]
    pts = []
    for r_idx, ncols in enumerate(rows_config):
        y = float(r_idx)
        x_start = 0.5 if ncols == 4 else 0.0
        for c in range(ncols):
            x = float(c) + x_start
            pts.append((x, y))
            
    xs = [p[0] for p in pts]
    ys = [p[1] for p in pts]
    
    # Normalize to fit within [0.12, 0.88] with some margin
    min_x, max_x = min(xs), max(xs)
    xs = [0.12 + 0.76 * (x - min_x) / (max_x - min_x) for x in xs]
    min_y, max_y = min(ys), max(ys)
    ys = [0.12 + 0.76 * (y - min_y) / (max_y - min_y) for y in ys]
    
    v0 = []
    for i in range(len(pts)):
        v0.extend([xs[i], ys[i], 0.085])  # Initial radius 0.085
    return np.array(v0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    v0 = get_initial_guess()
    
    # Add small random perturbation to escape symmetric local minima
    v0 += np.random.randn(len(v0)) * 0.002
    v0[v0 < 0.0] = 0.01
    v0[v0 > 1.0] = 0.99
    v0.reshape(-1, 3)[:, 2] = np.maximum(v0.reshape(-1, 3)[:, 2], 0.01)
    
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    cons = (
        {'type': 'ineq', 'fun': eval_boundary},
        {'type': 'ineq', 'fun': eval_overlap}
    )
    
    res = minimize(objective, v0, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'ftol': 1e-12, 'maxiter': 5000})
    
    centers = res.x.reshape(-1, 3)[:, :2]
    radii = res.x.reshape(-1, 3)[:, 2]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
