            forces[i, 1] += (r - y) * 100.0
        # Top wall

            forces[i, 1] -= (y - (1 - r)) * 100.0

                # If centers coincide, apply random small force to separate

                forces[j] -= np.random.normal(0, 0.01, 2)
                
    return forces
