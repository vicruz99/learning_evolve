import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty method with L-BFGS-B optimization.
    """
    n = 26
    
    # --- Initialization Strategy ---
    # We aim for a hexagonal-like arrangement which is denser than square grid.
    # For 26 circles, a configuration of 6 rows with counts 5, 4, 5, 4, 5, 3 
    # or similar might work, but let's try to distribute them evenly in a rectangle.
    # A 5x5 grid has 25 circles. We add one.
    # To maximize sum of radii, we want radii to be roughly equal.
    # Initial guess: 5x5 grid with one extra point in a gap, small radii.
    
    # Generate a 5x5 grid centered in [0,1]
    # Coordinates: 0.1, 0.3, 0.5, 0.7, 0.9
    # But this leaves no room for expansion if r=0.1.
    # Let's start with smaller radii and centers.
    
    x_coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    y_coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    
    centers = []
    for x in x_coords:
        for y in y_coords:
            centers.append([x, y])
    
    # Add the 26th circle. 
    # A good spot might be near the center but shifted, or in a corner gap.
    # Let's place it at (0.2, 0.2) roughly, or center (0.5, 0.5) shifted.
    # Actually, (0.5, 0.5) is occupied. 
    # Let's place it at (0.25, 0.25)? No, that's close to (0.1,0.1), (0.3,0.1), etc.
    # Let's try (0.5, 0.2) or something.
    # A random perturbation or a specific gap location.
    # Let's place it at (0.5, 0.2) -- between row y=0.1 and y=0.3.
    centers.append([0.5, 0.2])
    
    centers = np.array(centers)
    
    # Initial radii: small enough to be non-overlapping
    # Grid spacing is 0.2. So r=0.05 is safe.
    radii = np.full(n, 0.05)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    # Actually, easier to keep centers and radii separate in logic, 
    # but optimizer needs a 1D vector.
    # Let's use [x_0, ..., x_n, y_0, ..., y_n, r_0, ..., r_n]
    # Or interleaved [x0, y0, r0, x1, y1, r1...]. Interleaved is easier for indexing.
    
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # --- Penalty Method Setup ---
    # We maximize sum(r_i) <=> minimize -sum(r_i)
    # Subject to:
    # 1. Boundary: r_i <= x_i <= 1-r_i, r_i <= y_i <= 1-r_i
    # 2. Non-overlap: dist(i,j) >= r_i + r_j
    
    # We use a penalty function: f(x) + mu * sum(violation^2)
    # where violation = max(0, constraint_value).
    
    penalty_weight = 1000.0
    
    def objective_and_penalty(vars):
        # Reshape
        # vars is 1D array of length 3*n
        # x_i = vars[3*i], y_i = vars[3*i+1], r_i = vars[3*i+2]
        
        # Extract arrays
        cx = vars[0::3]
        cy = vars[1::3]
        cr = vars[2::3]
        
        # Objective: -sum(r)
        obj = -np.sum(cr)
        
        penalty = 0.0
        
        # Boundary constraints
        # x >= r  =>  r - x <= 0
        # x <= 1-r => x + r - 1 <= 0
        # y >= r  =>  r - y <= 0
        # y <= 1-r => y + r - 1 <= 0
        
        # Vectorized operations
        # v1 = max(0, r - x)
        v_boundary_x_low = cr - cx
        v_boundary_x_high = cx + cr - 1.0
        v_boundary_y_low = cr - cy
        v_boundary_y_high = cy + cr - 1.0
        
        # Square positive parts
        penalty += np.sum(np.maximum(0, v_boundary_x_low)**2)
        penalty += np.sum(np.maximum(0, v_boundary_x_high)**2)
        penalty += np.sum(np.maximum(0, v_boundary_y_low)**2)
        penalty += np.sum(np.maximum(0, v_boundary_y_high)**2)
        
        # Overlap constraints
        # dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        # Violation: (r_i + r_j)^2 - dist^2
        
        # We need to compute pairwise.
        # To speed up, we can iterate.
        # Or use broadcasting.
        
        # cx is (n,), cy is (n,), cr is (n,)
        # Create matrices
        # X matrix (n, n)
        # dist_sq matrix
        
        # This part is O(n^2). n=26 is small.
        
        # Compute squared distances
        # diff_x[i, j] = cx[i] - cx[j]
        diff_x = cx[:, np.newaxis] - cx[np.newaxis, :]
        diff_y = cy[:, np.newaxis] - cy[np.newaxis, :]
        dist_sq = diff_x**2 + diff_y**2
        
        # Sum of radii matrix
        sum_r = cr[:, np.newaxis] + cr[np.newaxis, :]
        sum_r_sq = sum_r**2
        
        # Violation: max(0, sum_r_sq - dist_sq)
        # We only care about upper triangle (i < j) to avoid double counting and self (0)
        # But since dist_sq[i,i]=0 and sum_r_sq[i,i] = (2r)^2 > 0, self constraint would be huge.
        # We must exclude i=j.
        
        # Create mask for i < j
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        violation_overlap = sum_r_sq - dist_sq
        # Set lower triangle and diagonal to 0 to ignore
        violation_overlap = np.where(mask, violation_overlap, 0.0)
        
        # Square positive parts
        penalty += np.sum(np.maximum(0, violation_overlap)**2)
        
        return obj + penalty_weight * penalty

    # --- Optimization ---
    # We might need to increase penalty weight if constraints are violated.
    # Let's try a few runs or adaptive penalty.
    
    # Initial run
    res = minimize(objective_and_penalty, x0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 1000, 'ftol': 1e-12})
    
    # If result is not valid, we might need higher penalty or re-optimization.
    # But L-BFGS-B with high penalty usually works for packing if started well.
    
    # Extract solution
    final_vars = res.x
    final_centers = np.column_stack((final_vars[0::3], final_vars[1::3]))
    final_radii = final_vars[2::3]
    
    # Post-processing: Clip negative radii if any (shouldn't happen with bounds)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Validate and print sum
    sum_radii = np.sum(final_radii)
    
    # Just in case, we can try to refine with higher penalty if sum is low or invalid.
    # But we don't have a loop here, just one function call.
    # Let's trust the optimization.
    
    return final_centers, final_radii, sum_radii

# To allow running without calling run_packing directly in some contexts
if __name__ == "__main__":
    centers, radii, s = run_packing()
    print(f"Sum of radii: {s}")
