import numpy as np
from scipy.optimize import minimize

N = 26

def compute_objective(vars, lam):
    """
    Computes the objective: -sum(radii) + lambda * penalty.
    Vectorized implementation for efficiency.
    """
    pts = vars[:2*N].reshape(N, 2)
    rad = vars[2*N:]
    
    # Primary objective: maximize sum of radii
    obj = -np.sum(rad)
    
    # Pairwise distance matrix
    diff = pts[:, None, :] - pts[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)  # Ignore self-distance
    
    # Overlap penalty
    overlap = np.maximum(0, rad[:, None] + rad[None, :] - dist)
    pen_ov = np.sum(overlap**2)
    
    # Boundary penalty (left, right, bottom, top)
    pen_bd = np.sum(np.maximum(0, rad - pts[:, 0])**2)
    pen_bd += np.sum(np.maximum(0, pts[:, 0] + rad - 1)**2)
    pen_bd += np.sum(np.maximum(0, rad - pts[:, 1])**2)
    pen_bd += np.sum(np.maximum(0, pts[:, 1] + rad - 1)**2)
    
    return obj + lam * (pen_ov + pen_bd)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum_r = -1.0
    best_vars = None
    
    # Bounds: centers in [0,1], radii in [1e-6, 0.5]
    bnds = [(0.0, 1.0) for _ in range(2*N)] + [(1e-6, 0.5) for _ in range(N)]
    
    # Multiple random restarts to avoid poor local minima
    for restart in range(5):
        pts0 = np.random.rand(N, 2) * 0.8 + 0.1
        rad0 = np.full(N, 0.08)
        vars0 = np.zeros(3*N)
        vars0[:2*N] = pts0.flatten()
        vars0[2*N:] = rad0
        
        lam = 50.0
        current_vars = vars0.copy()
        
        # Progressive penalty increase
        for step in range(15):
            res = minimize(compute_objective, current_vars, args=(lam,), method='L-BFGS-B', 
                           bounds=bnds, options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
            current_vars = res.x
            lam *= 3.0
            
        pts_final = current_vars[:2*N].reshape(N, 2)
        rad_final = current_vars[2*N:]
        
        # Strict internal validation
        valid = True
        if np.any(rad_final > pts_final[:, 0] + 1e-9) or np.any(rad_final > pts_final[:, 1] + 1e-9):
            valid = False
        if np.any(rad_final > 1.0 - pts_final[:, 0] + 1e-9) or np.any(rad_final > 1.0 - pts_final[:, 1] + 1e-9):
            valid = False
            
        dists = np.sqrt(np.sum((pts_final[:, None, :] - pts_final[None, :, :])**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        if np.any(dists < rad_final[:, None] + rad_final[None, :] - 1e-9):
            valid = False
            
        if valid:
            s = np.sum(rad_final)
            if s > best_sum_r:
                best_sum_r = s
                best_vars = current_vars.copy()
                
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    return centers, radii, float(np.sum(radii))
