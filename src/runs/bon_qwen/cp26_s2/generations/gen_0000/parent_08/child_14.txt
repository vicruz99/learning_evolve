import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize sum of radii.
    Uses an iterative grow-and-relax strategy with a hexagonal initialization.
    """
    n = 26
    np.random.seed(42)

    # 1. Initialization: Hexagonal lattice pattern
    # Generate a dense grid and select 26 points
    # This serves as a robust starting point for the optimizer
    grid_x = np.linspace(0.1, 0.9, 7)
    grid_y = np.linspace(0.1, 0.9, 7)
    xx, yy = np.meshgrid(grid_x, grid_y)
    points = np.vstack([xx.ravel(), yy.ravel()]).T
    
    # Select 26 points closest to the center to utilize the central space
    center = np.array([0.5, 0.5])
    dists = np.linalg.norm(points - center, axis=1)
    sorted_indices = np.argsort(dists)[:n]
    
    centers = points[sorted_indices]
    radii = np.full(n, 0.02) # Start with small non-overlapping radii

    # 2. Optimization Loop
    # Iteratively increase radii and optimize positions to resolve overlaps
    current_centers = centers
    current_radii = radii
    
    # Adaptive growth parameters
    growth_step = 0.0005 
    max_iterations = 1000
    
    for step in range(max_iterations):
        # Try to grow radii
        new_radii = current_radii + growth_step
        
        # Safety check: Radii cannot exceed 0.5
        if np.any(new_radii > 0.5):
            break
            
        # Define bounds for the optimizer
        # Each circle's center (x, y) must be in [r, 1-r]
        bounds = []
        for r in new_radii:
            bounds.append((r, 1 - r)) # x bound
            bounds.append((r, 1 - r)) # y bound
            
        # Flatten centers for the optimizer
        x0 = current_centers.flatten()
        
        # Objective function: Minimize squared overlaps
        # Vectorized calculation of pairwise distances
        def objective(x):
            c = x.reshape(n, 2)
            diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            
            # Radii sum matrix
            r_sum = new_radii[:, np.newaxis] + new_radii[np.newaxis, :]
            
            # Overlap penalty: max(0, r_i + r_j - dist)^2
            # We only care about upper triangle (i < j)
            mask = np.triu(np.ones((n, n), dtype=bool), k=1)
            overlap = np.maximum(0, r_sum - dists)
            penalty = np.sum(overlap[mask] ** 2)
            return penalty

        # Run local optimization to resolve overlaps
        # L-BFGS-B is efficient for bound-constrained problems
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 200})
            
            min_penalty = res.fun
            
            # If overlaps are negligible, accept the new larger radii
            if min_penalty < 1e-6:
                current_radii = new_radii
                current_centers = res.x.reshape(n, 2)
                # Occasionally increase growth step to speed up convergence
                if step % 50 == 0 and step > 0:
                    growth_step = min(growth_step * 1.05, 0.002)
            else:
                # If we can't resolve overlaps, the radii might be too large or 
                # we are stuck in a local minimum. 
                # We keep the previous valid state.
                # Optionally reduce growth step if we fail repeatedly
                pass
        except Exception:
            pass

    # 3. Final Cleanup and Validation
    # Ensure all radii are valid and return
    final_sum = np.sum(current_radii)
    
    return current_centers, current_radii, final_sum

if __name__ == "__main__":
    centers, radii, s = run_packing()
    print(f"Sum of radii: {s}")
