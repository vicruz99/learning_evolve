To maximize the sum of radii for 26 circles in a unit square, we treat this as a nonlinear optimization problem. The objective is to maximize $\sum r_i$, which is equivalent to minimizing $-\sum r_i$. The constraints are:
1. Boundary: $0 \le x_i - r_i$ and $x_i + r_i \le 1$ (similarly for $y$)
2. Non-overlap: $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$ for all $i < j$

We use `scipy.optimize.minimize` with the SLSQP algorithm, which handles inequality constraints and bounds efficiently.

- **Bounds**: $x, y \in [0, 1]$ and $r \in [0, 0.5]$ ensure physical validity.
- **No Closures/Lambdas**: All helper functions are defined at module level. Global constants are used for configuration to strictly comply with the rules.

```python
import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
_PAIR_INDICES = np.triu_indices(N_CIRCLES, k=1)

def get_objective(x):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2::3])

def get_constraints(x):
    """
    Returns an array of constraint values. All must be >= 0 for feasibility.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    n = N_CIRCLES
    pts = x.reshape((n, 3))
    xc = pts[:, 0]
    yc = pts[:, 1]
    r = pts[:, 2]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = [xc - r, 1.0 - xc - r, yc - r, 1.0 - yc - r]
    
    # Pairwise distance squared vs required radius sum squared
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dr = r[:, None] + r[None, :]
    
    dist_sq = dx**2 + dy**2
    req_sq = dr**2
    
    # Extract upper triangle for i < j
    c.append(dist_sq[_PAIR_INDICES] - req_sq[_PAIR_INDICES])
    
    return np.concatenate(c)

def run_packing():
    n = N_CIRCLES
    np.random.seed(42)
    
    # Generate initial hexagonal packing configuration
    row_counts = [6, 5, 6, 5, 4]
    s = 1.0
    pts = []
    y_curr = 0.0
    for i, cnt in enumerate(row_counts):
        offset = (i % 2) * s / 2.0
        for j in range(cnt):
            pts.append([offset + j * s, y_curr, 0.08])
        y_curr += s * np.sqrt(3.0) / 2.0
        
    pts = np.array(pts)
    x_min, x_max = pts[:, 0].min(), pts[:, 0].max()
    y_min, y_max = pts[:, 1].min(), pts[:, 1].max()
    
    # Scale to fit within [0.1, 0.9] with margin
    pts[:, 0] = 0.1 + (pts[:, 0] - x_min) / (x_max - x_min) * 0.8
    pts[:, 1] = 0.1 + (pts[:, 1] - y_min) / (y_max - y_min) * 0.8
    
    # Add slight random perturbation to break symmetry
    pts[:, :2] += np.random.uniform(-0.015, 0.015, size=pts[:, :2].shape)
    pts[:, 0] = np.clip(pts[:, 0], 0.05, 0.95)
    pts[:, 1] = np.clip(pts[:, 1], 0.05, 0.95)
    
    x0 = pts.flatten()
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    # Run SLSQP optimization
    res = minimize(get_objective, x0, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False})
    
    centers = res.x.reshape((n, 2))
    radii = res.x[2::3]
    radii = np.maximum(radii, 0.0)  # Ensure non-negative radii
    
    return centers, radii, np.sum(radii)
```
