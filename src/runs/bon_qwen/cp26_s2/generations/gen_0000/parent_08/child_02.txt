import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(x):
    """Objective function to maximize sum of radii (minimize negative sum)."""
    radii = x[2::3]
    return -np.sum(radii)

def constraint_func(x):
    """Compute constraint values. All returned values must be >= 0."""
    num_boundary = N_CIRCLES * 4
    num_overlap = N_CIRCLES * (N_CIRCLES - 1) // 2
    cons = np.empty(num_boundary + num_overlap)
    
    idx = 0
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    for i in range(N_CIRCLES):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        cons[idx] = xi - ri
        cons[idx+1] = 1.0 - xi - ri
        cons[idx+2] = yi - ri
        cons[idx+3] = 1.0 - yi - ri
        idx += 4
        
    # Overlap constraints: distance between centers >= sum of radii
    for i in range(N_CIRCLES):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        for j in range(i + 1, N_CIRCLES):
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            min_dist = ri + rj
            cons[idx] = dist_sq - min_dist**2
            idx += 1
            
    return cons

def run_packing():
    # Bounds for variables: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_val = -np.inf
    best_x = None
    
    # Multiple restarts to escape local minima and improve packing density
    for k in range(10):
        np.random.seed(k)
        
        # Initial feasible configuration
        # Place centers randomly in [0.1, 0.9] to keep away from boundaries initially
        centers = np.random.rand(N_CIRCLES, 2) * 0.8 + 0.1
        # Start with small radii to guarantee initial feasibility
        radii = np.full(N_CIRCLES, 0.02)
        
        x0 = np.zeros(3 * N_CIRCLES)
        for i in range(N_CIRCLES):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        try:
            res = minimize(objective_func, x0, bounds=bounds, constraints=cons,
                           method='SLSQP', options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False})
            
            # Verify constraints are satisfied with numerical tolerance
            c_vals = constraint_func(res.x)
            if np.min(c_vals) >= -1e-7:
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        raise RuntimeError("Optimization failed to find a valid packing")
        
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.zeros(N_CIRCLES)
    for i in range(N_CIRCLES):
        centers[i] = best_x[3*i:3*i+2]
        radii[i] = best_x[3*i+2]
        
    return centers, radii, best_val
