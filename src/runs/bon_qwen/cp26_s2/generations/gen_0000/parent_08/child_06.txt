import numpy as np

def solve_radii(centers):
    """
    Given centers, solve the LP to find optimal radii maximizing sum of radii.
    Constraints:
    1. r_i >= 0
    2. r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    3. r_i + r_j <= distance(c_i, c_j)
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r) => c = [-1, -1, ..., -1]
    c = -np.ones(n)
    
    # Inequality constraints A_ub * r <= b_ub
    # We will build lists of rows for A_ub and values for b_ub
    rows = []
    bounds_list = []
    
    # Boundary constraints for each circle
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        bounds_list.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        bounds_list.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        bounds_list.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        bounds_list.append(1.0 - y)

    # Pairwise distance constraints: r_i + r_j <= dist
    # Only need i < j
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            bounds_list.append(dist)
            
    A_ub = np.array(rows)
    b_ub = np.array(bounds_list)
    
    # Bounds for r_i: r_i >= 0
    # linprog expects bounds as (min, max)
    r_bounds = [(0, None) for _ in range(n)]
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
        if res.success:
            return res.fun, res.x # returns min sum (negative), so -fun is max sum
        else:
            return -1e9, np.zeros(n) # Infeasible or error
    except Exception:
        return -1e9, np.zeros(n)
