import numpy as np
from scipy.optimize import minimize


def penalty_obj(x, n, penalty):
    centers_local = x[:2 * n].reshape((n, 2))
    radii_local = x[2 * n:]

    val = -np.sum(radii_local)

    for i in range(n):
        xi, yi = centers_local[i]
        ri = radii_local[i]
        val += penalty * max(0, ri - xi) ** 2
        val += penalty * max(0, xi + ri - 1) ** 2
        val += penalty * max(0, ri - yi) ** 2
        val += penalty * max(0, yi + ri - 1) ** 2
        val += penalty * max(0, -ri) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers_local[i, 0] - centers_local[j, 0]
            dy = centers_local[i, 1] - centers_local[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-12)
            overlap = max(0, radii_local[i] + radii_local[j] - dist)
            val += penalty * overlap ** 2

    return val


def obj_func(x, n):
    return -np.sum(x[2 * n:])


def boundary_con(x, i, n):
    ri = x[2 * n + i]
    xi = x[2 * i]
    yi = x[2 * i + 1]
    return min(xi - ri, 1 - xi - ri, yi - ri, 1 - yi - ri)


def pair_con(x, i, j, n):
    xi, yi = x[2 * i], x[2 * i + 1]
    xj, yj = x[2 * j], x[2 * j + 1]
    ri, rj = x[2 * n + i], x[2 * n + j]
    return (xi - xj) ** 2 + (yi - yj) ** 2 - (ri + rj) ** 2


def radius_con(x, i, n):
    return x[2 * n + i]


def generate_hex_initial(n, spacing, offset_x, offset_y, perturbation):
    centers = np.zeros((n, 2))
    idx = 0

    for row in range(7):
        y = offset_y + row * spacing * np.sqrt(3) / 2
        if y > 0.95 and row > 0:
            break
        ncols = 5 if row % 2 == 0 else 4
        for col in range(ncols):
            if idx >= n:
                break
            if row % 2 == 0:
                x = offset_x + col * spacing
            else:
                x = offset_x + spacing / 2 + col * spacing
            x += np.random.randn() * perturbation
            y += np.random.randn() * perturbation
            centers[idx] = [np.clip(x, 0.02, 0.98), np.clip(y, 0.02, 0.98)]
            idx += 1
            if idx >= n:
                break
        if idx >= n:
            break

    return centers


def run_packing():
    n = 26

    best_sum = -np.inf
    best_x = None

    # Try many different initial configurations
    for restart in range(40):
        np.random.seed(restart * 137 + 42)

        # Vary parameters for diversity
        spacing = 0.18 + 0.06 * np.random.rand()
        offset_x = 0.02 + 0.08 * np.random.rand()
        offset_y = 0.02 + 0.08 * np.random.rand()
        perturbation = 0.02 + 0.02 * np.random.rand()

        centers = generate_hex_initial(n, spacing, offset_x, offset_y, perturbation)
        radii = np.ones(n) * (0.07 + 0.03 * np.random.rand())

        # Ensure initial radii are feasible
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            radii[i] = min(radii[i], max_r * 0.5)

        x0 = np.concatenate([centers.flatten(), radii])

        # Phase 1: Penalty-based optimization with increasing penalties
        x = x0.copy()
        penalties = [50, 200, 1000, 5000, 20000, 100000, 500000]

        for penalty in penalties:
            result = minimize(
                penalty_obj, x, args=(n, penalty),
                method='L-BFGS-B',
                bounds=[(0.001, 0.999)] * (2 * n) + [(0.001, 0.5)] * n,
                options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            x = result.x

        # Check feasibility and compute sum
        centers_tmp = x[:2 * n].reshape((n, 2))
        radii_tmp = x[2 * n:]

        # Verify approximate feasibility
        feasible = True
        for i in range(n):
            if (centers_tmp[i, 0] < radii_tmp[i] + 1e-8 or
                centers_tmp[i, 0] > 1 - radii_tmp[i] - 1e-8 or
                centers_tmp[i, 1] < radii_tmp[i] + 1e-8 or
                centers_tmp[i, 1] > 1 - radii_tmp[i] - 1e-8):
                feasible = False
                break

        if feasible:
            # Check pair constraints
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers_tmp[i, 0] - centers_tmp[j, 0]
                    dy = centers_tmp[i, 1] - centers_tmp[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < radii_tmp[i] + radii_tmp[j] - 1e-8:
                        feasible = False
                        break
                if not feasible:
                    break

        current_sum = np.sum(radii_tmp)
        if feasible and current_sum > best_sum:
            best_sum = current_sum
            best_x = x.copy()

    if best_x is None:
        # Fallback: use last result regardless of feasibility
        best_x = x

    # Phase 2: Refine with SLSQP constrained optimization
    centers_opt = best_x[:2 * n].reshape((n, 2))
    radii_opt = best_x[2 * n:]
    x0 = np.concatenate([centers_opt.flatten(), radii_opt])

    # Build constraints for SLSQP
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': boundary_con, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': radius_con, 'args': (i, n)})

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': pair_con, 'args': (i, j, n)})

    try:
        result = minimize(
            obj_func, x0, args=(n,),
            method='SLSQP',
            constraints=cons,
            options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
        )

        if result.success:
            centers_opt = result.x[:2 * n].reshape((n, 2))
            radii_opt = result.x[2 * n:]
            best_sum = -result.fun
    except Exception:
        pass

    # Final cleanup: ensure strict feasibility
    for i in range(n):
        margin = 1e-10
        radii_opt[i] = max(radii_opt[i], 1e-10)
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], radii_opt[i] + margin, 1 - radii_opt[i] - margin)
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], radii_opt[i] + margin, 1 - radii_opt[i] - margin)

    # Iteratively reduce radii if any overlaps remain
    for iteration in range(100):
        max_overlap = 0
        overlap_pair = None

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii_opt[i] + radii_opt[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
                    overlap_pair = (i, j)

        if max_overlap < 1e-12:
            break

        i, j = overlap_pair
        reduction = max_overlap / 2 + 1e-10
        radii_opt[i] = max(radii_opt[i] - reduction, 1e-10)
        radii_opt[j] = max(radii_opt[j] - reduction, 1e-10)

    # Recompute final sum
    final_sum = np.sum(radii_opt)

    return centers_opt, radii_opt, float(final_sum)
