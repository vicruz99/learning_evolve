import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Minimize negative sum of radii to maximize sum of radii."""
    return -np.sum(v[2*N:])

def boundary_constraints(v):
    """Check that all circles are within [0,1]x[0,1]."""
    c = np.empty(4*N)
    for i in range(N):
        c[4*i]     = v[2*i] - v[2*N+i]           # x - r >= 0
        c[4*i+1]   = 1.0 - v[2*i] - v[2*N+i]     # 1 - x - r >= 0
        c[4*i+2]   = v[2*i+1] - v[2*N+i]         # y - r >= 0
        c[4*i+3]   = 1.0 - v[2*i+1] - v[2*N+i]   # 1 - y - r >= 0
    return c

def overlap_constraints(v):
    """Check that no two circles overlap."""
    n_con = N*(N-1)//2
    c = np.empty(n_con)
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            dx = v[2*i] - v[2*j]
            dy = v[2*i+1] - v[2*j+1]
            c[idx] = np.hypot(dx, dy) - (v[2*N+i] + v[2*N+j])
            idx += 1
    return c

def run_packing():
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.04)
    
    # Initialize with a hexagonal-like grid pattern for better convergence
    rows = [5, 6, 5, 6, 4]
    idx = 0
    for row_idx, count in enumerate(rows):
        y = (row_idx + 0.5) * (1.0 / 5.0)
        x_start = 0.5 - (count * 0.2) / 2.0
        for col_idx in range(count):
            x = x_start + col_idx * 0.2
            centers[idx] = [x, y]
            idx += 1
            
    # Add controlled perturbation to break symmetry and avoid grid-lock
    np.random.seed(42)
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    x0 = np.concatenate([centers.flatten(), radii])
    bnds = [(0, 1)] * (2*N) + [(0, 0.5)] * N
    cons = [{'type': 'ineq', 'fun': boundary_constraints},
            {'type': 'ineq', 'fun': overlap_constraints}]
            
    best_res = None
    best_val = -np.inf
    
    # Multiple restarts to find a better local optimum
    for seed in range(5):
        np.random.seed(seed + 100)
        x0_trial = x0 + np.random.uniform(-0.01, 0.01, x0.shape)
        x0_trial = np.clip(x0_trial, 0.01, 0.99)
        x0_trial[2*N:] = 0.04  # Ensure small initial radii for feasibility
        
        try:
            res = minimize(objective, x0_trial, method='SLSQP', 
                           bounds=bnds, constraints=cons, 
                           options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})
            if res.success and -res.fun > best_val:
                best_val = -res.fun
                best_res = res
        except Exception:
            continue
            
    if best_res is not None:
        final_centers = best_res.x[:2*N].reshape(N, 2)
        final_radii = best_res.x[2*N:]
        return final_centers, final_radii, np.sum(final_radii)
    else:
        # Fallback to initial configuration if optimization fails
        return centers, radii, np.sum(radii)
