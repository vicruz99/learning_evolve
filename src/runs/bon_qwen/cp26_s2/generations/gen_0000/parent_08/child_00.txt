import numpy as np
from scipy.optimize import minimize
import math


def run_packing():
    n = 26
    
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    for seed in range(50):
        np.random.seed(seed)
        
        centers, radii = make_initial(n, seed)
        
        # Force-based pre-optimization to get good starting point
        centers, radii = force_optimize(centers, radii, n)
        
        # Validate after force optimization
        if not is_valid_packing(centers, radii):
            # Shrink slightly to fix numerical issues
            for i in range(n):
                radii[i] *= 0.999
            if not is_valid_packing(centers, radii):
                continue
        
        # Transform to optimization variables
        x0 = init_x0(centers, radii, n)
        
        # Run scipy optimization
        result = run_optimization(x0, n)
        
        if result is not None:
            centers_opt, radii_opt = result
            
            if is_valid_packing(centers_opt, radii_opt):
                s = np.sum(radii_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
    
    return best_centers, best_radii, best_sum


def make_initial(n, seed):
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.06)
    
    # Hexagonal-like arrangement: 5 rows with 6,5,6,5,4 circles
    rows = [6, 5, 6, 5, 4]
    idx = 0
    num_rows = 5
    
    for row_i, count in enumerate(rows):
        y = (row_i + 0.5) / num_rows
        for col in range(count):
            if row_i % 2 == 0:
                # Even rows: centered spacing
                x = (col + 0.5) / count
            else:
                # Odd rows: shifted for hexagonal packing
                x = (col + 0.75) / count
            centers[idx] = [x, y]
            idx += 1
    
    # Add random perturbation
    noise = np.random.randn(*centers.shape) * 0.02
    centers = centers + noise
    centers = np.clip(centers, 0.08, 0.92)
    
    return centers, radii


def force_optimize(centers, radii, n, steps=8000):
    centers = centers.copy()
    radii = radii.copy()
    
    dt = 0.005
    growth_rate = 0.00008
    
    for step in range(steps):
        # Gradually grow radii
        for i in range(n):
            radii[i] += growth_rate
        
        # Calculate forces
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = math.sqrt(dist_sq)
                
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-12:
                    # Repulsive force proportional to overlap / distance
                    overlap = min_dist - dist
                    force_mag = overlap / (dist + 1e-12) * 0.5
                    fx = force_mag * dx / dist
                    fy = force_mag * dy / dist
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
            
            # Boundary repulsion
            for dim in range(2):
                if centers[i, dim] - radii[i] < 0:
                    forces[i, dim] += (radii[i] - centers[i, dim]) * 5.0
                if centers[i, dim] + radii[i] > 1.0:
                    forces[i, dim] -= (centers[i, dim] + radii[i] - 1.0) * 5.0
        
        # Update positions
        centers = centers + forces * dt
        
        # Enforce hard boundary constraints
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))
        
        # Reduce growth rate over time for convergence
        if step % 1000 == 0 and step > 0:
            growth_rate *= 0.7
    
    return centers, radii


def init_x0(centers, radii, n):
    u = np.zeros((n, 2))
    for i in range(n):
        denom = 1.0 - 2.0 * radii[i]
        if denom < 1e-10:
            denom = 1e-10
        u[i, 0] = (centers[i, 0] - radii[i]) / denom
        u[i, 1] = (centers[i, 1] - radii[i]) / denom
    # Clip u to [0, 1]
    u = np.clip(u, 0.0, 1.0)
    x0 = np.concatenate([u.flatten(), radii])
    return x0


def u_and_r_from_x(x, n):
    u = x[:2 * n].reshape((n, 2))
    r = x[2 * n:]
    return u, r


def centers_from_u_r(u, r, n):
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i, 0] = r[i] + (1.0 - 2.0 * r[i]) * u[i, 0]
        centers[i, 1] = r[i] + (1.0 - 2.0 * r[i]) * u[i, 1]
    return centers


def constraint_func(x, args):
    n, i, j = args
    u, r = u_and_r_from_x(x, n)
    
    xi = r[i] + (1.0 - 2.0 * r[i]) * u[i, 0]
    yi = r[i] + (1.0 - 2.0 * r[i]) * u[i, 1]
    xj = r[j] + (1.0 - 2.0 * r[j]) * u[j, 0]
    yj = r[j] + (1.0 - 2.0 * r[j]) * u[j, 1]
    
    dx = xi - xj
    dy = yi - yj
    dist = math.sqrt(dx * dx + dy * dy)
    
    return dist - r[i] - r[j]


def obj_func(x, n):
    r = x[2 * n:]
    return -np.sum(r)


def run_optimization(x0, n):
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-9, 0.49)] * n
    
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': constraint_func,
                'args': (n, i, j)
            })
    
    result = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                     constraints=constraints, args=(n,),
                     options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
    
    if not np.isnan(result.fun) and result.fun < -0.05:
        x_opt = result.x
        u_opt = x_opt[:2 * n].reshape((n, 2))
        r_opt = x_opt[2 * n:]
        centers_opt = centers_from_u_r(u_opt, r_opt, n)
        return centers_opt, r_opt
    
    return None


def is_valid_packing(centers, radii):
    n = centers.shape[0]
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-12:
            return False
        if x - r < -1e-9 or x + r > 1.0 + 1e-9:
            return False
        if y - r < -1e-9 or y + r > 1.0 + 1e-9:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    
    return True
