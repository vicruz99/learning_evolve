import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # --- 1. Initialization ---
    # Start with a 5x5 grid configuration (25 circles)
    # Place 25 circles in a grid with initial radius 0.09 to leave room for the 26th
    grid_radii = 0.09
    centers = []
    radii = []
    
    # 5x5 grid positions
    step = (1.0 - 2 * grid_radii) / 4.0
    start = grid_radii
    
    for i in range(5):
        for j in range(5):
            cx = start + i * step
            cy = start + j * step
            centers.append([cx, cy])
            radii.append(grid_radii)
            
    # Add 26th circle in the center of the grid gap
    # The gap is roughly at (0.5, 0.5) if grid is centered, but our grid starts at 0.09
    # Grid centers are at 0.09, 0.3225, 0.555, 0.7875, 1.02? No.
    # With step ~0.206, centers: 0.09, 0.296, 0.502, 0.708, 0.914.
    # Gaps are around 0.193, 0.4, 0.607, 0.813.
    # Center gap approx (0.5, 0.5).
    # Let's place 26th at (0.5, 0.5) with a small radius
    centers.append([0.5, 0.5])
    radii.append(0.02) # Small initial radius
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds for variables: x,y in [0,1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # --- 2. Objective and Constraints ---
    
    def objective(vars):
        # Maximize sum of radii => Minimize negative sum
        r_sum = 0.0
        for i in range(n):
            r_sum += vars[3*i+2]
        return -r_sum

    def constraint_boundary(vars):
        # Constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        # Equivalent to: r <= x, x <= 1-r, r <= y, y <= 1-r
        # Or: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        cons = []
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i+1]
            r = vars[3*i+2]
            cons.append(x - r)
            cons.append(1.0 - x - r)
            cons.append(y - r)
            cons.append(1.0 - y - r)
        return cons

    def constraint_no_overlap(vars):
        # Constraint: dist(i, j) >= r_i + r_j
        # Equivalent to: dist(i, j) - r_i - r_j >= 0
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i = vars[3*i], vars[3*i+1]
                x_j, y_j = vars[3*j], vars[3*j+1]
                r_i, r_j = vars[3*i+2], vars[3*j+2]
                
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                cons.append(dist - r_i - r_j)
        return cons

    # Combine constraints
    def all_constraints(vars):
        c_b = constraint_boundary(vars)
        c_n = constraint_no_overlap(vars)
        return c_b + c_n

    # Define constraint dictionary for scipy
    # Type 'ineq' means value >= 0
    cons_dict = {'type': 'ineq', 'fun': all_constraints}

    # --- 3. Optimization ---
    
    # Use SLSQP which handles bounds and constraints
    # Nelder-Mead or COBYLA could also work but SLSQP is robust for this
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints=cons_dict, 
                       options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})
        final_vars = res.x
    except Exception:
        # Fallback if optimization fails
        final_vars = x0
        
    # --- 4. Extract Results ---
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_vars[3*i]
        final_centers[i, 1] = final_vars[3*i+1]
        final_radii[i] = final_vars[3*i+2]
        
    sum_radii = np.sum(final_radii)
    
    # Small adjustment to handle numerical precision issues if any (clip radii)
    final_radii = np.clip(final_radii, 0.0, None)
    
    return final_centers, final_radii, sum_radii
