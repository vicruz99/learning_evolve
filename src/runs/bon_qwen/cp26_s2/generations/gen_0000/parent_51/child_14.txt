My approach is based on the key insight that for fixed circle centers, computing optimal radii is a Linear Program (LP). This allows me to:

1. **Separate the optimization**: Optimize centers using a non-linear optimizer, and for each center configuration, solve an LP to find the maximum sum of radii.

2. **Multiple initial configurations**: Try several starting arrangements including hexagonal packing, grid packing, and random perturbations.

3. **Iterative refinement**: Use the LP-based evaluation with Nelder-Mead or COBYLA optimizer on centers, then extract optimal radii.

4. **Post-processing**: Apply local adjustments to squeeze out more radius sum.

The LP formulation for fixed centers:
- Maximize sum(r_i)
- Subject to: r_i <= distance to each boundary, r_i + r_j <= distance(i,j) for all pairs, r_i >= 0

This is convex and solved efficiently by scipy's linprog. The overall objective (sum of optimal radii as a function of centers) is continuous but not differentiable, making derivative-free optimizers ideal.

I'll use multiple restarts with different initial configurations to avoid local optima, and apply multiple passes of optimization with increasing precision.

import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution
from scipy.spatial.distance import pdist, squareform


def get_optimal_radii(centers):
    """Given centers, compute optimal radii using LP to maximize sum of radii."""
    n = centers.shape[0]
    if n == 0:
        return np.array([])
    
    dist_matrix = squareform(pdist(centers))
    
    c = -np.ones(n)
    
    A_ub_list = []
    b_ub_list = []
    
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1
        A_ub_list.append(row)
        b_ub_list.append(x)
        
        row = np.zeros(n)
        row[i] = 1
        A_ub_list.append(row)
        b_ub_list.append(1 - x)
        
        row = np.zeros(n)
        row[i] = 1
        A_ub_list.append(row)
        b_ub_list.append(y)
        
        row = np.zeros(n)
        row[i] = 1
        A_ub_list.append(row)
        b_ub_list.append(1 - y)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub_list.append(row)
            b_ub_list.append(dist_matrix[i, j])
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and np.all(result.x >= -1e-10):
        return np.maximum(result.x, 0)
    else:
        return np.ones(n) * 0.01


def objective_from_centers(centers, n):
    """Return negative sum of optimal radii for given centers."""
    radii = get_optimal_radii(centers)
    return -np.sum(radii)


def hexagonal_init(n, r):
    """Create hexagonal packing initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        y = r + row * r * np.sqrt(3)
        if row % 2 == 0:
            ncols = min(6, n - idx)
            sx = r
        else:
            ncols = min(5, n - idx)
            sx = 2 * r
        for col in range(ncols):
            if idx >= n:
                break
            centers[idx, 0] = sx + col * 2 * r
            centers[idx, 1] = y
            idx += 1
        row += 1
    return centers


def grid_init(n, spacing):
    """Create grid packing initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(10):
        for j in range(10):
            if idx >= n:
                return centers
            centers[idx, 0] = spacing * (i + 1)
            centers[idx, 1] = spacing * (j + 1)
            idx += 1
    return centers


def random_init(n, low=0.05, high=0.95):
    """Create random initialization."""
    centers = np.random.uniform(low, high, size=(n, 2))
    return centers


def run_packing():
    n = 26
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    initial_configs = []
    
    # Hexagonal with various radii
    for r in [0.07, 0.08, 0.09, 0.095]:
        initial_configs.append(hexagonal_init(n, r))
    
    # Grid initializations
    for sp in [0.1, 0.12, 0.15, 0.2]:
        initial_configs.append(grid_init(n, sp))
    
    # Random initializations
    for _ in range(5):
        initial_configs.append(random_init(n))
    
    # Optimized hexagonal (6-5-6-5-4 pattern)
    centers_hex = np.zeros((n, 2))
    r0 = 0.085
    row_configs = [(6, r0), (5, 2*r0), (6, r0), (5, 2*r0), (4, r0)]
    idx = 0
    for row, (ncols, sx) in enumerate(row_configs):
        y = r0 + row * r0 * np.sqrt(3)
        for col in range(ncols):
            if idx >= n:
                break
            centers_hex[idx, 0] = sx + col * 2 * r0
            centers_hex[idx, 1] = y
            idx += 1
    initial_configs.append(centers_hex)
    
    for init_idx, init_centers in enumerate(initial_configs):
        # Ensure centers are in valid range
        init_centers = np.clip(init_centers, 0.01, 0.99)
        
        # Phase 1: Coarse optimization with Nelder-Mead
        def obj(v):
            return objective_from_centers(v.reshape(n, 2), n)
        
        x0 = init_centers.flatten()
        
        try:
            result = minimize(
                obj, x0,
                method='Nelder-Mead',
                options={
                    'maxiter': 5000,
                    'xatol': 1e-6,
                    'fatol': 1e-6,
                    'adaptive': True
                }
            )
            centers_opt = result.x.reshape(n, 2)
        except Exception:
            centers_opt = init_centers
        
        # Phase 2: Fine optimization with COBYLA
        try:
            result2 = minimize(
                obj, centers_opt.flatten(),
                method='COBYLA',
                options={
                    'maxiter': 3000,
                    'rhobeg': 0.01,
                    'rhoend': 1e-6,
                    'catol': 1e-8
                }
            )
            centers_opt = result2.x.reshape(n, 2)
        except Exception:
            pass

                    'rhobeg': 0.005,
                    'rhoend': 1e-8,
                    'catol': 1e-10
                }
            )

        except Exception:
            pass

        centers_opt = np.clip(centers_opt, 0.001, 0.999)

        radii_opt = get_optimal_radii(centers_opt)
        
        current_sum = np.sum(radii_opt)

            best_sum = current_sum

    # Additional refinement: local perturbations

    radii_refined = best_radii.copy()
    
    for perturbation_pass in range(3):
        for i in range(n):
            best_local_sum = np.sum(radii_refined)

                           (0.01, 0.01), (-0.01, -0.01), (0.01, -0.01), (-0.01, 0.01)]:
                centers_test = centers_refined.copy()
                centers_test[i] += [dx, dy]
                centers_test = np.clip(centers_test, 0.001, 0.999)
                
                radii_test = get_optimal_radii(centers_test)
                test_sum = np.sum(radii_test)
                
                if test_sum > best_local_sum + 1e-10:
                    best_local_sum = test_sum
                    centers_refined = centers_test.copy()
                    radii_refined = radii_test.copy()
    
    best_centers = centers_refined
    best_radii = radii_refined
    best_sum = np.sum(best_radii)
    
    # Final validation and cleanup
    best_radii = np.maximum(best_radii, 0)
    for i in range(n):
        r = best_radii[i]
        best_centers[i, 0] = max(r, min(1 - r, best_centers[i, 0]))
        best_centers[i, 1] = max(r, min(1 - r, best_centers[i, 1]))
    
    return best_centers, best_radii, best_sum
