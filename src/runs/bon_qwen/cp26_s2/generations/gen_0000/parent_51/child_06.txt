import numpy as np
import scipy.optimize as opt

def get_hexagonal_centers(n):
    """Generates n centers in a hexagonal pattern inside the unit square."""
    centers = []
    r_est = 0.1 # Initial estimate
    spacing = 2 * r_est
    
    # Hexagonal packing parameters
    dx = spacing
    dy = spacing * np.sqrt(3) / 2
    
    row = 0
    while len(centers) < n:
        for col in range(n): # Iterate enough columns
            x = col * dx + (row % 2) * (dx / 2)
            y = row * dy
            
            # Adjust to center in square roughly
            # We will normalize later
            centers.append([x, y])
            if len(centers) >= n:
                break
        row += 1
    
    centers = np.array(centers[:n])
    
    # Normalize to fit in [0, 1] x [0, 1]
    min_c = centers.min(axis=0)
    max_c = centers.max(axis=0)
    size = max_c - min_c
    if size.max() > 0:
        scale = 0.95 / size.max() # Leave some margin
        centers = (centers - min_c) * scale + 0.5 * (1 - 0.95)
    else:
        centers = np.full((n, 2), 0.5)
        
    return centers

def calculate_radii(centers, penalty=1e6):
    """
    Calculates radii for a given set of centers using a penalty-based relaxation.
    This approximates the solution to the LP: max sum(r) s.t. r_i + r_j <= dist, r_i <= boundary.
    """
    n = centers.shape[0]
    radii = np.ones(n) * 0.1 # Initial radii
    
    # Iterative update to find stable radii
    # We treat constraints as forces pushing radii down or positions apart.
    # Here we just solve for radii given fixed positions.
    # r_i <= min( dist(i, boundary), min_j( dist(i, j) - r_j ) )
    # This is a system of inequalities. We can solve it via fixed-point iteration or LP.
    # For speed in optimization loop, let's use a simple iterative relaxation.
    
    # Initialize radii based on distance to boundary
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(x, 1-x, y, 1-y)
    
    # Iterate to satisfy pairwise constraints r_i + r_j <= d_ij
    for _ in range(50):
        for i in range(n):
            r_i = radii[i]
            for j in range(n):
                if i == j: continue
                d_ij = np.linalg.norm(centers[i] - centers[j])

            x, y = centers[i]
            bound = min(x, 1-x, y, 1-y)
            if radii[i] > bound:
                radii[i] = bound
                
    return radii

        return 1e9 # Penalty for bad centers

        return 1e9

    # Calculate overlap violations just to guide optimizer away from bad regions
    # though calculate_radii reduces radii to avoid overlap.
    # We want to maximize sum(radii).
    
    sum_radii = np.sum(radii)
    
    return -sum_radii

def run_packing():
    n = 26
    
    # 1. Initialize centers
    centers = get_hexagonal_centers(n)
    
    # 2. Optimize centers
