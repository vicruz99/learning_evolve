    attempts.append((np.array(c_hex), np.array(r_hex)))
    
    # Random attempts
    for _ in range(5):
        c_rand = np.random.uniform(0.05, 0.95, (n, 2))
        r_rand = np.ones(n) * 0.05
        attempts.append((c_rand, r_rand))
        
    for c_init, r_init in attempts:
        centers = c_init.copy()
        radii = r_init.copy()
        
        lr = 0.01 
        steps = 1000
        
        for step in range(steps):
            for i in range(n):
                max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
                for j in range(n):
                    if i == j: continue
                    dist = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                    max_r = min(max_r, dist - radii[j])
                
                target_r = max(0.0, max_r)
                radii[i] = radii[i] + 0.1 * (target_r - radii[i])
            
            for i in range(n):
                fx, fy = 0.0, 0.0
                for j in range(n):
                    if i == j: continue
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    if dist < 1e-9: dist = 1e-9
                    contact_dist = radii[i] + radii[j]
                    if dist < contact_dist:
                        repulsion = (contact_dist - dist) / dist
                        fx += repulsion * dx
                        fy += repulsion * dy
                
                centers[i, 0] += lr * fx
                centers[i, 1] += lr * fy
                
                # Clip to valid range
                r_i = radii[i]
                centers[i, 0] = np.clip(centers[i, 0], r_i, 1.0 - r_i)
                centers[i, 1] = np.clip(centers[i, 1], r_i, 1.0 - r_i)
            
            if step % 200 == 0:
                lr *= 0.8

        if check_validity(centers, radii, n):
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()
    
    if best_centers is None:
        centers = []
        radii = []
        for i in range(5):
            for j in range(5):
                centers.append([0.1 + i*0.2, 0.1 + j*0.2])
                radii.append(0.1)
        centers.append([0.2, 0.2])
        radii.append(0.04)
        best_centers = np.array(centers)
        best_radii = np.array(radii)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
