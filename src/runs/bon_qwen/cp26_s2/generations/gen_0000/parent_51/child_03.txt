import numpy as np
from scipy.optimize import minimize, NonlinearConstraint, LinearConstraint

def run_packing():
    n = 26
    num_vars = 3 * n  # x1, y1, r1, x2, y2, r2, ...

    # 1. Initialization: Hexagonal Grid
    # We generate a hexagonal lattice of points.
    # Spacing 's' determines density. We want ~26 points in [0,1]x[0,1].
    # Area per point ~ 1/26 ~ 0.038. Hex area ~ s^2 * sqrt(3)/2.
    # s ~ sqrt(2*0.038/1.732) ~ 0.21.
    s = 0.205
    
    pts = []
    y = s
    row = 0
    while y <= 1.0 - s:
        x_offset = 0.0
        if row % 2 == 1:
            x_offset = s * 0.5
        
        x = x_offset + s
        while x <= 1.0 - s:
            pts.append([x, y])
            x += s
        y += s * np.sqrt(3) / 2
        row += 1
    
    # Ensure we have enough points
    if len(pts) < n:
        # Fallback to smaller spacing if grid didn't generate enough
        pts = []
        s = 0.15
        y = s
        row = 0
        while y <= 1.0 - s:
            x_offset = 0.0
            if row % 2 == 1:
                x_offset = s * 0.5
            x = x_offset + s
            while x <= 1.0 - s:
                pts.append([x, y])
                x += s
            y += s * np.sqrt(3) / 2
            row += 1

    # Select first 26 points
    initial_centers = np.array(pts[:n])
    initial_radii = np.full(n, 0.05) # Start with small radii to be feasible

    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(num_vars)
    for i in range(n):
        x0[3*i] = initial_centers[i, 0]
        x0[3*i+1] = initial_centers[i, 1]
        x0[3*i+2] = initial_radii[i]

    # 2. Bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * num_vars
    for i in range(n):
        bounds[3*i+2] = (0, 0.5)

    # 3. Linear Constraints (Boundary conditions)
    # Constraints:
    # x_i - r_i >= 0
    # x_i + r_i <= 1
    # y_i - r_i >= 0
    # y_i + r_i <= 1
    # Formatted as lb <= A*x <= ub
    
    A_lin = np.zeros((4*n, num_vars))
    lb_lin = -np.inf * np.ones(4*n)
    ub_lin = np.inf * np.ones(4*n)

    for i in range(n):
        idx_x = 3*i
        idx_y = 3*i + 1
        idx_r = 3*i + 2

        # x - r >= 0
        row = 4*i
        A_lin[row, idx_x] = 1.0
        A_lin[row, idx_r] = -1.0
        lb_lin[row] = 0.0

        # x + r <= 1
        row = 4*i + 1
        A_lin[row, idx_x] = 1.0
        A_lin[row, idx_r] = 1.0
        ub_lin[row] = 1.0

        # y - r >= 0
        row = 4*i + 2
        A_lin[row, idx_y] = 1.0
        A_lin[row, idx_r] = -1.0
        lb_lin[row] = 0.0

        # y + r <= 1
        row = 4*i + 3
        A_lin[row, idx_y] = 1.0
        A_lin[row, idx_r] = 1.0
        ub_lin[row] = 1.0

    lin_constraint = LinearConstraint(A_lin, lb_lin, ub_lin)

    # 4. Nonlinear Constraints (Pairwise non-overlap)
    # dist(i, j) - r_i - r_j >= 0
    
    def nonlinear_fun(vars):
        centers = vars[:2*n].reshape(n, 2)
        radii = vars[2*n:]
        
        # Compute pairwise distances
        # diff[i, j] = centers[i] - centers[j]
        # Using broadcasting
        c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
        dists = np.sqrt(np.sum(c_diff**2, axis=2)) # (n, n)
        
        # We need upper triangle entries i < j
        # Create mask or index
        idxs = np.triu_indices(n, k=1)
        d_vals = dists[idxs]
        
        # Constraint values: dist - (r_i + r_j)
        r_sum = radii[idxs[0]] + radii[idxs[1]]
        return d_vals - r_sum

    def nonlinear_jac(vars):
        centers = vars[:2*n].reshape(n, 2)
        radii = vars[2*n:]
        
        # Compute distances for pairs
        idxs = np.triu_indices(n, k=1)
        i_indices = idxs[0]
        j_indices = idxs[1]
        
        c_diff = centers[i_indices] - centers[j_indices] # (m, 2) where m = n(n-1)/2
        dists = np.sqrt(np.sum(c_diff**2, axis=1)) # (m,)
        
        # Avoid division by zero
        dists_safe = np.where(dists > 1e-12, dists, 1e-12)
        
        # Jacobian is a matrix of size (m, num_vars)
        # It's sparse, but we return dense for simplicity given small size
        jac = np.zeros((len(i_indices), num_vars))
        
        # Gradients for centers
        # d(dist)/d(c_xi) = (xi - xj) / dist
        # d(dist)/d(c_xj) = -(xi - xj) / dist
        
        dx = c_diff[:, 0]
        dy = c_diff[:, 1]
        
        grad_dx_i = dx / dists_safe
        grad_dy_i = dy / dists_safe
        grad_dx_j = -dx / dists_safe
        grad_dy_j = -dy / dists_safe
        
        # Fill Jacobian rows
        # Variables order: x, y, r for each circle
        
        # For circle i
        jac[np.arange(len(i_indices)), 3*i_indices] = grad_dx_i
        jac[np.arange(len(i_indices)), 3*i_indices + 1] = grad_dy_i
        jac[np.arange(len(i_indices)), 3*i_indices + 2] = -1.0 # dr_i
        
        # For circle j
        jac[np.arange(len(i_indices)), 3*j_indices] = grad_dx_j
        jac[np.arange(len(i_indices)), 3*j_indices + 1] = grad_dy_j
        jac[np.arange(len(i_indices)), 3*j_indices + 2] = -1.0 # dr_j
        
        return jac

    nonlin_constraint = NonlinearConstraint(nonlinear_fun, 0, np.inf, jac=nonlinear_jac)

    # 5. Optimization
    # Minimize -sum(radii)
    def objective(vars):
        return -np.sum(vars[2*n:])

    try:
        res = minimize(objective, x0, method='trust-constr', bounds=bounds,
                       constraints=[lin_constraint, nonlin_constraint],
                       options={'maxiter': 1000, 'verbose': 0})
        
        if res.success:
            best_x = res.x
        else:
            # If optimization failed, try to recover from initial or last step
            # But usually it converges. Fallback to initial.
            best_x = x0 
    except Exception:
        best_x = x0

    # 6. Extract Results
    final_centers = best_x[:2*n].reshape(n, 2)
    final_radii = best_x[2*n:]
    
    # Clip radii to be non-negative (just in case)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Clip centers to [0, 1] (just in case)
    final_centers = np.clip(final_centers, 0, 1)
    
    # Adjust radii if they exceed boundaries slightly due to numerical errors
    # though constraints should handle it.
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        # Enforce boundary strictly
        r = min(r, x, 1-x, y, 1-y)
        if r < 0: r = 0
        final_radii[i] = r
        final_centers[i] = (x, y)

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
