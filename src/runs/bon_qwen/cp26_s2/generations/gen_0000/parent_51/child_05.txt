import numpy as np
from scipy.optimize import minimize, linprog

def initialize_centers(n):
    """Initialize centers in a hexagonal lattice pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.15
    for row in range(7):
        if row % 2 == 0:
            x_start = 0.15
            count = 5
        else:
            x_start = 0.25
            count = 4
        for col in range(count):
            if idx < n:
                centers[idx] = [x_start + col * 0.2, y]
                idx += 1
        y += 0.15 * np.sqrt(3)
    while idx < n:
        centers[idx] = np.random.rand(2) * 0.8 + 0.1
        idx += 1
    return centers

def compute_loss(params, n):
    """Objective function: maximize sum of radii with penalties for violations."""
    c = params[:n*2].reshape((n, 2))
    r = params[n*2:]
    pen = 0.0
    for i in range(n):
        ri = r[i]
        # Boundary penalties
        for k in [0, 1]:
            if c[i, k] - ri < 0:
                pen += (c[i, k] - ri)**2
            if c[i, k] + ri > 1:
                pen += (c[i, k] + ri - 1)**2
        # Overlap penalties
        for j in range(i+1, n):
            d = np.sqrt(np.sum((c[i] - c[j])**2))
            if d < r[i] + r[j]:
                pen += (r[i] + r[j] - d)**2
    return -np.sum(r) + 1000.0 * pen

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds_opt = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    # Multiple restarts to find global optimum
    for _ in range(15):
        centers0 = initialize_centers(n)
        centers0 += np.random.randn(n, 2) * 0.02
        centers0 = np.clip(centers0, 0.05, 0.95)
        
        radii0 = np.full(n, 0.08)
        x0 = np.concatenate([centers0.flatten(), radii0])
        
        res = minimize(compute_loss, x0, args=(n,), method='L-BFGS-B', bounds=bounds_opt, 
                       options={'maxiter': 2000, 'ftol': 1e-12})
        
        opt_centers = res.x[:n*2].reshape((n, 2))
        
        # Compute pairwise distances
        dists = np.sqrt(np.sum((opt_centers[:, np.newaxis, :] - opt_centers[np.newaxis, :, :])**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        
        # Formulate LP: maximize sum(r_i) s.t. r_i + r_j <= d_ij, r_i <= boundary_dist_i, r_i >= 0
        A_ub = []
        b_ub = []
        for i in range(n):
            for j in range(i+1, n):
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dists[i, j])
                
        bounds_lp = []
        for i in range(n):
            b_lim = min(opt_centers[i,0], 1-opt_centers[i,0], 
                        opt_centers[i,1], 1-opt_centers[i,1])
            bounds_lp.append((0, max(b_lim, 0.0)))
            
        c_obj = -np.ones(n)
        res_lp = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        
        if res_lp.success:
            curr_sum = np.sum(res_lp.x)
            if curr_sum > best_sum:

                best_centers = opt_centers.copy()
                best_radii = res_lp.x.copy()
                
    if best_radii is None:
        best_centers = initialize_centers(n)
        best_radii = np.full(n, 0.05)
        
    # Apply safety margin for numerical validation
    best_radii = np.maximum(best_radii - 1e-12, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
