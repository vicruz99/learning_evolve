import numpy as np
from scipy.optimize import minimize

def compute_radii(centers):
    """
    Computes the maximum possible radii for a given set of fixed centers
    such that circles remain within the unit square and do not overlap.
    Uses an iterative relaxation method equivalent to solving the dual problem.
    """
    n = centers.shape[0]
    radii = np.full(n, 0.5)
    for _ in range(100):
        new_radii = np.zeros(n)
        for i in range(n):
            # Constraint from boundaries
            br = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
            # Constraint from other circles
            min_d = np.inf
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                # r_i + r_j <= dist  =>  r_i <= dist - r_j
                val = dist - radii[j]
                if val < min_d:
                    min_d = val
            new_radii[i] = min(br, min_d)
        radii = np.maximum(new_radii, 0.0)
    return radii

def objective(centers_flat):
    centers = centers_flat.reshape(-1, 2)
    radii = compute_radii(centers)
    # We maximize the sum of radii, so minimize negative sum
    return -np.sum(radii)

def run_packing():

    # Initialize with a hexagonal lattice configuration
    centers_init = np.zeros((n, 2))
    idx = 0
    s = 0.18  # Base spacing for initialization
    for row in range(5):
        # Row lengths: 6, 5, 6, 5, 4 -> Total 26
        ncols = 6 if row % 2 == 0 else 5
        if row == 4:
            ncols = 4
        for col in range(ncols):
            x = col * s + (row % 2) * s * 0.5
            y = row * s * np.sqrt(3) / 2
            # Scale to fit within [0,1] roughly
            centers_init[idx, 0] = x / (5.0 * s)
            centers_init[idx, 1] = y / (4.0 * s * np.sqrt(3) / 2)
            idx += 1
            
    # Ensure initial centers are strictly inside the square
    centers_init = np.clip(centers_init, 0.05, 0.95)
    
    best_score = np.inf
    best_centers = centers_init

    for _ in range(15):
        c = centers_init + np.random.randn(n, 2) * 0.01
        c = np.clip(c, 0.02, 0.98)
        
        res = minimize(objective, c.flatten(), method='L-BFGS-B', 
                       bounds=[(0.01, 0.99)] * (2 * n),
                       options={'ftol': 1e-12, 'maxiter': 500})
        
        if res.fun < best_score:
            best_score = res.fun
            best_centers = res.x.reshape(n, 2)
            
    radii = compute_radii(best_centers)
    radii = np.maximum(radii, 0.0)
    
    return best_centers, radii, np.sum(radii)
