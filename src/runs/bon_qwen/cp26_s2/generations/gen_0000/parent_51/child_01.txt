def compute_forces_vectorized(centers, radii, n):
    """Compute repulsion forces between all circle pairs and from boundaries."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diff ** 2, axis=2) + 1e-20

    dirs = diff / dists[:, :, np.newaxis]
    force_contributions = overlaps[:, :, np.newaxis] * dirs
    forces = np.sum(force_contributions, axis=1)
    
    return forces


def init_hexagonal_packing(n=26, r=None):
    """Initialize circles in a hexagonal arrangement."""
    row_counts = [5, 4, 5, 4, 5, 3]
    centers = []
    
    if r is None:
        r = 0.085
    
    for row_idx, count in enumerate(row_counts):
        y = r + row_idx * r * np.sqrt(3)
        if row_idx % 2 == 0:
            x_start = r
        else:
            x_start = 2 * r
        
        for i in range(count):
            centers.append([x_start + i * 2 * r, y])
    
    centers = np.array(centers)
    radii = np.full(n, r)
    return centers, radii


def init_random_packing(n=26, seed=42):
    """Initialize circles randomly in the unit square."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.15, 0.85, size=(n, 2))
    radii = np.full(n, 0.02)
    return centers, radii


def run_single_optimization(centers, radii, n, max_iter=150000, growth_rate=1.00008, 
                            step_size=0.008, add_jitter=True, jitter_freq=2000,
                            jitter_mag=0.002):
    """Run force-based optimization for a single configuration."""
    for iteration in range(max_iter):
        # Gradually increase radii
        radii *= growth_rate
        
        # Compute repulsion forces
        forces = compute_forces_vectorized(centers, radii, n)
        
        # Boundary repulsion forces
        for i in range(n):
            for d in range(2):
                if centers[i, d] < radii[i]:
                    forces[i, d] += (radii[i] - centers[i, d]) * 50
                elif centers[i, d] > 1 - radii[i]:
                    forces[i, d] -= (centers[i, d] - (1 - radii[i])) * 50
        
        # Add random jitter to escape local minima
        if add_jitter and iteration % jitter_freq == 0:
            rng = np.random.RandomState(iteration)
            forces += rng.uniform(-jitter_mag, jitter_mag, size=(n, 2))
        
        # Adaptive step size (decrease over time)
        current_step = step_size * max(0.1, 1 - iteration / (max_iter * 1.5))
        
        # Update positions
        centers += forces * current_step
        
        # Clamp positions to valid range

            for d in range(2):
                centers[i, d] = max(radii[i], min(1 - radii[i], centers[i, d]))
    
    return centers, radii, np.sum(radii)

def fix_small_violations(centers, radii, n):
    """Fix small constraint violations by adjusting radii."""
    for i in range(n):
        radii[i] = min(radii[i], centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = max(radii[i], 0)
