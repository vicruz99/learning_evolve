import numpy as np
from scipy.optimize import minimize
from functools import partial


def bound_xmin_fn(x, n, i):
    return x[2 * i] - x[2 * n + i]


def bound_xmax_fn(x, n, i):
    return 1.0 - x[2 * i] - x[2 * n + i]


def bound_ymin_fn(x, n, i):
    return x[2 * i + 1] - x[2 * n + i]


def bound_ymax_fn(x, n, i):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def non_overlap_fn(x, n, i, j):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy + 1e-20)
    return dist - x[2 * n + i] - x[2 * n + j]


def non_neg_rad_fn(x, n, i):
    return x[2 * n + i]


def objective_fn(x, n):
    return -np.sum(x[2 * n:])


def compute_initial_radii(centers, n):
    radii = np.zeros(n)
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                max_r = min(max_r, dist)
        radii[i] = max(0.001, max_r / 2.0)
    return radii


def init_hexagonal(n):
    r_init = 0.085
    centers_list = []

    row = 0
    while len(centers_list) < n:
        col = 0
        row_done = False
        while len(centers_list) < n and not row_done:
            x = r_init + col * 2 * r_init + (row % 2) * r_init
            y = r_init + row * np.sqrt(3) * r_init
            if x + r_init <= 1.0 and y + r_init <= 1.0:
                centers_list.append([x, y])
            col += 1
            if col * 2 * r_init + 2 * r_init > 1.0 and col > 0:
                row_done = True
        row += 1
        if row * np.sqrt(3) * r_init + 2 * r_init > 1.0 and row > 0:
            break

    centers = np.array(centers_list[:n])
    if len(centers) < n:
        remaining = n - len(centers)
        extra = np.tile([0.5, 0.5], (remaining, 1))
        centers = np.vstack([centers, extra])

    radii = compute_initial_radii(centers, n)
    return centers, radii


def init_grid(n, nx, ny):
    centers = np.zeros((n, 2))
    count = 0
    for row in range(ny):
        for col in range(nx):
            if count < n:
                centers[count, 0] = (col + 0.5) / nx
                centers[count, 1] = (row + 0.5) / ny
                count += 1
    radii = compute_initial_radii(centers, n)
    return centers, radii


def init_random_with_repulsion(n, seed=42):
    rng = np.random.RandomState(seed)
    centers = rng.rand(n, 2) * 0.8 + 0.1
    radii = np.ones(n) * 0.01

    step = 0.05
    for iteration in range(2000):
        forces = np.zeros_like(centers)
        target_dist = 0.15

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-20)
                if dist < target_dist:
                    f = (target_dist - dist) / dist
                    forces[i, 0] += dx * f
                    forces[i, 1] += dy * f
                    forces[j, 0] -= dx * f
                    forces[j, 1] -= dy * f

        for i in range(n):
            margin = 0.05
            if centers[i, 0] < margin:
                forces[i, 0] += margin - centers[i, 0]
            if centers[i, 0] > 1.0 - margin:
                forces[i, 0] -= centers[i, 0] - (1.0 - margin)
            if centers[i, 1] < margin:
                forces[i, 1] += margin - centers[i, 1]
            if centers[i, 1] > 1.0 - margin:
                forces[i, 1] -= centers[i, 1] - (1.0 - margin)

        centers += forces * step * 0.1
        centers = np.clip(centers, 0.01, 0.99)

    radii = compute_initial_radii(centers, n)
    return centers, radii


def build_constraints(n):
    constraints = []

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': partial(bound_xmin_fn, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(bound_xmax_fn, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(bound_ymin_fn, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(bound_ymax_fn, n=n, i=i)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': partial(non_overlap_fn, n=n, i=i, j=j)})

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': partial(non_neg_rad_fn, n=n, i=i)})

    return constraints


def run_optimization_from(x0, n, constraints):
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    result = minimize(
        partial(objective_fn, n=n),
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 5000, 'ftol': 1e-15}
    )

    return result


def fix_solution(centers, radii, n):
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.maximum(radii, 0.0)

    for _ in range(100):
        changed = False
        for i in range(n):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            max_r = max(0.0, max_r)
            if radii[i] > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break

    return centers, radii


def run_packing():
    n = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    constraints = build_constraints(n)

    initializations = []

    initializations.append(('hex', init_hexagonal(n)))

    for nx, ny in [(5, 6), (6, 5), (4, 7), (7, 4)]:
        name = f'grid_{nx}x{ny}'
        initializations.append((name, init_grid(n, nx, ny)))

    for seed in [42, 123, 456, 789, 101]:
        name = f'random_{seed}'
        initializations.append((name, init_random_with_repulsion(n, seed)))

    for name, (centers_init, radii_init) in initializations:
        x0 = np.concatenate([centers_init.flatten(), radii_init])

        result = run_optimization_from(x0, n, constraints)

        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]

        centers_fixed, radii_fixed = fix_solution(centers_opt, radii_opt, n)
        current_sum = np.sum(radii_fixed)

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_fixed.copy()
            best_radii = radii_fixed.copy()

    best_centers, best_radii = fix_solution(best_centers, best_radii, n)
    final_sum = np.sum(best_radii)

    return best_centers, best_radii, final_sum
