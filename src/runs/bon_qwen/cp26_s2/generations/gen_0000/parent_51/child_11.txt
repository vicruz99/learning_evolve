import numpy as np
from scipy.optimize import minimize
import time

def compute_cost(vars_array, N=26, weight=1000.0):
    """
    Computes the cost function: -sum(radii) + penalty for overlaps/boundaries.
    """
    # Reshape variables: x, y, r for each circle
    # vars_array shape: (78,) -> 26 circles * 3 vars
    centers = vars_array[:-N].reshape(N, 2)
    radii = vars_array[-N:]

    # 1. Calculate sum of radii (to maximize, so we minimize negative sum)
    sum_radii = np.sum(radii)
    
    penalty = 0.0
    
    # 2. Boundary penalties
    # x must be >= r and <= 1-r => r-x <= 0 and r-(1-x) <= 0
    # Violation is max(0, val). We square it for smoothness.
    
    # Left boundary: x >= r
    violation_left = np.maximum(0, radii - centers[:, 0])
    penalty += np.sum(violation_left**2)
    
    # Right boundary: x <= 1 - r => r + x - 1 <= 0
    violation_right = np.maximum(0, radii + centers[:, 0] - 1.0)
    penalty += np.sum(violation_right**2)
    
    # Bottom boundary: y >= r
    violation_bottom = np.maximum(0, radii - centers[:, 1])
    penalty += np.sum(violation_bottom**2)
    
    # Top boundary: y <= 1 - r => r + y - 1 <= 0
    violation_top = np.maximum(0, radii + centers[:, 1] - 1.0)
    penalty += np.sum(violation_top**2)

    # 3. Pairwise overlap penalties
    # dist(i, j) >= r_i + r_j
    # Violation: r_i + r_j - dist(i, j) > 0
    
    # Vectorized distance calculation
    # centers shape (N, 2)
    # diff shape (N, N, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # radii sums matrix (N, N)
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violation matrix
    violations = r_sums - dists
    
    # We only care about upper triangle to avoid double counting and self-interaction
    # mask for upper triangle (excluding diagonal)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    
    # Get valid violations
    valid_violations = violations[mask]
    
    # Penalty for overlaps
    overlap_pen = np.maximum(0, valid_violations)
    penalty += np.sum(overlap_pen**2)
    
    total_cost = -sum_radii + weight * penalty
    return total_cost

            dy = 1.0 / (rows + 1)

    for i in range(N):

                # Shrink the larger circle or both
                if radii[i] >= radii[j]:

                else:
                    radii[j] -= overlap / 2.0

    # Recalculate sum
    final_sum = np.sum(radii)
    
    return centers, radii, final_sum
