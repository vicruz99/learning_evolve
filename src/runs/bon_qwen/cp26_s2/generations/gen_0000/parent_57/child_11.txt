import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    def optimize_from(centers_init: np.ndarray, radii_init: np.ndarray) -> tuple[np.ndarray, np.ndarray, float]:
        x0 = np.concatenate([centers_init.flatten(), radii_init])

        def objective(x: np.ndarray) -> float:
            return -np.sum(x[2 * n:])

        constraints = []

        for i in range(n):
            # r >= epsilon
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[2 * n + i] - 1e-8
            })
            # x >= r
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[2 * i] - x[2 * n + i]
            })
            # x + r <= 1
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: 1.0 - x[2 * i] - x[2 * n + i]
            })
            # y >= r
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[2 * i + 1] - x[2 * n + i]
            })
            # y + r <= 1
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: 1.0 - x[2 * i + 1] - x[2 * n + i]
            })

        for i in range(n):
            for j in range(i + 1, n):
                def overlap_con(x: np.ndarray, i: int = i, j: int = j) -> float:
                    xi = x[2 * i]
                    yi = x[2 * i + 1]
                    xj = x[2 * j]
                    yj = x[2 * j + 1]
                    ri = x[2 * n + i]
                    rj = x[2 * n + j]
                    dist_sq = (xi - xj) ** 2 + (yi - yj) ** 2
                    return dist_sq - (ri + rj) ** 2
                constraints.append({
                    'type': 'ineq',
                    'fun': overlap_con
                })

        result = minimize(
            objective,
            x0,
            method='SLSQP',
            constraints=constraints,
            options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False}
        )

        final_centers = result.x[:2 * n].reshape(n, 2)
        final_radii = result.x[2 * n:]

        # Ensure non-negative radii
        final_radii = np.maximum(final_radii, 1e-10)

        # Ensure centers are within valid bounds
        for i in range(n):
            r = final_radii[i]
            final_centers[i, 0] = max(r, min(1 - r, final_centers[i, 0]))
            final_centers[i, 1] = max(r, min(1 - r, final_centers[i, 1]))

        current_sum = np.sum(final_radii)
        return final_centers, final_radii, current_sum

    def make_hexagonal_init(offset: float = 0.0) -> tuple[np.ndarray, np.ndarray]:
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.09 + offset)

        idx = 0
        for row in range(7):
            for col in range(5):
                if idx >= n:
                    break
                x = 0.09 + col * 0.175
                if row % 2 == 1:
                    x += 0.0875
                y = 0.09 + row * 0.145
                centers[idx] = [x, y]
                idx += 1
            if idx >= n:
                break

        return centers, radii

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Try multiple initial configurations with different perturbations
    for trial in range(30):
        np.random.seed(trial * 7 + 3)

        centers, radii = make_hexagonal_init(offset=0.0)

        # Add random perturbation
        centers += 0.015 * np.random.randn(n, 2)
        centers = np.clip(centers, 0.05, 0.95)

        # Slightly vary initial radii
        radii = radii * (1 + 0.1 * np.random.randn(n))
        radii = np.clip(radii, 0.05, 0.15)

        try:
            final_centers, final_radii, current_sum = optimize_from(centers, radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = final_centers.copy()
                best_radii = final_radii.copy()
        except Exception:
            continue

    # If no valid solution found, use a fallback
    if best_centers is None:
        centers, radii = make_hexagonal_init()
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    # Post-processing: ensure strict validity with a tiny margin
    # Shrink radii slightly if there are any near-violations
    for i in range(n):
        r = best_radii[i]
        x, y = best_centers[i]
        # Ensure boundary constraints
        r = min(r, x, 1 - x, y, 1 - y)

    for i in range(n):
        for j in range(i + 1, n):
            dx = best_centers[j, 0] - best_centers[i, 0]
            dy = best_centers[j, 1] - best_centers[i, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = best_radii[i] + best_radii[j]
            if dist < min_dist:
                # Reduce radii to fix overlap
                reduction = (min_dist - dist + 1e-10) / 2
                best_radii[i] = max(best_radii[i] - reduction, 1e-10)
                best_radii[j] = max(best_radii[j] - reduction, 1e-10)

    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
