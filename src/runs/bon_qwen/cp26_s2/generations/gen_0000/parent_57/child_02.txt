import numpy as np
from scipy.optimize import minimize
import math

def objective(vars, n):
    radii = vars[2*n:]
    return -np.sum(radii)

def penalty(vars, n, lam):
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    p = 0.0
    
    # Boundary penalties
    p += lam * np.sum(np.maximum(0, radii - centers[:,0])**2)
    p += lam * np.sum(np.maximum(0, radii - (1 - centers[:,0]))**2)
    p += lam * np.sum(np.maximum(0, radii - centers[:,1])**2)
    p += lam * np.sum(np.maximum(0, radii - (1 - centers[:,1]))**2)
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            dist = math.sqrt(dx*dx + dy*dy)
            ov = radii[i] + radii[j] - dist
            if ov > 0:
                p += lam * ov * ov
                
    return objective(vars, n) + p

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    for trial in range(20):
        np.random.seed(trial)
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        
        # Staggered hexagonal initialization
        k = 0
        rows = 6
        cols = 5
        for r in range(rows):
            for c in range(cols):
                if k >= n: break
                x = (c + 0.5) / cols
                y = (r + 0.5) / rows
                if r % 2 == 1:
                    x = (c + 1.0) / cols
                # Add controlled jitter to break symmetry
                centers[k] = [x + np.random.normal(0, 0.008), y + np.random.normal(0, 0.008)]
                radii[k] = 0.05 + np.random.uniform(-0.005, 0.005)
                k += 1
        while k < n:
            centers[k] = np.random.uniform(0.15, 0.85, 2)
            radii[k] = 0.03
            k += 1
            
        centers = np.clip(centers, 0.05, 0.95)
        radii = np.clip(radii, 0.02, 0.3)
        
        vars0 = np.concatenate([centers.flatten(), radii])
        current_vars = vars0.copy()
        
        # Continuation on penalty parameter
        lam = 50.0
        for step in range(12):
            res = minimize(penalty, current_vars, args=(n, lam), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-13, 'disp': False})
            current_vars = res.x
            lam *= 12.0
            
        c_opt = current_vars[:2*n].reshape(n, 2)
        r_opt = current_vars[2*n:]
        s = np.sum(r_opt)
        
        if s > best_sum:
            best_sum = s
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Final safety shrink to guarantee strict validity within tolerance
    best_radii = np.clip(best_radii, 1e-9, None)
    
    return best_centers, best_radii, float(best_sum)
