from scipy.optimize import minimize

def compute_penalty(pos, r):
    """
    Computes the squared penalty for boundary violations and circle overlaps.
    pos: numpy array of shape (n, 2)
    r: radius of circles
    """

    x, y = pos[:, 0], pos[:, 1]
    p = 0.0

    p += np.sum(np.maximum(0.0, r - x)**2)

    p += np.sum(np.maximum(0.0, r - y)**2)
    p += np.sum(np.maximum(0.0, y - (1.0 - r))**2)

    np.fill_diagonal(dists, np.inf)  # Ignore self-distances
    p += np.sum(np.maximum(0.0, 2.0*r - dists)**2)
    
    return p
