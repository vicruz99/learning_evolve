import numpy as np
from scipy.optimize import minimize

def objective_func(z):
    """Objective: minimize negative sum of radii."""
    return -np.sum(z[2::3])

def constraint_boundary(z):
    """Constraints: circles must stay inside [0,1]x[0,1]."""
    n = len(z) // 3
    vals = np.zeros(4 * n)
    for i in range(n):
        x, y, r = z[3*i], z[3*i+1], z[3*i+2]
        vals[4*i] = x - r
        vals[4*i+1] = 1.0 - x - r
        vals[4*i+2] = y - r
        vals[4*i+3] = 1.0 - y - r
    return vals

def constraint_pairs(z):
    """Constraints: circles must not overlap."""
    n = len(z) // 3
    pts = z.reshape(-1, 3)
    diff = pts[:, np.newaxis, :2] - pts[np.newaxis, :, :2]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    radii_sum = pts[:, np.newaxis, 2] + pts[np.newaxis, :, 2]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return dists[mask] - radii_sum[mask]

def run_packing():
    n = 26
    # Initial radius smaller than optimal to ensure strict feasibility
    r_init = 0.09
    centers = []
    idx = 0
    dy = np.sqrt(3) * r_init
    dx = 2 * r_init
    
    # Generate hexagonal lattice positions
    for row in range(6):
        y = r_init + row * dy
        if y + r_init > 1.0:
            continue
        offset = r_init if row % 2 == 1 else 0
        x = offset + r_init
        while x + r_init <= 1.0 and idx < n:
            centers.append([x, y])
            x += dx
            idx += 1
            
    centers = np.array(centers[:n])
    radii = np.ones(n) * r_init
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: coordinates in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
    
    # Define constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_pairs}
    ]
    
    # Run optimization
    res = minimize(
        objective_func, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 1000, 'ftol': 1e-10}
    )
    
    # Extract results
    centers_opt = res.x[:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]
    radii_opt = np.maximum(radii_opt, 0.0)  # Ensure non-negativity
    
    return centers_opt, radii_opt, np.sum(radii_opt)
