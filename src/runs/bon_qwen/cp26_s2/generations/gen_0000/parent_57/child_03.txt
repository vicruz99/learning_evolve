import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(v[2::3])

def boundary_constraints(v):
    """Constraints: circles must stay within [0,1]x[0,1]"""
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    c = np.empty(4 * N)
    c[0::4] = xs - rs          # x - r >= 0
    c[1::4] = 1 - xs - rs      # 1 - x - r >= 0
    c[2::4] = ys - rs          # y - r >= 0
    c[3::4] = 1 - ys - rs      # 1 - y - r >= 0
    return c

def pairwise_constraints(v):
    """Constraints: circles must not overlap"""
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Vectorized pairwise distances and radius sums
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dist_sq = dx**2 + dy**2
    sum_r_sq = dr**2
    
    # Extract upper triangle (i < j)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    return (dist_sq - sum_r_sq)[mask].ravel()

def force_relax(centers, radii, steps=3000):
    """Physics-based pre-optimization to find a dense feasible packing"""
    for _ in range(steps):
        radii += 1e-5
        forces = np.zeros_like(centers)
        
        # Boundary repulsion
        left = radii - centers[:, 0]
        forces[left > 0, 0] += left[left > 0]
        right = radii - (1 - centers[:, 0])
        forces[right > 0, 0] -= right[right > 0]
        bottom = radii - centers[:, 1]
        forces[bottom > 0, 1] += bottom[bottom > 0]
        top = radii - (1 - centers[:, 1])
        forces[top > 0, 1] -= top[top > 0]
        
        # Pairwise repulsion
        for i in range(N):
            for j in range(i+1, N):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < radii[i] + radii[j]:
                    overlap = radii[i] + radii[j] - dist
                    if dist > 1e-9:
                        f = diff / dist * overlap
                        forces[i] += f
                        forces[j] -= f
                        
        centers += 0.1 * forces
        centers = np.clip(centers, 0, 1)
    return centers, radii

def run_packing():
    # 1. Initialize with hexagonal lattice
    centers = np.zeros((N, 2))
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx >= N: break
            x = 0.12 + j * 0.19 + (i % 2) * 0.095
            y = 0.12 + i * 0.19 * np.sqrt(3)/2
            centers[idx] = [x, y]
            idx += 1
        if idx >= N: break
        
    radii = np.full(N, 0.04)
    
    # 2. Force relaxation to get close to optimal density
    centers, radii = force_relax(centers, radii, steps=3000)
    
    # Shrink slightly to ensure strict feasibility before optimization
    radii *= 0.98
    
    # 3. Prepare for SLSQP optimization
    v0 = np.zeros(3 * N)
    v0[0::3] = centers[:, 0]
    v0[1::3] = centers[:, 1]
    v0[2::3] = radii
    
    bounds = [(0, 1), (0, 1), (0, 0.5)] * N
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': pairwise_constraints}
    ]
    
    # Run constrained optimization
    res = minimize(objective, v0, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': 1500, 'ftol': 1e-10})
    
    # 4. Extract and validate results
    v_opt = res.x
    centers_opt = np.column_stack((v_opt[0::3], v_opt[1::3]))
    radii_opt = np.maximum(v_opt[2::3], 1e-9) # Ensure non-negative
    
    # Final sanity clamp to [0,1]
    centers_opt = np.clip(centers_opt, 0, 1)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
