import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialize centers on a perturbed grid to distribute circles evenly
    rng = np.random.default_rng(42)
    cols = int(np.ceil(np.sqrt(n * 1.2))) # Use slightly wider grid to allow for perturbation
    rows = int(np.ceil(n / cols))
    
    grid_x = np.linspace(0.05, 0.95, cols)
    grid_y = np.linspace(0.05, 0.95, rows)
    
    centers_init = []
    for i in range(n):
        c = i % cols
        r = i // cols
        # Add small perturbation
        cx = grid_x[c] + rng.uniform(-0.01, 0.01)
        cy = grid_y[r] + rng.uniform(-0.01, 0.01)
        centers_init.append([cx, cy])
        
    centers_init = np.array(centers_init)
    
    # 2. Setup initial variables: [x0, y0, r0, x1, y1, r1, ...]
    # Initial radii should be small to satisfy constraints easily
    r_init = np.ones(n) * 0.01
    x_init = centers_init[:, 0]
    y_init = centers_init[:, 1]
    
    x0 = np.concatenate([x_init, y_init, r_init])
    
    # 3. Define objective: Maximize sum of radii => Minimize negative sum
    def objective(vars):
        r = vars[2 * n:] # Radii are the last n variables
        return -np.sum(r)
    
    # 4. Define constraints
    constraints = []
    
    # Boundary constraints: 0 <= x-r, x+r <= 1, 0 <= y-r, y+r <= 1
    # Simplified to: r <= x <= 1-r and r <= y <= 1-r
    # We implement these as r - x <= 0, x - (1-r) <= 0, etc.
    
    for i in range(n):
        idx_x = i
        idx_y = n + i
        idx_r = 2 * n + i
        
        # x - r >= 0  => r - x <= 0
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[idx_x] - vars[idx_r]})
        
        # x + r <= 1  => x + r - 1 <= 0
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: 1 - vars[idx_x] - vars[idx_r]})
        
        # y - r >= 0  => r - y <= 0
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[idx_y] - vars[idx_r]})
        
        # y + r <= 1  => y + r - 1 <= 0
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: 1 - vars[idx_y] - vars[idx_r]})
        
        # r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[idx_r]})

    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = i, n + i, 2 * n + i
            idx_xj, idx_yj, idx_rj = j, n + j, 2 * n + j
            
            def non_overlap(vars, i=i, j=j):
                dx = vars[idx_xi] - vars[idx_xj]
                dy = vars[idx_yi] - vars[idx_yj]
                dr = vars[idx_ri] + vars[idx_rj]
                return dx*dx + dy*dy - dr*dr
                
            constraints.append({'type': 'ineq', 'fun': non_overlap})

    # 5. Optimization
    # Using SLSQP which handles bounds and constraints
    result = minimize(objective, x0, method='SLSQP', constraints=constraints, 
                      options={'maxiter': 1000, 'ftol': 1e-12})
    
    # 6. Extract results
    if result.success:
        final_x = result.x[:n]
        final_y = result.x[n:2*n]
        final_r = result.x[2*n:]
        
        # Ensure non-negative radii (solver might drift slightly negative due to float precision)
        final_r = np.maximum(final_r, 0.0)
        
        final_centers = np.column_stack((final_x, final_y))
        sum_radii = np.sum(final_r)
        
        return final_centers, final_r, sum_radii
    else:
        # Fallback to a simple grid if optimization fails
        # 5x5 grid + 1 circle in center is not valid, so 5x5 grid with smaller radii
        centers = np.linspace(0.1, 0.9, 5)
        xs, ys = np.meshgrid(centers, centers)
