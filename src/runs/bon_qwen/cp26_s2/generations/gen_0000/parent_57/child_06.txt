def compute_gradient(x):
    """Compute gradient of the objective function (-sum of radii)."""
    n = len(x) // 3
    grad = np.zeros(n * 3)
    # Gradient of -sum(r_i) is -1 for each r component
    # x is [x0, y0, r0, x1, y1, r1, ...]
    # r components are at indices 2, 5, 8, ...
    for i in range(n):
        grad[i * 3 + 2] = -1.0
    return grad

def compute_constraints(x):

        c.append(xi - ri)
        c.append(1.0 - xi - ri)
        c.append(yi - ri)
        c.append(1.0 - yi - ri)
    
    # Non-overlap constraints: dist^2 - (ri + rj)^2 >= 0
    for i in range(n):

    return np.array(c)
