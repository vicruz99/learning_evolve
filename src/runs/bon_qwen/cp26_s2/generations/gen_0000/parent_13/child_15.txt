import numpy as np
from scipy.optimize import minimize

def run_packing():
    n_circles = 26
    
    # 1. Generate Initial Configuration (Hexagonal Packing)
    # We create a grid of points based on hexagonal lattice
    # Basis vectors: (1, 0) and (0.5, sqrt(3)/2)
    
    # Generate a large set of candidate points
    # We want to cover the [0,1]x[0,1] square
    # Let's try a grid with spacing roughly 0.15
    spacing = 0.15
    rows = int(1.0 / spacing) + 2
    cols = int(1.0 / spacing) + 2
    
    points = []
    for r in range(rows):
        for c in range(cols):
            # Hexagonal offset for odd rows
            x = c * spacing
            y = r * spacing * np.sqrt(3) / 2
            if r % 2 == 1:
                x += spacing / 2
            
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
    
    # Select 26 points that are well distributed
    # Simply taking the first 26 from the list might be clustered, 
    # but since we optimize later, it's a decent start.
    # To be safer, let's shuffle and take 26, or just take a subset.
    # Actually, just taking the first 26 valid points from the dense list is fine.
    # But let's ensure they are somewhat spread.
    # We can just filter points that are not too close to each other?
    # Or just use the list.
    
    if len(points) < n_circles:
        # Fallback to random if grid is too sparse (unlikely with spacing 0.15)
        rng = np.random.default_rng(42)
        centers_init = rng.uniform(0.05, 0.95, (n_circles, 2))
    else:
        centers_init = np.array(points[:n_circles])
    
    # Initial radii: small enough to be valid
    radii_init = np.full(n_circles, 0.04)
    
    # Combine into initial vector: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]

    # 2. Define Bounds
    # x, y in [0, 1], r in [0, 1]
    # Tighter bounds for r can help, but [0, 1] is safe.
    bounds = []
    for i in range(n_circles):
        bounds.extend([(0, 1), (0, 1), (0, 1)])

    # 3. Define Constraints
    # Inequality constraints: g(x) >= 0
    
    def boundary_constraints(vars):
        c = []
        for i in range(n_circles):
            xi = vars[3*i]
            yi = vars[3*i+1]
            ri = vars[3*i+2]
            # x - r >= 0
            c.append(xi - ri)
            # 1 - x - r >= 0
            c.append(1.0 - xi - ri)
            # y - r >= 0
            c.append(yi - ri)
            # 1 - y - r >= 0
            c.append(1.0 - yi - ri)
        return c

    def overlap_constraints(vars):
        c = []
        # We need to check all pairs
        # To speed up, we can vectorize, but loop is clear
        for i in range(n_circles):
            xi = vars[3*i]
            yi = vars[3*i+1]
            ri = vars[3*i+2]
            
            # Precompute square of radius for this circle? 
            # Actually we need (r_i + r_j)^2
            # Let's just compute directly
            for j in range(i + 1, n_circles):
                xj = vars[3*j]
                yj = vars[3*j+1]
                rj = vars[3*j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx*dx + dy*dy
                r_sum = ri + rj
                # Constraint: dist_sq >= (r_i + r_j)^2
                # i.e., dist_sq - (r_sum)^2 >= 0
                c.append(dist_sq - r_sum*r_sum)
        return c

    # Prepare constraints for scipy
    # SLSQP expects constraints as dictionaries or a list of dicts
    # type 'ineq' means g(x) >= 0
    
    # Boundary constraints (4 per circle)
    # We can combine them into a single function returning an array
    # scipy handles array returned constraints element-wise
    
    cons_boundary = {
        'type': 'ineq',
        'fun': boundary_constraints
    }
    
    cons_overlap = {
        'type': 'ineq',
        'fun': overlap_constraints
    }
    
    constraints = [cons_boundary, cons_overlap]

    # 4. Objective Function
    def objective(vars):
        # Minimize negative sum of radii
        # Radii are at indices 2, 5, 8, ...
        radii = vars[2::3]
        return -np.sum(radii)

    # 5. Run Optimization with multiple restarts
    best_result = None
    best_val = -np.inf
    
    # Try a few different initial seeds or perturbations
    rng = np.random.default_rng(123)
    
    for restart in range(10):
        # Perturb initial positions slightly
        current_x0 = x0.copy()
        if restart > 0:
            # Add some noise
            noise = rng.normal(0, 0.01, size=current_x0.shape)
            current_x0 += noise
            
            # Clip to bounds roughly
            # x, y should stay in [0, 1]
            for k in range(0, len(current_x0), 3):
                current_x0[k] = np.clip(current_x0[k], 0, 1)
                current_x0[k+1] = np.clip(current_x0[k+1], 0, 1)
            
            # Reduce radii slightly to ensure validity after move
            for k in range(2, len(current_x0), 3):
                current_x0[k] = max(0.01, current_x0[k] * 0.9)
        
        try:
            res = minimize(objective, current_x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 500, 'ftol': 1e-8})
            
            if res.success or res.fun < best_val: # Note: best_val tracks min of negative sum (so max sum)
                # Check if result is actually better (higher sum of radii)
                current_sum = -res.fun
                if current_sum > best_val:
                    best_val = current_sum
                    best_result = res
        except Exception as e:
            print(f"Optimization failed: {e}")
            continue

    # 6. Extract and return best solution
    if best_result is not None:
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        for i in range(n_circles):
            centers[i, 0] = best_result.x[3*i]
            centers[i, 1] = best_result.x[3*i+1]
            radii[i] = best_result.x[3*i+2]
        
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        # Fallback if optimization fails completely
        # Return initial valid config
        centers = centers_init
        radii = radii_init
        return centers, radii, np.sum(radii)
