import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

def initialize_hexagonal(n, seed):
    rng = np.random.default_rng(seed)

    # 5 rows with 6, 5, 6, 5, 4 circles
    rows = [6, 5, 6, 5, 4]

    centers = []
    radii = []

    r = 0.075

    for row_idx, num_circles in enumerate(rows):
        y = r + row_idx * np.sqrt(3) * r

        if row_idx % 2 == 0:
            x_start = r
        else:
            x_start = 2 * r

        for col_idx in range(num_circles):
            x = x_start + col_idx * 2 * r
            x += rng.uniform(-0.01, 0.01)
            y += rng.uniform(-0.01, 0.01)

            centers.append([x, y])
            radii.append(r)

    return np.array(centers), np.array(radii)


def non_overlap_constraint(x, params):
    i, j, n = params
    ci = np.array([x[2 * i], x[2 * i + 1]])
    cj = np.array([x[2 * j], x[2 * j + 1]])
    ri = x[2 * n + i]
    rj = x[2 * n + j]
    dist = np.sqrt(np.sum((ci - cj) ** 2))
    return dist - ri - rj


def boundary_left_constraint(x, params):
    i, n = params
    return x[2 * i] - x[2 * n + i]


def boundary_right_constraint(x, params):
    i, n = params
    return 1.0 - x[2 * i] - x[2 * n + i]


def boundary_bottom_constraint(x, params):
    i, n = params
    return x[2 * i + 1] - x[2 * n + i]


def boundary_top_constraint(x, params):
    i, n = params
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def optimize_packing(centers, radii):
    n = len(radii)

    x0 = np.concatenate([centers.flatten(), radii])

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    def objective(x):
        return -np.sum(x[2 * n:])

    constraints = []

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': non_overlap_constraint,
                'args': ((i, j, n),)
            })

    # Boundary constraints
    for i in range(n):
        constraints.append({
            'type': 'ineq',
            'fun': boundary_left_constraint,
            'args': ((i, n),)
        })
        constraints.append({
            'type': 'ineq',
            'fun': boundary_right_constraint,
            'args': ((i, n),)
        })
        constraints.append({
            'type': 'ineq',
            'fun': boundary_bottom_constraint,
            'args': ((i, n),)
        })
        constraints.append({
            'type': 'ineq',
            'fun': boundary_top_constraint,
            'args': ((i, n),)
        })

    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': 5000,
            'ftol': 1e-15,
            'disp': False
        }
    )

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]

    radii_opt = np.maximum(radii_opt, 0)

    return centers_opt, radii_opt


def ensure_valid(centers, radii, n):
    # Clip radii to be non-negative
    radii = np.maximum(radii, 0)

    # Iteratively fix any constraint violations
    for iteration in range(200):
        max_violation = 0

        # Check overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                overlap = radii[i] + radii[j] - dist
                if overlap > max_violation:
                    max_violation = overlap

        # Check boundaries
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            if x - r < 0:
                max_violation = max(max_violation, r - x)
            if x + r > 1:
                max_violation = max(max_violation, x + r - 1)
            if y - r < 0:
                max_violation = max(max_violation, r - y)
            if y + r > 1:
                max_violation = max(max_violation, y + r - 1)

        if max_violation < 1e-12:
            break

        # Reduce all radii slightly to resolve violations
        scale = 0.999
        radii *= scale

    # Final boundary clipping
    for i in range(n):
        r = radii[i]
        centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1 - r, centers[i, 1]))

    return centers, radii
