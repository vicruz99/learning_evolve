import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_constraints(vars):
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    n_constraints = n * 4 + n * (n - 1) // 2
    c = np.empty(n_constraints)
    idx = 0
    
    # Boundary constraints: x - r >= 0
    c[idx:idx+n] = x - r
    idx += n
    # 1 - x - r >= 0
    c[idx:idx+n] = 1.0 - x - r
    idx += n
    # y - r >= 0
    c[idx:idx+n] = y - r
    idx += n
    # 1 - y - r >= 0
    c[idx:idx+n] = 1.0 - y - r
    idx += n
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    dist_sq = dx**2 + dy**2
    sum_r_sq = dr**2
    
    i, j = np.triu_indices(n, k=1)
    c[idx:] = dist_sq[i, j] - sum_r_sq[i, j]
    
    return c

def objective(vars):
    r = vars[2::3]
    return -np.sum(r)

def constraint_fun(vars):
    return compute_constraints(vars)

def get_initial_config(method):
    n = N_CIRCLES
    x, y, r = [], [], []
    
    if method == 0:
        # Dense grid initialization
        cols = 6
        rows = 5
        spacing_x = 1.0 / cols
        spacing_y = 1.0 / rows
        count = 0
        for j in range(rows):
            for i in range(cols):
                if count < n:
                    x.append((i + 0.5) * spacing_x)
                    y.append((j + 0.5) * spacing_y)
                    r.append(0.08)
                    count += 1
                    
    elif method == 1:
        # Hexagonal-like staggered initialization
        row = 0
        count = 0
        while count < n:
            y_c = (row + 0.5) * (1.0 / 5.0)
            is_odd = row % 2 == 1
            ncols = 5 if not is_odd else 4
            spacing_x = 1.0 / 5.0
            for i in range(ncols):
                if count < n:
                    x_c = (i + 0.5) * spacing_x if is_odd else i * spacing_x + 0.1
                    x.append(x_c)
                    y.append(y_c)
                    r.append(0.09)
                    count += 1
            row += 1
            
    else:
        # Random feasible initialization
        for _ in range(n):
            placed = False
            for _ in range(100):
                cx = np.random.uniform(0.15, 0.85)
                cy = np.random.uniform(0.15, 0.85)
                cr = np.random.uniform(0.06, 0.11)
                
                if cx - cr < 0 or cx + cr > 1 or cy - cr < 0 or cy + cr > 1:
                    continue
                    
                overlap = False
                for k in range(len(x)):
                    if (cx - x[k])**2 + (cy - y[k])**2 < (cr + r[k])**2:
                        overlap = True
                        break
                        
                if not overlap:
                    x.append(cx)
                    y.append(cy)
                    r.append(cr)
                    placed = True
                    break
                    
            if not placed:
                x.append(0.5)
                y.append(0.5)
                r.append(0.05)
                
    return np.array(x + y + r)

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(1e-5, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    # Run optimization from multiple initial configurations
    for method in range(3):
        for trial in range(4):
            x0 = get_initial_config(method)
            # Add small perturbation to escape symmetric local optima
            x0 = x0 + np.random.normal(0, 0.003, size=x0.shape)
            
            # Project to bounds
            x0[:2*n] = np.clip(x0[:2*n], 0.0, 1.0)
            x0[2*n:] = np.clip(x0[2*n:], 1e-5, 0.5)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
                
                if res.success:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
            except Exception:
                continue
                
    # Fallback in case optimization fails
    if best_centers is None:
        x0 = get_initial_config(0)
        best_centers = np.column_stack((x0[0::3], x0[1::3]))
        best_radii = x0[2::3]
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
