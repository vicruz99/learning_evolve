import numpy as np
from scipy.optimize import minimize


def compute_loss(params, n, penalty):
    """Compute the objective: negative sum of radii plus penalty for constraint violations."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    loss = -np.sum(radii)

    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        loss += penalty * max(0.0, r - x) ** 2
        loss += penalty * max(0.0, r - (1.0 - x)) ** 2
        loss += penalty * max(0.0, r - y) ** 2
        loss += penalty * max(0.0, r - (1.0 - y)) ** 2

    # Non-overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            min_dist = radii[i] + radii[j]
            if dist_sq < min_dist * min_dist:
                dist = np.sqrt(dist_sq)
                overlap = min_dist - dist
                loss += penalty * overlap * overlap

    return loss


class ObjectiveFunction:
    """Callable wrapper for the loss function without closures."""
    def __init__(self, n, penalty):
        self.n = n
        self.penalty = penalty

    def __call__(self, params):
        return compute_loss(params, self.n, self.penalty)


def make_initial_arrangement(n):
    """Create initial hexagonal lattice arrangement for n circles."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07

    # Hexagonal pattern with 6 rows: 5, 4, 5, 4, 5, 3 = 26
    row_counts = [5, 4, 5, 4, 5, 3]
    spacing_x = 0.22
    spacing_y = spacing_x * np.sqrt(3) / 2

    idx = 0
    for row_i, count in enumerate(row_counts):
        y = 0.08 + row_i * spacing_y
        if row_i % 2 == 0:
            x_start = 0.08
        else:
            x_start = 0.08 + spacing_x / 2.0

        for col_i in range(count):
            if idx < n:
                x = x_start + col_i * spacing_x
                centers[idx] = [x, y]
                idx += 1

    return centers, radii


def ensure_validity(centers, radii, n):
    """Post-process to ensure all constraints are strictly satisfied."""
    # Ensure boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y) - 1e-10
        radii[i] = min(r, max_r)

    # Fix overlaps iteratively
    for _ in range(200):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                required = radii[i] + radii[j] + 1e-10
                if dist < required:
                    scale = dist / required
                    radii[i] *= np.sqrt(scale)
                    radii[j] *= np.sqrt(scale)
                    overlap_found = True
        if not overlap_found:
            break

    return centers, radii


def run_packing():
    """Pack 26 circles in a unit square to maximize sum of radii."""
    n = 26

    centers, radii = make_initial_arrangement(n)
    x0 = np.concatenate([centers.flatten(), radii])

    # Multi-stage optimization with increasing penalties
    current_x = x0.copy()
    penalties = [50, 200, 1000, 5000, 20000, 100000, 500000]

    for penalty in penalties:
        bounds = [(0.001, 0.999) for _ in range(2 * n)]
        bounds += [(0.001, 0.5) for _ in range(n)]

        obj = ObjectiveFunction(n, penalty)

        result = minimize(
            obj,
            current_x,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 3000, "ftol": 1e-15, "gtol": 1e-10, "iprint": -1}
        )

        current_x = result.x

    centers_opt = current_x[:2 * n].reshape(n, 2)
    radii_opt = current_x[2 * n:]

    # Enforce hard constraints
    centers_opt, radii_opt = ensure_validity(centers_opt, radii_opt, n)

    total = np.sum(radii_opt)

    return centers_opt, radii_opt, total
