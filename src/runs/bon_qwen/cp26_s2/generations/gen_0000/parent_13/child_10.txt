import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    def penalty_function(centers, radii, penalty_weight=1000.0):
        """
        Calculates a penalty score based on constraint violations.
        Returns a score where 0 is valid, positive is violation.
        """
        score = 0.0
        
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Distance to boundaries
            dist_left = x - r
            dist_right = 1.0 - x - r
            dist_bottom = y - r
            dist_top = 1.0 - y - r
            
            violations = [
                max(0, -dist_left),
                max(0, -dist_right),
                max(0, -dist_bottom),
                max(0, -dist_top)
            ]
            for v in violations:
                score += penalty_weight * (v ** 2)
                
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                req_dist = radii[i] + radii[j]
                
                if dist < req_dist:
                    overlap = req_dist - dist
                    score += penalty_weight * (overlap ** 2)
                    
        return score

    def objective(vars_arr, penalty_weight=1000.0):
        """
        Objective function to minimize.
        We want to maximize sum(radii), so we minimize -sum(radii).
        We add a penalty for constraint violations.
        """
        centers = vars_arr[:n*2].reshape(n, 2)
        radii = vars_arr[n*2:]
        
        # Penalty term
        p = penalty_function(centers, radii, penalty_weight)
        
        # Objective: negative sum of radii
        obj = -np.sum(radii) + p
        
        return obj

                'type': 'ineq',

            })

            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[i*2 + 1] - v[n*2 + i]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[i*2 + 1] - v[n*2 + i]
            })
            
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: \
                        (v[i*2+0] - v[j*2+0])**2 + (v[i*2+1] - v[j*2+1])**2 - (v[n*2+i] + v[n*2+j])**2
                })
                
        return constraints

    def run_optimization(initial_vars, method='SLSQP'):
        """
        Runs the optimization process.
        """

        cons = get_constraints(initial_vars)

            objective,
            initial_vars,
            method=method,
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9}
        )
        return result

        return np.array(centers[:n])
