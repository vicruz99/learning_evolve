import numpy as np
from scipy.optimize import linprog

def solve_lp_for_radii(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= dist(i, j) for all i < j
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Inequality constraints A_ub @ r <= b_ub
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    # We can build these efficiently
    # There are n*(n-1)/2 pairs
    num_pairs = n * (n - 1) // 2
    
    # Precompute distances
    # centers is (n, 2)
    # diff = centers[:, None, :] - centers[None, :, :] # (n, n, 2)
    # dists = np.linalg.norm(diff, axis=2)
    # To save memory/time, we can iterate or use broadcasting carefully
    # But for n=26, n^2 is 676, small enough.
    
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    
    # Build sparse matrix or list of tuples for linprog? 
    # linprog accepts dense or sparse. Dense is fine for 26 vars.
    # We need to populate rows for r_i + r_j <= d_ij
    
    # To avoid huge matrix construction in loop, let's build indices
    rows = []
    cols = []
    vals = []
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            # Row index in A_ub
            row_idx = idx
            idx += 1
            # r_i coeff 1, r_j coeff 1
            rows.append(row_idx)
            cols.append(i)
            vals.append(1.0)
            
            rows.append(row_idx)
            cols.append(j)
            vals.append(1.0)
            
            b_ub.append(dists[i, j])
    
    # Boundary constraints
    # r_i <= x_i  => 1*r_i <= x_i
    # r_i <= 1-x_i => 1*r_i <= 1-x_i
    # r_i <= y_i
    # r_i <= 1-y_i
    # These are simple upper bounds, can be handled by bounds or A_ub.
    # linprog supports bounds directly: (lower, upper)
    
    # We'll use bounds for simplicity
    bounds = [(0, None)] * n
    
    # Construct A_ub matrix
    # A_ub has shape (num_pairs, n)
    A_ub_matrix = np.zeros((num_pairs, n))
    for r, c, v in zip(rows, cols, vals):
        A_ub_matrix[r, c] = v
        
    b_ub = np.array(b_ub)
    
    # Solve
    # method='highs' is usually good
    res = linprog(c_obj, A_ub=A_ub_matrix, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        # Fallback or error
        return 0.0, np.zeros(n)
