def compute_pairwise_distances(centers):
    """Compute pairwise distances between circle centers."""
    n = centers.shape[0]
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)
    return dists


def objective_function(x, n):
    """Negative sum of radii (to be minimized)."""
    radii = x[2 * n:]
    return -np.sum(radii)


def constraint_function(x, n):
    """All constraints as a single array (must be >= 0)."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    constraints = []
    
    # Boundary constraints: 0 <= x - r and x + r <= 1, same for y
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i])
        constraints.append(1.0 - centers[i, 0] - radii[i])
        constraints.append(centers[i, 1] - radii[i])
        constraints.append(1.0 - centers[i, 1] - radii[i])
    
    # Non-overlap constraints using squared distances
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            min_dist_sq = (radii[i] + radii[j]) ** 2
            constraints.append(dist_sq - min_dist_sq)
    
    return np.array(constraints)


def initialize_hexagonal(n, r):
    """Create a hexagonal arrangement of n circles with radius r."""
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    
    # Determine row configuration for 26 circles
    # Use pattern that fits well in unit square
    # 5 rows: 5, 5, 5, 5, 6 = 26 (but offset rows have fewer)
    # Better: 4, 5, 4, 5, 4, 4 = 26
    
    row_counts = [5, 4, 5, 4, 5, 3]  # Sum = 26
    idx = 0
    
    sqrt3 = np.sqrt(3)
    vertical_spacing = r * sqrt3
    
    for row in range(6):
        y = r + row * vertical_spacing
        count = row_counts[row] if row < len(row_counts) else 5
        
        # Even rows start at x = r, odd rows start at x = 2r (offset)
        if row % 2 == 0:
            x_start = r
        else:
            x_start = 2 * r
        
        for col in range(count):
            if idx < n:
                x = x_start + col * 2 * r
                centers[idx] = [x, y]
                idx += 1
    
    return centers, radii


def force_optimize(centers, radii, n, num_iterations=15000):
    """Force-based optimization to pack circles tightly."""
    centers = centers.copy()
    radii = radii.copy()
    
    dt = 0.003
    spring_k = 500.0
    boundary_k = 2000.0
    damping = 0.85
    
    velocities = np.zeros((n, 2))
    
    for iteration in range(num_iterations):
        # Gradually increase target radius
        progress = iteration / num_iterations
        target_r = radii[0] + 0.05 * progress  # Grow from initial to ~0.05 larger
        
        # Compute repulsion forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                
                if dist < 1e-12:
                    # Avoid division by zero, push apart randomly
                    dx = np.random.uniform(-0.01, 0.01)
                    dy = np.random.uniform(-0.01, 0.01)
                    dist = np.sqrt(dx * dx + dy * dy)
                
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = overlap * spring_k
                    fx = (dx / dist) * force_mag
                    fy = (dy / dist) * force_mag
                    velocities[i] -= [fx, fy] * dt
                    velocities[j] += [fx, fy] * dt
        
        # Boundary repulsion forces
        for i in range(n):
            for d in range(2):
                if centers[i, d] - radii[i] < 0:
                    penetration = radii[i] - centers[i, d]
                    velocities[i, d] += penetration * boundary_k * dt
                if centers[i, d] + radii[i] > 1.0:
                    penetration = centers[i, d] + radii[i] - 1.0
                    velocities[i, d] -= penetration * boundary_k * dt
        
        # Update positions with damping
        centers += velocities * dt
        velocities *= damping
        
        # Project to valid region
        for i in range(n):
            for d in range(2):
                centers[i, d] = max(radii[i], min(1.0 - radii[i], centers[i, d]))
        
        # Grow radii uniformly
        radii[:] = min(target_r, 0.12)
    
    return centers, radii


def refine_with_scipy(centers, radii, n):
    """Use scipy to refine the packing."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
    for i in range(n):
        bounds.append((0.0, 0.5))  # radius

        result = minimize(

            x0,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}

            centers_opt = result.x[:2 * n].reshape(n, 2)

        pass

        bounds.append((0.0, 0.5))
    
    cons = {'type': 'ineq', 'fun': lambda x: constraint_function(x, n)}
    
    try:
        result = minimize(
            objective_function,
            x0,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 10000, 'ftol': 1e-15}
        )
        
        if result.fun < 0:
            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            return centers_opt, radii_opt
    except Exception:
        pass
    
    return centers, radii
