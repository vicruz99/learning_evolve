import numpy as np
from scipy.optimize import minimize

def get_pairwise_dists(centers):
    """Compute matrix of pairwise Euclidean distances."""
    diff = centers[:, None, :] - centers[None, :, :]
    return np.sqrt(np.sum(diff**2, axis=2))

def resolve_overlaps(centers, radii, iters=500):
    """Iteratively push circles apart to satisfy non-overlap and boundary constraints."""
    n = centers.shape[0]
    for _ in range(iters):
        moved = False
        d_mat = get_pairwise_dists(centers)
        for i in range(n):
            for j in range(i + 1, n):
                d = d_mat[i, j]
                min_d = radii[i] + radii[j]
                if d < min_d - 1e-9:
                    diff = centers[i] - centers[j]
                    if d < 1e-12:
                        push = np.array([1.0, 0.0]) * 1e-6
                    else:
                        push = (diff / d) * (min_d - d) / 2.0
                    centers[i] += push
                    centers[j] -= push
                    moved = True
        if not moved:
            break
            
    # Project onto boundary constraints
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    return centers

def expand_and_optimize(n, init_centers, init_radii):
    """Grow radii until they are limited by geometry, resolving overlaps at each step."""
    centers = init_centers.copy()
    radii = init_radii.copy()
    for step in range(300):
        max_r = np.full(n, 1.0)
        d_mat = get_pairwise_dists(centers)
        for i in range(n):
            mr = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    mr = min(mr, d_mat[i, j] / 2.0)
            max_r[i] = mr
        
        # Blend current radii with max possible to grow smoothly
        radii = 0.85 * radii + 0.15 * max_r
        centers = resolve_overlaps(centers, radii, iters=150)
    return centers, radii
