import numpy as np
from scipy.optimize import minimize
import math

# Precompute pair indices for efficiency during penalty calculation
N_CIRCLES = 26
PAIR_INDICES = [(i, j) for i in range(N_CIRCLES) for j in range(i + 1, N_CIRCLES)]

def compute_objective(v, W=20000.0):
    """
    Computes the objective function: -sum(radii) + W * penalty.
    Penalty includes squared overlaps and boundary violations.
    """
    c = v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    r = v[2*N_CIRCLES:]
    
    obj = -np.sum(r)
    pen = 0.0

    # Overlap penalty
    for i, j in PAIR_INDICES:
        dx = c[i, 0] - c[j, 0]
        dy = c[i, 1] - c[j, 1]
        d = math.sqrt(dx*dx + dy*dy)
        gap = r[i] + r[j] - d
        if gap > 0:
            pen += gap * gap

    # Boundary penalty
    for i in range(N_CIRCLES):
        ri = r[i]
        xi, yi = c[i]
        if ri > xi: 
            pen += (ri - xi)**2
        if ri > 1 - xi: 
            pen += (ri - (1 - xi))**2
        if ri > yi: 
            pen += (ri - yi)**2
        if ri > 1 - yi: 
            pen += (ri - (1 - yi))**2

    return obj + W * pen

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -1.0
    best_v = None
    
    # Bounds for variables: x,y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * (2*N_CIRCLES) + [(1e-6, 0.5)] * N_CIRCLES

    # Multiple restarts to escape local minima
    for seed in range(20):
        np.random.seed(seed * 100 + 7)
        # Initialize centers in a safe interior region
        c_init = np.random.uniform(0.15, 0.85, size=(N_CIRCLES, 2))
        r_init = np.full(N_CIRCLES, 0.08)
        v_init = np.concatenate([c_init.ravel(), r_init])

        res = minimize(compute_objective, v_init, args=(20000.0,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-10})

        r_opt = res.x[2*N_CIRCLES:]
        s = np.sum(r_opt)
        
        # Estimate penalty from the returned objective value
        # obj = -s + W*pen  =>  pen = (obj + s)/W
        pen = (res.fun + s) / 20000.0
        
        # Accept if penalty is negligible and improves the sum
        if pen < 1e-6 and s > best_sum:
            best_sum = s
            best_v = res.x.copy()

    # Fallback if no strictly valid configuration was found during search
    if best_v is None:
        best_v = res.x.copy()

    c_final = best_v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    r_final = best_v[2*N_CIRCLES:]

    # Strict boundary clamping
    for i in range(N_CIRCLES):
        r_final[i] = min(r_final[i], c_final[i, 0], 1 - c_final[i, 0], 
                         c_final[i, 1], 1 - c_final[i, 1])

    # Safety scaling to guarantee no overlaps within tolerance
    for _ in range(100):
        valid = True
        for i, j in PAIR_INDICES:
            dist = math.sqrt((c_final[i, 0] - c_final[j, 0])**2 + 
                             (c_final[i, 1] - c_final[j, 1])**2)
            if dist < r_final[i] + r_final[j] - 1e-12:
                valid = False
                break
        if valid:
            break
        # Shrink radii uniformly to resolve overlaps
        r_final *= 0.998

    return c_final, r_final, float(np.sum(r_final))
