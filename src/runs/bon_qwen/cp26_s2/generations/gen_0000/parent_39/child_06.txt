        r_target = 0.095

        # Rows approx 1 / (r * sqrt(3))

        num_cols = 5

        radii = []

        x_step = 2 * r_target

        jitter = seed_offset * 0.01
        
        current_r = r_target

            y_base = current_r + r_idx * y_step
            # Check if row fits in height
            if y_base + current_r > 1.0:
                break
            
            # Offset every other row
            x_offset = current_r if r_idx % 2 == 1 else current_r

                if x_pos + current_r > 1.0:
                    break

                    break

                            np.random.rand() * (1 - 2*r_fill) + r_fill])
