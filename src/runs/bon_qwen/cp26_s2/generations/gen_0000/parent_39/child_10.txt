import numpy as np
from scipy.optimize import minimize

def generate_initial_positions(n):
    """
    Generates an initial hexagonal packing of n circles.
    """
    positions = []
    r_init = 0.09  # Initial tentative radius
    
    row = 0
    col = 0
    y = r_init
    
    while len(positions) < n:
        x = r_init
        # Shift every other row
        if row % 2 == 1:
            x += r_init
            
        while len(positions) < n and x + r_init <= 1.0:
            positions.append([x, y, r_init])
            x += 2 * r_init
            col += 1
        
        y += r_init * np.sqrt(3)
        row += 1
        if y + r_init > 1.0:
            # Reset row if we hit bottom, slightly shifting to pack more
            row = 0
            y = r_init
            col = 0
            if not positions: break # Safety break

    return np.array(positions)

def run_packing():
    n = 26
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None

    # Generate multiple initial configurations to avoid local minima
    # We will try a hexagonal packing and a few random perturbations
    initial_guesses = []
    
    # 1. Hexagonal packing
    hex_pos = generate_initial_positions(n)
    # Flatten to 1D array [x1, y1, r1, x2, y2, r2, ...]
    initial_guesses.append(hex_pos.flatten())
    
    # 2. Perturbed versions
    for _ in range(5):
        perturbed = hex_pos.copy()
        # Small random noise
        noise = np.random.uniform(-0.02, 0.02, perturbed.shape)
        # Ensure noise doesn't push out of bounds too much initially
        perturbed += noise
        # Clip positions to [0.05, 0.95] roughly to keep valid r
        perturbed[:, 0] = np.clip(perturbed[:, 0], 0.05, 0.95)
        perturbed[:, 1] = np.clip(perturbed[:, 1], 0.05, 0.95)
        perturbed[:, 2] = 0.09 # Reset radius
        initial_guesses.append(perturbed.flatten())

    for guess in initial_guesses:
        # Define objective: Maximize sum of radii => Minimize negative sum
        def objective(x_vars):
            radii = x_vars[2::3]
            return -np.sum(radii)

        # Define constraints
        constraints = []

        # 1. Boundary constraints
        for i in range(n):
            idx_x = i * 3
            idx_y = i * 3 + 1
            idx_r = i * 3 + 2
            
            # x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx_x] - x[idx_r]})
            # x + r <= 1 => 1 - (x + r) >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - (x[idx_x] + x[idx_r])})
            # y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx_y] - x[idx_r]})
            # y + r <= 1
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - (x[idx_y] + x[idx_r])})
            
            # Non-negative radius
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx_r]})

        # 2. Non-overlap constraints
        # dist_ij >= r_i + r_j
        # sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = i * 3, i * 3 + 1, i * 3 + 2
                idx_xj, idx_yj, idx_rj = j * 3, j * 3 + 1, j * 3 + 2
                
                # To avoid sqrt in constraint function for better performance/gradient,
                # we can square it, but careful with signs.
                # Since dist >= 0 and r >= 0, dist >= sum_r is equivalent to dist^2 >= sum_r^2
                # However, direct sqrt is safer for numerical stability if dist is small.
                # Let's use the direct form.
                
                def constraint_overlap(x, i=i, j=j):
                    xi, yi = x[idx_xi], x[idx_yi]
                    xj, yj = x[idx_xj], x[idx_yj]
                    ri, rj = x[idx_ri], x[idx_rj]
                    
                    dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                    return dist - (ri + rj)
                
                constraints.append({'type': 'ineq', 'fun': constraint_overlap})

        # Bounds for variables
        # x, y in [0, 1], r in [0, 0.5] (approx max possible)
        bounds = []
        for i in range(n):
            bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

        try:
            # Run optimizer
            res = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-9})
            
            if res.success or res.fun < -best_sum_radii:
                current_sum = -res.fun
                # Validate result manually to be safe before accepting
                centers = res.x.reshape(n, 3)[:, :2]
                radii = res.x.reshape(n, 3)[:, 2]
                
                # Quick validation
                valid = True
                for i in range(n):
                    if radii[i] < 0: valid = False; break
                    if centers[i][0] - radii[i] < -1e-9 or centers[i][0] + radii[i] > 1 + 1e-9: valid = False; break
                    if centers[i][1] - radii[i] < -1e-9 or centers[i][1] + radii[i] > 1 + 1e-9: valid = False; break
                
                if valid:
                    for i in range(n):
                        for j in range(i + 1, n):
                            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                            if dist < radii[i] + radii[j] - 1e-9:
                                valid = False
                                break
                        if not valid: break
                
                if valid and current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_centers = centers
                    best_radii = radii

        except Exception as e:
            pass

    if best_centers is None:
        # Fallback to simple grid if optimization fails
        # This shouldn't happen with the logic above
        centers = np.array([[0.1, 0.1]])
        radii = np.array([0.1])
        best_sum_radii = 0.1

    return best_centers, best_radii, best_sum_radii
