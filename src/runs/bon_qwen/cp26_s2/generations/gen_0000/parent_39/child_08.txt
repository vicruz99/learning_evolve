This can be solved efficiently using `scipy.optimize.linprog`.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist
import random

# Constants
N_CIRCLES = 26

def get_optimal_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Returns (sum_radii, radii_array).
    """
    n = N_CIRCLES
    
    # Compute pairwise distances
    # pdist returns condensed distance matrix (upper triangle)
    dists = pdist(centers, 'euclidean')
    
    # Build LP matrices
    # Variables: r_0, ..., r_{n-1}
    # Objective: Maximize sum(r_i) <=> Minimize -sum(r_i)
    c_obj = np.ones(n) * -1.0
    
    # Inequality constraints A_ub * r <= b_ub
    # r_i + r_j <= dist_ij
    m_constraints = n * (n - 1) // 2
    A_ub = np.zeros((m_constraints, n))
    b_ub = np.zeros(m_constraints)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[idx]
            idx += 1
            
    # Bounds for radii: 0 <= r_i <= distance to nearest boundary
    bounds = []
    for i in range(n):
        x, y = centers[i]
        # Distance to boundaries: x, 1-x, y, 1-y
        # If center is outside, min_dist could be negative, but radius must be >= 0
        # We clip at 0 to ensure feasible bounds
        min_dist = min(x, 1.0 - x, y, 1.0 - y)
        upper = max(0.0, min_dist)
        bounds.append((0.0, upper))
        
    # Solve LP
    # Using 'highs' method is generally fast and robust
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            # radii are in res.x
            radii = res.x
            total_sum = -res.fun # because we minimized -sum
            return total_sum, radii
        else:
            # Fallback or error handling
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for optimizer.
    Returns negative sum of radii (since minimize finds minimum).
    """
    centers = np.array(centers_flat).reshape(N_CIRCLES, 2)
    
    # Ensure centers are valid (though optimizer should keep them valid)
    # If a center is far outside, LP bounds handle it (upper=0), sum=0.
    
    sum_r, _ = get_optimal_radii(centers)
    return -sum_r

             pass

            x += step_x
        if len(hex_centers) >= N_CIRCLES:
            break
        y += step_y
    
    # Pad or trim to exactly 26

        hex_centers.append([0.5, 0.5])
    hex_centers = hex_centers[:N_CIRCLES]
    
    initial_configs = [np.array(hex_centers).flatten()]

    for _ in range(5):
