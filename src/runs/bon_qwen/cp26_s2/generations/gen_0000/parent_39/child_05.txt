import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initialize with a hexagonal-like packing (staggered rows)
    # Pattern: 5, 4, 5, 4, 5, 3 = 26 circles
    # Approximate radius 0.1
    centers = []
    radii = []
    r_init = 0.1
    dy = np.sqrt(3) * r_init
    
    # Row configuration
    row_counts = [5, 4, 5, 4, 5, 3]
    
    for row_idx, count in enumerate(row_counts):
        y = r_init + row_idx * dy
        # Determine x range and step
        # For 5 circles: width 1.0, centers at 0.1, 0.3, 0.5, 0.7, 0.9
        # For 4 circles: width 0.8, centers at 0.2, 0.4, 0.6, 0.8
        # For 3 circles: width 0.6, centers at 0.3, 0.5, 0.7
        
        if count == 5:
            x_start = r_init
            x_end = 1.0 - r_init
            step = (x_end - x_start) / (count - 1) if count > 1 else 0
            x_vals = np.linspace(x_start, x_end, count)
        elif count == 4:
            x_start = 2 * r_init
            x_end = 1.0 - 2 * r_init
            step = (x_end - x_start) / (count - 1) if count > 1 else 0
            x_vals = np.linspace(x_start, x_end, count)
        elif count == 3:
            x_start = 3 * r_init
            x_end = 1.0 - 3 * r_init
            step = (x_end - x_start) / (count - 1) if count > 1 else 0
            x_vals = np.linspace(x_start, x_end, count)
        else:
            x_vals = np.linspace(0, 1, count) # Fallback

        for x in x_vals:
            centers.append([x, y])
            radii.append(r_init)

    centers = np.array(centers)
    radii = np.array(radii)
    
    # Flatten variables for optimization: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]

    def objective(vars):
        # Maximize sum of radii => Minimize negative sum
        r_sum = sum(vars[3*i + 2] for i in range(n))
        return -r_sum

    def get_radii(vars):
        return np.array([vars[3*i + 2] for i in range(n)])
    
    def get_centers(vars):
        xs = np.array([vars[3*i] for i in range(n)])
        ys = np.array([vars[3*i + 1] for i in range(n)])
        return np.column_stack((xs, ys))

    # Constraints
    constraints = []

    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(n):
        # x >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i + 2]})
        # 1 - x >= r  =>  1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i] - v[3*i + 2]})
        # y >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i + 1] - v[3*i + 2]})
        # 1 - y >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i + 1] - v[3*i + 2]})
        
    # Non-overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i_x, idx_i_y, idx_i_r = 3*i, 3*i + 1, 3*i + 2
            idx_j_x, idx_j_y, idx_j_r = 3*j, 3*j + 1, 3*j + 2
            
            def non_overlap(v, i=i, j=j, ix=idx_i_x, iy=idx_i_y, ir=idx_i_r, jx=idx_j_x, jy=idx_j_y, jr=idx_j_r):
                dx = v[ix] - v[jx]
                dy = v[iy] - v[jy]
                r_sum = v[ir] + v[jr]
                return dx*dx + dy*dy - r_sum*r_sum
            
            constraints.append({'type': 'ineq', 'fun': non_overlap})

    # Bounds for variables
    # x in [0, 1], y in [0, 1], r >= 0
    bounds = []
    for _ in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 1)) # r

    # Optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 1000})
    
    if res.success:
        final_vars = res.x
    else:
        # Fallback to initial if optimization fails or returns nonsense
        final_vars = x0

    # Extract results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_vars[3*i]
        final_centers[i, 1] = final_vars[3*i + 1]
        final_radii[i] = final_vars[3*i + 2]

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
