def solve_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns radii array and sum of radii.
    """
    n = centers.shape[0]
    
    # Variables: r_0, r_1, ..., r_{n-1}
    # We want to maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # And r_i >= 0 (handled by bounds)
    
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x  => r_i + x <= 1
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - y)
        
    # Non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: [0, infinity)
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        radii = res.x
        return radii, -res.fun
