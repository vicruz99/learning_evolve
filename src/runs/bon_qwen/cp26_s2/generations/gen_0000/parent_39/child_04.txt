def get_overlap_forces(centers, radii, n):

    Calculates repulsive forces for overlapping circles and boundary violations.
    Returns forces vector (n, 2) and a flag indicating if any violation exists.
    """

    dists = np.sqrt(np.sum(diff**2, axis=2)) # (n, n)

    # Apply forces: circle i pushed away from j by norm_diff[i,j] * mag[i,j]

    return forces, violation

def generate_hex_grid(n):
    """Generates n points in a hexagonal grid inside [0,1]x[0,1]"""
    points = []
    # Estimate spacing
    # Area per point ~ 1/n. Hex area ~ sqrt(3)/2 * s^2.
    # s ~ sqrt(2 / (sqrt(3)*n))
    s = 1.0 / np.sqrt(n) 
    y = s * np.sqrt(3) / 2 # Row height
    
    row = 0
    while len(points) < n:
        row_shift = (row % 2) * (s / 2.0)
        x = s / 2.0 + row_shift
        while x <= 1.0 - s/2.0 and len(points) < n:
            points.append([x, y * row + s/2.0])
            x += s
        row += 1
        # Safety break for infinite loop if spacing too small
        if row > 20: break
        
    # Normalize/Scale to fit better?
    # The above generates points starting from s/2.
    # Let's scale them to fill [0,1].
    pts = np.array(points[:n])
    if len(pts) < n:
        # Fallback to random if grid generation fails to fill
        pts = np.random.rand(n, 2)
        
    # Scale to fit in [0.1, 0.9] initially to avoid walls
    min_c = np.min(pts, axis=0)
    max_c = np.max(pts, axis=0)
    pts = (pts - min_c) / (max_c - min_c) * 0.8 + 0.1
    
    return pts
