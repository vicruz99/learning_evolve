import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, sum_radii) for the optimal packing of 26 circles.
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Pre-calculate indices for constraints to speed up LP
    pair_indices = []
    for i in range(n):
        for j in range(i + 1, n):
            pair_indices.append((i, j))
    
    num_pairs = len(pair_indices)
    num_boundary = 4 * n

    # 1. Optimization of Centers
    # Using Differential Evolution to find center configurations
    def fitness(x_flat):
        centers = x_flat.reshape((n, 2))
        # Objective is the negative of the LP result (maximize -> minimize negative)
        lp_result = solve_lp_radii(centers, n, pair_indices, num_pairs, num_boundary)
        return -lp_result

    # Bounds for centers [0, 1]
    bounds = [(0.0, 1.0)] * (n * 2)

    # Run differential evolution
    # n_init=10 gives 10 * 10 * 52 = 5200 function evaluations, manageable
    result = opt.differential_evolution(fitness, bounds, maxiter=1000, popsize=15, 
                                        mutation=(0.5, 1.0), recombination=0.9, 
                                        tol=1e-4, seed=42, polish=True)

    best_centers = result.x.reshape((n, 2))
    
    # 2. Final LP to get radii
    final_sum = solve_lp_radii(best_centers, n, pair_indices, num_pairs, num_boundary)
    
    # Solve LP again to retrieve the actual radii values
    A_ub, b_ub = build_lp_constraints(best_centers, n, pair_indices, num_pairs, num_boundary)
    c_obj = -np.ones(n)
    
    # Solve using high precision
    lp_sol = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if not lp_sol.success:
        # Fallback to grid if optimization fails
        grid_coords = []
        for r in range(5):
            for c in range(5):
                grid_coords.append([0.1 + 0.2 * r, 0.1 + 0.2 * c])
        # Add 26th circle at center
        grid_coords.append([0.5, 0.5])
        best_centers = np.array(grid_coords[:26])
        lp_sol = opt.linprog(c_obj, A_ub=build_lp_constraints(best_centers, n, pair_indices, num_pairs, num_boundary)[0], 
                             b_ub=build_lp_constraints(best_centers, n, pair_indices, num_pairs, num_boundary)[1], 
                             bounds=(0, None), method='highs')

    final_radii = lp_sol.x
    
    # 3. Validation and Correction
    valid = validate_packing(best_centers, final_radii)
    if not valid:
        # If validation fails, shrink radii slightly to satisfy tolerance
        factor = 0.999
        final_radii *= factor
        best_sum = np.sum(final_radii)
    else:
        best_sum = np.sum(final_radii)

    return (best_centers, final_radii, best_sum)

def build_lp_constraints(centers, n, pair_indices, num_pairs, num_boundary):
    """
    Builds the matrix A_ub and vector b_ub for the LP:
    Maximize sum(r_i)
    Subject to:
      r_i <= boundary_limit
      r_i + r_j <= distance(i, j)
    """
    num_vars = n
    num_constraints = num_boundary + num_pairs
    A_ub = np.zeros((num_constraints, num_vars))
    b_ub = np.zeros(num_constraints)
    
    row = 0
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[row, i] = 1.0
        b_ub[row] = x
        row += 1
        # r_i <= 1 - x
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - x
        row += 1
        # r_i <= y
        A_ub[row, i] = 1.0
        b_ub[row] = y
        row += 1
        # r_i <= 1 - y
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - y
        row += 1

    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for (i, j) in pair_indices:
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        A_ub[row, i] = 1.0
        A_ub[row, j] = 1.0
        b_ub[row] = dist
        row += 1
        
    return A_ub, b_ub

def solve_lp_radii(centers, n, pair_indices, num_pairs, num_boundary):
    """
    Solves the LP for radii and returns the maximum sum.
    """
    A_ub, b_ub = build_lp_constraints(centers, n, pair_indices, num_pairs, num_boundary)
    c_obj = -np.ones(n)
    
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return res.fun * -1 # Return positive sum
        else:
            return 0.0
    except Exception:
        return 0.0

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    import numpy as np # Re-import locally for safety if needed
    n = centers.shape[0]

    if np.isnan(centers).any(): return False
    if np.isnan(radii).any(): return False

    for i in range(n):
        if radii[i] < -1e-9: return False
        
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False

    return True

if __name__ == "__main__":
    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
    print(f"Valid: {validate_packing(c, r)}")
