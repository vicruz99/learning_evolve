import numpy as np
from scipy.optimize import minimize

def penalty_function(vars, n, weight):
    """
    Computes the objective: negative sum of radii + penalty for constraint violations.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Primary objective: maximize sum of radii
    obj = -np.sum(radii)
    
    # Pairwise overlap penalty
    i, j = np.triu_indices(n, k=1)
    dists = np.linalg.norm(centers[i] - centers[j], axis=1)
    overlaps = radii[i] + radii[j] - dists
    obj += weight * np.sum(np.maximum(overlaps, 0)**2)
    
    # Boundary violation penalty
    obj += weight * np.sum(np.maximum(radii - centers[:,0], 0)**2)
    obj += weight * np.sum(np.maximum(centers[:,0] + radii - 1, 0)**2)
    obj += weight * np.sum(np.maximum(radii - centers[:,1], 0)**2)
    obj += weight * np.sum(np.maximum(centers[:,1] + radii - 1, 0)**2)
    
    return obj

def run_packing():
    n = 26
    best_vars = None
    best_obj = np.inf
    
    np.random.seed(42)
    
    # 1. Hexagonal lattice initialization for dense packing baseline
    pts = []
    for row in range(8):
        for col in range(6):
            if len(pts) >= n:
                break
            x = col * 2.0 + (row % 2) * 1.0
            y = row * np.sqrt(3)
            pts.append([x, y])
        if len(pts) >= n:
            break
            
    pts = np.array(pts[:n])
    pts -= pts.min(axis=0)
    pts /= pts.max(axis=0)
    pts = pts * 0.8 + 0.1  # Keep away from exact edges initially
    
    # Bounds for positions [0,1] and radii [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # 2. Multi-start optimization
    for trial in range(10):
        init_vars = np.zeros(3*n)
        init_vars[:2*n] = pts.flatten().copy()
        
        # Add stochastic perturbation for diversity
        if trial > 0:
            init_vars[:2*n] += np.random.uniform(-0.05, 0.05, (2*n))
            init_vars[:2*n] = np.clip(init_vars[:2*n], 0.05, 0.95)
        init_vars[2*n:] = 0.06  # Initial radius guess
        
        # Run optimizer with high penalty weight to enforce constraints
        weight = 2000
        res = minimize(penalty_function, init_vars, args=(n, weight), 
                       method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 5000, 'ftol': 1e-14})
                       
        if res.fun < best_obj:
            best_obj = res.fun
            best_vars = res.x
            
    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]
    
    return centers, radii, float(np.sum(radii))
