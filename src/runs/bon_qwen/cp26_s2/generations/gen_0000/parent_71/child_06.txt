import numpy as np

def run_packing():
    np.random.seed(42)
    N = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Try multiple initial configurations to escape local minima
    inits = [
        _grid_init(N),
        _hex_init(N),
        _random_init(N),
        _random_init(N),
        _random_init(N)
    ]

        if s > best_sum:
            best_sum = s
            best_centers = centers
            best_radii = radii
            
    return best_centers, best_radii, float(best_sum)

def _grid_init(N):
    """Initialize with a 5x5 grid plus center, shuffled."""
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    pts.append([0.5, 0.5])
    np.random.shuffle(pts)
    return np.array(pts[:N])

def _hex_init(N):
    """Initialize with a hexagonal lattice pattern."""
    pts = []
    r = 0.15
    dy = r * 1.5
    dx = r * np.sqrt(3)
    y = 0.1
    row = 0
    while len(pts) < N:
        x = 0.1 + (row % 2) * dx / 2
        while x < 1.0 - 0.1 and len(pts) < N:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
    np.random.shuffle(pts)
    return np.array(pts[:N])

def _random_init(N):
    """Initialize with random positions in the center region."""
    return np.random.rand(N, 2) * 0.8 + 0.1

def _inflate_and_optimize(N, initial_centers):
    """Run the force-directed packing simulation with growing radii."""
    centers = initial_centers.copy()
    radii = np.full(N, 0.04)
    
    lr = 0.02
    growth = 0.00015
    steps = 10000
    
    for step in range(steps):
        # Slowly increase radii to expand the packing
        radii += growth
        
        forces = np.zeros((N, 2))
        
        # Vectorized pairwise repulsion
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff ** 2, axis=2)
        dist = np.sqrt(dist_sq + 1e-12)
        min_dist = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        overlap = min_dist - dist
        mask = overlap > 0
        
        # Force proportional to overlap distance
        k = 5.0
        force_mag = np.where(mask, k * overlap / (dist + 1e-12), 0.0)
        
        # Apply repulsive forces
        force_vec = diff * force_mag[:, :, np.newaxis]
        forces = np.sum(force_vec, axis=1)
        
        # Vectorized boundary repulsion
        # Left boundary
        overlap_x = radii - centers[:, 0]
        mask_xl = overlap_x > 0
        forces[mask_xl, 0] += 10.0 * overlap_x[mask_xl]
        
        # Right boundary
        overlap_x = centers[:, 0] + radii - 1.0
        mask_xr = overlap_x > 0
        forces[mask_xr, 0] -= 10.0 * overlap_x[mask_xr]
        
        # Bottom boundary
        overlap_y = radii - centers[:, 1]
        mask_yb = overlap_y > 0
        forces[mask_yb, 1] += 10.0 * overlap_y[mask_yb]
        
        # Top boundary
        overlap_y = centers[:, 1] + radii - 1.0
        mask_yt = overlap_y > 0
        forces[mask_yt, 1] -= 10.0 * overlap_y[mask_yt]
        
        # Update positions
        centers += lr * forces
        
        # Strictly enforce boundary constraints
        r_min = radii[:, np.newaxis]
        r_max = 1.0 - r_min
        centers = np.clip(centers, r_min, r_max)
        
        # Add small random jitter periodically to escape local minima
        if step % 1000 == 0 and step > 0:
            jitter = np.random.normal(0, 0.002, centers.shape)
            centers += jitter
            centers = np.clip(centers, r_min, r_max)
            
    return centers, radii, np.sum(radii)
