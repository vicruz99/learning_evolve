import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    N = 26
    # Initial guess: Hexagonal grid
    # We estimate a radius that fits roughly 26 circles. 
    # A 5x5 grid fits 25 with r=0.1. For 26, we might need slightly smaller or better packing.
    # Let's start with r = 0.09 to be safe and let the optimizer expand them.
    r_init = 0.09
    
    centers = np.zeros((N, 2))
    radii = np.full(N, r_init)
    
    # Fill hexagonal grid
    idx = 0
    row = 0
    while idx < N:
        # Y coordinate for current row
        y = r_init + row * (np.sqrt(3) * r_init)
        if y + r_init > 1.0:
            break
            
        # X coordinates
        # Even rows (0, 2, ...) start at r_init
        # Odd rows (1, 3, ...) start at 2*r_init (shifted)
        if row % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
            
        col = 0
        x = x_start
        while x + r_init <= 1.0 and idx < N:
            centers[idx] = [x, y]
            idx += 1
            col += 1
            x += 2 * r_init
        row += 1
        
    # If we didn't fit 26 in the loop (unlikely with this logic unless r is too big),
    # we just have a subset. But with r=0.09, it should fit.
    # Just in case, if we have fewer than 26, we would need a fallback, 
    # but 0.09 is small enough.
    
    # Flatten variables for optimization: [x0, y0, r0, x1, y1, r1, ...]
    # Actually, scipy.optimize handles bounds and constraints on specific indices, 
    # but it's easier to pass a single vector.
    # Let's use a vector of size 3*N.
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(N):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Constraints
    def bound_constraints(vars):
        """Constraints to keep circles inside the square"""
        cons = []
        for i in range(N):
            x = vars[3*i]
            y = vars[3*i+1]
            r = vars[3*i+2]
            # x - r >= 0
            cons.append(x - r)
            # y - r >= 0
            cons.append(y - r)
            # x + r <= 1
            cons.append(1.0 - (x + r))
            # y + r <= 1
            cons.append(1.0 - (y + r))
        return cons

    def overlap_constraints(vars):
        """Constraints to prevent overlap"""
        cons = []
        for i in range(N):
            for j in range(i + 1, N):
                xi = vars[3*i]
                yi = vars[3*i+1]
                ri = vars[3*i+2]
                xj = vars[3*j]
                yj = vars[3*j+1]
                rj = vars[3*j+2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                # dist >= ri + rj  =>  dist^2 >= (ri+rj)^2
                # To avoid sqrt and keep it smooth/differentiable, we can use dist^2 - (r+r)^2 >= 0
                # However, this is not strictly equivalent if distances are negative (impossible here) 
                # but numerically safer. 
                # Actually, constraint is sqrt(dist_sq) - (ri + rj) >= 0
                # Using squared form: dist_sq - (ri + rj)**2 >= 0
                # This is valid since dist, ri, rj >= 0.
                val = dist_sq - (ri + rj)**2
                cons.append(val)
        return cons

    # Objective: Maximize sum of radii -> Minimize negative sum
    def objective(vars):
        r_sum = sum(vars[3*i + 2] for i in range(N))
        return -r_sum

    # Combine constraints
    constraints = []
    
    # Boundary constraints (inequality >= 0)
    # SLSQP expects constraints in the form cons(x) >= 0 or cons(x) == 0
    # We can pass a function that returns a list of values, each >= 0.
    # But scipy.optimize.minimize with SLSQP expects constraints as dicts.
    
    # Helper to create constraint dict
    def make_constraint(func):
        return {'type': 'ineq', 'fun': func}

    # However, passing a function that returns a list is not directly supported for 'ineq' in a single dict 
    # unless we use 'nonlinear_constraint' logic or split them. 
    # Actually, scipy supports returning an array for 'ineq'.
    
    constraints.append(make_constraint(bound_constraints))
    constraints.append(make_constraint(overlap_constraints))

    # Optimization
    # SLSQP is a good choice.
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        final_vars = res.x
    except Exception as e:
        # Fallback if optimization fails or throws
        final_vars = x0
        
    # Extract results
    final_centers = np.zeros((N, 2))
    final_radii = np.zeros(N)
    
    for i in range(N):
        final_centers[i, 0] = final_vars[3*i]
        final_centers[i, 1] = final_vars[3*i + 1]
        final_radii[i] = final_vars[3*i + 2]
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
