import numpy as np
import scipy.optimize as opt

def run_packing():
    n = 26
    
    def objective(vars):
        # Objective is to minimize negative sum of radii
        radii = vars[2*n:]
        return -np.sum(radii)

    def constraints(vars):
        centers = vars[:2*n].reshape(n, 2)
        radii = vars[2*n:]
        cons = []
        
        # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            cons.append(x - r)
            cons.append(r - x)
            cons.append(y - r)
            cons.append(r - y)
            cons.append(r) # r >= 0
            
        # Overlap constraints: dist >= r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx**2 + dy**2
                rad_sum = radii[i] + radii[j]
                # Constraint: dist_sq - (r_i + r_j)^2 >= 0
                cons.append(dist_sq - rad_sum**2)
                
        return np.array(cons)

    def get_initial_guess(seed=None):
        rng = np.random.default_rng(seed)
        hex_centers = []
        row_height = np.sqrt(3) / 2
        
        # Approximate hexagonal packing
        rows = 6
        cols = 5
        count = 0
        for r in range(rows):
            y = r * row_height
            for c in range(cols):
                if count >= n:
                    break
                x = c * 1.0 + (y % 2) * 0.5
                hex_centers.append([x, y])
                count += 1
        
        # Normalize to unit square with padding
        if len(hex_centers) >= n:
            centers = np.array(hex_centers[:n])
            min_c, max_c = centers.min(axis=0), centers.max(axis=0)
            centers = (centers - min_c) / (max_c - min_c) * 0.9 + 0.05
        else:
            centers = rng.random((n, 2)) * 0.8 + 0.1
            
        radii = np.full(n, 0.1)
        return np.concatenate([centers.flatten(), radii])

    best_result = None
    best_val = -np.inf

    # Run multiple restarts
    for i in range(10):
        try:
            x0 = get_initial_guess(seed=i)
            bounds = [(0, 1)] * (2*n) + [(0, 1)] * n
            
            # Use SLSQP with a callback to monitor progress is optional, 
            # but we just let it run for a fixed time/iteration limit if needed.
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds,

                best_result = res

        radii = best_result.x[2*n:]
        
    return centers, radii, np.sum(radii)
