import numpy as np

def run_packing():
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    Uses a force-directed growing circles algorithm.
    """
    n = 26
    np.random.seed(42)  # For reproducibility

    # 1. Initialization
    # Start with a grid-like distribution to spread them out, then add jitter
    # A 5x5 grid has 25 spots. We need 26.
    # Let's place 25 in a 5x5 grid and 1 in the center or random.
    # Actually, random initialization with low radii is often safer to avoid bias,
    # but grid is good for initial spacing.
    
    # Let's try a hexagonal-ish initial placement or just random.
    # Random is robust.
    centers = np.random.uniform(0.2, 0.8, size=(n, 2))
    radii = np.full(n, 0.02) # Start small

    # Simulation parameters
    num_steps = 2000
    growth_rate = 1.0005  # Slow growth
    repulsion_k = 50.0    # Force constant
    wall_k = 100.0        # Wall force constant
    dt = 0.005            # Time step for movement

    # To prevent circles from getting stuck, we can also try to optimize
    # by allowing radii to grow unevenly if some are constrained?
    # But uniform growth is a good baseline.

    for step in range(num_steps):
        # 1. Grow radii
        # We grow them slightly. If they overlap, forces will push centers apart.
        # Increasing radii too fast might cause chaos, but with high repulsion it's ok.
        radii *= growth_rate
        
        # Cap radii at 0.5 to prevent numerical issues, though constraints should prevent this.
        radii = np.clip(radii, 0, 0.5)

        # 2. Calculate Forces
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        # Vectorized distance calculation
        # centers shape (n, 2)
        # diff shape (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # Avoid self-interaction (diagonal is 0)
        np.fill_diagonal(dist_sq, np.inf)
        
        dist = np.sqrt(dist_sq)
        dist[dist < 1e-10] = 1e-10 # Prevent division by zero

        # Sum of radii
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        # Overlap amount (positive if overlapping)
        overlap = r_sum - dist
        
        # Force magnitude: proportional to overlap
        # We only apply force if overlap > 0
        # To make it smooth, we can just use overlap directly or squared.
        # Linear force is standard for repulsion.
        force_mag = np.maximum(0, overlap) * repulsion_k
        
        # Force direction: unit vector * magnitude
        # unit vector = diff / dist
        # But we need to be careful with dist=0.
        # We can normalize diff.
        # normalized_diff = diff / dist[:, :, np.newaxis]
        # However, dist is (n, n). diff is (n, n, 2).
        
        # Efficient way:
        # force_vector = diff * (force_mag / dist_sq)[:, :, np.newaxis] 
        # because diff/dist = diff / sqrt(dist_sq) = diff * sqrt(dist_sq) / dist_sq ?
        # No. diff / dist.
        # force = diff * (force_mag / dist)
        
        # Let's compute unit vectors safely
        # dist has inf on diagonal, so safe.
        inv_dist = 1.0 / dist
        unit_vectors = diff * inv_dist[:, :, np.newaxis]
        
        # Accumulate forces
        # force_mag is (n, n). We need to sum over neighbors.
        # forces[i] += sum_j (unit_vectors[i, j] * force_mag[i, j])
        
        # Vectorized sum
        # Expand dims for broadcasting
        # unit_vectors: (n, n, 2)
        # force_mag: (n, n, 1)
        force_vectors = unit_vectors * force_mag[:, :, np.newaxis]
        
        # Sum over axis 1 (the neighbor dimension)
        repulsive_forces = np.sum(force_vectors, axis=1)
        
        forces += repulsive_forces

        # 3. Wall Forces
        # If x < r, push right. If x > 1-r, push left.
        # x coordinate: centers[:, 0]
        x = centers[:, 0]
        y = centers[:, 1]
        
        # Left wall
        dist_left = x - radii
        penetration_left = np.maximum(0, -dist_left)
        forces[:, 0] += wall_k * penetration_left
        
        # Right wall
        dist_right = (1 - x) - radii
        penetration_right = np.maximum(0, -dist_right)
        forces[:, 0] -= wall_k * penetration_right
        
        # Bottom wall
        dist_bottom = y - radii
        penetration_bottom = np.maximum(0, -dist_bottom)
        forces[:, 1] += wall_k * penetration_bottom
        
        # Top wall
        dist_top = (1 - y) - radii
        penetration_top = np.maximum(0, -dist_top)
        forces[:, 1] -= wall_k * penetration_top

        # 4. Update Positions
        centers += dt * forces
        
        # 5. Clamp positions to valid range strictly
        # This handles numerical drift or strong forces pushing out
        # The valid range for center is [r, 1-r]
        # But since we just added forces, maybe just clamp to [0, 1] is safer?
        # Actually, if we clamp to [0,1], the wall force logic might fight it.
        # Better to clamp to [0, 1] and rely on wall forces.
        centers[:, 0] = np.clip(centers[:, 0], 0, 1)
        centers[:, 1] = np.clip(centers[:, 1], 0, 1)
        
        # However, strict constraint is x >= r and x <= 1-r.
        # If a circle is pushed out, we must pull it back or shrink it.
        # Shrinking radii is bad for objective.
        # Let's strictly enforce center bounds relative to current radius.
        # This acts as a hard constraint projection.
        
        # Project x
        min_x = radii
        max_x = 1 - radii
        centers[:, 0] = np.clip(centers[:, 0], min_x, max_x)
        
        # Project y
        min_y = radii
        max_y = 1 - radii
        centers[:, 1] = np.clip(centers[:, 1], min_y, max_y)

    # Final check and slight adjustment if needed
    # Sometimes clipping might cause overlaps if radii grew too fast in last step.
    # But with small growth rate and projection, it should be stable.
    
    # Calculate sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)
