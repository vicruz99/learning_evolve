import numpy as np
import math
from scipy.optimize import minimize

def run_packing():

    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    # Helper function to generate initial hexagonal-like grid
    def generate_initial_hex(n_circles, square_size=1.0):
        # Approximate radius for equal circles
        # Density ~ 0.9, Area = n * pi * r^2 <= 0.9 * 1
        # r ~ sqrt(0.9 / (n * pi))
        r_est = math.sqrt(0.9 / (n_circles * math.pi))
        
        centers = []
        radii = []
        
        # Try to fill rows
        # Vertical spacing for hex packing is r * sqrt(3)
        # Horizontal spacing is 2r
        # Let's try to fit into 1x1
        
        # We'll use a slightly smaller r to ensure they fit initially
        r_init = r_est * 0.95
        
        count = 0
        row = 0
        while count < n_circles:
            y = r_init + row * r_init * math.sqrt(3)
            if y + r_init > 1.0:
                # Shift rows down if needed, or break
                break
            
            # Horizontal offset for alternating rows
            offset = r_init if row % 2 == 1 else 0
            x = r_init + offset
            
            col = 0
            while x + r_init <= 1.0 and count < n_circles:
                centers.append([x, y])
                radii.append(r_init)
                count += 1
                x += 2 * r_init
                col += 1
            row += 1
        
        # If we didn't fit enough, fall back to random or grid
        if count < n_circles:
            centers = []
            radii = []
            # Random placement
            for _ in range(n_circles):
                centers.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
                radii.append(0.05) # Small safe radius
        
        return np.array(centers), np.array(radii)

    # Objective: Maximize sum of radii => Minimize -sum(radii)
    def objective(vars):
        # vars contains [x0, y0, r0, x1, y1, r1, ...]
        radii = vars[2::3]
        return -np.sum(radii)

    # Constraints
    def constraints_factory(n):
        cons = []
        
        # Boundary constraints
        # x_i - r_i >= 0
        # r_i - x_i + 1 >= 0  => 1 - x_i - r_i >= 0 ? No.
        # x_i <= 1 - r_i => x_i + r_i <= 1 => 1 - x_i - r_i >= 0
        
        def bound_x_lower(vars):
            return vars[0::3] - vars[2::3] # x - r >= 0

            return 1.0 - vars[0::3] - vars[2::3] # 1 - x - r >= 0
            
        def bound_y_lower(vars):
            return vars[1::3] - vars[2::3] # y - r >= 0
            
        def bound_y_upper(vars):
            return 1.0 - vars[1::3] - vars[2::3] # 1 - y - r >= 0
            
        def non_negative_r(vars):
            return vars[2::3] # r >= 0

        cons.append({'type': 'ineq', 'fun': bound_x_lower})
        cons.append({'type': 'ineq', 'fun': bound_x_upper})
        cons.append({'type': 'ineq', 'fun': bound_y_lower})
        cons.append({'type': 'ineq', 'fun': bound_y_upper})
        cons.append({'type': 'ineq', 'fun': non_negative_r})

        # Non-overlap constraints: dist(i, j) >= r_i + r_j

        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        
        def non_overlap(vars):
            vals = []
            for i in range(n):

                    xi = vars[3*i]
                    yi = vars[3*i+1]

                    rj = vars[3*j+2]

                    vals.append(dist_sq - r_sum_sq)
            return np.array(vals)

        cons.append({'type': 'ineq', 'fun': non_overlap})
        
        return cons
