import numpy as np
import scipy.optimize as opt
import itertools

def get_initial_guess(n, seed=0):
    """Generates various initial guesses for circle centers."""
    np.random.seed(seed)
    guesses = []
    
    # 1. Perturbed Grid (5x5 + 1)
    # 25 circles in a 5x5 grid
    grid_centers = []
    for i in range(5):
        for j in range(5):
            grid_centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    # Add one circle in the center or random spot
    grid_centers.append([0.5, 0.5])
    
    # Take first n
    centers = np.array(grid_centers[:n])
    # Add small random perturbation
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    # Clip to [0.01, 0.99] to ensure valid radii initialization
    centers = np.clip(centers, 0.01, 0.99)
    guesses.append(centers)
    
    # 2. Hexagonal Lattice
    hex_centers = []
    r_est = 0.1 # Initial guess for radius to determine spacing
    spacing_x = 2 * r_est
    spacing_y = np.sqrt(3) * r_est
    y = r_est
    count = 0
    while count < n and y + r_est <= 1.0:
        x = r_est
        # Shift every other row
        shift = (int((y - r_est) / spacing_y) % 2) * r_est
        while x + r_est <= 1.0 and count < n:
            hex_centers.append([x + shift, y])
            count += 1
            x += spacing_x
        y += spacing_y
    hex_centers = np.array(hex_centers[:n])
    hex_centers += np.random.uniform(-0.005, 0.005, hex_centers.shape)
    hex_centers = np.clip(hex_centers, 0.01, 0.99)
    guesses.append(hex_centers)
    
    # 3. Random
    random_centers = np.random.uniform(0.1, 0.9, (n, 2))
    guesses.append(random_centers)
    
    return guesses

def objective_func(vars, n):
    """Objective: Minimize negative sum of radii."""
    r = vars[2*n:]
    return -np.sum(r)

def boundary_constraints(vars, n):
    """Constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    Formulated as inequality >= 0.
    x - r >= 0 => x - r >= 0
    1 - x - r >= 0
    y - r >= 0
    1 - y - r >= 0
    """
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    constraints = []
    for i in range(n):
        constraints.append(x[i] - r[i])
        constraints.append(1.0 - x[i] - r[i])
        constraints.append(y[i] - r[i])
        constraints.append(1.0 - y[i] - r[i])
    return np.array(constraints)

def overlap_constraints(vars, n):
    """Constraints: dist(i,j) >= r_i + r_j
    (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    """
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = (x[i] - x[j])**2 + (y[i] - y[j])**2
            rad_sum = r[i] + r[j]
            constraints.append(dist_sq - rad_sum**2)
    return np.array(constraints)

def run_optimization(initial_centers, n):
    """Runs the optimization for a given set of initial centers."""
    n_circles = n
    # Variables: x (n), y (n), r (n) -> 3n
    x0 = np.concatenate([initial_centers[:, 0], initial_centers[:, 1], np.full(n_circles, 0.05)])
    
    # Bounds
    # x, y in [0, 1]
    # r >= 0
    bounds = [(0, 1)] * n_circles + [(0, 1)] * n_circles + [(0, 1)] * n_circles
    
    # Constraints
    cons = []
    
    # Boundary constraints (linear)
    # x_i - r_i >= 0
    # 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0
    # We can define these as dictionary constraints for SLSQP
    # But SLSQP takes a list of dicts.
    
    # Let's combine all constraints into one function for efficiency or list of dicts
    # SLSQP supports list of dicts.
    
    # Linear constraints can be specified in dict as 'linear' or func.
    # Using func is safer for mixed types.
    
    def get_boundary_cons(vars, n):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        c = []
        for i in range(n):
            c.append(x[i] - r[i])
            c.append(1.0 - x[i] - r[i])
            c.append(y[i] - r[i])
            c.append(1.0 - y[i] - r[i])
        return np.array(c)

    def get_overlap_cons(vars, n):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        c = []
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = (x[i] - x[j])**2 + (y[i] - y[j])**2
                rad_sum = r[i] + r[j]
                c.append(dist_sq - rad_sum**2)
        return np.array(c)

    cons = [
        {'type': 'ineq', 'fun': lambda v: get_boundary_cons(v, n_circles)},
        {'type': 'ineq', 'fun': lambda v: get_overlap_cons(v, n_circles)}
    ]
    
    try:
        res = opt.minimize(
            objective_func, 
            x0, 
            args=(n_circles,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
        )
        return res
    except Exception:
        return None

def run_packing():

    best_sum_r = -1.0

                        valid = False
                        break
                # Check overlaps

                        best_centers = centers.copy()

    # If optimization didn't find anything good, fallback to a known valid simple packing
