For this circle packing problem maximizing sum of radii for 26 circles in a unit square, I'll use a coordinate descent approach:

1. **Initialization**: Start with circles placed in a hexagonal grid pattern, which is known to be optimal for equal circle packing density.

2. **Coordinate Descent**: Alternate between:
   a. Fix positions, optimize radii using Linear Programming (LP) - this is exact and fast since constraints are linear in radii
   b. Fix radii, optimize positions using force-based relaxation with repulsive forces for overlaps and boundary violations

3. **Multiple restarts**: Try different random perturbations to escape local minima

4. **Key insight**: The LP formulation for fixed positions makes the radius optimization exact and efficient. The force-based position optimization handles the non-convex geometry naturally.

5. **Force design**: Use strong repulsion for overlapping circles, moderate repulsion for near-circles, and boundary repulsion to keep circles inside the square.

import numpy as np
from scipy.optimize import linprog


def run_packing():
    n = 26

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Try multiple trials with different random perturbations
    for trial in range(30):
        centers = initialize_hexagonal(n, trial)
        radii = np.ones(n) * 0.09

        # Coordinate descent: alternate between optimizing radii (LP) and positions (forces)
        for outer_iter in range(60):
            radii = maximize_radii_lp(centers, n)
            if radii is None or np.sum(radii) < 1e-10:
                break
            centers = optimize_positions(centers, radii, n)

        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    return best_centers, best_radii, best_sum


def initialize_hexagonal(n, seed):
    """Initialize circles in a hexagonal grid pattern with small random perturbations."""
    np.random.seed(seed * 42 + 7)
    centers = np.zeros((n, 2))
    idx = 0
    row_spacing = 0.12
    col_spacing = 0.18

    for row in range(12):
        cols = 5
        if idx + cols > n:
            cols = n - idx
        for col in range(cols):
            x = 0.08 + col * col_spacing
            if row % 2 == 1:
                x += col_spacing / 2
            y = 0.08 + row * row_spacing
            # Add small random perturbation
            x += np.random.uniform(-0.005, 0.005)
            y += np.random.uniform(-0.005, 0.005)
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    return centers


def maximize_radii_lp(centers, n):
    """
    Given fixed positions, find optimal radii using Linear Programming.
    Maximize sum(r_i) subject to:
    - r_i + r_j <= dist(c_i, c_j) for all pairs
    - r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i for all i
    - r_i >= 0
    """
    # Objective: minimize -sum(r_i)
    c_obj = -np.ones(n)

    A_ub_list = []
    b_ub_list = []

    # Pair constraints: r_i + r_j <= dist(c_i, c_j)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub_list.append(row)
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            b_ub_list.append(dist)

    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        bounds_for_i = [
            centers[i][0],
            1 - centers[i][0],
            centers[i][1],
            1 - centers[i][1]
        ]
        for bound in bounds_for_i:
            row = np.zeros(n)
            row[i] = 1
            A_ub_list.append(row)
            b_ub_list.append(bound)

    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    bounds = [(0, None)] * n

    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success and np.all(result.x >= -1e-10):
        return np.maximum(result.x, 0)
    else:
        # Fallback: greedy computation
        radii = compute_radii_greedy(centers, n)
        return radii


def compute_radii_greedy(centers, n):
    """Fallback greedy radius computation if LP fails."""
    radii = np.zeros(n)
    for i in range(n):
        max_r = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                max_r = min(max_r, dist - radii[j])
        radii[i] = max(0, max_r)
    return radii


def optimize_positions(centers, radii, n):
    """
    Given fixed radii, optimize positions using force-based relaxation.
    Uses repulsive forces for overlapping/near circles and boundary violations.
    """
    temperature = 0.008

    for iteration in range(4000):
        forces = np.zeros((n, 2))

        # Inter-circle forces
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]

                if dist < min_dist + 0.03 and dist > 1e-12:
                    if dist < min_dist:
                        # Strong repulsion for overlaps
                        overlap = min_dist - dist
                        force_mag = overlap * 80
                    else:
                        # Moderate repulsion to maintain spacing and create room
                        excess = dist - min_dist
                        force_mag = 0.5 / (1 + excess * 50)

                    forces[i] += diff / dist * force_mag
                    forces[j] -= diff / dist * force_mag

            # Boundary repulsion forces
            for axis in range(2):
                if centers[i][axis] - radii[i] < 0:
                    forces[i][axis] += (-centers[i][axis] + radii[i]) * 100
                if centers[i][axis] + radii[i] > 1:
                    forces[i][axis] -= (centers[i][axis] + radii[i] - 1) * 100

        # Adaptive step size
        step = 0.008 / (1 + iteration * 0.003)
        centers += forces * step

        # Simulated annealing: random perturbation
        centers += np.random.randn(n, 2) * temperature
        temperature *= 0.998

        # Clamp positions to valid range
        for i in range(n):
            centers[i] = np.clip(centers[i], radii[i], 1 - radii[i])

    return centers
