import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n):
    """Objective function for scipy.optimize.minimize.
    Maximizes radius r while penalizing overlaps and boundary violations.
    """
    cs = vars[:2*n].reshape(n, 2)
    r = vars[2*n]
    
    if r <= 0:
        return 1e6
        
    penalty = 0.0
    
    # Pairwise distances
    diff = cs[:, np.newaxis, :] - cs[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    np.fill_diagonal(dists, np.inf)
    
    # Overlap penalty: push circles apart if distance < 2r
    overlap = np.maximum(0, 2*r - dists)
    penalty += np.sum(overlap**2)
    
    # Boundary penalty: keep circles inside [0,1]
    penalty += np.sum(np.maximum(0, r - cs)**2)
    penalty += np.sum(np.maximum(0, cs - (1.0 - r))**2)
    
    # Objective: maximize r (minimize -r) + weighted penalty
    return -r + 5000 * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialize on a hexagonal lattice
    centers = np.zeros((n, 2))
    idx = 0
    # Row counts for a dense hexagonal arrangement
    rows = [6, 5, 6, 5, 4]
    sp = 0.15
    dy = sp * np.sqrt(3) / 2
    
    for r_idx, count in enumerate(rows):
        for c_idx in range(count):
            centers[idx] = [c_idx * sp - count * sp / 2 + sp/2, r_idx * dy]
            idx += 1
            
    # Normalize to fit comfortably inside [0,1]
    min_c = centers.min(axis=0)
    max_c = centers.max(axis=0)
    centers = (centers - min_c) / (max_c - min_c) * 0.8 + 0.1
    
    # 2. Force-directed expansion
    dt = 0.01
    k_rep = 100.0
    target_r = 0.05
    
    for _ in range(1500):
        forces = np.zeros_like(centers)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        np.fill_diagonal(dists, np.inf)
        
        threshold = 2 * target_r
        i_idx, j_idx = np.where(dists < threshold)
        
        for i, j in zip(i_idx, j_idx):
            if i < j:
                d = dists[i, j]
                f_mag = (threshold - d) * k_rep
                dir_vec = (centers[i] - centers[j]) / d
                forces[i] += dir_vec * f_mag
                forces[j] -= dir_vec * f_mag
                
        # Boundary repulsion
        for dim in range(2):
            mask_l = centers[:, dim] < target_r
            forces[mask_l, dim] += k_rep * (target_r - centers[mask_l, dim])
            mask_r = centers[:, dim] > 1.0 - target_r
            forces[mask_r, dim] -= k_rep * (centers[mask_r, dim] - (1.0 - target_r))
            
        centers += forces * dt
        target_r += 0.0003
        dt *= 0.995
        
    # 3. Precise optimization with scipy
    x0 = np.concatenate([centers.flatten(), [target_r]])
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)]
    
    res = minimize(
        compute_objective, 
        x0, 
        args=(n,), 
        method='L-BFGS-B', 
        bounds=bounds, 
        options={'maxiter': 4000, 'ftol': 1e-9}
    )
    
    opt_centers = res.x[:2*n].reshape(n, 2)
    opt_r = res.x[2*n]
    
    # 4. Strict validity enforcement
    diff = opt_centers[:, np.newaxis, :] - opt_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    np.fill_diagonal(dists, np.inf)
    min_d = np.min(dists)
    
    # Radius is limited by half min distance, and distance to boundaries
    actual_r = min(opt_r, min_d / 2.0, np.min(opt_centers), np.min(1.0 - opt_centers))
    actual_r = max(actual_r, 1e-5)
    
    # Final projection to guarantee constraints
    opt_centers = np.clip(opt_centers, actual_r, 1.0 - actual_r)
    
    radii = np.full(n, actual_r)
    return opt_centers, radii, float(np.sum(radii))
