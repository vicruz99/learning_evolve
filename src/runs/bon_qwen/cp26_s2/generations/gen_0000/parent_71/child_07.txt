import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_vec(vars):
    """Vectorized constraint function for SLSQP. Returns constraint values >= 0."""
    n = 26
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: r <= x <= 1-r  and  r <= y <= 1-r
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    
    i, j = np.triu_indices(n, k=1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - (r[i] + r[j])**2])
    
    return c

def run_packing():
    n = 26
    
    # 1. Initial hex-like placement
    centers = np.zeros((n, 2))
    row_counts = [6, 5, 6, 5, 4]  # Sums to 26
    idx = 0
    y = 0.1
    for count in row_counts:
        x_start = (1.0 - (count - 1) * 0.2) / 2.0
        for _ in range(count):
            centers[idx] = [x_start + 0.1, y + 0.1]
            x_start += 0.2
            idx += 1
        y += 0.173205081  # sqrt(3)/2 * 0.2
        
    # Slight random perturbation to break symmetry
    rng = np.random.default_rng(42)
    centers += rng.uniform(-0.01, 0.01, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Flatten to optimization variables [x0, y0, r0, x1, y1, r1, ...]
    vars0 = np.zeros(3 * n)
    for i in range(n):
        vars0[3*i] = centers[i, 0]
        vars0[3*i+1] = centers[i, 1]
        vars0[3*i+2] = 0.08  # Initial radius guess
        
    bounds = [(0.0, 1.0)] * 2*n + [(0.0, 0.5)] * n
    
    # 2. Optimization
    res = minimize(
        objective, 
        vars0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints={'type': 'ineq', 'fun': constraint_vec}, 
        options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
    )
    
    # 3. Extract and ensure validity
    vars_opt = res.x
    centers_opt = np.array([[vars_opt[3*i], vars_opt[3*i+1]] for i in range(n)])
    radii_opt = np.array([vars_opt[3*i+2] for i in range(n)])
    
    # Deterministic safety clamp to satisfy validation function strictly
    # Boundaries
    radii_opt = np.minimum(radii_opt, np.minimum(centers_opt[:,0], 1.0 - centers_opt[:,0]))
    radii_opt = np.minimum(radii_opt, np.minimum(centers_opt[:,1], 1.0 - centers_opt[:,1]))
    
    # Pairwise distances
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt((centers_opt[i,0]-centers_opt[j,0])**2 + (centers_opt[i,1]-centers_opt[j,1])**2)
            allowed = dist - radii_opt[j]
            if radii_opt[i] > allowed:
                radii_opt[i] = max(0.0, allowed)
                
    sum_radii = float(np.sum(radii_opt))
    return centers_opt, radii_opt, sum_radii
