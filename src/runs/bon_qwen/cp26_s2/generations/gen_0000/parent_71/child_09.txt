import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Uniform Grid inside valid region
    # We choose a grid that fits within [0.1, 0.9] x [0.1, 0.9]
    # This ensures that with radius 0.09, circles are strictly inside [0,1].
    # We generate a 6x5 grid (30 points) and select the first 26.
    
    grid_x = np.linspace(0.12, 0.88, 6)
    grid_y = np.linspace(0.12, 0.88, 5)
    cx, cy = np.meshgrid(grid_x, grid_y)
    points = np.column_stack((cx.flatten(), cy.flatten()))
    centers_arr = points[:n]
    
    r_init = 0.09
    
    # 2. Optimization Setup
    # Variables vector: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    initial_vars = np.zeros(n * 3)
    for i in range(n):
        initial_vars[3*i] = centers_arr[i, 0]
        initial_vars[3*i+1] = centers_arr[i, 1]
        initial_vars[3*i+2] = r_init
        
    # Objective function: Minimize -sum(r) + Penalties
    def objective(vars_flat):
        vars_2d = vars_flat.reshape(n, 3)
        x = vars_2d[:, 0]
        y = vars_2d[:, 1]
        r = vars_2d[:, 2]
        
        # Maximize sum of radii -> Minimize negative sum
        loss = -np.sum(r)
        penalty = 0.0
        
        # --- Boundary Penalties ---
        # Constraints: r <= x <= 1-r  and  r <= y <= 1-r
        # Equivalent to: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        
        # Violation for x < r: max(0, r - x)
        # Violation for x > 1-r: max(0, x + r - 1)
        # Same for y
        
        # We use a quadratic penalty for violations
        # Coefficients need to be high enough to enforce constraints
        
        # X lower bound violation (x < r)
        v_x_low = r - x
        penalty += 1000 * np.sum(np.maximum(0, v_x_low)**2)
        
        # X upper bound violation (x > 1-r)
        v_x_high = x + r - 1.0
        penalty += 1000 * np.sum(np.maximum(0, v_x_high)**2)
        
        # Y lower bound violation (y < r)
        v_y_low = r - y
        penalty += 1000 * np.sum(np.maximum(0, v_y_low)**2)
        
        # Y upper bound violation (y > 1-r)
        v_y_high = y + r - 1.0
        penalty += 1000 * np.sum(np.maximum(0, v_y_high)**2)
        
        # Non-negative radius
        penalty += 1000 * np.sum(np.maximum(0, -r)**2)
        
        # --- Overlap Penalties ---
        # Constraint: dist(i, j) >= r_i + r_j
        # Violation: max(0, r_i + r_j - dist(i, j))
        
        # Compute pairwise distances
        # C is (n, 2)
        C = vars_2d[:, :2]
        # Diff matrix shape (n, n, 2)
        diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Radius sums matrix (n, n)
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        # We only care about pairs i < j to avoid double counting
        # Create a mask for upper triangle (excluding diagonal)
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        # Overlap amount (positive means overlap)
        overlap = r_sum - dists
        violation = np.maximum(0, overlap) * mask
        
        # High penalty for overlaps to ensure valid packing
        penalty += 5000 * np.sum(violation**2)
        
        return loss + penalty

    # 3. Run Optimization
    # Using Nelder-Mead as it is derivative-free and robust to non-smooth penalty functions.
    # We rely on the penalty function to keep variables within bounds.
    # Note: Nelder-Mead does not support explicit bounds, but the penalty acts as soft bounds.
    
    res = minimize(objective, initial_vars, method='Nelder-Mead', 
                   options={'maxiter': 20000, 'xatol': 1e-6, 'fatol': 1e-6})
    
    final_vars = res.x
    
    # 4. Extract Results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_vars[3*i]
        final_centers[i, 1] = final_vars[3*i+1]
        final_radii[i] = max(0.0, final_vars[3*i+2])
        
    # Post-processing: Ensure strict validity
    # Clip centers to [0, 1] (should be already satisfied by penalty)
    final_centers = np.clip(final_centers, 0.0, 1.0)
    
    # Adjust radii to be strictly inside boundaries if needed
    # Although penalty should have handled it, numerical precision might require this.
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        # Max possible radius at (x,y) inside [0,1]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if r > max_r + 1e-9:
            final_radii[i] = max_r
            
    # Final sum
    total_sum = np.sum(final_radii)
    
    return final_centers, final_radii, total_sum

    penalty = 0.0

    v1 = r - x

    v5 = -r
    penalty += 1000 * np.sum(np.maximum(0, v5)**2)

    # Constraint: dist(i, j) >= r_i + r_j
