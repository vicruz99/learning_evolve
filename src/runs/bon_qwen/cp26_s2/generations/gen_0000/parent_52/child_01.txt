

Looking at this circle packing problem, I need to pack 26 circles in a unit square to maximize the sum of radii (target: 2.636).

<strategy>
My strategy involves:

1. **Initial Configuration**: Start with a hexagonal grid arrangement, which is known to be near-optimal for circle packing. For 26 circles, I'll use a 5×5 grid plus 1 extra circle, arranged in hexagonal rows.

2. **Optimization Approach**: Use scipy.optimize.minimize with SLSQP (Sequential Least Squares Programming) which handles inequality constraints well. The constraints are:
   - Boundary constraints: each circle must stay within [0,1]²
   - Non-overlap constraints: distance between any two circles ≥ sum of their radii

3. **Multiple Restarts**: Run optimization from multiple initial configurations (hexagonal grid, perturbed grids, random) to avoid local minima.

4. **Iterative Refinement**: After initial optimization, perform additional refinement passes to squeeze out more sum of radii.

5. **Key Insight**: The optimal arrangement will likely have unequal circles - some larger near corners/edges where they can "borrow" space from walls, and smaller ones filling gaps. The optimization will discover this automatically.

The objective is to maximize Σrᵢ, which is equivalent to minimizing -Σrᵢ subject to all constraints.

import numpy as np
from scipy.optimize import minimize

def compute_obj_and_cons(params, n=26):
    """Compute objective and constraint values."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    obj = -np.sum(radii)
    
    cons = []
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        # Boundary constraints
        cons.append(x - r)        # x >= r
        cons.append(1 - x - r)    # x <= 1 - r
        cons.append(y - r)        # y >= r
        cons.append(1 - y - r)    # y <= 1 - r
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            cons.append(dist - radii[i] - radii[j])  # dist >= r_i + r_j
    
    return obj, np.array(cons)


def penalty_objective(params, n=26, weight=1e6):
    """Penalty-based objective function for unconstrained optimization."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    obj = -np.sum(radii)
    
    # Boundary penalty
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        if x - r < 0:
            obj += weight * (x - r) ** 2
        if x + r > 1:
            obj += weight * (x + r - 1) ** 2
        if y - r < 0:
            obj += weight * (y - r) ** 2
        if y + r > 1:
            obj += weight * (y + r - 1) ** 2
    
    # Overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                obj += weight * overlap ** 2
    
    return obj


def create_hexagonal_grid(n=26):
    """Create a hexagonal grid initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    
    idx = 0
    spacing_x = 0.18
    spacing_y = spacing_x * np.sqrt(3) / 2
    offset_x = spacing_x / 2
    
    for row in range(8):
        n_cols = 5 if row < 5 else (4 if row == 5 else (3 if row == 6 else 1))
        for col in range(n_cols):
            if idx >= n:
                return centers[:n], radii[:n]
            
            x = 0.1 + col * spacing_x
            y = 0.1 + row * spacing_y
            
            if row % 2 == 1:
                x += offset_x
            
            # Clamp to valid range
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            
            centers[idx] = [x, y]
            radii[idx] = 0.085
            idx += 1
    
    return centers[:n], radii[:n]


def create_random_config(n=26, seed=42):

    np.random.seed(seed)
    centers = np.random.rand(n, 2) * 0.6 + 0.2
    radii = np.full(n, 0.08)
    return centers, radii


def optimize_with_slqp(centers, radii, n=26, max_iter=2000):
    """Optimize using SLSQP with hard constraints."""
    params0 = np.concatenate([centers.flatten(), radii])
    
    # Build constraint list
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*i] - p[2*n+i]})
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[2*i] - p[2*n+i]})
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*i+1] - p[2*n+i]})
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[2*i+1] - p[2*n+i]})
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq', 
                'fun': lambda p, i=i, j=j: np.sqrt(np.sum((p[2*i:2*i+2] - p[2*j:2*j+2]) ** 2)) - p[2*n+i] - p[2*n+j]
            })
    
    def obj(params):
        return -np.sum(params[2*n:])
    
    result = minimize(
        obj, params0,
        method='SLSQP',
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'disp': False}
    )
    
    return result


def optimize_with_penalty(centers, radii, n=26, weight=1e6, max_iter=3000):
    """Optimize using penalty method with BFGS."""
    params0 = np.concatenate([centers.flatten(), radii])
    
    result = minimize(
        penalty_objective, params0,
        args=(n, weight),
        method='BFGS',
        options={'maxiter': max_iter, 'gtol': 1e-12}
    )
    
    return result


def refine_packing(centers, radii, n=26):
    """Refine packing with multiple optimization passes."""
    for iteration in range(3):
        # Small perturbation to escape local minima
        noise = np.random.randn(n, 2) * 0.001
        centers = centers + noise
        centers = np.clip(centers, 0.001, 0.999)
        
        params = np.concatenate([centers.flatten(), radii])
        
        # Gradient-based refinement
        result = minimize(
            penalty_objective, params,
            args=(n, 1e7),
            method='L-BFGS-B',
            bounds=[(0.001, 0.999)] * 2*n + [(0.0001, 0.5)] * n,
            options={'maxiter': 2000, 'ftol': 1e-15}
        )
        
        centers = result.x[:2*n].reshape(n, 2)
        radii = np.maximum(result.x[2*n:], 0.0001)
    
    return centers, radii
