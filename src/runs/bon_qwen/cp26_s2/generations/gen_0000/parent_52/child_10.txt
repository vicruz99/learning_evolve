import numpy as np
from scipy.optimize import linprog

def get_max_radii_and_sum(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize sum(r_i).
    Constraints: r_i >= 0, r_i <= wall_dist_i, r_i + r_j <= dist_ij
    """
    n = centers.shape[0]
    
    # 1. Wall constraints -> upper bounds for r_i
    x = centers[:, 0]
    y = centers[:, 1]
    # Distance to boundaries
    upper_bounds = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    bounds = [(0, ub) for ub in upper_bounds]
    
    # 2. Pairwise constraints: r_i + r_j <= distance(i, j)
    # We construct the matrix A_ub and vector b_ub
    # A_ub * r <= b_ub
    
    pairs = []
    dists = []
    
    # Optimization: Only consider pairs that are "close enough" to potentially constrain radii?
    # For n=26, 325 constraints is small, so we include all.
    # Using list comprehension for speed
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            pairs.append((i, j))
            dists.append(d)
            
    pairs = np.array(pairs)
    dists = np.array(dists)
    
    # Build A_ub matrix (num_pairs x n)
    # Since n is small, dense matrix is fine
    num_pairs = len(pairs)
    A_ub = np.zeros((num_pairs, n))
    for idx, (i, j) in enumerate(pairs):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        
    b_ub = dists
    
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c = -np.ones(n)
    
    try:
        # Using 'highs' method for speed
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            sum_radii = -res.fun
            return radii, sum_radii
        else:
            # Fallback if LP fails (rare)
            # Return valid but small radii
            radii = np.array([min(ub, 0.001) for ub in upper_bounds])
            return radii, np.sum(radii)
    except Exception:
        # Fallback
        radii = np.array([min(ub, 0.001) for ub in upper_bounds])
        return radii, np.sum(radii)
