2.  **Gradient-based Center Optimization:** To find the optimal centers, I will calculate the sensitivity of the total sum of radii to the positions of the centers. I will then use a conjugate gradient method to move the centers in the direction of steepest ascent, effectively pushing circles apart from each other and from the boundaries.
3.  **Restart Strategy:** To escape local optima, the solver will perform multiple restarts with different initial center distributions (uniform random and grid-based) before returning the best valid configuration found.

import numpy as np
from scipy.optimize import linprog, minimize

def get_lp_components(centers):
    """Constructs matrices for the radii optimization LP."""
    n = centers.shape[0]
    # Objective: maximize sum of radii -> minimize -sum
    c_obj = -np.ones(n)
    
    A_ineq = []
    b_ineq = []
    
    # Boundary constraints: 0 <= x-r, x+r <= 1, 0 <= y-r, y+r <= 1
    # Translated to: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        A_ineq.append(row)
        b_ineq.append(centers[i, 0])
        
        A_ineq.append(row)
        b_ineq.append(1 - centers[i, 0])
        
        A_ineq.append(row)
        b_ineq.append(centers[i, 1])
        
        A_ineq.append(row)
        b_ineq.append(1 - centers[i, 1])
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ineq.append(row)
            b_ineq.append(dist)
            
    return c_obj, np.array(A_ineq), np.array(b_ineq)

def solve_radii(centers):
    """Solves for max radii given fixed centers."""
    c_obj, A_ub, b_ub = get_lp_components(centers)
    # Bounds for radii: r >= 0
    bounds = [(0, None) for _ in range(centers.shape[0])]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        radii = -res.x
        return radii, res.ineqlin.marginals
    return None, None

def objective_and_gradient(center_vec, n):
    """Computes negative sum of radii and its gradient w.r.t centers."""
    centers = center_vec.reshape((n, 2))
    radii, duals = solve_radii(centers)
    
    if radii is None:
        return 0, np.zeros_like(center_vec)
        
    total_sum = np.sum(radii)
    grad = np.zeros_like(centers)
    
    # Marginals correspond to constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    # Indices: 0-3n: boundaries, 3n-end: pairwise
    # For boundaries (r <= c_x), dual > 0 means r is limited by boundary.
    # Gradient contribution: +dual to the coordinate pushing the center away from boundary.
    for i in range(n):
        # Boundary constraints indices
        idx_x = 4 * i
        idx_1x = 4 * i + 1
        idx_y = 4 * i + 2
        idx_1y = 4 * i + 3
        
        # If r_i is limited by x_i (r_i <= x_i), we want to increase x_i
        if duals[idx_x] > 1e-12:
            grad[i, 0] += duals[idx_x]
        # If r_i is limited by 1-x_i (r_i <= 1-x_i), we want to decrease x_i
        if duals[idx_1x] > 1e-12:
            grad[i, 0] -= duals[idx_1x]
            
        if duals[idx_y] > 1e-12:
            grad[i, 1] += duals[idx_y]
        if duals[idx_1y] > 1e-12:
            grad[i, 1] -= duals[idx_1y]

    # Pairwise constraints: r_i + r_j <= dist(i, j)
    # Marginals are stored after boundary constraints (4*n)
    k = 4 * n
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dual = duals[k + pair_idx]
            pair_idx += 1
            if dual > 1e-12:
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist > 1e-12:
                    vec = (centers[i] - centers[j]) / dist
                    grad[i] += dual * vec
                    grad[j] -= dual * vec
                    
    return -total_sum, grad.flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    def objective_func(x):
        val, _ = objective_and_gradient(x, n)
        return val
    
    def gradient_func(x):
        _, grad = objective_and_gradient(x, n)
        return grad

    # Generate multiple starting configurations
    inits = []
    
    # 1. Random restarts
    for _ in range(5):
        c = np.random.rand(n, 2) * 0.8 + 0.1
        inits.append(c.flatten())
        
    # 2. Grid-based initializations
    grids = [
        np.linspace(0.15, 0.85, 5),
        np.linspace(0.1, 0.9, 5)
    ]
    
    for g in grids:
        c = np.array([[x, y] for x in g for y in g])
        if len(c) < n:
            # Fill remainder
            extra = np.random.rand(n - len(c), 2) * 0.8 + 0.1
            c = np.vstack([c, extra])
        inits.append(c.flatten())

    for x0 in inits:
        # Conjugate gradient optimization (fmin_cg)
        res = minimize(objective_func, x0, jac=gradient_func, method='L-BFGS-B', 
                       bounds=[(0, 1) for _ in range(2*n)], options={'maxiter': 2000})
        
        if res.success:
            centers_opt = res.x.reshape((n, 2))
            radii_opt, _ = solve_radii(centers_opt)
            if radii_opt is not None:
                s = np.sum(radii_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt
                    best_radii = radii_opt

    return best_centers, best_radii, float(best_sum)
