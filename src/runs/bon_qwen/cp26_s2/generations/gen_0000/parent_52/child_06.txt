import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_initial_hexagonal(n):
    """Generate a hexagonal grid initialization for n circles."""
    centers = []
    r_est = 0.09 # Initial guess for radius
    
    # Hexagonal packing parameters
    row_spacing = r_est * np.sqrt(3)
    col_spacing = 2 * r_est
    
    y = r_est
    row_idx = 0
    while len(centers) < n:
        x = r_est
        offset = (row_idx % 2) * col_spacing / 2
        while x <= 1 - r_est and len(centers) < n:
            centers.append([x + offset, y])
            x += col_spacing
        y += row_spacing
        row_idx += 1
        
    return np.array(centers[:n])

def get_initial_random(n):
    """Generate random initialization."""
    centers = np.random.rand(n, 2)
    return centers

def objective(z, n):
    """Objective function to minimize: -sum(radii)."""
    radii = z[2::3]
    return -np.sum(radii)

def constraint_boundary(z, n):
    """Constraints for circles staying inside the unit square."""
    constraints = []
    for i in range(n):
        x = z[3*i]
        y = z[3*i+1]
        r = z[3*i+2]
        
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[3*i] - z[3*i+2]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[3*i] - z[3*i+2]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[3*i+1] - z[3*i+2]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[3*i+1] - z[3*i+2]})
    return constraints

def constraint_overlap(z, i, j):
    """Constraint for non-overlapping between circle i and j."""
    # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
    # We use the squared form to avoid sqrt, but it's non-convex.
    # Alternatively: dist >= ri + rj.
    # Let's use the distance form as it's often handled better by gradients if careful,
    # but squared is polynomial.
    # However, (ri+rj)^2 - dist^2 <= 0
    xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
    xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
    
    dist_sq = (xi - xj)**2 + (yi - yj)**2
    sum_r = ri + rj
    
    # If dist_sq is very small, we might have issues with gradients if we use sqrt,
    # but here we compare squares.
    # Constraint: dist_sq - sum_r^2 >= 0
    return dist_sq - sum_r**2

def run_optimization(n, initial_centers, seed):
    """Run optimization for a specific initial configuration."""
    np.random.seed(seed)
    
    # Initialize radii to a small valid value based on min distance to boundary
    initial_radii = []
    for c in initial_centers:
        r = min(c[0], 1-c[0], c[1], 1-c[1])
        initial_radii.append(max(r * 0.1, 0.001)) # Start small to be safe
    
    # Combine into single vector: [x0, y0, r0, x1, y1, r1, ...]
    z0 = np.zeros(3 * n)
    for i in range(n):
        z0[3*i] = initial_centers[i, 0]
        z0[3*i+1] = initial_centers[i, 1]
        z0[3*i+2] = initial_radii[i]

    # Bounds for variables
    # x, y in [0, 1], r >= 0
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r (cannot exceed 0.5)

    # Constraints
    cons = []
    
    # Boundary constraints
    for i in range(n):
        cons.extend(constraint_boundary(z0, n)) # Note: lambdas capture i correctly
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            # Define a function for each pair
            cons.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: constraint_overlap(z, i, j)})

    try:
        res = minimize(
            objective, 
            z0, 
            args=(n,),
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9}
        )
        
        if res.success:
            # Extract solution
            centers = np.zeros((n, 2))
            radii = np.zeros(n)
            for i in range(n):
                centers[i] = [res.x[3*i], res.x[3*i+1]]
                radii[i] = res.x[3*i+2]
            return centers, radii, -res.fun
        else:
            return None, None, 0.0
            
    except Exception as e:
        return None, None, 0.0

def jitter_optimization(n, centers, radii, iterations=2000):
    """
    Simple iterative expansion/jitter to improve packing if solver fails or to refine.
    """
    for _ in range(iterations):
        # Pick random circle
        i = np.random.randint(0, n)
        
        # Try to expand radius
        current_r = radii[i]
        max_r = current_r
        
        # Check boundaries
        max_r = min(max_r, centers[i, 0])
        max_r = min(max_r, 1 - centers[i, 0])
        max_r = min(max_r, centers[i, 1])
        max_r = min(max_r, 1 - centers[i, 1])
        
        # Check other circles
        for j in range(n):
            if i == j: continue
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            # Max r_i such that r_i + r_j <= dist => r_i <= dist - r_j
            limit = dist - radii[j]
            if limit < max_r:
                max_r = limit
        
        # Update radius if we can increase it significantly
        if max_r > current_r + 1e-6:
            radii[i] = max_r
        
        # Jitter center slightly to open space
        # Move in random direction
        move_mag = 0.005
        dx = np.random.uniform(-move_mag, move_mag)
        dy = np.random.uniform(-move_mag, move_mag)
        
        new_pos = centers[i] + np.array([dx, dy])
        
        # Check if new position is valid with current radius (and slightly larger)
        # Actually, just check if it allows radius to be at least current
        # Or even simpler: if we can maintain current radius
        
        valid = True
        # Boundary check
        if new_pos[0] < radii[i] or new_pos[0] > 1 - radii[i] or \
           new_pos[1] < radii[i] or new_pos[1] > 1 - radii[i]:
            valid = False
        else:
            # Overlap check
            for j in range(n):
                if i == j: continue
                dist = np.sqrt(np.sum((new_pos - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
        
        if valid:
            centers[i] = new_pos

    return centers, radii
