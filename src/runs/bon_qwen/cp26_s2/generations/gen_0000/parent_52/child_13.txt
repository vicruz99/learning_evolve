import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP optimization with multiple restarts.
    """
    n = 26
    
    # Objective function: minimize negative sum of radii
    def objective(v):
        return -np.sum(v[2*n:])
    
    # Constraint function: returns vector of constraint values (must be >= 0)
    def constraint_func(v):
        x = v[:n]
        y = v[n:2*n]
        r = v[2*n:3*n]
        
        # List to accumulate constraints
        c = []
        
        # 1. Boundary Constraints:
        # x - r >= 0
        c.append(x - r)
        # 1 - x - r >= 0
        c.append(1 - x - r)
        # y - r >= 0
        c.append(y - r)
        # 1 - y - r >= 0
        c.append(1 - y - r)
        
        # 2. Non-overlap Constraints:
        # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
        # Vectorized computation using broadcasting
        # dx shape (n, n)
        dx = x[:, np.newaxis] - x[np.newaxis, :]
        dy = y[:, np.newaxis] - y[np.newaxis, :]
        dist_sq = dx**2 + dy**2
        
        # radii sum matrix
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        r_sum_sq = r_sum**2
        
        # Overlap condition: dist_sq - r_sum_sq >= 0
        # We only care about i != j. 
        # Using a mask to extract upper triangle (excluding diagonal)
        # Mask for upper triangle excluding diagonal
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        overlaps = dist_sq[mask] - r_sum_sq[mask]
        c.append(overlaps)
        
        return np.concatenate(c)

    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    # r <= 0.5 is loose but safe (max diameter 1)
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n

    cons = ({'type': 'ineq', 'fun': constraint_func})

    num_restarts = 15

                    pts.append((x, y))

        r_init = np.full(n, 0.01)
