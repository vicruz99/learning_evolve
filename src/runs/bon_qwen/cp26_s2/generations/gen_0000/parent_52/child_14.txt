import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # --- Helper Functions (Top Level) ---

    def objective(params):
        """Objective function: minimize -r (to maximize sum of radii)"""
        return -params[-1]

    def constraint_overlap(params):
        """Ensures all circles are at least 2r apart."""
        centers = params[:-1].reshape(-1, 2)
        r = params[-1]
        dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
        # Lower triangle to avoid duplicates and self-check
        mask = np.triu(np.ones((26, 26), dtype=bool), k=1)
        return dists[mask] - 2 * r

    def constraint_boundaries(params):
        """Ensures circles stay within [0,1]x[0,1]."""
        centers = params[:-1].reshape(-1, 2)
        r = params[-1]
        lower = centers - r
        upper = 1 - centers - r
        return np.concatenate([lower, upper])

    # --- Initialization ---
    n = 26
    centers = np.zeros((n, 2))
    
    # Initialize in a hexagonal lattice pattern
    row = 0
    count = 0
    # Row sizes: 5, 4, 5, 4, 5, 3 (Total 26)
    # This provides a good starting density for optimization
    row_sizes = [5, 4, 5, 4, 5, 3]
    
    y_step = 1.732 / 7  # Vertical spacing estimate
    x_step = 1.0 / 6    # Horizontal spacing estimate
    
    for size in row_sizes:
        y = 0.15 + row * y_step
        for i in range(size):
            x = 0.1 + i * x_step + (row % 2) * 0.05 # Add shift for hex pattern
            if count < n:
                centers[count, 0] = x
                centers[count, 1] = y
                count += 1
        row += 1

    # Flatten centers and add initial radius guess
    x0 = np.concatenate([centers.flatten(), [0.1]])
    
    # --- Optimization ---
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] # x, y in [0,1], r in [0, 0.5]
    
    constraints = [
        {"type": "ineq", "fun": constraint_overlap},
        {"type": "ineq", "fun": constraint_boundaries}
    ]

    res = minimize(objective, x0, bounds=bounds, constraints=constraints, method='SLSQP', 
                   options={'ftol': 1e-10, 'maxiter': 2000})

    # --- Results ---
    optimal_r = res.x[-1]
    optimal_centers = res.x[:-1].reshape(-1, 2)
    optimal_radii = np.full(n, optimal_r)
    total_sum = np.sum(optimal_radii)

    return optimal_centers, optimal_radii, total_sum
