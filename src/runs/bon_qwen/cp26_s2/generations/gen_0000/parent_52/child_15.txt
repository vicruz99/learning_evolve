    Solves the circle packing problem for 26 circles in a unit square
    to maximize the sum of radii.

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # r <= x, r <= y, x <= 1-r, y <= 1-r
            # Equivalent to: x - r >= 0, y - r >= 0, 1 - x - r >= 0, 1 - y - r >= 0
            cons.append(x - r)
            cons.append(y - r)
            cons.append(1 - x - r)
            cons.append(1 - y - r)
        return cons

    def non_overlap_constraints(params):
        # Returns list of constraint values >= 0
        # dist^2 >= (r_i + r_j)^2
        centers = params[:2 * n].reshape(n, 2)
        radii = params[2 * n:]
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((centers[i] - centers[j]) ** 2)
                r_sum = radii[i] + radii[j]
                cons.append(dist_sq - r_sum**2)
        return cons

    def get_init_guess(seed=None):

            idx_c_j = 2 * j
            idx_r_i = 2 * n + i

                    xj, yj = p[2*j], p[2*j+1]

                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    return dist_sq - (ri + rj)**2
                return con_func
