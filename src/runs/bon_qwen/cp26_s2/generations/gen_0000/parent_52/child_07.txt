import numpy as np

def repulsive_force(c1, c2, r1, r2):
    dx = c1[0] - c2[0]
    dy = c1[1] - c2[1]
    dist = np.sqrt(dx**2 + dy**2)
    if dist < r1 + r2 and dist > 1e-9:
        overlap = (r1 + r2) - dist
        fx = (dx / dist) * overlap
        fy = (dy / dist) * overlap
        return fx, fy
    return 0.0, 0.0

def wall_force(center, radius):
    fx = 0.0
    fy = 0.0
    x, y = center
    # Left wall
    if x < radius:
        fx += (radius - x)
    # Right wall
    if x > 1 - radius:
        fx -= (x - (1 - radius))
    # Bottom wall
    if y < radius:
        fy += (radius - y)
    # Top wall
    if y > 1 - radius:
        fy -= (y - (1 - radius))
    return fx, fy
