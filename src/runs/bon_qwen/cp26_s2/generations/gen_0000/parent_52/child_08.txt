    def solve_radii(centers):
        """
        Solves for max sum of radii given fixed centers.
        Maximize sum(r_i)
        s.t.
        r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        r_i + r_j <= ||c_i - c_j||
        r_i >= 0

        # r_i <= x_i  => 1*r_i <= x_i
        # r_i <= 1-x_i => 1*r_i <= 1-x_i
        # r_i <= y_i  => 1*r_i <= y_i
        # r_i <= 1-y_i => 1*r_i <= 1-y_i
        
        A_ub = []
        b_ub = []
        
        for i in range(n):
            x, y = centers[i]
            # r_i <= x
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(x)
            
            # r_i <= 1-x
            A_ub.append(row)
            b_ub.append(1.0 - x)

            A_ub.append(row)
            b_ub.append(y)

            A_ub.append(row)
            b_ub.append(1.0 - y)
