import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns a packing of 26 circles in a unit square that maximizes the sum of radii.
    """
    n = 26
    
    # Initial configuration: Hexagonal grid
    # We try to fit rows with varying counts to approximate 26 circles
    # A 6-row configuration: 5, 4, 5, 4, 5, 3 (Sum = 26)
    # This is slightly wider than a square grid, but we will optimize.
    
    rows_counts = [5, 4, 5, 4, 5, 3]
    initial_r = 0.08  # Start small to ensure no overlaps
    
    centers = []
    y = initial_r
    for i, count in enumerate(rows_counts):
        # Calculate x spacing for this row
        # Available width for centers is 1 - 2*r, but we approximate for init
        if count > 0:
            # For hex packing, offset odd rows by r
            x_start = initial_r
            if i % 2 == 1:
                x_start += initial_r
            
            # Space available for 'count' circles
            # We place them evenly
            if count == 1:
                x_vals = [0.5]
            else:
                spacing = (1 - 2 * initial_r) / (count - 1)
                x_vals = [x_start + j * spacing for j in range(count)]
        else:
            x_vals = []
            
        for x in x_vals:
            centers.append([x, y])
        
        y += initial_r * np.sqrt(3)
    
    centers = np.array(centers)
    radii = np.full(n, initial_r)
    
    # Optimization Function
    # Variables: [x0, y0, r0, x1, y1, r1, ...]
    # But scipy.optimize works better if we fix the structure.
    # We will optimize centers and radii together.
    
    def objective(vars):
        # We want to maximize sum of radii -> minimize negative sum
        rs = vars[2::3]
        return -np.sum(rs)

    def constraints(vars):
        cs = np.column_stack((vars[0::3], vars[1::3]))
        rs = vars[2::3]
        cons = []
        
        # Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
        # Implemented as: r - x <= 0, x + r - 1 <= 0, etc.
        # scipy expects constraint function >= 0
        # So: x - r >= 0 -> x - r
        
        for i in range(n):
            x, y = cs[i]
            r = rs[i]
            cons.append(x - r)
            cons.append(1 - x - r)
            cons.append(y - r)
            cons.append(1 - y - r)
            
        # Pairwise distance constraints: dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # For numerical stability in optimization, we use a softened constraint 
        # or just check dist - r_i - r_j >= 0.
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((cs[i] - cs[j])**2))
                cons.append(dist - rs[i] - rs[j])
                
        return cons

    # Initial guess
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
    
    # Bounds: x, y in [0, 1], r >= 0
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 1)]) # x, y, r (r upper bound 1)
        
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Use SLSQP for constrained optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        if res.success:
            final_centers = np.column_stack((res.x[0::3], res.x[1::3]))
            final_radii = res.x[2::3]
        else:
            # Fallback to the optimized grid if optimization fails
            # Re-run with different init or return current
            final_centers = centers
            final_radii = radii
            
    except Exception:
        final_centers = centers
        final_radii = radii
        
    # Final Validation and small corrections if needed
    # Ensure radii are non-negative
    final_radii = np.maximum(final_radii, 1e-9)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
