import numpy as np
from scipy.optimize import minimize
import math


def run_packing():
    n = 26

    best_centers = None
    best_radii = None
    best_sum = 0.0

    def evaluate(x, n_local=n):
        centers = x[:2 * n_local].reshape(n_local, 2)
        radii = x[2 * n_local:]

        obj = -np.sum(radii)
        penalty = 0.0
        penalty_weight = 5000.0

        for i in range(n_local):
            x_coord = centers[i, 0]
            y_coord = centers[i, 1]
            r = radii[i]

            if x_coord < r:
                penalty += penalty_weight * (r - x_coord) ** 2
            if x_coord > 1 - r:
                penalty += penalty_weight * (x_coord - (1 - r)) ** 2
            if y_coord < r:
                penalty += penalty_weight * (r - y_coord) ** 2
            if y_coord > 1 - r:
                penalty += penalty_weight * (y_coord - (1 - r)) ** 2

        for i in range(n_local):
            for j in range(i + 1, n_local):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                sum_r = radii[i] + radii[j]
                if dist < sum_r:
                    penalty += penalty_weight * (sum_r - dist) ** 2

        return obj + penalty

    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    def make_grid_initial():
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.09
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n:
                    centers[idx] = [(j + 1) / 6.0, (i + 1) / 6.0]
                    idx += 1
        while idx < n:
            centers[idx] = [0.5, 0.5]
            idx += 1
        return np.concatenate([centers.flatten(), radii])

    def make_hex_initial():
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.09
        idx = 0
        spacing = 0.18
        row_height = spacing * math.sqrt(3) / 2

        for row in range(8):
            y = row_height * row + spacing
            x_start = spacing if row % 2 == 0 else spacing * 1.5

            for col in range(6):
                if idx >= n:
                    break
                x = x_start + col * 2 * spacing
                if x <= 1 - spacing and y <= 1 - spacing:
                    centers[idx] = [x, y]
                    idx += 1

        while idx < n:
            centers[idx] = [0.5, 0.5]
            idx += 1
        return np.concatenate([centers.flatten(), radii])

    def make_corner_initial():
        centers = np.zeros((n, 2))
        radii = np.zeros(n)

        # Place 4 corner circles with larger radii
        corner_r = 0.2
        corners = [[corner_r, corner_r], [1 - corner_r, corner_r],
                    [corner_r, 1 - corner_r], [1 - corner_r, 1 - corner_r]]
        for i, c in enumerate(corners):
            centers[i] = c
            radii[i] = corner_r

        # Fill remaining with grid-like
        idx = 4
        spacing = 0.2
        for i in range(7):
            for j in range(7):
                if idx >= n:
                    break
                x = 0.1 + j * spacing
                y = 0.1 + i * spacing
                if x <= 0.9 and y <= 0.9:
                    # Check if too close to corners
                    too_close = False
                    for ci in range(4):
                        dx = x - centers[ci, 0]
                        dy = y - centers[ci, 1]
                        if math.sqrt(dx*dx + dy*dy) < radii[ci] + 0.12:
                            too_close = True
                            break
                    if not too_close:
                        centers[idx] = [x, y]
                        radii[idx] = 0.08
                        idx += 1

        while idx < n:
            centers[idx] = [0.5, 0.5]
            radii[idx] = 0.08
            idx += 1

        return np.concatenate([centers.flatten(), radii])

    def make_staggered_initial():
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.1
        idx = 0

        # Staggered rows
        for row in range(6):
            y = 0.15 + row * 0.15
            num_in_row = 5 if row % 2 == 0 else 4
            x_start = 0.125 if row % 2 == 0 else 0.25

            for col in range(num_in_row):
                if idx >= n:
                    break
                x = x_start + col * 0.2
                if x <= 0.9:
                    centers[idx] = [x, y]
                    idx += 1

        while idx < n:
            centers[idx] = [0.5, 0.5]
            idx += 1

        return np.concatenate([centers.flatten(), radii])

    initial_configs = [
        make_grid_initial(),
        make_hex_initial(),
        make_corner_initial(),
        make_staggered_initial(),
    ]

    for config_idx, x0_base in enumerate(initial_configs):
        for trial in range(8):
            np.random.seed(config_idx * 100 + trial * 7 + 42)
            x0 = x0_base.copy()

            if trial > 0:
                perturbation = np.random.randn(3 * n) * 0.015
                x0 = x0 + perturbation
                x0 = np.clip(x0, np.array([0] * 2 * n + [0] * n),
                             np.array([1] * 2 * n + [0.5] * n))

            result = minimize(evaluate, x0, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 150000, 'ftol': 1e-15, 'gtol': 1e-12})

            centers = result.x[:2 * n].reshape(n, 2)
            radii = result.x[2 * n:]

            # Post-process: ensure boundary validity
            for i in range(n):
                r = radii[i]
                centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
                centers[i, 1] = max(r, min(1 - r, centers[i, 1]))

            # Post-process: fix overlaps by reducing radii
            for _ in range(200):
                changed = False
                for i in range(n):
                    for j in range(i + 1, n):
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = math.sqrt(dx * dx + dy * dy)
                        if dist < radii[i] + radii[j] - 1e-12:
                            overlap = radii[i] + radii[j] - dist
                            reduction = overlap / 2 + 1e-14
                            radii[i] = max(0, radii[i] - reduction)
                            radii[j] = max(0, radii[j] - reduction)
                            changed = True
                if not changed:
                    break

            # Re-clamp after radius changes
            for i in range(n):
                r = radii[i]
                centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
                centers[i, 1] = max(r, min(1 - r, centers[i, 1]))

            current_sum = np.sum(radii)

            # Verify validity
            valid = True
            for i in range(n):
                x_coord, y_coord = centers[i]
                r = radii[i]
                if x_coord < r - 1e-10 or x_coord > 1 - r + 1e-10:
                    valid = False
                    break
                if y_coord < r - 1e-10 or y_coord > 1 - r + 1e-10:
                    valid = False
                    break
                if not valid:
                    break

            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = math.sqrt(dx * dx + dy * dy)
                        if dist < radii[i] + radii[j] - 1e-10:
                            valid = False
                            break
                    if not valid:
                        break

            if valid and current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()

    return best_centers, best_radii, best_sum
