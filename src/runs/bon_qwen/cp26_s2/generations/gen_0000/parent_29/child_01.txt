import numpy as np
from scipy.optimize import minimize

def get_objective_and_grad(vars, n):
    """
    Computes the objective function and its gradient for the packing problem.
    vars: array of shape (n, 4) -> [r, xi_x, xi_y, dummy]
    Actually we will flatten to 3n variables: r_i, xi_ix, xi_iy
    """
    # vars shape (3n,)
    # Split into r (n,), xi_x (n,), xi_y (n,)
    r = vars[:n]
    xi_x = vars[n:2*n]
    xi_y = vars[2*n:3*n]
    
    # Recover centers
    # x = r + xi_x * (1 - 2r)
    # y = r + xi_y * (1 - 2r)
    factor = 1 - 2 * r
    x = r + xi_x * factor
    y = r + xi_y * factor
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    obj = -np.sum(r)
    
    # Penalty for overlaps
    # Penalty = lambda * sum_{i<j} max(0, (r_i + r_j)^2 - dist^2)^2
    lambda_p = 5000.0
    
    penalty = 0.0
    grad_r = np.zeros(n)
    grad_xi_x = np.zeros(n)
    grad_xi_y = np.zeros(n)
    
    # Compute pairwise distances and penalties
    # Vectorized approach for efficiency
    # r is (n,), x is (n,), y is (n,)
    
    # Broadcast r, x, y
    r_i = r[:, np.newaxis]
    r_j = r[np.newaxis, :]
    x_i = x[:, np.newaxis]
    x_j = x[np.newaxis, :]
    y_i = y[:, np.newaxis]
    y_j = y[np.newaxis, :]
    
    # Mask for upper triangle i < j
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
    sum_r_sq = (r_i + r_j)**2
    
    overlap = sum_r_sq - dist_sq
    violation = np.maximum(0, overlap)
    penalty = lambda_p * np.sum(violation**2)
    
    obj += penalty
    
    # Gradient computation
    # d(obj)/d(r_k) = -1 + d(penalty)/d(r_k)
    # d(obj)/d(xi_k) = d(penalty)/d(xi_k)
    
    # We need to compute gradients of penalty.
    # Let V_k = sum_{l!=k} 2 * violation_{kl} * d(violation_{kl})/d(param_k)
    # violation_{kl} = max(0, (r_k+r_l)^2 - dist_{kl}^2)
    
    # Gradient of overlap term w.r.t variables
    # Let D_{kl} = (r_k+r_l)^2 - ((x_k-x_l)^2 + (y_k-y_l)^2)
    # If D_{kl} > 0, term is 2 * D_{kl} * lambda_p * (derivative of D_{kl})
    
    # Derivatives of D_{kl}
    # dD/dr_k = 2(r_k + r_l) - 2(x_k - x_l) * dx_k/dr_k - 2(y_k - y_l) * dy_k/dr_k
    # dD/dxi_k = - 2(x_k - x_l) * dx_k/dxi_k - 2(y_k - y_l) * dy_k/dxi_k
    
    # Derivatives of coordinates w.r.t parameters:
    # x_k = r_k + xi_k * (1 - 2r_k)
    # dx_k/dr_k = 1 + xi_k * (-2) = 1 - 2*xi_k
    # dx_k/dxi_k = 1 - 2*r_k
    
    # Similarly for y
    
    # Precompute derivatives of coords
    dx_dr = 1 - 2 * xi_x
    dy_dr = 1 - 2 * xi_y
    dx_dxi = 1 - 2 * r
    dy_dxi = 1 - 2 * r
    
    # We can compute gradients using matrix operations
    # Gradient w.r.t r_k
    # Sum over l (l != k) of 2 * lambda_p * violation_{kl} * dD_{kl}/dr_k
    
    # Let's compute dD_{kl}/dr_k, dD_{kl}/dr_l etc.
    # dD_{kl}/dr_k = 2(r_k + r_l) - 2(x_k - x_l)(1 - 2*xi_kx) - 2(y_k - y_l)(1 - 2*xi_ky)
    # dD_{kl}/dr_l = 2(r_k + r_l) - 2(x_l - x_k)(1 - 2*xi_lx) - 2(y_l - y_k)(1 - 2*xi_ly)
    # Note: x_l - x_k = -(x_k - x_l)
    
    # Gradient w.r.t xi_kx
    # dD_{kl}/dxi_kx = - 2(x_k - x_l)(1 - 2*r_k)
    # dD_{kl}/dxi_ky = - 2(y_k - y_l)(1 - 2*r_k)
    
    # Initialize gradient arrays
    grad_r_penalty = np.zeros(n)
    grad_xi_x_penalty = np.zeros(n)
    grad_xi_y_penalty = np.zeros(n)
    
    # Compute components
    # x_diff[k, l] = x_k - x_l
    x_diff = x_i - x_j
    y_diff = y_i - y_j
    
    # Coefficients for derivatives
    # For r_k:
    # term = 2(r_k+r_l) - 2 x_diff (1-2 xi_kx) - 2 y_diff (1-2 xi_ky)
    # Note: x_diff[k,l] is used for k, -x_diff[k,l] for l
    
    # Let's compute the "force" contributions
    # F = 2 * lambda_p * violation
    
    # Reshape violation for broadcasting
    # violation is (n, n)
    
    # Coefficients for dD/dr_k
    # A = 2(r_k + r_l)
    A = 2 * (r_i + r_j)
    # B = 2 x_diff
    B = 2 * x_diff
    # C = 2 y_diff
    C = 2 * y_diff
    
    # Derivatives of coords
    # dx_k/dr_k = 1 - 2 xi_kx
    dx_k_dr = 1 - 2 * xi_x[:, np.newaxis]
    dy_k_dr = 1 - 2 * xi_y[:, np.newaxis]
    dx_l_dr = 1 - 2 * xi_x[np.newaxis, :]
    dy_l_dr = 1 - 2 * xi_y[np.newaxis, :]
    
    # dD/dr_k = A - B * dx_k_dr - C * dy_k_dr
    # But B, C depend on k,l. B[k,l] = 2(x_k - x_l).
    # So for pair (k,l), contribution to grad_r[k] is F[k,l] * (A[k,l] - B[k,l]*dx_k_dr[k,k] - C[k,l]*dy_k_dr[k,k])
    # Contribution to grad_r[l] is F[k,l] * (A[k,l] - B[k,l]*dx_l_dr[l,l] - C[k,l]*dy_l_dr[l,l]) 
    # Wait, dD/dr_l involves derivatives of x_l, y_l.
    # x_l - x_k = - (x_k - x_l) = -x_diff[k,l].
    # So dD/dr_l = 2(r_k+r_l) - 2(x_l-x_k) dx_l/dr_l - ...
    # = A - (-B) dx_l_dr - (-C) dy_l_dr = A + B dx_l_dr + C dy_l_dr
    
    # Let's compute these terms
    
    # Term for k
    dD_dr_k = A - B * dx_k_dr - C * dy_k_dr
    
    # Term for l
    # x_l - x_k = -B/2. So 2(x_l-x_k) = -B.
    dD_dr_l = A + B * dx_l_dr + C * dy_l_dr
    
    # Multiply by violation and lambda
    # Only consider upper triangle
    # grad_r_penalty[k] += sum_l (F[k,l] * dD_dr_k[k,l])
    
    grad_r_penalty += np.sum(F * dD_dr_k * mask, axis=1)
    grad_r_penalty += np.sum(F * dD_dr_l * mask, axis=0) # axis 0 sums over k for fixed l
    
    # Gradient w.r.t xi_kx
    # dD/dxi_kx = - 2(x_k - x_l) * (1 - 2*r_k) = -B * dx_k_dxi_k
    # dx_k/dxi_k = 1 - 2*r_k
    dx_k_dxi = 1 - 2 * r[:, np.newaxis]
    dy_k_dxi = 1 - 2 * r[:, np.newaxis] # Same for y
    
    dD_dxi_x_k = -B * dx_k_dxi
    dD_dxi_y_k = -C * dy_k_dxi
    
    # For l:
    # x_l - x_k = -B/2. 2(x_l-x_k) = -B.
    # dD/dxi_lx = - 2(x_l - x_k) (1 - 2*r_l) = -(-B) dx_l_dxi_l = B * dx_l_dxi_l
    dx_l_dxi = 1 - 2 * r[np.newaxis, :]
    
    dD_dxi_x_l = B * dx_l_dxi
    dD_dxi_y_l = C * dx_l_dxi # Wait, y term uses C? No.
    # dD/dxi_ly = - 2(y_l - y_k) (1 - 2*r_l) = -(-C) dy_l_dxi_l = C * dy_l_dxi_l
    
    grad_xi_x_penalty += np.sum(F * dD_dxi_x_k * mask, axis=1)
    grad_xi_x_penalty += np.sum(F * dD_dxi_x_l * mask, axis=0)
    
    grad_xi_y_penalty += np.sum(F * dD_dxi_y_k * mask, axis=1)
    grad_xi_y_penalty += np.sum(F * dD_dxi_y_l * mask, axis=0)
    
    # Total gradients
    grad_r = -np.ones(n) + 2 * lambda_p * grad_r_penalty
    grad_xi_x = 2 * lambda_p * grad_xi_x_penalty
    grad_xi_y = 2 * lambda_p * grad_xi_y_penalty
    
    grad = np.concatenate([grad_r, grad_xi_x, grad_xi_y])
    
    return obj, grad

def run_packing():
    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None
    
    # Number of restarts
    restarts = 20
    
    for _ in range(restarts):
        # Random initialization
        r_init = np.random.uniform(0.05, 0.15, n)
        xi_x_init = np.random.uniform(0, 1, n)
        xi_y_init = np.random.uniform(0, 1, n)
        
        vars_init = np.concatenate([r_init, xi_x_init, xi_y_init])
        
        # Bounds: r in [0, 0.5], xi in [0, 1]
        bounds = []
        for _ in range(n):
            bounds.append((0, 0.5)) # r
            bounds.append((0, 1))   # xi_x
            bounds.append((0, 1))   # xi_y
            
        # Optimization
        # We define a wrapper to pass n
        def objective_wrapper(vars):
            return get_objective_and_grad(vars, n)[0]
        
        def gradient_wrapper(vars):
            return get_objective_and_grad(vars, n)[1]
            
        try:
            res = minimize(objective_wrapper, vars_init, jac=gradient_wrapper, 
                           method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success or res.fun < 0: # fun is -sum_r + penalty, so negative means we found something
                # Extract parameters
                r_opt = res.x[:n]
                xi_x_opt = res.x[n:2*n]
                xi_y_opt = res.x[2*n:3*n]
                
                # Calculate centers
                factor = 1 - 2 * r_opt
                x_opt = r_opt + xi_x_opt * factor
                y_opt = r_opt + xi_y_opt * factor
                
                # Check validity roughly (penalty should be small)
                # Sum of radii
                sum_r = np.sum(r_opt)
                
                if sum_r > best_sum_r:
                    # Validate fully
                    centers = np.column_stack((x_opt, y_opt))
                    radii = r_opt
                    
                    # Quick validation check (same as provided function logic)
                    valid = True
                    # Boundary
                    if np.any(radii < -1e-12) or np.any(radii > 0.5 + 1e-12): # Radii bounds
                         valid = False
                    if np.any(centers - radii[:, np.newaxis] < -1e-12) or \
                       np.any(centers + radii[:, np.newaxis] > 1 + 1e-12):
                         valid = False
                    
                    if valid:
                        best_sum_r = sum_r
                        best_centers = centers
                        best_radii = radii
                        
        except Exception:
            continue
            
    return best_centers, best_radii, best_sum_r
