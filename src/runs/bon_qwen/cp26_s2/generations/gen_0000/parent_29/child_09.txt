import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed center positions.
    Returns (max_sum_radii, radii_array).
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Preallocate constraint matrices
    num_pairwise = n * (n - 1) // 2
    num_boundary = 4 * n
    A_ub = np.zeros((num_boundary + num_pairwise, n))
    b_ub = np.zeros(num_boundary + num_pairwise)
    
    idx = 0
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for lim in (x, 1.0 - x, y, 1.0 - y):
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(0.0, lim)
            idx += 1
            
    # Pairwise constraints: r_i + r_j <= distance(c_i, c_j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None) for _ in range(n)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, -res.x
    return 0.0, np.zeros(n)

def run_packing():
    n = 26
    np.random.seed(42)
    
    best_centers = None
    best_radii = None
    best_val = 0.0
    
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal packing arrangement (dense baseline)
    centers_hex = np.zeros((n, 2))
    row_counts = [6, 5, 6, 5, 4]
    k = 0
    for r_idx, cnt in enumerate(row_counts):
        y = 0.12 + r_idx * 0.18
        x_start = (1.0 - (cnt - 1) * 0.18) / 2.0
        for c_idx in range(cnt):
            x = x_start + c_idx * 0.18
            if r_idx % 2 == 1:
                x += 0.09
            centers_hex[k] = [x, y]
            k += 1
    configs.append(centers_hex.copy())
    
    # 2. Perturbed hexagonal
    configs.append(centers_hex + np.random.normal(0, 0.02, centers_hex.shape))
    
    # 3. Uniform random
    configs.append(np.random.uniform(0.1, 0.9, (n, 2)))
    
    # Optimize each configuration
    for init_c in configs:
        curr_c = np.clip(init_c, 0.05, 0.95).copy()
        curr_sum, curr_radii = solve_radii(curr_c)
        
        step = 0.06
        for _ in range(2500):
            improved = False
            for i in range(n):
                # Try deterministic and random moves
                directions = [
                    [step, 0], [-step, 0], [0, step], [0, -step],
                    [np.random.normal(0, step), np.random.normal(0, step)]
                ]
                for dx, dy in directions:
                    trial_c = curr_c.copy()
                    trial_c[i, 0] = np.clip(trial_c[i, 0] + dx, 1e-3, 1.0 - 1e-3)
                    trial_c[i, 1] = np.clip(trial_c[i, 1] + dy, 1e-3, 1.0 - 1e-3)
                    
                    s, _ = solve_radii(trial_c)
                    if s > curr_sum + 1e-8:
                        curr_c = trial_c
                        curr_sum = s
                        improved = True
                if improved:
                    break
                    
            if not improved:
                step *= 0.85
                if step < 1e-5:
                    break
        
        if curr_sum > best_val:
            best_val = curr_sum
            best_centers = curr_c.copy()
            best_radii = curr_radii.copy()
            
    # Final exact solve to ensure consistency
    final_sum, final_radii = solve_radii(best_centers)
    return best_centers, final_radii, float(final_sum)
