import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_obj(vars, penalty):
    """Objective function: maximize sum of radii with penalty for constraint violations."""
    c = vars[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    r = vars[2*N_CIRCLES:]
    
    # Boundary violations: circles must stay within [0,1]
    v_b = np.maximum(0, r - c[:,0]) + np.maximum(0, r - (1.0 - c[:,0]))
    v_b += np.maximum(0, r - c[:,1]) + np.maximum(0, r - (1.0 - c[:,1]))
    
    # Overlap violations: distance between centers must be >= sum of radii
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Ignore self-distance
    dists[np.arange(N_CIRCLES), np.arange(N_CIRCLES)] = 1e9
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    v_o = np.maximum(0, r_sum - dists)
    
    total_viol = np.sum(v_b**2) + np.sum(v_o**2)
    return -np.sum(r) + penalty * total_viol

def get_initial_guess(seed):
    """Generate a staggered hexagonal-like initial layout."""
    np.random.seed(seed)
    c = np.zeros((N_CIRCLES, 2))
    r = np.ones(N_CIRCLES) * 0.07
    
    idx = 0
    y = 0.12
    for row in range(5):
        # Alternate row sizes to approximate hexagonal packing
        count = 6 if row % 2 == 0 else 5
        if idx + count > N_CIRCLES:
            count = N_CIRCLES - idx
            
        x_start = 0.1
        step = (0.8) / (count - 1) if count > 1 else 0.0
        shift = 0.04 if row % 2 == 1 else 0.0
        
        for k in range(count):
            c[idx] = [x_start + shift + k*step, y]
            idx += 1
        y += 0.22
        
    # Add small perturbations to break symmetry and help optimization
    c += np.random.uniform(-0.01, 0.01, c.shape)
    c = np.clip(c, 0.02, 0.98)
    return c, r

def run_packing():
    """Optimize circle packing to maximize sum of radii."""
    bounds = [(0.0, 1.0)] * (2*N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    best_c, best_r, best_sum = None, None, -1.0
    
    # Multiple starts to avoid local minima
    seeds = [42, 137, 256, 999]
    # Continuation: gradually increase penalty to enforce constraints strictly
    penalties = [20, 100, 500, 2000]
    
    for seed in seeds:
        c_init, r_init = get_initial_guess(seed)
        x = np.concatenate([c_init.flatten(), r_init])
        
        for pen in penalties:
            res = minimize(compute_obj, x, args=(pen,), method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-13, 'disp': False})
            x = res.x
            
        c_opt = x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
        r_opt = x[2*N_CIRCLES:]
        
        # Projection step: guarantee strict feasibility for validation
        valid_r = np.zeros(N_CIRCLES)
        for i in range(N_CIRCLES):
            # Max radius limited by boundaries
            lim = min(c_opt[i,0], 1.0 - c_opt[i,0], c_opt[i,1], 1.0 - c_opt[i,1])
            # Max radius limited by other circles
            for j in range(N_CIRCLES):
                if i != j:
                    d = np.sqrt(np.sum((c_opt[i] - c_opt[j])**2))
                    lim = min(lim, d / 2.0)
            valid_r[i] = max(0.0, lim - 1e-11)
            
        s = np.sum(valid_r)
        if s > best_sum:
            best_sum = s
            best_c = c_opt
            best_r = valid_r
            
    return best_c, best_r, best_sum
