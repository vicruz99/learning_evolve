def get_overlap_energy(centers, radii):
    """
    Calculates the sum of squared overlaps between all pairs of circles.
    Returns 0 if no overlaps.
    """
    n = len(centers)
    energy = 0.0
    # Using double loop for clarity, can be vectorized but n=26 is small
    for i in range(n):
        r_i = radii[i]
        c_i = centers[i]
        for j in range(i + 1, n):
            r_j = radii[j]
            c_j = centers[j]
            dist_sq = np.sum((c_i - c_j)**2)
            dist = np.sqrt(dist_sq)
            sum_r = r_i + r_j
            if dist < sum_r:
                overlap = sum_r - dist
                energy += overlap**2
    return energy
