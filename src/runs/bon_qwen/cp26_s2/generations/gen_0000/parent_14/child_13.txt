import numpy as np
from scipy.optimize import linprog

def run_packing():

    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    np.random.seed(42)
    
    # 1. Initialize centers
    # Start with a 5x5 grid (25 circles) and place the 26th in a gap
    grid_size = 5
    step = 1.0 / (grid_size + 1) # 1/6 approx 0.166, but we want tight packing
    # Actually for 5x5 grid of radius 0.1, centers are at 0.1, 0.3, 0.5, 0.7, 0.9
    coords = np.linspace(0.1, 0.9, 5)
    centers = []
    for y in coords:
        for x in coords:
            centers.append([x, y])
    centers = np.array(centers)
    
    # Add 26th circle in a gap, e.g., at (0.2, 0.2) which is between (0.1,0.1) and (0.3,0.3) etc.
    # Actually (0.2, 0.2) is center of gap formed by (0.1,0.1), (0.3,0.1), (0.1,0.3), (0.3,0.3)
    # But we need to fit it. Let's place it at (0.2, 0.2) initially.
    # To avoid exact overlap or singularity, add tiny noise
    extra_center = np.array([0.2, 0.2]) + np.random.uniform(-0.001, 0.001, 2)
    centers = np.vstack([centers, extra_center])
    
    # 2. Function to solve LP for radii given centers
    def solve_radii(centers):
        n_c = centers.shape[0]
        # LP Variables: r_0, ..., r_{n-1}
        # Maximize sum(r_i) <=> Minimize -sum(r_i)
        c_obj = -np.ones(n_c)
        
        # Constraints:
        # 1. r_i <= x_i
        # 2. r_i <= 1 - x_i
        # 3. r_i <= y_i
        # 4. r_i <= 1 - y_i
        # 5. r_i + r_j <= dist(i, j) for all i < j
        
        # Build A_ub and b_ub
        # We will collect all linear constraints: A_ub @ r <= b_ub
        
        constraints_rows = []
        constraints_vals = []
        
        # Boundary constraints
        for i in range(n_c):
            x, y = centers[i]
            # r_i <= x
            row = np.zeros(n_c); row[i] = 1.0; constraints_rows.append(row); constraints_vals.append(x)
            # r_i <= 1 - x
            row = np.zeros(n_c); row[i] = 1.0; constraints_rows.append(row); constraints_vals.append(1.0 - x)
            # r_i <= y
            row = np.zeros(n_c); row[i] = 1.0; constraints_rows.append(row); constraints_vals.append(y)
            # r_i <= 1 - y
            row = np.zeros(n_c); row[i] = 1.0; constraints_rows.append(row); constraints_vals.append(1.0 - y)
            
        # Distance constraints
        for i in range(n_c):
            for j in range(i + 1, n_c):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                row = np.zeros(n_c)
                row[i] = 1.0
                row[j] = 1.0
                constraints_rows.append(row)
                constraints_vals.append(dist)
                
        A_ub = np.array(constraints_rows)
        b_ub = np.array(constraints_vals)
        
        # Bounds for r_i: 0 <= r_i
        bounds = [(0, None) for _ in range(n_c)]
        
        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            return radii, -res.fun
        else:
            # Fallback if LP fails (shouldn't happen for feasible centers)
            return np.zeros(n_c), 0.0

                best_radii = radii.copy()

    return centers_out, radii_out, float(final_sum)
