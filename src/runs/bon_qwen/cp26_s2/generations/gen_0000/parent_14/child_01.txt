def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Arrange circles in a grid-like structure to start
    # 6 columns x 5 rows = 30 slots, we need 26

    cols = 6
    rows = 5

    centers = np.array([[xi, yi] for yi in y for xi in x])

    # Remove the last 4
    centers = centers[:n]

    radii = np.full(n, 0.01)

    # Using a simple physics simulation approach

    repulsion_strength = 10.0
    damping = 0.9
    
    velocities = np.zeros_like(centers)

                    forces[i] += force

        centers = np.clip(centers, 0.0, 1.0)
