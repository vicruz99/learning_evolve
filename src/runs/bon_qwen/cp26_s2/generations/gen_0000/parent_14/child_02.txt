import numpy as np
import math
import random

def compute_radii(centers):
    """
    Computes the maximum valid radii for a given set of centers.
    Radii are determined by the distance to the nearest neighbor or boundary.
    """
    n = centers.shape[0]
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        # Distance to boundaries
        r = min(x, 1 - x, y, 1 - y)
        
        # Distance to other circles
        for j in range(n):
            if i == j:
                continue
            dist = math.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            # The radius is limited by half the distance to the center of circle j
            # because if r_i + r_j <= dist, and we maximize symmetrically or locally
            # the tightest constraint for r_i given r_j is unknown is r_i <= dist/2
            # (assuming r_j could be 0). 
            # However, to be strictly safe with the derived radii logic:
            # If we set r_i = min(dist_ij / 2), then r_i + r_j <= dist_ij/2 + dist_ji/2 = dist_ij.
            # This holds for all pairs.
            if dist / 2.0 < r:
                r = dist / 2.0
        
        radii[i] = r
    return radii

def get_score(centers):
    """Calculates the sum of radii for the current centers."""
    radii = compute_radii(centers)
    return np.sum(radii), radii

def run_packing() -> tuple:
    """
    Runs the optimization to pack 26 circles.
    Returns (centers, radii, sum_radii).
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal-like grid perturbed
    # We try to fit roughly 5-6 circles per row
    centers = np.zeros((n_circles, 2))
    idx = 0
    
    # Create a rough grid
    # 5 rows, alternating counts
    # Rows: 6, 5, 6, 5, 4 = 26
    row_counts = [6, 5, 6, 5, 4]
    
    y_spacing = 0.9 / (len(row_counts) - 1) # Spread vertically
    y_curr = 0.05
    
    for count in row_counts:
        # x spacing
        x_spacing = 0.9 / count if count > 1 else 0
        x_start = (1.0 - (count - 1) * x_spacing) / 2 if count > 1 else 0.5
        
        for k in range(count):
            x = x_start + k * x_spacing
            # Add small random perturbation to avoid symmetry traps
            x += (random.random() - 0.5) * 0.05
            y = y_curr + (random.random() - 0.5) * 0.05
            
            # Clamp strictly inside
            x = max(0.001, min(0.999, x))
            y = max(0.001, min(0.999, y))
            
            centers[idx] = [x, y]
            idx += 1
        
        y_curr += y_spacing

    # 2. Optimization: Simulated Annealing / Local Search
    current_centers = centers.copy()
    best_centers = current_centers.copy()
    
    current_score, current_radii = get_score(current_centers)
    best_score = current_score
    
    temperature = 0.1 # Initial "step size" for perturbations
    min_temperature = 1e-6
    decay = 0.995
    
    # Initial optimization run
    iterations = 20000 # Number of steps
    
    for step in range(iterations):
        # Pick a random circle to move
        i = random.randint(0, n_circles - 1)
        
        # Propose a new position
        # The step size should be relative to current radius to allow fine tuning
        # Or just a small fixed fraction of the box
        step_size = temperature * 0.1 
        
        dx = (random.random() - 0.5) * step_size
        dy = (random.random() - 0.5) * step_size
        
        new_x = current_centers[i, 0] + dx
        new_y = current_centers[i, 1] + dy
        
        # Keep inside bounds (strictly inside to allow radius > 0)
        new_x = max(1e-5, min(1.0 - 1e-5, new_x))
        new_y = max(1e-5, min(1.0 - 1e-5, new_y))
        
        # Apply move
        old_pos = current_centers[i].copy()
        current_centers[i] = [new_x, new_y]
        
        new_score, new_radii = get_score(current_centers)
        
        delta = new_score - current_score
        
        # Acceptance criterion
        if delta > 0 or random.random() < math.exp(delta / max(temperature, 1e-9)):

            if current_score > best_score:
                best_score = current_score
                best_centers = current_centers.copy()
                # best_radii = current_radii # Recompute later
        else:
            # Revert
            current_centers[i] = old_pos
            
        # Cool down
        temperature *= decay

    # Final check and radius computation
    final_centers = best_centers
    final_radii = compute_radii(final_centers)
    final_sum = np.sum(final_radii)

def _fallback_packing(n):
    """Fallback simple grid packing if optimization fails."""
    centers = np.zeros((n, 2))
    k = int(math.ceil(math.sqrt(n)))
    step = 1.0 / (k + 1)
    idx = 0
    for r in range(k):
        for c in range(k):
            if idx < n:
                centers[idx] = [step * (c + 1), step * (r + 1)]
                idx += 1
    radii = compute_radii(centers)
    return centers, radii, np.sum(radii)
