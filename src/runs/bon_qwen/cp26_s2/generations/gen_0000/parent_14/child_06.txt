import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_hex_layout(n_circles):
    """
    Generates a hexagonal layout for n_circles.
    Tries to distribute circles into rows to maximize minimum radius.
    """
    # We want to minimize the max number of circles in a row to reduce width constraint,
    # but minimize number of rows to reduce height constraint.
    # sqrt(26) ~ 5.1.
    # Try 5 rows: 6, 5, 5, 5, 5 (sum 26) -> max width 6 circles.
    # Try 6 rows: 5, 5, 5, 5, 4, 2? No, balanced is better.
    # 26 / 6 = 4.33. Maybe 5, 5, 5, 5, 4, 2 is unbalanced.
    # Let's try 5 rows with counts [6, 5, 5, 5, 5]
    
    row_counts = [6, 5, 5, 5, 5]
    if sum(row_counts) != n_circles:
        # Fallback logic if n changes, though fixed at 26 here
        row_counts = []
        rows = int(np.ceil(np.sqrt(n_circles)))
        count = n_circles
        for i in range(rows):
            c = count // (rows - i) + (1 if count % (rows - i) != 0 else 0)
            row_counts.append(min(c, 7)) # cap columns
            count -= c
            if count == 0: break
    
    points = []
    # Assume a radius r=1 for layout generation, scale later
    r = 1.0
    dy = r * np.sqrt(3)
    
    current_y = 0
    for i, count in enumerate(row_counts):
        # Shift odd rows
        if i % 2 == 1:
            dx_offset = r
        else:
            dx_offset = 0
        
        # Generate points for this row
        # To center the row, we need to know the width.
        # Width of row with k circles is (k-1)*2r + 2r = 2kr.
        # But we just place them relative to a start.
        # Let's just place them starting at x=0, then center the whole blob later.
        
        for j in range(count):
            x = j * 2 * r + dx_offset
            y = current_y
            points.append([x, y])
        
        current_y += dy
        
    # Center the points in [0, 1] x [0, 1] roughly
    # Actually, let's just return relative positions, optimizer will scale.
    # But we need a valid initial radius.
    # Let's compute bounding box of this shape and scale to fit in [0,1]x[0,1] with some margin.
    
    pts = np.array(points)
    min_x, min_y = np.min(pts[:, 0]), np.min(pts[:, 1])
    max_x, max_y = np.max(pts[:, 0]), np.max(pts[:, 1])
    
    width = max_x - min_x
    height = max_y - min_y
    
    # We want to fit this in 1x1.
    # Current radius is 1.
    # Required radius r_fit such that bounding box + 2*r_fit <= 1?
    # Actually, the points are centers.
    # If we scale everything by s, new radius is s*1 = s.
    # New width is s*width.
    # We need s*width + 2*s <= 1 => s(width+2) <= 1 => s <= 1/(width+2).
    # Same for height: s <= 1/(height+2).
    
    s = 1.0 / (max(width, height) + 2)
    
    # Shift to center
    center_x = (max_x + min_x) / 2
    center_y = (max_y + min_y) / 2
    
    # Normalize to [0, 1] box
    # Scale and shift
    pts_scaled = (pts - np.array([center_x, center_y])) * s + 0.5
    
    # Radius
    r_init = s
    
    return pts_scaled, r_init

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Generate initial layout
    centers_init, r_init = get_hex_layout(n)
    
    # Flatten centers to 1D array
    # Order: x1, y1, r1, x2, y2, r2, ...
    # Total variables: 3 * n
    vars_init = np.zeros(3 * n)
    for i in range(n):
        vars_init[3*i] = centers_init[i, 0]
        vars_init[3*i+1] = centers_init[i, 1]
        vars_init[3*i+2] = r_init
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    
    # Optimization Objective: Minimize -sum(radii)
    def objective(vars):
        radii = vars[2::3]
        return -np.sum(radii)
    
    # Constraints
    # 1. Boundary: x - r >= 0, 1 - x - r >= 0, etc.
    # 2. Overlap: dist^2 - (r_i + r_j)^2 >= 0
    
    def constraints(vars):
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        cons = []
        
        # Boundary constraints
        # x >= r  => x - r >= 0
        cons.append(x - r)
        # x <= 1 - r => 1 - x - r >= 0
        cons.append(1.0 - x - r)
        # y >= r => y - r >= 0
        cons.append(y - r)
        # y <= 1 - r => 1 - y - r >= 0
        cons.append(1.0 - y - r)
        
        # Overlap constraints
        # Only check pairs i < j
        # Vectorized approach
        # x, y, r are arrays of size n
        # Create upper triangular matrix of indices
        idx_i, idx_j = np.triu_indices(n, k=1)
        
        # Calculate squared distances
        dx = x[idx_i] - x[idx_j]
        dy = y[idx_i] - y[idx_j]
        dist_sq = dx**2 + dy**2
        
        # Calculate squared sum of radii
        r_sum = r[idx_i] + r[idx_j]
        r_sum_sq = r_sum**2
        
        cons.append(dist_sq - r_sum_sq)
        
        return np.concatenate(cons)
    
    constraint = {
        'type': 'ineq',
        'fun': constraints
    }
    
    # Run optimization
    # Use SLSQP as it handles constraints well
    try:
        res = minimize(
            objective, 
            vars_init, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=[constraint],
            options={'maxiter': 1000, 'ftol': 1e-12}
        )
        
        # Extract results
        final_vars = res.x
        final_centers = np.column_stack((final_vars[0::3], final_vars[1::3]))
        final_radii = final_vars[2::3]
        
        # Ensure non-negative radii (solver might drift slightly below 0 due to numerical issues, though bounds prevent it)
        final_radii = np.maximum(final_radii, 0.0)
        
        # Validate
        if validate_packing(final_centers, final_radii):
            return final_centers, final_radii, np.sum(final_radii)
        else:
            # If validation fails, try to recover or return initial valid config
            # This might happen if solver finds a point on boundary that fails strict check?
            # Usually solver respects constraints.
            # Let's try to project radii slightly smaller if needed?
            # But let's trust the solver first.
            pass
            
    except Exception as e:
        print(f"Optimization failed: {e}")
    
    # Fallback: return the initial valid layout if optimization fails or validation fails
    # We need to ensure the initial layout is valid.
    # The generated layout might have overlaps if r_init is too large.
    # Let's compute a valid r for the initial centers.
    
    # Compute pairwise distances and boundary distances
    # Simple fallback: set radii to min( boundary_dist, min_overlap_dist / 2 )
    # But that's complex.
    # Instead, let's just scale down the initial layout to be safe.
    # The r_init calculated was based on bounding box, should be valid.
    # But let's verify.
    
    if not validate_packing(centers_init, np.ones(n)*r_init):
        # If initial is invalid, reduce radius drastically
        r_safe = 0.01
        # Random positions
        np.random.seed(42)
        centers_fallback = np.random.rand(n, 2)
        radii_fallback = np.ones(n) * r_safe
        # This is a valid packing (tiny circles don't overlap much if sparse, but might overlap if random)
        # Better: grid
        k = int(np.ceil(np.sqrt(n)))
        grid = np.meshgrid(np.linspace(0.05, 0.95, k), np.linspace(0.05, 0.95, k))
        pts = np.column_stack((grid[0].flatten(), grid[1].flatten()))[:n]
        radii_fallback = np.ones(n) * 0.04
        if validate_packing(pts, radii_fallback):
            return pts, radii_fallback, np.sum(radii_fallback)
            
    return centers_init, np.ones(n)*r_init, n*r_init
