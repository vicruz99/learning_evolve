import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n_circles = 26
    penalty_weight = 1000.0
    
    def objective(params):
        """Objective function: negative sum of radii + penalty for constraints."""
        centers = params[:2 * n_circles].reshape(-1, 2)
        radii = params[2 * n_circles:]
        
        # 1. Objective term: Maximize sum of radii (minimize negative sum)
        val = -np.sum(radii)
        
        # 2. Boundary Penalty
        # Circles must be within [0, 1] x [0, 1]
        # x - r >= 0  =>  r - x <= 0
        # x + r <= 1  =>  x + r - 1 <= 0
        val += penalty_weight * np.sum(np.maximum(0, radii - centers[:, 0])**2)
        val += penalty_weight * np.sum(np.maximum(0, centers[:, 0] + radii - 1)**2)
        val += penalty_weight * np.sum(np.maximum(0, radii - centers[:, 1])**2)
        val += penalty_weight * np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2)
        
        # 3. Overlap Penalty
        # Distance between centers >= sum of radii
        # dist^2 >= (r_i + r_j)^2
        # Penalty for dist < r_i + r_j
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    val += penalty_weight * (min_dist - dist)**2
                    
        return val

    # --- Initialization: Hexagonal Lattice ---
    centers_init = []
    
    # Approximate number of columns and rows for 26 circles
    # A 5x5 grid holds 25. We need 26.
    # We can try 5 rows of 5, plus one extra in a 6th row or inserted in a gap.
    # A 6-row structure might be better for density.
    # Let's try to fit 26 circles in a hexagonal pattern.
    
    # Estimated radius for 26 equal circles is ~0.1.
    # We start with 0.08 to be safe.
    r_start = 0.08
    
    # Hexagonal packing coordinates
    # Row 0: 5 circles
    # Row 1: 5 circles (shifted)
    # Row 2: 5 circles
    # Row 3: 5 circles
    # Row 4: 5 circles
    # Row 5: 1 circle
    # Total 26.
    
    # Vertical spacing for hex packing is sqrt(3)/2 * 2r = sqrt(3)r ~ 1.732r
    # Horizontal spacing is 2r
    # Shift for odd rows is r
    
    y_coords = []
    row_counts = [5, 5, 5, 5, 5, 1]
    
    # We need to center these in the square [0,1]x[0,1]
    # Total height roughly: 2*r + 5*sqrt(3)*r
    # With r=0.08, height ~ 0.16 + 5*0.138 = 0.85. Fits well.
    
    current_y = r_start
    for count in row_counts:
        for k in range(count):
            x = r_start + k * 2 * r_start
            # Shift odd rows
            if len(y_coords) % 2 == 1: # 1st row index is 1 (0-based) -> odd
                 x += r_start
            centers_init.append([x, current_y])
        current_y += np.sqrt(3) * r_start
        
    # Center the whole configuration in the square
    centers_arr = np.array(centers_init)
    min_x, min_y = centers_arr.min(axis=0)
    max_x, max_y = centers_arr.max(axis=0)
    
    # Shift to center
    shift_x = (1 - (max_x - min_x)) / 2 - min_x
    shift_y = (1 - (max_y - min_y)) / 2 - min_y
    
    centers_init = centers_arr + [shift_x, shift_y]
    
    radii_init = np.full(n_circles, r_start)
    
    # --- Optimization ---
    # Variables: x1, y1, r1, x2, y2, r2, ...
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
        
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Run optimization
    # Using L-BFGS-B as it handles bounds efficiently
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 2000, 'ftol': 1e-12})
    
    # Extract results
    final_params = result.x
    final_centers = final_params[:2 * n_circles].reshape(-1, 2)
    final_radii = final_params[2 * n_circles:]
    
    # Ensure non-negative radii (though bounds should handle it)
    final_radii = np.maximum(final_radii, 0)
    
    return final_centers, final_radii, np.sum(final_radii)
