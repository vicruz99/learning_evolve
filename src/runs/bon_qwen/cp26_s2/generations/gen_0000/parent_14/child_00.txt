import numpy as np
from scipy.optimize import minimize

def get_init_positions(n=26):
    """Generate initial positions based on a dense hexagonal pattern."""
    positions = []
    r_est = 0.10
    x, y = 0.0, 0.0
    row = 0
    
    while len(positions) < n:
        if row % 2 == 0:
            x_offset = 0.0
        else:
            x_offset = r_est * np.sqrt(3)
            
        col = 0
        while True:
            x = x_offset + col * (2 * r_est * np.sqrt(3))
            y = row * (3 * r_est)
            
            if 0 <= x <= 1 and 0 <= y <= 1:
                positions.append([x, y])
                if len(positions) == n:
                    break
            col += 1
            if x > 1:
                break
        row += 1
        if y > 1:
            break
            
    # Fill any missing slots randomly if the pattern was tight
    while len(positions) < n:
        positions.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        
    return np.array(positions[:n])

def objective_and_penalty(params, n=26):
    """
    Computes the negative sum of radii plus a penalty for constraint violations.
    params format: [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = params[3 * i]
        centers[i, 1] = params[3 * i + 1]
        radii[i] = params[3 * i + 2]
        
    # Penalty for negative radii (though we clamp, this helps gradient)
    if np.any(radii < 0):
        return 1e9, None
        
    # 1. Maximize sum of radii (so we minimize negative sum)
    loss = -np.sum(radii)
    
    # 2. Boundary penalties
    # x - r >= 0 => penalty if r > x
    # x + r <= 1 => penalty if x + r > 1
    # Same for y
    boundary_pen = 0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0: r = 0 # For distance calc
        
        # Left/Bottom
        dist_left = x - r
        if dist_left < 0:
            boundary_pen += dist_left**2
        dist_bottom = y - r
        if dist_bottom < 0:
            boundary_pen += dist_bottom**2
            
        # Right/Top
        dist_right = 1 - (x + r)
        if dist_right < 0:
            boundary_pen += dist_right**2
        dist_top = 1 - (y + r)
        if dist_top < 0:
            boundary_pen += dist_top**2
            
    # 3. Overlap penalties
    overlap_pen = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                           (centers[i, 1] - centers[j, 1])**2)
            req_dist = radii[i] + radii[j]
            if dist < req_dist:
                overlap_pen += (req_dist - dist)**2
                
    total_loss = loss + 100.0 * boundary_pen + 100.0 * overlap_pen
    
    return total_loss, np.array([centers, radii, -total_loss]) # Return centers, radii, neg_loss (score)

def run_packing():
    best_score = -np.inf
    best_params = None
    
    n = 26
    num_trials = 15
    
    for _ in range(num_trials):
        # Mix of hexagonal and random starts
        if np.random.random() > 0.5:
            centers_init = get_init_positions(n)
        else:
            centers_init = np.random.uniform(0.1, 0.9, (n, 2))
            
        radii_init = np.full(n, 0.05) # Start small to avoid immediate heavy penalty
        
        # Pack into optimization array
        params_init = []
        for i in range(n):
            params_init.extend([centers_init[i, 0], centers_init[i, 1], radii_init[i]])
            
        # Bounds: x,y in [0,1], r >= 0
        bounds = []
        for i in range(n):
            bounds.extend([(0, 1), (0, 1), (0, None)])
            
        # Optimize
        try:
            res = minimize(objective_and_penalty, params_init, args=(n,), 
                           method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 2000, 'ftol': 1e-9})
            
            # Extract results
            centers_opt = np.zeros((n, 2))
            radii_opt = np.zeros(n)
            for i in range(n):
                centers_opt[i, 0] = res.x[3 * i]
                centers_opt[i, 1] = res.x[3 * i + 1]
                radii_opt[i] = max(0, res.x[3 * i + 2])
            
            current_sum = np.sum(radii_opt)
            
            # Check validity roughly
            if res.success or res.nit > 500: 
                 if current_sum > best_score:
                    best_score = current_sum
                    best_params = (centers_opt, radii_opt, current_sum)
        except Exception:
            continue

    if best_params is None:
        # Fallback to simple grid
        centers = np.zeros((26, 2))
        radii = np.full(26, 0.05)
        return centers, radii, 1.3

    # Final clean up to ensure strict non-negative and bounds
    c, r, s = best_params
    r = np.clip(r, 0, None)
    c[:, 0] = np.clip(c[:, 0], 0, 1)
    c[:, 1] = np.clip(c[:, 1], 0, 1)
    
    # Recompute sum based on clipped values
    final_sum = np.sum(r)
    
    return c, r, final_sum
