import numpy as np
from scipy.optimize import minimize, linprog

def get_initial_config(n, seed):
    """
    Generates an initial configuration of n circles in a unit square.
    Uses a perturbed hexagonal grid for better packing density.
    """
    np.random.seed(seed)
    
    # Approximate spacing for hexagonal packing
    # Area per circle in hex lattice is approx sqrt(3)/2 * (2r)^2.
    # We want n circles to cover area ~1.
    # s = 2r. Area ~ n * sqrt(3)/2 * s^2 = 1 => s ~ sqrt(2 / (n * sqrt(3)))
    s = np.sqrt(2.0 / (n * np.sqrt(3.0)))
    
    centers = []
    y = s * np.sqrt(3.0) / 2.0
    row = 0
    
    # Generate points in a hexagonal pattern
    while len(centers) < n:
        x = 0.0
        if row % 2 == 1:
            x = s / 2.0
        
        while len(centers) < n:
            centers.append([x, y * row])
            x += s
            if x > 1.0 + s: 
                break
        row += 1
        if y * row > 1.0 + s:
            break
            
    centers = np.array(centers[:n])
    
    # Normalize and center the points within [0, 1]
    if centers.size > 0:
        pts_min = centers.min(axis=0)
        pts_max = centers.max(axis=0)
        dim_max = (pts_max - pts_min).max()
        if dim_max > 0:
            centers -= pts_min
            centers /= dim_max
            centers += 0.5
        
        # Add noise to break symmetry and help escape local minima
        centers += np.random.uniform(-0.05, 0.05, centers.shape)
        centers = np.clip(centers, 0.0, 1.0)
    
    # Initialize with small radii
    radii = np.ones(n) * 0.01
    return centers, radii

def objective_function(params, n, penalty_weight):
    """
    Objective function for optimization.
    Minimizes: -Sum(radii) + Penalty(Overlaps + BoundaryViolations)
    """
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]
    
    # Primary objective: Maximize sum of radii (minimize negative sum)
    cost = -np.sum(radii)
    
    # Penalty for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                cost += penalty_weight * (overlap ** 2)
        
        # Penalty for boundaries
        x, y = centers[i]
        r = radii[i]
        
        # Left and Right boundaries
        if r > x:
            cost += penalty_weight * (r - x) ** 2
        if r > 1 - x:
            cost += penalty_weight * (r - (1 - x)) ** 2
            
        # Bottom and Top boundaries
        if r > y:
            cost += penalty_weight * (r - y) ** 2
        if r > 1 - y:
            cost += penalty_weight * (r - (1 - y)) ** 2
            
    return cost

def solve_radii_lp(centers, n):
    """
    Given fixed centers, solve an LP to maximize sum of radii subject to non-overlap and boundary constraints.
    """
    c = -np.ones(n)  # Minimize -sum(r) => Maximize sum(r)
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    for i in range(n):
        x, y = centers[i]
        bounds = [x, 1 - x, y, 1 - y]
        for b in bounds:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(b)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback to small radii if LP fails (should not happen with feasible centers)
            return np.ones(n) * 1e-6
    except Exception:
        return np.ones(n) * 1e-6

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Main function to pack 26 circles in a unit square.
    Returns (centers, radii, sum_radii).

    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    num_restarts = 15

        # 1. Initialize

        x0 = np.concatenate([centers.flatten(), radii])

        # 2. Optimize Centers and Radii with Penalty
        res = minimize(objective_function, x0, args=(n, penalty_weight), 
                       method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 2000, 'ftol': 1e-12})
        
        centers_opt = res.x[:2 * n].reshape((n, 2))
        
        # 3. Refine Radii with LP
        # This ensures radii are maximal for the found centers
        radii_opt = solve_radii_lp(centers_opt, n)
        
        # 4. Validation
        current_sum = np.sum(radii_opt)
        valid = True
        
        # Check overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j]) ** 2))
                if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break

            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        
        if valid and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
            
    return best_centers, best_radii, best_sum
