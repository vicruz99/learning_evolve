import numpy as np
from scipy.optimize import minimize

def objective(x):
    """
    Objective function to minimize: negative sum of radii.
    x is a flat array [x1, y1, r1, x2, y2, r2, ...]
    """
    # Radii are at indices 2, 5, 8, ...
    return -np.sum(x[2::3])

def boundary_constraints(x):
    """
    Constraints for circles to be inside the unit square.
    Returns an array of values that must be >= 0.
    """
    n = len(x) // 3
    # Reshape to (n, 3) where columns are x, y, r
    cr = x.reshape((n, 3))
    
    constraints = []
    
    # x - r >= 0
    constraints.append(cr[:, 0] - cr[:, 2])
    # 1 - x - r >= 0
    constraints.append(1.0 - cr[:, 0] - cr[:, 2])
    # y - r >= 0
    constraints.append(cr[:, 1] - cr[:, 2])
    # 1 - y - r >= 0
    constraints.append(1.0 - cr[:, 1] - cr[:, 2])
    
    return np.concatenate(constraints)

def overlap_constraints(x):
    """
    Constraints for circles not to overlap.
    dist(i, j) >= r_i + r_j  =>  dist(i, j) - (r_i + r_j) >= 0
    Returns an array of values for all pairs i < j.
    """
    n = len(x) // 3
    cr = x.reshape((n, 3))
    
    # Compute pairwise Euclidean distances
    # dx[i, j] = x_i - x_j
    dx = cr[:, 0][:, np.newaxis] - cr[:, 0][np.newaxis, :]
    dy = cr[:, 1][:, np.newaxis] - cr[:, 1][np.newaxis, :]
    dist = np.sqrt(dx**2 + dy**2)
    
    # Compute sum of radii for all pairs
    # r_sum[i, j] = r_i + r_j
    r_sum = cr[:, 2][:, np.newaxis] + cr[:, 2][np.newaxis, :]
    
    # Constraint violation
    diff = dist - r_sum
    
    # We only need constraints for i < j (upper triangle)
    idx = np.triu_indices(n, k=1)
    return diff[idx]

def generate_hex_grid(n, spacing=0.18):
    """
    Generates initial positions in a hexagonal lattice pattern.
    """
    centers = []
    # Row spacing is spacing * sqrt(3) / 2
    row_h = spacing * np.sqrt(3) / 2
    
    y = 0.1 + row_h / 2 # Start slightly offset to fit better
    row_idx = 0
    
    while len(centers) < n:
        # Offset every other row
        offset = (row_idx % 2) * (spacing / 2)
        x = 0.1 + offset
        
        while x <= 0.9 and len(centers) < n:
            centers.append((x, y))
            x += spacing
        
        y += row_h
        row_idx += 1
        
    return centers[:n]
