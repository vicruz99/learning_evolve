import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n, mu):
    """
    Penalized objective function for circle packing.
    Minimizes this function to maximize sum of radii while avoiding overlaps/boundaries.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]

    # Pairwise distances (vectorized)
    c_diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(c_diff**2, axis=2))

    # Overlap penalty
    i, j = np.triu_indices(n, k=1)
    overlaps = np.maximum(0.0, radii[i] + radii[j] - dists[i, j])
    loss_pair = np.sum(overlaps**2)

    # Boundary violation penalty
    loss_bound = np.sum(np.maximum(0.0, -centers[:, 0] + radii)**2)
    loss_bound += np.sum(np.maximum(0.0, centers[:, 0] + radii - 1.0)**2)
    loss_bound += np.sum(np.maximum(0.0, -centers[:, 1] + radii)**2)
    loss_bound += np.sum(np.maximum(0.0, centers[:, 1] + radii - 1.0)**2)

    # Objective: maximize sum of radii <=> minimize -sum(radii) + penalty
    return -np.sum(radii) + mu * (loss_pair + loss_bound)

def run_packing():
    n = 26
    rng = np.random.default_rng(42)

    # 1. Initialize positions on a hexagonal/triangular lattice
    pts = []
    a = 0.22  # lattice spacing
    h = a * np.sqrt(3) / 2
    for j in range(15):
        for i in range(15):
            x = i * a + (j % 2) * 0.5 * a
            y = j * h
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                pts.append([x, y])
    pts = np.array(pts[:n])
    # Add small random noise to break symmetry
    pts += rng.uniform(-0.01, 0.01, size=pts.shape)

    # 2. Initialize radii
    r0 = 0.09 * np.ones(n)
    x0 = np.concatenate([pts.flatten(), r0])

    # 3. Set bounds for L-BFGS-B
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    mu = 25000.0  # Penalty weight

    # 4. Optimize
    res = minimize(compute_loss, x0, args=(n, mu), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': 4000, 'ftol': 1e-12})

    centers = res.x[:2*n].reshape(n, 2)
    radii = res.x[2*n:]

    # 5. Post-process to guarantee strict validity
    c_diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(c_diff**2, axis=2))
    i, j = np.triu_indices(n, k=1)
    overlaps = (radii[i] + radii[j]) - dists[i, j]
    max_overlap = np.max(overlaps)

    min_margin = min(
        np.min(centers[:, 0] - radii),
        np.min(1.0 - centers[:, 0] - radii),
        np.min(centers[:, 1] - radii),
        np.min(1.0 - centers[:, 1] - radii)
    )

    # Shrink radii slightly if any constraint is violated
    if max_overlap > 1e-9 or min_margin < -1e-9:
        scale = min(1.0 - max_overlap / 2.5, 1.0 + min_margin / 2.5)
        radii *= max(0.0, scale)

    sum_r = np.sum(radii)
    return centers, radii, sum_r
