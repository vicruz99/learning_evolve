import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    N = 26
    best_centers = None
    best_radii = None
    max_sum_r = -1.0

    # Helper to create initial hexagonal lattice configuration
    def create_hex_lattice(n, rotation=0.0, scale=1.0):
        # Approximate radius based on area density for N circles
        # Area ~ N * pi * r^2. Density ~ 0.9069.
        # N * pi * r^2 <= 0.9069 * 1 => r <= sqrt(0.9069 / (N * pi))
        est_r = np.sqrt(0.9069 / (N * np.pi)) * scale
        
        rows = []
        current_n = 0
        row_idx = 0
        
        # Try to fill rows
        while current_n < N:
            # Number of circles in this row
            # Alternating row sizes can help fit more
            # Start with roughly sqrt(N)
            base_count = int(np.ceil(N / 5.0)) # 5 rows roughly
            if row_idx % 2 == 0:
                count = base_count
            else:
                count = base_count - 1 # Shifted rows often have 1 less
            
            # Don't exceed N
            if current_n + count > N:
                count = N - current_n
            
            row_y = row_idx * est_r * np.sqrt(3) # Vertical spacing for hex packing
            # If row is shifted (odd index), shift x by est_r
            x_offset = est_r if row_idx % 2 == 1 else 0
            
            row_circles = []
            for c in range(count):
                x = x_offset + c * (2 * est_r)
                y = row_y + est_r # Add r to y to start from r, not 0
                row_circles.append([x, y])
            
            rows.append(row_circles)
            current_n += count
            row_idx += 1
            
            # Safety break
            if row_idx > 10:
                break
        
        centers = np.array([p for r_c in rows for p in r_c])
        
        # Rotate if needed
        if rotation != 0:
            c = np.array([0.5, 0.5])
            angle = np.radians(rotation)
            R = np.array([[np.cos(angle), -np.sin(angle)],
                          [np.sin(angle), np.cos(angle)]])
            centers = (centers - c) @ R.T + c
            
        # Normalize/Scale to fit in [0,1]
        min_c = np.min(centers, axis=0)
        max_c = np.max(centers, axis=0)
        width = max_c[0] - min_c[0]
        height = max_c[1] - min_c[1]
        
        if width > 0 and height > 0:
            scale_factor = 1.0 / max(width, height)
            # Center it
            centers = centers * scale_factor
            # Adjust to center in [0,1]
            current_min = np.min(centers, axis=0)
            current_max = np.max(centers, axis=0)
            shift = (1.0 - (current_max - current_min)) / 2.0 - current_min
            centers += shift
            
        # Estimate radii based on min distance to boundary and neighbors
        # A rough estimate: 0.5 * min(distance to other centers, distance to boundary)
        # But for initialization, uniform radius is easier.
        # Let's estimate r from the grid spacing we used.
        r_init = est_r * scale_factor
        
        # Clip r to fit boundaries roughly
        dist_to_wall = np.min([np.min(centers[:,0]), np.min(1-centers[:,0]), 
                               np.min(centers[:,1]), np.min(1-centers[:,1])])
        r_init = min(r_init, dist_to_wall * 0.9)
        
        radii = np.full(N, r_init)
        return centers[:N], radii[:N]

    def objective(x_vars):
        # x_vars contains [x1, y1, r1, x2, y2, r2, ...]
        # But optimization is easier if we separate centers and radii
        # Here x_vars is just a flat array
        # We will reshape it in the constraint/callback or just access directly
        # Let's assume structure: centers (N, 2) then radii (N,)
        # Actually, passing (centers, radii) to minimize is not direct.
        # We flatten them.
        centers = x_vars[:N*2].reshape(N, 2)
        radii = x_vars[N*2:]
        return -np.sum(radii)

    def constraint_overlap(x_vars):
        centers = x_vars[:N*2].reshape(N, 2)
        radii = x_vars[N*2:]
        cons = []
        for i in range(N):
            for j in range(i + 1, N):
                # dist^2 >= (r_i + r_j)^2
                dist_sq = np.sum((centers[i] - centers[j])**2)
                r_sum = radii[i] + radii[j]
                # dist_sq - r_sum^2 >= 0
                cons.append(dist_sq - r_sum**2)
        return np.array(cons)

    def constraint_boundary(x_vars):
        centers = x_vars[:N*2].reshape(N, 2)
        radii = x_vars[N*2:]
        cons = []
        for i in range(N):
            # x - r >= 0
            cons.append(centers[i, 0] - radii[i])
            # 1 - x - r >= 0
            cons.append(1 - centers[i, 0] - radii[i])
            # y - r >= 0
            cons.append(centers[i, 1] - radii[i])
            # 1 - y - r >= 0
            cons.append(1 - centers[i, 1] - radii[i])
        return np.array(cons)

    def constraint_positive_r(x_vars):
        radii = x_vars[N*2:]
        return radii

    # SLSQP expects constraints as dictionaries
    # 'type': 'ineq', 'fun': function returning >= 0

    # but SLSQP handles list of dicts better for different types.

    # No, we need to optimize r.

    # To avoid passing huge arrays, we can use a helper that packs constraints.
    
    def all_constraints(x_vars):
        c = []
        # Overlap
        centers = x_vars[:N*2].reshape(N, 2)
        radii = x_vars[N*2:]
        for i in range(N):
            for j in range(i + 1, N):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                r_sum = radii[i] + radii[j]
                c.append(dist_sq - r_sum**2)
        # Boundary
        for i in range(N):
            c.append(centers[i, 0] - radii[i])
            c.append(1 - centers[i, 0] - radii[i])
            c.append(centers[i, 1] - radii[i])
            c.append(1 - centers[i, 1] - radii[i])
        # Positive radii (small epsilon)
        for i in range(N):
            c.append(radii[i] - 1e-6)
        return np.array(c)

    best_sum = -np.inf
    best_sol = None

    # Try multiple initial configurations
    configs_to_try = [
        (0.0, 1.0),    # Standard hex
        (15.0, 1.0),   # Rotated
        (30.0, 1.0),
        (45.0, 1.0),
        (0.0, 1.1),    # Scaled up (will be constrained)
        (0.0, 0.9),    # Scaled down
    ]

    # Also try random perturbations
    for _ in range(3):

        # Add noise
        centers += np.random.normal(0, 0.01, centers.shape)
        # Clip
        centers = np.clip(centers, 0.02, 0.98)
        configs_to_try.append(('random', centers, radii))

    for item in configs_to_try:
        if item[0] == 'random':
            init_centers = item[1]
            init_radii = item[2]
        else:
            angle, scale = item
            init_centers, init_radii = create_hex_lattice(N, rotation=angle, scale=scale)
        
        # Ensure valid init
        init_radii = np.clip(init_radii, 1e-4, 0.5)
        # Center validity check (rough)
        # If init is bad, optimizer might fail.
        # Let's ensure centers are within bounds for init radii
        for i in range(N):
            x, y = init_centers[i]
            r = init_radii[i]
            init_centers[i, 0] = np.clip(x, r, 1-r)
            init_centers[i, 1] = np.clip(y, r, 1-r)
        
        x0 = np.concatenate([init_centers.flatten(), init_radii])
        
        try:
            # Bounds: x, y in [0, 1], r in [0, 0.5]
            bounds = [(0, 1)] * (2 * N) + [(1e-5, 0.5)] * N
            
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints={'type': 'ineq', 'fun': all_constraints},
                options={'maxiter': 2000, 'ftol': 1e-9}
            )
            
            if res.success or res.fun < best_sum: # res.fun is negative sum
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_sol = res.x
                    best_centers = best_sol[:N*2].reshape(N, 2)
                    best_radii = best_sol[N*2:]
        except Exception:
            continue

    # If optimization failed or didn't find better, fallback to initial hex
    if best_centers is None:
        best_centers, best_radii = create_hex_lattice(N, rotation=0.0, scale=1.0)
        best_sum = np.sum(best_radii)

    # Final cleanup and validation logic check (internal)
    # The constraints should hold, but let's ensure strict positivity
    best_radii = np.maximum(best_radii, 1e-6)
    
    # Re-calculate sum
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, final_sum
