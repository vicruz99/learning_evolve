        x0[3*i+2] = radii[i]

    def objective(variables):
        return -np.sum(variables[2::3])

    bounds = []

    for i in range(n_circles):
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        # 1 - x - r >= 0
