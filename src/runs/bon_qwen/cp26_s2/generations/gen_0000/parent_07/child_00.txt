import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Assumes optimal packing uses equal radii arranged in a dense pattern.
    """
    n_circles = 26
    
    # --- 1. Initialization: Hexagonal Lattice ---
    # We arrange circles in rows with counts 6, 5, 6, 5, 4 to sum to 26.
    # This mimics a hexagonal packing which is denser than a grid.
    rows = [6, 5, 6, 5, 4]
    
    centers = []
    # Initial guess for radius (grid gives 0.1, hex might allow slightly more)
    r_init = 0.1 
    
    # Vertical spacing for hexagonal packing is sqrt(3)/2 * diameter = sqrt(3)*r
    # We will scale this later, but for init, let's just place them roughly.
    y_curr = r_init
    for i, count in enumerate(rows):
        # Horizontal spacing is 2*r. 
        # Total width for 'count' circles is (count-1)*2*r + 2*r = 2*r*count
        # To center them in [0,1]: start_x = (1 - 2*r*count) / 2 + r
        # But for init, let's just spread them evenly.
        
        # We want to fit 'count' circles. 
        # Available width for centers is 1 - 2*r.
        # Spacing dx = (1 - 2*r) / (count - 1) if count > 1 else 0
        
        if count == 0:
            continue
            
        if count == 1:
            x_vals = [0.5]
        else:
            # Center the row
            # We'll just place them with spacing roughly 2*r initially
            # and let the optimizer fix it.
            # To fit in [0,1], let's map 0..count-1 to r..1-r
            x_vals = np.linspace(r_init, 1 - r_init, count)
            
        # Shift alternate rows for hexagonal effect
        if i % 2 == 1:
            # Shift by half spacing
            # This is approximate for init
            x_vals = x_vals - (x_vals[1] - x_vals[0]) / 2
            
        for x in x_vals:
            centers.append([x, y_curr])
            
        # Move to next row
        y_curr += np.sqrt(3) * r_init

    centers = np.array(centers[:n_circles]) # Ensure we have exactly n_circles
    
    # --- 2. Optimization Setup ---
    # Variables: x1, y1, ..., x26, y26, r
    # Shape: 26*2 + 1 = 53
    
    def objective(vars):
        # Return negative radius to maximize
        return -vars[-1]

    def constraint_boundary(vars):
        x = vars[0::2]
        y = vars[1::2]
        r = vars[-1]
        # Constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        # Returns values that must be >= 0
        return np.concatenate([
            x - r,
            (1 - r) - x,
            y - r,
            (1 - r) - y
        ])

    def constraint_overlap(vars):
        x = vars[0::2]
        y = vars[1::2]
        r = vars[-1]
        
        # Distance between i and j must be >= 2r
        # dist^2 >= 4r^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - 4r^2 >= 0
        
        cons = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist_sq = dx*dx + dy*dy
                cons.append(dist_sq - 4*r*r)
        return np.array(cons)

    # Initial guess vector
    x0 = np.zeros(2 * n_circles + 1)
    x0[0::2] = centers[:, 0]
    x0[1::2] = centers[:, 1]
    x0[-1] = r_init
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n_circles) + [(1e-6, 0.5)]
    
    # Define constraints for SLSQP
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    # --- 3. Run Optimizer ---
    # SLSQP is a good choice for constrained optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
        
        if res.success:
            final_centers = np.zeros((n_circles, 2))
            final_centers[:, 0] = res.x[0::2]
            final_centers[:, 1] = res.x[1::2]
            final_r = res.x[-1]
            final_radii = np.full(n_circles, final_r)
            final_sum = np.sum(final_radii)
            
            # Return result
            return final_centers, final_radii, final_sum
        else:
            # Fallback to initial if optimization fails
            print("Optimization failed, using initial config.")
            return centers, np.full(n_circles, r_init), n_circles * r_init
            
    except Exception as e:
        print(f"Optimization error: {e}")
        return centers, np.full(n_circles, r_init), n_circles * r_init
