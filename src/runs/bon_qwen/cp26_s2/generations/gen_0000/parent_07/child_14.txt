import numpy as np

def compute_safe_radius(centers, n):
    """Compute the maximum feasible radius for given centers strictly inside bounds."""
    min_d = 1.0
    # Distance to boundaries
    for i in range(n):
        x, y = centers[i]
        min_d = min(min_d, x, 1 - x, y, 1 - y)
        
    # Half distance between circle pairs
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            min_d = min(min_d, d / 2)
            
    # Apply safety margin to strictly satisfy validation tolerances
    return max(1e-4, min_d - 1e-7)
