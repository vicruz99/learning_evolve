import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization followed by numerical optimization.
    """

    # 1. Generate initial configuration using a hexagonal lattice pattern
    centers = np.zeros((n, 2))

    # We aim for a radius slightly above 0.1. 
    # Vertical spacing is sqrt(3)/2 * diameter = sqrt(3) * r
    # Horizontal spacing is diameter = 2 * r
    
    # Estimate initial radius based on density arguments.
    # 26 circles, target sum ~ 2.64 => avg r ~ 0.1015
    r_init = 0.102
    
    # Calculate dimensions for the lattice
    row_spacing = np.sqrt(3) * r_init
    col_spacing = 2.0 * r_init
    
    # Determine grid size to fit 26 circles
    # Try to fit in a 5x6 or similar staggered grid
    # Pattern: 5 rows. Row counts: 6, 5, 6, 5, 4 -> Total 26
    # Or 5, 5, 5, 5, 6 (staggered)
    
    # Let's construct a 5-row pattern
    # Row 0: 6 circles
    # Row 1: 5 circles (offset)
    # Row 2: 6 circles
    # Row 3: 5 circles (offset)
    # Row 4: 4 circles
    # Total: 26
    
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    
    # Calculate total width and height to center the pattern
    max_width = 0
    max_height = 0
    
    # Pre-calculate positions relative to (0,0)
    temp_centers = []
    for row_idx, count in enumerate(row_counts):
        y = row_idx * row_spacing
        # Offset every other row by r_init
        x_offset = r_init if row_idx % 2 == 1 else 0.0
        for col_idx in range(count):
            x = col_idx * col_spacing + x_offset
            temp_centers.append([x, y])
            
    temp_centers = np.array(temp_centers)
    
    # Compute bounding box
    min_x, min_y = np.min(temp_centers, axis=0)
    max_x, max_y = np.max(temp_centers, axis=0)
    
    # Center the pattern in the unit square [0, 1]
    # We want the extent of circles (center +/- r) to be within [0, 1]
    # The current extent of centers is [min_x, max_x]
    # We want final centers to be in [r_init, 1-r_init]
    # Scale/Translate
    
    current_width = max_x - min_x
    current_height = max_y - min_y
    
    # Desired width for centers to allow radius r_init
    # Available width for centers is 1 - 2*r_init
    avail_w = 1.0 - 2.0 * r_init
    avail_h = 1.0 - 2.0 * r_init
    
    # Scale factor
    if current_width > 0:
        scale_x = avail_w / current_width
    else:
        scale_x = 1.0
        
    if current_height > 0:
        scale_y = avail_h / current_height
    else:
        scale_y = 1.0
    
    # Use the smaller scale to fit in both dimensions
    scale = min(scale_x, scale_y)
    
    # Apply scaling and centering
    # Shift to origin first
    temp_centers -= [min_x, min_y]
    # Scale
    temp_centers *= scale
    # Center in available space
    # New width/height after scaling
    new_w = current_width * scale
    new_h = current_height * scale
    # Shift to center of available space [r, 1-r]
    shift_x = r_init + (avail_w - new_w) / 2.0
    shift_y = r_init + (avail_h - new_h) / 2.0
    
    centers = temp_centers + [shift_x, shift_y]
    
    # Initialize radii
    radii = np.full(n, r_init)
    
    # 2. Optimization
    # We optimize both centers and radii.
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total 3 * n variables.
    
    x0 = []
    for i in range(n):
        x0.extend([centers[i, 0], centers[i, 1], radii[i]])
    x0 = np.array(x0)
    
    # Bounds: x, y in [0, 1], r >= 0
    # Actually, for optimization stability, we can bound x,y in [0,1] and r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    # Objective function: Maximize sum of radii => Minimize -sum(radii)
    # Penalty for violations
    def objective(vars):
        sum_r = 0.0
        penalty = 0.0
        penalty_weight = 1000.0
        
        current_centers = np.zeros((n, 2))
        current_radii = np.zeros(n)
        
        for i in range(n):
            idx = 3 * i
            cx, cy, r = vars[idx], vars[idx+1], vars[idx+2]
            current_centers[i] = [cx, cy]
            current_radii[i] = r
            sum_r += r
            
            # Boundary penalty
            if cx < r: penalty += penalty_weight * (cx - r)**2
            if cx > 1 - r: penalty += penalty_weight * (cx - (1 - r))**2
            if cy < r: penalty += penalty_weight * (cy - r)**2
            if cy > 1 - r: penalty += penalty_weight * (cy - (1 - r))**2
            
        # Overlap penalty
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((current_centers[i, 0] - current_centers[j, 0])**2 +

                min_dist = current_radii[i] + current_radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    penalty += penalty_weight * overlap**2
                    
        return -sum_r + penalty

    # Use L-BFGS-B for bounded optimization
    # It handles bounds well and is efficient
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 2000, 'ftol': 1e-12})
    
    # Extract results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        idx = 3 * i
        final_centers[i, 0] = result.x[idx]
        final_centers[i, 1] = result.x[idx+1]
        final_radii[i] = result.x[idx+2]
        
    # Clip radii to ensure non-negative (though bounds should handle it)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Calculate sum
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
