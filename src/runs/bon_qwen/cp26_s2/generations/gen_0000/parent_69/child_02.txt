import numpy as np
import scipy.optimize as opt

def compute_objective(v, n, mu):
    """Computes the penalized objective function for circle packing."""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    val = -np.sum(r)  # We maximize sum of radii
    
    # Pairwise non-overlap penalties
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
            pen = r[i] + r[j] - d
            if pen > 0:
                val += mu * pen**2
                
        # Boundary penalties
        if r[i] - c[i,0] > 0: val += mu * (r[i] - c[i,0])**2
        if c[i,0] + r[i] - 1 > 0: val += mu * (c[i,0] + r[i] - 1)**2
        if r[i] - c[i,1] > 0: val += mu * (r[i] - c[i,1])**2
        if c[i,1] + r[i] - 1 > 0: val += mu * (c[i,1] + r[i] - 1)**2
        
    return val

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    # Hexagonal-inspired initialization
    centers = []
    row_configs = [6, 5, 6, 5, 4]
    for i, num in enumerate(row_configs):
        y = 0.12 + i * 0.16
        width = (num - 1) * 0.20
        start_x = (1.0 - width) / 2
        for j in range(num):
            x = start_x + j * 0.20
            if i % 2 == 1:
                x += 0.10
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.04)
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    mu = 20.0
    x_curr = x0
    
    # Iterative penalty optimization
    for step in range(8):
        res = opt.minimize(compute_objective, x_curr, args=(n, mu), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 1000, 'ftol': 1e-10})
        x_curr = res.x
        mu *= 2.5
        
        # Escape local minima with small jitter
        if step < 6:
            x_curr[:2*n] += rng.normal(0, 0.0005, (2*n))
            x_curr[:2*n] = np.clip(x_curr[:2*n], 0.05, 0.95)
            
    centers_opt = x_curr[:2*n].reshape(n, 2)
    radii_opt = x_curr[2*n:]
    
    # Strict validity enforcement
    for _ in range(20):
        valid = True
        max_viol = 0.0
        
        for i in range(n):
            # Boundaries
            radii_opt[i] = max(0, min(radii_opt[i], centers_opt[i,0], 1-centers_opt[i,0], 
                                      centers_opt[i,1], 1-centers_opt[i,1]))
            
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(centers_opt[i,0]-centers_opt[j,0], centers_opt[i,1]-centers_opt[j,1])
                v = radii_opt[i] + radii_opt[j] - d
                if v > 0:
                    valid = False
                    if v > max_viol: max_viol = v
                    
        if valid or max_viol < 1e-14:
            break
            
        # Global scale down if microscopic overlaps persist
        radii_opt *= (1 - 1.1 * max_viol / np.max(radii_opt))
        
    return centers_opt, radii_opt, np.sum(radii_opt)
