import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Returns (radii, sum_radii).
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0

    # Objective: maximize sum(r) -> minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints lists
    A_rows = []
    b_vals = []
    
    # 1. Pairwise distance constraints: r_i + r_j <= dist(i, j)
    # Optimization: If dist >= 1.0, constraint is redundant because r_i <= 0.5
    # (derived from boundary constraints x_i >= r_i and x_i <= 1-r_i => 2r_i <= 1)
    for i in range(n):
        xi, yi = centers[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            dist = np.hypot(xi - xj, yi - yj)
            
            if dist < 1.0 - 1e-9: # Only add if not trivially satisfied
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_rows.append(row)
                b_vals.append(dist)
    
    # 2. Boundary constraints
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    for i in range(n):
        xi, yi = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        
        # x-bound
        A_rows.append(row)
        b_vals.append(xi)
        A_rows.append(row)
        b_vals.append(1.0 - xi)
        
        # y-bound
        A_rows.append(row)
        b_vals.append(yi)
        A_rows.append(row)
        b_vals.append(1.0 - yi)

    if not A_rows:
        # If no constraints (impossible for n>1 inside square), return max possible
        # But this branch is likely unreachable with valid centers
        return np.full(n, 0.5), 0.5 * n

    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Bounds: 0 <= r_i <= 0.5
    bounds = [(0, 0.5) for _ in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback if LP fails
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def calculate_forces(centers, radii, tight_tol=1e-6):
    """
    Calculates repulsive forces on centers based on tight constraints.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    for i in range(n):
        fx, fy = 0.0, 0.0
        xi, yi = centers[i]
        ri = radii[i]
        
        # Check boundary constraints
        # If touching left wall (x - r approx 0), push right
        if xi - ri < tight_tol:
            fx += 1.0
        # If touching right wall (x + r approx 1), push left
        if xi + ri > 1.0 - tight_tol:
            fx -= 1.0
        # If touching bottom (y - r approx 0), push up
        if yi - ri < tight_tol:
            fy += 1.0
        # If touching top (y + r approx 1), push down
        if yi + ri > 1.0 - tight_tol:
            fy -= 1.0
            
        # Check pairwise constraints
        for j in range(n):
            if i == j:
                continue
            xj, yj = centers[j]
            rj = radii[j]
            dx = xi - xj
            dy = yi - yj
            dist = np.hypot(dx, dy)
            
            if dist < 1e-9:
                # Coincident centers, push randomly
                dx, dy = 1.0, 0.0
                dist = 1.0
            
            # If touching (r_i + r_j approx dist), push apart
            # Force direction is along vector (i - j)
            if ri + rj > dist - tight_tol:
                # Normalize force
                inv_dist = 1.0 / dist
                fx += dx * inv_dist
                fy += dy * inv_dist
        
        forces[i] = [fx, fy]
        
    return forces

def optimize_packing():
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # We will try multiple initializations
    num_trials = 20
    
    for trial in range(num_trials):

        # Run relaxation loop

            radii, current_sum = solve_radii(centers)
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
            
            if current_sum > 2.635: # Close to target
                break
                
            # 2. Calculate forces
            forces = calculate_forces(centers, radii)
            
            # 3. Update centers
            # Step size decay
            step_size = 0.02 * (1.0 - step / 200.0)
            
            # Normalize forces to avoid huge jumps
            magnitudes = np.linalg.norm(forces, axis=1, keepdims=True)
            magnitudes[magnitudes == 0] = 1.0
            normalized_forces = forces / magnitudes
            
            centers = centers + step_size * normalized_forces
            
            # 4. Add random jitter to escape local optima
            jitter = 0.005 * (1.0 - step / 200.0)
            centers += np.random.uniform(-jitter, jitter, centers.shape)
            
            # 5. Clip to valid region [0, 1]
            # To prevent radius becoming 0, keep centers away from edges slightly?
            # But LP handles radius=0 if center is on edge.
            # Better to clamp strictly inside [0,1]
            centers = np.clip(centers, 0.0, 1.0)
            
            # If stuck (force 0), perturb randomly
            if np.all(magnitudes < 1e-5):
                 centers += np.random.uniform(-0.01, 0.01, centers.shape)
                 centers = np.clip(centers, 0.0, 1.0)

        # Random restart if best not updated much (optional, handled by loop)
        
    # Final safety clamp and validation check (optional, but good practice)
    if best_centers is not None:
        # Re-solve radii for best centers one last time to ensure consistency
        final_radii, final_sum = solve_radii(best_centers)
        
        # Apply tiny shrink to ensure strict validity against numerical errors

        final_radii = final_radii * 0.999999999
        
        return best_centers, final_radii, np.sum(final_radii)
    
    return np.zeros((26, 2)), np.zeros(26), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    return optimize_packing()
