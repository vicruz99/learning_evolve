import numpy as np
from scipy.optimize import linprog, differential_evolution

def compute_lp(centers):
    """
    Solve the linear program to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to r_i + r_j <= dist(c_i, c_j) and boundary constraints.
    """
    n = centers.shape[0]
    n_pairs = n * (n - 1) // 2
    n_constraints = n_pairs + n
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    # Pairwise non-overlap constraints
    for i in range(n):
        diffs = centers - centers[i]
        dists = np.sqrt(np.sum(diffs**2, axis=1))
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[j]
            idx += 1
            
    # Boundary containment constraints
    for i in range(n):
        xi, yi = centers[i]
        b_ub[idx] = min(xi, 1.0 - xi, yi, 1.0 - yi)
        A_ub[idx, i] = 1.0
        idx += 1
        
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    # Use HiGHS solver for speed and robustness
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective(centers_flat):
    """Objective function for optimization: negative of max sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    val, _ = compute_lp(centers)
    return -val

def _get_radii(centers):
    """Extract optimal radii for a given set of centers."""
    _, radii = compute_lp(centers)
    return radii

def run_packing():
    n = 26
    bounds = [(0, 1)] * (n * 2)
    
    # Differential Evolution searches the 52D space of center coordinates
    result = differential_evolution(
        objective, 
        bounds, 
        seed=42, 
        popsize=15, 
        maxiter=200, 
        tol=1e-9, 
        polish=True, 
        workers=1,
        mutation=(0.5, 1.5),
        recombination=0.9
    )
    
    centers = result.x.reshape(n, 2)
    radii = _get_radii(centers)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
