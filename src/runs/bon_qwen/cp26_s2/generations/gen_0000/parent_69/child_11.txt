from scipy.optimize import linprog, minimize

warnings.filterwarnings("ignore")

    def solve_lp(centers):

                rows_A.append(row); vec_b.append(max(y, 1e-7))

                    row = np.zeros(n)
                    row[i], row[j] = 1.0, 1.0

                return -res.fun, res.x

        return -s, r
