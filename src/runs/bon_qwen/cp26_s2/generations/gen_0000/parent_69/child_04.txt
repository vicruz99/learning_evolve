import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, n):
    """Compute penalty for boundary violations and overlaps."""
    penalty = 0.0
    margin = 1e-7
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r + margin: 
            penalty += (r + margin - x)**2
        if x > 1 - r - margin: 
            penalty += (x - (1 - r - margin))**2
        if y < r + margin: 
            penalty += (r + margin - y)**2
        if y > 1 - r - margin: 
            penalty += (y - (1 - r - margin))**2
            
    # Pairwise overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            min_dist = radii[i] + radii[j] + margin
            if dist < min_dist:
                penalty += (min_dist - dist)**2
    return penalty

def objective(params, n):
    """Objective function: maximize sum of radii, penalize violations."""
    centers = params[:2*n].reshape((n, 2))
    radii = params[2*n:]
    # Minimize negative sum (maximize sum) plus weighted penalty
    return -np.sum(radii) + 1000.0 * compute_penalty(centers, radii, n)

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.02)
    
    # Hexagonal grid initialization
    idx = 0

        y = 0.15 + row * 0.18
        cols = 5 if row % 2 == 0 else 6
        for col in range(cols):
            if idx >= n: break
            x = 0.12 + col * 0.18 + (0.09 if row % 2 == 1 else 0.0)
            centers[idx] = [x, y]
            idx += 1
        if idx >= n: break
        
    # Simulation parameters
    dt = 0.02
    stiffness = 100.0
    damping = 0.8
    growth_rate = 0.00004
    steps = 10000
    
    velocities = np.zeros_like(centers)
    
    for _ in range(steps):
        # Expand radii
        radii += growth_rate
        
        forces = np.zeros_like(centers)
        
        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x < r: forces[i, 0] += stiffness * (r - x)
            elif x > 1 - r: forces[i, 0] -= stiffness * (x - (1 - r))
            if y < r: forces[i, 1] += stiffness * (r - y)
            elif y > 1 - r: forces[i, 1] -= stiffness * (y - (1 - r))
            
        # Repulsion forces
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                min_dist = radii[i] + radii[j]
                if dist_sq < min_dist**2 and dist_sq > 1e-12:
                    dist = np.sqrt(dist_sq)
                    overlap = min_dist - dist
                    fx = (dx / dist) * stiffness * overlap
                    fy = (dy / dist) * stiffness * overlap
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    
        velocities = velocities * damping + forces * dt
        centers += velocities * dt
        centers = np.clip(centers, 0.0, 1.0)
        
    # Fine-tuning with scipy
    params = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1) if i < 2*n else (0, 1)] * (3*n)
    
    res = minimize(
        objective, 
        params, 
        args=(n,), 
        method='L-BFGS-B', 
        bounds=bounds, 
        options={'maxiter': 1500, 'ftol': 1e-15, 'gtol': 1e-12}
    )
    
    final_centers = res.x[:2*n].reshape((n, 2))
    final_radii = np.maximum(res.x[2*n:], 0.0)
    
    total_radius = np.sum(final_radii)
    return final_centers, final_radii, total_radius
