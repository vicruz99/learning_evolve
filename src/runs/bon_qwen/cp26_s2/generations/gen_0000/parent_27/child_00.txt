import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization followed by SLSQP optimization.
    """
    n = 26

    # --- 1. Initialization: Hexagonal Lattice ---
    # A hexagonal lattice is denser than a square grid.
    # We generate points and select 26 that fit best.
    
    # Parameters for the lattice
    # We aim for a radius around 0.1 initially.
    # Spacing s_x = 2*r, s_y = sqrt(3)/2 * 2*r
    # Let's try to fit roughly 5x5 with offsets.
    
    rows = 6
    cols = 5
    points = []
    
    # Generate a grid of potential centers
    # We will use a spacing that allows r ~ 0.1
    # x step 0.2, y step sqrt(3)*0.1
    dx = 0.2
    dy = 0.2 * np.sqrt(3) / 2  # ~0.1732
    
    # Start with some padding
    pad = 0.1
    
    for r in range(rows):
        for c in range(cols):
            x = pad + c * dx + (r % 2) * (dx / 2)
            y = pad + r * dy
            # Keep points inside [0,1]
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    
    # If we generated too many or too few, we adjust or slice.
    # We need exactly 26.
    if len(points) > n:
        # Select the first n points (should be valid)
        points = points[:n]
    elif len(points) < n:
        # Fill remaining randomly if needed (unlikely with this grid)
        while len(points) < n:
            rx = np.random.uniform(0, 1)
            ry = np.random.uniform(0, 1)
            points.append([rx, ry])
            
    centers_init = np.array(points[:n])
    
    # Initialize radii to a small valid value to ensure feasibility
    # A safe initial radius for this grid spacing is roughly 0.09
    radii_init = np.full(n, 0.09)

    # --- 2. Optimization Setup ---
    # Variables: x_0, y_0, r_0, x_1, y_1, r_1, ...
    # Total 3 * n variables.
    
    def objective(vars_flat):
        # Negative sum of radii (for minimization)
        radii = vars_flat[2::3]
        return -np.sum(radii)

    def gradient(vars_flat):
        # Analytical gradient for the objective
        # d/dx_i (-sum r) = 0
        # d/dy_i (-sum r) = 0
        # d/dr_i (-sum r) = -1
        grad = np.zeros_like(vars_flat)
        grad[2::3] = -1.0
        return grad

    # Constraints
    constraints = []
    
    # 1. Boundary Constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    # Inequality form: g(vars) >= 0
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r],
            'jac': lambda v, i=i: np.zeros(3*n).__setitem__(3*i, 1) or np.zeros(3*n).__setitem__(3*i+2, -1) or np.zeros(3*n) # Simplified below
        })
        # Actually, defining jacobian for each constraint individually in a loop is verbose.
        # Let's define a single function for boundary constraints returning a vector.
        
    # Let's restructure constraints for efficiency
    
    def boundary_constraints(vars_flat):
        val = np.zeros(4 * n)
        for i in range(n):
            idx = 3 * i
            x = vars_flat[idx]
            y = vars_flat[idx+1]
            r = vars_flat[idx+2]
            
            # x - r >= 0
            val[4*i] = x - r
            # 1 - x - r >= 0
            val[4*i+1] = 1 - x - r
            # y - r >= 0
            val[4*i+2] = y - r
            # 1 - y - r >= 0
            val[4*i+3] = 1 - y - r
        return val

    def boundary_constraints_jac(vars_flat):
        jac = np.zeros((4 * n, 3 * n))
        for i in range(n):
            idx = 3 * i
            # d/dx (x-r) = 1, d/dr = -1
            jac[4*i, idx] = 1.0
            jac[4*i, idx+2] = -1.0
            
            # d/dx (1-x-r) = -1, d/dr = -1
            jac[4*i+1, idx] = -1.0
            jac[4*i+1, idx+2] = -1.0
            
            # d/dy (y-r) = 1, d/dr = -1
            jac[4*i+2, idx+1] = 1.0
            jac[4*i+2, idx+2] = -1.0
            
            # d/dy (1-y-r) = -1, d/dr = -1
            jac[4*i+3, idx+1] = -1.0
            jac[4*i+3, idx+2] = -1.0
        return jac

    # 2. Distance Constraints: dist_ij >= r_i + r_j
    # dist_ij^2 >= (r_i + r_j)^2  <-- Non-linear, but standard form is dist - sum_r >= 0
    # Let's use dist - (r_i + r_j) >= 0
    
    def distance_constraints(vars_flat):
        centers = vars_flat[0:3*n:3].reshape(n, 2) # x
        centers_y = vars_flat[1:3*n:3].reshape(n, 2) # y - Wait, indexing is interleaved
        # vars_flat structure: [x0, y0, r0, x1, y1, r1, ...]
        # x coords: 0, 3, 6... -> slice [::3]
        # y coords: 1, 4, 7... -> slice [1::3]
        # r coords: 2, 5, 8... -> slice [2::3]
        
        xs = vars_flat[0::3]
        ys = vars_flat[1::3]
        rs = vars_flat[2::3]
        
        # Compute pairwise distances
        # Using broadcasting
        # xs_i - xs_j
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dists = np.sqrt(dx**2 + dy**2)
        
        # Sum of radii
        sum_r = rs[:, None] + rs[None, :]
        
        # Constraint: dist - sum_r >= 0
        # We only need upper triangle to avoid duplicates and self-checks (0)
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        vals = (dists - sum_r)[mask]
        
        return vals

    def distance_constraints_jac(vars_flat):
        # Jacobian matrix of size (n*(n-1)/2, 3*n)
        # This is dense and expensive to compute fully if n is large, but for 26 it's manageable.
        # However, numerical jacobian might be safer/easier to implement bug-free.
        # SLSQP with finite differences for Jacobian is acceptable for N=26.
        return None 

    # Combine constraints
    # We can pass a list of constraint dicts or a single function.
    # Given the number of distance constraints (26*25/2 = 325), a single function returning a vector is better.
    
    constraints_list = [
        {
            'type': 'ineq',
            'fun': distance_constraints,
            # 'jac': distance_constraints_jac # Rely on numerical diff for distance constraints
        },
        {
            'type': 'ineq',
            'fun': boundary_constraints,
            'jac': boundary_constraints_jac
        }
    ]

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r

    # Initial guess
    x0 = np.zeros(3 * n)
    x0[0::3] = centers_init[:, 0]
    x0[1::3] = centers_init[:, 1]
    x0[2::3] = radii_init

    # --- 3. Solve ---
    # Use SLSQP
    # To improve performance, we might limit maxiter, but default is usually fine.
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        jac=gradient, 
        bounds=bounds, 
        constraints=constraints_list,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )

    # --- 4. Extract Result ---
    if res.success:
        final_centers = np.column_stack((res.x[0::3], res.x[1::3]))
        final_radii = res.x[2::3]
    else:
        # Fallback to initial if optimization fails (unlikely)
        final_centers = centers_init
        final_radii = radii_init

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
