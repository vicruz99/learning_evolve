import numpy as np
from scipy.optimize import differential_evolution

def compute_radii_array(centers):
    """Compute the maximum possible radius for each circle given fixed centers."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        # Start with boundary constraints
        r = centers[i, 0]
        if 1.0 - centers[i, 0] < r: r = 1.0 - centers[i, 0]
        if centers[i, 1] < r: r = centers[i, 1]
        if 1.0 - centers[i, 1] < r: r = 1.0 - centers[i, 1]
        
        # Constrain by pairwise distances
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                if d * 0.5 < r:
                    r = d * 0.5
        radii[i] = r
    return radii

def objective(x):
    """Objective function for optimization: minimize negative sum of radii."""
    return -np.sum(compute_radii_array(x.reshape(26, 2)))

def run_packing():
    # Define bounds for each coordinate in the unit square
    bounds = [(0.0, 1.0)] * 52
    
    # 1. Global search using Differential Evolution
    res = differential_evolution(
        objective, 
        bounds, 
        popsize=25, 
        maxiter=400, 
        polish=True, 
        seed=42, 
        tol=1e-9
    )
    centers = res.x.reshape(26, 2)
    
    # 2. Local refinement using force-directed relaxation
    lr = 0.005
    for step in range(2000):
        radii = compute_radii_array(centers)
        forces = np.zeros_like(centers)
        
        for i in range(26):
            fx, fy = 0.0, 0.0
            
            # Repulsion from boundaries
            if centers[i, 0] < radii[i]: fx += (radii[i] - centers[i, 0]) * 10.0
            if centers[i, 0] > 1.0 - radii[i]: fx -= (centers[i, 0] - (1.0 - radii[i])) * 10.0
            if centers[i, 1] < radii[i]: fy += (radii[i] - centers[i, 1]) * 10.0
            if centers[i, 1] > 1.0 - radii[i]: fy -= (centers[i, 1] - (1.0 - radii[i])) * 10.0
            
            # Repulsion from other circles
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                overlap = radii[i] + radii[j] - dist
                
                if overlap > 0:
                    if dist < 1e-12:
                        dx, dy = 0.001, 0.001
                        dist = np.hypot(dx, dy)
                    f = overlap * 5.0 / dist
                    fx += dx * f
                    fy += dy * f
                    forces[j, 0] -= dx * f
                    forces[j, 1] -= dy * f
                    
            forces[i] = [fx, fy]
            
        # Update positions and project back to [0, 1]
        centers += forces * lr
        centers = np.clip(centers, 0.0, 1.0)
        lr *= 0.998  # Gradually reduce step size
        
    # Final radius computation
    final_radii = compute_radii_array(centers)
    total_sum = np.sum(final_radii)
    
    return centers, final_radii, total_sum
