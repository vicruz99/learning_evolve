import numpy as np
import scipy.optimize as opt

def generate_hexagonal_cluster(n):
    """Generates n points in a hexagonal lattice centered at origin."""
    points = []
    # Generate points in a range that covers enough neighbors
    limit = 10
    for i in range(-limit, limit + 1):
        for j in range(-limit, limit + 1):
            x = i + 0.5 * j
            y = (np.sqrt(3) / 2) * j
            points.append((x, y))
    
    # Sort by distance from origin
    points.sort(key=lambda p: p[0]**2 + p[1]**2)
    return np.array(points[:n])

def rotate_points(points, angle_rad):
    """Rotates points around their centroid."""
    centroid = np.mean(points, axis=0)
    points_centered = points - centroid
    cos_a = np.cos(angle_rad)
    sin_a = np.sin(angle_rad)
    rotation_matrix = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
    points_rotated = (rotation_matrix @ points_centered.T).T
    return points_rotated

def find_max_radius_for_cluster(points, angle_rad, square_size=1.0):
    """
    Finds the maximum radius r for a given set of rotated points 
    to fit in a square of size square_size.
    """
    pts = rotate_points(points, angle_rad)
    
    # Calculate bounding box of the points
    min_x, min_y = np.min(pts, axis=0)
    max_x, max_y = np.max(pts, axis=0)
    
    width = max_x - min_x
    height = max_y - min_y
    
    # The span of centers must fit in (square_size - 2*r)
    # width * scale + 2*r <= square_size? 
    # Actually, if points are lattice coords with spacing 1, 
    # real coordinates are r * pts (if we consider 2r distance? No.)
    # Let's define lattice spacing as 1 unit corresponds to 2r distance.
    # So real coords = r * pts.
    # Then distance between neighbors is r. Wait, lattice neighbors are dist 1.
    # We want distance 2r. So scale factor is 2r.
    # Real coords = 2r * pts.
    # Bounding box size = 2r * width.
    # Condition: 2r * width + 2r <= square_size => 2r(width + 1) <= square_size
    # r <= square_size / (2 * (width + 1))
    
    # However, we need to check both x and y constraints
    r_x = square_size / (2 * (width + 1))
    r_y = square_size / (2 * (height + 1))
    
    return min(r_x, r_y), pts

def get_initial_packing(n=26):
    """Generates an initial packing using hexagonal lattice optimization."""
    pts_lattice = generate_hexagonal_cluster(n)
    
    best_r = 0
    best_pts_scaled = None
    
    # Test rotations from 0 to 30 degrees (pi/6)
    angles = np.linspace(0, np.pi/6, 61)
    
    for angle in angles:
        r, pts_rotated = find_max_radius_for_cluster(pts_lattice, angle)
        if r > best_r:
            best_r = r
            # Scale points to actual coordinates
            # Scale factor is 2r (since lattice dist 1 -> real dist 2r)
            best_pts_scaled = 2 * r * pts_rotated
            
    # Center the configuration in the unit square
    min_x, min_y = np.min(best_pts_scaled, axis=0)
    max_x, max_y = np.max(best_pts_scaled, axis=0)
    width = max_x - min_x
    height = max_y - min_y
    
    # Target center (0.5, 0.5)
    current_center = (min_x + max_x) / 2, (min_y + max_y) / 2
    shift_x = 0.5 - current_center[0]
    shift_y = 0.5 - current_center[1]
    
    centers = best_pts_scaled + np.array([shift_x, shift_y])
    radii = np.full(n, best_r)
    
    return centers, radii

def validate_and_fix(centers, radii):
    """Basic validation and repair for numerical stability."""
    n = len(radii)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Clip to boundaries
        r = min(r, x, 1-x, y, 1-y)
        radii[i] = max(0, r)
        centers[i][0] = np.clip(x, radii[i], 1 - radii[i])
        centers[i][1] = np.clip(y, radii[i], 1 - radii[i])
        
    # Resolve simple overlaps by shrinking radii if necessary
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-9:
                overlap = (radii[i] + radii[j] - dist) / 2 + 1e-9
                radii[i] -= overlap
                radii[j] -= overlap
                radii[i] = max(0, radii[i])
                radii[j] = max(0, radii[j])
    return centers, radii

def run_optimization(centers, radii):
    """
    Uses scipy to optimize positions and radii.
    Objective: Maximize sum(radii)
    Constraints: Non-overlap, boundaries.
    Implemented via penalty function.
    """
    n = len(radii)
    vars_init = np.concatenate([centers.flatten(), radii])
    bounds = []
    for i in range(n):
        # x bounds [0, 1]
        bounds.append((0, 1))
        # y bounds [0, 1]
        bounds.append((0, 1))
        # r bounds [0, 1]
        bounds.append((1e-6, 1.0))

    def objective(vars_flat):
        c = vars_flat[:2*n].reshape(n, 2)
        r = vars_flat[2*n:]
        return -np.sum(r) # Minimize negative sum

    def penalty(vars_flat):
        c = vars_flat[:2*n].reshape(n, 2)
        r = vars_flat[2*n:]
        p = 0.0
        
        # Boundary penalty
        for i in range(n):
            x, y = c[i]
            ri = r[i]
            if x - ri < 0: p += (x - ri)**2
            if x + ri > 1: p += (x + ri - 1)**2
            if y - ri < 0: p += (y - ri)**2
            if y + ri > 1: p += (y + ri - 1)**2
            
        # Overlap penalty

            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j])**2))
                req_dist = r[i] + r[j]
                if dist < req_dist:
                    p += (req_dist - dist)**2
                    
        return p

        return objective(vars_flat) + 1000 * penalty(vars_flat)

                       options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
    
    c_opt = res.x[:2*n].reshape(n, 2)
    r_opt = res.x[2*n:]
    
    return c_opt, r_opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
