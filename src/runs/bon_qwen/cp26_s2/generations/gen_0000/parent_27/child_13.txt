import numpy as np
import scipy.optimize as opt

N_CIRCLES = 26

def compute_objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def compute_constraints(vars):
    """Compute all inequality constraints: g(x) >= 0."""
    n = N_CIRCLES
    n_con = 4 * n + n * (n - 1) // 2
    con_vals = np.empty(n_con)
    idx = 0
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    for i in range(n):
        x = vars[3 * i]
        y = vars[3 * i + 1]
        r = vars[3 * i + 2]
        con_vals[idx] = x - r
        con_vals[idx + 1] = 1.0 - x - r
        con_vals[idx + 2] = y - r
        con_vals[idx + 3] = 1.0 - y - r
        idx += 4

    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        xi = vars[3 * i]
        yi = vars[3 * i + 1]
        ri = vars[3 * i + 2]
        for j in range(i + 1, n):
            dx = xi - vars[3 * j]
            dy = yi - vars[3 * j + 1]
            dr = ri + vars[3 * j + 2]
            con_vals[idx] = dx * dx + dy * dy - dr * dr
            idx += 1
    return con_vals

def generate_hex_init():
    """Generate a feasible hexagonal lattice initialization."""
    pts = []
    s = 0.15
    for row in range(15):
        y = row * s * np.sqrt(3) / 2
        shift = s / 2 if row % 2 == 1 else 0
        for col in range(15):
            x = col * s + shift
            pts.append([x, y])
            if len(pts) >= N_CIRCLES:
                break
        if len(pts) >= N_CIRCLES:
            break
            
    pts = np.array(pts[:N_CIRCLES])
    mx, my = pts.min(axis=0)
    Mx, My = pts.max(axis=0)
    pts = pts - [mx, my]
    span = max(Mx - mx, My - my)
    if span > 1e-6:
        pts *= 0.8 / span
    pts += 0.05
    
    # Compute min distance to set safe initial radii
    min_d = 1.0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            d = np.sqrt(np.sum((pts[i] - pts[j]) ** 2))
            if d < min_d:
                min_d = d
                
    r0 = 0.4 * min_d
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    x0[2::3] = r0
    return x0

def generate_random_init():
    """Generate a feasible random initialization."""
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = np.random.rand(N_CIRCLES) * 0.8 + 0.1
    x0[1::3] = np.random.rand(N_CIRCLES) * 0.8 + 0.1
    # Start with small radii to guarantee feasibility
    x0[2::3] = 0.01 * np.ones(N_CIRCLES)
    return x0

def check_validity(centers, radii):
    """Mirrors the validation function to ensure feasibility."""
    n = len(radii)
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing():
    best_sum = 0.0
    best_centers = np.zeros((N_CIRCLES, 2))
    best_radii = np.zeros(N_CIRCLES)
    
    # Define bounds: x,y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0) for _ in range(3 * N_CIRCLES)]
    for i in range(N_CIRCLES):
        bounds[3 * i + 2] = (1e-6, 0.5)

    constraint_dict = {'type': 'ineq', 'fun': compute_constraints}
    
    trials = 30
    for t in range(trials):
        if t % 3 == 0:
            x0 = generate_hex_init()
        else:
            x0 = generate_random_init()
            
        # Slight perturbation to diversify restarts
        x0 += np.random.normal(0, 0.005, size=x0.shape)
        x0[0::3] = np.clip(x0[0::3], 0.01, 0.99)
        x0[1::3] = np.clip(x0[1::3], 0.01, 0.99)
        x0[2::3] = np.clip(x0[2::3], 1e-5, 0.2)

        try:
            res = opt.minimize(
                compute_objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint_dict,
                options={'maxiter': 1500, 'ftol': 1e-10, 'disp': False}
            )
            
            centers = res.x.reshape((N_CIRCLES, 3))[:, :2]
            radii = res.x.reshape((N_CIRCLES, 3))[:, 2]
            
            if check_validity(centers, radii):
                current_sum = np.sum(radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
        except Exception:
            continue
            
    # Final refinement pass
    final_x0 = np.zeros(3 * N_CIRCLES)
    final_x0[0::3] = best_centers[:, 0]
    final_x0[1::3] = best_centers[:, 1]
    final_x0[2::3] = best_radii
    
    final_x0 += np.random.normal(0, 0.002, size=final_x0.shape)
    final_x0[2::3] = np.clip(final_x0[2::3], 1e-5, best_radii.max() + 0.01)
    
    try:
        res = opt.minimize(
            compute_objective,
            final_x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint_dict,
            options={'maxiter': 2000, 'ftol': 1e-11, 'disp': False}
        )
        if check_validity(res.x.reshape((N_CIRCLES, 3))[:, :2], res.x.reshape((N_CIRCLES, 3))[:, 2]):
            s = np.sum(res.x[2::3])
            if s > best_sum:
                best_sum = s
                best_centers = res.x.reshape((N_CIRCLES, 3))[:, :2]
                best_radii = res.x.reshape((N_CIRCLES, 3))[:, 2]
    except Exception:
        pass

    return best_centers, best_radii, float(best_sum)
