import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Uses LP sensitivity analysis (dual variables) to guide gradient ascent on centers.
    """
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Pre-define pair indices for the 26 circles
    pair_indices = []
    for i in range(n):
        for j in range(i + 1, n):
            pair_indices.append((i, j))
    n_pairs = len(pair_indices) # 325
    n_constraints = n_pairs + 4 * n # 325 + 104 = 429

    def solve_lp_and_get_gradient(centers):
        """
        Solves the LP for optimal radii given centers and computes the gradient 
        of the sum of radii with respect to the centers.
        """
        # Initialize LP components
        A_ub = np.zeros((n_constraints, n))
        b_ub = np.zeros(n_constraints)
        
        # 1. Pairwise constraints: r_i + r_j <= dist(i, j)
        for k, (i, j) in enumerate(pair_indices):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            b_ub[k] = dist
            
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0

        # 2. Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
        for i in range(n):
            x, y = centers[i]
            row_base = n_pairs + 4 * i
            
            # r_i <= x
            A_ub[row_base, i] = 1.0
            b_ub[row_base] = x
            
            # r_i <= 1 - x
            A_ub[row_base + 1, i] = 1.0
            b_ub[row_base + 1] = 1.0 - x
            
            # r_i <= y
            A_ub[row_base + 2, i] = 1.0
            b_ub[row_base + 2] = y
            
            # r_i <= 1 - y
            A_ub[row_base + 3, i] = 1.0
            b_ub[row_base + 3] = 1.0 - y

        # Objective: Maximize sum(r) -> Minimize -sum(r)
        c_obj = -np.ones(n)
        
        # Solve LP
        # Using 'highs' for efficiency and reliable marginals
        try:
            res = linprog(c=c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        except:
            return None, None, np.zeros((n, 2))
            
        if not res.success:
            return None, None, np.zeros((n, 2))
            
        radii = res.x
        sum_radii = np.sum(radii)
        marginals = res.ineqlin.marginals

        # Compute Gradient of Sum of Radii wrt Centers
        grad = np.zeros((n, 2))
        
        # Pairwise contributions
        for k, (i, j) in enumerate(pair_indices):
            mu = marginals[k]
            if mu > 1e-9:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist > 1e-12:
                    ux = dx / dist
                    uy = dy / dist
                    grad[i, 0] += mu * ux
                    grad[i, 1] += mu * uy
                    grad[j, 0] -= mu * ux
                    grad[j, 1] -= mu * uy
        
        # Boundary contributions
        for i in range(n):
            row_base = n_pairs + 4 * i
            # Marginals for r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
            mu_x_pos = marginals[row_base]
            mu_x_neg = marginals[row_base + 1]
            mu_y_pos = marginals[row_base + 2]
            mu_y_neg = marginals[row_base + 3]
            
            # Gradient wrt x: mu(x) * d(x)/dx + mu(1-x) * d(1-x)/dx = mu_x_pos - mu_x_neg
            grad[i, 0] += mu_x_pos - mu_x_neg
            # Gradient wrt y: mu(y) - mu(1-y)
            grad[i, 1] += mu_y_pos - mu_y_neg

        return radii, sum_radii, grad

    def optimize_from_start(start_centers):
        centers = start_centers.copy()
        best_local_sum = 0.0
        best_local_radii = None
        best_local_centers = None

        learning_rate = 0.02
        decay = 0.995
        lr = learning_rate
        iterations = 500

                break

                best_local_sum = curr_sum
                best_local_radii = radii.copy()
                best_local_centers = centers.copy()

            centers += lr * grad
            centers = np.clip(centers, 0.0, 1.0)
            lr *= decay

        return best_local_centers, best_local_radii, best_local_sum

    np.random.seed(42)
    n_runs = 15

            start += np.random.uniform(-0.02, 0.02, start.shape)

            start = np.random.uniform(0.1, 0.9, (n, 2))

            start += np.random.uniform(-0.05, 0.05, start.shape)
            start = np.clip(start, 0.01, 0.99)

        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt

    # Final validation and clamping
    if best_centers is None:
        # Fallback if something goes wrong
        best_centers = np.random.rand(n, 2)

    return best_centers, best_radii, best_sum
