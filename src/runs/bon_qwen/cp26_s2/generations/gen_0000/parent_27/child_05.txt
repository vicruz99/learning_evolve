import numpy as np
from scipy.optimize import minimize


def boundary_constraint_x_ge_r(x, i, n):
    xi = 2 * i
    ri = 2 * n + i
    return x[xi] - x[ri]


def boundary_constraint_x_le_1mr(x, i, n):
    xi = 2 * i
    ri = 2 * n + i
    return 1 - x[xi] - x[ri]


def boundary_constraint_y_ge_r(x, i, n):
    yi = 2 * i + 1
    ri = 2 * n + i
    return x[yi] - x[ri]


def boundary_constraint_y_le_1mr(x, i, n):
    yi = 2 * i + 1
    ri = 2 * n + i
    return 1 - x[yi] - x[ri]


def overlap_constraint(x, i, j, n):
    xi, yi = 2 * i, 2 * i + 1
    xj, yj = 2 * j, 2 * j + 1
    ri, rj = 2 * n + i, 2 * n + j
    dx = x[xi] - x[xj]
    dy = x[yi] - x[yj]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[ri] - x[rj]


def objective_func(x, n):
    return -np.sum(x[2 * n:])


def initialize_hexagonal(n, rows, cols):
    centers = []
    idx = 0
    spacing_x = 1.0 / (cols + 1)
    spacing_y = 1.0 / (rows + 1)
    for row in range(rows + 1):
        for col in range(cols + 1):
            if idx >= n:
                break
            x = (col + 1) * spacing_x
            y = (row + 1) * spacing_y
            if row % 2 == 1:
                x += spacing_x / 2
            if 0 < x < 1 and 0 < y < 1:
                centers.append([x, y])
                idx += 1
        if idx >= n:
            break
    centers = np.array(centers[:n])
    min_sp = min(spacing_x, spacing_y * np.sqrt(3) / 2)
    radii = np.full(n, min_sp / 2 - 0.002)
    return centers, radii


def initialize_grid(n, rows, cols):
