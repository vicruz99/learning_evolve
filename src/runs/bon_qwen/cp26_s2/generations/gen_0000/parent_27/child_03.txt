import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # We will perform multiple optimization attempts
    n_attempts = 50
    
    # Penalty weights
    # These need to be high enough to enforce constraints but not break optimization
    penalty_overlap = 1000.0
    penalty_boundary = 1000.0
    
    # Define the objective function
    # We minimize: -sum(radii) + penalty(violations)
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    def objective(vars_array):
        centers = vars_array[:2*n].reshape(n, 2)
        radii = vars_array[2*n:]
        
        # Main objective: maximize sum of radii -> minimize negative sum
        obj = -np.sum(radii)
        
        # Penalty for boundary violations
        # Circle i at (x, y) with radius r
        # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        # Violations: max(0, r-x), etc.
        
        # Vectorized boundary check
        x = centers[:, 0]
        y = centers[:, 1]
        r = radii
        
        # Lower bounds: x >= r -> r - x <= 0
        viol_lx = np.maximum(0, r - x)
        # Upper bounds: x <= 1-r -> x + r - 1 <= 0
        viol_ux = np.maximum(0, x + r - 1.0)
        # Lower bounds y
        viol_ly = np.maximum(0, r - y)
        # Upper bounds y
        viol_uy = np.maximum(0, y + r - 1.0)
        
        penalty = penalty_boundary * np.sum(viol_lx**2 + viol_ux**2 + viol_ly**2 + viol_uy**2)
        
        # Penalty for overlaps
        # For each pair (i, j), dist >= ri + rj
        # dist^2 >= (ri + rj)^2
        # Violation: (ri + rj) - dist
        
        # Compute pairwise distances
        # This is O(N^2), N=26 is small (650 pairs)
        # Using broadcasting for speed
        # centers shape (n, 2)
        # diff shape (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Lower triangle indices to avoid double counting and self
        i_indices, j_indices = np.triu_indices(n, k=1)
        
        sum_r = radii[i_indices] + radii[j_indices]
        dists_ij = dists[i_indices, j_indices]
        
        # Overlap violation
        overlap_viol = np.maximum(0, sum_r - dists_ij)
        penalty += penalty_overlap * np.sum(overlap_viol**2)
        
        return obj + penalty

    # Bounds for variables

    bounds = []

        bounds.append((0.0, 0.5)) # r

        radii_init = np.ones(n) * 0.02

        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        # Run optimizer
        # L-BFGS-B is good for bound-constrained problems
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,

                        valid = False
                        break

                            if dist < radii_res[i] + radii_res[j] - 1e-12:
                                valid = False
                                break
                        if not valid: break
            
            if valid:

                    best_centers = centers_res.copy()
                    best_radii = radii_res.copy()
