import numpy as np
from scipy.optimize import minimize

def compute_constraints(x):
    n = 26
    C = x.reshape(n, 3)
    X, Y, R = C[:, 0], C[:, 1], C[:, 2]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    bound_constr = np.concatenate([
        X - R,
        1.0 - X - R,
        Y - R,
        1.0 - Y - R
    ])
    
    # Pairwise non-overlap constraints: dist^2 >= (r1+r2)^2
    X_i, X_j = X[:, None], X[None, :]
    Y_i, Y_j = Y[:, None], Y[None, :]
    R_i, R_j = R[:, None], R[None, :]
    
    dist_sq = (X_i - X_j)**2 + (Y_i - Y_j)**2
    rad_sum_sq = (R_i + R_j)**2
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    pair_constr = dist_sq[i_idx, j_idx] - rad_sum_sq[i_idx, j_idx]
        
    return np.concatenate([bound_constr, pair_constr])

def run_packing():
    n = 26
    
    # Helper to build x0 vector
    def build_x0(centers, radii):
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
        return x0

    # 1. Hexagonal initial guess
    row_counts = [6, 5, 6, 5, 4]
    centers_hex = []
    r_init = 0.06
    y = 0.15
    dy = 0.15 * np.sqrt(3)
    idx = 0
    for i, cnt in enumerate(row_counts):
        x_start = 0.15 + (i % 2) * 0.075
        for j in range(cnt):
            if idx < n:
                x = x_start + j * 0.15
                centers_hex.append([x, y])
                idx += 1
        y += dy
    centers_hex = np.array(centers_hex[:n])
    radii_hex = np.full(n, r_init)
    x0_hex = build_x0(centers_hex, radii_hex)
    
    # 2. Grid initial guess
    centers_grid = []
    for r in range(6):
        for c in range(5):
            if len(centers_grid) < n:
                centers_grid.append([0.12 + c*0.18, 0.1 + r*0.18])
    centers_grid = np.array(centers_grid)
    radii_grid = np.full(n, 0.05)
    x0_grid = build_x0(centers_grid, radii_grid)
    
    # 3. Perturbed hex guess
    np.random.seed(42)
    x0_pert = x0_hex + np.random.normal(0, 0.015, 3*n)
    x0_pert[2::3] = np.clip(x0_pert[2::3], 0.02, 0.4) # Ensure positive radii
    
    def objective(x):
        return -np.sum(x[2::3])
        
    cons = {'type': 'ineq', 'fun': compute_constraints}
    bounds = [(0.0, 1.0) if k % 3 != 2 else (0.0, 0.5) for k in range(3*n)]
    
    best_val = -np.inf
    best_x = x0_hex
    
    initial_guesses = [x0_hex, x0_grid, x0_pert]
    
    for guess in initial_guesses:
        try:
            res = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'ftol': 1e-13, 'maxiter': 5000, 'disp': False})
            if -res.fun > best_val:
                best_val = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue
            
    centers_opt = best_x.reshape(n, 3)[:, :2]
    radii_opt = best_x.reshape(n, 3)[:, 2]
    
    # Safety clipping for numerical stability
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    radii_opt = np.clip(radii_opt, 0.0, 0.5)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
