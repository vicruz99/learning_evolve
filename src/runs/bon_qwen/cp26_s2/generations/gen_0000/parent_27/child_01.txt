import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_constraints(n):
    """Returns a list of constraint dictionaries for scipy.optimize"""
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        # x_i >= r_i  => x_i - r_i >= 0
        idx_x = 3 * i
        idx_r = 3 * i + 2
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r],
            'jac': lambda v, i=i: np.zeros(3 * n, dtype=float) 
        })
        # 1 - x_i >= r_i => 1 - x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r],
            'jac': lambda v, i=i: np.zeros(3 * n, dtype=float)
        })
        # y_i >= r_i
        idx_y = 3 * i + 1
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r],
            'jac': lambda v, i=i: np.zeros(3 * n, dtype=float)
        })
        # 1 - y_i >= r_i
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r],
            'jac': lambda v, i=i: np.zeros(3 * n, dtype=float)
        })

    # Pairwise non-overlap constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # => (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3 * i, 3 * i + 1, 3 * i + 2
            idx_xj, idx_yj, idx_rj = 3 * j, 3 * j + 1, 3 * j + 2
            
            def constraint_fun(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

            # Jacobian
            def constraint_jac(v, i=i, j=j):
                jac = np.zeros(3 * n)
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                
                dx = xi - xj
                dy = yi - yj
                r_sum = ri + rj
                
                jac[idx_xi] = 2 * dx
                jac[idx_yi] = 2 * dy
                jac[idx_ri] = -2 * r_sum
                
                jac[idx_xj] = -2 * dx
                jac[idx_yj] = -2 * dy
                jac[idx_rj] = -2 * r_sum
                return jac

            constraints.append({
                'type': 'ineq',
                'fun': constraint_fun,
                'jac': constraint_jac
            })
            
    return constraints

def objective_function(v, n):
    """Objective: Minimize negative sum of radii"""
    radii = v[2::3]
    return -np.sum(radii)

def generate_hexagonal_init(n, rows=None):
    """Generate initial positions in a hexagonal pattern"""
    if rows is None:
        # Estimate rows for n circles
        # Hexagonal packing density ~0.9. Area ~ n * pi * r^2.
        # Rough estimate for grid-like structure
        rows = int(np.ceil(np.sqrt(n * np.sqrt(3) / 2)))
    
    positions = []
    count = 0
    r_est = 0.09 # Estimated radius for packing
    
    # Hexagonal row height is sqrt(3)/2 * diameter = sqrt(3)*r
    # But we don't know r yet, so we just place centers and let optimizer scale r
    # Let's assume a spacing of 0.2 for initialization
    
    spacing = 0.2
    h_spacing = spacing * np.sqrt(3) / 2
    
    y = spacing # start from bottom
    row_idx = 0
    while count < n:
        # Determine x start
        if row_idx % 2 == 0:
            x_start = spacing
        else:
            x_start = spacing * 1.5 # Shifted
        
        x = x_start
        while x < 1.0 - spacing and count < n:
            positions.append([x, y])
            count += 1
            x += spacing
        y += h_spacing
        row_idx += 1
        
    return np.array(positions[:n])

def generate_grid_init(n):
    """Generate initial positions in a grid"""
    # Find grid dimensions close to sqrt(n)
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    
    positions = []
    count = 0
    for r in range(rows):
        for c in range(cols):
            if count >= n:
                break
            # Place in center of grid cells
            x = (c + 0.5) / cols
            y = (r + 0.5) / rows
            positions.append([x, y])
            count += 1
    return np.array(positions)
