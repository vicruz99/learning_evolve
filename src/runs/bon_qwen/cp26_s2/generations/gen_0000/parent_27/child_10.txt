2.  **Local Optimization**: Use `scipy.optimize.minimize` with the `trust-constr` method. This solver is robust for handling the complex non-linear inequality constraints required for circle packing (non-overlap and boundary containment).

    *   **Constraints**:

import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_constraints(centers_init, radii_init):
    """
    Define the constraints for the optimization problem.
    """
    n = len(radii_init)
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # This is equivalent to x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
    # We define these as NonlinearConstraint or NonlinearConstraint is for equality/inequality
    # trust-constr accepts NonlinearConstraint(type='ineq', fun, lb, ub) where fun(x) >= lb
    
    # 1. r_i <= x_i  => x_i - r_i >= 0
    def fun_x_lower(x):
        c = x[0::3]
        r = x[2::3]
        return c - r
    constraints.append(opt.nonlinear_constraint.NonlinearConstraint(fun_x_lower, 0, np.inf))

    # 2. x_i <= 1 - r_i => x_i + r_i <= 1
    def fun_x_upper(x):
        c = x[0::3]
        r = x[2::3]
        return c + r
    constraints.append(opt.nonlinear_constraint.NonlinearConstraint(fun_x_upper, -np.inf, 1))

    # 3. r_i <= y_i => y_i - r_i >= 0
    def fun_y_lower(x):
        c = x[1::3]
        r = x[2::3]
        return c - r
    constraints.append(opt.nonlinear_constraint.NonlinearConstraint(fun_y_lower, 0, np.inf))

    # 4. y_i <= 1 - r_i => y_i + r_i <= 1
    def fun_y_upper(x):
        c = x[1::3]
        r = x[2::3]
        return c + r
    constraints.append(opt.nonlinear_constraint.NonlinearConstraint(fun_y_upper, -np.inf, 1))
    
    # 5. Non-overlap: dist(i,j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # This is a vectorized function
    
    def fun_non_overlap(x):
        # x shape (3n,)
        # centers shape (n, 2), radii shape (n,)
        c = np.column_stack((x[0::3], x[1::3]))
        r = x[2::3]
        
        # Compute pairwise squared distances
        # c: (n, 2)
        # c[:, None, :] - c[None, :, :] -> (n, n, 2)
        # np.sum(...**2, axis=2) -> (n, n)
        diff = c[:, None, :] - c[None, :, :]
        sq_dist = np.sum(diff**2, axis=2)
        
        # r_i + r_j
        r_sum = r[:, None] + r[None, :]
        
        # Constraint: sq_dist >= r_sum^2
        val = sq_dist - r_sum**2
        
        # We only need upper triangular part to avoid duplicates and self
        # But trust-constr expects a vector.
        # Flatten and return.
        # Mask for upper triangular
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        return val[mask]

    constraints.append(opt.nonlinear_constraint.NonlinearConstraint(fun_non_overlap, 0, np.inf))
    
    return constraints

def objective_function(x):
    """
    Objective: Maximize sum of radii.
    Minimize negative sum.
    """
    r = x[2::3]
    return -np.sum(r)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Helper to create initial hexagonal packing
    def create_initial_config(perturbation=0.0):
        centers = []
        # Hexagonal packing parameters
        # Assume initial radius 0.1, diameter 0.2
        # Horizontal spacing 0.2, vertical spacing 0.1732
        # We want to fill 1x1 square.

        # Or just random
        
        # Better: Start with a random valid config

            rng = np.random.RandomState(perturbation) # different seed

        rows = 6
        cols = 5

        dx = 1.0 / (cols + 1)
        dy = 1.0 / (rows + 1) * 1.5 # slightly tighter vertical for hex

        y = 0.2

            row_count = 0

                centers.append([x + (row_count % 2) * 0.1, y])
                x += 0.2
                count += 1
                row_count += 1
            y += dy
            if count >= n:
                break

            centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])

        centers = centers[:n]
        
        # Perturb slightly

            centers = np.array(centers) + rng.uniform(-0.05, 0.05, size=(n, 2))
            # Clip
            centers = np.clip(centers, 0.1, 0.9)
        
        # Initial radii
        radii = np.full(n, 0.05)
        
        return centers, radii

    # Function to optimize
    def solve(centers_init, radii_init):
        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(n * 3)
        for i in range(n):

        # Actually r is constrained by geometry, but bounds help solver
        lb = np.zeros(n * 3)
        ub = np.ones(n * 3)
        # r can be at most 0.5
        lb[2::3] = 0
        ub[2::3] = 0.5
        # x, y bounds
        lb[0::3] = 0; ub[0::3] = 1
        lb[1::3] = 0; ub[1::3] = 1
        
        # Constraints
        constraints = get_constraints(centers_init, radii_init)

            res = opt.minimize(
                objective_function,
                x0,
                method='trust-constr',
                bounds=list(zip(lb, ub)),
                constraints=constraints,
                options={'maxiter': 200, 'verbose': 0}
            )
            return res.x, res.success
        except Exception:
            return None, False

            # Check validity (numerical noise might cause tiny violations)
            # Adjust radii slightly to ensure strict validity if needed
            # But trust-constr usually respects constraints well.
            
            # Validate
            if validate_packing(c_final, r_final):
                s = np.sum(r_final)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_final
                    best_radii = r_final
            else:
                # Try to fix small violations by shrinking radii slightly
                # This is a heuristic fix
                # Not ideal but ensures we return something valid
                pass

    if best_centers is None:
        # Fallback to a simple grid
        best_centers, best_radii = create_initial_config(0)
        best_sum = np.sum(best_radii)
        # Ensure valid
        if not validate_packing(best_centers, best_radii):
             best_radii = np.full(26, 0.0)
             best_sum = 0.0

    # Final polish: Expand radii locally
    # If we have a valid packing, we can try to increase radii uniformly until they hit constraints
    # This is equivalent to finding the "bottleneck" constraints.
    # But since we optimized sum, this should be near optimal.
