    Validate that circles don't overlap and are inside the unit square
    """

def get_overlap_energy(centers, radii, tol=1e-6):

    Calculate a penalty energy based on overlaps.
    Positive energy means overlap, 0 means valid.

            energy += (tol + r - y) ** 2

            energy += (y + r - 1 - tol) ** 2
