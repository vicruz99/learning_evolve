def calculate_cost(centers_flat, radii_flat, n, penalty_weight):
    """
    Calculates the objective function: negative sum of radii + penalty terms.
    """
    centers = centers_flat.reshape((n, 2))
    radii = radii_flat
    
    cost = -np.sum(radii)
    
    # 1. Boundary Penalties
    # Constraints: r <= x <= 1-r, r <= y <= 1-r
    # i.e., x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    # We penalize violation squared.
    violations = []
    
    # Left wall: x >= r => r - x <= 0
    violations.extend(radii - centers[:, 0])
    # Right wall: 1 - x >= r => r - (1 - x) <= 0 => r + x - 1 <= 0
    violations.extend(radii + centers[:, 0] - 1)
    # Bottom wall: y >= r => r - y <= 0
    violations.extend(radii - centers[:, 1])
    # Top wall: 1 - y >= r => r + y - 1 <= 0
    violations.extend(radii + centers[:, 1] - 1)
    
    penalty_boundary = penalty_weight * np.sum(np.maximum(violations, 0)**2)
    
    # 2. Overlap Penalties
    # Constraints: ||c_i - c_j|| >= r_i + r_j
    # Violation: r_i + r_j - ||c_i - c_j|| > 0
    penalty_overlap = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty_overlap += penalty_weight * (overlap**2)
                
    return cost + penalty_boundary + penalty_overlap

def get_jacobian(centers_flat, radii_flat, n, penalty_weight):

        grad_centers[active_right, 0] += grad # d/dx
        grad_radii[active_right] += grad      # d/dr
        
    # Bottom wall: r - y <= 0. Violation v = r - y.
    v_bottom = radii - centers[:, 1]
    active_bottom = v_bottom > 0
    if np.any(active_bottom):
        grad = 2 * penalty_weight * v_bottom[active_bottom]
        grad_centers[active_bottom, 1] -= grad
        grad_radii[active_bottom] += grad
        
    # Top wall: r + y - 1 <= 0. Violation v = r + y - 1.
    v_top = radii + centers[:, 1] - 1
    active_top = v_top > 0
    if np.any(active_top):
        grad = 2 * penalty_weight * v_top[active_top]
        grad_centers[active_top, 1] += grad
        grad_radii[active_top] += grad
        
    # --- Overlap Gradients ---
    # Overlap: r_i + r_j - dist <= 0. Violation v = r_i + r_j - dist.
    # Term: w * v^2.
    # d/dr_i: w * 2v * 1. d/dr_j: w * 2v * 1.
    # d/dx_i: w * 2v * (-d(dist)/dx_i) = w * 2v * (-(x_i - x_j)/dist).
    # d/dy_i: w * 2v * (-(y_i - y_j)/dist).

            dx = centers[i, 0] - centers[j, 0]

            dist = np.sqrt(dx*dx + dy*dy)
            
            if dist < 1e-10:

            overlap = radii[i] + radii[j] - dist
            if overlap > 0:

                grad_radii[i] += term
                grad_radii[j] += term
                
                # Gradient for centers
                # Force direction is along the vector connecting them
                # To reduce overlap (increase dist), we move them apart.
                # The derivative of -dist w.r.t c_i is -(c_i - c_j)/dist.
                # The penalty term increases with overlap, so we want to decrease overlap.
                # Wait, cost = penalty. Gradient points in direction of increasing cost.
                # We want to minimize cost.
                # d(cost)/d(c_i) = 2*w*overlap * d(overlap)/d(c_i)
                # d(overlap)/d(c_i) = - (c_i - c_j) / dist
                # So gradient is -2*w*overlap * (c_i - c_j) / dist.
                # This means the force pushes c_i away from c_j (negative of vector).

                fy = -term * (dy / dist)
                
                grad_centers[i, 0] += fx
                grad_centers[i, 1] += fy
                grad_centers[j, 0] -= fx
                grad_centers[j, 1] -= fy
                
    return np.concatenate([grad_centers, grad_radii])

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
