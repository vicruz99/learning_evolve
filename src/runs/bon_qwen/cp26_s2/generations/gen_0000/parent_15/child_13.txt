import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the sum of radii for 26 circles in a unit square.
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Optimization helper functions
    def objective(vars):
        # vars contains flattened [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum of radii, so minimize negative sum
        radii = vars[2::3]
        return -np.sum(radii)

    def boundary_constraints(vars):
        constraints = []
        for i in range(n):
            x = vars[3 * i]
            y = vars[3 * i + 1]
            r = vars[3 * i + 2]
            # Circle must be inside [0, 1]
            constraints.append(x - r)      # left
            constraints.append(1 - (x + r)) # right
            constraints.append(y - r)      # bottom
            constraints.append(1 - (y + r)) # top
        return constraints

    def overlap_constraints(vars):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i = vars[3 * i], vars[3 * i + 1]
                x_j, y_j = vars[3 * j], vars[3 * j + 1]
                r_i, r_j = vars[3 * i + 2], vars[3 * j + 2]
                
                dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
                r_sum = r_i + r_j
                
                # dist >= r_i + r_j  =>  dist^2 >= (r_i + r_j)^2
                # dist^2 - (r_i + r_j)^2 >= 0
                constraints.append(dist_sq - r_sum**2)
        return constraints

    def get_constraints():
        bc = [{'type': 'ineq', 'fun': lambda v, k=k: boundary_constraints(v)[k]} for k in range(4 * n)]
        oc = [{'type': 'ineq', 'fun': lambda v, k=k: overlap_constraints(v)[k]} for k in range(n * (n - 1) // 2)]
        return bc + oc

    def run_optimization(init_vars):
        # Bounds for x, y in [0, 1], r >= 0
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r (max radius in unit square is 0.5)
        
        cons = get_constraints()
        
        # Use SLSQP for constrained optimization
        res = minimize(objective, init_vars, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 1000})
        return res

    # Run multiple trials with different initial configurations
    trials = 5

                # Space them out roughly

                y_pos = y_center + np.random.uniform(-0.02, 0.02)

                x_pos = np.clip(x_pos, 0.05, 0.95)
                y_pos = np.clip(y_pos, 0.05, 0.95)
                
                init_vars[3 * idx] = x_pos
                init_vars[3 * idx + 1] = y_pos
                init_vars[3 * idx + 2] = 0.09 # Initial radius guess
                
                idx += 1

            idx += 1

    # Final cleanup: ensure radii are non-negative (solver might drift slightly)
    if best_radii is not None:
        best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, best_sum
