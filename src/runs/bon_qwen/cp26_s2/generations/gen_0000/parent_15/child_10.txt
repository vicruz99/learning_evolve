import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, lam):
    """Compute the penalized objective function for circle packing."""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Vectorized pairwise distances
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # Lower triangle indices for unique circle pairs
    i, j = np.tril_indices(n, -1)
    min_d = r[i] + r[j]
    overlap = np.maximum(0, min_d - dist[i, j])**2
    
    # Boundary penalties (left, bottom, right, top)
    x, y = c[:, 0], c[:, 1]
    bound_pen = (np.maximum(0, r - x)**2 + np.maximum(0, r - y)**2 + 
                 np.maximum(0, x + r - 1)**2 + np.maximum(0, y + r - 1)**2)
                 
    return -np.sum(r) + lam * (np.sum(overlap) + np.sum(bound_pen))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    best_sum = -1.0
    best_vars = None
    
    # Variable bounds: x,y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Initialization strategies
    inits = []
    
    # 1. Hexagonal-like grid initialization
    c_grid = np.zeros((n, 2))
    idx = 0
    for row in range(5):
        cols = 6 if row == 4 else 5
        shift = 0.5 if row % 2 == 1 else 0.0
        for col in range(cols):
            c_grid[idx, 0] = (col + 0.5 + shift) / (cols + 1)
            c_grid[idx, 1] = (row + 0.5) / 5.0
            idx += 1
    inits.append(np.hstack([c_grid.ravel(), np.full(n, 0.08)]))
    
    # 2 & 3. Random initializations for diversity
    for _ in range(2):
        c_rand = np.random.rand(n, 2)
        r_rand = np.full(n, 0.05)
        inits.append(np.hstack([c_rand.ravel(), r_rand]))
        
    # Optimize each initialization
    for x0 in inits:
        curr = x0.copy()
        # Penalty annealing schedule
        for lam in [50.0, 500.0, 5000.0, 20000.0]:
            res = minimize(compute_objective, curr, args=(n, lam), 
                           method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-10, 'disp': False})
            curr = res.x
            
        r_res = curr[2*n:]
        s = np.sum(r_res)
        if s > best_sum:
            best_sum = s
            best_vars = curr.copy()
            
    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]
    
    # Final validity enforcement (handles tiny numerical drift)
    for _ in range(10):
        min_clearance = 1.0
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            min_clearance = min(min_clearance, x - r, y - r, 1 - x - r, 1 - y - r)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                min_clearance = min(min_clearance, d - radii[i] - radii[j])
                
        if min_clearance < -1e-12:
            # Scale radii down just enough to satisfy constraints
            avg_r = np.mean(radii) + 1e-9
            factor = 1.0 + min_clearance / avg_r
            radii *= factor
        else:
            break
            
    return centers, radii, float(np.sum(radii))
