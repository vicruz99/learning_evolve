import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n, penalty):
    """
    Computes the negative sum of radii plus penalty terms for overlaps and boundary violations.
    """
    pos = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    loss = -np.sum(r)
    
    # Overlap penalty: penalize when distance < sum of radii
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(pos[i,0]-pos[j,0], pos[i,1]-pos[j,1])
            overlap = r[i] + r[j] - d
            if overlap > 0:
                loss += penalty * overlap**2
                
    # Boundary penalty: penalize when circle extends outside [0,1]x[0,1]
    for i in range(n):
        v = r[i] - pos[i,0]
        if v > 0: loss += penalty * v**2
        v = pos[i,0] + r[i] - 1.0
        if v > 0: loss += penalty * v**2
        v = r[i] - pos[i,1]
        if v > 0: loss += penalty * v**2
        v = pos[i,1] + r[i] - 1.0
        if v > 0: loss += penalty * v**2
        
    return loss

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_sum = -np.inf
    best_vars = None
    
    # Multi-start optimization with deterministic variations
    for seed in range(5):
        positions = np.zeros((n, 2))
        radii = np.full(n, 0.09)
        
        # Hexagonal lattice initialization
        r_grid = 0.06
        idx = 0
        jitter = 0.005 * (seed + 1)
        
        for row in range(12):
            for col in range(12):
                if idx >= n: break
                x = (col + 0.5) * 2 * r_grid + jitter * np.sin(row * 0.5)
                y = (row + 0.5 * (col % 2)) * np.sqrt(3) * r_grid + jitter * np.cos(col * 0.5)
                if x < 0.95 and y < 0.95:
                    positions[idx] = [x, y]
                    idx += 1
            if idx >= n: break
            
        # Deterministically fill remaining slots if hex grid doesn't provide enough
        k = 0
        while idx < n:
            positions[idx] = [0.15 + 0.7 * (k % 5) / 4, 0.15 + 0.7 * (k // 5) / 4]
            radii[idx] = 0.08
            idx += 1
            k += 1
            
        x0 = np.concatenate([positions.flatten(), radii])
        
        # Continuation method: gradually increase penalty to strictly enforce constraints
        current_vars = x0.copy()
        for alpha in [100, 500, 2000, 10000]:
            res = minimize(compute_loss, current_vars, args=(n, alpha), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-11, 'gtol': 1e-6})
            current_vars = res.x
            
        current_sum = np.sum(current_vars[2*n:])
        if current_sum > best_sum:
            best_sum = current_sum
            best_vars = current_vars.copy()

    centers = best_vars[:2*n].reshape(n, 2)
    radii_final = best_vars[2*n:]
    
    # Strict validity enforcement
    radii_final = np.maximum(radii_final, 1e-7)
    centers[:, 0] = np.clip(centers[:, 0], radii_final, 1.0 - radii_final)
    centers[:, 1] = np.clip(centers[:, 1], radii_final, 1.0 - radii_final)
    
    # Iteratively resolve any remaining overlaps by shrinking radii
    for _ in range(20):
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii_final[i] + radii_final[j]:
                    reduce = (radii_final[i] + radii_final[j] - d) / 2.0 + 1e-9
                    radii_final[i] -= reduce
                    radii_final[j] -= reduce
                    changed = True
        if not changed:
            break
            
    radii_final = np.maximum(radii_final, 1e-7)
    sum_r = float(np.sum(radii_final))
    
    return centers, radii_final, sum_r
