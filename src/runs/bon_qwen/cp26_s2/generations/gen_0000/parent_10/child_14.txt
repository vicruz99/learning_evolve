def evaluate_constraints(vars):

    Computes boundary and pairwise constraints.
    Returns a flattened array of constraint values.
    Inequality constraints: c(x) >= 0.
    """
    n = 26
    centers = vars.reshape(-1, 3)[:, :2] # (26, 2)
    radii = vars.reshape(-1, 3)[:, 2]    # (26,)

    constraints = []

    # 1. Boundary constraints
    # x - r >= 0  => x >= r
    # 1 - x - r >= 0 => x + r <= 1
    # same for y
    # We formulate as:
    # x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    
    # Vectorized boundary constraints
    # Left
    constraints.append(centers[:, 0] - radii)
    # Right
    constraints.append(1.0 - centers[:, 0] - radii)
    # Bottom
    constraints.append(centers[:, 1] - radii)
    # Top
    constraints.append(1.0 - centers[:, 1] - radii)

    # 2. Pairwise non-overlap constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # dist_sq - (r_i + r_j)^2 >= 0
    
    # Compute pairwise squared distances
    # centers shape (26, 2)
    # diff shape (26, 26, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]

    # r_sum shape (26, 26)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint matrix
    pair_constraints = dist_sq - r_sum_sq

    # np.triu with k=1 gives upper triangle excluding diagonal
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    constraints.append(pair_constraints[mask])

    # Each element in the list is an array
    flat_constraints = []
    for c in constraints:
        flat_constraints.append(c.flatten())
    
    return np.concatenate(flat_constraints)
