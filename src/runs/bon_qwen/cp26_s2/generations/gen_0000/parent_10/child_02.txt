import numpy as np
from scipy.optimize import minimize


def compute_violation(centers, radii):
    """Compute total constraint violation squared."""
    n = len(radii)
    violations = 0.0

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        violations += max(0.0, r - x) ** 2
        violations += max(0.0, r - (1.0 - x)) ** 2
        violations += max(0.0, r - y) ** 2
        violations += max(0.0, r - (1.0 - y)) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            violations += max(0.0, radii[i] + radii[j] - dist) ** 2

    return violations


def objective_penalty(x, n_circles, penalty_w):
    """Objective: negative sum of radii + penalty for violations."""
    centers = x[:2 * n_circles].reshape(n_circles, 2)
    radii = np.maximum(x[2 * n_circles:], 1e-10)

    violation = compute_violation(centers, radii)
    return -np.sum(radii) + penalty_w * violation


def is_valid(centers, radii):
    """Check if packing is valid."""
    n = len(radii)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True


def force_based_optimize(centers, radii, n_circles):
    """Run force-based optimization to grow circles and resolve overlaps."""
    dt = 0.003
    growth_rate = 0.0003

    for iteration in range(80000):
        radii += growth_rate

        forces = np.zeros((n_circles, 2))

        # Boundary repulsion
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]

            if x - r < 0:
                forces[i, 0] += 200.0 * (r - x)
            if x + r > 1:
                forces[i, 0] -= 200.0 * (x + r - 1)
            if y - r < 0:
                forces[i, 1] += 200.0 * (r - y)
            if y + r > 1:
                forces[i, 1] -= 200.0 * (y + r - 1)

        # Overlap repulsion
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10

                if dist < radii[i] + radii[j]:
                    overlap = radii[i] + radii[j] - dist
                    force_mag = 200.0 * overlap / dist
                    forces[i, 0] += force_mag * dx
                    forces[i, 1] += force_mag * dy
                    forces[j, 0] -= force_mag * dx
                    forces[j, 1] -= force_mag * dy

        centers += dt * forces

        # Clip to valid range
        for i in range(n_circles):
            centers[i, 0] = max(radii[i] + 1e-12, min(1 - radii[i] - 1e-12, centers[i, 0]))
            centers[i, 1] = max(radii[i] + 1e-12, min(1 - radii[i] - 1e-12, centers[i, 1]))

        # Decay growth rate
        if iteration % 20000 == 0 and iteration > 0:
            growth_rate *= 0.85

    return centers, radii


def optimize_with_slsqp(centers, radii, n_circles):
    """Refine packing using scipy SLSQP optimizer."""
    x0 = np.concatenate([centers.flatten(), radii])

    def obj(x):
        return -np.sum(x[2 * n_circles:])

    constraints = []

    # Boundary constraints
    for i in range(n_circles):
        def make_c_x_min(idx):
            def c(x):
                return x[2 * idx] - x[2 * n_circles + idx]
            return c
        constraints.append({'type': 'ineq', 'fun': make_c_x_min(i)})

        def make_c_x_max(idx):
            def c(x):
                return 1 - x[2 * idx] - x[2 * n_circles + idx]
            return c
        constraints.append({'type': 'ineq', 'fun': make_c_x_max(i)})

        def make_c_y_min(idx):
            def c(x):
                return x[2 * idx + 1] - x[2 * n_circles + idx]
            return c
        constraints.append({'type': 'ineq', 'fun': make_c_y_min(i)})

        def make_c_y_max(idx):
            def c(x):
                return 1 - x[2 * idx + 1] - x[2 * n_circles + idx]
            return c
        constraints.append({'type': 'ineq', 'fun': make_c_y_max(i)})

        # Non-negative radius
        def make_c_r_pos(idx):
            def c(x):
                return x[2 * n_circles + idx]
            return c
        constraints.append({'type': 'ineq', 'fun': make_c_r_pos(i)})

    # Overlap constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            def make_overlap_c(idx_i, idx_j):
                def c(x):
                    c1 = np.array([x[2 * idx_i], x[2 * idx_i + 1]])
                    c2 = np.array([x[2 * idx_j], x[2 * idx_j + 1]])
                    r1 = x[2 * n_circles + idx_i]
                    r2 = x[2 * n_circles + idx_j]
                    dist = np.sqrt(np.sum((c1 - c2) ** 2))
                    return dist - r1 - r2
                return c
            constraints.append({'type': 'ineq', 'fun': make_overlap_c(i, j)})

    bounds = [(0, 1)] * (2 * n_circles) + [(1e-12, 0.5)] * n_circles

    try:
        result = minimize(obj, x0, method='SLSQP', constraints=constraints,
                          bounds=bounds, options={'maxiter': 500, 'ftol': 1e-14})

        centers_opt = result.x[:2 * n_circles].reshape(n_circles, 2)
        radii_opt = np.maximum(result.x[2 * n_circles:], 1e-12)
        return centers_opt, radii_opt
    except Exception:
        return centers, radii


def hexagonal_init(n_circles, variation=0.0):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n_circles, 2))
    radii = np.ones(n_circles) * 0.08

    idx = 0
    spacing_x = 0.19 + variation * 0.01
    spacing_y = spacing_x * np.sqrt(3) / 2

    for row in range(7):
        for col in range(5):
            if idx >= n_circles:
                break
            x = (col + 0.5 + (row % 2) * 0.5) * spacing_x
            y = (row + 0.5) * spacing_y
            centers[idx] = [x, y]
            radii[idx] = 0.08 + variation * 0.005
            idx += 1
        if idx >= n_circles:
            break

    return centers, radii


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n_circles = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    for trial in range(15):
        # Phase 1: Initialize with hexagonal grid
        centers, radii = hexagonal_init(n_circles, variation=float(trial))

        # Phase 2: Force-based optimization
        centers, radii = force_based_optimize(centers, radii, n_circles)

        # Check validity after force optimization
        if is_valid(centers, radii):
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()

        # Phase 3: SLSQP refinement
        centers_opt, radii_opt = optimize_with_slsqp(centers, radii, n_circles)

        if is_valid(centers_opt, radii_opt):
            current_sum = np.sum(radii_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    # If no valid solution found yet, try one more round
    if best_sum < 2.5:
        centers, radii = hexagonal_init(n_circles, variation=2.0)
        centers, radii = force_based_optimize(centers, radii, n_circles)
        centers_opt, radii_opt = optimize_with_slsqp(centers, radii, n_circles)

        if is_valid(centers_opt, radii_opt):
            best_sum = np.sum(radii_opt)
            best_centers = centers_opt
            best_radii = radii_opt

    if best_centers is None:
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
