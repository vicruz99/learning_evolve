import numpy as np

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Hexagonal grid initialization for high initial density
    centers = []
    r_start = 0.05
    dx = 0.13
    dy = dx * np.sqrt(3) / 2
    row = 0
    count = 0
    while count < n:
        y = r_start + row * dy
        if y + r_start > 1.0:
            # Fill remaining circles randomly if grid fills up
            while len(centers) < n:
                centers.append([np.random.uniform(0.15, 0.85), np.random.uniform(0.15, 0.85)])
            break
        col = 0
        while count < n:
            x = r_start + col * dx + (row % 2) * (dx / 2)
            if x + r_start > 1.0:
                break
            centers.append([x, y])
            count += 1
            col += 1
        row += 1
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_start)
    
    # Simulation parameters
    dt = 0.008
    k_repulse = 20.0
    k_boundary = 20.0
    max_steps = 150000
    expand_step = 200
    expand_rate = 1.0015
    
    for step in range(max_steps):
        forces = np.zeros_like(centers)
        max_overlap = 0.0
        
        # Compute repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    if overlap > max_overlap:
                        max_overlap = overlap
                    if dist > 1e-8:
                        f_mag = k_repulse * overlap
                        forces[i] += f_mag * diff / dist
                        forces[j] -= f_mag * diff / dist
                        
        # Boundary forces to keep circles inside [0,1]x[0,1]
        for i in range(n):
            if centers[i,0] < radii[i]:
                forces[i,0] += k_boundary * (radii[i] - centers[i,0])
            if centers[i,0] > 1 - radii[i]:
                forces[i,0] -= k_boundary * (centers[i,0] - (1 - radii[i]))
            if centers[i,1] < radii[i]:
                forces[i,1] += k_boundary * (radii[i] - centers[i,1])
            if centers[i,1] > 1 - radii[i]:
                forces[i,1] -= k_boundary * (centers[i,1] - (1 - radii[i]))
                
        # Update positions
        centers += dt * forces
        # Enforce hard boundary constraints
        centers = np.clip(centers, radii[:, None], 1 - radii[:, None])
        
        # Expand radii if the configuration is stable (low overlap)
        if step % expand_step == 0 and max_overlap < 1e-4:
            radii *= expand_rate
            
        # Occasional perturbation to escape local minima
        if step % 10000 == 0 and step > 0:
            centers += np.random.uniform(-0.002, 0.002, centers.shape)
            centers = np.clip(centers, 0.05, 0.95)
            
        # Adaptive time step for stability
        force_norm = np.linalg.norm(forces)
        if force_norm > 5.0:
            dt *= 0.8
        elif force_norm < 0.05:
            dt = min(dt * 1.02, 0.02)
        dt = max(dt, 1e-6)
        
    # Final relaxation phase to strictly satisfy constraints
    for _ in range(10000):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-8:
                    f_mag = k_repulse * (min_dist - dist)
                    forces[i] += f_mag * diff / dist
                    forces[j] -= f_mag * diff / dist
            if centers[i,0] < radii[i]:
                forces[i,0] += k_boundary * (radii[i] - centers[i,0])
            if centers[i,0] > 1 - radii[i]:
                forces[i,0] -= k_boundary * (centers[i,0] - (1 - radii[i]))
            if centers[i,1] < radii[i]:
                forces[i,1] += k_boundary * (radii[i] - centers[i,1])
            if centers[i,1] > 1 - radii[i]:
                forces[i,1] -= k_boundary * (centers[i,1] - (1 - radii[i]))
        centers += dt * forces
        centers = np.clip(centers, radii[:, None], 1 - radii[:, None])
        
    return centers, radii, float(np.sum(radii))
