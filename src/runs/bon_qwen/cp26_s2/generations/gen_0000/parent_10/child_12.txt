import numpy as np
from scipy.optimize import minimize

def objective_func(v):
    n = len(v) // 3
    return -np.sum(v[:n])

def constraint_func(v):
    n = len(v) // 3
    r = v[:n]
    x = v[n:2*n]
    y = v[2*n:3*n]
    
    cons = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)
    
    # Pairwise constraints: dist(i,j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    triu_idx = np.triu_indices(n, k=1)
    cons.append(dist[triu_idx] - r_sum[triu_idx])
    
    return np.concatenate(cons)

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial configuration: Hexagonal packing
    r_init = 0.09
    centers = []
    radii = []
    row_y = r_init
    current_n = 0
    
    for row in range(6):
        cols = 5 if row % 2 == 0 else 4
        if row == 5:
            cols = 3
        if current_n + cols > n:
            cols = n - current_n
            
        shift_x = r_init if row % 2 != 0 else 0
        for c in range(cols):
            cx = r_init + c * 2 * r_init + shift_x
            cy = row_y
            centers.append([cx, cy])
            radii.append(r_init)
            current_n += 1
        row_y += np.sqrt(3) * r_init
        
    centers = np.array(centers)
    radii = np.array(radii)
    
    x0 = np.concatenate([radii, centers[:, 0], centers[:, 1]])
    bounds = [(0.0, 0.5)]*n + [(0.0, 1.0)]*n + [(0.0, 1.0)]*n
    
    best_res = None
    best_val = np.inf
    
    # Run optimization with a few restarts to escape local minima
    for trial in range(3):
        x0_trial = x0.copy()
        if trial > 0:
            # Perturb positions slightly
            noise = np.random.uniform(-0.02, 0.02, size=2*n)
            x0_trial[n:] += noise
            
        try:
            res = minimize(objective_func, x0_trial, bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraint_func},
                           method='SLSQP', options={'maxiter': 3000, 'ftol': 1e-12})
            if res.success and -res.fun > -best_val:
                best_val = -res.fun
                best_res = res
        except Exception:
            continue
            
    if best_res is None:
        return centers, radii, np.sum(radii)
        
    r_opt = best_res.x[:n]
    x_opt = best_res.x[n:2*n]
    y_opt = best_res.x[2*n:3*n]
    centers_opt = np.column_stack((x_opt, y_opt))
    
    # Post-processing to guarantee strict feasibility
    # Clamp radii to boundary limits
    r_opt = np.minimum(r_opt, np.minimum(x_opt, 1.0 - x_opt))
    r_opt = np.minimum(r_opt, np.minimum(y_opt, 1.0 - y_opt))
    
    # Ensure r_opt is non-negative
    r_opt = np.maximum(r_opt, 0.0)
    
    # Slight shrinkage to account for numerical tolerance in pairwise constraints
    min_ratio = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(x_opt[i] - x_opt[j], y_opt[i] - y_opt[j])
            req_sum = r_opt[i] + r_opt[j]
            if req_sum > 1e-12 and d < req_sum:
                ratio = d / req_sum
                if ratio < min_ratio:
                    min_ratio = ratio
                    
    if min_ratio < 1.0:
        r_opt *= min_ratio * 0.999
        
    return centers_opt, r_opt, np.sum(r_opt)
