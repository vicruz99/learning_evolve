import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None

    # Helper to generate initial configurations
    def get_initial_configs():
        configs = []
        
        # 1. Hexagonal Lattice Approximation
        # We generate a dense hex lattice and pick points that fit best
        # or just create a specific pattern for N=26
        # Pattern: Rows with counts 6, 5, 6, 5, 4 (sum=26)
        centers = []
        row_counts = [6, 5, 6, 5, 4]
        y_step = 0.15 # Initial guess for vertical spacing
        y_base = 0.1
        
        for i, count in enumerate(row_counts):
            y = y_base + i * y_step
            x_step = 1.0 / (count + 1)
            for j in range(count):
                x = x_step * (j + 1)
                # Shift odd rows slightly for hex effect
                if i % 2 == 1:
                    x += x_step * 0.5
                centers.append([x, y])
        configs.append(np.array(centers))
        
        # 2. Grid 5x5 + 1 in center (perturbed)
        grid_centers = []
        for i in range(5):
            for j in range(5):
                grid_centers.append([0.1 + 0.2*i, 0.1 + 0.2*j])
        # Add one in the center of a gap
        grid_centers.append([0.2, 0.3])
        configs.append(np.array(grid_centers)[:n])

        # 3. Random with low seed
        rng = np.random.RandomState(42)
        rand_centers = rng.rand(n, 2) * 0.8 + 0.1
        configs.append(rand_centers)
        
        # 4. Random with different seed
        rng2 = np.random.RandomState(123)
        rand_centers2 = rng2.rand(n, 2) * 0.8 + 0.1
        configs.append(rand_centers2)

        return configs

    def push_apart(centers, r, steps=500, k=0.01):
        """Heuristic force-directed separation"""
        centers = centers.copy()
        for _ in range(steps):
            moved = False
            for i in range(n):
                xi, yi = centers[i]
                fx, fy = 0.0, 0.0
                
                # Boundary forces
                if xi < r: fx += k * (r - xi)
                elif xi > 1 - r: fx -= k * (xi - (1 - r))
                if yi < r: fy += k * (r - yi)
                elif yi > 1 - r: fy -= k * (yi - (1 - r))
                
                # Pairwise forces
                for j in range(n):
                    if i == j: continue
                    dx = xi - centers[j, 0]
                    dy = yi - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    if dist < 1e-9: dist = 1e-9
                    if dist < 2*r:
                        force = k * (2*r - dist)
                        fx += force * (dx / dist)
                        fy += force * (dy / dist)
                        
                centers[i, 0] += fx
                centers[i, 1] += fy
                centers[i, 0] = max(r, min(1-r, centers[i, 0]))
                centers[i, 1] = max(r, min(1-r, centers[i, 1]))
                if abs(fx) > 1e-6 or abs(fy) > 1e-6:
                    moved = True
        return centers

    def is_valid(centers, r):
        """Quick validity check"""
        for i in range(n):
            x, y = centers[i]
            if x < r - 1e-9 or x > 1 - r + 1e-9 or y < r - 1e-9 or y > 1 - r + 1e-9:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                d2 = (centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2
                if d2 < (2*r)**2 - 1e-9:
                    return False
        return True

    # Optimization wrapper
    def optimize_from(start_centers):
        # Phase 1: Grow and Push to find a decent r
        r = 0.01
        centers = start_centers.copy()
        
        # Normalize centers to be well inside initially
        centers = centers * 0.8 + 0.1
        
        for _ in range(2000):
            r *= 1.001
            centers = push_apart(centers, r, steps=20, k=0.05)
            if not is_valid(centers, r):
                # If invalid, we grew too fast, shrink slightly and push more
                r /= 1.002
                centers = push_apart(centers, r, steps=100, k=0.02)
                
        # Phase 2: Scipy Refinement
        # Variables: x1, y1, ..., xn, yn, r
        # We maximize r => minimize -r
        # Constraints: dist >= 2r, boundary
        
        vars0 = np.concatenate([centers.flatten(), [r]])
        
        def objective(vars):
            return -vars[-1] # Minimize -r

        def constraint_boundary(vars):
            # x >= r, x <= 1-r, y >= r, y <= 1-r
            # vars[0] = x1, vars[1] = y1, ..., vars[-2] = yn, vars[-1] = r
            c_vals = []
            for i in range(n):
                x = vars[2*i]
                y = vars[2*i+1]
                rad = vars[-1]
                c_vals.append(x - rad)
                c_vals.append(1 - x - rad)
                c_vals.append(y - rad)
                c_vals.append(1 - y - rad)
            return np.array(c_vals)

        def constraint_overlap(vars):
            c_vals = []
            for i in range(n):
                xi, yi = vars[2*i], vars[2*i+1]
                for j in range(i + 1, n):
                    xj, yj = vars[2*j], vars[2*j+1]
                    d2 = (xi - xj)**2 + (yi - yj)**2
                    rad = vars[-1]
                    # d >= 2r => d^2 >= 4r^2
                    c_vals.append(d2 - 4*rad*rad)
            return np.array(c_vals)

        cons = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_overlap}
        ]
        
        try:
            res = minimize(objective, vars0, method='SLSQP', constraints=cons, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success or res.fun < -best_sum_r/26:
                final_centers = res.x[:-1].reshape(n, 2)
                final_r = res.x[-1]
                return final_centers, final_r
        except:
            pass
        return None, None

    # Run optimization for each strategy
    for init_config in get_initial_configs():
        centers, r = optimize_from(init_config)
        if centers is not None and r > 0:
            current_sum = 26 * r
            if current_sum > best_sum_r:
                best_sum_r = current_sum
                best_centers = centers
                best_radii = np.full(n, r)

    # Fallback to best heuristic result if scipy failed
    if best_centers is None:
        # Just return the last heuristic result roughly
        # This shouldn't happen with the code above
        best_centers = np.random.rand(n, 2) * 0.5 + 0.25
        best_radii = np.full(n, 0.05)
        best_sum_r = 26 * 0.05

    return best_centers, best_radii, best_sum_r
