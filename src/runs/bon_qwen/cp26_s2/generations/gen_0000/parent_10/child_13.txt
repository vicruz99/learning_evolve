def max_safe_radius(centers, radii, i, n):
    """Compute maximum safe radius for circle i given current configuration."""
    max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])

    return max(0, max_r)
