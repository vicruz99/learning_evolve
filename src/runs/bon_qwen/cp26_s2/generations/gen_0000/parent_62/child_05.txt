import numpy as np
from scipy.optimize import minimize
import random

def check_validity(centers, radii):
    """
    Quick validation check to ensure constraints are satisfied.
    Returns True if valid, False otherwise.
    """
    n = centers.shape[0]
    # Check bounds
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9:
            return False
        if y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
        if r < -1e-9:
            return False
            
    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Define constraints function for SLSQP
    # Variables order: x0, y0, r0, x1, y1, r1, ...
    def objective(x):
        # x is flattened array of size 3*n
        radii = x[2::3]
        return -np.sum(radii)

    def constraint_boundary(x):
        # Returns array of constraint values >= 0
        # Constraints:
        # 1. x_i - r_i >= 0
        # 2. 1 - x_i - r_i >= 0
        # 3. y_i - r_i >= 0
        # 4. 1 - y_i - r_i >= 0
        vals = []
        for i in range(n):
            x_i = x[3*i]
            y_i = x[3*i+1]
            r_i = x[3*i+2]
            vals.append(x_i - r_i)
            vals.append(1.0 - x_i - r_i)
            vals.append(y_i - r_i)
            vals.append(1.0 - y_i - r_i)
        return np.array(vals)

    def constraint_overlap(x):
        # Returns array of constraint values >= 0
        # dist(i, j) - (r_i + r_j) >= 0
        vals = []
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i = x[3*i], x[3*i+1]
                x_j, y_j = x[3*j], x[3*j+1]
                r_i, r_j = x[3*i+2], x[3*j+2]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                vals.append(dist - (r_i + r_j))
        return np.array(vals)

    # List of constraints for SLSQP
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n

    def optimize_config(init_centers, init_radii):
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = init_centers[i, 0]
            x0[3*i+1] = init_centers[i, 1]
            x0[3*i+2] = init_radii[i]
        
        # Use SLSQP
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        if res.success:
            final_centers = np.array([[res.x[3*i], res.x[3*i+1]] for i in range(n)])
            final_radii = np.array([res.x[3*i+2] for i in range(n)])
            
            # Validate with tolerance
            if check_validity(final_centers, final_radii):
                s = np.sum(final_radii)
                return res.x, final_centers, final_radii, s
        return None, None, None, -1.0

        pass
