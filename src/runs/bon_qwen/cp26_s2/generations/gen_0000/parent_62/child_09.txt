import numpy as np
from scipy.optimize import minimize, linprog


def boundary_x_constraint(params, n, i):
    return params[2 * i] - params[2 * n + i]


def boundary_y_constraint(params, n, i):
    return params[2 * i + 1] - params[2 * n + i]


def boundary_1x_constraint(params, n, i):
    return 1.0 - params[2 * i] - params[2 * n + i]


def boundary_1y_constraint(params, n, i):
    return 1.0 - params[2 * i + 1] - params[2 * n + i]


def overlap_constraint(params, n, i, j):
    dx = params[2 * i] - params[2 * j]
    dy = params[2 * i + 1] - params[2 * j + 1]
    dist_sq = dx * dx + dy * dy
    if dist_sq < 1e-20:
        dist_sq = 1e-20
    return np.sqrt(dist_sq) - params[2 * n + i] - params[2 * n + j]


def objective_neg_sum_radii(params, n):
    return -np.sum(params[2 * n:])


def create_hex_centers(counts, spacing):
    centers = []
    y = spacing * np.sqrt(3) / 2
    for row_idx, count in enumerate(counts):
        if count == 0:
            continue
        x_start = (1.0 - (count - 1) * spacing) / 2.0
        offset = spacing * 0.5 if row_idx % 2 == 1 else 0
        for col_idx in range(count):
            x = x_start + col_idx * spacing + offset
            centers.append([x, y])
        y += spacing * np.sqrt(3) / 2
    return np.array(centers)


def solve_radii_lp(centers, n):
    given centers, find optimal radii via linear programming.
    Returns (radii, sum_radii) or (zeros, 0) on failure.
    """
    A_ub_rows = []
    b_ub_vals = []

    for i in range(n):
        b_i = min(centers[i, 0], centers[i, 1], 1.0 - centers[i, 0], 1.0 - centers[i, 1])
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_vals.append(max(b_i, 1e-10))

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d_ij = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(d_ij)

    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)

    c = -np.ones(n)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success and result.fun is not None:
        radii = np.maximum(result.x, 0)
        return radii, np.sum(radii)
    return np.zeros(n), 0.0


def barrier_objective(params, n, mu):
    Compute smooth barrier objective for L-BFGS-B optimization.
    """
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-10)

    obj = -np.sum(radii)

    for i in range(n):
        vals = [
            centers[i, 0] - radii[i],
            centers[i, 1] - radii[i],
            1.0 - centers[i, 0] - radii[i],
            1.0 - centers[i, 1] - radii[i],
        ]
        for v in vals:
            if v < 0:
                obj += mu * v * v

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-20)
            v = dist - radii[i] - radii[j]
            if v < 0:
                obj += mu * v * v

    return obj


def compute_sum_radii_from_centers(centers, n):
    For fixed centers, compute max sum of radii via LP.
    """
    _, s = solve_radii_lp(centers, n)
    return s


def perturb_centers(centers, rng, scale=0.01):
    Add small random perturbation to centers.
    """
    perturbed = centers.copy()
    perturbed += rng.uniform(-scale, scale, size=centers.shape)
    return perturbed


def run_slsqp_optimization(params0, n):
    Run SLSQP optimization with explicit constraints.
    """
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': boundary_x_constraint, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_constraint, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': boundary_1x_constraint, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': boundary_1y_constraint, 'args': (n, i)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': overlap_constraint, 'args': (n, i, j)})

    bounds = [(0.001, 0.999)] * (2 * n) + [(1e-8, 0.5)] * n

    result = minimize(
        objective_neg_sum_radii,
        params0,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
    )
    return result


def run_lbfgsb_optimization(params0, n, mu=500):
    Run L-BFGS-B optimization with barrier method.
    """
    bounds = [(0.001, 0.999)] * (2 * n) + [(1e-8, 0.5)] * n

    result = minimize(
        barrier_objective,
        params0,
        args=(n, mu),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 10000, 'ftol': 1e-15}
    )
    return result


def clamp_to_square(centers, radii, n):
    Ensure all circles are within the unit square by clamping.
    """
    for i in range(n):
        centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
    return centers


def check_feasibility(centers, radii, n):
    Check if packing is feasible.
    """
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True
