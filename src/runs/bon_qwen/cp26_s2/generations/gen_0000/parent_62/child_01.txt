import numpy as np
from scipy.optimize import minimize

def compute_objective_and_grad(z, n, penalty_coeff):
    """
    Computes the objective function value (negative sum of radii + penalties).
    Returns (f, grad).
    """
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    
    # Initialize gradient
    grad = np.zeros_like(z)
    
    # Objective: -sum(radii)
    f = -np.sum(radii)
    grad[2*n:] = -1.0  # Gradient of -sum(r) wrt r is -1
    
    # --- Boundary Constraints ---
    # Constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    # Penalty terms: max(0, r - x)^2, etc.
    # Gradients are 2*(r-x)*(-1) wrt x, 2*(r-x)*(1) wrt r, etc. if r > x
    
    # Left: x - r >= 0 => penalty if r > x
    diff_l = radii - centers[:, 0]
    mask_l = diff_l > 0
    if np.any(mask_l):
        term = diff_l[mask_l]**2
        f += penalty_coeff * np.sum(term)
        grad[mask_l, 0] -= 2 * penalty_coeff * diff_l[mask_l] # d/dx
        grad[mask_l + 2*n] += 2 * penalty_coeff * diff_l[mask_l] # d/dr
        
    # Right: 1 - x - r >= 0 => penalty if x + r > 1
    diff_r = centers[:, 0] + radii - 1
    mask_r = diff_r > 0
    if np.any(mask_r):
        term = diff_r[mask_r]**2
        f += penalty_coeff * np.sum(term)
        grad[mask_r, 0] += 2 * penalty_coeff * diff_r[mask_r] # d/dx
        grad[mask_r + 2*n] += 2 * penalty_coeff * diff_r[mask_r] # d/dr
        
    # Bottom: y - r >= 0
    diff_b = radii - centers[:, 1]
    mask_b = diff_b > 0
    if np.any(mask_b):
        term = diff_b[mask_b]**2
        f += penalty_coeff * np.sum(term)
        grad[mask_b, 1] -= 2 * penalty_coeff * diff_b[mask_b] # d/dy
        grad[mask_b + 2*n] += 2 * penalty_coeff * diff_b[mask_b] # d/dr
        
    # Top: 1 - y - r >= 0
    diff_t = centers[:, 1] + radii - 1
    mask_t = diff_t > 0
    if np.any(mask_t):
        term = diff_t[mask_t]**2
        f += penalty_coeff * np.sum(term)
        grad[mask_t, 1] += 2 * penalty_coeff * diff_t[mask_t] # d/dy
        grad[mask_t + 2*n] += 2 * penalty_coeff * diff_t[mask_t] # d/dr
        
    # --- Overlap Constraints ---
    # dist_ij >= r_i + r_j => penalty if r_i + r_j > dist_ij
    # Use upper triangular indices for efficiency
    i_idx, j_idx = np.triu_indices(n, k=1)
    
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16) # epsilon to avoid div by zero
    
    r_sum = radii[i_idx] + radii[j_idx]
    overlap = r_sum - dist
    mask_ov = overlap > 0
    
    if np.any(mask_ov):
        ov_vals = overlap[mask_ov]
        term = ov_vals**2
        f += penalty_coeff * np.sum(term)
        
        # Gradients for i and j
        idx_i = i_idx[mask_ov]
        idx_j = j_idx[mask_ov]
        
        dx_ov = dx[mask_ov]
        dy_ov = dy[mask_ov]
        dist_ov = dist[mask_ov]
        
        # Chain rule: d/dx_i (r_i + r_j - d)^2 = 2*(r_i+r_j-d) * (-(x_i-x_j)/d)
        grad_i_x = -2 * penalty_coeff * ov_vals * (dx_ov / dist_ov)
        grad_i_y = -2 * penalty_coeff * ov_vals * (dy_ov / dist_ov)
        grad_i_r = 2 * penalty_coeff * ov_vals
        
        grad_j_x = 2 * penalty_coeff * ov_vals * (dx_ov / dist_ov)
        grad_j_y = 2 * penalty_coeff * ov_vals * (dy_ov / dist_ov)
        grad_j_r = 2 * penalty_coeff * ov_vals
        
        np.add.at(grad[:, 0], idx_i, grad_i_x)
        np.add.at(grad[:, 1], idx_i, grad_i_y)
        np.add.at(grad[2*n:], idx_i, grad_i_r)
        
        np.add.at(grad[:, 0], idx_j, grad_j_x)
        np.add.at(grad[:, 1], idx_j, grad_j_y)
        np.add.at(grad[2*n:], idx_j, grad_j_r)
        
    return f, grad

def run_packing():
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Parameters for optimization
    penalty_coeff = 5000.0
    n_restarts = 100
    
    # Precompute indices for overlap
    i_idx, j_idx = np.triu_indices(n, k=1)
    
    def objective(z):
        return compute_objective_and_grad(z, n, penalty_coeff)[0]
    
    def jac(z):
        return compute_objective_and_grad(z, n, penalty_coeff)[1]

    for restart in range(n_restarts):
        # --- Initialization Strategy ---
        # Mix of grid, hexagonal, and random to explore space
        strategy = restart % 3
        
        if strategy == 0:
            # Grid initialization
            xs = np.linspace(0.12, 0.88, 5)
            ys = np.linspace(0.12, 0.88, 5)
            pts = np.array([(x, y) for y in ys for x in xs])
            # Add 26th point near center
            pts = np.vstack([pts, [0.5, 0.5]])
            np.random.shuffle(pts)
            centers = pts[:n].copy()
        elif strategy == 1:
            # Hexagonal-ish
            centers = np.random.uniform(0.1, 0.9, (n, 2))
        else:
            # Random dense
            centers = np.random.uniform(0.05, 0.95, (n, 2))
            
        # Add noise
        centers += np.random.normal(0, 0.02, centers.shape)
        centers = np.clip(centers, 0.02, 0.98)
        
        # Initial radii: small but non-zero
        radii = np.full(n, 0.04 + 0.01 * np.random.rand(n))
        
        x0 = np.concatenate([centers.flatten(), radii])
        
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        try:
            res = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', 
                           jac=jac, options={'maxiter': 1000, 'ftol': 1e-15, 'gtol': 1e-12})
            
            c = res.x[:2*n].reshape(n, 2)
            r = res.x[2*n:]
            
            # Check validity
            # Use a small epsilon for validation to be safe against numerical errors
            valid = True
            # Boundary check
            if np.any(r < 1e-9) or np.any(c < r) or np.any(c + r > 1):
                valid = False
            # Overlap check
            if valid:
                dx = c[:, 0][:, None] - c[:, 0][None, :]
                dy = c[:, 1][:, None] - c[:, 1][None, :]
                dist = np.sqrt(dx**2 + dy**2)
                r_sum = r[:, None] + r[None, :]
                # Ignore diagonal
                np.fill_diagonal(dist, np.inf)
                if np.any(dist < r_sum - 1e-10):
                    valid = False
                    
            if valid:
                s = np.sum(r)
                if s > best_sum:
                    best_sum = s
                    best_centers = c.copy()
                    best_radii = r.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails (unlikely)
    if best_centers is None:
        # Generate a safe grid packing
        pts = np.linspace(0.1, 0.9, 5)
        centers = np.array([(x, y) for y in pts for x in pts])
        # Adjust for 26 circles
        centers = np.vstack([centers, [0.5, 0.5]])
        r_val = 0.08
        best_centers = centers
        best_radii = np.full(26, r_val)
        best_sum = 26 * r_val

    return best_centers, best_radii, best_sum
