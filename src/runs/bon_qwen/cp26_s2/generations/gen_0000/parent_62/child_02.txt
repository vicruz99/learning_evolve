import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
        sum_radii: float sum of radii
    """
    n = 26
    
    # --- 1. Initialization: Hexagonal Packing ---
    # We start with a valid configuration to ensure the optimizer has a good baseline.
    # A hexagonal lattice is the densest way to pack equal circles.
    centers = []
    radii = np.zeros(n)
    r_init = 0.08  # Initial radius, safely small enough to fit 26 circles
    
    y = r_init
    row_idx = 0
    while len(centers) < n:
        # Offset x for odd rows to create hexagonal pattern
        x_start = r_init if row_idx % 2 == 0 else 2 * r_init
        x = x_start
        while x <= 1 - r_init and len(centers) < n:
            centers.append([x, y])
            x += 2 * r_init
        y += r_init * np.sqrt(3)
        row_idx += 1
    
    centers = np.array(centers[:n])
    radii[:] = r_init
    
    # Flatten variables for optimization: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: 0 <= x, y <= 1; 0 <= r <= 0.5
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    # --- 2. Optimization Setup ---
    
    def objective(v):
        # We want to maximize sum of radii, so we minimize negative sum
        return -np.sum(v[2::3])

    def penalty_function(v):
        centers = v[:2*n].reshape((n, 2))
        radii = v[2*n:]
        
        penalty = 0.0
        
        # Boundary penalties: max(0, r - dist_to_edge)^2
        # Left: x - r >= 0 -> violation if x < r
        # Right: 1 - x - r >= 0 -> violation if 1 - x < r
        # Down: y - r >= 0 -> violation if y < r
        # Up: 1 - y - r >= 0 -> violation if 1 - y < r
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left
            val = x - r
            if val < 0: penalty += val**2
            # Right
            val = 1 - x - r
            if val < 0: penalty += val**2
            # Down
            val = y - r
            if val < 0: penalty += val**2
            # Up
            val = 1 - y - r
            if val < 0: penalty += val**2

        # Overlap penalties: max(0, r_i + r_j - dist)^2
        # dist = sqrt((xi-xj)^2 + (yi-yj)^2)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                               (centers[i, 1] - centers[j, 1])**2)
                val = dist - (radii[i] + radii[j])
                if val < 0:
                    penalty += val**2
                    
        return penalty

    def augmented_objective(v):
        # High penalty weight to enforce constraints strictly
        return objective(v) + 1000 * penalty_function(v)

    # --- 3. Run Optimization ---
    # L-BFGS-B is efficient for large-scale bound-constrained optimization
    res = minimize(augmented_objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 2000})
    
    # Extract results
    final_centers = res.x[:2*n].reshape((n, 2))
    final_radii = res.x[2*n:]
    
    # Clean up very small negative radii due to numerical noise
    final_radii = np.maximum(final_radii, 0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
