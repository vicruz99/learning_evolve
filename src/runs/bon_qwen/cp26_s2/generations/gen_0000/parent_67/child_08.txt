def get_initial_hex_packing(n=26):
    # Get raw hexagonal points
    raw_centers = hexagonal_layout(n)
    
    # We need to scale these to fit in [-0.5, 0.5] x [-0.5, 0.5] roughly
    # then shift to [0,1]
    # But actually, we will optimize radii, so we just need a valid non-overlapping config.
    # Let's scale them to fit in the unit square with some margin.
    
    # Find bounding box of raw_centers
    mins = np.min(raw_centers, axis=0)
    maxs = np.max(raw_centers, axis=0)
    width = maxs[0] - mins[0]
    height = maxs[1] - mins[1]
    
    # Scale to fit in [0,1] with a safety margin (e.g., 90% of space)
    # We leave 5% margin on each side for radius
    margin = 0.05
    max_dim = max(width, height)
    scale = (1 - 2*margin) / max_dim
    
    scaled_centers = raw_centers * scale + 0.5 # Center in square
    
    # Initialize radii. 
    # A safe initial radius is small, or we can estimate based on spacing.
    # Min distance between points in scaled config
    dists = []
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(scaled_centers[i] - scaled_centers[j])
            dists.append(d)
    min_dist = min(dists)
    # Also distance to walls
    wall_dists = []
    for i in range(n):
        x, y = scaled_centers[i]
        wall_dists.append(min(x, 1-x, y, 1-y))
    min_wall = min(wall_dists)
    
    # Initial radius: half of min spacing? No, that would make them touch.
    # Let's start with a fraction to ensure validity
    r_init = min(min_dist, min_wall) * 0.1
    
    return scaled_centers, r_init

def constraints_function(vars, n):
    """
    vars: [x0, y0, r0, x1, y1, r1, ...]
    """

    radii = np.zeros(n)

    # x - r >= 0  => r - x <= 0 (inequality constraint form: c(x) >= 0)
    # SLSQP expects 'ineq' meaning constraint >= 0
    # So x - r >= 0 -> x - r
    # 1 - x - r >= 0 -> 1 - x - r
    
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]}) # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i] - v[3*i+2]}) # 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]}) # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i+1] - v[3*i+2]}) # 1 - y - r >= 0
        
        # Non-negative radius
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+2]})

    # 2. Non-overlap constraints
    # dist(c_i, c_j) >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq', 
                'fun': lambda v, i=i, j=j: \
                    np.sqrt((v[3*i]-v[3*j])**2 + (v[3*i+1]-v[3*j+1])**2) - v[3*i+2] - v[3*j+2]
            })
            
    return cons

def objective_function(vars):
    """Maximize sum of radii -> Minimize negative sum"""
    n = len(vars) // 3
    radii = vars[2::3] # Every 3rd element starting from index 2
    return -np.sum(radii)
