import numpy as np
from scipy.optimize import minimize

# Global constants for the problem
NUM_CIRCLES = 26
PENALTY_WEIGHT = 10000.0

def compute_objective(params):
    """
    Computes the objective function value.
    Minimizes negative sum of radii plus penalties for constraint violations.
    
    Args:
        params: numpy array of shape (3 * NUM_CIRCLES,) containing [x0, y0, r0, x1, y1, r1, ...]
        
    Returns:
        float: objective value
    """
    # Extract centers and radii
    # params is flat: [x0, y0, r0, x1, y1, r1, ...]
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    # Primary objective: Maximize sum of radii (minimize negative sum)
    obj_val = -np.sum(rs)
    
    # 1. Boundary Penalties
    # Circles must be within [0, 1] x [0, 1].
    # Constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    # Violations: r - x > 0, x + r - 1 > 0, r - y > 0, y + r - 1 > 0
    
    v_x1 = np.maximum(rs - xs, 0.0)       # Left boundary violation
    v_x2 = np.maximum(xs + rs - 1.0, 0.0) # Right boundary violation
    v_y1 = np.maximum(rs - ys, 0.0)       # Bottom boundary violation
    v_y2 = np.maximum(ys + rs - 1.0, 0.0) # Top boundary violation
    
    penalty_boundary = np.sum(v_x1**2) + np.sum(v_x2**2) + np.sum(v_y1**2) + np.sum(v_y2**2)
    
    # 2. Overlap Penalties
    # Circles must not overlap: dist(i, j) >= ri + rj
    # Violation: ri + rj - dist(i, j) > 0
    
    # Compute pairwise distances using broadcasting
    # xs shape (N,), xs[:, None] shape (N, 1)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Compute sum of radii for all pairs
    r_sums = rs[:, None] + rs[None, :]
    
    # Calculate violations matrix
    # We only care about pairs where violation > 0
    violations = np.maximum(r_sums - dists, 0.0)
    
    # Sum squares of violations for unique pairs (upper triangle, excluding diagonal)
    # Using np.triu to get upper triangle
    upper_tri_violations = np.triu(violations, k=1)
    penalty_overlap = np.sum(upper_tri_violations**2)
    
    # Total penalty
    total_penalty = PENALTY_WEIGHT * (penalty_boundary + penalty_overlap)
    
    return obj_val + total_penalty

        rows = 5
        
        # Spacing
        # Add some margin
        margin = 0.05
        x_step = (1.0 - 2 * margin) / (cols - 1) if cols > 1 else 0
        y_step = (1.0 - 2 * margin) / (rows - 1) if rows > 1 else 0
        
        idx = 0
        for r in range(rows):
            for c in range(cols):
                if idx >= n:
                    break

                x_offset = (x_step / 2) if (r % 2 == 1) else 0

                y = margin + r * y_step

                    y += rng.uniform(-0.02, 0.02)

                r_init = 0.04
