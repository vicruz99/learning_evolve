- Objective: Maximize $\sum r_i$ (minimize $-\sum r_i$).
- Constraints:

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(v):
    """Objective function: maximize sum of radii => minimize negative sum."""
    return -np.sum(v[2::3])

def compute_constraints(v):
    """Compute all inequality constraints as an array."""
    n = N_CIRCLES
    num_constraints = 4 * n + n * (n - 1) // 2
    c = np.empty(num_constraints)
    idx = 0
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    for i in range(n):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        c[idx] = xi - ri
        c[idx+1] = 1.0 - xi - ri
        c[idx+2] = yi - ri
        c[idx+3] = 1.0 - yi - ri
        idx += 4
        
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        for j in range(i+1, n):
            xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
            dx = xi - xj
            dy = yi - yj
            c[idx] = dx*dx + dy*dy - (ri + rj)**2
            idx += 1
            
    return c

def run_packing():
    n = N_CIRCLES
    cons = [{'type': 'ineq', 'fun': compute_constraints}]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Multi-start optimization with perturbed hexagonal initialization
    for seed in range(5):
        np.random.seed(seed)
        cols = 6
        rows = 5
        x0 = np.zeros(n)
        y0 = np.zeros(n)
        r0 = np.full(n, 0.05)  # Reasonable starting radius
        
        idx = 0
        for r in range(rows):
            for c in range(cols):
                if idx >= n: 
                    break
                sx = 1.0 / (cols + 1)
                sy = sx * np.sqrt(3) / 2.0
                x = (c + 1) * sx
                y = (r + 1) * sy
                if r % 2 == 1:
                    x += sx / 2.0
                x0[idx] = x
                y0[idx] = y
                idx += 1
                
        # Add small random perturbation to break symmetry and escape trivial minima
        x0 += np.random.uniform(-0.02, 0.02, n)
        y0 += np.random.uniform(-0.02, 0.02, n)
        
        vars0 = np.concatenate([x0, y0, r0])
        vars0[:2*n] = np.clip(vars0[:2*n], 0.05, 0.95)
        
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-9})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x[:2*n].reshape(-1, 2)
                best_radii = res.x[2*n:]
        except Exception:
            continue
            
    # Fallback to a trivial valid packing if optimization fails
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.01)
        best_sum = 0.26
        
    return best_centers, best_radii, float(best_sum)
