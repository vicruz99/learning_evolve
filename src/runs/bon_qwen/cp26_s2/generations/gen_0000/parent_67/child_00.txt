import numpy as np
from scipy.optimize import minimize

def objective(vars_arr):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    n = len(vars_arr) // 3
    radii = vars_arr[2*n:]
    return -np.sum(radii)

def constraint_func(vars_arr):
    """Vectorized inequality constraints for boundaries and pairwise distances"""
    n = len(vars_arr) // 3
    centers = vars_arr[:2*n].reshape(n, 2)
    radii = vars_arr[2*n:]
    
    c = []
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    c.append(centers[:, 0] - radii)
    c.append(1.0 - centers[:, 0] - radii)
    c.append(centers[:, 1] - radii)
    c.append(1.0 - centers[:, 1] - radii)
    
    # Inter-circle non-overlap constraints: dist(i,j) >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.append(dists[mask] - r_sum[mask])
    
    return np.concatenate(c)

def run_packing():
    n = 26
    best_val = -np.inf
    best_vars = None
    
    starts = []
    
    # 1. Grid initialization
    grid = np.zeros((n, 2))
    idx = 0
    for r in range(5):
        for c in range(5):
            if idx < n:
                grid[idx] = [0.15 + c * 0.175, 0.15 + r * 0.175]
                idx += 1
    if idx < n:
        grid[idx] = [0.5, 0.5]
    starts.append(grid)
    
    # 2. Hexagonal lattice initialization
    hex_pts = []
    s = 0.14
    y = 0.15
    while len(hex_pts) < n:
        row_offset = (len(hex_pts) // 6) % 2 * (s / 2)
        x = 0.15 + row_offset
        while x < 0.85 and len(hex_pts) < n:
            hex_pts.append([x, y])
            x += s
        y += s * np.sqrt(3) / 2
    starts.append(np.array(hex_pts[:n]))
    
    # 3. Random initializations for diversity
    np.random.seed(42)
    for _ in range(4):
        starts.append(np.random.uniform(0.2, 0.8, (n, 2)))
        
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    for start_centers in starts:
        x0 = np.zeros(3*n)
        x0[:2*n] = start_centers.flatten()
        x0[2*n:] = 0.04  # Small initial radii to start feasible
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 3000, 'ftol': 1e-9})
        
        if res.fun < best_val:
            best_val = res.fun
            best_vars = res.x.copy()
            
    # Extract and format results
    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]
    
    # Strict post-processing to guarantee validity within tolerance
    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if radii[i] > max_r:
            radii[i] = max_r
            
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            req_dist = radii[i] + radii[j]
            if dist < req_dist - 1e-12:
                overlap = req_dist - dist
                radii[i] -= overlap / 2
                radii[j] -= overlap / 2
                
    radii = np.maximum(radii, 1e-7)
    return centers, radii, np.sum(radii)
