import numpy as np
from scipy.optimize import minimize

def compute_energy(vars, penalty_weight, N):
    pos = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    
    energy = -np.sum(r)
    penalty = 0.0
    
    for i in range(N):
        xi, yi = pos[i]
        ri = r[i]
        
        # Boundary penalties
        for b in [xi, 1.0 - xi, yi, 1.0 - yi]:
            diff = ri - b
            if diff > 0:
                penalty += diff * diff
                
        # Overlap penalties
        for j in range(i + 1, N):
            diff_vec = pos[i] - pos[j]
            dist = np.sqrt(diff_vec[0]**2 + diff_vec[1]**2)
            overlap = r[i] + r[j] - dist
            if overlap > 0:
                penalty += overlap * overlap
                
    return energy + penalty_weight * penalty

def run_packing():
    np.random.seed(42)
    N = 26
    
    best_val = float('inf')
    best_vars = None
    
    inits = []
    
    # Hexagonal initialization for better spread
    row_width = 0.12
    col_height = row_width * np.sqrt(3) / 2
    hex_pos = []
    count = 0
    for i in range(8):
        for j in range(8):
            if count >= N: break
            x = 0.1 + j * row_width + (0.5 * row_width if i % 2 else 0)
            y = 0.1 + i * col_height
            if x < 0.9 and y < 0.9:
                hex_pos.append([x, y])
                count += 1
        if count >= N: break
    hex_pos = np.array(hex_pos)
    
    # Create multiple starts with jitter
    for _ in range(3):
        jittered = hex_pos + np.random.randn(N, 2) * 0.03
        inits.append(jittered)
        
    # Random starts
    for _ in range(2):
        inits.append(np.random.rand(N, 2) * 0.8 + 0.1)
        
    r_init = np.ones(N) * 0.05
    bounds = [(0.0, 1.0) for _ in range(2*N)] + [(1e-5, 0.5) for _ in range(N)]
    
    for pos_init in inits:
        vars0 = np.concatenate([pos_init.flatten(), r_init])
        
        # Phase 1: Moderate penalty to explore configuration space
        res1 = minimize(compute_energy, vars0, args=(100.0, N), method='L-BFGS-B', bounds=bounds, options={'maxiter': 800, 'ftol': 1e-12})
        
        # Phase 2: High penalty to enforce constraints tightly
        res2 = minimize(compute_energy, res1.x, args=(10000.0, N), method='L-BFGS-B', bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-12})
        
        if res2.fun < best_val:
            best_val = res2.fun
            best_vars = res2.x
            
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    
    # Feasibility projection to guarantee strict compliance with tolerance
    for _ in range(100):
        changed = False
        for i in range(N):
            xi, yi = centers[i]
            ri = radii[i]
            
            for b in [xi, 1.0 - xi, yi, 1.0 - yi]:
                if ri > b + 1e-12:
                    radii[i] = b + 1e-12
                    changed = True
                    
            for j in range(i + 1, N):
                diff = centers[i] - centers[j]
                dist = np.sqrt(diff[0]**2 + diff[1]**2)
                req = radii[i] + radii[j]
                if dist < req - 1e-12:
                    scale = (dist + 1e-12) / req
                    radii[i] *= scale
                    radii[j] *= scale
                    changed = True
        if not changed:
            break
            
    return centers, radii, np.sum(radii)
