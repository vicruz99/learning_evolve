import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # --- 1. Initialization ---
    # We initialize circles in a grid pattern. 
    # A 6x5 grid gives 30 positions; we take the first 26.
    # We space them slightly apart to start with a valid configuration.
    
    # Create a grid of points
    # x range: 0.1 to 0.9 (avoiding immediate boundary issues)
    # y range: 0.1 to 0.9
    x_coords = np.linspace(0.1, 0.9, 6)
    y_coords = np.linspace(0.1, 0.9, 5)
    
    xx, yy = np.meshgrid(x_coords, y_coords)
    grid_points = np.column_stack([xx.flatten(), yy.flatten()])
    
    # Select first 26 points
    initial_centers = grid_points[:n]
    
    # Initial radii: small enough to fit in the grid spacing (0.2)
    # r=0.05 implies diameter 0.1, fitting easily with 0.2 spacing.
    initial_radii = np.full(n, 0.05)
    
    # Combine into a single vector for the optimizer
    # Order: x1, y1, r1, x2, y2, r2, ...
    x0 = np.zeros(3 * n)
    x0[0::3] = initial_centers[:, 0]
    x0[1::3] = initial_centers[:, 1]
    x0[2::3] = initial_radii
    
    # --- 2. Bounds ---
    # x, y must be in [0, 1]
    # r must be in [0, 0.5] (since max radius in unit square is 0.5)
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    
    # --- 3. Constraints ---
    # We define a function that returns a vector of constraint values.
    # For 'ineq' constraints, the solver expects values >= 0.
    
    def constraint_func(v):
        # Reshape variables
        xs = v[0::3]
        ys = v[1::3]
        rs = v[2::3]
        
        # Boundary Constraints:
        # 1. x - r >= 0  => x >= r
        # 2. 1 - x - r >= 0 => x + r <= 1
        # 3. y - r >= 0  => y >= r
        # 4. 1 - y - r >= 0 => y + r <= 1
        
        c_bound = np.concatenate([
            xs - rs,
            1.0 - xs - rs,
            ys - rs,
            1.0 - ys - rs
        ])
        
        # Pairwise Non-overlap Constraints:
        # dist(i, j)^2 >= (r_i + r_j)^2
        # dist^2 - (r_i + r_j)^2 >= 0
        
        # Compute pairwise differences using broadcasting
        # dx[i, j] = x_i - x_j
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dr = rs[:, None] + rs[None, :]
        
        dist_sq = dx**2 + dy**2
        rad_sum_sq = dr**2
        
        # We only need constraints for i < j (upper triangle)
        # Create a mask for the upper triangle excluding diagonal
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        c_pairwise = dist_sq[mask] - rad_sum_sq[mask]
        
        return np.concatenate([c_bound, c_pairwise])
    
    constraints = {
        'type': 'ineq',
        'fun': constraint_func
    }
    
    # --- 4. Objective Function ---
    # Maximize sum of radii -> Minimize negative sum
    def objective(v):
        return -np.sum(v[2::3])
    
    # --- 5. Optimization ---
    # Use SLSQP method which supports bounds and constraints
    # maxiter increased to allow convergence
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )
    
    # --- 6. Post-processing ---
    if res.success:
        best_v = res.x
    else:
        # If optimization fails, we still return the best found or initial
        # In a robust solution, we might retry, but for this context,
        # we proceed with the result.
        best_v = res.x 
    
    # Extract results
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    
    # Ensure non-negative radii (numerical safety)
    radii = np.maximum(radii, 0.0)
    
    # Ensure centers are within bounds (numerical safety)
    centers[:, 0] = np.clip(centers[:, 0], 0.0, 1.0)
    centers[:, 1] = np.clip(centers[:, 1], 0.0, 1.0)
    
    # Adjust radii if they slightly violate boundaries due to noise
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    r_max_from_bounds = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    radii = np.minimum(radii, r_max_from_bounds)
    
    # Calculate sum of radii
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
