import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(v):
    """Maximize radius r (minimize -r)"""
    return -v[-1]

def constraint_fun(v):
    """Vectorized constraint function returning inequality constraints >= 0"""
    n = N_CIRCLES
    r = v[-1]
    cx = v[0::2]
    cy = v[1::2]
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    b1 = cx - r
    b2 = 1.0 - cx - r
    b3 = cy - r
    b4 = 1.0 - cy - r
    b_constraints = np.concatenate([b1, b2, b3, b4])
    
    # Distance constraints: dist^2 >= 4r^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    
    # Extract lower triangular part without diagonal for unique pairs
    idx = np.tril_indices(n, -1)
    d_constraints = dist2[idx] - 4.0 * r * r
    
    return np.concatenate([b_constraints, d_constraints])

def run_packing():
    # 1. Initial hexagonal placement
    centers = np.zeros((N_CIRCLES, 2))
    r_init = 0.085  # Start feasible
    
    # Pattern: 6 rows with counts [5, 5, 5, 5, 4, 2] sums to 26
    row_counts = [5, 5, 5, 5, 4, 2]
    idx = 0
    for row, count in enumerate(row_counts):
        for col in range(count):
            x = r_init + col * 2 * r_init + (r_init if row % 2 == 1 else 0)
            y = r_init + row * r_init * np.sqrt(3)
            centers[idx] = [x, y]
            idx += 1
            
    # 2. Prepare optimization variables and bounds
    x0 = np.concatenate([centers.flatten(), [r_init]])
    bounds = [(0, 1)] * (2 * N_CIRCLES) + [(0, 0.5)]
    
    # 3. Run constrained optimization
    cons = {'type': 'ineq', 'fun': constraint_fun}
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 2000, 'ftol': 1e-10}
    )
    
    # 4. Extract and validate results
    r_final = res.x[-1]
    c_final = res.x[:-1].reshape((N_CIRCLES, 2))
    
    # Apply tiny margin to strictly satisfy validation tolerance (1e-12)
    margin = 1e-8
    c_final = np.clip(c_final, r_final + margin, 1.0 - r_final - margin)
    
    radii = np.full(N_CIRCLES, r_final)
    sum_radii = float(np.sum(radii))
    
    return c_final, radii, sum_radii
