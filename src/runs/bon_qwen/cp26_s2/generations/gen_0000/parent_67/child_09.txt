import numpy as np
from scipy.optimize import minimize


def compute_violations(centers, radii):
    """Compute total constraint violation (boundary + overlap)."""
    n = len(radii)
    violation = 0.0
    
    # Boundary violations (vectorized)
    violation += np.sum(np.maximum(0.0, radii - centers[:, 0])**2)
    violation += np.sum(np.maximum(0.0, centers[:, 0] + radii - 1.0)**2)
    violation += np.sum(np.maximum(0.0, radii - centers[:, 1])**2)
    violation += np.sum(np.maximum(0.0, centers[:, 1] + radii - 1.0)**2)
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violation += (min_dist - dist) ** 2
    
    return violation


def objective(params, n, penalty_weight):
    """Objective: maximize sum of radii with penalty for constraint violations."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    obj_val = -np.sum(radii)
    violation = compute_violations(centers, radii)
    return obj_val + penalty_weight * violation


def init_hexagonal_pattern(n, r):
    """Initialize circles in hexagonal packing pattern."""
    centers_list = []
    radii_list = []
    
    row_configs = [
        [5, 4, 5, 4, 5, 3],
        [5, 4, 5, 4, 4, 4],
        [4, 5, 4, 5, 4, 4],
        [5, 5, 5, 5, 3, 3],
        [6, 5, 5, 5, 4, 1],
        [5, 5, 5, 4, 4, 3],
    ]
    
    idx = 0
    for row, count in enumerate(row_configs[0]):
        y = r + row * r * np.sqrt(3)
        for col in range(count):
            if row % 2 == 0:
                x = r + col * 2.0 * r
            else:
                x = 2.0 * r + col * 2.0 * r
            if idx < n:
                centers_list.append([x, y])
                radii_list.append(r)
                idx += 1
    
    return np.array(centers_list), np.array(radii_list)


def init_staggered_grid(n, r):
    """Initialize with a staggered grid that may fit better."""
    centers_list = []
    radii_list = []
    
    rows = 6
    base_cols = 5
    
    idx = 0
    for row in range(rows):
        y = r + row * r * 1.732050808
        num_cols = base_cols if row % 2 == 0 else base_cols - 1
        if row >= 4:
            num_cols = max(num_cols - 1, 3)
        
        for col in range(num_cols):
            if idx >= n:
                break
            if row % 2 == 0:
                x = r + col * 2.0 * r
            else:
                x = r + r + col * 2.0 * r
            centers_list.append([x, y])
            radii_list.append(r)
            idx += 1
    
    if idx < n:
        while len(centers_list) < n:
            centers_list.append([0.5, 0.5])
            radii_list.append(0.01)
    
    return np.array(centers_list[:n]), np.array(radii_list[:n])


def optimize_with_sequential_penalty(centers, radii, n):
    """Optimize using sequential penalty method."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    penalty = 100.0
    for _ in range(20):
        result = minimize(
            objective,
            x0,
            args=(n, penalty),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * 2 * n + [(1e-10, 0.5)] * n,
            options={'maxiter': 15000, 'ftol': 1e-18, 'gtol': 1e-12}
        )
        x0 = result.x
        
        centers_temp = x0[:2 * n].reshape(n, 2)
        radii_temp = x0[2 * n:]
        violation = compute_violations(centers_temp, radii_temp)
        
        if violation < 1e-14:
            break
        
        penalty *= 3.0
    
    centers_opt = x0[:2 * n].reshape(n, 2)
    radii_opt = x0[2 * n:]
    
    return centers_opt, radii_opt


def optimize_refined(centers, radii, n):
    """Additional refinement with very high penalties."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    for penalty in [50000.0, 100000.0, 200000.0]:
        result = minimize(
            objective,
            x0,
            args=(n, penalty),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * 2 * n + [(1e-10, 0.5)] * n,
            options={'maxiter': 20000, 'ftol': 1e-18}
        )
        x0 = result.x
    
    centers_opt = x0[:2 * n].reshape(n, 2)
    radii_opt = x0[2 * n:]
    
    return centers_opt, radii_opt


def try_random_init(n, seed):
    """Try optimization from random initialization."""
    rng = np.random.RandomState(seed)
    
    centers = rng.rand(n, 2) * 0.7 + 0.15
    radii = rng.rand(n) * 0.06 + 0.04
    
    centers_opt, radii_opt = optimize_with_sequential_penalty(centers, radii, n)
    return centers_opt, radii_opt


def try_perturbed_hexagonal(n, seed, r_base):
    """Try optimization from perturbed hexagonal initialization."""
    rng = np.random.RandomState(seed)
    
    centers, radii = init_hexagonal_pattern(n, r_base)
    
    centers_pert = centers + rng.randn(n, 2) * 0.02
    radii_pert = radii + rng.randn(n) * 0.008
    radii_pert = np.maximum(radii_pert, 1e-6)
    
    centers_pert[:, 0] = np.clip(centers_pert[:, 0], 0.02, 0.98)
    centers_pert[:, 1] = np.clip(centers_pert[:, 1], 0.02, 0.98)
    
    centers_opt, radii_opt = optimize_with_sequential_penalty(centers_pert, radii_pert, n)
    return centers_opt, radii_opt
