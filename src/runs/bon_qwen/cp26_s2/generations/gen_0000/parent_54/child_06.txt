import numpy as np
from scipy.optimize import minimize
import warnings

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    n = len(vars) // 3
    return -np.sum(vars[2::3])

def constraints(vars):
    """Constraint function: returns array of constraint values >= 0."""
    n = len(vars) // 3
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con_list = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    # Pairwise non-overlap constraints: dist^2 >= (r1+r2)^2
    dx = x[:, np.newaxis] - x[np.newaxis, :]
    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]
    
    dist_sq = dx**2 + dy**2
    rad_sum_sq = dr**2
    
    # Extract only upper triangular pairs (i < j)
    idx_i, idx_j = np.triu_indices(n, k=1)
    pair_cons = dist_sq[idx_i, idx_j] - rad_sum_sq[idx_i, idx_j]
    con_list.append(pair_cons)
    
    return np.concatenate(con_list)

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    warnings.filterwarnings('ignore')
    n = 26
    
    # Initialize with a 5x5 grid + center circle
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.125 + 0.25*i, 0.125 + 0.25*j]
            idx += 1
    centers[25] = [0.5, 0.5]
    
    # Initial radii
    radii = np.full(n, 0.1)
    
    # Flatten to 1D for optimizer: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3*n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    cons = {'type': 'ineq', 'fun': constraints}
    bounds = [(0, 1)] * (2*n) + [(0, 1)] * n
    
    # Run SLSQP optimization
    res = minimize(objective, x0, method='SLSQP', 
                   bounds=bounds, constraints=cons,
                   options={'ftol': 1e-10, 'maxiter': 5000, 'disp': False})
                   
    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = res.x[2*n:]
    
    # Apply tiny safety margin to satisfy strict validation tolerance
    final_radii = np.maximum(final_radii - 1e-7, 0.0)
    
    return final_centers, final_radii, np.sum(final_radii)
