import numpy as np
from scipy.optimize import minimize

N = 26
PENALTY_WEIGHT = 2000

def objective(x):
    """Objective: maximize sum of radii (minimize negative sum) with penalty for constraint violations."""
    c = x[:2 * N].reshape(N, 2)
    r = x[2 * N:]
    
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties
    for i in range(N):
        for coord in range(2):
            lower_violation = c[i, coord] - r[i]
            if lower_violation < 0:
                pen += PENALTY_WEIGHT * lower_violation ** 2
            upper_violation = c[i, coord] + r[i] - 1.0
            if upper_violation > 0:
                pen += PENALTY_WEIGHT * upper_violation ** 2
    
    # Overlap penalties
    for i in range(N):
        for j in range(i + 1, N):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = r[i] + r[j] - dist
            if overlap > 0:
                pen += PENALTY_WEIGHT * overlap ** 2
    
    return obj + pen

def hexagonal_init():
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((N, 2))
    idx = 0
    
    row_spacing = 0.175
    col_spacing = 0.2
    
    for row in range(6):
        cols_in_row = 5 if row % 2 == 0 else 4
        offset = col_spacing / 2.0 if row % 2 == 1 else 0.0
        for col in range(cols_in_row):
            if idx >= N:
                break
            x = (col + 0.5) * col_spacing + offset
            y = (row + 0.5) * row_spacing
            centers[idx] = [x, y]
            idx += 1
    
    return centers

def regular_grid_init():
    """Initialize circles in a regular grid pattern."""
    centers = np.zeros((N, 2))
    
    # 6x5 grid = 30 slots, use first 26
    row_count = 5
    col_count = 6
    spacing_x = 1.0 / (col_count + 1)
    spacing_y = 1.0 / (row_count + 1)
    
    idx = 0
    for row in range(row_count):
        for col in range(col_count):
            if idx >= N:
                break
            x = (col + 1) * spacing_x
            y = (row + 1) * spacing_y
            centers[idx] = [x, y]
            idx += 1
    
    return centers

def run_packing():
    """Pack 26 circles in a unit square to maximize sum of radii."""
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Try multiple initializations
    init_functions = [hexagonal_init, regular_grid_init]
    
    for init_idx, init_func in enumerate(init_functions):
        centers = init_func()
        radii = np.ones(N) * 0.08
        
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Try with different penalty weights
        for pw_idx, pw in enumerate([1000, 2000, 4000]):
            global PENALTY_WEIGHT
            PENALTY_WEIGHT = pw
            
            # Add slight perturbation for diversity
            x_init = x0.copy()
            x_init[:2 * N] += np.random.randn(2 * N) * 0.015
            
            result = minimize(
                objective,
                x_init,
                method='BFGS',
                options={'maxiter': 3000, 'gtol': 1e-10}
            )
            
            centers_opt = result.x[:2 * N].reshape(N, 2)
            radii_opt = np.maximum(result.x[2 * N:], 1e-10)
            
            # Enforce boundary constraints
            for i in range(N):
                centers_opt[i, 0] = np.clip(centers_opt[i, 0], radii_opt[i], 1 - radii_opt[i])
                centers_opt[i, 1] = np.clip(centers_opt[i, 1], radii_opt[i], 1 - radii_opt[i])
            
            current_sum = np.sum(radii_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
    
    # Refinement: do a final optimization with high penalty to tighten constraints
    if best_centers is not None:
        PENALTY_WEIGHT = 8000
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        
        result = minimize(
            objective,
            x0,
            method='BFGS',
            options={'maxiter': 2000, 'gtol': 1e-12}
        )
        
        centers_opt = result.x[:2 * N].reshape(N, 2)
        radii_opt = np.maximum(result.x[2 * N:], 1e-10)
        
        for i in range(N):
            centers_opt[i, 0] = np.clip(centers_opt[i, 0], radii_opt[i], 1 - radii_opt[i])
            centers_opt[i, 1] = np.clip(centers_opt[i, 1], radii_opt[i], 1 - radii_opt[i])
        
        final_sum = np.sum(radii_opt)
        if final_sum > best_sum:
            best_centers = centers_opt
            best_radii = radii_opt
            best_sum = final_sum
    
    return best_centers, best_radii, best_sum
