import numpy as np
import math
from scipy.optimize import minimize

# Helper functions

def compute_violations(centers, radii, penalty_weight):
    """
    Computes the total penalty for overlaps and boundary violations.
    
    Args:
        centers: np.array (n, 2)
        radii: np.array (n)
        penalty_weight: float, weight for the penalty term
        
    Returns:
        float: Total penalty
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary violations
    # x - r >= 0  =>  r - x <= 0
    # x + r <= 1  =>  x + r - 1 <= 0
    # y - r >= 0
    # y + r <= 1
    
    # We penalize if r > x or r > (1-x) or r > y or r > (1-y)
    # Equivalent to max(0, r - x)^2 + max(0, x + r - 1)^2 ...
    
    # Vectorized boundary check
    # Left
    viol_left = np.maximum(0, radii - centers[:, 0])
    # Right
    viol_right = np.maximum(0, centers[:, 0] + radii - 1.0)
    # Bottom
    viol_bottom = np.maximum(0, radii - centers[:, 1])
    # Top
    viol_top = np.maximum(0, centers[:, 1] + radii - 1.0)
    
    boundary_penalty = np.sum(viol_left**2 + viol_right**2 + viol_bottom**2 + viol_top**2)
    penalty += boundary_penalty

    # Overlap violations
    # dist(c_i, c_j) >= r_i + r_j
    # r_i + r_j - dist <= 0
    # Violation: max(0, r_i + r_j - dist)
    
    # Compute pairwise distances
    # Efficiently compute distance matrix
    # diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    # dists = np.sqrt(np.sum(diff**2, axis=2))
    # This is O(N^3) memory for diff, O(N^2) for dists. N=26 is small.
    
    # Optimized pairwise loop or vectorization
    # Using broadcasting
    c1 = centers[:, np.newaxis, :] # (26, 1, 2)
    c2 = centers[np.newaxis, :, :] # (1, 26, 2)
    diff = c1 - c2
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Radii sum matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount
    overlap = r_sum - dists
    
    # Only consider upper triangle to avoid double counting and self
    # Or just sum all, self overlap is 0 (r+r - 0 = 2r, but dist=0... wait)
    # dist[i,i] = 0. r[i]+r[i] - 0 = 2r[i]. This would be a huge penalty!
    # We must exclude i==j.
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap_upper = np.where(mask, overlap, 0.0)
    
    # Positive overlaps are violations
    viol_overlap = np.maximum(0, overlap_upper)
    overlap_penalty = np.sum(viol_overlap**2)
    
    penalty += overlap_penalty
    
    return penalty * penalty_weight

def objective_function(x, penalty_weight):
    """
    Objective function to minimize.
    We want to maximize sum(radii), so we minimize -sum(radii) + penalty.
    
    Args:
        x: flattened array of [x1, y1, r1, x2, y2, r2, ...]
        penalty_weight: float
        
    Returns:
        float: value to minimize
    """
    n = 26
    centers = x[:2*n].reshape((n, 2))
    radii = x[2*n:]
    
    sum_radii = np.sum(radii)
    penalty = compute_violations(centers, radii, penalty_weight)
    
    return -sum_radii + penalty

def run_packing():
