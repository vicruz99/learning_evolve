import numpy as np
from scipy.optimize import minimize

def objective(vars, n, lam):
    """
    Objective function with penalty terms for boundary and overlap constraints.
    Minimizes -sum(radii) + penalty.
    """
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    # Boundary penalties: r <= x, r <= 1-x, r <= y, r <= 1-y
    pen = 0.0
    pen += np.sum(np.maximum(0, r - x)**2)
    pen += np.sum(np.maximum(0, r - (1.0 - x))**2)
    pen += np.sum(np.maximum(0, r - y)**2)
    pen += np.sum(np.maximum(0, r - (1.0 - y))**2)
    
    # Inter-circle penalties: r_i + r_j <= distance(i, j)
    cx = x[:, None] - x[None, :]
    cy = y[:, None] - y[None, :]
    dist = np.sqrt(cx**2 + cy**2)
    
    ri = r[:, None] + r[None, :]
    viol = np.maximum(0, ri - dist)
    
    # Only consider each pair once (upper triangle)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pen += np.sum((viol * mask)**2)
    
    return -np.sum(r) + lam * pen

def run_packing():
    n = 26
    np.random.seed(42)
    
    best_vars = None
    best_sum = -1.0
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # Multiple random restarts to escape local minima
    for _ in range(50):
        # Initialize centers away from boundaries to allow initial growth
        x = np.random.uniform(0.2, 0.8, n)
        y = np.random.uniform(0.2, 0.8, n)
        r = np.full(n, 0.02)
        v0 = np.concatenate([x, y, r])
        
        v = v0
        # Gradually increase penalty weight to tighten constraints
        for lam in [50, 200, 1000, 2000]:
            res = minimize(objective, v, args=(n, lam), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-15})
            v = res.x
            
        r_final = v[2*n:]
        current_sum = np.sum(r_final)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_vars = v.copy()
            
    centers = np.column_stack((best_vars[:n], best_vars[n:2*n]))
    radii = best_vars[2*n:]
    return centers, radii, np.sum(radii)
