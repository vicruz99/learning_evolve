import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization followed by numerical optimization.
    """
    n = 26
    
    # --- Step 1: Initialization ---
    # We generate points on a hexagonal grid.
    # Tentative radius for initialization
    r_init = 0.09
    
    # Hexagonal grid parameters
    # Vertical distance between rows
    dy = np.sqrt(3) * r_init
    # Horizontal distance between centers in a row
    dx = 2 * r_init
    
    centers_list = []
    
    # Generate rows
    # y starts at r_init
    y = r_init
    while y + r_init <= 1.0 + 1e-9:
        # Determine offset for this row
        # Even rows (0-indexed relative to y start) start at x = r_init
        # Odd rows are offset by dx/2 = r_init, so they start at x = 2*r_init
        # However, to fit better, let's check which pattern allows more points.
        # Standard hex packing: Row 0 at x=r, Row 1 at x=2r (offset r), Row 2 at x=r...
        # Actually, offset is usually half spacing. Spacing is 2r. Offset is r.
        # If Row 0 starts at r, Row 1 starts at r+r = 2r?
        # Let's stick to: Row 0 x = r, 3r, 5r...
        # Row 1 x = 2r, 4r, 6r... (shifted by r)
        
        # Calculate row index to determine offset
        # We can approximate row index by (y - r_init) / dy
        row_idx = int(round((y - r_init) / dy))
        
        x_start = r_init if row_idx % 2 == 0 else 2 * r_init
        
        x = x_start
        while x + r_init <= 1.0 + 1e-9:
            centers_list.append([x, y])
            x += dx
        
        y += dy
    
    # We might have generated more or fewer than 26 points.
    # We want exactly 26.
    # Let's trim or pad if necessary, but with r=0.09 we should have plenty (around 30).
    # We select the first 26 to maintain a dense packing in one corner/region?
    # Better to select points that are most central or just first 26 if they fill the space.
    # The generation order fills from bottom-left. This is fine for initialization.
    
    if len(centers_list) > n:
        # Select a subset. To maximize symmetry and fill, maybe remove corners?
        # For simplicity and robustness, let's just take the first n.
        # The optimizer will move them.
        centers_list = centers_list[:n]
    elif len(centers_list) < n:
        # If fewer, add random points or grid points.
        # Unlikely with r=0.09.
        pass

    centers_init = np.array(centers_list)
    radii_init = np.array([r_init] * n)
    
    # --- Step 2: Optimization ---
    # We want to maximize sum(radii) => minimize -sum(radii)
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 3 * n
    
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]
        
    # Bounds for variables:
    # x, y in [0, 1]
    # r >= 0 (and implicitly <= 1)
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 1)) # r
    
    # Constraints:
    # 1. Boundary constraints:
    #    x - r >= 0  => x >= r
    #    x + r <= 1  => x <= 1 - r
    #    y - r >= 0  => y >= r
    #    y + r <= 1  => y <= 1 - r
    #    These can be formulated as:
    #    x - r >= 0
    #    1 - x - r >= 0
    #    y - r >= 0
    #    1 - y - r >= 0
    
    # 2. Non-overlap constraints:
    #    dist(i, j) >= r_i + r_j
    #    sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
    #    To avoid sqrt in constraints (differentiability/efficiency), we can use squared distance,
    #    but scipy constraints handle general functions. 
    #    dist - (ri + rj) >= 0 is fine.
    
    cons = []
    
    # Add boundary constraints for each circle
    for i in range(n):
        idx_x = 3*i
        idx_y = 3*i + 1
        idx_r = 3*i + 2
        
        # x >= r
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y >= r
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })
        
    # Add non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i + 1, 3*i + 2

            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: np.sqrt((v[idx_xi] - v[idx_xj])**2 + (v[idx_yi] - v[idx_yj])**2) - (v[idx_ri] + v[idx_rj])
            })
            
    # Objective function
    def objective(v):
        # Sum of radii, negative because we minimize
        return -sum(v[3*i + 2] for i in range(n))
        
    # Run optimizer
    # SLSQP is suitable for constrained optimization
    # We might need to run it a few times or use a robust method.
    # Given the constraints are non-convex, we rely on the good initialization.
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})
        
        if res.success or res.status == 0:
            final_params = res.x
        else:
            # Fallback to initial if optimization fails, though unlikely
            final_params = x0
            
    except Exception:
        final_params = x0
        
    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = final_params[3*i]
        centers[i, 1] = final_params[3*i+1]
        radii[i] = final_params[3*i+2]
        
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
