import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # Helper to validate constraints inside the optimization loop isn't strictly necessary 
    # if constraints are defined in the optimizer, but useful for debugging or final check.
    # The optimizer constraints handle the logic.

    def objective(vars):
        # vars shape: (78,) -> [x1, y1, r1, ..., x26, y26, r26]
        radii = vars[2::3]
        return -np.sum(radii)  # Minimize negative sum

    def boundary_constraints(vars):
        constraints_list = []
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i + 1]
            r = vars[3*i + 2]
            # x - r >= 0
            constraints_list.append(x - r)
            # x + r <= 1 => 1 - (x + r) >= 0
            constraints_list.append(1.0 - (x + r))
            # y - r >= 0
            constraints_list.append(y - r)
            # y + r <= 1 => 1 - (y + r) >= 0
            constraints_list.append(1.0 - (y + r))
            # r >= 0 (handled by bounds usually, but explicit constraint ok)
            constraints_list.append(r)
        return constraints_list

    def non_overlap_constraints(vars):
        constraints_list = []
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i = vars[3*i], vars[3*i + 1]
                x_j, y_j = vars[3*j], vars[3*j + 1]
                r_i, r_j = vars[3*i + 2], vars[3*j + 2]
                
                dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
                min_dist_sq = (r_i + r_j)**2
                
                # dist >= r_i + r_j  =>  dist^2 >= (r_i + r_j)^2
                # We use a slightly relaxed constraint or exact?
                # Exact: dist_sq - min_dist_sq >= 0
                constraints_list.append(dist_sq - min_dist_sq)
        return constraints_list

    # Initial guess generation
    # We want a dense initial configuration to help the optimizer start near a good solution.
    # A jittered grid or hexagonal packing is good.
    def get_initial_guess():
        # Try to place circles in a hexagonal-like grid
        # Estimate radius ~ 0.1
        r_init = 0.08 
        
        # 5 rows, varying counts to sum to 26
        # Pattern: 5, 4, 5, 4, 5, 3 -> sum 26? 5+4+5+4+5+3 = 26.
        # Or 6 rows?
        # Let's try a simple grid first and let optimizer move them.
        # 6x5 grid has 30 spots. Pick 26.
        
        cols = 6
        rows = 5
        x_coords = np.linspace(0.1, 0.9, cols) # Spread within [0.1, 0.9]
        y_coords = np.linspace(0.1, 0.9, rows)
        
        centers = []
        count = 0
        for y in y_coords:
            for x in x_coords:
                if count < n:
                    centers.append([x, y])
                    count += 1
        
        # If we didn't fill enough (should not happen with 6x5=30), add more
        # But 30 > 26, so we just take first 26.
        centers = np.array(centers[:n])
        
        # Add some jitter to break symmetry
        jitter = np.random.uniform(-0.02, 0.02, size=(n, 2))
        centers += jitter
        
        # Clip to valid range
        centers = np.clip(centers, 0.05, 0.95)
        
        radii = np.full(n, r_init)
        
        vars = np.zeros(3 * n)
        for i in range(n):
            vars[3*i] = centers[i, 0]
            vars[3*i + 1] = centers[i, 1]
            vars[3*i + 2] = radii[i]
            
        return vars

    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])

    # Define constraints for optimizer
    # SLSQP expects a list of constraint dictionaries
    # type 'ineq' means func(x) >= 0
    
    # Note: Defining constraints as functions inside the loop might be slow if called repeatedly
    # But for n=26, 300 constraints is manageable.
    
    # To speed up, we can define a single constraint function that returns all values
    # But SLSQP handles list of dicts or a single dict with vector function?
    # scipy.optimize.minimize constraints can be a dict with 'type': 'ineq' and 'fun' returning vector.
    
    def all_constraints(vars):
        vals = []
        # Boundary
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i + 1]
            r = vars[3*i + 2]
            vals.append(x - r)
            vals.append(1.0 - (x + r))
            vals.append(y - r)
            vals.append(1.0 - (y + r))
            vals.append(r) # r >= 0

                dx = vars[3*i] - vars[3*j]

                dr = vars[3*i+2] + vars[3*j+2]
                vals.append(dx*dx + dy*dy - dr*dr)
                
        return np.array(vals)

                x0,

                constraints=cons,

        # 5x5 grid radius 0.1 -> 25 circles. Add one small?

        return centers, radii, best_sum

    # Extract results
    best_vars = best_vars.reshape(-1, 3)
    centers = best_vars[:, :2]
    radii = best_vars[:, 2]
    total_sum = np.sum(radii)

    return centers, radii, total_sum
