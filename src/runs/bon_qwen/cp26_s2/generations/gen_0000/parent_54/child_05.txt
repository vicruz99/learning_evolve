import numpy as np
from scipy.optimize import minimize

def bound_constraints(x, n):
    """Returns array of boundary constraint values. Must be >= 0."""
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    cons = []
    for i in range(n):
        # x - r >= 0
        cons.append(centers[i, 0] - radii[i])
        # 1 - x - r >= 0
        cons.append(1.0 - centers[i, 0] - radii[i])
        # y - r >= 0
        cons.append(centers[i, 1] - radii[i])
        # 1 - y - r >= 0
        cons.append(1.0 - centers[i, 1] - radii[i])
        # r >= 0
        cons.append(radii[i])
    return np.array(cons)

def overlap_constraints(x, n):
    """Returns array of non-overlap constraint values. Must be >= 0."""
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            d2 = np.sum((centers[i] - centers[j]) ** 2)
            r_sum = radii[i] + radii[j]
            # d^2 - (ri+rj)^2 >= 0
            cons.append(d2 - r_sum ** 2)
    return np.array(cons)

def neg_sum_radii(x, n):
    """Objective function to minimize (negative sum of radii)."""
    return -np.sum(x[2*n:])

def run_packing():
    n = 26
    best_sum = -1.0
    best_x = None
    
    # Run multiple optimizations with different initial seeds/perturbations
    for trial in range(15):
        # Hexagonal grid initialization provides a dense starting configuration
        cols = 6
        rows = 5
        xs = np.linspace(0.15, 0.85, cols)
        ys = np.linspace(0.15, 0.85, rows)
        centers = []
        for r_idx in range(rows):
            offset = (r_idx % 2) * (xs[1] - xs[0]) / 2.0
            for c_idx in range(cols):
                cx = xs[c_idx] + offset
                cy = ys[r_idx]
                if 0 <= cx <= 1:
                    centers.append([cx, cy])
                    if len(centers) >= n:
                        break
            if len(centers) >= n:
                break
        centers = np.array(centers[:n])
        
        # Add controlled random noise to break symmetry and escape trivial local optima
        np.random.seed(trial * 13 + 7)
        centers += np.random.randn(n, 2) * 0.02
        centers = np.clip(centers, 0.12, 0.88)
        
        # Start with feasible radii
        r_init = 0.065 + trial * 0.001
        radii = np.full(n, r_init)
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Variable bounds
        bnds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        cons = [
            {'type': 'ineq', 'fun': bound_constraints, 'args': (n,)},
            {'type': 'ineq', 'fun': overlap_constraints, 'args': (n,)}
        ]
        
        try:
            res = minimize(neg_sum_radii, x0, args=(n,), method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            
            current_sum = np.sum(res.x[2*n:])
            if current_sum > best_sum:
                best_sum = current_sum
                best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        # Fallback (should not trigger)
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.full(n, 0.01)
        return centers, radii, np.sum(radii)
        
    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]
    
    # Ensure strict non-negativity and bounds compliance
    radii = np.maximum(radii, 1e-10)
    centers = np.clip(centers, 1e-10, 1.0 - 1e-10)
    
    return centers, radii, np.sum(radii)
