from scipy.optimize import minimize

def run_packing():

        # To fit 'count' circles of diameter 2r, we need width approx 2r + (count-1)*2r = count*2r
        # But we can scale to fit within [0,1] roughly.

        # and rely on optimizer to fix positions.
        # Better: spread them with spacing 2*r_init.
        spacing = 2.0 * r_init * 1.1 # Slight gap
        
        x_start = (1.0 - (count - 1) * spacing) / 2.0
        xs = [x_start + i * spacing + shift_x for i in range(count)]
        return xs

    # but clustered to avoid immediate massive overlap, or a simple grid.

    # Let's use a 5x5 grid (25 points) and add one at (0.5, 0.5) perturbed?
    # No, 5x5 is dense.

    # Rows: 5, 5, 6, 5, 5 is hard for width.
    # Let's try 4, 5, 4, 5, 4, 4? 6 rows.
    # y_0 = r, y_1 = r + sqrt(3)r, ...
    
    # Simplest robust init: Random in [0.1, 0.9] with small radius
    np.random.seed(42)
    centers_list = []
    for _ in range(n):
        centers_list.append([0.1 + 0.8 * np.random.rand(), 0.1 + 0.8 * np.random.rand()])
    centers = np.array(centers_list)
    radii = np.full(n, 0.05) # Start small
    
    # --- Optimization Setup ---
    
    def objective(vars):
        # Unpack variables
        # vars: [x1, y1, r1, x2, y2, r2, ...]
        c = vars[:2*n].reshape(n, 2)
        r = vars[2*n:]
        
        # Objective: Maximize sum(r) => Minimize -sum(r)
        obj = -np.sum(r)
        
        # Penalty Terms
        penalty = 0.0
        
        # 1. Overlap Penalty
        # dist >= r_i + r_j  =>  r_i + r_j - dist <= 0
        # We penalize max(0, r_i + r_j - dist)^2
        # Vectorized computation for efficiency
        # Compute pairwise distances
        # c_diff shape (n, n, 2)
        c_diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(c_diff**2, axis=2))
        
        # r_sum shape (n, n)
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        # Overlap violations
        violations = r_sum - dists
        # We only care about upper triangle i < j
        # Mask for i < j
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        overlaps = violations[mask]
        
        # Smooth penalty
        # penalty += sum(max(0, overlap)^2)
        # Using np.maximum for vectorization
        penalty += np.sum(np.maximum(0, overlaps)**2)
        
        # 2. Boundary Penalty
        # Constraints:
        # x - r >= 0  => r - x <= 0
        # x + r <= 1  => x + r - 1 <= 0
        # y - r >= 0  => r - y <= 0
        # y + r <= 1  => y + r - 1 <= 0
        
        # x constraints
        penalty += np.sum(np.maximum(0, r - c[:, 0])**2)
        penalty += np.sum(np.maximum(0, c[:, 0] + r - 1)**2)
        
        # y constraints
        penalty += np.sum(np.maximum(0, r - c[:, 1])**2)
        penalty += np.sum(np.maximum(0, c[:, 1] + r - 1)**2)
        
        # Weighted penalty. 
        # High weight ensures constraints are respected.
        # As we optimize, we might want to adjust, but constant high weight usually works for L-BFGS-B
        # if initialized well.
        penalty_weight = 1000.0
        
        return obj + penalty_weight * penalty

    # Bounds
    # x, y in [0, 1]
    # r in [0, 1] (loosely bounded, effectively limited by geometry)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (cannot be > 0.5)
        
    # Initial state vector
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Run optimization
    # Using L-BFGS-B for bounds
    # We might run multiple restarts or just one good run.
    # Given the penalty method, one run from a reasonable start should converge.
    
    # To improve chances, we can run a few iterations with increasing penalty or just a strong penalty.
    # Let's try a strong penalty.
    
    res = minimize(
        objective, 
        x0, 
        method='L-BFGS-B', 
        bounds=bounds, 
        options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    
    # Extract results
    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = res.x[2*n:]
    
    # Post-check: if penalty is high, we might need to project or adjust.
    # But with high weight, it should be low.
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
