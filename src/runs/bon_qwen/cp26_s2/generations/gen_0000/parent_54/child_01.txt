import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP optimization on centers and radii simultaneously.
    """
    n = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    # Helper to generate hexagonal grid initialization
    def get_hex_grid_init(n_circles, seed_offset=0):
        # Estimate radius for packing n circles in unit square
        # Density approx 0.9. Area pi * r^2 * n approx 0.9
        # r approx sqrt(0.9 / (pi * n))
        r_est = np.sqrt(0.9 / (np.pi * n_circles))
        d = 2 * r_est
        
        centers = np.zeros((n_circles, 2))
        idx = 0
        
        # Hexagonal packing generation
        # Row spacing: sqrt(3)/2 * d
        # Col spacing: d (shifted by d/2 in alternate rows)
        y_step = np.sqrt(3) / 2 * d
        x_step = d
        
        y = r_est # Start at radius distance from bottom
        row = 0
        
        while idx < n_circles and y + r_est <= 1.0 + 1e-6:
            # Determine shift for this row
            shift = (0.5 * d) if row % 2 == 1 else 0
            
            # Add small random perturbation to avoid symmetry traps
            x_start = r_est + shift + seed_offset * 0.001 
            
            x = x_start
            while idx < n_circles and x + r_est <= 1.0 + 1e-6:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
                x += x_step
            
            y += y_step
            row += 1
            
        # If we didn't get enough circles, fill randomly in valid spots
        # (Though hex grid usually fits 26 with small r)
        while idx < n_circles:
            # Place in a grid if hex failed to fill
            grid_idx = idx
            r_grid = 0.1
            cx = (grid_idx % 6 + 0.5) / 6.0 + seed_offset * 0.001
            cy = (grid_idx // 6 + 0.5) / 6.0 + seed_offset * 0.001
            centers[idx] = [cx, cy]
            idx += 1
            
        return centers, r_est

    # Objective function: minimize negative sum of radii
    def objective(vars_flat):
        radii = vars_flat[2*n:]
        return -np.sum(radii)

    def constraints_func(vars_flat):
        centers = vars_flat[:2*n].reshape(n, 2)
        radii = vars_flat[2*n:]
        con_vals = []

        # x - r >= 0  =>  x - r
        # 1 - x - r >= 0 => 1 - x - r
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            con_vals.append(x - r)
            con_vals.append(1.0 - x - r)
            con_vals.append(y - r)
            con_vals.append(1.0 - y - r)

        # 2. Non-overlap constraints: dist(c_i, c_j)^2 >= (r_i + r_j)^2
        # dx^2 + dy^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                r_sum = radii[i] + radii[j]
                con_vals.append(dist_sq - r_sum*r_sum)
        
        return np.array(con_vals)

    constraints = {'type': 'ineq', 'fun': constraints_func}

    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    num_restarts = 5

        # Flatten variables
        x0 = np.concatenate([centers_init.flatten(), radii_init])

        # Optimize
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints, 
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )

                current_sum = -res.fun
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_centers = res.x[:2*n].reshape(n, 2)
                    best_radii = res.x[2*n:]
        except Exception:
            continue

    # Final validation and cleanup
    # Ensure radii are not negative (though bounds should handle it)
    if best_radii is not None:
        best_radii = np.maximum(best_radii, 0.0)

    if best_centers is None:

                    idx += 1
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
