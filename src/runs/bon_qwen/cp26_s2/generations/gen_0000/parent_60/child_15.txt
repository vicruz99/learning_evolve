import numpy as np
from scipy.optimize import minimize
import math
import random

def get_initial_guess(n):
    """Generate a hexagonal lattice initial configuration."""
    pts = []
    h = math.sqrt(3) / 2
    # Create a hexagonal pattern
    for row in range(5):
        y = row * h
        if row % 2 == 0:
            for col in range(6):
                pts.append((col, y))
        else:
            for col in range(5):
                pts.append((col + 0.5, y))
                
    xs = [p[0] for p in pts]
    ys = [p[1] for p in pts]
    min_x, max_x = min(xs), max(xs)
    min_y, max_y = min(ys), max(ys)
    
    # Scale and center within [0.1, 0.9] to leave boundary margin
    scale_x = 0.8 / (max_x - min_x)
    scale_y = 0.8 / (max_y - min_y)
    scale = min(scale_x, scale_y)
    
    centers = np.zeros((n, 2))
    for i, (x, y) in enumerate(pts):
        centers[i, 0] = 0.1 + (x - min_x) * scale
        centers[i, 1] = 0.1 + (y - min_y) * scale
        
    # Initial radii chosen to fit the scaled spacing
    radii = np.full(n, 0.08)
    
    # Flatten to optimization vector: [x0, y0, r0, x1, y1, r1, ...]
    p0 = np.zeros(3 * n)
    for i in range(n):
        p0[i*3] = centers[i, 0]
        p0[i*3+1] = centers[i, 1]
        p0[i*3+2] = radii[i]
        
    return p0

def objective_func(p, n, penalty):
    """
    Objective function: Minimize -sum(radii) + penalty * constraint_violations^2
    """
    centers = p.reshape(n, 3)[:, :2]
    radii = p.reshape(n, 3)[:, 2]
    
    # Start with negative sum of radii (since we minimize)
    loss = -np.sum(radii)
    
    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    v1 = radii - centers[:, 0]
    v2 = radii - (1.0 - centers[:, 0])
    v3 = radii - centers[:, 1]
    v4 = radii - (1.0 - centers[:, 1])
    
    v_boundary = np.maximum(0.0, v1)**2 + np.maximum(0.0, v2)**2 + \
                 np.maximum(0.0, v3)**2 + np.maximum(0.0, v4)**2
    loss += penalty * np.sum(v_boundary)
    
    # Overlap constraints: dist >= ri + rj  =>  ri + rj - dist <= 0
    # Vectorized pairwise distances
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    sum_radii = radii[:, None] + radii[None, :]
    
    v_overlap = np.maximum(0.0, sum_radii - dists)
    # Zero out lower triangle and diagonal to avoid double counting
    mask = np.triu(np.ones((n, n)), k=1).astype(bool)
    v_overlap[~mask] = 0.0
    
    loss += penalty * np.sum(v_overlap**2)
    
    return loss

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Set seeds for reproducibility
    np.random.seed(42)
    random.seed(42)
    
    n = 26
    best_loss = np.inf
    best_p = None
    
    # Generate base initial guess
    p0 = get_initial_guess(n)
    
    # Define bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    # Multi-start optimization
    num_restarts = 5
    for restart in range(num_restarts):
        if restart > 0:
            # Perturb initial guess slightly
            pert = np.random.randn(3 * n) * 0.01
            p_current = np.clip(p0 + pert, 1e-5, 0.999)
            # Ensure radii bounds
            p_current[2::3] = np.clip(p_current[2::3], 1e-5, 0.5)
        else:
            p_current = p0.copy()
            
        # Annealing penalty schedule
        penalties = [100, 500, 2000, 10000]
        for pen in penalties:
            res = minimize(objective_func, p_current, args=(n, pen), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12})
            p_current = res.x
            
        # Evaluate final loss with a very high penalty to check quality
        final_loss = objective_func(p_current, n, 50000)
        if final_loss < best_loss:
            best_loss = final_loss
            best_p = p_current.copy()
            
    # Extract results
    centers = best_p.reshape(n, 3)[:, :2]
    radii = best_p.reshape(n, 3)[:, 2]
    
    # Ensure strict validity by checking and shrinking if necessary
    # This handles potential numerical edge cases
    valid = True
    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            valid = False
            break
            
    # Check overlaps
    if valid:
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
                
    # If not valid (should be rare with high penalty), shrink radii slightly
    if not valid:
        scale = 0.99
        radii *= scale
        centers *= scale # Not necessary but safe
        
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
