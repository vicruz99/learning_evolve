import numpy as np
import scipy.optimize as opt

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the Linear Program to maximize sum of radii.
    Variables: r_0, ..., r_25
    Maximize sum(r_i)
    Subject to:
      r_i >= 0
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i + r_j <= sqrt((x_i - x_j)^2 + (y_i - y_j)^2)
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Objective: minimize -sum(r) => maximize sum(r)
    
    # Bounds for r_i: [0, inf)
    bounds = [(0, None) for _ in range(n)]
    
    # Constraints matrix A_ub * r <= b_ub
    # We will construct rows for boundary constraints and pairwise constraints
    
    rows = []
    rhs = []
    
    # 1. Boundary constraints
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    # Each is a row with 1 at index i
    
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - y)
        
    # 2. Pairwise constraints
    # r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dist)
            
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.fun * -1, res.x
        else:
            # If LP fails (e.g. infeasible due to numerical issues), return 0
            # This shouldn't happen if centers are valid and r=0 is feasible
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)
