This is a circle packing optimization problem: pack 26 circles in [0,1]² to maximize sum of radii.

**Key insights:**
1. For equal circles in hexagonal packing, r ≈ 0.1 for 26 circles, giving sum ≈ 2.6
2. The target is 2.636, so we need to optimize both positions and potentially allow non-equal radii
3. Corners and edges are valuable - circles there can be larger since they have fewer neighbors to avoid
4. A penalty method approach with L-BFGS-B optimization should work well

**Strategy:**
1. Generate diverse initial configurations (hexagonal, rectangular, perturbed)
2. Use sequential quadratic programming via L-BFGS-B with increasing penalty weights
3. The penalty function penalizes boundary violations and circle overlaps
4. After optimization, repair any remaining constraint violations
5. Return the best valid configuration found across all restarts

**Implementation details:**
- Penalty method: objective = -sum(radii) + weight × (boundary_violations² + overlap_violations²)
- Sequential penalties with weights [200, 1000, 5000, 15000, 30000] to gradually enforce constraints
- Multiple initial configurations to escape local minima
- Post-processing repair function to fix minor violations

import numpy as np
from scipy.optimize import minimize


def run_packing():
    """Pack 26 circles in a unit square to maximize sum of radii."""
    n = 26

    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum = 0.0

    # Generate diverse initial configurations
    configs = []

    # Hexagonal patterns with different radii
    for r0 in [0.075, 0.08, 0.085, 0.09, 0.095, 0.1]:
        configs.append(hex_init(n, r0))

    # Rectangular patterns
    for r0 in [0.075, 0.08, 0.085, 0.09]:
        configs.append(rect_init(n, r0))

    # Perturbed hexagonal configurations
    np.random.seed(42)
    for _ in range(15):
        c, r = hex_init(n, 0.087)
        c = c + np.random.uniform(-0.02, 0.02, c.shape)
        c = np.clip(c, 0.03, 0.97)
        configs.append((c, r))

    # Perturbed rectangular configurations
    for _ in range(10):
        c, r = rect_init(n, 0.087)
        c = c + np.random.uniform(-0.02, 0.02, c.shape)
        c = np.clip(c, 0.03, 0.97)
        configs.append((c, r))

    # Try each configuration
    for c0, r0 in configs:
        x0 = np.concatenate([c0.flatten(), r0])
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

        # Sequential penalty optimization with increasing weights
        x = x0.copy()
        for weight in [100, 500, 2000, 8000, 20000, 50000]:
            result = minimize(
                objective_function,
                x,
                args=(n, weight),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 50000, "ftol": 1e-15, "gtol": 1e-10},
            )
            x = result.x

        centers = x[: 2 * n].reshape(n, 2)
        radii = x[2 * n:]

        # Repair any remaining constraint violations
        centers, radii = repair_packing(centers, radii, n)

        # Validate and track best
        if is_valid(centers, radii, n):
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()

    return best_centers, best_radii, best_sum


def objective_function(params, n, weight):
    """Objective: maximize sum of radii with penalty for constraint violations."""
    centers = params[: 2 * n].reshape(n, 2)
    radii = params[2 * n :]

    # Primary objective: maximize sum of radii (minimize negative)
    obj = -np.sum(radii)

    # Penalty for constraint violations
    penalty = 0.0

    # Boundary penalties: circle must be within [0,1]^2
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]

        if x < r:
            penalty += weight * (r - x) ** 2
        if x + r > 1.0:
            penalty += weight * (x + r - 1.0) ** 2
        if y < r:
            penalty += weight * (r - y) ** 2
        if y + r > 1.0:
            penalty += weight * (y + r - 1.0) ** 2

    # Overlap penalties: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]

            if dist_sq < r_sum * r_sum:
                dist = np.sqrt(dist_sq)
                if dist < 1e-14:
                    dist = 1e-14
                penalty += weight * (r_sum - dist) ** 2

    return obj + penalty


def hex_init(n, r0):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r0

    idx = 0
    y = r0
    row = 0
    while idx < n:
        x = r0
        offset = r0 * np.sqrt(3) / 2 if row % 2 == 1 else 0.0
        while x + offset <= 1.0 - r0 + 0.001 and idx < n:
            centers[idx, 0] = np.clip(x + offset, r0, 1.0 - r0)
            centers[idx, 1] = np.clip(y, r0, 1.0 - r0)
            idx += 1
            x += r0 * np.sqrt(3)
        y += r0 * 0.95
        row += 1

    return centers, radii


def rect_init(n, r0):
    """Initialize circles in a rectangular grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r0

    # Determine grid dimensions
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))

    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = (col + 0.5) * (1.0 - 2.0 * r0) / cols + r0
            y = (row + 0.5) * (1.0 - 2.0 * r0) / rows + r0
            centers[idx, 0] = np.clip(x, r0, 1.0 - r0)
            centers[idx, 1] = np.clip(y, r0, 1.0 - r0)
            idx += 1

    return centers, radii


def repair_packing(centers, radii, n):
    """Iteratively repair constraint violations."""
    for _ in range(2000):
        max_violation = 0.0

        # Fix boundary violations by moving centers
        for i in range(n):
            x = centers[i, 0]
            y = centers[i, 1]
            r = radii[i]

            if x < r:
                centers[i, 0] = r
                max_violation = max(max_violation, r - x)
            if x + r > 1.0:
                centers[i, 0] = 1.0 - r
                max_violation = max(max_violation, x + r - 1.0)
            if y < r:
                centers[i, 1] = r
                max_violation = max(max_violation, r - y)
            if y + r > 1.0:
                centers[i, 1] = 1.0 - r
                max_violation = max(max_violation, y + r - 1.0)

        # Fix overlap violations by shrinking radii equally
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                r_sum = radii[i] + radii[j]

                if dist < r_sum - 1e-14:
                    excess = (r_sum - dist) / 2.0
                    radii[i] -= excess
                    radii[j] -= excess
                    max_violation = max(max_violation, excess)

        # Ensure non-negative radii
        radii = np.maximum(radii, 0.0)

        if max_violation < 1e-15:
            break

    return centers, radii


def is_valid(centers, radii, n):
    """Check if packing satisfies all constraints."""
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]

        if r < 0:
            return False
        if x < r - 1e-12 or x + r > 1.0 + 1e-12:
            return False
        if y < r - 1e-12 or y + r > 1.0 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True
