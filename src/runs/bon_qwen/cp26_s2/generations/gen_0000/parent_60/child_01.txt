import numpy as np
from scipy.optimize import minimize

def get_initial_guess():
    """Generates a 5x5 grid plus a central circle as a strong feasible starting point."""
    centers = []
    radii = []
    # 5x5 grid
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
            radii.append(0.095)
    # Center circle
    centers.append([0.5, 0.5])
    radii.append(0.15)
    return np.array(centers), np.array(radii)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[52:])

def boundary_constraints(vars):
    """Ensures circles remain inside [0,1]x[0,1]."""
    x = vars[:26]
    y = vars[26:52]
    r = vars[52:]
    return np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])

def overlap_constraints(vars):
    """Ensures circles do not overlap."""
    x = vars[:26]
    y = vars[26:52]
    r = vars[52:]
    cons = np.zeros(26 * 25 // 2)
    idx = 0
    for i in range(26):
        for j in range(i + 1, 26):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dr = r[i] + r[j]
            cons[idx] = dx**2 + dy**2 - dr**2
            idx += 1
    return cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]
    
    # Bounds for x, y in [0, 1] and r in [0, 0.5]
    bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)] * 26
    
    np.random.seed(42)
    
    # Run optimization with multiple restarts
    for _ in range(15):
        c0, r0 = get_initial_guess()
        
        # Add small random perturbations to break symmetry
        c0 += np.random.normal(0, 0.002, c0.shape)
        r0 += np.random.normal(0, 0.001, r0.shape)
        r0 = np.maximum(r0, 0.01)
        
        # Ensure initial guess respects boundaries
        c0[:, 0] = np.clip(c0[:, 0], r0, 1.0 - r0)
        c0[:, 1] = np.clip(c0[:, 1], r0, 1.0 - r0)
        
        x0 = np.concatenate([c0.flatten(), r0])
        
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 1500, 'ftol': 1e-10, 'disp': False}
        )
        
        curr_centers = res.x[:52].reshape(26, 2)
        curr_radii = res.x[52:]
        
        # Strictly project boundaries to guarantee feasibility
        curr_centers[:, 0] = np.clip(curr_centers[:, 0], curr_radii, 1.0 - curr_radii)
        curr_centers[:, 1] = np.clip(curr_centers[:, 1], curr_radii, 1.0 - curr_radii)
        
        # Validate overlaps with a safe tolerance
        valid = True
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.hypot(curr_centers[i, 0] - curr_centers[j, 0], 
                                curr_centers[i, 1] - curr_centers[j, 1])
                if dist < curr_radii[i] + curr_radii[j] - 1e-9:
                    valid = False
                    break
            if not valid:
                break
                
        if valid and np.sum(curr_radii) > best_sum:
            best_sum = np.sum(curr_radii)
            best_centers = curr_centers.copy()
            best_radii = curr_radii.copy()
            
    # Fallback to initial guess if optimization fails
    if best_centers is None:
        return get_initial_guess()
        
    return best_centers, best_radii, best_sum
