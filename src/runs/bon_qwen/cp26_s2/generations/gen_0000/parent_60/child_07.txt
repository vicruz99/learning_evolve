import numpy as np
from scipy.optimize import minimize

def calculate_penalty(centers, radii, mu):
    """
    Calculates the total penalty for boundary violations and overlaps.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        
        # Boundary penalties: Circle must be strictly inside [0, 1]x[0, 1]
        # Constraint: r <= x <= 1-r  =>  x >= r  AND  x + r <= 1
        # If x < r, violation is r - x
        if xi < ri:
            penalty += mu * (ri - xi)**2
        if xi + ri > 1:
            penalty += mu * (xi + ri - 1)**2
        
        # Constraint: r <= y <= 1-r
        if yi < ri:
            penalty += mu * (ri - yi)**2
        if yi + ri > 1:
            penalty += mu * (yi + ri - 1)**2
            
        # Overlap penalties: distance between centers >= r_i + r_j
        for j in range(i + 1, n):
            dx = xi - centers[j][0]
            dy = yi - centers[j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            min_dist = ri + radii[j]
            
            if dist < min_dist:
                penalty += mu * (min_dist - dist)**2
                
    return penalty

def objective_function(vars, mu):
    """
    Objective function to minimize.
    Minimizes: -sum(radii) + penalty
    """
    n = 26
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]

    obj = -np.sum(radii)
    
    # Add penalty for constraints
    obj += calculate_penalty(centers, radii, mu)
    
    return obj

def generate_hexagonal_init(n, size=1.0):

    """
    centers = []
    # Hexagonal spacing factors
    # Horizontal spacing 2*r, Vertical spacing sqrt(3)*r

    s = 0.15 # Initial spacing
    
    y = s
    while len(centers) < n:
        x = s
        is_odd_row = (int((y - s) / (s * np.sqrt(3))) % 2 == 1)
        if is_odd_row:
            x = s + s/2 # Shift for odd rows
        
        while x < size - s and len(centers) < n:
            centers.append([x, y])
            x += 2 * s
        
        y += s * np.sqrt(3)
        
    centers = np.array(centers[:n])

    if centers.size > 0:
        # Shift to center in [0,1]
        min_x, min_y = np.min(centers[:, 0]), np.min(centers[:, 1])
        max_x, max_y = np.max(centers[:, 0]), np.max(centers[:, 1])

        height = max_y - min_y

        scale_y = (0.8) / height if height > 0 else 1.0
        scale = min(scale_x, scale_y)
        
        centers = (centers - [min_x, min_y]) * scale
        # Center the whole block
        current_max = np.max(centers, axis=0)
        shift = (1.0 - current_max) / 2.0
        centers += shift
        
    return centers
