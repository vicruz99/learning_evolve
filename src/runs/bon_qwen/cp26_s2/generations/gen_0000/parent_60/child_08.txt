import numpy as np
from scipy.optimize import minimize

def objective(X, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(X[2*n:])

def constraints(X, n):
    """
    Constraint function for circle packing.
    Returns a 1D array of constraint values, all must be >= 0.
    """
    x = X[:n]
    y = X[n:2*n]
    r = X[2*n:3*n]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, same for y
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise non-overlap constraints: ||p_i - p_j||^2 >= (r_i + r_j)^2
    X_mat = np.vstack((x, y)).T
    diff = X_mat[:, None, :] - X_mat[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    r_sum = r[:, None] + r[None, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.append(dist_sq[mask] - r_sum[mask]**2)
    
    return np.concatenate(c)

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initial Hexagonal Layout
    pts = []
    s = 0.22
    dy = s * np.sqrt(3) / 2
    for row in range(7):
        y = row * dy
        if y > 1.0: 
            break
        offset = (row % 2) * (s / 2)
        for col in range(6):
            x = col * s + offset
            if 0 <= x <= 1:
                pts.append((x, y))
        if len(pts) >= n:
            break
            
    xs = np.array([p[0] for p in pts[:n]])
    ys = np.array([p[1] for p in pts[:n]])
    
    # 2. Break Symmetry & Ensure Strict Feasibility
    xs += np.random.uniform(-0.01, 0.01, n)
    ys += np.random.uniform(-0.01, 0.01, n)
    xs = np.clip(xs, 0.02, 0.98)
    ys = np.clip(ys, 0.02, 0.98)
    
    rs = np.full(n, 0.05)  # Conservative initial radius
    X0 = np.concatenate([xs, ys, rs])
    
    # 3. Setup Optimization
    bounds = [(0.0, 1.0)]*n + [(0.0, 1.0)]*n + [(0.0, 0.5)]*n
    cons = {'type': 'ineq', 'fun': constraints, 'args': (n,)}
    
    res = minimize(objective, X0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
                   
    x_opt = res.x[:n]
    y_opt = res.x[n:2*n]
    r_opt = res.x[2*n:3*n]
    
    # 4. Post-processing & Safety Check
    # Compute minimum violation to handle potential numerical drift
    min_viol = 1.0
    
    # Check boundaries
    min_viol = min(min_viol, np.min(x_opt - r_opt), np.min(1.0 - x_opt - r_opt),
                   np.min(y_opt - r_opt), np.min(1.0 - y_opt - r_opt))
                   
    # Check pairwise overlaps
    X_mat = np.vstack((x_opt, y_opt)).T
    dist = np.sqrt(np.sum((X_mat[:, None] - X_mat[None, :])**2, axis=2))
    r_sum = r_opt[:, None] + r_opt[None, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    gaps = dist[mask] - r_sum[mask]
    min_viol = min(min_viol, np.min(gaps))
    
    # If any constraint is violated beyond tolerance, scale radii down slightly
    if min_viol < -1e-9:
        scale_factor = 1.0 + min_viol / (2.0 * np.max(r_opt) + 1e-9)
        r_opt *= max(scale_factor, 0.9)
        
    # Ensure non-negative radii
    r_opt = np.maximum(r_opt, 1e-9)
    
    centers = np.column_stack((x_opt, y_opt))
    sum_radii = float(np.sum(r_opt))
    
    return centers, r_opt, sum_radii
