import math

def force_based_optimization(centers, radii, n, max_iterations=80000):
    """Force-based optimization to increase radii and resolve overlaps."""
    dt = 0.015
    stiffness = 150.0
    damping = 0.85
    temperature = 0.005
    
    velocities = np.zeros((n, 2))
    
    for iteration in range(max_iterations):
        forces = np.zeros((n, 2))
        
        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0:
                forces[i, 0] += stiffness * (r - x)
            elif x - r < 0.01:
                forces[i, 0] += stiffness * 0.01 * (0.01 - (x - r)) * 5
            
            if x + r > 1:
                forces[i, 0] -= stiffness * (x + r - 1)
            elif x + r > 0.99:
                forces[i, 0] -= stiffness * 0.01 * (0.99 - (1 - (x + r))) * 5
            
            if y - r < 0:
                forces[i, 1] += stiffness * (r - y)
            elif y - r < 0.01:
                forces[i, 1] += stiffness * 0.01 * (0.01 - (y - r)) * 5
            
            if y + r > 1:
                forces[i, 1] -= stiffness * (y + r - 1)
            elif y + r > 0.99:
                forces[i, 1] -= stiffness * 0.01 * (0.99 - (1 - (y + r))) * 5
        
        # Pairwise repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = math.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-12:
                    overlap = min_dist - dist
                    force_mag = stiffness * overlap
                    nx = dx / dist
                    ny = dy / dist
                    forces[i, 0] += force_mag * nx
                    forces[i, 1] += force_mag * ny
                    forces[j, 0] -= force_mag * nx
                    forces[j, 1] -= force_mag * ny
        
        # Update velocities and positions
        velocities = damping * velocities + dt * forces
        centers += dt * velocities
        
        # Clamp positions
        centers[:, 0] = np.clip(centers[:, 0], 1e-8, 1 - 1e-8)
        centers[:, 1] = np.clip(centers[:, 1], 1e-8, 1 - 1e-8)
        
        # Gradually increase radii
        if iteration % 50 == 0:
            growth_factor = 1.00005
            radii *= growth_factor
            
            # Constrain radii to valid values
            for i in range(n):
                x, y = centers[i]
                max_r = min(x, 1 - x, y, 1 - y)
                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = math.sqrt(dx * dx + dy * dy)
                        available = dist - radii[j]
                        if available < max_r:
                            max_r = available
                radii[i] = max(1e-10, min(radii[i], max_r - 1e-12))
        
        # Add random perturbation periodically

            perturbation = np.random.randn(n, 2) * temperature * 0.1
            centers += perturbation
            centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
            centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)
    
    return centers, radii


def objective_function(params, n):
    """Objective: minimize negative sum of radii."""
    radii = params[2 * n:]
    return -np.sum(radii)


def boundary_inequality(params, n):
    """Boundary constraints: circles inside [0,1]^2."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    result = np.empty(4 * n)
    for i in range(n):
        result[4 * i] = centers[i, 0] - radii[i]
        result[4 * i + 1] = 1.0 - centers[i, 0] - radii[i]
        result[4 * i + 2] = centers[i, 1] - radii[i]
        result[4 * i + 3] = 1.0 - centers[i, 1] - radii[i]
    return result


def non_overlap_inequality(params, n):
    """Non-overlap constraints: distance >= r_i + r_j."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    num_pairs = n * (n - 1) // 2
    result = np.empty(num_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            result[idx] = dist - radii[i] - radii[j]
            idx += 1
    return result


def radius_inequality(params, n):
    """Non-negative radii."""
    return params[2 * n:]


def scipy_refinement(centers, radii, n, max_iter=20000):
    """Refine using scipy SLSQP optimizer."""
    params = np.concatenate([centers.flatten(), radii])
    
    constraints = [
        {'type': 'ineq', 'fun': lambda p: boundary_inequality(p, n)},
        {'type': 'ineq', 'fun': lambda p: non_overlap_inequality(p, n)},
        {'type': 'ineq', 'fun': lambda p: radius_inequality(p, n)},
    ]
    
    bounds_list = []
    for i in range(2 * n):
        bounds_list.append((0.0, 1.0))
    for i in range(n):
        bounds_list.append((1e-10, 0.5))
    
    result = minimize(
        objective_function,
        params,
        args=(n,),
        method='SLSQP',
        bounds=bounds_list,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'disp': False}
    )
    
    if result.success or result.fun < -np.sum(radii) - 1e-6:
        return result.x[:2 * n].reshape(n, 2), result.x[2 * n:]
    return centers, radii


def validate_and_clean(centers, radii, n):
    """Ensure all constraints are satisfied."""
    centers = centers.copy()
    radii = radii.copy()
    
    for i in range(n):

        r = radii[i]
        
        # Ensure inside square
        max_r = min(x, 1 - x, y, 1 - y)
        if radii[i] > max_r:
            radii[i] = max(1e-10, max_r - 1e-12)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            sum_r = radii[i] + radii[j]
            if dist < sum_r - 1e-12:
                # Reduce the smaller radius
                overlap = sum_r - dist
                min_r = min(radii[i], radii[j])
                radii[i] -= overlap * radii[i] / sum_r
                radii[j] -= overlap * radii[j] / sum_r
                if radii[i] < 1e-10:
                    radii[i] = 1e-10
                if radii[j] < 1e-10:
                    radii[j] = 1e-10
    
    return centers, radii
