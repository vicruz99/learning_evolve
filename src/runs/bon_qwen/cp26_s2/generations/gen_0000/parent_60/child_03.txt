import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    def objective(vars):
        # vars: [x1, y1, r1, ..., x26, y26, r26]
        # We maximize sum of radii, so we minimize negative sum
        radii = vars[2::3]
        return -np.sum(radii)

    def constraints_factory(vars, n):
        cons = []
        # Boundary and non-negativity constraints
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i + 1]
            r = vars[3*i + 2]
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx + 2]}) # r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx] - v[3*idx + 2]}) # x >= r
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1 - v[3*idx] - v[3*idx + 2]}) # x + r <= 1
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx + 1] - v[3*idx + 2]}) # y >= r
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1 - v[3*idx + 1] - v[3*idx + 2]}) # y + r <= 1
        
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: 
                        np.sqrt((v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2) - (v[3*i+2] + v[3*j+2])
                })
        return cons

    cons = constraints_factory(np.zeros(3*n), n)
    bounds = [(0, 1)] * (2*n) + [(0, 1)] * n

    # Initialization 1: Hexagonal packing
    x1, y1, r1 = np.zeros(n), np.zeros(n), np.zeros(n)
    row_counts = [6, 5, 6, 5, 4]
    r_init = 0.09
    idx = 0
    for row_idx, count in enumerate(row_counts):
        y = 0.1 + row_idx * 0.17
        x_start = (1 - (count - 1) * 2 * r_init) / 2
        if row_idx % 2 == 1: x_start += r_init
        for k in range(count):
            if idx < n:
                x1[idx] = x_start + k * 2 * r_init
                y1[idx] = y
                r1[idx] = r_init
                idx += 1

    # Initialization 2: Perturbed grid
    x2 = np.random.uniform(0.05, 0.95, n)
    y2 = np.random.uniform(0.05, 0.95, n)
    r2 = np.full(n, 0.05)

    def optimize(x0, y0, r0):
        x0_flat = np.zeros(3*n)
        x0_flat[0::3] = x0
        x0_flat[1::3] = y0
        x0_flat[2::3] = r0
        
        res = minimize(objective, x0_flat, method='COBYLA', args=(n,), bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'rhobeg': 0.01})
        return res.x, -res.fun

    res1, sum1 = optimize(x1, y1, r1)
    res2, sum2 = optimize(x2, y2, r2)

    if sum1 > sum2:
        final_vars = res1
    else:
        final_vars = res2

    centers = np.column_stack((final_vars[0::3], final_vars[1::3]))
    radii = final_vars[2::3]
    
    # Final cleanup to ensure non-negative and valid
    radii = np.maximum(radii, 0.0)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)

    return centers, radii, np.sum(radii)
