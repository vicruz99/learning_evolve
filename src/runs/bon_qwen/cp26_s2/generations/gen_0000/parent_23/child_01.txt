import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n, P):
    """Computes objective (-sum radii) plus quadratic penalty for constraint violations."""
    centers = vars[:n*2].reshape(n, 2)
    radii = vars[n*2:]
    loss = -np.sum(radii)
    
    # Boundary constraints: circles must stay within [0,1]x[0,1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r > x:
            loss += P * (r - x)**2
        if r > 1 - x:
            loss += P * (r - (1 - x))**2
        if r > y:
            loss += P * (r - y)**2
        if r > 1 - y:
            loss += P * (r - (1 - y))**2
            
    # Overlap constraints: distance between centers >= sum of radii
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            rj = radii[j]
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx*dx + dy*dy)
            sum_r = ri + rj
            if sum_r > dist:
                loss += P * (sum_r - dist)**2
                
    return loss

def init_centers_radii(n):

    centers = []
    radii = []
    np.random.seed(42)

        best_pos = None
        best_r = 0.0

            x, y = np.random.rand(2)

                r = min(r, dist - radii[k])
            if r > best_r:
                best_r = r
                best_pos = (x, y)
        centers.append(best_pos)
        radii.append(max(0.01, best_r))
    return np.array(centers), np.array(radii)
