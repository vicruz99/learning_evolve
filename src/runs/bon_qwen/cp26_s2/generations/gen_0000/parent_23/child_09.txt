import numpy as np
from scipy.optimize import minimize

def run_packing():

    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    """
    n = 26
    
    # --- 1. Generate Initial Hexagonal Packing ---
    centers = []
    # Hexagonal rows: 5, 6, 5, 6, 4 = 26 circles
    # Row 0: 5 circles
    for i in range(5):
        centers.append([0.15 + i * 0.2, 0.15])
    # Row 1: 6 circles (shifted)
    for i in range(6):
        centers.append([0.25 + i * 0.2, 0.30])
    # Row 2: 5 circles
    for i in range(5):
        centers.append([0.15 + i * 0.2, 0.45])
    # Row 3: 6 circles (shifted)
    for i in range(6):
        centers.append([0.25 + i * 0.2, 0.60])
    # Row 4: 4 circles
    for i in range(4):
        centers.append([0.15 + i * 0.2, 0.75])
        
    centers = np.array(centers)
    # Initial radii (safe start)
    radii = np.full(n, 0.05)
    
    # --- 2. Define Objective and Constraints ---
    def objective(x_flat):
        # x_flat[0:52] are centers, x_flat[52:78] are radii
        return -np.sum(x_flat[52:])

    def constraints_func(x_flat):
        centers_opt = x_flat[:n*2].reshape(n, 2)
        radii_opt = x_flat[52:]
        
        # Boundary constraints: x - r >= 0, 1 - (x + r) >= 0, same for y
        x, y = centers_opt[:, 0], centers_opt[:, 1]
        r = radii_opt
        c1 = x - r
        c2 = 1 - (x + r)
        c3 = y - r
        c4 = 1 - (y + r)
        
        # Non-overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        # Vectorized calculation for all pairs (i, j) where i < j
        diff = centers_opt[:, np.newaxis, :] - centers_opt[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        # Extract upper triangle
        indices = np.triu_indices(n, k=1)
        c_overlap = dist_sq[indices] - r_sum[indices]**2
        
        return np.concatenate([c1, c2, c3, c4, c_overlap])

    # --- 3. Setup and Run Optimization ---
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for variables: centers in [0,1], radii in [0, 0.5]
    bnds = [(0.0, 1.0)] * (n*2) + [(0.0, 0.5)] * n
    
    # Non-linear constraints dict
    cons = ({'type': 'ineq', 'fun': constraints_func})
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons, 
                   options={'ftol': 1e-8, 'maxiter': 500, 'disp': False})
    
    # --- 4. Post-processing and Validation ---
    best_centers = res.x[:n*2].reshape(n, 2)
    best_radii = res.x[52:]
    
    # Clip tiny negative radii due to numerical noise
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, np.sum(best_radii)
