from scipy.optimize import minimize

def compute_all_constraints(x):

    constraints = []

    for i in range(n):

        r = x[3 * i + 2]
        constraints.append(cx - r)          # left boundary: cx >= r
        constraints.append(1.0 - cx - r)    # right boundary: cx + r <= 1
        constraints.append(cy - r)          # bottom boundary: cy >= r
        constraints.append(1.0 - cy - r)    # top boundary: cy + r <= 1
        
    # Distance constraints: 1 per pair (non-overlapping)
    for i in range(n):
        for j in range(i + 1, n):
            cx1 = x[3 * i]
            cy1 = x[3 * i + 1]
            r1 = x[3 * i + 2]

            cy2 = x[3 * j + 1]
            r2 = x[3 * j + 2]

            rad_sum_sq = (r1 + r2) ** 2
            constraints.append(dist_sq - rad_sum_sq) # dist >= r1 + r2
            
    return np.array(constraints)
