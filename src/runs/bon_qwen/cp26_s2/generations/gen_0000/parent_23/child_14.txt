Key insights:
1. **Initialization**: A hexagonal lattice packing is the densest arrangement in the plane. Starting with a hexagonal grid provides a much better initial configuration than a random or square grid, placing circles close to optimal spacing.
2. **Penalty Method**: Directly handling ~425 inequality constraints with standard solvers can be slow or prone to stalling. 3. **Annealing Schedule**: We iteratively increase the penalty weight `mu`. Starting with a moderate `mu` allows circles to move freely into a good configuration, while gradually increasing it forces the solution to strictly satisfy constraints.
4. **Multi-start Optimization**: The landscape has many local minima. Running the optimization from 5 different randomized seeds (perturbing the hex grid) ensures we escape poor local configurations and find a globally competitive packing.

import numpy as np
from scipy.optimize import minimize

def packing_objective(v, n, mu):
    """
    Computes the penalized objective for circle packing.
    Minimizes negative sum of radii + penalty for constraint violations.
    """
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    obj = -np.sum(r)
    
    # Penalty terms for constraints
    pen = 0.0
    
    # Boundary constraints: circles must be inside [0,1]x[0,1]
    # Left: x >= r  => r - x <= 0
    pen += np.sum(np.maximum(0.0, r - c[:,0])**2)
    # Right: x + r <= 1 => x + r - 1 <= 0
    pen += np.sum(np.maximum(0.0, c[:,0] + r - 1.0)**2)
    # Bottom: y >= r => r - y <= 0
    pen += np.sum(np.maximum(0.0, r - c[:,1])**2)
    # Top: y + r <= 1 => y + r - 1 <= 0
    pen += np.sum(np.maximum(0.0, c[:,1] + r - 1.0)**2)
    
    # Non-overlap constraints: distance between centers >= sum of radii
    # Compute pairwise distances efficiently using broadcasting
    diff = c[:, None, :] - c[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # Upper triangular mask for unique pairs (i, j) with i < j
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = r[:, None] + r[None, :] - dist
    pen += np.sum(np.maximum(0.0, overlaps[mask])**2)
    
    return obj + mu * pen

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Use multiple random restarts to escape local minima
    for seed in range(5):
        np.random.seed(seed + 100)
        
        # Initialize centers in a hexagonal pattern for good starting density
        centers = np.zeros((n, 2))
        r_init = 0.095
        idx = 0
        
        for row in range(8):
            y = r_init + row * r_init * np.sqrt(3)
            if y + r_init > 1.0:
                break
            x_start = r_init if row % 2 == 0 else r_init * 1.5
            x = x_start
            while x + r_init <= 1.0 and idx < n:
                centers[idx] = [x, y]
                x += 2.0 * r_init
                idx += 1
            if idx == n:
                break
                
        # Fill remaining spots randomly if hex grid didn't cover all
        if idx < n:
            centers[idx:] = np.random.rand(n - idx, 2) * 0.8 + 0.1
            
        # Add slight noise to break exact symmetry and improve gradient flow
        centers += np.random.normal(0, 0.002, centers.shape)
        centers = np.clip(centers, 0.01, 0.99)
        
        # Initialize radii
        radii = np.full(n, 0.05)
        
        # Flatten to 1D array for optimizer
        vars0 = np.hstack([centers.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        # Optimization loop with increasing penalty weight (annealing)
        v_curr = vars0
        mu = 50.0
        for step in range(10):
            res = minimize(packing_objective, v_curr, args=(n, mu), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-12})
            v_curr = res.x
            mu *= 1.5
            
        c_opt = v_curr[:2*n].reshape(n, 2)
        r_opt = v_curr[2*n:]
        cur_sum = np.sum(r_opt)
        
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    # Ensure strict non-negativity and boundary adherence
    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    return best_centers, best_radii, best_sum
