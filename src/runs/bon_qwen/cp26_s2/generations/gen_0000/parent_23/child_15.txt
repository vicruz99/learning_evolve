import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint, Bounds

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Returns centers (26, 2), radii (26,), and sum of radii.
    """
    n = 26
    
    # Variable ordering in the optimization vector:
    # [x_1, ..., x_n, y_1, ..., y_n, r_1, ..., r_n]
    # Total 3 * n variables
    
    # 1. Define Bounds
    # x, y must be in [0, 1]
    # r must be in [0, 0.5] (since max diameter is 1)
    lb = np.zeros(3 * n)
    ub = np.ones(3 * n)
    lb[2 * n:] = 0.0
    ub[2 * n:] = 0.5
    
    bounds = Bounds(lb, ub)
    
    # 2. Define Linear Constraints (Boundaries)
    # Constraints:
    # x_i >= r_i  =>  -x_i + r_i <= 0
    # x_i <= 1 - r_i => x_i + r_i <= 1
    # y_i >= r_i  =>  -y_i + r_i <= 0
    # y_i <= 1 - r_i => y_i + r_i <= 1
    
    # Matrix A for A @ vars <= b
    # We need 4 * n constraints
    A_lin = np.zeros((4 * n, 3 * n))
    b_lin = np.zeros(4 * n)
    
    for i in range(n):
        row_base = 4 * i
        
        # -x_i + r_i <= 0
        A_lin[row_base, i] = -1.0
        A_lin[row_base, 2 * n + i] = 1.0
        b_lin[row_base] = 0.0
        
        # x_i + r_i <= 1
        A_lin[row_base + 1, i] = 1.0
        A_lin[row_base + 1, 2 * n + i] = 1.0
        b_lin[row_base + 1] = 1.0
        
        # -y_i + r_i <= 0
        A_lin[row_base + 2, n + i] = -1.0
        A_lin[row_base + 2, 2 * n + i] = 1.0
        b_lin[row_base + 2] = 0.0
        
        # y_i + r_i <= 1
        A_lin[row_base + 3, n + i] = 1.0
        A_lin[row_base + 3, 2 * n + i] = 1.0
        b_lin[row_base + 3] = 1.0
        
    lin_con = LinearConstraint(A_lin, -np.inf, b_lin)
    
    # 3. Define Non-Linear Constraints (Non-overlap)
    # For each pair (i, j) with i < j:
    # dist(i, j)^2 >= (r_i + r_j)^2
    # => (r_i + r_j)^2 - ( (x_i - x_j)^2 + (y_i - y_j)^2 ) <= 0
    
    # Precompute indices for all pairs
    i_idx, j_idx = np.triu_indices(n, k=1)
    num_pairs = len(i_idx)
    
    def nl_constraints(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        
        dx = x[i_idx] - x[j_idx]
        dy = y[i_idx] - y[j_idx]
        dist_sq = dx**2 + dy**2
        
        sum_r = r[i_idx] + r[j_idx]
        sum_r_sq = sum_r**2
        
        # We want sum_r_sq - dist_sq <= 0
        return sum_r_sq - dist_sq

    def nl_jac(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        
        # Jacobian matrix size: (num_pairs, 3*n)
        jac = np.zeros((num_pairs, 3 * n))
        
        dx = x[i_idx] - x[j_idx]
        dy = y[i_idx] - y[j_idx]
        sum_r = r[i_idx] + r[j_idx]
        
        # Analytical gradients for constraint c_k = (r_i+r_j)^2 - (dx^2 + dy^2)
        # dc/dx_i = -2*dx
        # dc/dx_j = 2*dx
        # dc/dy_i = -2*dy
        # dc/dy_j = 2*dy
        # dc/dr_i = 2*sum_r
        # dc/dr_j = 2*sum_r
        
        # Vectorized assignment
        # Rows correspond to pairs, Columns to variables
        
        # x variables (cols 0 to n-1)
        jac[np.arange(num_pairs), i_idx] -= 2.0 * dx
        jac[np.arange(num_pairs), j_idx] += 2.0 * dx
        
        # y variables (cols n to 2n-1)
        jac[np.arange(num_pairs), n + i_idx] -= 2.0 * dy
        jac[np.arange(num_pairs), n + j_idx] += 2.0 * dy
        
        # r variables (cols 2n to 3n-1)
        jac[np.arange(num_pairs), 2*n + i_idx] += 2.0 * sum_r
        jac[np.arange(num_pairs), 2*n + j_idx] += 2.0 * sum_r
        
        return jac

    nl_con = NonlinearConstraint(nl_constraints, -np.inf, 0.0, jac=nl_jac)
    
    # 4. Objective Function
    # Maximize sum(r) <=> Minimize -sum(r)
    def objective(vars):
        return -np.sum(vars[2*n:])
    
    def obj_grad(vars):
        grad = np.zeros(3 * n)
        grad[2*n:] = -1.0
        return grad

    # 5. Initial Guess Generation (Hexagonal Packing)
    # Create a hexagonal grid of points to serve as centers
    def generate_hex_init():
        points = []
        # Estimate a starting radius small enough to fit many, but large enough to be useful
        # 26 circles, area ~ 1. r ~ 0.1.
        # Let's try to pack with spacing approx 0.2
        r_start = 0.09
        y = r_start
        row = 0
        while y + r_start <= 1.0:
            if row % 2 == 0:
                x_start = r_start
            else:
                x_start = r_start + r_start # Shift by r for hex packing? 
                # Standard hex: row 0 at x=0, row 1 at x=r (if spacing 2r). 
                # But boundary is r. So row 0 first pt at r.
                # Row 1 first pt at r + r = 2r?
                # Distance between (r, y0) and (2r, y1) is sqrt(r^2 + (sqrt(3)r)^2) = 2r. Correct.
                x_start = 2 * r_start
            
            x = x_start
            while x + r_start <= 1.0:
                points.append([x, y])
                x += 2 * r_start
            y += np.sqrt(3) * r_start
            row += 1
            if len(points) >= 26:
                break
        
        # If we generated fewer than 26 (unlikely with 0.09), or more, slice
        pts = np.array(points[:26])
        
        # Fill if necessary (fallback to random if grid fails, though unlikely)
        if len(pts) < 26:
            extra = 26 - len(pts)
            for _ in range(extra):
                pts = np.vstack([pts, [0.5, 0.5]]) # Fallback center
        
        vars = np.zeros(3 * 26)
        vars[:26] = pts[:, 0]
        vars[26:52] = pts[:, 1]
        vars[52:] = r_start
        return vars

    init_guess = generate_hex_init()
    
    # 6. Optimization Loop with Restart
    best_result = None
    best_sum_r = -np.inf
    
    constraints = [lin_con, nl_con]
    
    # Run multiple times to escape local minima
    for restart in range(5):
        # Perturb initial guess
        current_guess = init_guess.copy()
        # Add small noise
        noise = np.random.normal(0, 0.005, size=current_guess.shape)
        # Apply noise to x, y. Keep r fixed for first few iterations? Or noise all.
        # Let's noise positions mostly.
        current_guess[:n] += noise[:n] * 0.1
        current_guess[n:2*n] += noise[n:2*n] * 0.1
        # Don't noise r much, keep it valid.
        
        # Project to bounds roughly to ensure feasibility for solver start
        current_guess[:n] = np.clip(current_guess[:n], 0.01, 0.99)
        current_guess[n:2*n] = np.clip(current_guess[n:2*n], 0.01, 0.99)
        # Ensure x >= r, etc. roughly
        current_r = current_guess[2*n:]
        current_guess[:n] = np.maximum(current_guess[:n], current_r + 1e-6)
        current_guess[n:2*n] = np.maximum(current_guess[n:2*n], current_r + 1e-6)
        current_guess[:n] = np.minimum(current_guess[:n], 1.0 - current_r - 1e-6)
        current_guess[n:2*n] = np.minimum(current_guess[n:2*n], 1.0 - current_r - 1e-6)

        try:
            res = minimize(
                objective, 
                current_guess, 
                method='trust-constr',
                jac=obj_grad,
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'verbose': 0}
            )
            
            if res.success or res.fun < best_sum_r: # fun is -sum(r), so smaller is better
                # Check constraints validity roughly
                # The solver should respect them, but let's check
                current_vars = res.x
                centers = np.column_stack((current_vars[:n], current_vars[n:2*n]))
                radii = current_vars[2*n:]
                
                # Quick validity check (loose)
                valid = True
                # Boundary
                if np.any(radii < 0): valid = False
                if np.any(centers[:, 0] - radii < -1e-5) or np.any(centers[:, 0] + radii > 1 + 1e-5): valid = False
                if np.any(centers[:, 1] - radii < -1e-5) or np.any(centers[:, 1] + radii > 1 + 1e-5): valid = False
                # Overlap (sampling or full check)
                # Full check is O(N^2) but N=26 is small
                dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
                sum_r_mat = radii[:, np.newaxis] + radii[np.newaxis, :]
                # Upper triangle
                mask = np.triu(np.ones((n, n), dtype=bool), k=1)
                if np.any(dists[mask] < sum_r_mat[mask] - 1e-5):
                    valid = False
                
                if valid:
                    sum_r = np.sum(radii)
                    if sum_r > best_sum_r:
                        best_sum_r = sum_r
                        best_result = (centers, radii, sum_r)
                        # print(f"Restart {restart}: New best sum_r = {sum_r:.5f}")
                else:
                    # Solver might return infeasible point if not fully converged?
                    # trust-constr usually respects constraints.
                    pass
        except Exception as e:
            # print(f"Restart {restart} failed: {e}")
            pass

    if best_result is None:
        # Fallback to initial guess if optimization failed completely
        centers = np.column_stack((init_guess[:n], init_guess[n:2*n]))
        radii = init_guess[2*n:]
        best_result = (centers, radii, np.sum(radii))

    centers, radii, sum_r = best_result
    return centers, radii, sum_r

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    print(f"Sum of radii: {sum_r}")
