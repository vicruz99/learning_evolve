def soft_min(values, alpha=100.0):
    """
    Approximates the minimum of a list of values using the log-sum-exp trick.
    This provides a smooth, differentiable approximation to the min function.
    """
    # Subtract max for numerical stability

    return max_val - (1.0 / alpha) * np.log(sum_exp)

def get_radii(centers, alpha=100.0, smooth=True):
    """
    Computes the radius for each circle given centers.

        x, y = centers[i]

        d_bound = np.array([x, 1 - x, y, 1 - y])
        
        # Distances to other circles (half distance)
        dists_to_others = np.array([
            0.5 * np.linalg.norm(centers[i] - centers[j]) 
            for j in range(n) if i != j
        ])

        constraints = np.concatenate([d_bound, dists_to_others])

            radii[i] = np.min(constraints)
            
    return radii
