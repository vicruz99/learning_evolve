To maximize the sum of radii for 26 circles in a unit square, we formulate this as a continuous non-linear optimization problem. The variables are the $(x, y)$ centers and radii $r_i$ for each circle. The objective is to maximize $\sum r_i$, subject to:
1. Boundary constraints: $r_i \le x_i, y_i \le 1-r_i$
2. Non-overlap constraints: $\|c_i - c_j\|^2 \ge (r_i + r_j)^2$

We use `scipy.optimize.minimize` with the SLSQP algorithm, which handles inequality constraints effectively. To avoid poor local minima, we employ a multi-start strategy:
- Generate multiple feasible initial configurations using a fast overlap-resolution heuristic.
- Run SLSQP from each start point.
- Keep the best valid solution across all runs.

The constraint function is vectorized using NumPy broadcasting for efficiency. A strict post-optimization validation ensures numerical robustness before accepting a solution. Bounds are set to keep coordinates in $[0,1]$ and radii in $[10^{-5}, 0.5]$.

import numpy as np
from scipy.optimize import minimize

N = 26

def _generate_feasible_start(n):
    """Generates a non-overlapping initial configuration for n circles."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.02)
    centers[:, 0] = np.random.uniform(0.1, 0.9, n)
    centers[:, 1] = np.random.uniform(0.1, 0.9, n)
    
    for _ in range(200):
        overlap = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist + 1e-5:
                    overlap = True
                    if dist < 1e-8:
                        dist = 1e-8
                        dx, dy = np.random.randn(2)
                    norm = 1.0 / dist
                    push = (min_dist - dist) * 0.5 + 1e-4
                    centers[i, 0] += dx * norm * push
                    centers[i, 1] += dy * norm * push
                    centers[j, 0] -= dx * norm * push
                    centers[j, 1] -= dy * norm * push
                    
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
        if not overlap:
            break
    return centers, radii

def _objective(v):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(v[2*N:])

def _constraints(v):
    """Returns array of inequality constraints >= 0."""
    n = N
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    # 4N boundary constraints
    cons = np.empty(4*n + n*(n-1)//2)
    cons[:n] = centers[:, 0] - radii
    cons[n:2*n] = centers[:, 1] - radii
    cons[2*n:3*n] = 1.0 - centers[:, 0] - radii
    cons[3*n:4*n] = 1.0 - centers[:, 1] - radii
    
    # N(N-1)/2 pairwise distance constraints (vectorized)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    tri_idx = np.triu_indices(n, k=1)
    cons[4*n:] = dist_sq[tri_idx] - r_sum[tri_idx]**2
    
    return cons

def run_packing():
    np.random.seed(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Bounds: x in [0,1], y in [0,1], r in [1e-5, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)] * N
    
    for seed in range(80):
        np.random.seed(seed)
        c_init, r_init = _generate_feasible_start(N)
        x0 = np.concatenate([c_init.flatten(), r_init])
        
        res = minimize(_objective, x0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': _constraints},
                       options={'maxiter': 300, 'ftol': 1e-12})
                       
        if res.success:
            c_opt = res.x[:2*N].reshape(N, 2)
            r_opt = res.x[2*N:]
            
            # Strict validation to filter out numerical drift
            valid = True
            if np.any(r_opt < 0):
                valid = False
            elif np.any(c_opt[:, 0] - r_opt < -1e-9) or np.any(c_opt[:, 0] + r_opt > 1 + 1e-9):
                valid = False
            elif np.any(c_opt[:, 1] - r_opt < -1e-9) or np.any(c_opt[:, 1] + r_opt > 1 + 1e-9):
                valid = False
                
            if valid:
                diff = c_opt[:, np.newaxis, :] - c_opt[np.newaxis, :, :]
                dist_sq = np.sum(diff**2, axis=2)
                r_sum = r_opt[:, np.newaxis] + r_opt[np.newaxis, :]
                tri_idx = np.triu_indices(N, k=1)
                overlaps = dist_sq[tri_idx] < (r_sum[tri_idx] - 1e-9)**2
                if np.any(overlaps):
                    valid = False
                    
            if valid:
                s = np.sum(r_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
                    
    # Fallback (should not be reached with 80 starts)
    if best_centers is None:
        c_f, r_f = _generate_feasible_start(N)
        best_centers = c_f
        best_radii = r_f
        best_sum = np.sum(r_f)
        
    return best_centers, best_radii, best_sum
