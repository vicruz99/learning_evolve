import numpy as np
from scipy.optimize import minimize

def compute_penalty(vars, alpha, n):
    """
    Computes the penalized objective function.
    vars: flattened array [x0, y0, r0, x1, y1, r1, ...]
    alpha: penalty weight
    n: number of circles
    """
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    
    # Objective: Minimize negative sum of radii
    obj = -np.sum(radii)
    
    penalty = 0.0
    
    # 1. Boundary Constraints
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    xs = centers[:, 0]
    ys = centers[:, 1]
    
    v1 = np.maximum(0.0, radii - xs)
    v2 = np.maximum(0.0, radii - (1.0 - xs))
    v3 = np.maximum(0.0, radii - ys)
    v4 = np.maximum(0.0, radii - (1.0 - ys))
    
    penalty += alpha * np.sum(v1**2 + v2**2 + v3**2 + v4**2)
    
    # 2. Overlap Constraints
    # distance(i, j) >= r_i + r_j
    # Vectorized pairwise distance calculation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    dist = np.sqrt(np.maximum(dist_sq, 0.0))
    
    sum_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount (positive if overlapping)
    overlap = np.maximum(0.0, sum_radii - dist)
    np.fill_diagonal(overlap, 0.0)
    
    penalty += alpha * np.sum(overlap**2)
    
    return obj + penalty

                x_shifted = x + 0.1
                if x_shifted <= 1.0:
                    all_pts.append([x_shifted, y])
            else:
                all_pts.append([x, y])
    
    # Take first 26 points
    if len(all_pts) >= n:
        centers_init = all_pts[:n]
    else:
        # Fallback to random if not enough grid points
        np.random.seed(42)

    centers_init = np.array(centers_init)

    # Flatten variables
    vars_init = np.concatenate([centers_init.flatten(), radii_init])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    alpha = 10.0
    vars_current = vars_init
