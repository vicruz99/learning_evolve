def softplus(x):
    """Smooth approximation of max(0, x)"""
    # log(1 + exp(x)) is smooth max(0, x)
    # To avoid overflow, clip x
    x = np.clip(x, -50, 50)
    return np.log1p(np.exp(x))

def objective_function(params, n):
    """
    Minimizes -sum(radii) + penalty(overlaps) + penalty(boundaries)
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: Maximize sum of radii => Minimize -sum(radii)
    obj = -np.sum(radii)
    
    penalty_weight = 100.0
    penalty_sum = 0.0
    
    # Boundary constraints:
    # x_i - r_i >= 0  => -(x_i - r_i) <= 0
    # 1 - x_i - r_i >= 0 => -(1 - x_i - r_i) <= 0
    # Same for y
    
    # We penalize violations. 
    # Violation v = max(0, r_i - x_i). We want to minimize softplus(v).
    # Actually, if r_i > x_i, we have a violation.
    
    # Left wall: r_i - x_i
    # Right wall: r_i - (1 - x_i) = r_i + x_i - 1
    # Bottom: r_i - y_i
    # Top: r_i + y_i - 1
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Boundary violations
        v_left = r - x
        v_right = r + x - 1.0
        v_bottom = r - y
        v_top = r + y - 1.0
        
        penalty_sum += softplus(penalty_weight * v_left)
        penalty_sum += softplus(penalty_weight * v_right)
        penalty_sum += softplus(penalty_weight * v_bottom)
        penalty_sum += softplus(penalty_weight * v_top)
        
    # Overlap constraints:
    # dist(c_i, c_j) >= r_i + r_j
    # r_i + r_j - dist <= 0
    # Violation: r_i + r_j - dist
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            
            v_overlap = r_sum - dist
            penalty_sum += softplus(penalty_weight * v_overlap)
            
    return obj + penalty_sum
