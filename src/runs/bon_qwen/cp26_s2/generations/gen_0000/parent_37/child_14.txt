import numpy as np
from scipy.optimize import minimize
import warnings

# Suppress scipy warnings for cleaner execution output
warnings.filterwarnings('ignore')

def compute_overlap(centers, r):
    """Calculate the sum of squared overlaps between all circle pairs."""
    # Compute pairwise distances efficiently using broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    # Overlap penalty: positive only when circles intersect (dist < 2r)
    overlap = np.maximum(0, 2 * r - dists)
    return np.sum(overlap**2)

def objective(centers_flat, r, n):
    """Objective function for scipy.optimize.minimize."""
    centers = centers_flat.reshape((n, 2))
    return compute_overlap(centers, r)

def optimize_positions(r, n, seed):
    """
    Optimize circle positions for a fixed radius r to minimize overlaps.
    Uses multiple random restarts to avoid local minima.
    """
    rng = np.random.default_rng(seed)
    best_val = np.inf
    best_sol = None
    # Hard boundary constraints: centers must be within [r, 1-r]
    bounds = [(r, 1 - r)] * (2 * n)
    
    for _ in range(5):
        x0 = rng.uniform(r, 1 - r, size=2 * n)
        res = minimize(objective, x0, args=(r, n), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-8})
        if res.fun < best_val:
            best_val = res.fun
            best_sol = res.x
            
    return best_sol.reshape((n, 2)), best_val

def find_packing():
    """
    Binary search for the maximum equal radius r such that 26 circles
    can be packed in the unit square without overlap.
    """
    n = 26
    low, high = 0.05, 0.15
    best_r = low
    # Fallback initialization
    rng = np.random.default_rng(0)
    best_centers = rng.uniform(low, 1 - low, size=(n, 2))
    seed = 42
    
    for _ in range(25):
        mid = (low + high) / 2
        centers, val = optimize_positions(mid, n, seed)
        seed += 1
        
        # If overlap penalty is sufficiently small, the radius is feasible
        if val < 1e-5:
            best_r = mid
            best_centers = centers
            low = mid
        else:
            high = mid
            
    return best_r, best_centers

def run_packing():
    r, centers = find_packing()
    radii = np.full(26, r)
    return centers, radii, np.sum(radii)
