import numpy as np
from scipy.optimize import minimize
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective(params, n):
    """
    Objective function: negative sum of radii.
    params layout: [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = params[2::3]
    return -np.sum(radii)

def get_constraints(n):
    """
    Returns a list of constraint dictionaries for the optimization problem.
    """
    constraints = []

    # Boundary constraints: 0 <= x - r, x + r <= 1, 0 <= y - r, y + r <= 1
    # Equivalent to: r <= x <= 1-r and r <= y <= 1-r
    # We can enforce x >= r, y >= r, x <= 1-r, y <= 1-r
    # In scipy constraints: fun(x) >= 0
    # x - r >= 0
    # y - r >= 0
    # (1 - r) - x >= 0  => 1 - r - x >= 0
    # (1 - r) - y >= 0  => 1 - r - y >= 0
    
    # We will add these as bounds or constraints. Bounds are better for simple box constraints.
    # However, bounds in scipy are for variables, not expressions.
    # So we use constraints.
    
    for i in range(n):
        idx = 3 * i
        # x_i >= r_i  => x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, idx=idx: p[idx] - p[idx+2]
        })
        # y_i >= r_i  => y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, idx=idx: p[idx+1] - p[idx+2]
        })
        # x_i <= 1 - r_i => 1 - r_i - x_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, idx=idx: 1 - p[idx+2] - p[idx]
        })
        # y_i <= 1 - r_i => 1 - r_i - y_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, idx=idx: 1 - p[idx+2] - p[idx+1]
        })

    # Non-overlap constraints: dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            def make_ineq(idx_i, idx_j):
                def ineq(p):
                    x1, y1, r1 = p[idx_i], p[idx_i+1], p[idx_i+2]
                    x2, y2, r2 = p[idx_j], p[idx_j+1], p[idx_j+2]
                    dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                    sum_r_sq = (r1 + r2)**2
                    return dist_sq - sum_r_sq
                return ineq

            constraints.append({
                'type': 'ineq',
                'fun': make_ineq(idx_i, idx_j)
            })

    return constraints

def generate_hexagonal_init(n, num_rows):
    """
    Generates an initial guess using a hexagonal packing pattern.
    """
    centers = []
    radii = []
    
    # Estimate radius based on area
    # N * pi * r^2 ~ 1 * density (approx 0.9)
    r_est = math.sqrt(0.9 / (n * math.pi))
    
    # Try to fit into rows
    # Approx circles per row
    cols = math.ceil(n / num_rows)
    
    # Width and height available
    # For hex packing, row height is sqrt(3)/2 * diameter
    # Let's just place them and let optimizer fix
    
    y_step = math.sqrt(3) * r_est * 0.95 # slightly compressed
    x_step = 2 * r_est * 0.95
    
    count = 0
    for row in range(num_rows):
        y = r_est + row * y_step
        # Shift odd rows
        shift = (r_est * 0.95) if row % 2 == 1 else 0
        num_in_row = cols
        if count + num_in_row > n:
            num_in_row = n - count
            
        for col in range(num_in_row):
            x = r_est + shift + col * x_step
            if count < n:
                centers.append([x, y])
                radii.append(r_est)
                count += 1
    
    # If we didn't fill n, adjust or just pad? 
    # This function assumes we pick right num_rows/cols.
    # Let's make a simpler grid init if this fails.
    while len(centers) < n:
        centers.append([0.5, 0.5])
        radii.append(0.1)
        
    return np.array(centers[:n]), np.array(radii[:n])

def generate_grid_init(n):
    """
    Generates a grid-based initial guess.
    """
    cols = math.ceil(math.sqrt(n))
    rows = math.ceil(n / cols)
    
    # Space out
    step_x = 1.0 / (cols + 1)
    step_y = 1.0 / (rows + 1)
    
    centers = []
    radii = []
    
    # Estimate radius
    r_est = min(step_x, step_y) / 2 * 0.9
    
    count = 0
    for r in range(rows):
        for c in range(cols):
            if count >= n:
                break
            x = step_x * (c + 1)
            y = step_y * (r + 1)
            centers.append([x, y])
            radii.append(r_est)
            count += 1
            
    return np.array(centers[:n]), np.array(radii[:n])

def run_packing():

    Runs the optimization to pack 26 circles.
    """

    # Initial guesses
    inits = []

    rng = np.random.RandomState(42)

        radii = np.clip(radii, 0.01, 0.2)
        inits.append(np.hstack([centers.ravel(), radii]))

    constraints = get_constraints(n)
    
    best_sol = None
    best_val = -np.inf
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    # Actually r can be up to 0.5, but practically smaller.
    # Let's use [0, 1] for x,y and [0, 0.5] for r.
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r
        
    for init in inits:
        try:
            res = minimize(
                objective,
                init,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if res.success or res.fun < best_val: # Note: fun is negative sum
                current_sum = -res.fun
                if current_sum > best_val:
                    best_val = current_sum
                    best_sol = res.x
        except Exception as e:
            # If optimization fails, skip
            continue

    if best_sol is not None:
        # Extract centers and radii
        centers = best_sol[:2*n].reshape(n, 2)
        radii = best_sol[2*n:]
        
        # Final cleanup to ensure strict constraints if needed

        if not validate_packing(centers, radii):
            # Fallback: scale down radii slightly until valid

            radii = radii * scale
            # If still invalid, return a safe default

                # Generate a safe grid packing
                centers_safe, radii_safe = generate_grid_init(n)
                return centers_safe, radii_safe, np.sum(radii_safe)
        
        return centers, radii, np.sum(radii)
    else:
        # Fallback
        centers_safe, radii_safe = generate_grid_init(n)
        return centers_safe, radii_safe, np.sum(radii_safe)
