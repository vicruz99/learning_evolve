import numpy as np
from scipy.optimize import minimize

def compute_objective(theta, n, alpha):
    """
    Objective function: minimize -sum(radii) + alpha * penalty
    """
    centers = theta[:n*2].reshape(n, 2)
    radii = theta[n*2:]
    
    # Base objective: maximize sum of radii
    obj = -np.sum(radii)
    
    # Boundary penalties: circles must stay within [0,1]x[0,1]
    # x >= r  => r - x <= 0
    # x <= 1-r => x + r - 1 <= 0
    p = np.maximum(0.0, radii - centers[:, 0])**2
    p += np.maximum(0.0, centers[:, 0] + radii - 1.0)**2
    p += np.maximum(0.0, radii - centers[:, 1])**2
    p += np.maximum(0.0, centers[:, 1] + radii - 1.0)**2
    obj += alpha * np.sum(p)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            val = radii[i] + radii[j] - dist
            if val > 0:
                obj += alpha * val**2
                
    return obj

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0) for _ in range(n * 2)] + [(0.01, 0.4) for _ in range(n)]
    
    best_theta = None
    best_score = np.inf
    rng = np.random.default_rng(42)
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Random starts
    for _ in range(50):
        centers = rng.random((n, 2))
        radii = np.full(n, 0.05)
        inits.append(np.concatenate([centers.flatten(), radii]))
        
    # 2. 5x5 Grid + 1 center
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.095)
    idx = 0
    for r in range(5):
        for c in range(5):
            centers[idx, 0] = 0.1 + c * 0.2
            centers[idx, 1] = 0.1 + r * 0.2
            idx += 1
    centers[25, :] = [0.5, 0.5]
    inits.append(np.concatenate([centers.flatten(), radii]))
    
    # 3. Quasi-hexagonal packing
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)
    row_y = 0.1
    idx = 0
    row_idx = 0
    while idx < n:
        row_x = 0.1
        shift = 0.08 if row_idx % 2 == 1 else 0.0
        while idx < n and row_x + 0.08 <= 1.0:
            centers[idx, 0] = row_x + shift
            centers[idx, 1] = row_y
            idx += 1
            row_x += 0.16
        row_y += 0.14
        row_idx += 1
    inits.append(np.concatenate([centers.flatten(), radii]))
    
    # Annealing schedule: progressively enforce constraints
    alphas = [100, 1000, 5000, 10000]
    current_candidates = list(inits)
    
    for alpha in alphas:
        next_candidates = []
        max_iter = 3000 if alpha == 10000 else 2000
        
        for theta0 in current_candidates:
            try:
                res = minimize(
                    compute_objective, theta0, method='L-BFGS-B', 
                    bounds=bounds, args=(n, alpha),
                    options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
                )
                if res.fun < best_score:
                    best_score = res.fun
                    best_theta = res.x.copy()
                next_candidates.append(res.x)
            except Exception:
                continue
        current_candidates = next_candidates
        
    if best_theta is not None:
        centers = best_theta[:n*2].reshape(n, 2)
        radii = best_theta[n*2:]
        
        # Safety corrections to guarantee validity within 1e-12 tolerance
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            # Clamp radius based on distance to boundaries
            r = min(r, x, y, 1.0 - x, 1.0 - y)
            radii[i] = max(1e-6, r)
            
        sum_r = float(np.sum(radii))
        return centers, radii, sum_r
    else:
        # Fallback (should not be reached)
        centers = np.random.rand(n, 2)
        radii = np.full(n, 0.01)
        return centers, radii, float(np.sum(radii))
