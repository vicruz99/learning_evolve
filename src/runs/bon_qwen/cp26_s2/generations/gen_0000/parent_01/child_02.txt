2.  **Simulated Annealing**: We define a cost function as the negative sum of radii. In each step:

import numpy as np
from scipy.optimize import linprog
import math
import random

def get_max_radii_and_sum(centers):
    """
    Uses Linear Programming to find the maximum possible sum of radii 
    for a given set of circle centers.
    
    Constraints:
    1. 0 <= x - r, x + r <= 1  => r <= x, r <= 1-x
    2. 0 <= y - r, y + r <= 1  => r <= y, r <= 1-y
    3. r_i + r_j <= dist(i, j) for all pairs
    4. r >= 0
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum => Minimize negative sum

    A_ub = []
    b_ub = []

    # 1. Boundary constraints (4 per circle)
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = [0.0] * n
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x
        A_ub.append(row)
        b_ub.append(1.0 - x)
        
        # r_i <= y
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        A_ub.append(row)
        b_ub.append(1.0 - y)

    # 2. Pairwise non-overlap constraints (r_i + r_j <= d_ij)
    for i in range(n):
        for j in range(i + 1, n):
            row = [0.0] * n
            row[i] = 1.0
            row[j] = 1.0
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub.append(row)
            b_ub.append(dist)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]

    # Solve LP
    # We use 'highs' method if available (default in newer scipy), else simplex
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds)

    if res.success:
        radii = res.x
        return radii, -res.fun
    else:
        # Fallback if LP fails (rare), return small radii
        radii = np.zeros(n)
        return radii, 0.0

def generate_initial_centers(n=26):
    """Generates a hexagonal packing of points in the unit square."""
    centers = []
    # Estimate radius for n=26 to set spacing
    # Approx r ~ 0.1. Spacing 2r ~ 0.2.
    r_est = 0.1
    h = r_est * math.sqrt(3)
    
    rows = 0
    while len(centers) < n:
        y = r_est + rows * h
        if y + r_est > 1.0:
            break
        
        # Determine x-offset for hexagonal shift
        if rows % 2 == 1:
            x_start = r_est + 2 * r_est # Shift by diameter to avoid wall issues? 
            # Actually standard hex shift is r (0.1), so x starts at 0.2 for odd rows if even starts at 0.1
            # Let's use shift of 2*r (0.2) for safety or just 0.1
            # If x starts at 0.1, odd row starts at 0.1 + 0.1 = 0.2
            x_start = r_est + r_est 
        else:
            x_start = r_est
            
        x = x_start
        while x + r_est <= 1.0:
            if len(centers) < n:
                centers.append([x, y])
            x += 2 * r_est
        rows += 1
        
    # If we didn't get enough (should not happen with these params), fill randomly
    while len(centers) < n:
        centers.append([random.uniform(0.01, 0.99), random.uniform(0.01, 0.99)])
        
    return np.array(centers[:n])

def run_packing():
    # Set seed for reproducibility if needed, but random is okay
    np.random.seed(42)
    random.seed(42)
    
    # 1. Initialize centers
    centers = generate_initial_centers(26)
    
    # Get initial state
    radii, current_sum = get_max_radii_and_sum(centers)
    best_sum = current_sum
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    # 2. Simulated Annealing
    n_iterations = 3000
    temperature = 0.05  # Initial temperature (radius units)
    cooling_rate = 0.995
    
    for step in range(n_iterations):
        # Pick a random circle to move
        idx = random.randint(0, 25)
        
        # Propose a move: Gaussian perturbation
        # Scale step size with temperature to allow large jumps early
        move_scale = temperature * 0.1 
        delta_x = np.random.normal(0, move_scale)
        delta_y = np.random.normal(0, move_scale)
        
        new_x = centers[idx, 0] + delta_x
        new_y = centers[idx, 1] + delta_y
        
        # Clip to [0, 1]
        new_x = np.clip(new_x, 0.0, 1.0)
        new_y = np.clip(new_y, 0.0, 1.0)
        
        # Backup and apply
        old_x, old_y = centers[idx, 0], centers[idx, 1]
        centers[idx, 0] = new_x
        centers[idx, 1] = new_y
        
        # Evaluate new state
        _, new_sum = get_max_radii_and_sum(centers)
        
        delta_sum = new_sum - current_sum
        
        # Acceptance criteria
        if delta_sum > 0:
            current_sum = new_sum
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                radii, _ = get_max_radii_and_sum(best_centers)
                best_radii = radii
        else:
            # Metropolis criterion
            if random.random() < math.exp(delta_sum / temperature):
                current_sum = new_sum
            else:
                # Revert move
                centers[idx, 0] = old_x
                centers[idx, 1] = old_y
        
        # Cool down
        temperature *= cooling_rate

    # Final polish: Run a few more iterations with very low temp or just return best
    # Return the best found configuration
    return best_centers, best_radii, best_sum
