def get_objective(v):
    """
    Objective function: Maximize sum of radii.
    v is a 1D array of shape (N*3,).
    Indices 2, 5, 8, ... correspond to radii.
    We minimize the negative sum.
    """
    r = v[2::3]
    return -np.sum(r)

def get_objective_jac(v):

    Gradient of the objective function.
