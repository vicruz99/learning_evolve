import numpy as np
from scipy.optimize import minimize

def get_obj_and_grad(x, r, n):
    """
    Computes the objective function (sum of squared overlaps) and its gradient
    for a given flattened center array x, target radius r, and number of circles n.
    """
    centers = x.reshape(n, 2)
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dist, np.inf)
    
    overlap = np.maximum(2.0 * r - dist, 0.0)
    obj = np.sum(overlap**2)
    
    grad = np.zeros_like(x)
    if np.any(overlap > 0):
        dir_vec = diff / (dist[:, :, None] + 1e-9)
        force_mag = -2.0 * overlap / (dist + 1e-9)
        grad_centers = np.sum(force_mag[:, :, None] * dir_vec, axis=1)
        grad[:] = grad_centers.ravel()
        
    return obj, grad

def obj_func(x, r, n):
    return get_obj_and_grad(x, r, n)[0]

def jac_func(x, r, n):
    return get_obj_and_grad(x, r, n)[1]

def run_packing():
    n = 26
    
    # Initialize centers in a hexagonal grid pattern
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.1
    row = 0
    while idx < n:
        x_start = 0.15 if row % 2 == 1 else 0.05
        cols = 6 if row % 2 == 0 else 5
        for c in range(cols):
            if idx >= n: break
            centers[idx] = [x_start + c * 0.14, y]
            idx += 1
        y += 0.13
        row += 1
        
    r_low, r_high = 0.05, 0.15
    best_r = r_low
    best_centers = centers.copy()
    
    # Binary search for the maximum feasible radius
    for _ in range(20):
        r_mid = (r_low + r_high) / 2
        x0 = best_centers.flatten()
        bounds = [(r_mid, 1 - r_mid)] * (2 * n)
        
        res = minimize(
            fun=obj_func,
            jac=jac_func,
            x0=x0,
            method='L-BFGS-B',
            bounds=bounds,
            args=(r_mid, n),
            options={'maxiter': 2000, 'ftol': 1e-15}
        )
        
        # If optimization successfully resolves overlaps below threshold, r is feasible
        if res.fun < 1e-6:
            best_r = r_mid
            best_centers = res.x.reshape(n, 2)
            r_low = r_mid
        else:
            r_high = r_mid
            
    # Ensure strict boundary compliance
    best_centers = np.clip(best_centers, best_r, 1 - best_r)
    radii = np.full(n, best_r)
    return best_centers, radii, np.sum(radii)
