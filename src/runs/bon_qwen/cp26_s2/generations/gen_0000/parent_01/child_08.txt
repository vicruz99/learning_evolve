import numpy as np
from scipy.optimize import minimize
from functools import partial


def objective(x_flat, n):
    """Objective: maximize sum of radii (minimize negative)"""
    return -np.sum(x_flat[2::3])


def boundary_constraint(x, i, ctype):
    """Boundary constraint for circle i"""
    idx = 3 * i
    if ctype == 0:
        return x[idx] - x[idx + 2]
    elif ctype == 1:
        return 1 - x[idx] - x[idx + 2]
    elif ctype == 2:
        return x[idx + 1] - x[idx + 2]
    else:
        return 1 - x[idx + 1] - x[idx + 2]


def nonoverlap_constraint(x, i, j):
    """Non-overlap constraint between circles i and j"""
    idx_i = 3 * i
    idx_j = 3 * j
    dx = x[idx_i] - x[idx_j]
    dy = x[idx_i + 1] - x[idx_j + 1]
    dist = np.sqrt(max(1e-20, dx * dx + dy * dy))
    r_sum = x[idx_i + 2] + x[idx_j + 2]
    return dist - r_sum


def hexagonal_initial(n, scale=1.0):
    """Generate initial hexagonal placement for n circles"""
    centers = []
    idx = 0

    # For 26 circles, use a hexagonal grid
    # Try different grid dimensions
    best_config = None
    best_coverage = 0

    for cols in range(4, 8):
        rows = int(np.ceil(n / cols))
        if cols * rows >= n:
            coverage = n / (cols * rows)
            if coverage > best_coverage:
                best_coverage = coverage
                best_config = (cols, rows)

    if best_config is None:
        best_config = (6, 5)

    cols, rows = best_config

    # Hexagonal spacing
    spacing_x = 1.0 / (cols + 1) * scale
    spacing_y = spacing_x * np.sqrt(3) / 2

    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = spacing_x * (col + 1)
            y = spacing_y * row + spacing_x * 0.5
            if row % 2 == 1:
                x += spacing_x * 0.5
            centers.append([x, y])
            idx += 1
        if idx >= n:
            break

    centers = np.array(centers[:n])

    # Normalize to fit well in unit square
    if len(centers) > 0:
        cx = centers.copy()
        x_min, x_max = cx[:, 0].min(), cx[:, 0].max()
        y_min, y_max = cx[:, 1].min(), cx[:, 1].max()

        margin = 0.03
        x_range = x_max - x_min
        y_range = y_max - y_min

        if x_range > 1e-10:
            cx[:, 0] = margin + (cx[:, 0] - x_min) / x_range * (1 - 2 * margin)
        if y_range > 1e-10:
            cx[:, 1] = margin + (cx[:, 1] - y_min) / y_range * (1 - 2 * margin)

        centers = cx

    return centers


def force_relax(centers, radii, n, max_iter=3000):
    """Force-based relaxation to expand radii and resolve overlaps"""
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(max_iter):
        # Gradually expand radii
        expansion_factor = 1.0 + 2e-5
        radii *= expansion_factor

        # Apply repulsion forces for multiple sub-iterations
        for sub_iter in range(20):
            forces = np.zeros_like(centers)

            # Repulsion between circles
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist_sq = dx * dx + dy * dy
                    dist = np.sqrt(max(1e-20, dist_sq))

                    min_dist = radii[i] + radii[j]
                    if dist < min_dist and dist > 1e-10:
                        overlap = min_dist - dist
                        force_mag = overlap * 0.3 / max(dist, 0.02)
                        fx = dx * force_mag
                        fy = dy * force_mag
                        forces[i][0] += fx
                        forces[i][1] += fy
                        forces[j][0] -= fx
                        forces[j][1] -= fy

            # Boundary repulsion
            for i in range(n):
                x, y = centers[i]
                r = radii[i]

                if x - r < 1e-10:
                    forces[i][0] += max(0, r - x) * 20
                if x + r > 1 - 1e-10:
                    forces[i][0] -= max(0, x + r - 1) * 20
                if y - r < 1e-10:
                    forces[i][1] += max(0, r - y) * 20
                if y + r > 1 - 1e-10:
                    forces[i][1] -= max(0, y + r - 1) * 20

            centers += forces

            # Project into feasible region
            for i in range(n):
                r = radii[i]
                centers[i][0] = max(r + 1e-8, min(1 - r - 1e-8, centers[i][0]))
                centers[i][1] = max(r + 1e-8, min(1 - r - 1e-8, centers[i][1]))

        # Check validity and contract if needed
        valid = True
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x < r or x > 1 - r or y < r or y > 1 - r:
                valid = False
                break

        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break

        if not valid:
            radii *= 0.98

    return centers, radii


def run_packing():
    n = 26

    best_sum = 0
    best_centers = None
    best_radii = None

    # Try multiple initial configurations
    for seed in range(8):
        np.random.seed(seed)

        # Generate initial placement with slight variations
        scale = 1.0 + (seed - 3) * 0.15
        centers = hexagonal_initial(n, scale=scale)

        # Add small random perturbation
        perturbation = np.random.randn(n, 2) * 0.02
        centers = centers + perturbation
        centers = np.clip(centers, 0.05, 0.95)

        # Initial radius
        initial_radius = 0.04 + seed * 0.005
        radii = np.full(n, initial_radius)

        # Force-based relaxation
        centers, radii = force_relax(centers, radii, n, max_iter=2000)

        # Build optimization variables
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3 * i] = centers[i][0]
            x0[3 * i + 1] = centers[i][1]
            x0[3 * i + 2] = radii[i]

        # Bounds for each variable
        bounds = []
        for i in range(n):
            bounds.extend([(1e-6, 1 - 1e-6), (1e-6, 1 - 1e-6), (1e-6, 0.3)])

        # Constraints
        constraints = []
        for i in range(n):
            for ctype in range(4):
                constraints.append({
                    'type': 'ineq',
                    'fun': partial(boundary_constraint, i=i, ctype=ctype)
                })

        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({
                    'type': 'ineq',
                    'fun': partial(nonoverlap_constraint, i=i, j=j)
                })

        # Optimize
        result = minimize(
            objective,
            x0,
            args=(n,),
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 15000, 'ftol': 1e-15, 'disp': False}
        )

        x_opt = result.x
        opt_centers = np.column_stack([x_opt[0::3], x_opt[1::3]])
        opt_radii = x_opt[2::3]
        opt_sum = np.sum(opt_radii)

        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_centers.copy()
            best_radii = opt_radii.copy()

    return best_centers, best_radii, best_sum
