import numpy as np
from scipy.optimize import minimize

N = 26

def objective_func(v):
    # Maximize sum of radii -> minimize negative sum
    # Radii are stored at indices 2, 5, 8, ...
    return -np.sum(v[2::3])

def constraint_func(v):
    cons = []
    # Boundary constraints: r <= coord <= 1-r
    for i in range(N):
        xi = v[3*i]
        yi = v[3*i+1]
        ri = v[3*i+2]
        cons.append(xi - ri)
        cons.append(1.0 - xi - ri)
        cons.append(yi - ri)
        cons.append(1.0 - yi - ri)
        
    # Circle-circle non-overlap constraints
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Compute pairwise squared distances and required squared distances
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dist_sq = dx**2 + dy**2
    req_dist_sq = dr**2
    
    # Extract upper triangle (i < j)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    cons.extend(dist_sq[mask] - req_dist_sq[mask])
    
    return np.array(cons)

def run_packing():
    # 1. Initial configuration: Hexagonal packing
    r_init = 0.08
    coords = []
    dy = r_init * np.sqrt(3)
    dx = 2 * r_init
    
    row = 0
    while len(coords) < N:
        y = r_init + row * dy
        if y + r_init > 0.95:
            row += 1
            continue

        col = 0
        while len(coords) < N:
            x_offset = r_init if is_even else r_init + dx/2
            x = x_offset + col * dx
            if x + r_init <= 0.95:
                coords.append([x, y])
                col += 1
            else:
                break
        row += 1
        if row > 15: break
        
    coords = np.array(coords[:N])

    x0 = np.zeros(3*N)

        x0[3*i+1] = coords[i, 1]
        x0[3*i+2] = r_init

    np.random.seed(42)

    constraints = {'type': 'ineq', 'fun': constraint_func}

    centers = np.column_stack((v_final[0::3], v_final[1::3]))
    radii = v_final[2::3]
    
    return centers, radii, np.sum(radii)
