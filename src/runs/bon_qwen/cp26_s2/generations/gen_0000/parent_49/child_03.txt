def eval_and_grad(params, n, penalty_weight):

    Compute objective and gradient for circle packing.
    params: [x1...xN, y1...yN, r1...rN]
    """

    radii = params[2*n:]
    
    cost = 0.0
    grad = np.zeros_like(params)
    
    # Gradient of -sum(radii)
    grad[2*n:] = -1.0
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left
        if r > x:
            err = r - x
            cost += err**2
            grad[2*n + i] += 2 * penalty_weight * err
            grad[i] -= 2 * penalty_weight * err
        # Right
        if r > 1 - x:
            err = r - (1 - x)
            cost += err**2
            grad[2*n + i] += 2 * penalty_weight * err
            grad[i] += 2 * penalty_weight * err

            err = r - y
            cost += err**2
            grad[2*n + i] += 2 * penalty_weight * err
            grad[n + i] -= 2 * penalty_weight * err
        # Top

            err = r - (1 - y)
            cost += err**2
            grad[2*n + i] += 2 * penalty_weight * err
            grad[n + i] += 2 * penalty_weight * err

    for i in range(n):
        for j in range(i + 1, n):
