import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(x):
    """
    Objective function: Minimize negative sum of radii.
    x is a 1D array of shape (3 * N_CIRCLES) containing [x1, y1, r1, x2, y2, r2, ...]
    """
    # Radii are at indices 2, 5, 8, ... (step 3 starting from 2)
    radii = x[2::3]
    return -np.sum(radii)

def constraints_func(x):
    """
    Constraint function: Returns a vector of constraint values >= 0.
    Constraints:
    1. Boundary: 0 <= x <= 1, 0 <= y <= 1, r >= 0
       Specifically: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    2. Non-overlap: distance(c_i, c_j) >= r_i + r_j
       Equivalent to: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    """
    # Reshape to (N, 3) -> [x, y, r]
    c = x.reshape(N_CIRCLES, 3)
    
    x_coords = c[:, 0]
    y_coords = c[:, 1]
    r_coords = c[:, 2]
    
    # --- Boundary Constraints ---
    # x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    # We combine these into a single array.
    
    b_cons = np.concatenate([
        x_coords - r_coords,
        1.0 - x_coords - r_coords,
        y_coords - r_coords,
        1.0 - y_coords - r_coords
    ])
    
    # --- Overlap Constraints ---
    # We need (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0 for all i < j
    
    # Compute pairwise squared distances efficiently using broadcasting
    # diff_x shape: (N, N)
    diff_x = x_coords[:, None] - x_coords[None, :]
    diff_y = y_coords[:, None] - y_coords[None, :]
    dist_sq = diff_x**2 + diff_y**2
    
    # Compute pairwise sum of radii squared
    # sum_r shape: (N, N)
    sum_r = r_coords[:, None] + r_coords[None, :]
    sum_r_sq = sum_r**2
    
    # We only need the upper triangular part (i < j) to avoid duplicates and self-checks
    # np.triu_indices returns indices for the upper triangle
    idx_i, idx_j = np.triu_indices(N_CIRCLES, k=1)
    
    overlap_cons = dist_sq[idx_i, idx_j] - sum_r_sq[idx_i, idx_j]
    
    # Concatenate all constraints
    return np.concatenate([b_cons, overlap_cons])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = N_CIRCLES

    # Create a hexagonal grid initialization.
    # This is a dense packing pattern and serves as a good starting point.
    
    initial_points = []
    r_init = 0.04  # Small initial radius to ensure points are valid
    
    y = r_init
    row = 0
    while len(initial_points) < n:
        # Hexagonal packing: rows are shifted by r_init relative to each other
        # Row 0 (even): starts at x = r_init, steps 2*r_init
        # Row 1 (odd):  starts at x = 2*r_init (shifted by r_init), steps 2*r_init
        
        if row % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
        
        x = x_start
        while x + r_init <= 1.0:
            if len(initial_points) < n:
                initial_points.append([x, y, r_init])
            else:
                break
            x += 2 * r_init
        
        # Vertical step for hexagonal packing is r * sqrt(3)
        y += r_init * np.sqrt(3)
        row += 1
    
    # Flatten to 1D array for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.array(initial_points).flatten()

    # x, y in [0, 1], r in [0, 0.5]

        ])

    result = minimize(
        objective_func,
        x0,
        method='SLSQP',
