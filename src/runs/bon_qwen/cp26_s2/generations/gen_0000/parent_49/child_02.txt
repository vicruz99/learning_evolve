import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii (maximize sum of radii)"""
    return -N_CIRCLES * vars[-1]

def constraint_boundary(vars):
    """Boundary constraints: circles must be inside [0,1]x[0,1]"""
    r = vars[-1]
    x = vars[0::2]
    y = vars[1::2]
    # Returns array: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    return np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])

def constraint_overlap(vars):
    """Non-overlap constraints: distance between centers >= 2*r"""
    r = vars[-1]
    x = vars[0::2]
    y = vars[1::2]
    cons = []
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            cons.append(dx*dx + dy*dy - 4*r*r)
    return np.array(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    
    # 1. Initial Hexagonal Lattice Configuration
    # Start with a feasible radius slightly below optimal to ensure validity
    r_init = 0.085
    centers = []
    dy = np.sqrt(3) * r_init
    
    # 5 rows of 5 circles
    for i in range(5):
        y = r_init + i * dy
        for j in range(5):
            x = r_init + j * 2 * r_init + (i % 2) * r_init
            centers.append([x, y])
            
    # 26th circle placed in the central gap
    centers.append([0.5, 0.5])
    centers = np.array(centers)
    
    # Flatten centers and append initial radius
    x0 = np.concatenate([centers.flatten(), [r_init]])
    
    # Bounds: centers in [0, 1], radius in [0.01, 0.2]
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    bounds.append((0.01, 0.2))
    
    # Define constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    # 2. Optimization
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-12}
    )
    
    # 3. Extract Results
    if res.success:
        final_centers = res.x[:-1].reshape(n, 2)
        final_radii = np.full(n, res.x[-1])
    else:
        # Fallback to initial configuration if optimization fails
        final_centers = centers
        final_radii = np.full(n, r_init)
        
    sum_r = float(np.sum(final_radii))
    return final_centers, final_radii, sum_r
