import numpy as np

def compute_forces(centers, radii, n):
    """Compute repulsive forces between circles and from boundaries."""
    forces = np.zeros((n, 2))
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_d = radii[i] + radii[j]
            
            if dist < min_d:
                overlap = min_d - dist
                # Normalize direction, avoid division by zero
                norm = overlap / (dist + 1e-9)
                fx, fy = norm * dx, norm * dy
                forces[i, 0] += fx
                forces[i, 1] += fy
                forces[j, 0] -= fx
                forces[j, 1] -= fy
                
        # Boundary repulsion
        for ax in range(2):
            if centers[i, ax] - radii[i] < 0:
                forces[i, ax] += (radii[i] - centers[i, ax]) * 10.0
            if centers[i, ax] + radii[i] > 1:
                forces[i, ax] -= (centers[i, ax] + radii[i] - 1) * 10.0
    return forces

def simulate_step(centers, radii, n, steps, dt):
    """Run force-directed simulation for a fixed number of steps."""
    c = centers.copy()
    r = radii.copy()
    for _ in range(steps):
        f = compute_forces(c, r, n)
        c += f * dt
        # Strictly enforce boundary constraints
        c[:, 0] = np.clip(c[:, 0], r, 1 - r)
        c[:, 1] = np.clip(c[:, 1], r, 1 - r)
    return c

def check_max_overlap(centers, radii, n):
    """Return the maximum overlap violation in the current configuration."""
    max_ov = 0.0
    for i in range(n):
        # Boundary check
        for ax in range(2):
            if centers[i, ax] - radii[i] < -1e-9:
                return 1.0
            if centers[i, ax] + radii[i] > 1 + 1e-9:
                return 1.0
        # Pairwise check
        for j in range(i + 1, n):
            d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                        (centers[i, 1] - centers[j, 1])**2)
            ov = radii[i] + radii[j] - d
            if ov > max_ov:
                max_ov = ov
    return max_ov
