import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the placement of 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty-based optimization approach with L-BFGS-B.
    """
    N = 26
    
    # --- 1. Initialization ---
    # Generate a grid-like initial configuration to ensure a compact start.
    # We try to place points in a dense pattern.
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    
    # Create a hexagonal-like grid distribution
    # We aim to fit roughly 5x5 circles.
    # Spacing approx 0.2.
    pts = []
    # Generate a coarse grid of potential centers
    for i in range(6):
        for j in range(6):
            x = 0.1 + i * 0.18
            y = 0.1 + j * 0.18
            pts.append([x, y])
    
    # Add some offset points for hexagonal packing feel
    for i in range(6):
        for j in range(5):
            x = 0.1 + i * 0.18 + 0.09
            y = 0.1 + j * 0.18 + 0.09
            pts.append([x, y])
            
    # Filter points that are roughly inside [0.05, 0.95] to avoid immediate wall issues
    valid_pts = []
    for p in pts:
        if 0.05 <= p[0] <= 0.95 and 0.05 <= p[1] <= 0.95:
            valid_pts.append(p)
            
    # Select first N points
    if len(valid_pts) >= N:
        centers = np.array(valid_pts[:N])
    else:
        # Fallback to random if grid is sparse (unlikely here)
        centers = np.random.rand(N, 2) * 0.8 + 0.1
        
    # Initialize radii slightly larger than 0.1 to force rearrangement
    # A square grid of 5x5 fits r=0.1 exactly. Pushing higher forces denser packing.
    radii = np.ones(N) * 0.11
    
    # Combine into a single parameter vector for the optimizer
    # Structure: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(N * 3)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    def objective(params, lambda_bound=1000, lambda_overlap=1000):
        # Unpack parameters
        # We reshape to (N, 3) -> [x, y, r]
        pts = params.reshape(N, 3)
        xs = pts[:, 0]
        ys = pts[:, 1]
        rs = pts[:, 2]
        
        # Objective: Maximize sum of radii -> Minimize negative sum
        cost = -np.sum(rs)
        
        # Penalty 1: Boundary constraints
        # Circle i must be within [0, 1]x[0, 1]
        # Violation: r > x, r > 1-x, r > y, r > 1-y
        # We use a squared penalty for smoothness
        violations_bound = np.zeros(N)
        violations_bound += np.maximum(0, rs - xs)       # Left wall
        violations_bound += np.maximum(0, rs - (1.0 - xs)) # Right wall
        violations_bound += np.maximum(0, rs - ys)       # Bottom wall
        violations_bound += np.maximum(0, rs - (1.0 - ys)) # Top wall
        
        cost += lambda_bound * np.sum(violations_bound ** 2)
        
        # Penalty 2: Overlap constraints
        # Distance between i and j must be >= r_i + r_j
        # Violation: (r_i + r_j) - dist > 0
        # Vectorized calculation for performance
        # Centers matrix shape (N, 2)
        center_mat = np.column_stack((xs, ys))
        
        # Compute pairwise distance matrix
        # Use broadcasting: (N, 1, 2) - (1, N, 2)
        diff = center_mat[:, np.newaxis, :] - center_mat[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff ** 2, axis=2))
        
        # Compute sum of radii matrix
        rad_sum = rs[:, np.newaxis] + rs[np.newaxis, :]
        
        # Overlap violation: rad_sum - dists
        # We only care about upper triangle (i < j) to avoid double counting and self
        # dists is symmetric, diagonal is 0.
        # rad_sum diagonal is 2r.
        
        violations = rad_sum - dists
        np.fill_diagonal(violations, 0) # Self distance is 0, self overlap not checked
        
        # Take upper triangle
        triu_indices = np.triu_indices(N, k=1)
        overlaps = violations[triu_indices]
        
        # Penalize positive overlaps (where dist < r1 + r2)
        overlap_violations = np.maximum(0, overlaps)
        cost += lambda_overlap * np.sum(overlap_violations ** 2)
        
        return cost

    # --- 3. Optimization Loop ---

    current_params = x0
    bounds = [(0, 1) for _ in range(N)] + [(0, 1) for _ in range(N)] + [(0, 0.5) for _ in range(N)]

    stages = [
        {'lb': 100, 'lo': 100, 'iters': 500},
        {'lb': 500, 'lo': 500, 'iters': 1000},
        {'lb': 2000, 'lo': 2000, 'iters': 1500},
        {'lb': 10000, 'lo': 10000, 'iters': 2000},
        {'lb': 50000, 'lo': 50000, 'iters': 3000},
    ]

    for stage in stages:
        res = minimize(
            objective,
            current_params,
            args=(stage['lb'], stage['lo']),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': stage['iters'], 'ftol': 1e-9, 'gtol': 1e-6}
        )
        current_params = res.x
        
        # Check if we are making progress or stuck
        # If cost is dominated by penalties, we might need to reduce radii
        # But the optimizer handles this via the objective trade-off.

    # Unpack final parameters
    final_pts = current_params.reshape(N, 3)
    final_centers = final_pts[:, :2]
    final_radii = final_pts[:, 2]

    # Ensure radii are non-negative (bounds handle this, but just in case)
    final_radii = np.maximum(0, final_radii)

    # Final Safety Check & Correction
    # If there are tiny violations due to numerical precision, shrink radii slightly
    # This ensures the validation function passes.
    
    # Check boundaries
    for i in range(N):
        x, y = final_centers[i]
        r = final_radii[i]
        # Max allowed radius by walls
        max_r = min(x, 1-x, y, 1-y)

                    # Distribute reduction
                    final_radii[i] -= reduction * 0.5

        if not overlap_found:
            break
            
    # Ensure radii didn't go negative
    final_radii = np.maximum(0, final_radii)
    
    # Re-check boundaries after overlap reduction (radii might be smaller, so safe)

        r = final_radii[i]
