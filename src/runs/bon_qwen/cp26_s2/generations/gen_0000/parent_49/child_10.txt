import numpy as np
from scipy.optimize import minimize
import itertools

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def penalty_function(params, n):
    """
    Calculate the penalty for constraint violations.
    params contains [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = params[0:2*n].reshape(n, 2)
    radii = params[2*n:3*n]
    
    penalty = 0.0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += (x - r) ** 2
        if x + r > 1:
            penalty += (x + r - 1) ** 2
        if y - r < 0:
            penalty += (y - r) ** 2
        if y + r > 1:
            penalty += (y + r - 1) ** 2
            
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j]) ** 2)
            min_dist = radii[i] + radii[j]
            if dist_sq < min_dist ** 2:
                overlap = min_dist - np.sqrt(dist_sq)
                penalty += overlap ** 2
                
    # Penalty for small radii (encourage maximization)
    penalty -= np.sum(radii) * 0.01 
    
    return penalty

def get_bounds(n):
    """
    Return bounds for optimization variables.
    """
    bounds = []
    for _ in range(n):
        # x in [0, 1]
        bounds.append((0, 1))
        # y in [0, 1]
        bounds.append((0, 1))
        # r in [0, 0.5] (max possible radius)
        bounds.append((0, 0.5))
    return bounds

def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Strategy: Try multiple initial configurations
    num_trials = 50
    
    for trial in range(num_trials):

            grid_x = np.linspace(0.1, 0.9, 5)
            grid_y = np.linspace(0.1, 0.9, 5)
            cx = []
            cy = []
            for gx in grid_x:
                for gy in grid_y:
                    cx.append(gx + np.random.uniform(-0.01, 0.01))
                    cy.append(gy + np.random.uniform(-0.01, 0.01))
            # Add one more circle
            cx.append(np.random.uniform(0.1, 0.9))
            cy.append(np.random.uniform(0.1, 0.9))
            cx = np.array(cx)
            cy = np.array(cy)
            r = np.full(n, 0.08) # Start small
        else:
            # Random initialization
            cx = np.random.uniform(0.1, 0.9, n)
            cy = np.random.uniform(0.1, 0.9, n)
            r = np.full(n, 0.05)
            
        centers = np.column_stack((cx, cy))
        
        # Prepare parameters vector: [x1, y1, r1, x2, y2, r2, ...]

        params[0:2*n] = centers.flatten()
        params[2*n:3*n] = r
        
        bounds = get_bounds(n)
        
        # Optimize
        res = minimize(penalty_function, params, args=(n,), method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000})

            opt_centers = res.x[0:2*n].reshape(n, 2)
            opt_radii = res.x[2*n:3*n]

            if current_sum > best_sum:

                if validate_packing(opt_centers, opt_radii):

            best_centers = final_centers

            best_sum = np.sum(best_radii)
