import numpy as np
from scipy.optimize import minimize

N = 26


def objective(x):
    return -np.sum(x[:N])


def constraint_x_lower(x, i):
    return x[N + i] - x[i]


def constraint_x_upper(x, i):
    return 1.0 - x[N + i] - x[i]


def constraint_y_lower(x, i):
    return x[2 * N + i] - x[i]


def constraint_y_upper(x, i):
    return 1.0 - x[2 * N + i] - x[i]


def constraint_non_overlap(x, pair):
    i, j = pair
    dx = x[N + i] - x[N + j]
    dy = x[2 * N + i] - x[2 * N + j]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[i] - x[j]


def constraint_non_negative(x, i):
    return x[i]


def build_constraints():
    constraints = []
    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': constraint_x_lower, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_x_upper, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_lower, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_upper, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_non_negative, 'args': (i,)})
    
    pairs = [(i, j) for i in range(N) for j in range(i + 1, N)]
    for pair in pairs:
        constraints.append({'type': 'ineq', 'fun': constraint_non_overlap, 'args': (pair,)})
    
    return constraints


def initialize_hexagonal(offset_x=0.08, offset_y=0.08):
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.05)
    
    counts = [6, 5, 6, 5, 4]
    idx = 0
    
    for row, count in enumerate(counts):
        if idx >= N:
            break
        actual_count = min(count, N - idx)
        
        if actual_count > 1:
            spacing_x = (1.0 - 2.0 * offset_x) / (actual_count - 1)
        else:
            spacing_x = 0.0
        
        spacing_y = (1.0 - 2.0 * offset_y) / 4.0
        
        for col in range(actual_count):
            if row % 2 == 1:
                x = offset_x + (col + 0.5) * spacing_x
            else:
                x = offset_x + col * spacing_x
            y = offset_y + row * spacing_y
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def initialize_grid_variant(offset=0.12):
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.04)
    
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [offset + i * 0.2, offset + j * 0.2]
            idx += 1
    
    centers[25] = [0.5, 0.5]
    
    return centers, radii


def optimize_with_init(centers_init, radii_init, constraints):
    x0 = np.concatenate([radii_init, centers_init.flatten()])
    
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        constraints=constraints,
        options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
    )
    
    return result


def run_packing():
    constraints = build_constraints()
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    configs = [
        (0.08, 0.08),
        (0.10, 0.10),
        (0.06, 0.06),
        (0.12, 0.12),
        (0.09, 0.11),
        (0.11, 0.09),
    ]
    
    for ox, oy in configs:
        centers_init, radii_init = initialize_hexagonal(offset_x=ox, offset_y=oy)
        result = optimize_with_init(centers_init, radii_init, constraints)
        
        r_opt = result.x[:N]
        cx_opt = result.x[N:2*N]
        cy_opt = result.x[2*N:3*N]
        
        current_sum = np.sum(r_opt)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = np.column_stack([cx_opt, cy_opt])
            best_radii = r_opt
    
    centers_grid, radii_grid = initialize_grid_variant()
    result_grid = optimize_with_init(centers_grid, radii_grid, constraints)
    
    r_grid = result_grid.x[:N]
    cx_grid = result_grid.x[N:2*N]
    cy_grid = result_grid.x[2*N:3*N]
    grid_sum = np.sum(r_grid)
    
    if grid_sum > best_sum:
        best_sum = grid_sum
        best_centers = np.column_stack([cx_grid, cy_grid])
        best_radii = r_grid
    
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, final_sum
