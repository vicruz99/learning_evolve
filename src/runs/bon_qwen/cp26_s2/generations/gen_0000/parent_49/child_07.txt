import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    for attempt in range(40):
        centers, radii = build_init(n, attempt)
        params = np.concatenate([centers.ravel(), radii])

        for pw in [5, 25, 100, 500, 2500, 12000, 50000]:
            res = minimize(pen_obj, params, args=(n, pw),
                           method='L-BFGS-B',
                           options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-10})
            params = res.x

        c = params[:2 * n].reshape(-1, 2)
        r = np.maximum(params[2 * n:], 1e-10)

        c, r = fix_constraints(c, r, n)
        s = r.sum()

        if is_valid(c, r, n) and s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()

    if best_centers is None:
        best_centers = c
        best_radii = r
        best_sum = r.sum()

    return best_centers, best_radii, best_sum


def build_init(n, seed):
    np.random.seed(seed * 13 + 7)
    c = np.zeros((n, 2))
    r = np.ones(n) * 0.045

    sp = 0.155
    idx = 0
    for row in range(7):
        for col in range(5):
            if idx >= n:
                break
            x = 0.11 + col * sp + (sp / 2 if row % 2 == 1 else 0)
            y = 0.11 + row * sp * np.sqrt(3) / 2
            c[idx] = [x, y]
            idx += 1

    c += np.random.uniform(-0.003, 0.003, (n, 2))
    c[:, 0] = np.clip(c[:, 0], 0.12, 0.88)
    c[:, 1] = np.clip(c[:, 1], 0.12, 0.88)
    return c, r


def pen_obj(p, n, pw):
    c = p[:2 * n].reshape(-1, 2)
    r = np.maximum(p[2 * n:], 1e-10)

    val = -r.sum()

    for i in range(n):
        x, y = c[i]
        ri = r[i]
        val += pw * max(0, ri - x) ** 2
        val += pw * max(0, ri - (1 - x)) ** 2
        val += pw * max(0, ri - y) ** 2
        val += pw * max(0, ri - (1 - y)) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            ov = r[i] + r[j] - d
            if ov > 0:
                val += pw * ov ** 2

    return val


def fix_constraints(centers, radii, n):
    for _ in range(200):
        valid = True
        for i in range(n):
            x, y = centers[i]
            ri = radii[i]

            if x - ri < -1e-11:
                centers[i, 0] = ri - 1e-12
                valid = False
            if x + ri > 1 + 1e-11:
                centers[i, 0] = 1 - ri + 1e-12
                valid = False
            if y - ri < -1e-11:
                centers[i, 1] = ri - 1e-12
                valid = False
            if y + ri > 1 + 1e-11:
                centers[i, 1] = 1 - ri + 1e-12
                valid = False

            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-11:
                    if dist > 1e-14:
                        overlap = (radii[i] + radii[j] - dist + 2e-12) / 2
                        nx = dx / dist
                        ny = dy / dist
                        centers[i, 0] += overlap * nx
                        centers[i, 1] += overlap * ny
                        centers[j, 0] -= overlap * nx
                        centers[j, 1] -= overlap * ny
                    else:
                        radii[i] *= 0.95
                        radii[j] *= 0.95
                    valid = False
        if valid:
            break

    centers[:, 0] = np.clip(centers[:, 0], 1e-10, 1 - 1e-10)
    centers[:, 1] = np.clip(centers[:, 1], 1e-10, 1 - 1e-10)
    return centers, radii


def is_valid(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        ri = radii[i]
        if x - ri < -1e-9 or x + ri > 1 + 1e-9 or y - ri < -1e-9 or y + ri > 1 + 1e-9:
            return False
        for j in range(i + 1, n):
            d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            if d < radii[i] + radii[j] - 1e-9:
                return False
    return True
